namespace AspenTech.HtmlHelp.Logging
{
	public class CHtmlHelpLogging
	{
		private string licKeyString = "SLM_TR_ALL_ProdHelp";

		private string licKeyVer = string.Empty;

		private int licCookie;

		private string licKeyLgString = "SLM_TR_ALL_TRANSLATE";

		private string licKeyLgVer = string.Empty;

		private int licLgCookie;

		private SecServer licSecSrv;

		public void PauseLogging()
		{
			if (licSecSrv != null)
			{
				if (licLgCookie != 0)
				{
					licSecSrv.ReleaseLicense(licLgCookie);
				}
				licLgCookie = 0;
				if (licCookie != 0)
				{
					licSecSrv.ReleaseLicense(licCookie);
				}
				licCookie = 0;
			}
		}

		public string GetLCIDForLog(string strLanguage)
		{
			string empty = string.Empty;
			switch (strLanguage)
			{
			case "fr":
				return "fr-FR";
			case "it":
				return "it-IT";
			case "de":
				return "de-DE";
			case "es":
				return "es-ES";
			case "ja":
				return "ja-JP";
			case "zh-CHS":
			case "zh-CHT":
				return "zh-HANS";
			case "ko":
				return "ko-KR";
			case "ru":
				return "ru-RU";
			case "pt":
				return "pt-BR";
			default:
				return "en-US";
			}
		}

		public void HelpDisplayLangHandle(string strHelpDisplayLang)
		{
			try
			{
				if (licSecSrv is ISLMUsageLog2 iSLMUsageLog)
				{
					string lCIDForLog = GetLCIDForLog(strHelpDisplayLang);
					iSLMUsageLog.SetLanguage(lCIDForLog);
					licSecSrv.CheckoutLicenseDeferred(licKeyLgString, licKeyLgVer, out licLgCookie, null, LicenseImportance_enum.liNotCritical);
				}
			}
			catch (Exception)
			{
			}
		}

		public void StopLogging()
		{
			licSecSrv = null;
		}

		public bool InitLogging()
		{
			if (licSecSrv == null)
			{
				licSecSrv = (SecServer)Activator.CreateInstance(Marshal.GetTypeFromCLSID(new Guid("1460FDB4-752B-11D4-B067-0090271CF383")));
			}
			if (licSecSrv == null)
			{
				return false;
			}
			licSecSrv.CheckoutLicenseDeferred(licKeyString, licKeyVer, out licCookie, null, LicenseImportance_enum.liNotCritical);
			return true;
		}
	}
}
