	public class MolecularEditorPresenter : IMolecularEditorPresenter, IMoleculeDrawingProxy
	{
		private IMoleculeRenderEngine _moleculeRenderEngine;

		private IMoleculeDrawingEngine _moleculeDrawingEngine;

		private IMoleculeRenderEngine MoleculeRenderEngine => _moleculeRenderEngine ?? (_moleculeRenderEngine = MolecularEditorNetEngine.CreateMoleculeRenderEngine());

		private IMoleculeDrawingEngine MoleculeDrawingEngine => _moleculeDrawingEngine ?? (_moleculeDrawingEngine = MolecularEditorNetEngine.CreateMoleculeDrawingEngine());

		public void SimplyOpenMolecularEditor()
		{
			MoleculeDrawingEngine.SimplyOpenMolecularEditor();
		}

		public string[] OpenMolecularEditor(string[] molefileContentAsLines, bool forTDE, ref string fragmentName, out int fragmentConsistency, out bool changed)
		{
			return MoleculeDrawingEngine.OpenMolecularEditor(molefileContentAsLines, forTDE, ref fragmentName, ref fragmentConsistency, ref changed);
		}

		public string[] CheckMolecularEditor(string[] molefileContentAsLines, out int fragmentConsistency)
		{
			return MoleculeDrawingEngine.CheckMolecularEditor(molefileContentAsLines, ref fragmentConsistency);
		}

		public object CreateBitmapForMolefile(string[] molefileContentAsLines)
		{
			return MoleculeDrawingEngine.CreateBitmapForMolefile(molefileContentAsLines);
		}

		public object CreateBitmapForMolefileInSpecialSize(string[] molefileContentAsLines, double width, double height)
		{
			return MoleculeDrawingEngine.CreateBitmapForMolefileInSpecialSize(molefileContentAsLines, width, height);
		}

		public void GenerateConnectionTable(string[] molefileContentAsLines, out int[] startAtomIndexes, out int[] endAtomIndexes, out string[] startAtomSymbols, out string[] endAtomSymbols, out int[] bondTypes, out string[] formulaAtomSymbols, out int[] formularAtomCounts)
		{
			MoleculeDrawingEngine.GenerateConnectionTable(molefileContentAsLines, ref startAtomIndexes, ref endAtomIndexes, ref startAtomSymbols, ref endAtomSymbols, ref bondTypes, ref formulaAtomSymbols, ref formularAtomCounts);
		}

		string IMoleculeDrawingProxy.CompressText(string text)
		{
			return MoleStructCompressor.CompressText(text);
		}

		string IMoleculeDrawingProxy.DecompressText(string text)
		{
			return MoleStructCompressor.DecompressText(text);
		}

		string[] IMoleculeDrawingProxy.OpenMolecularEditor(string[] molefileContentAsLines, bool forTDE, ref string fragmentName, out int fragmentConsistency, out bool changed)
		{
			return OpenMolecularEditor(molefileContentAsLines, forTDE, ref fragmentName, out fragmentConsistency, out changed);
		}

		string[] IMoleculeDrawingProxy.CheckMolecularEditor(string[] molefileContentAsLines, out int fragmentConsistency)
		{
			return CheckMolecularEditor(molefileContentAsLines, out fragmentConsistency);
		}

		ImageSource IMoleculeDrawingProxy.CreateBitmapForMolefileInSpecialSize(string[] molefileContentAsLines, double width, double height)
		{
			IMoleculeImage moleculeImage = MoleculeRenderEngine.GetMoleculeImage(molefileContentAsLines);
			return ImageHelper.GetRenderTargetBitmapLessThanTargetSize(moleculeImage.VectorImage, moleculeImage.Size, new Size(width, height));
		}

		void IMoleculeDrawingProxy.GenerateConnectionTable(string[] molefileContentAsLines, out int[] startAtomIndexes, out int[] endAtomIndexes, out string[] startAtomSymbols, out string[] endAtomSymbols, out int[] bondTypes, out string[] formulaAtomSymbols, out int[] formularAtomCounts)
		{
			GenerateConnectionTable(molefileContentAsLines, out startAtomIndexes, out endAtomIndexes, out startAtomSymbols, out endAtomSymbols, out bondTypes, out formulaAtomSymbols, out formularAtomCounts);
		}

		public string[] GenerateMolStringsFromSMILES(string smiles)
		{
			return MoleculeDrawingEngine.GenerateMolStringsFromSMILES(smiles);
		}
	}
