namespace AspenHelper;

public class AspenHelper : IDisposable
{
	private IHapp App;

	public AspenHelper(string strFilePath)
	{
		ref IHapp app = ref App;
		object obj = Interaction.CreateObject("Apwn.Document." + GetCurrentVersion());
		app = (IHapp)((obj is IHapp) ? obj : null);
		App.InitFromFile(ref strFilePath, (object)true);
		App.Visible = true;
	}

	private static string GetCurrentVersion()
	{
		RegistryKey registryKey = Registry.LocalMachine.OpenSubKey("Software\\Aspentech\\Aspen Plus\\CurVer");
		if (registryKey != null)
		{
			return registryKey.GetValue(string.Empty).ToString();
		}
		return string.Empty;
	}

	public void Dispose()
	{
		if (App != null)
		{
			App.Close((object)0);
			Marshal.FinalReleaseComObject(App);
			App = null;
			Thread.Sleep(2000);
		}
	}

	~AspenHelper()
	{
	}

	public void Save(string strFileName)
	{
		if (App != null)
		{
			App.SaveAs(ref strFileName, (object)true);
		}
	}

	public static string GetConvertFilePath()
	{
		return Registry.LocalMachine.OpenSubKey("Software\\Aspentech\\Aspen Plus\\" + GetCurrentVersion() + "\\aplus").GetValue("asptop").ToString() + "\\Xeq\\atp3.exe";
	}

	public static string CheckStreamName(string strFilePath)
	{
		FileStream fileStream = new FileStream(strFilePath, FileMode.Open, FileAccess.Read);
		StreamReader streamReader = new StreamReader(fileStream);
		string str = streamReader.ReadToEnd();
		streamReader.Close();
		fileStream.Close();
		int num = str.IndexOf("UNIT OPERATIONS");
		if (num > 0)
		{
			int nIndex = 0;
			while (true)
			{
				string streamName = GetStreamName(ref str, ref nIndex);
				if (string.IsNullOrEmpty(streamName) || nIndex > num)
				{
					break;
				}
				if (streamName.IndexOf('.') > 0)
				{
					return streamName;
				}
			}
		}
		return "";
	}

	private static string GetStreamName(ref string str, ref int nIndex)
	{
		if (nIndex == 0)
		{
			nIndex = str.IndexOf("STREAM DATA");
			if (nIndex < 0)
			{
				return "";
			}
		}
		int num = 0;
		while (num >= 0)
		{
			num = str.IndexOf("STREAM", nIndex);
			if (num <= 0)
			{
				continue;
			}
			if (str[num - 1] == 'F')
			{
				num += 6;
				continue;
			}
			num = str.IndexOf('=', num + 6);
			if (num > 0)
			{
				int num2 = str.IndexOf(',', num + 1);
				if (num2 < 0)
				{
					num2 = str.IndexOf('\r', num + 1);
				}
				if (num2 > 0)
				{
					string text = str.Substring(num + 1, num2 - num - 1);
					nIndex = num2;
					return text.Trim();
				}
			}
		}
		return "";
	}

	public static bool CutBlockName(string strFilePath, string strNewFilePath)
	{
		FileStream fileStream = new FileStream(strFilePath, FileMode.Open, FileAccess.Read);
		StreamReader streamReader = new StreamReader(fileStream, Encoding.Default);
		string str = streamReader.ReadToEnd();
		streamReader.Close();
		fileStream.Close();
		int num = 0;
		List<string> blocks = new List<string>();
		do
		{
			num = FindBlockName(ref str, num, ref blocks);
		}
		while (num != -1);
		int nIndex = 0;
		List<string> list = new List<string>();
		foreach (string item in blocks)
		{
			list.Add(ConvertBlockName(item, ref nIndex));
		}
		for (int i = 0; i < blocks.Count; i++)
		{
			if (blocks[i] != list[i])
			{
				str = str.Replace(blocks[i], list[i]);
			}
		}
		FileStream fileStream2 = new FileStream(strNewFilePath, FileMode.Create);
		StreamWriter streamWriter = new StreamWriter(fileStream2);
		streamWriter.Write(str);
		streamWriter.Close();
		fileStream2.Close();
		return true;
	}

	private static string ConvertBlockName(string strName, ref int nIndex)
	{
		int num = nIndex.ToString().Length + 1;
		if (num >= 3)
		{
			return nIndex.ToString();
		}
		string result = $"{strName.Substring(0, 4 - num)}_{nIndex.ToString()}";
		nIndex++;
		return result;
	}

	private static int FindBlockName(ref string str, int nIndex, ref List<string> blocks)
	{
		int num = str.IndexOf("LNGHX", nIndex);
		if (num > 0)
		{
			num = str.IndexOf("UID", num + 5);
			if (num > 0)
			{
				num = str.IndexOf('=', num + 3);
				if (num > 0)
				{
					int num2 = str.IndexOf(',', num + 1);
					if (num2 == -1)
					{
						num2 = str.IndexOf('\r', num + 1);
					}
					if (num2 != -1)
					{
						string text = str.Substring(num + 1, num2 - num - 1);
						blocks.Add(text.Trim());
						return num;
					}
				}
			}
		}
		return -1;
	}
}
