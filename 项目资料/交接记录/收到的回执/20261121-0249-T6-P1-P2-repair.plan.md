3. **固定摘要输入。** 以本件最终存储的报告和正文组成明确清单，逐文件取原样字节 SHA-256。按 POSIX 文件 basename 升序拼接 UTF-8 的 `basename=小写64位sha256;`，再计算 SHA-256。basename 从规范的库内相对路径提取；Windows 本地路径使用 `Path.name`。现有 `product_sha` 用 `rsplit("/",1)` 拆本地路径，反斜杠可能被算进文件名；须用 Windows 与 POSIX 路径的同字节夹具验证。单件有重复 basename 时显式报出。复用已发布 blob 时保持字节原样，避免解码再编码或换行、BOM、尾部换行变化。`.gz` 按实际存储字节算指纹，解压内容另行验收。

## 4. 可执行的独立读回

下面的脚本放在本地检查目录，参数为固定提交的库根内容快照。用认证连接器将两个 CSV 和本件清单所需的已发布文件按相对路径、原字节写到新的普通目录；可额外用树 API 核对 blob ID。脚本不会运行快照中的程序。执行前后分别调用 `python audit_p1p2.py <H0快照>` 和 `python audit_p1p2.py <H1快照>`。基线预期 P1=38，P2 以实际输出为准；H1 唯一通过行是 `rows=58 files=158 P1=0 P2=0 INDEX=0`，退出码 0。

```python
import csv
import hashlib
import sys
from pathlib import Path, PurePosixPath

root = Path(sys.argv[1]).resolve()
base = PurePosixPath("analysis/20261120-batch-driver")
roles = {"CLR": (("il", ".il.txt"), ("decompiled", ".cs.txt")),
         "native": (("disasm", ".asm.txt"), ("decompiled", ".c.txt")), "nonPE": ()}

def read_csv(name):
    with (root / base / name).open(encoding="utf-8-sig", newline="") as f:
        return list(csv.DictReader(f))

def digest(data):
    return hashlib.sha256(data).hexdigest()

def require(value, message):
    if not value:
        raise ValueError(message)

ledger = read_csv("archive-ledger.csv")
index_rows = read_csv("index.csv")
index = {row["sha256"]: row for row in index_rows}
keys = [row["sha256"] for row in ledger]
require(len(set(keys)) == len(keys) == len(index_rows) == len(index) == 58, "key_count")
require(set(keys) == set(index), "index_keys")
seen = set()
p1 = p2 = index_errors = 0
for row in ledger:
    report = PurePosixPath(row["report_md"])
    rel = report.relative_to(base)
    require(".." not in rel.parts and len(rel.parts) >= 2, "report_path")
    p1 += (rel.parts[0] != row["product"])
    name = row["source_name"]
    require(PurePosixPath(name).name == name and "\\" not in name, "source_name")
    paths = [report]
    for folder, ext in roles[row["kind"]]:
        plain = report.parent / folder / (name + ext)
        choices = [p for p in (plain, PurePosixPath(str(plain) + ".gz")) if (root / p).is_file()]
        require(len(choices) == 1, "missing_or_duplicate_companion")
        paths.append(choices[0])
    require(len(paths) == int(row["product_files"]), "member_count")
    require(len({p.name for p in paths}) == len(paths), "duplicate_basename")
    total = 0
    entries = []
    for path in sorted(paths, key=lambda p: p.name):
        require(path not in seen, "shared_member")
        seen.add(path)
        target = (root / path).resolve()
        require(root in target.parents, "outside_snapshot")
        data = target.read_bytes()
        entries.append(path.name + "=" + digest(data) + ";")
        total += len(data)
    actual = digest("".join(entries).encode("utf-8"))
    p2 += (actual != row["product_sha256"])
    idx = index[row["sha256"]]
    index_errors += (idx["product_sha256"] != actual or int(idx["bytes"]) != total or
                     any(idx[k] != row[k] for k in ("report_md", "product", "kind", "status")))
print(f"rows={len(ledger)} files={len(seen)} P1={p1} P2={p2} INDEX={index_errors}")
sys.exit(int(len(seen) != 158 or bool(p1 or p2 or index_errors)))
```
