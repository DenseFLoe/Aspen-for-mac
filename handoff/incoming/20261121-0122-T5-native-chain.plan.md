### `tool/ghidra_scripts/ExportAll.java`

```java
// Export the functions Ghidra identified in one imported program.
//@category Aspen
import ghidra.app.decompiler.DecompInterface;
import ghidra.app.decompiler.DecompileResults;
import ghidra.app.util.headless.HeadlessScript;
import ghidra.program.model.listing.*;
import ghidra.program.model.mem.MemoryBlock;
import java.io.BufferedWriter;
import java.nio.file.*;

public class ExportAll extends HeadlessScript {
    private String oneLine(Object value) {
        return String.valueOf(value).replace('\r', ' ').replace('\n', ' ').replace('\t', ' ');
    }

    private void line(BufferedWriter out, String value) throws Exception {
        out.write(value);
        out.write('\n');
    }

    @Override
    public void run() throws Exception {
        Path dir = Paths.get(getScriptArgs()[0]);
        String stem = currentProgram.getName();
        Files.createDirectories(dir);
        if (analysisTimeoutOccurred()) {
            Files.writeString(dir.resolve(stem + ".timeout"), "timeout\n");
            return;
        }
        FunctionManager fm = currentProgram.getFunctionManager();
        Listing listing = currentProgram.getListing();
        int count = 0;
        // Take an address inventory independently of the export loop.
        try (BufferedWriter addresses = Files.newBufferedWriter(dir.resolve(stem + ".addresses.txt"))) {
            for (Function f : fm.getFunctions(true)) {
                line(addresses, f.getEntryPoint().toString(true));
                count++;
            }
        }
        DecompInterface decompiler = new DecompInterface();
        try (BufferedWriter asm = Files.newBufferedWriter(dir.resolve(stem + ".asm.txt"));
             BufferedWriter pseudo = Files.newBufferedWriter(dir.resolve(stem + ".c.txt"))) {
            if (!decompiler.openProgram(currentProgram)) {
                throw new IllegalStateException("decompiler_open_failed");
            }
            line(asm, "# LANG " + currentProgram.getLanguageID());
            line(asm, "# BASE " + currentProgram.getImageBase());
            line(asm, "# BLOCKS " + currentProgram.getMemory().getBlocks().length);
            line(asm, "# FORMAT " + oneLine(currentProgram.getExecutableFormat()));
            line(asm, "# COMPILER " + currentProgram.getCompilerSpec().getCompilerSpecID());
            for (MemoryBlock b : currentProgram.getMemory().getBlocks()) {
                line(asm, "# BLOCK " + oneLine(b.getName()) + " " + b.getStart() + " " +
                    b.getEnd() + " " + (b.isRead() ? "r" : "-") +
                    (b.isWrite() ? "w" : "-") + (b.isExecute() ? "x" : "-"));
            }
            for (Function f : fm.getFunctions(true)) {
                monitor.checkCancelled();
                String addr = f.getEntryPoint().toString(true);
                String label = addr + " " + oneLine(f.getName(true));
                line(asm, "FUNC " + label);
                InstructionIterator instructions = listing.getInstructions(f.getBody(), true);
                int n = 0;
                while (instructions.hasNext()) {
                    Instruction ins = instructions.next();
                    line(asm, "  " + ins.getAddress().toString(true) + ": " + oneLine(ins));
                    n++;
                }
                line(asm, "# INSNS " + n);
                line(asm, "ENDFUNC");
                String body = null;
                String reason = "no_c_text";
                try {
                    DecompileResults r = decompiler.decompileFunction(f, 30, monitor);
                    if (r != null && r.decompileCompleted() && r.getDecompiledFunction() != null) {
                        body = r.getDecompiledFunction().getC();
                    }
                    else if (r != null) {
                        reason = r.isTimedOut() ? "function_timeout" :
                            (r.isCancelled() ? "cancelled" : "decompiler_failed");
                    }
                }
                catch (RuntimeException e) {
                    reason = e.getClass().getSimpleName();
                }
                monitor.checkCancelled();
                if (body == null || body.isBlank()) {
                    line(pseudo, "// DECOMP-FAIL " + label + " " + reason);
                }
                else {
                    line(pseudo, "// FUNC " + label);
                    for (String s : body.lines().toList()) {
                        line(pseudo, "    " + s);
                    }
                    line(pseudo, "// ENDFUNC");
                }
            }
        }
        finally {
            decompiler.dispose();
        }
        Files.writeString(dir.resolve(stem + ".finished.txt"), count + "\n", StandardOpenOption.CREATE_NEW);
    }
}
```

### `out/R1N/_tmp/audit_native.py`

```python
import gzip
import sys
from pathlib import Path


def require(test, reason):
    if not test:
        raise ValueError(reason)


def rows(path):
    opener = gzip.open if path.suffix == ".gz" else open
    with opener(path, "rt", encoding="utf-8") as data:
        for row in data:
            yield row.rstrip("\r\n")


def verify(catalog, finished, asm, pseudo):
    addresses = list(rows(catalog))
    n = int(finished.read_text(encoding="ascii").strip())
    require(n > 0 and len(addresses) == len(set(addresses)) == n, "inventory")
    seq = iter(rows(asm))
    require(next(seq).startswith("# LANG ") and next(seq).startswith("# BASE "), "language_base")
    blocks = next(seq)
    require(blocks.startswith("# BLOCKS "), "block_header")
    expected_blocks = int(blocks.split()[-1])
    names = {}
    current = None
    count = empty = found_blocks = 0
    declared = None
    for line in seq:
        if line.startswith("# BLOCK "):
            found_blocks += 1
        elif line.startswith("FUNC "):
            require(current is None, "nested_asm")
            _, current, name = line.split(" ", 2)
            require(current not in names and name.strip(), "asm_name")
            names[current] = name
            count, declared = 0, None
        elif line.startswith("  "):
            require(current is not None and ": " in line, "instruction")
            count += 1
        elif line.startswith("# INSNS "):
            declared = int(line.split()[-1])
        elif line == "ENDFUNC":
            require(current is not None and declared == count, "instruction_count")
            empty += count == 0
            current = None
    require(current is None and expected_blocks == found_blocks > 0, "asm_end_or_blocks")
    require(list(names) == addresses, "asm_addresses")
    outcomes = []
    failed = 0
    body = False
    for line in rows(pseudo):
        if line.startswith("// FUNC "):
            require(current is None, "nested_c")
            _, _, current, name = line.split(" ", 3)
            require(names.get(current) == name, "c_name")
            outcomes.append(current)
            body = False
        elif line == "// ENDFUNC":
            require(current is not None and body, "empty_c")
            current = None
        elif line.startswith("// DECOMP-FAIL "):
            _, _, addr, detail = line.split(" ", 3)
            require(current is None and addr in names and
                    detail.startswith(names[addr] + " ") and
                    detail[len(names[addr]):].strip(), "failure_detail")
            outcomes.append(addr)
            failed += 1
        else:
            require(current is not None and line.startswith("    "), "unexpected_c_line")
            body = body or bool(line.strip())
    require(current is None and outcomes == addresses, "c_addresses")
    return n, n - failed, failed, empty


if __name__ == "__main__":
    if len(sys.argv) != 5:
        raise SystemExit("usage: audit_native.py addresses.txt finished.txt asm.txt[.gz] c.txt[.gz]")
    print("D2 functions=%d ok=%d decomp_fail=%d no_instructions=%d" %
          verify(*(Path(s) for s in sys.argv[1:])))
```
