	internal sealed class ExeCOMServer
	{
		private static ExeCOMServer _instance = new ExeCOMServer();

		private object syncRoot;

		private bool _bRunning;

		private uint _nMainThreadID;

		private int _nLockCnt;

		private System.Threading.Timer _gcTimer;

		private uint _cookieSimpleObj;

		public static ExeCOMServer Instance => _instance;

		private ExeCOMServer()
		{
			syncRoot = RuntimeHelpers.GetObjectValue(new object());
			_bRunning = false;
			_nMainThreadID = 0u;
			_nLockCnt = 0;
		}

		private static void GarbageCollect(object stateInfo)
		{
			GC.Collect();
		}

		private void PreMessageLoop()
		{
			Guid rclsid = new Guid("B9927A7C-CD92-47E5-BCDE-96C909D3744B");
			int num = COMNative.CoRegisterClassObject(ref rclsid, new ApplicationClassFactory(), CLSCTX.LOCAL_SERVER, REGCLS.MULTIPLEUSE | REGCLS.SUSPENDED, out _cookieSimpleObj);
			if (num != 0)
			{
				throw new ApplicationException("CoRegisterClassObject failed w/err 0x" + num.ToString("X"));
			}
			num = COMNative.CoResumeClassObjects();
			if (num != 0)
			{
				if ((long)_cookieSimpleObj != 0L)
				{
					COMNative.CoRevokeClassObject(_cookieSimpleObj);
				}
				throw new ApplicationException("CoResumeClassObjects failed w/err 0x" + num.ToString("X"));
			}
			_nMainThreadID = NativeMethod.GetCurrentThreadId();
			_nLockCnt = 0;
			_gcTimer = new System.Threading.Timer(GarbageCollect, null, 5000, 5000);
		}

		private void RunMessageLoop()
		{
			MSG lpMsg;
			while (NativeMethod.GetMessage(out lpMsg, IntPtr.Zero, 0u, 0u))
			{
				MSG lpMsg2 = lpMsg;
				NativeMethod.TranslateMessage(ref lpMsg2);
				lpMsg2 = lpMsg;
				NativeMethod.DispatchMessage(ref lpMsg2);
			}
		}

		private void PostMessageLoop()
		{
			if ((long)_cookieSimpleObj != 0L)
			{
				COMNative.CoRevokeClassObject(_cookieSimpleObj);
			}
			if (_gcTimer != null)
			{
				_gcTimer.Dispose();
			}
			Thread.Sleep(1000);
		}

		public void Run()
		{
			object obj = syncRoot;
			ObjectFlowControl.CheckForSyncLockOnValueType(obj);
			bool lockTaken = false;
			try
			{
				Monitor.Enter(obj, ref lockTaken);
				if (_bRunning)
				{
					return;
				}
				_bRunning = true;
			}
			finally
			{
				if (lockTaken)
				{
					Monitor.Exit(obj);
				}
			}
			try
			{
				LegacyRecallLogger.Logger = new LegacyRecallLogger(isFullLog: true);
				LegacyRecallLogger.Logger.Log("Beginning logging", Category.Info, Priority.Low);
				PreMessageLoop();
				RunMessageLoop();
				PostMessageLoop();
			}
			catch (Exception ex)
			{
				ProjectData.SetProjectError(ex);
				Exception e = ex;
				LegacyRecallLogger.LogExceptionStackTrace(string.Empty, e);
				ProjectData.ClearProjectError();
			}
			finally
			{
				_bRunning = false;
			}
		}

		public int Lock()
		{
			return Interlocked.Increment(ref _nLockCnt);
		}

		public int Unlock()
		{
			int num = Interlocked.Decrement(ref _nLockCnt);
			if (num == 0)
			{
				NativeMethod.PostThreadMessage(_nMainThreadID, 18u, UIntPtr.Zero, IntPtr.Zero);
			}
			return num;
		}

		public int GetLockCount()
		{
			return _nLockCnt;
		}

				try
				{
					string contents = $"[{DateTime.Now.ToString(_DateFormat)}]:{message}{Environment.NewLine}";
					File.AppendAllText(_filePath, contents);
				}
				catch (Exception ex)
				{
					ProjectData.SetProjectError(ex);
					Exception ex2 = ex;
					ProjectData.ClearProjectError();
				}
			}
		}

		void ILoggerFacade.Log(string message, Category category, Priority priority)
		{
			//ILSpy generated this explicit interface implementation from .override directive in Log
			this.Log(message, category, priority);
		}

		public static void LogExceptionStackTrace(string preStackMessage, Exception e)
		{
			if (e != null)
			{
				if (string.IsNullOrEmpty(preStackMessage))
				{
					preStackMessage = "Exception";
				}
				string text = string.Format("{3}: {0}{1}{2}", e.Message, Environment.NewLine, e.StackTrace, preStackMessage);
				while (e.InnerException != null)
				{
					e = e.InnerException;
					text += string.Format("{1}  Inner exception {0}{1}{2}", e.Message, Environment.NewLine, e.StackTrace);
				}
				Logger.Log(text, Category.Exception, Priority.High);
			}
		}
