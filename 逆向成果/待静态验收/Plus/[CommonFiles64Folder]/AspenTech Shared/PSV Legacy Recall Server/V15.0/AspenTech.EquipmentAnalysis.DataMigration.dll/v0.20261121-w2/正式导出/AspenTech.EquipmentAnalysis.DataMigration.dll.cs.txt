	private List<IProtoReliefDevice> ProtoPrds = new List<IProtoReliefDevice>();

	public readonly PSVDatabaseMgr DB;

	private IAdbTop AdbTop;

	private ProtoCaseConfiguration ProtoCaseConfiguration;

	public DataTranscriber(string databasePath, string caseKey)
	{
		//IL_0014: Unknown result type (might be due to invalid IL or missing references)
		//IL_001e: Expected O, but got Unknown
		DB = new PSVDatabaseMgr();
		UnitConversion.ParseUnitsFile();
		DB.Initialize(databasePath, caseKey);
	}

	public void RecallAndTranscribe(IAdbTop adbTop)
	{
		AdbTop = adbTop;
		RecallDefaults();
		RecallCaseConfiguration();
		RecallTanks();
		RecallPrdSystems();
	}

	public void CloseAndCleanUp()
	{
		DB.DatabaseCleanUp();
	}

	private void RecallDefaults()
	{
		PSVDBKey simulationSettingsKey = DB.GetAllKeysOfType(eRecordType.SimulationSettings).Single();
		ProtoDefaultValueDefiner protoDefaultValueDefiner = new ProtoDefaultValueDefiner(DB, simulationSettingsKey);
		protoDefaultValueDefiner.InstantiateFields();
		protoDefaultValueDefiner.CopyData(SafetyAnalysisCase.ActiveDefaultDefiner);
	}

	private void RecallCaseConfiguration()
	{
		PSVDBKey simulationSettingsKey = DB.GetAllKeysOfType(eRecordType.SimulationSettings).Single();
		ProtoCaseConfiguration = new ProtoCaseConfiguration(DB, simulationSettingsKey);
		ProtoCaseConfiguration.InstantiateFields();
		ProtoCaseConfiguration.CopyData(SafetyAnalysisCase.ActiveConfiguration);
	}

	private void RecallTanks()
	{
		PSVDBKey[] allKeysOfType = DB.GetAllKeysOfType(eRecordType.Utility);
		List<ProtoStorageTank> list = allKeysOfType.Select((PSVDBKey uk) => new ProtoStorageTank(DB, uk)).ToList();
		foreach (ProtoStorageTank item in list)
		{
			item.InstantiateFields();
		}
		foreach (ProtoStorageTank item2 in list)
		{
			StorageTank storageTank = StorageTank.CreateNew(AdbTop);
			storageTank.FixUpLegacyKey(item2.UtilityKey.Value);
			item2.CopyData(storageTank);
		}
	}

	private void RecallPrdSystems()
	{
		PSVDBKey[] allKeysOfType = DB.GetAllKeysOfType(eRecordType.PRDSystem);
		List<ProtoPressureProtectionRequirement> list = allKeysOfType.Select((PSVDBKey prdsk) => new ProtoPressureProtectionRequirement(DB, prdsk)).ToList();
		foreach (ProtoPressureProtectionRequirement item in list)
		{
			item.InstantiateFields();
		}
		foreach (ProtoPressureProtectionRequirement item2 in list)
		{
			PressureProtectionRequirement pressureProtectionRequirement = PressureProtectionRequirement.CreateNew(AdbTop);
			pressureProtectionRequirement.FixUpLegacyKey(item2.PRDSystemKey.Value);
			item2.CopyData(pressureProtectionRequirement);
			RecallPrds(item2, pressureProtectionRequirement);
			RecallScenarios(item2, pressureProtectionRequirement);
			ProtoDischargeLocation protoDischargeLocation = new ProtoDischargeLocation(DB, item2.PRDSystemKey);
			protoDischargeLocation.InstantiateFields();
			AspenTech.EquipmentAnalysis.Models.DischargeLocation dischargeLocation = AspenTech.EquipmentAnalysis.Models.DischargeLocation.CreateNew(AdbTop);
			protoDischargeLocation.CopyData(dischargeLocation);
			pressureProtectionRequirement.DischargeLocation = dischargeLocation;
		}
	}

	private void RecallPrds(ProtoPressureProtectionRequirement pppr, PressureProtectionRequirement ppr)
	{
		ProtoReliefDeviceConfiguration protoReliefDeviceConfiguration = new ProtoReliefDeviceConfiguration(DB, pppr.PRDSystemKey);
		protoReliefDeviceConfiguration.InstantiateFields();
		protoReliefDeviceConfiguration.CopyData(ppr.SelectedReliefDeviceConfiguration);
		PSVDBKey[] childrenKeys = DB.GetChildrenKeys(pppr.PRDSystemKey, eRecordType.PRD);
		ReliefDeviceType value = ppr.ReliefDeviceType.Value.Value;
		List<IProtoReliefDevice> list = CreateProtoReliefDevices(childrenKeys, value);
		foreach (IProtoReliefDevice item in list)
		{
			ReliefDevice reliefDevice = ppr.CreateReliefDevice(value);
			if (value == ReliefDeviceType.Psv)
			{
				(item as ProtoPressureSafetyValve).CopyData(reliefDevice as PressureSafetyValve);
			}
			else
			{
				(item as ProtoRuptureDisk).CopyData(reliefDevice as RuptureDisk);
			}
			if (item.IsOverallModel())
			{
				ppr.SelectedReliefDeviceConfiguration.TotalOrifice = reliefDevice;
			}
			if (item.IsSpare())
			{
				ppr.SelectedReliefDeviceConfiguration.SpareReliefDevices.Add(reliefDevice);
			}
			else if ((list.Where((IProtoReliefDevice x) => !x.IsSpare()).Count() == 1 && ppr.SelectedReliefDeviceConfiguration.ReliefDevices.Count == 0) || !item.IsOverallModel())
			{
				ppr.SelectedReliefDeviceConfiguration.ReliefDevices.Add(reliefDevice);
			}
		}
	}

	private List<IProtoReliefDevice> CreateProtoReliefDevices(PSVDBKey[] keys, ReliefDeviceType type)
	{
		List<IProtoReliefDevice> list = new List<IProtoReliefDevice>();
		if (type == ReliefDeviceType.Psv)
		{
			foreach (PSVDBKey pRDKey in keys)
			{
				list.Add(new ProtoPressureSafetyValve(DB, pRDKey));
			}
		}
		else
		{
			foreach (PSVDBKey pRDKey2 in keys)
			{
				list.Add(new ProtoRuptureDisk(DB, pRDKey2));
			}
		}
		foreach (IProtoReliefDevice item in list)
		{
			item.InstantiateFields();
		}
		ProtoPrds.AddRange(list);
		ProtoPrds = ProtoPrds.Distinct().ToList();
		return list;
	}

	private void RecallScenarios(ProtoPressureProtectionRequirement pppr, PressureProtectionRequirement ppr)
	{
		PSVDBKey[] childrenKeys = DB.GetChildrenKeys(pppr.PRDSystemKey, eRecordType.ReliefLoad);
		List<ProtoReliefScenario> list = new List<ProtoReliefScenario>();
		HashSet<(ProtoPrdConstraintsDefinition, ProtoPipeLineConstraintsDefinition)> hashSet = new HashSet<(ProtoPrdConstraintsDefinition, ProtoPipeLineConstraintsDefinition)>();
		PSVDBKey[] array = childrenKeys;
		foreach (PSVDBKey reliefLoadKey in array)
		{
			ProtoReliefScenario protoReliefScenario = new ProtoReliefScenario(DB, reliefLoadKey);
			protoReliefScenario.InstantiateFields();
			list.Add(protoReliefScenario);
		}
		foreach (ProtoReliefScenario item in list)
		{
			ReliefScenario reliefScenario = ppr.CreateReliefScenario();
			item.CopyData(reliefScenario);
			if (item.IsSizingCase())
			{
				ppr.SetSizingCase(reliefScenario);
			}
			reliefScenario.ReliefDescription.EmergencyType.Clear(new UserSpecifier());
			ProtoReliefDescription protoReliefDescription = new ProtoReliefDescription(DB, item.ReliefLoadKey);
			protoReliefDescription.InstantiateFields();
			protoReliefDescription.CopyData(reliefScenario.ReliefDescription);
			ProtoReliefLoad protoReliefLoad = new ProtoReliefLoad(DB, item.ReliefLoadKey);
			protoReliefLoad.InstantiateFields();
			ReliefLoad reliefLoad = reliefScenario.ReliefDescription.ReliefLoads[reliefScenario.PressureProtectionRequirement.SelectedReliefDeviceConfiguration];
			protoReliefLoad.CopyData(reliefLoad);
			item.RecallFixupOverpressure(reliefScenario, ProtoPrds, item.IsSizingCase());
			ProtoReliefScenario.UpdateReliefPressure(reliefScenario);
			RecallReliefLoadCalculations(protoReliefLoad, reliefLoad);
			RecallOrificeCalculations(item, reliefScenario, hashSet);
		}
		foreach (var (protoPrdConstraintsDefinition, protoPipeLineConstraintsDefinition) in hashSet)
		{
			protoPrdConstraintsDefinition.RecallFixup();
			protoPipeLineConstraintsDefinition.RecallFixup();
		}
	}

	private void RecallReliefLoadCalculations(ProtoReliefLoad prl, ReliefLoad rl)
	{
		if (((IEnumerable<RLCFanFailureRow>)DB.DS.RLCFanFailure).SingleOrDefault((RLCFanFailureRow row) => row.ReliefLoadID == prl.ReliefLoadKey.Value) != null)
		{
			ProtoFanFailure protoFanFailure = new ProtoFanFailure(DB, prl.ReliefLoadKey);
			protoFanFailure.InstantiateFields();
			FanFailure orCreateReliefLoadCalculation = GetOrCreateReliefLoadCalculation(rl, ReliefLoadCalculationType.FanFailure, FanFailure.CreateNew);
			protoFanFailure.CopyData(orCreateReliefLoadCalculation);
		}
		if (((IEnumerable<RLCRefluxFailureRow>)DB.DS.RLCRefluxFailure).SingleOrDefault((RLCRefluxFailureRow row) => row.ReliefLoadID == prl.ReliefLoadKey.Value) != null)
		{
			ProtoRefluxFailure protoRefluxFailure = new ProtoRefluxFailure(DB, prl.ReliefLoadKey);
			protoRefluxFailure.InstantiateFields();
			RefluxFailure orCreateReliefLoadCalculation2 = GetOrCreateReliefLoadCalculation(rl, ReliefLoadCalculationType.RefluxFailure, RefluxFailure.CreateNew);
			protoRefluxFailure.CopyData(orCreateReliefLoadCalculation2);
		}
		if (((IEnumerable<RLCValveFailureRow>)DB.DS.RLCValveFailure).SingleOrDefault((RLCValveFailureRow row) => row.ReliefLoadID == prl.ReliefLoadKey.Value) != null)
		{
			ProtoControlValveFailure protoControlValveFailure = new ProtoControlValveFailure(DB, prl.ReliefLoadKey);
			protoControlValveFailure.InstantiateFields();
			ControlValveFailure orCreateReliefLoadCalculation3 = GetOrCreateReliefLoadCalculation(rl, ReliefLoadCalculationType.ControlValveFailure, ControlValveFailure.CreateNew);
			protoControlValveFailure.CopyData(orCreateReliefLoadCalculation3);
		}
		if (((IEnumerable<RLCThermalExpansionRow>)DB.DS.RLCThermalExpansion).SingleOrDefault((RLCThermalExpansionRow row) => row.ReliefLoadID == prl.ReliefLoadKey.Value) != null)
		{
			ProtoThermalExpansion protoThermalExpansion = new ProtoThermalExpansion(DB, prl.ReliefLoadKey);
			protoThermalExpansion.InstantiateFields();
			ThermalExpansion orCreateReliefLoadCalculation4 = GetOrCreateReliefLoadCalculation(rl, ReliefLoadCalculationType.ThermalExpansion, ThermalExpansion.CreateNew);
			protoThermalExpansion.CopyData(orCreateReliefLoadCalculation4);
		}
		if (((IEnumerable<RLCTubeRuptureRow>)DB.DS.RLCTubeRupture).SingleOrDefault((RLCTubeRuptureRow row) => row.ReliefLoadID == prl.ReliefLoadKey.Value) != null)
		{
			ProtoTubeRupture protoTubeRupture = new ProtoTubeRupture(DB, prl.ReliefLoadKey);
			protoTubeRupture.InstantiateFields();
			TubeRupture orCreateReliefLoadCalculation5 = GetOrCreateReliefLoadCalculation(rl, ReliefLoadCalculationType.TubeRupture, TubeRupture.CreateNew);
			protoTubeRupture.CopyData(orCreateReliefLoadCalculation5);
		}
		if (((IEnumerable<RLCFireRow>)DB.DS.RLCFire).SingleOrDefault((RLCFireRow row) => row.ReliefLoadID == prl.ReliefLoadKey.Value) != null)
		{
			ProtoFire protoFire = new ProtoFire(DB, prl.ReliefLoadKey);
			protoFire.InstantiateFields();
			Fire orCreateReliefLoadCalculation6 = GetOrCreateReliefLoadCalculation(rl, ReliefLoadCalculationType.Fire, Fire.CreateNew);
			protoFire.CopyData(orCreateReliefLoadCalculation6);
			orCreateReliefLoadCalculation6.CreateFlashManagerIfNull();
			ProtoFireFlashManager protoFireFlashManager = new ProtoFireFlashManager(DB, prl.ReliefLoadKey);
			protoFireFlashManager.InstantiateFields();
			protoFireFlashManager.CopyData(orCreateReliefLoadCalculation6.FireFlashManager);
			RecallFireVessels(prl, orCreateReliefLoadCalculation6);
		}
		rl.UpdateReliefLoadCalculation();
	}

	private T GetOrCreateReliefLoadCalculation<T>(ReliefLoad rl, ReliefLoadCalculationType type, Func<IAdbTop, T> createFunc) where T : ReliefLoadCalculation
	{
		if (rl.ReliefLoadCalculations.ContainsKey(type))
		{
			return (T)rl.ReliefLoadCalculations[type];
		}
		T val = createFunc(AdbTop);
		rl.ReliefLoadCalculations.Add(type, val);
		return val;
	}

	private void RecallFireVessels(ProtoReliefLoad prl, Fire f)
	{
		List<OrificeCalcRow> list = ((IEnumerable<OrificeCalcRow>)DB.DS.OrificeCalc).Where((OrificeCalcRow row) => row.ReliefLoadID == prl.ReliefLoadKey.Value).ToList();
		int overallOrificeCalcKeyValue;
		if (list.Count == 1)
		{
			overallOrificeCalcKeyValue = list.First().ID;
		}
		else
		{
			if (list.Count <= 1)
			{
				return;
			}
			int prdSystemKeyValue = DB.DS.ReliefLoad.FindByID(prl.ReliefLoadKey.Value).PRDSystemID;
			int overallPrdKeyValue = ((IEnumerable<PRDRow>)DB.DS.PRD).SingleOrDefault((PRDRow row) => row.IsOverallModel && row.PRDSystemID == prdSystemKeyValue).ID;
			overallOrificeCalcKeyValue = ((IEnumerable<OrificeCalcRow>)DB.DS.OrificeCalc).SingleOrDefault((OrificeCalcRow row) => row.PRDID == overallPrdKeyValue && row.ReliefLoadID == prl.ReliefLoadKey.Value).ID;
		}
		IEnumerable<int> enumerable = (from row in (IEnumerable<ConnectedObjectRow>)DB.DS.ConnectedObject
			where row.LHSTable == ((DataTable)(object)DB.DS.OrificeCalc).TableName && row.LHSID == overallOrificeCalcKeyValue && row.RHSTable == ((DataTable)(object)DB.DS.Equipment).TableName
			select row.RHSID).Distinct();
		int num = 0;
		foreach (int item in enumerable)
		{
			ProtoVesselExposedToFire protoVesselExposedToFire = new ProtoVesselExposedToFire(DB, new PSVDBKey(eRecordType.Equipment, item));
			protoVesselExposedToFire.InstantiateFields();
			VesselExposedToFire model = f.CreateVessel();
			protoVesselExposedToFire.CopyData(model);
			num++;
			if (num == 3)
			{
				break;
			}
		}
	}

	private void RecallOrificeCalculations(ProtoReliefScenario prs, ReliefScenario rs, HashSet<(ProtoPrdConstraintsDefinition ppcd, ProtoPipeLineConstraintsDefinition pplcd)> constraints)
	{
		IEnumerable<IProtoReliefDevice> enumerable = ProtoPrds.Where((IProtoReliefDevice p) => p.ProtoReliefDevice.RealModel.PressureProtectionRequirement == rs.PressureProtectionRequirement);
		bool flag = rs.PressureProtectionRequirement.ReliefDevices.Count == 1;
		foreach (IProtoReliefDevice pprd in enumerable)
		{
			OrificeCalcRow val = ((IEnumerable<OrificeCalcRow>)DB.DS.OrificeCalc).SingleOrDefault((OrificeCalcRow row) => row.ReliefLoadID == prs.ReliefLoadKey.Value && row.PRDID == pprd.PRDKey.Value);
			if (val == null)
			{
				continue;
			}
			ProtoPrdConstraintsDefinition ppcd = new ProtoPrdConstraintsDefinition(DB, new PSVDBKey(eRecordType.OrificeCalc, val.ID));
			ppcd.InstantiateFields();
			PrdConstraintsDefinition orCreatePrdConstraints = rs.GetOrCreatePrdConstraints(pprd.ProtoReliefDevice.RealModel);
			ppcd.CopyData(orCreatePrdConstraints);
			orCreatePrdConstraints.FixUpLegacyKey(ppcd.OrificeCalcKey.Value);
			orCreatePrdConstraints.ValidateOrificeSizingModelType();
			if (((IEnumerable<OrificeCalcPSVRow>)DB.DS.OrificeCalcPSV).SingleOrDefault((OrificeCalcPSVRow row) => row.OrificeCalcID == ppcd.OrificeCalcKey.Value) != null && orCreatePrdConstraints.OrificeSizing is PsvOrificeSizing model)
			{
				ProtoPsvOrificeSizing protoPsvOrificeSizing = new ProtoPsvOrificeSizing(DB, ppcd.OrificeCalcKey);
				protoPsvOrificeSizing.InstantiateFields();
				protoPsvOrificeSizing.CopyData(model);
				ProtoDetailedReliefDevice protoDetailedReliefDevice = new ProtoDetailedReliefDevice(DB, ppcd.OrificeCalcKey);
				protoDetailedReliefDevice.InstantiateFields();
				protoDetailedReliefDevice.CopyData(orCreatePrdConstraints.GetOrCreateDetailedReliefDevice());
				SetReliefOrifice(orCreatePrdConstraints, ppcd, pprd, protoPsvOrificeSizing.ProtoOrificeSizing, protoPsvOrificeSizing);
			}
			if (((IEnumerable<OrificeCalcKrMethodRow>)DB.DS.OrificeCalcKrMethod).SingleOrDefault((OrificeCalcKrMethodRow row) => row.OrificeCalcID == ppcd.OrificeCalcKey.Value) != null && orCreatePrdConstraints.OrificeSizing is KrMethodOrificeSizing model2)
			{
				ProtoKrMethodOrificeSizing protoKrMethodOrificeSizing = new ProtoKrMethodOrificeSizing(DB, ppcd.OrificeCalcKey);
				protoKrMethodOrificeSizing.InstantiateFields();
				protoKrMethodOrificeSizing.CopyData(model2);
				SetReliefOrifice(orCreatePrdConstraints, ppcd, pprd, protoKrMethodOrificeSizing.ProtoOrificeSizing);
			}
			if (ppcd.IsSizingCase())
			{
				ProtoPipeLine protoPipeLine = new ProtoPipeLine(DB, ppcd.OrificeCalcKey);
				protoPipeLine.InstantiateFields();
				protoPipeLine.CopyData(pprd.ProtoReliefDevice.RealModel.PipeLine);
			}
			LineSizingRow val2 = ((IEnumerable<LineSizingRow>)DB.DS.LineSizing).SingleOrDefault((LineSizingRow row) => row.OrificeCalcID == ppcd.OrificeCalcKey.Value);
			if (val2 != null)
			{
				ProtoPipeLineConstraintsDefinition protoPipeLineConstraintsDefinition = new ProtoPipeLineConstraintsDefinition(DB, new PSVDBKey(eRecordType.LineSizing, val2.ID));
				protoPipeLineConstraintsDefinition.InstantiateFields();
				PipeLineConstraintsDefinition orCreatePipeLineConstraintsDefinition = rs.GetOrCreatePipeLineConstraintsDefinition(pprd.ProtoReliefDevice.RealModel.PipeLine);
				protoPipeLineConstraintsDefinition.CopyData(orCreatePipeLineConstraintsDefinition);
				RecallLineRatingCalculations(protoPipeLineConstraintsDefinition, orCreatePipeLineConstraintsDefinition);
				if (prs.IsSizingCase() || !double.IsNaN(_GetLineSizingResult(protoPipeLineConstraintsDefinition.RealModel.LineRating)))
				{
					protoPipeLineConstraintsDefinition.RealModel.IsRatingCalculated.Specify(true, new UserSpecifier());
				}
				else
				{
					protoPipeLineConstraintsDefinition.RealModel.IsRatingCalculated.Specify(false, new UserSpecifier());
				}
				orCreatePipeLineConstraintsDefinition.ReliefScenario.LineSizingCalculationMethod.Specify(TranslateEnum.LineSizingMethod.Enum(val2.LineSizingMethod), new UserSpecifier());
				ProtoReliefScenario.ValidateLineSizingMethod(orCreatePipeLineConstraintsDefinition.ReliefScenario);
				constraints.Add((ppcd, protoPipeLineConstraintsDefinition));
			}
		}
	}

	private double _GetLineSizingResult(LineRating lr)
	{
		if (lr is HomogeneousLineRating homogeneousLineRating)
		{
			return homogeneousLineRating.OutletPressureDrop.ValueIn(Units.Pa);
		}
		if (lr is SinglePhaseLineRating singlePhaseLineRating)
		{
			return singlePhaseLineRating.OutletPressureDrop.ValueIn(Units.Pa);
		}
		if (lr is OmegaMethodLineRating omegaMethodLineRating)
		{
			return omegaMethodLineRating.OutletMaxEquivalentLength.ValueIn(Units.m);
		}
		return double.NaN;
	}

	private void SetReliefOrifice(PrdConstraintsDefinition pcd, ProtoPrdConstraintsDefinition ppcd, IProtoReliefDevice pprd, ProtoOrificeSizing pos, ProtoPsvOrificeSizing ppos = null)
	{
		if (pprd.ProtoReliefDevice.RealModel.ReliefDeviceType == ReliefDeviceType.Psv)
		{
			if (ppos != null && ppos.SelectedOrificeDesignation() != string.Empty && !double.IsNaN(ppos.SelectedOrificeArea_cm2()))
			{
				string selectedValveName = ProtoCaseConfiguration.OrificeName(ppos.SelectedOrificeDesignation(), ppos.SelectedOrificeArea_cm2());
				ReliefOrifice value = SafetyAnalysisCase.ActiveConfiguration.ReliefOrifices.SingleOrDefault((KeyValuePair<string, ReliefOrifice> pair) => pair.Key == selectedValveName).Value;
				pcd.SetReliefOrifice(value, ppcd.IsSizingCase());
			}
			return;
		}
		Distance distance = pos?.SelectedDiameter?.Value;
		RuptureDiskOrifice ruptureDiskOrifice = ((distance != null) ? SafetyAnalysisCase.ActiveConfiguration.GetRdBuiltInReliefOrifice(distance) : null);
		pcd.SetReliefOrifice(ruptureDiskOrifice, ppcd.IsSizingCase());
		if (ruptureDiskOrifice != null)
		{
			pcd.DetailedReliefDevice.InletFlangeNominalDiameter.TrySpecifyFrom(ruptureDiskOrifice.Diameter);
			pcd.DetailedReliefDevice.OutletFlangeNominalDiameter.TrySpecifyFrom(ruptureDiskOrifice.Diameter);
		}
	}

	private void RecallLineRatingCalculations(ProtoPipeLineConstraintsDefinition pplcd, PipeLineConstraintsDefinition plcd)
	{
		if (((IEnumerable<LineSizing1PhaseRow>)DB.DS.LineSizing1Phase).SingleOrDefault((LineSizing1PhaseRow row) => row.LineSizingID == pplcd.LineSizingKey.Value) != null)
		{
			ProtoSinglePhaseLineRating protoSinglePhaseLineRating = new ProtoSinglePhaseLineRating(DB, pplcd.LineSizingKey);
			protoSinglePhaseLineRating.InstantiateFields();
			SinglePhaseLineRating orCreateLineRatingCalculation = GetOrCreateLineRatingCalculation(plcd, LineSizingMethod.SinglePhase, SinglePhaseLineRating.CreateNew);
			protoSinglePhaseLineRating.CopyData(orCreateLineRatingCalculation);
		}
		if (((IEnumerable<LineSizingHomogeneousRow>)DB.DS.LineSizingHomogeneous).SingleOrDefault((LineSizingHomogeneousRow row) => row.LineSizingID == pplcd.LineSizingKey.Value) != null)
		{
			ProtoHomogeneousLineRating protoHomogeneousLineRating = new ProtoHomogeneousLineRating(DB, pplcd.LineSizingKey);
			protoHomogeneousLineRating.InstantiateFields();
			HomogeneousLineRating orCreateLineRatingCalculation2 = GetOrCreateLineRatingCalculation(plcd, LineSizingMethod.Homogeneous, HomogeneousLineRating.CreateNew);
			protoHomogeneousLineRating.CopyData(orCreateLineRatingCalculation2);
		}
		if (((IEnumerable<LineSizingOmegaRow>)DB.DS.LineSizingOmega).SingleOrDefault((LineSizingOmegaRow row) => row.LineSizingID == pplcd.LineSizingKey.Value) != null)
		{
			ProtoOmegaMethodLineRating protoOmegaMethodLineRating = new ProtoOmegaMethodLineRating(DB, pplcd.LineSizingKey);
			protoOmegaMethodLineRating.InstantiateFields();
			OmegaMethodLineRating orCreateLineRatingCalculation3 = GetOrCreateLineRatingCalculation(plcd, LineSizingMethod.Omega, OmegaMethodLineRating.CreateNew);
			protoOmegaMethodLineRating.CopyData(orCreateLineRatingCalculation3);
		}
	}

	private T GetOrCreateLineRatingCalculation<T>(PipeLineConstraintsDefinition plcd, LineSizingMethod type, Func<IAdbTop, T> createFunc) where T : LineRating
	{
		if (plcd.AllLineRatings.ContainsKey(type))
		{
			return (T)plcd.AllLineRatings[type];
		}
		T val = createFunc(AdbTop);
		plcd.AllLineRatings.Add(type, val);
		return val;
	}

		}
		if ((model.OrificeCalculationMethod.Value ?? OrificeSizingMethod.Vapor) == OrificeSizingMethod.Mixed_2)

	public void RecallFixupOverpressure(ReliefScenario model, IList<IProtoReliefDevice> protoPrds, bool isSizingCase)
	{
		PressureProtectionRequirement pressureProtectionRequirement = model.PressureProtectionRequirement;
		ReliefDeviceConfiguration selectedReliefDeviceConfiguration = pressureProtectionRequirement.SelectedReliefDeviceConfiguration;
		ReliefDevice controllingDevice = null;
		if (selectedReliefDeviceConfiguration.ReliefDevices.Count == 1 || !isSizingCase)
		{
			controllingDevice = model.TotalOrificeConstraints.ReliefDevice;
		}
		else
		{
			controllingDevice = selectedReliefDeviceConfiguration.ReliefDevices[0];
		}
		int majorVersionLoaded = DB.GetMajorVersionLoaded();
		bool forceSingleDevice = false;
		if (majorVersionLoaded < 35)
		{
			forceSingleDevice = !isSizingCase;
		}
		double defaultOverpressure = OverpressureDefaultCalculator.GetDefaultOverpressure(model.TotalOrificeConstraints, null, forceSingleDevice);
		IProtoReliefDevice protoPrd = protoPrds.Single((IProtoReliefDevice x) => x.ProtoReliefDevice.RealModel == controllingDevice);
		OrificeCalcRow val = ((IEnumerable<OrificeCalcRow>)DB.DS.OrificeCalc).SingleOrDefault((OrificeCalcRow row) => row.ReliefLoadID == ReliefLoadKey.Value && row.PRDID == protoPrd.PRDKey.Value);
		PSVDBKey recordDBKey = DB.GetRecordDBKey(eRecordType.OrificeCalc, (int)((DataRow)(object)val)[DB.DS.OrificeCalc.IDColumn]);
		double num = DB.GetValue<double>(recordDBKey, "OverPress") / 100.0;
		if (MyMath.ValuesDiffer(num, defaultOverpressure, 1E-06, 1E-12))
		{
			model.TotalOrificeConstraints.UseCustomOverpressure.Specify(true, new UserSpecifier());
			model.TotalOrificeConstraints.AllowableOverpressure.Specify(num, Units.One, new UserSpecifier());
			selectedReliefDeviceConfiguration.UseApiArrangement.Specify(false, new UserSpecifier());
		}
	}

	protected static ISet<LineSizingMethod> GetAllowedLineSizingModels(ReliefScenario model)
	{
		HashSet<LineSizingMethod> hashSet = new HashSet<LineSizingMethod>();
		OrificeSizingMethod valueOrDefault = model.OrificeCalculationMethod.Value.GetValueOrDefault();
		ReliefDeviceType valueOrDefault2 = model.PressureProtectionRequirement.ReliefDeviceType.Value.GetValueOrDefault();
		if (valueOrDefault2 == ReliefDeviceType.RuptureDisk && valueOrDefault == OrificeSizingMethod.KrMethod)
		{
			hashSet.Add(LineSizingMethod.Rigorous);
		}
		else if (valueOrDefault == OrificeSizingMethod.Liquid || valueOrDefault == OrificeSizingMethod.Vapor || valueOrDefault == OrificeSizingMethod.Steam)
		{
			hashSet.Add(LineSizingMethod.SinglePhase);
		}
		else
		{
			hashSet.Add(LineSizingMethod.Homogeneous);
			if (valueOrDefault2 != ReliefDeviceType.RuptureDisk && valueOrDefault != OrificeSizingMethod.Mixed_Direct)
			{
				hashSet.Add(LineSizingMethod.Omega);
			}
		}
		return hashSet;
	}

	protected static LineSizingMethod GetDefaultLineSizingModel(ReliefScenario model)
	{
		OrificeSizingMethod valueOrDefault = model.OrificeCalculationMethod.Value.GetValueOrDefault();
		ReliefDeviceType valueOrDefault2 = model.PressureProtectionRequirement.ReliefDeviceType.Value.GetValueOrDefault();
		if (valueOrDefault2 == ReliefDeviceType.RuptureDisk && valueOrDefault == OrificeSizingMethod.KrMethod)
		{
			return LineSizingMethod.Rigorous;
		}
		if (valueOrDefault == OrificeSizingMethod.Liquid || valueOrDefault == OrificeSizingMethod.Vapor || valueOrDefault == OrificeSizingMethod.Steam)
		{
			return LineSizingMethod.SinglePhase;
		}
		if (valueOrDefault == OrificeSizingMethod.Mixed_Direct || valueOrDefault2 == ReliefDeviceType.RuptureDisk)
		{
			return LineSizingMethod.Homogeneous;
		}
		return LineSizingMethod.Omega;
	}

	public static void ValidateLineSizingMethod(ReliefScenario model)
	{
		LineSizingMethod valueOrDefault = model.LineSizingCalculationMethod.Value.GetValueOrDefault();
		if (!GetAllowedLineSizingModels(model).Contains(valueOrDefault))
		{
			model.LineSizingCalculationMethod.Specify(GetDefaultLineSizingModel(model), new UserSpecifier());
		}
	}

	public static void UpdateReliefPressure(ReliefScenario model)
	{
		double number = model.TotalOrificeConstraints.AllowableOverpressure.Value.Number;
		double num = model.TotalOrificeConstraints.ReliefDevice.SetPressure.ValueIn(Units.bar, PressureReference.Gauge);
		double value = (1.0 + number) * num;
		model.ReliefPressure.Specify(new Pressure(value, Units.bar, PressureReference.Gauge), new CalculationSpecifier());
	}

		model.Fluid.SpecificHeatRatio.Specify(FluidSpecificHeatRatio.Value, new CalculationSpecifier());
		model.Fluid.MolarMass.Specify(FluidMolarMass.Value, new CalculationSpecifier());
		model.Fluid.Compressibility.Specify(FluidCompressibility.Value, new CalculationSpecifier());

		model.Fluid.MassEnthalpyOfVaporization.Specify(FluidMassEnthalpyOfVaporization.Value, new CalculationSpecifier());
		ReliefTemperatureSource valueOrDefault = model.ReliefTemperatureSource.Value.GetValueOrDefault();
		if ((valueOrDefault == AspenTech.EquipmentAnalysis.Infrastructure.ReliefTemperatureSource.Reference || valueOrDefault == AspenTech.EquipmentAnalysis.Infrastructure.ReliefTemperatureSource.Specified) && !model.VaporFraction.IsEmpty)
		{
			model.VaporFraction.Clear(new UserSpecifier());
		}
		if (model.ReliefLoadCalculation is Fire)
		{
			model.Fluid.LiquidMassEnthalpyOfVaporization.Specify(FluidLiquidMassEnthalpyOfVaporization.Value, new CalculationSpecifier());
			return;
		}
		OrificeSizingMethod? orificeSizingMethod = model.FindReliefScenario()?.OrificeCalculationMethod?.Value;
		if (orificeSizingMethod.HasValue && orificeSizingMethod == OrificeSizingMethod.Liquid)
		{
			model.Fluid.LiquidMassEnthalpyOfVaporization.Specify(FluidMassEnthalpyOfVaporization.Value, new CalculationSpecifier());
		}
	}

	public void RecallFixup()
	{
		PipeLineConstraintsDefinition realModel = base.RealModel;
		if (realModel.PipeLine.ReliefDevice.ReliefDeviceType != ReliefDeviceType.Psv)
		{
			return;
		}
		int majorVersionLoaded = DB.GetMajorVersionLoaded();
		if (majorVersionLoaded >= 35)
		{
			return;
		}
		ReliefScenario reliefScenario = realModel.ReliefScenario;
		LineSizingMethod? value = reliefScenario.LineSizingCalculationMethod.Value;
		if (!value.HasValue)
		{
			return;
		}
		LineSizingMethod valueOrDefault = value.GetValueOrDefault();
		if (valueOrDefault == LineSizingMethod.Omega)
		{
			realModel.LineRatingFlowSpec.Specify(AspenTech.EquipmentAnalysis.Infrastructure.LineRatingFlowSpec.Required, new UserSpecifier());
			return;
		}
		if (realModel.UpratedFlow.IsEmpty)
		{
			if (majorVersionLoaded < 30 && valueOrDefault == LineSizingMethod.Homogeneous)
			{
				realModel.LineRatingFlowSpec.Specify(AspenTech.EquipmentAnalysis.Infrastructure.LineRatingFlowSpec.Required, new UserSpecifier());
			}
			else
			{
				realModel.LineRatingFlowSpec.Clear(new UserSpecifier());
			}
			return;
		}
		double num = realModel.UpratedFlow.ValueIn(Units.kg_per_h);
		double num2 = 5E-05 * num;
		PrdConstraintsDefinition value2 = null;
		if (!realModel.ReliefScenario.PrdConstraints.TryGetValue(realModel.PipeLine.ReliefDevice, out value2))
		{
			return;
		}
		double num3 = value2.OrificeSizing.SelectedCapacity.ValueIn(Units.kg_per_h);
		double num4 = value2.FlowFraction.Value.Number * realModel.ReliefScenario.ReliefLoad.MassFlow.ValueIn(Units.kg_per_h);
		if (Math.Abs(num3 - num4) < num2 && Math.Abs(num3 - num) < 2.0 * num2)
		{
			if (realModel.ReliefScenario.ReliefDescription.EmergencyType.Value == EmergScenarioType.Fire || realModel.ReliefScenario.ReliefLoad.Phase.Value == FluidPhaseRegime.Liquid || realModel.ReliefScenario.OrificeCalculationMethod.Value == OrificeSizingMethod.Mixed_3)
			{
				realModel.LineRatingFlowSpec.Specify(AspenTech.EquipmentAnalysis.Infrastructure.LineRatingFlowSpec.Required, new UserSpecifier());
			}
			else
			{
				realModel.LineRatingFlowSpec.Specify(AspenTech.EquipmentAnalysis.Infrastructure.LineRatingFlowSpec.RatedArea, new UserSpecifier());
			}
		}
		else if (Math.Abs(num3 - num) < num2)
		{
			realModel.LineRatingFlowSpec.Specify(AspenTech.EquipmentAnalysis.Infrastructure.LineRatingFlowSpec.RatedArea, new UserSpecifier());
		}
		else if (Math.Abs(num4 - num) < num2)
		{
			realModel.LineRatingFlowSpec.Specify(AspenTech.EquipmentAnalysis.Infrastructure.LineRatingFlowSpec.Required, new UserSpecifier());
		}
		else
		{
			realModel.LineRatingFlowSpec.Specify(AspenTech.EquipmentAnalysis.Infrastructure.LineRatingFlowSpec.Manual, new UserSpecifier());
		}
	}
}

		model.Api520Part1Edition.Specify(Api520Part1Edition.Edition9, new UserSpecifier());
		model.LastSavedLocale.Specify("en-US", new CalculationSpecifier());
		model.ForceLocaleCheck();
		model.SetForceCalculateFlag();

	}

	private void TransferCustomOrifices(CaseConfiguration model)
	{
		for (int i = 0; i < 19; i++)
		{
			if (IsUserDefined(i))
			{
				string text = UserDesignation(i);
				string text2 = text;
				while (!model.IsUniqueReliefOrificeName(text2))
				{
					text2 += "*";
				}
				DetailedReliefDevice detailedReliefDevice = model.CreatePsvOrifice(text2, TranslateEnum.ApiDesignation.Enum(text), UserArea(i), string.Empty);
				detailedReliefDevice.InletFlangeNominalDiameter.Specify(UserInletFlange(i), new UserSpecifier());
				detailedReliefDevice.InletRating.Specify(UserInletRating(i), new UserSpecifier());
				detailedReliefDevice.OutletFlangeNominalDiameter.Specify(UserOutletFlange(i), new UserSpecifier());
				detailedReliefDevice.OutletRating.Specify(UserOutletRating(i), new UserSpecifier());
			}
		}
	}

	{
		return DB.GetValue<bool>(SimulationSettingsKey, $"GLOBUser({i + 1})");
	}

	public string OrificeName(string oldDesignation, double orificeSize_cm2)
	{
		CaseConfiguration activeConfiguration = SafetyAnalysisCase.ActiveConfiguration;
		AspenTech.EquipmentAnalysis.Infrastructure.ApiDesignation? oldDes = TranslateEnum.ApiDesignation.Enum(oldDesignation);
		List<KeyValuePair<string, ReliefOrifice>> source = activeConfiguration.ReliefOrifices.Where((KeyValuePair<string, ReliefOrifice> pair) => pair.Value.ReliefDeviceType == ReliefDeviceType.Psv && pair.Value.Designation.Value == oldDes).ToList();
		if (source.Count() == 1)
		{
			return source.Single().Key;
		}
		if (source.Count() == 0)
		{
			return string.Empty;
		}
		Func<ReliefOrifice, double> f = (ReliefOrifice ro) => AbsDiffFromOrificeSize(ro, orificeSize_cm2);
		return source.Select((KeyValuePair<string, ReliefOrifice> pair) => pair.Value).Aggregate(null, (ReliefOrifice currMin, ReliefOrifice ro) => (f(ro) < f(currMin)) ? ro : currMin, (ReliefOrifice min) => min.Name.Value);
	}

	private static double AbsDiffFromOrificeSize(ReliefOrifice ro, double orificeSize_cm2)
	{
		return (ro == null) ? double.MaxValue : Math.Abs(ro.OrificeArea.ValueIn(Units.cm2) - orificeSize_cm2);
	}

{
	public readonly PSVDatabaseMgr DB;

	public T RealModel { get; private set; }

	protected ProtoModel(PSVDatabaseMgr db)
	{
		DB = db;
	}

	protected virtual void InstantiateBoilerplateFields()
	{
	}

	protected virtual void InstantiateCustomFields()
	{
	}

	public void InstantiateFields()
	{
		InstantiateBoilerplateFields();
		InstantiateCustomFields();
	}

	protected virtual void CopyBoilerplateData(T model)
	{
	}

	protected virtual void CopyCustomData(T model)
	{
	}

	public void CopyData(T model)
	{
		RealModel = model;
		if (model != null)
		{
			CopyBoilerplateData(model);
			CopyCustomData(model);
		}
	}
}
