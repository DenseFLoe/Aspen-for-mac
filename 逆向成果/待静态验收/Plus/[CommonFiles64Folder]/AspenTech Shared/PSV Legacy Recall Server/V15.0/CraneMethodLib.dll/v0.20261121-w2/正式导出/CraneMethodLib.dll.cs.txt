	private static double LinearInterpolation(double X0, double Y0, double X1, double Y1, double X)
	{
		return Y0 + (Y1 - Y0) * (X - X0) / (X1 - X0);
	}

	public static double CalculateHomogeneousEquivalentDensity(double DensityL, double DensityG, double Quality)
	{
		return 1.0 / ((1.0 - Quality) / DensityL + Quality / DensityG);
	}

	}
