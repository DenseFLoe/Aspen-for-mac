	public static double ThermalResistanceOfPlaneWall(double Conductivity, double WallThickness, double Area)
	{
		try
		{
			if (Area == RealValue.FEMPTY || Area < 0.0)
			{
				Area = 1.0;
			}
			if (Conductivity <= 0.0)
			{
				throw new ArgumentOutOfRangeException("Conductivity", Conductivity, Resources.MustBeGreaterThanZero);
			}
			return WallThickness / (Conductivity * Area);
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "ThermalResistanceOfPlaneWall"), innerException);
		}
	}

	public static double ThermalResistanceOfCylinder(double Conductivity, double InnerDiam, double OuterDiam, double Length)
	{
		try
		{
			if (Length == RealValue.FEMPTY || Length < 0.0)
			{
				Length = 1.0;
			}
			if (InnerDiam <= 0.0 || InnerDiam > OuterDiam)
			{
				throw new ArgumentOutOfRangeException("InnerDiam", InnerDiam, Resources.MustBeGreaterThanZero2);
			}
			if (OuterDiam <= 0.0)
			{
				throw new ArgumentOutOfRangeException("OuterDiam", OuterDiam, Resources.MustBeGreaterThanZero);
			}
			if (Conductivity <= 0.0)
			{
				throw new ArgumentOutOfRangeException("Conductivity", Conductivity, Resources.MustBeGreaterThanZero);
			}
			return Math.Log(OuterDiam / InnerDiam) / (Math.PI * 2.0 * Conductivity * Length);
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "ThermalResistanceOfCylinder"), innerException);
		}
	}

	public static double ThermalResistanceOfBuriedCylinder(double Conductivity, double Diameter, double Depth)
	{
		double num = 0.0;
		try
		{
			if (Depth < 0.5 * Diameter)
			{
				throw new Exception(Resources.CylinderNotBuried);
			}
			double num2;
			if (Depth < 2.0 * Diameter)
			{
				num = Math.Cosh(2.0 * Depth / Diameter);
				num2 = Math.Log(Math.Abs(num + Math.Sqrt(num * num - 1.0)));
			}
			else if (Depth > 100.0 * Diameter)
			{
				num2 = Math.Log(4.0 * Depth / Diameter);
			}
			else
			{
				num = 2.0 * Depth / Diameter;
				num2 = Math.Log(num + Math.Sqrt(num * num - 1.0));
			}
			return num2 / (Math.PI * 2.0 * Conductivity);
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "ThermalResistanceOfBuriedCylinder"), innerException);
		}
	}

	public static double ThermalResistanceOfSphere(double Conductivity, double InnerRadius, double OuterRadius)
	{
		try
		{
			if (InnerRadius <= 0.0 || InnerRadius > OuterRadius)
			{
				throw new ArgumentOutOfRangeException("InnerRadius", InnerRadius, Resources.MustBeGreaterThanZero3);
			}
			if (OuterRadius <= 0.0)
			{
				throw new ArgumentOutOfRangeException("OuterRadius", OuterRadius, Resources.MustBeGreaterThanZero);
			}
			if (Conductivity <= 0.0)
			{
				throw new ArgumentOutOfRangeException("Conductivity", Conductivity, Resources.MustBeGreaterThanZero);
			}
			return 1.0 / (Math.PI * 4.0 * Conductivity) * (1.0 / InnerRadius - 1.0 / OuterRadius);
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "ThermalResistanceOfSphere"), innerException);
		}
	}

	public static double ThermalResistanceForConvection(double HTCoeff, double Area)
	{
		try
		{
			if (Area == RealValue.FEMPTY || Area < 0.0)
			{
				Area = 1.0;
			}
			if (HTCoeff <= 0.0)
			{
				throw new ArgumentOutOfRangeException("HTCoeff", HTCoeff, Resources.MustBeGreaterThanZero);
			}
			return 1.0 / (HTCoeff * Area);
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "ThermalResistanceForConvection"), innerException);
		}
	}

	public static double HeatFlux(double T1, double T2, double ThermalResistance)
	{
		try
		{
			if (ThermalResistance <= 0.0)
			{
				throw new ArgumentOutOfRangeException("ThermalResistance", ThermalResistance, Resources.MustBeGreaterThanZero);
			}
			return (T1 - T2) / ThermalResistance;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "HeatFlux"), innerException);
		}
	}

	public static double IntegrationBySimpsonRule(double Xmin, double Xmax, double PreAnswer, double Accuracy, double[] Y, out int N, out bool isTheAnswer)
	{

		{

			double num = (Xmax - Xmin) / (double)N;

			double num3 = 0.0;

			{

			}
