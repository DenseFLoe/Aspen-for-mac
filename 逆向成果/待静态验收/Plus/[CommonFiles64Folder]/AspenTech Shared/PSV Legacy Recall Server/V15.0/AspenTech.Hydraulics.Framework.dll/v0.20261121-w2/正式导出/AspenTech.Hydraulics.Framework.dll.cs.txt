[Guid("7728ADF7-EED3-4776-BB8E-0897ABBF0BA6")]
public class HydraulicPhase : Phase, IHydraulicPhase, IPhase, ICloneable
{
	private IRealValue massFlowRate;

	private IRealValue velocity;

	private IRealValue perCentAreaOccupied;

	public IRealValue MassFlowRate
	{
		get
		{
			return massFlowRate;
		}
		set
		{
			massFlowRate.Value = value.Value;
			if (massFlowRate is RealValueWithCallBack)
			{
				RealValueWithCallBack realValueWithCallBack = massFlowRate as RealValueWithCallBack;
				if (realValueWithCallBack.CallBack != null)
				{
					realValueWithCallBack.CallBack();
				}
			}
		}
	}

	public IRealValue DF_DP

	public HydraulicPhase(HydraulicPhase refPhase)
		: base(refPhase)
	{
		InitialiseRealValues();
		if (!refPhase.PerCentAreaOccupied.IsUnknown)
		{
			PerCentAreaOccupied.Value = refPhase.PerCentAreaOccupied.Value;
		}
		if (!refPhase.Velocity.IsUnknown)
		{
			Velocity.Value = refPhase.Velocity.Value;
		}
		if (!refPhase.MassFlowRate.IsUnknown)
		{
			MassFlowRate.Value = refPhase.MassFlowRate.Value;
		}
		if (!refPhase.DF_DP.IsUnknown)
		{
			DF_DP.Value = refPhase.DF_DP.Value;
		}
		if (!refPhase.DP_DF.IsUnknown)
		{
			DP_DF.Value = refPhase.DP_DF.Value;
		}
		IRealValue[,] array;
		if (refPhase.dFi_DPj != null)
		{
			int num = refPhase.dFi_DPj.GetUpperBound(0) + 1;
			array = new RealValue[num, num];
			dFi_DPj = array;
			for (int i = 0; i < num; i++)
			{
				for (int j = 0; j < num; j++)
				{
					dFi_DPj[j, i] = new RealValue();
					if (!refPhase.dFi_DPj[j, i].IsUnknown)
					{
						dFi_DPj[j, i].Value = refPhase.dFi_DPj[j, i].Value;
					}
				}
			}
		}
		if (refPhase.dPi_DFj == null)
		{
			return;
		}
		int num2 = refPhase.dPi_DFj.GetUpperBound(0) + 1;
		array = new RealValue[num2, num2];
		dPi_DFj = array;
		for (int k = 0; k < num2; k++)
		{
			for (int l = 0; l < num2; l++)
			{
				dPi_DFj[l, k] = new RealValue();
				if (!refPhase.dPi_DFj[l, k].IsUnknown)
				{
					dPi_DFj[l, k].Value = refPhase.dPi_DFj[l, k].Value;
				}
			}
		}
	}

	public HydraulicPhase()
	{
		InitialiseRealValues();
	}

	public HydraulicPhase(int NComponents)
		: base(NComponents)
	{
		InitialiseRealValues();
	}

	private void InitialiseRealValues()
	{
		massFlowRate = new RealValueWithCallBack(-32767.0);
		massFlowRate.Erase();
		velocity = new RealValue();
		perCentAreaOccupied = new RealValue();
		dF_DP = new RealValue();
		dP_DF = new RealValue();
	}

	internal void AssignDerivativesDimension(int arrayDim)
	{
		IRealValue[,] array = new RealValue[arrayDim, arrayDim];
		dFi_DPj = array;
		array = new RealValue[arrayDim, arrayDim];
		dPi_DFj = array;
		for (int i = 0; i < arrayDim; i++)
		{
			for (int j = 0; j < arrayDim; j++)
			{
				dFi_DPj[i, j] = new RealValue();
				dPi_DFj[i, j] = new RealValue();
			}
		}
	}

	public override object Clone()
	{
		return new HydraulicPhase(this);
	}

	public override void AvaragePhase(IPhase phase1, IPhase phase2, double localDistance, double totalDistance, bool overrideExistingData)
	{
		try
		{
			IRealValue realValue = null;
			IRealValue realValue2 = null;
			IRealValue realValue3 = null;
			IHydraulicPhase hydraulicPhase = null;
			IHydraulicPhase hydraulicPhase2 = null;
			hydraulicPhase = phase1 as IHydraulicPhase;
			hydraulicPhase2 = phase2 as IHydraulicPhase;
			base.AvaragePhase(phase1, phase2, localDistance, totalDistance, overrideExistingData);
			realValue = hydraulicPhase.MassFlowRate;
			realValue2 = hydraulicPhase2.MassFlowRate;
			realValue3 = MassFlowRate;
			Phase.PlugAverageValue(realValue, realValue2, realValue3, overrideExistingData, localDistance, totalDistance);
			realValue = hydraulicPhase.Velocity;
			realValue2 = hydraulicPhase2.Velocity;
			realValue3 = Velocity;
			Phase.PlugAverageValue(realValue, realValue2, realValue3, overrideExistingData, localDistance, totalDistance);
			realValue = hydraulicPhase.PerCentAreaOccupied;
			realValue2 = hydraulicPhase2.PerCentAreaOccupied;
			realValue3 = PerCentAreaOccupied;
			Phase.PlugAverageValue(realValue, realValue2, realValue3, overrideExistingData, localDistance, totalDistance);
			realValue = hydraulicPhase.DF_DP;
			realValue2 = hydraulicPhase2.DF_DP;
			realValue3 = DF_DP;
			Phase.PlugAverageValue(realValue, realValue2, realValue3, overrideExistingData, localDistance, totalDistance);
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodForObjectFailed, "AvaragePhase", "HydraulicPhase"), innerException);
		}
	}

	public override void UpdatePhaseByValue(IPhase refPhase)
	{
		base.UpdatePhaseByValue(refPhase);
		if (!(refPhase is HydraulicPhase hydraulicPhase))
		{
			return;
		}
		if (!hydraulicPhase.PerCentAreaOccupied.IsUnknown)
		{
			PerCentAreaOccupied.Value = hydraulicPhase.PerCentAreaOccupied.Value;
		}
		if (!hydraulicPhase.Velocity.IsUnknown)
		{
			Velocity.Value = hydraulicPhase.Velocity.Value;
		}
		if (!hydraulicPhase.MassFlowRate.IsUnknown)
		{
			MassFlowRate.Value = hydraulicPhase.MassFlowRate.Value;
		}
		if (!hydraulicPhase.DF_DP.IsUnknown)
		{
			DF_DP.Value = hydraulicPhase.DF_DP.Value;
		}
		if (!hydraulicPhase.DP_DF.IsUnknown)
		{
			DP_DF.Value = hydraulicPhase.DP_DF.Value;
		}
		IRealValue[,] array;
		if (hydraulicPhase.dFi_DPj != null)
		{
			int num = hydraulicPhase.dFi_DPj.GetUpperBound(0) + 1;
			array = new RealValue[num, num];
			dFi_DPj = array;
			for (int i = 0; i < num; i++)
			{
				for (int j = 0; j < num; j++)
				{
					dFi_DPj[j, i] = new RealValue();
					if (!hydraulicPhase.dFi_DPj[j, i].IsUnknown)
					{
						dFi_DPj[j, i].Value = hydraulicPhase.dFi_DPj[j, i].Value;
					}
				}
			}
		}
		if (hydraulicPhase.dPi_DFj == null)
		{
			return;
		}
		int num2 = hydraulicPhase.dPi_DFj.GetUpperBound(0) + 1;
		array = new RealValue[num2, num2];
		dPi_DFj = array;
		for (int k = 0; k < num2; k++)
		{
			for (int l = 0; l < num2; l++)
			{
				dPi_DFj[l, k] = new RealValue();
				if (!hydraulicPhase.dPi_DFj[l, k].IsUnknown)
				{
					dPi_DFj[l, k].Value = hydraulicPhase.dPi_DFj[l, k].Value;
				}
			}
		}
	}

[ComVisible(false)]
public class LayerCollection : ILayerCollection
{
	internal ArrayList insulationLayers;

	public IInsulationLayer this[int index] => insulationLayers[index] as IInsulationLayer;

	internal void SetInsulationLayers(int Nlayers)
	{
		IInsulationLayer insulationLayer = null;
		int i = 0;
		insulationLayers = new ArrayList(Nlayers);
		for (; i < Nlayers; i++)
		{
			insulationLayers.Add(new InsulationLayer());
		}
		insulationLayer = insulationLayers[0] as IInsulationLayer;
		insulationLayer.MaterialType = "PipeWall";
		for (i = 1; i < Nlayers; i++)
		{
			insulationLayer = insulationLayers[i] as IInsulationLayer;
			insulationLayer.MaterialType = "Empty";
		}
	}
}
[Guid("0C8320BF-97E9-42C4-888F-D09E3FF13831")]
[ComVisible(false)]
public class Insulation : IInsulation, ICloneable
{
	private int nLayers;

	private ILayerCollection layerCollection;

	private IRealValue internalDiameter;

	private IHydraulicFlowStream externalFlowStream;

	private IHydraulicFlowStream internalFlowStream;

	private IRealValue externalTemperature;

	private IRealValue internalTemperature;

	private ExternalMedia externalMedia;

	private IRealValue mEmissivity;

	private bool mIncludeRadiation;

	public int NLayers => nLayers;

	public IInsulationLayer PipeWall
	{
		get
		{
			IInsulationLayer result = null;
			for (int i = 0; i < NLayers; i++)
			{
				if (InsulationLayers[i].MaterialType == "PipeWall")
				{
					result = InsulationLayers[i];
					i = NLayers;
				}
			}
			return result;
		}
	}

	public IRealValue InternalDiameter
	{
		get
		{
			return internalDiameter;
		}
		set
		{
			internalDiameter = value;
		}
	}

	public ILayerCollection InsulationLayers => layerCollection;

	public double ExternalHT => this.externalHT(this, new object());

	public IRealValue ExternalTemperature => externalTemperature;

	public IHydraulicFlowStream ExternalFlowStream
	{
		get
		{
			return externalFlowStream;
		}
		set
		{
			externalFlowStream = value;
			externalTemperature = externalFlowStream.GetPhaseByID(0).Temperature;
		}
	}

	public IExternalMedia ExternalMedia => externalMedia;

	public IHydraulicFlowStream InternalFlowStream
	{
		get
		{
			return internalFlowStream;
		}
		set
		{
			internalFlowStream = value;
			internalTemperature = internalFlowStream.GetPhaseByID(0).Temperature;
		}
	}

	public IRealValue InternalTemperature => internalTemperature;

	public IRealValue Emissivity
	{
		get
		{
			return mEmissivity;
		}
		set
		{
			mEmissivity = value;
		}
	}

	public bool IncludeRadiation
	{
		get
		{
			return mIncludeRadiation;
		}
		set
		{
			mIncludeRadiation = value;
		}
	}

	private event HTCHandler externalHT;

	public Insulation()
	{
		InitialiseData();
		LayerCollection layerCollection = this.layerCollection as LayerCollection;
		layerCollection.SetInsulationLayers(1);
		nLayers = 1;
		externalMedia.WhatExternalMedia = ExternalMediaType.Buried;
		HTCHandler calcEXT = HTFSExternalHTC;
		AssignExternalHT(calcEXT);
	}

	public Insulation(int Nlayers)
	{
		InitialiseData();
		LayerCollection layerCollection = this.layerCollection as LayerCollection;
		layerCollection.SetInsulationLayers(Nlayers);
		nLayers = Nlayers;
	}

	private void InitialiseData()
	{
		layerCollection = new LayerCollection();
		internalTemperature = new RealValue();
		externalMedia = new ExternalMedia(this);
		externalTemperature = externalMedia.OuterMedia.ExternalTemperature;
		internalDiameter = new RealValue();
		mEmissivity = new RealValue(0.0);
		mIncludeRadiation = false;
	}

	public void AssignExternalHT(HTCHandler calcEXT)
	{
		this.externalHT = calcEXT;
	}

	public void AddInner()
	{
		LayerCollection layerCollection = this.layerCollection as LayerCollection;
		layerCollection.insulationLayers.Insert(0, new InsulationLayer());
		IInsulationLayer insulationLayer = InsulationLayers[0];
		IInsulationLayer insulationLayer2 = InsulationLayers[1];
		insulationLayer.ExternalTemperature = insulationLayer2.InternalTemperature;
		nLayers = NLayers + 1;
	}

	public void AddInner(IInsulationLayer insulationLayer)
	{
		LayerCollection layerCollection = this.layerCollection as LayerCollection;
		layerCollection.insulationLayers.Insert(0, insulationLayer);
		IInsulationLayer insulationLayer2 = InsulationLayers[0];
		IInsulationLayer insulationLayer3 = InsulationLayers[1];
		insulationLayer2.ExternalTemperature = insulationLayer3.InternalTemperature;
		nLayers = NLayers + 1;
	}

	public void AddOuter()
	{
		LayerCollection layerCollection = this.layerCollection as LayerCollection;
		layerCollection.insulationLayers.Add(new InsulationLayer());
		IInsulationLayer insulationLayer = InsulationLayers[nLayers];
		IInsulationLayer insulationLayer2 = InsulationLayers[nLayers - 1];
		insulationLayer.InternalTemperature = insulationLayer2.ExternalTemperature;
		nLayers = NLayers + 1;
	}

	public void AddOuter(IInsulationLayer insulationLayer)
	{
		LayerCollection layerCollection = this.layerCollection as LayerCollection;
		layerCollection.insulationLayers.Add(insulationLayer);
		IInsulationLayer insulationLayer2 = InsulationLayers[nLayers];
		IInsulationLayer insulationLayer3 = InsulationLayers[nLayers - 1];
		insulationLayer2.InternalTemperature = insulationLayer3.ExternalTemperature;
		nLayers = NLayers + 1;
	}

	public void ClearAll()
	{
		int num = 0;
		LayerCollection layerCollection = this.layerCollection as LayerCollection;
		while (NLayers > 1)
		{
			if (layerCollection[num].MaterialType != "PipeWall")
			{
				layerCollection.insulationLayers.RemoveAt(num);
				num = 0;
				nLayers--;
			}
			else
			{
				num++;
			}
		}
	}

	public void Remove(int iLayer)
	{
		LayerCollection layerCollection = this.layerCollection as LayerCollection;
		if (layerCollection[iLayer].MaterialType != "PipeWall")
		{
			layerCollection.insulationLayers.RemoveAt(iLayer);
			nLayers--;
			return;
		}
		throw new Exception(Resources.CantRemovePipeWall);
	}

	public void RemoveInner()
	{
		int iLayer = 0;
		Remove(iLayer);
	}

	public void RemoveOuter()
	{
		int iLayer = NLayers - 1;
		Remove(iLayer);
	}

	public object Clone()
	{
		int i = 0;
		Insulation insulation;
		for (insulation = new Insulation(NLayers); i < insulation.NLayers; i++)
		{
			LayerCollection layerCollection = insulation.layerCollection as LayerCollection;
			layerCollection.insulationLayers[i] = InsulationLayers[i].Clone() as IInsulationLayer;
		}
		insulation.AssignExternalHT(this.externalHT);
		if (ExternalFlowStream != null)
		{
			insulation.ExternalFlowStream = ExternalFlowStream.Clone() as IHydraulicFlowStream;
		}
		if (ExternalMedia != null)
		{
			insulation.externalMedia = ExternalMedia.Clone() as ExternalMedia;
		}
		if (ExternalTemperature != null)
		{
			insulation.ExternalTemperature.Value = ExternalTemperature.Value;
		}
		if (InternalFlowStream != null)
		{
			insulation.InternalFlowStream = InternalFlowStream.Clone() as IHydraulicFlowStream;
		}
		insulation.InternalDiameter = new RealValue(InternalDiameter.Value);
		return insulation;
	}

	public static double HeatTransferCalcZero(Insulation ins, object Void)
	{
		return 10.0;
	}

	public static double HTFSExternalHTC(Insulation ins, object Void)
	{
		double result = 0.0;
		ExternalMediaType whatExternalMedia = ins.ExternalMedia.WhatExternalMedia;
		double num = ins.InternalDiameter.Value;
		for (int i = 0; i < ins.NLayers; i++)
		{
			num += ins.InsulationLayers[i].Thickness.Value;
		}
		switch (whatExternalMedia)
		{
		case ExternalMediaType.Air:
		case ExternalMediaType.Sea:
		{
			ExternalFluid externalFluid = ins.ExternalMedia.OuterMedia as ExternalFluid;
			double velocity = externalFluid.Velocity;
			double density = externalFluid.Density;
			double viscosity = externalFluid.Viscosity;
			double cp = externalFluid.Cp;
			double conductivity2 = externalFluid.Conductivity;
			Phase phase = new Phase();
			phase.Density.Value = density;
			phase.Viscosity.Value = viscosity;
			phase.Cp.Value = cp;
			phase.Conductivity.Value = conductivity2;
			result = Convection.HTCOfExternalForcedConvectionOnCylinder(phase, num, velocity);
			break;
		}
		case ExternalMediaType.Buried:
		{
			Buried buried = ins.ExternalMedia.OuterMedia as Buried;
			double num2 = buried.BurialDepth;
			if (num2 < num / 2.0 + 0.001)
			{
				num2 = num / 2.0 + 0.001;
			}
			double conductivity = buried.Conductivity;
			double num3 = Conduction.ThermalResistanceOfBuriedCylinder(conductivity, num, num2);
			result = 1.0 / num3 / (Math.PI * num);
			break;
		}
		}
		return result;
	}

	public void CheckMissingPhase()
	{
		IFlowStream flowStream = base.PortCollection.Item(1).FlowStream;
		if (flowStream == null)
		{
			flowStream = base.PortCollection.Item(0).FlowStream;
		}
		if (flowStream == null)
		{
			return;
		}
		IPhase phaseByID = flowStream.GetPhaseByID(0);
		if (phaseByID == null)
		{
			return;
		}
		for (int i = 0; i < phaseByID.NumberOfCompounds; i++)
		{
			IComponent component = phaseByID.GetComponent(i);
			int j;
			for (j = 0; j < mMissingVapourPhase.NumberOfCompounds && mMissingVapourPhase.GetComponent(j) != component; j++)
			{
			}
			if (j == mMissingVapourPhase.NumberOfCompounds)
			{
				mMissingVapourPhase.AddComponent(component);
				mMissingVapourPhase.ComponentMoleFractions[j].Value = 0.0;
			}
		}

	{
		double result = mDiameter.Value;
		if (mLiquidLevel.Value > 0.0 && mLiquidLevel.Value < mDiameter.Value)
		{
			result = Math.Sqrt(4.0 * VapourSpace(mDiameter.Value, mLiquidLevel.Value) / Math.PI);
		}
		return result;
	}

	public double VapourSpace(double tDiameter, double tLiqLevel)
	{
		double num = 0.5 * tDiameter;
		double num2 = ((!(tLiqLevel > num)) ? tLiqLevel : (tDiameter - tLiqLevel));
		double num3;
		if (num == 0.0 || num2 == 0.0)
		{
			num3 = 0.0;
		}
		else
		{
			double num4 = (num - num2) / num;
			num3 = Math.Atan((0.0 - num4) / Math.Sqrt(1.0 - num4 * num4)) + 2.0 * Math.Atan(1.0);
		}
		double num5 = num * num * num3 - (num - num2) * Math.Sqrt(2.0 * num * num2 - num2 * num2);
		if (tLiqLevel > num)
		{
			return num5;
		}
		return Math.PI * num * num - num5;
	}
