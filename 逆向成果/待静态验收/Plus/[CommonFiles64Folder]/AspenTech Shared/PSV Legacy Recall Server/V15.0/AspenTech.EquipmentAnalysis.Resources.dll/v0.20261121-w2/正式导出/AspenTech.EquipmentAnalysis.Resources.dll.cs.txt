public abstract class BaseQuantity : IQuantity, IQuantitativeValue
{
	protected readonly double _value;

	public readonly BaseUnitType UnitType;

	IUnitType IQuantitativeValue.UnitType => UnitType;

	protected virtual double MinValue => double.MinValue;

	protected virtual double MaxValue => double.MaxValue;

	public bool IsValid
	{
		get
		{
			if (_value >= MinValue)
			{
				return _value <= MaxValue;
			}
			return false;
		}
	}

	protected BaseQuantity(BaseUnitType unitType, double value)
	{
		if (unitType == null)
		{
			throw new ArgumentNullException("unitType", "Cannot create a quantity with a null unit-type");
		}
		UnitType = unitType;
		_value = value;
	}

	double IQuantity.ValueIn(IUnitOfMeasure uom)
	{
		return GenericValueIn(uom);
	}

	protected virtual double GenericValueIn(IUnitOfMeasure uom)
	{
		if (((BaseUnitType)uom.UnitType).Dimensions != UnitType.Dimensions)
		{
			throw new ArgumentException($"Cannot return value from {UnitType.Name} quantity in {uom.Symbol} units");
		}
		return _valueIn(uom);
	}

	protected double _valueIn(IUnitOfMeasure uom)
	{
		return uom.Factor * _value / ((IUnitType)UnitType).InnateUnit.Factor;
	}

	public static RuntimeQuantity operator *(BaseQuantity lhs, BaseQuantity rhs)
	{
		return RuntimeQuantity.CreateInInnateUnits(lhs._value * rhs._value, lhs.UnitType.Dimensions + rhs.UnitType.Dimensions);
	}

	public static RuntimeQuantity operator /(BaseQuantity lhs, BaseQuantity rhs)
	{
		return RuntimeQuantity.CreateInInnateUnits(lhs._value / rhs._value, lhs.UnitType.Dimensions - rhs.UnitType.Dimensions);
	}

	public RuntimeQuantity ToPower(int exponent)
	{
		return RuntimeQuantity.CreateInInnateUnits(Math.Pow(_value, exponent), UnitType.Dimensions * exponent);
	}

	public RuntimeQuantity ToRoot(int root)
	{
		return RuntimeQuantity.CreateInInnateUnits(Math.Pow(_value, 1.0 / (double)root), UnitType.Dimensions / root);
	}

	public string ToString(IUnitOfMeasure unit)
	{
		return $"{GenericValueIn(unit)} {unit.Symbol}";
	}

	public string ValueToString(IUnitOfMeasure unit, DoubleRepresentation doubleRepresentation, CultureInfo cultureInfo = null)
	{
		if (cultureInfo == null)
		{
			cultureInfo = CultureInfo.InvariantCulture;
		}
		double d = GenericValueIn(unit);
		return doubleRepresentation.FormatNumber(d, cultureInfo);
	}
}
public abstract class BaseUnitType : IUnitType
{
	public readonly DimensionsDefinition Dimensions;

	private readonly double _relativeErrorThreshold;

	private readonly double _zeroThreshold;

	public string Name { get; private set; }

	IDimensionsDefinition IUnitType.Dimensions => Dimensions;

	IUnitOfMeasure IUnitType.InnateUnit => GenericInnateUnit;

	internal abstract IUnitOfMeasure GenericInnateUnit { get; }

	public double RelativeErrorThreshold => _relativeErrorThreshold;

	public double ZeroThreshold => _zeroThreshold;

	IEnumerable<IUnitOfMeasure> IUnitType.UnitsOfMeasure => GenericUnitsOfMeasure;

	protected abstract IEnumerable<IUnitOfMeasure> GenericUnitsOfMeasure { get; }

	protected BaseUnitType(string name, DimensionsDefinition dimensions, double relativeErrorThreshold, double zeroThreshold)
	{
		Name = name;
		Dimensions = dimensions;
		_relativeErrorThreshold = relativeErrorThreshold;
		_zeroThreshold = zeroThreshold;
	}

	public override int GetHashCode()
	{
		return Dimensions.GetHashCode() ^ _relativeErrorThreshold.GetHashCode() ^ _zeroThreshold.GetHashCode();
	}

	public override string ToString()
	{
		return Name;
	}
}

public struct DimensionsDefinition : IDimensionsDefinition
{
	private readonly IReadOnlyDictionary<int, int> _exponents;

	private IReadOnlyDictionary<int, int> Exponents => _exponents ?? new Dictionary<int, int>();

	public int this[IQuantityDimension qd]
	{
		get
		{
			if (!Exponents.TryGetValue(qd.Value, out var value))
			{
				return 0;
			}
			return value;
		}
	}

	public IEnumerable<IQuantityDimension> NonZeroDimensions => Exponents.Select((KeyValuePair<int, int> pair) => (QuantityDimension)pair.Key).Cast<IQuantityDimension>();

	private static List<IQuantityDimension> OrderedList => QuantityDimension.GetValues().ToList();

	internal DimensionsDefinition(params Tuple<IQuantityDimension, int>[] dimensionDefinitions)
	{
		_exponents = dimensionDefinitions.Where((Tuple<IQuantityDimension, int> pair) => pair.Item2 != 0).ToDictionary((Tuple<IQuantityDimension, int> dimensionDefinition) => dimensionDefinition.Item1.Value, (Tuple<IQuantityDimension, int> dimensionDefinition) => dimensionDefinition.Item2);
	}

	private DimensionsDefinition(IReadOnlyDictionary<int, int> dict1, IReadOnlyDictionary<int, int> dict2)
	{
		Dictionary<int, int> dictionary = dict1.ToDictionary((KeyValuePair<int, int> pair) => pair.Key, (KeyValuePair<int, int> pair) => pair.Value);
		foreach (KeyValuePair<int, int> item in dict2)
		{
			if (dictionary.ContainsKey(item.Key))
			{
				dictionary[item.Key] += item.Value;
				if (dictionary[item.Key] == 0)
				{
					dictionary.Remove(item.Key);
				}
			}
			else
			{
				dictionary.Add(item.Key, item.Value);
			}
		}
		_exponents = dictionary;
	}

	private DimensionsDefinition(IReadOnlyDictionary<int, int> dict)
	{
		_exponents = dict.ToDictionary((KeyValuePair<int, int> pair) => pair.Key, (KeyValuePair<int, int> pair) => pair.Value);
	}

	private static IReadOnlyDictionary<int, int> Multiply(IReadOnlyDictionary<int, int> dict, int i)
	{
		if (i == 0)
		{
			return new Dictionary<int, int>();
		}
		return dict.ToDictionary((KeyValuePair<int, int> pair) => pair.Key, (KeyValuePair<int, int> pair) => pair.Value * i);
	}

	private static IReadOnlyDictionary<int, int> Divide(DimensionsDefinition dd, int divisor)
	{
		if (dd.Exponents.Values.Any((int i) => i % divisor != 0))
		{
			throw new ArgumentException($"Cannot divide {dd} by {divisor} because this would result in a non-integer exponent");
		}
		return dd.Exponents.ToDictionary((KeyValuePair<int, int> pair) => pair.Key, (KeyValuePair<int, int> pair) => pair.Value / divisor);
	}

	public static bool operator ==(DimensionsDefinition lhs, IDimensionsDefinition rhs)
	{
		if (rhs != null)
		{
			return lhs.NonZeroDimensions.Union(rhs.NonZeroDimensions).All((IQuantityDimension key) => lhs[key] == rhs[key]);
		}
		return false;
	}

	public static bool operator !=(DimensionsDefinition lhs, IDimensionsDefinition rhs)
	{
		return !(lhs == rhs);
	}

	public bool Equals(IDimensionsDefinition other)
	{
		return this == other;
	}

	public override bool Equals(object obj)
	{
		if (obj == null)
		{
			return false;
		}
		if (obj is IDimensionsDefinition)
		{
			return Equals((IDimensionsDefinition)obj);
		}
		return false;
	}

	public override int GetHashCode()
	{
		if (Exponents != null)
		{
			return Exponents.Aggregate(0, (int current, KeyValuePair<int, int> pair) => current ^ (pair.Key.GetHashCode() * 41 + pair.Value * 17));
		}
		return 0;
	}

	public override string ToString()
	{
		if (Exponents == null || Exponents.Count == 0)
		{
			return "[]";
		}
		List<string> list = new List<string>();
		foreach (IQuantityDimension nonZeroDimension in NonZeroDimensions)
		{
			string text = "[" + nonZeroDimension.Name + "]";
			if (Exponents[nonZeroDimension.Value] != 1)
			{
				text = text + "^" + Exponents[nonZeroDimension.Value];
			}
			list.Add(text);
		}
		return string.Join(" ", list);
	}

	public static DimensionsDefinition operator +(DimensionsDefinition lhs, DimensionsDefinition rhs)
	{
		return new DimensionsDefinition(lhs.Exponents, rhs.Exponents);
	}

	public static DimensionsDefinition operator -(DimensionsDefinition dd)
	{
		return new DimensionsDefinition(Multiply(dd.Exponents, -1));
	}

	public static DimensionsDefinition operator -(DimensionsDefinition lhs, DimensionsDefinition rhs)
	{
		return lhs + -rhs;
	}

	public static DimensionsDefinition operator *(DimensionsDefinition lhs, int rhs)
	{
		return new DimensionsDefinition(Multiply(lhs.Exponents, rhs));
	}

	public static DimensionsDefinition operator *(int lhs, DimensionsDefinition rhs)
	{
		return rhs * lhs;
	}

	public static DimensionsDefinition operator /(DimensionsDefinition lhs, int rhs)
	{
		return new DimensionsDefinition(Divide(lhs, rhs));
	}
}

public enum NumberFormat
{
	SigFigs,
	Decimals,
	Scientific
}
public delegate double RoundingMethod(double d);
public struct DoubleRepresentation
{
	private readonly NumberFormat? _numberFormat;

	private readonly int? _numNumbers;

	private readonly int? _numWholeNumbers;

	public readonly bool ThousandsSeparator;

	public readonly bool ShowLeadingZeros;

	public readonly bool ShowTrailingZeros;

	public readonly RoundingMethod RoundingMethod;

	public NumberFormat NumberFormat => _numberFormat ?? NumberFormat.SigFigs;

	public int NumNumbers => _numNumbers ?? 2;

	public int NumWholeNumbers => _numWholeNumbers ?? 1;

	public DoubleRepresentation(NumberFormat numberFormat = NumberFormat.SigFigs, int numNumbers = 2, int numWholeNumbers = 1, bool thousandsSeparator = false, bool showLeadingZeros = false, bool showTrailingZeros = false, RoundingMethod roundingMethod = null)
	{
		_numberFormat = numberFormat;
		_numNumbers = numNumbers;
		_numWholeNumbers = numWholeNumbers;
		ThousandsSeparator = thousandsSeparator;
		ShowLeadingZeros = showLeadingZeros;
		ShowTrailingZeros = showTrailingZeros;
		RoundingMethod = roundingMethod;
	}

	public string FormatString(double value)
	{
		if (NumberFormat == NumberFormat.SigFigs)
		{
			int num = (int)Math.Floor(Math.Log10(Math.Abs(value)));
			bool flag = num < -1 || num >= NumNumbers;
			string text = (ThousandsSeparator ? ",0." : "0.");
			string text2 = (ShowTrailingZeros ? "0" : "#");
			int num2 = Math.Max(1, NumNumbers);
			if (!flag)
			{
				num2 -= num;
			}
			for (int i = 1; i < num2; i++)
			{
				text += text2;
			}
			if (flag)
			{
				text += (ShowLeadingZeros ? "e-000" : "e-0");
			}
			return text;
		}
		if (NumberFormat == NumberFormat.Decimals)
		{
			string text3 = "";
			if (ShowLeadingZeros)
			{
				string text4 = (ThousandsSeparator ? "," : "");
				for (int j = 0; j < Math.Max(1, NumWholeNumbers); j++)
				{
					text3 = "0" + text3;
					if (j % 3 == 2 && j < NumWholeNumbers - 1)
					{
						text3 = text4 + text3;
					}
				}
			}
			else
			{
				text3 = (ThousandsSeparator ? ",0" : "0");
			}
			text3 += ".";
			string text5 = (ShowTrailingZeros ? "0" : "#");
			for (int k = 0; k < Math.Max(0, NumNumbers); k++)
			{
				text3 += text5;
			}
			return text3;
		}
		if (NumberFormat == NumberFormat.Scientific)
		{
			string text6 = (ThousandsSeparator ? ",0." : "0.");
			string text7 = (ShowTrailingZeros ? "0" : "#");
			for (int l = 1; l < Math.Max(1, NumNumbers); l++)
			{
				text6 += text7;
			}
			return text6 + (ShowLeadingZeros ? "e-000" : "e-0");
		}
		throw new NotImplementedException("Need to write code for NumberFormat." + NumberFormat);
	}

	public override string ToString()
	{
		string text = $"{NumNumbers} {NumberFormat.ToString()} with";
		if (ShowLeadingZeros && NumberFormat == NumberFormat.Decimals)
		{
			text += $" {NumWholeNumbers} leading digits and";
		}
		text += string.Format("{0} thousands separator", ThousandsSeparator ? "" : " no");
		text += string.Format(" and {0} trailing zeros", ShowTrailingZeros ? "displayed" : "hidden");
		return text + string.Format(" and {0} leading zeros", ShowLeadingZeros ? "displayed" : "hidden");
	}

	public string FormatNumber(double d, IFormatProvider formatProvider)
	{
		if (RoundingMethod != null)
		{
			d = RoundingMethod(d);
		}
		return d.ToString(FormatString(d), formatProvider);
	}
}

public abstract class FloatingQuantity<T, TU> : BaseQuantity, IComparable<T> where T : FloatingQuantity<T, TU> where TU : UnitOfMeasure
{
	protected FloatingQuantity(BaseUnitType unitType)
		: base(unitType, double.NaN)
	{
	}

	protected FloatingQuantity(BaseUnitType unitType, double value, TU unit)
		: base(unitType, ConvertOnConstruction(unitType, value, unit))
	{
		if (unit == null)
		{
			throw new ArgumentNullException("Cannot create a quantity with a null unit of measure");
		}
	}

	private static double ConvertOnConstruction(BaseUnitType unitType, double value, TU unit)
	{
		if (unit == unitType.GenericInnateUnit)
		{
			return value;
		}
		if (unit.UnitType.Dimensions != unitType.Dimensions)
		{
			throw new ArgumentException($"Cannot assign {unitType.Name} quantity using {unit.Symbol} units because the dimensions do not match");
		}
		return unitType.GenericInnateUnit.Factor * value / unit.Factor;
	}

	protected FloatingQuantity(BaseUnitType unitType, IQuantity quantity)
		: base(unitType, quantity.ValueIn(quantity.UnitType.InnateUnit))
	{
		if (quantity == null)
		{
			throw new ArgumentNullException("Cannot create a quantity from a null quantity");
		}
		if (((BaseQuantity)quantity).UnitType.Dimensions != unitType.Dimensions)
		{
			throw new ArgumentException($"Cannot create {unitType.Name} Quantity from {quantity.UnitType.Name} Quantity because the dimensions do not match");
		}
	}

	protected abstract T CreateInInnateUnit(double value);

	public virtual double ValueIn(TU uom)
	{
		return _valueIn(uom);
	}

	public override string ToString()
	{
		return $"{_value} {UnitType.GenericInnateUnit.Symbol}";
	}

	protected virtual void CheckDimensions(T that)
	{
	}

	public bool DiffersFrom(T that)
	{
		if (UnitType != that.UnitType)
		{
			return true;
		}
		return MyMath.ValuesDiffer(_value, that._value, UnitType.RelativeErrorThreshold, UnitType.ZeroThreshold);
	}

	public static bool operator ==(FloatingQuantity<T, TU> lhs, T rhs)
	{
		if ((object)lhs == null && (object)rhs == null)
		{
			return true;
		}
		if ((object)lhs == null || (object)rhs == null)
		{
			return false;
		}
		return !lhs.DiffersFrom(rhs);
	}

	public static bool operator !=(FloatingQuantity<T, TU> lhs, T rhs)
	{
		return !(lhs == rhs);
	}

	protected bool Equals(T other)
	{
		if (other != null && object.Equals(UnitType, other.UnitType))
		{
			return !DiffersFrom(other);
		}
		return false;
	}

	public override bool Equals(object obj)
	{
		if (obj == null)
		{
			return false;
		}
		if (this == obj)
		{
			return true;
		}
		return Equals(obj as T);
	}

	public override int GetHashCode()
	{
		return (_value.GetHashCode() * 397) ^ ((UnitType != null) ? UnitType.GetHashCode() : 0);
	}

	public static bool operator >(FloatingQuantity<T, TU> lhs, T rhs)
	{
		lhs.CheckDimensions(rhs);
		if (lhs._value > rhs._value)
		{
			return lhs != rhs;
		}
		return false;
	}

	public static bool operator <(FloatingQuantity<T, TU> lhs, T rhs)
	{
		lhs.CheckDimensions(rhs);
		if (lhs._value < rhs._value)
		{
			return lhs != rhs;
		}
		return false;
	}

	public static bool operator >=(FloatingQuantity<T, TU> lhs, T rhs)
	{
		return !(lhs < rhs);
	}

	public static bool operator <=(FloatingQuantity<T, TU> lhs, T rhs)
	{
		return !(lhs > rhs);
	}

	public static T operator *(FloatingQuantity<T, TU> lhs, Scalar rhs)
	{
		return lhs.CreateInInnateUnit(lhs._value * rhs._value);
	}

	public static T operator *(Scalar lhs, FloatingQuantity<T, TU> rhs)
	{
		return rhs * lhs;
	}

	public static T operator /(FloatingQuantity<T, TU> lhs, Scalar rhs)
	{
		return lhs.CreateInInnateUnit(lhs._value / rhs._value);
	}

	public int CompareTo(T other)
	{
		return _value.CompareTo(other._value);
	}
}

public static class MyMath
{
	public static MolarStandardVolume StandardConversionFactor => new MolarStandardVolume(379.49, Units.SCF_per_lbmol);

	public static MolarEntropy UniversalGasConstant => new MolarEntropy(8.3144598, Units.J_per_mol_K);

	public static bool ValuesDiffer(double num1, double num2, double relativeErrorThreshold, double zeroThreshold)
	{
		if (num1 == num2)
		{
			return false;
		}
		if (double.IsNaN(num1) && double.IsNaN(num2))
		{
			return false;
		}
		if (double.IsNaN(num1) || double.IsNaN(num2))
		{
			return true;
		}
		if (double.IsInfinity(num1) && double.IsInfinity(num2))
		{
			return double.IsPositiveInfinity(num1) ^ double.IsPositiveInfinity(num2);
		}
		if (double.IsInfinity(num1) || double.IsInfinity(num2))
		{
			return true;
		}
		if (Math.Abs(num1) < zeroThreshold && Math.Abs(num2) < zeroThreshold)
		{
			return false;
		}
		double num3 = Math.Abs(num1 - num2);
		double num4 = Math.Abs(num1) * relativeErrorThreshold;
		double num5 = Math.Abs(num2) * relativeErrorThreshold;
		if (num3 > num4 && num3 > num5)
		{
			return true;
		}
		if (num3 < num4 && num3 < num5)
		{
			return false;
		}
		return 1.0 > relativeErrorThreshold * num1 / num3 * (relativeErrorThreshold * num2 / num3);
	}

	public static bool ValueExceeds(double num1, double num2, double relativeEqualityThreshold, double zeroThreshold)
	{
		if (num1 < num2)
		{
			return false;
		}
		if (double.IsNaN(num1) || double.IsNaN(num2))
		{
			return false;
		}
		if (double.IsPositiveInfinity(num1) && double.IsNegativeInfinity(num2))
		{
			return true;
		}
		return ValuesDiffer(num1, num2, relativeEqualityThreshold, zeroThreshold);
	}

	public static int Gcd(int a, int b)
	{
		a = Math.Abs(a);
		b = Math.Abs(b);
		while (b != 0)
		{
			int num = a;
			a = b;
			b = num % b;
		}
		return a;
	}

	public static int RoundedInt(double value, int resolution, MidpointRounding midpointRounding = MidpointRounding.AwayFromZero)
	{
		return (int)Math.Round(value / (double)resolution, midpointRounding) * resolution;
	}

	public static RationalNumber RoundedRationalNumber(double value, int inverseResolution, MidpointRounding midpointRounding = MidpointRounding.AwayFromZero)
	{
		return RationalNumber.Fraction((int)Math.Round(value * (double)inverseResolution, midpointRounding), inverseResolution);
	}

public class PipeDiameterDefinitionManager
{
	private readonly Dictionary<PipeDiameterEnglishKey, PipeDiameterDefinition> _english = new Dictionary<PipeDiameterEnglishKey, PipeDiameterDefinition>();

	private readonly Dictionary<PipeDiameterSiKey, PipeDiameterDefinition> _si = new Dictionary<PipeDiameterSiKey, PipeDiameterDefinition>();

	public IEnumerable<PipeDiameterDefinition> All => _english.Values.Union(_si.Values);

	public IEnumerable<PipeDiameterDefinition> English => _english.Values;

	public IEnumerable<PipeDiameterDefinition> SI => _si.Values;

	public bool AddNew(PipeSchedule schedule, IQuantity innerDiameter, IQuantity wallThickness, double? nominalDiameter_inch, double? nominalDiameter_mm)
	{
		return AddNew(schedule, new Distance(innerDiameter), new Distance(wallThickness), nominalDiameter_inch, nominalDiameter_mm);
	}

	public bool AddNew(PipeSchedule schedule, Distance innerDiameter, Distance wallThickness, RationalNumber? nominalDiameter_inch, int? nominalDiameter_mm)
	{
		PipeDiameterDefinition pipeDiameterDefinition = new PipeDiameterDefinition(innerDiameter, wallThickness, nominalDiameter_inch, nominalDiameter_mm, schedule);
		if (nominalDiameter_inch.HasValue)
		{
			_english.Add(new PipeDiameterEnglishKey(pipeDiameterDefinition), pipeDiameterDefinition);
		}
		if (nominalDiameter_mm.HasValue)
		{
			_si.Add(new PipeDiameterSiKey(pipeDiameterDefinition), pipeDiameterDefinition);
		}
		if (nominalDiameter_inch.HasValue)
		{
			return nominalDiameter_mm.HasValue;
		}
		return false;
	}

	public bool Remove(PipeDiameterDefinition item)
	{
		bool flag = false;
		if (item.NominalDiameter_inch.HasValue)
		{
			flag = _english.Remove(new PipeDiameterEnglishKey(item));
		}
		if (item.NominalDiameter_mm.HasValue)
		{
			flag = _si.Remove(new PipeDiameterSiKey(item)) || flag;
		}
		return flag;
	}
}

	{
		get
		{
			if (_pressureDifference.IsValid)

				return ValueIn(UnitTypes.Pressure.InnateUnit, PressureReference.Absolute) > 0.0;
			}

		}

	public static void UpdateAmbientPressure(Pressure ambientPressure, Session session = null)
	{
		if (ambientPressure == null)
		{
			throw new ArgumentNullException("ambientPressure", "There is no good reason to set the singleton ambient pressure to null");
		}
		session = session ?? Session.Instance;
		session.Ambient = ambientPressure;
		session.AmbientPressure = new PressureDifference(ambientPressure._pressureDifference);
	}

	public Pressure()
	{
		Reference = PressureReference.Absolute;
		_pressureDifference = new PressureDifference();
	}

	public Pressure(PressureDifference value, PressureReference reference)
	{
		if (value == null)
		{
			throw new ArgumentNullException("Cannot create Pressure from null PressureReference");
		}
		_pressureDifference = value;
		Reference = reference;
	}

	public Pressure(double value, PressureUnit unit, PressureReference reference)
	{
		_pressureDifference = new PressureDifference(value, unit);
		Reference = reference;
	}

	{
		return new Pressure(value, PressureReference.Gauge);
	}

	}

	{
		return new PressureDifference(ValueIn(UnitTypes.Pressure.InnateUnit, pressureReference, session), UnitTypes.Pressure.InnateUnit);
	}

	public double ValueIn(PressureUnit uom, PressureReference pressureReference, Session session = null)
	{
		if (Reference == pressureReference)
		{
			return _pressureDifference.ValueIn(uom);
		}
		return (_pressureDifference + _referencePressure(Reference, session) - _referencePressure(pressureReference, session)).ValueIn(uom);
	}

	private PressureDifference _referencePressure(PressureReference pressureReference, Session session)
	{
		session = session ?? Session.Instance;
		return pressureReference switch
		{
			PressureReference.Absolute => new PressureDifference(0.0, UnitTypes.Pressure.InnateUnit), 
			PressureReference.Gauge => session.AmbientPressure, 
			_ => throw new ArgumentException($"I don't know how to handle this value of PressureReference, {pressureReference.ToString()}"), 
		};
	}

public abstract class Quantity<T, TU> : FloatingQuantity<T, TU> where T : Quantity<T, TU> where TU : UnitOfMeasure
{
	protected Quantity(BaseUnitType unitType)
		: base(unitType)
	{
	}

	protected Quantity(BaseUnitType unitType, double value, TU unit)
		: base(unitType, value, unit)
	{
	}

	protected Quantity(BaseUnitType unitType, IQuantity quantity)
		: base(unitType, quantity)
	{
	}

	public static T operator -(Quantity<T, TU> q)
	{
		return q.CreateInInnateUnit(0.0 - q._value);
	}

	{

		{
			if (!_value.HasValue)
			{

		_value = value;
	}

	internal static QuantityDimension Create(string name, int value)
	{
		if (value == int.MinValue)
		{
			throw new ArgumentException("int.MinValue is a reserved value, indicating a missing value");
		}
		if (string.IsNullOrWhiteSpace(name))
		{
			throw new ArgumentException("Quantity dimension name must not be null, empty, nor only whitespace");
		}

		{
			throw new ArgumentException($"Cannot add QuantityDimensions with same name.  \"{value}\" is already registered");
		}

		ByValue.Add(value, quantityDimension);

public struct RationalNumber : IComparable<RationalNumber>
{
	public enum DisplayStyle
	{
		Simple,
		FancyUnicode,
		FancySuperscript
	}

	public readonly int Numerator;

	private readonly int _denominator;

	private const char unicodeFractionChar = '⁄';

	public int Denominator
	{
		get
		{
			if (_denominator != 0)
			{
				return _denominator;
			}
			return 1;
		}
	}

	public int Quotient => Numerator / Denominator;

	public int Remainder => Numerator % Denominator;

	private RationalNumber(int numerator, int denominator)
	{
		if (denominator == 0)
		{
			throw new DivideByZeroException("Cannot create RationalNumber with 0 as denominator");
		}
		if (denominator < 0)
		{
			numerator = -numerator;
			denominator = -denominator;
		}
		int num = MyMath.Gcd(numerator, denominator);
		Numerator = numerator / num;
		_denominator = denominator / num;
	}

	public static RationalNumber Fraction(int numerator, int denominator)
	{
		return new RationalNumber(numerator, denominator);
	}

	public static RationalNumber MixedNumber(int quotient, uint modulus, int denominator)
	{
		if (modulus > int.MaxValue - Math.Abs(quotient * denominator))
		{
			throw new OverflowException("resulting numerator is larger than int.MaxValue");
		}
		int num = (int)modulus;
		if ((quotient < 0) ^ (denominator < 0))
		{
			num = -num;
		}
		return new RationalNumber(quotient * denominator + num, Math.Abs(denominator));
	}

	public override string ToString()
	{
		if (Denominator == 1)
		{
			return Numerator.ToString();
		}
		return Numerator + "/" + Denominator;
	}

	public string ToString(DisplayStyle displayStyle)
	{
		if (Denominator == 1)
		{
			return Numerator.ToString();
		}
		return displayStyle switch
		{
			DisplayStyle.FancySuperscript => ToSuperscript(Numerator) + "/" + ToSubscript(Denominator), 
			DisplayStyle.FancyUnicode => Numerator + "⁄" + Denominator, 
			_ => ToString(), 
		};
	}

	public string ToMixedNumberString(DisplayStyle displayStyle = DisplayStyle.Simple)
	{
		if (Quotient == 0 || Denominator == 1)
		{
			return ToString(displayStyle);
		}
		return displayStyle switch
		{
			DisplayStyle.FancySuperscript => $"{Quotient} {ToSuperscript(Math.Abs(Remainder))}/{ToSubscript(Denominator)}", 
			DisplayStyle.FancyUnicode => $"{Quotient} {Math.Abs(Remainder)}⁄{Denominator}", 
			_ => $"{Quotient} {Math.Abs(Remainder)}/{Denominator}", 
		};
	}

	private static string ToSuperscript(int i)
	{
		return i.ToString().Replace("0", "⁰").Replace("1", "¹")
			.Replace("2", "²")
			.Replace("3", "³")
			.Replace("4", "⁴")
			.Replace("5", "⁵")
			.Replace("6", "⁶")
			.Replace("7", "⁷")
			.Replace("8", "⁸")
			.Replace("9", "⁹");
	}

	private static string ToSubscript(int i)
	{
		return i.ToString().Replace("0", "₀").Replace("1", "₁")
			.Replace("2", "₂")
			.Replace("3", "₃")
			.Replace("4", "₄")
			.Replace("5", "₅")
			.Replace("6", "₆")
			.Replace("7", "₇")
			.Replace("8", "₈")
			.Replace("9", "₉");
	}

	public static explicit operator double(RationalNumber n)
	{
		return (double)n.Numerator / (double)n.Denominator;
	}

	public static implicit operator RationalNumber(int i)
	{
		return new RationalNumber(i, 1);
	}

	public static RationalNumber operator +(RationalNumber lhs, RationalNumber rhs)
	{
		int numerator = lhs.Numerator * rhs.Denominator + rhs.Numerator * lhs.Denominator;
		int denominator = lhs.Denominator * rhs.Denominator;
		return new RationalNumber(numerator, denominator);
	}

	public static RationalNumber operator -(RationalNumber n)
	{
		return new RationalNumber(-n.Numerator, n.Denominator);
	}

	public static RationalNumber operator -(RationalNumber lhs, RationalNumber rhs)
	{
		return lhs + -rhs;
	}

	public static RationalNumber operator *(RationalNumber lhs, RationalNumber rhs)
	{
		return new RationalNumber(lhs.Numerator * rhs.Numerator, lhs.Denominator * rhs.Denominator);
	}

	public static RationalNumber operator /(RationalNumber lhs, RationalNumber rhs)
	{
		return new RationalNumber(lhs.Numerator * rhs.Denominator, lhs.Denominator * rhs.Numerator);
	}

	public static RationalNumber operator %(RationalNumber lhs, RationalNumber rhs)
	{
		return Fraction(lhs.Numerator * rhs.Denominator % (rhs.Numerator * lhs.Denominator), lhs.Denominator * rhs.Denominator);
	}

	public static bool operator ==(RationalNumber lhs, RationalNumber rhs)
	{
		if (lhs.Numerator == rhs.Numerator)
		{
			return lhs.Denominator == rhs.Denominator;
		}
		return false;
	}

	public static bool operator !=(RationalNumber lhs, RationalNumber rhs)
	{
		return !(lhs == rhs);
	}

	public bool Equals(RationalNumber that)
	{
		return this == that;
	}

	public override bool Equals(object obj)
	{
		if (obj is RationalNumber || obj is int)
		{
			return Equals((RationalNumber)obj);
		}
		if (obj is double || obj is float)
		{
			return (double)obj == (double)this;
		}
		return false;
	}

	public override int GetHashCode()
	{
		return 23 + 31 * (Numerator + 23 + 31 * Denominator);
	}

	public static bool operator <(RationalNumber lhs, RationalNumber rhs)
	{
		if (lhs.Quotient == rhs.Quotient)
		{
			return lhs.Remainder * rhs.Denominator < rhs.Remainder * lhs.Denominator;
		}
		return lhs.Quotient < rhs.Quotient;
	}

	public static bool operator >(RationalNumber lhs, RationalNumber rhs)
	{
		if (lhs.Quotient == rhs.Quotient)
		{
			return lhs.Remainder * rhs.Denominator > rhs.Remainder * lhs.Denominator;
		}
		return lhs.Quotient > rhs.Quotient;
	}

	public static bool operator <=(RationalNumber lhs, RationalNumber rhs)
	{
		return !(lhs > rhs);
	}

	public static bool operator >=(RationalNumber lhs, RationalNumber rhs)
	{
		return !(lhs < rhs);
	}

	public int CompareTo(RationalNumber other)
	{
		int num = Quotient.CompareTo(other.Quotient);
		if (num != 0)
		{
			return num;
		}
		return Remainder.CompareTo(other.Remainder);
	}
}

		//IL_0039: Invalid comparison between Unknown and I4

			if ((int)((Utf8JsonReader)(ref reader)).TokenType == 2 && ((Utf8JsonReader)(ref reader)).CurrentDepth == currentDepth)
			{

			}
			if ((int)((Utf8JsonReader)(ref reader)).TokenType != 5)
			{
				continue;
			}
			string text = ((Utf8JsonReader)(ref reader)).GetString();
			((Utf8JsonReader)(ref reader)).Read();
			switch (text)
			{
			case "PressureConverter":
				pressureConverter = JsonSerializer.Deserialize<PressureConverter>(ref reader, (JsonSerializerOptions)null);

			{

				session.RuntimeUnitsOfMeasure = source.ToDictionary((RuntimeUnitOfMeasureEntity e) => e.Symbol);
				break;
			}
			case "ByDimensions":
				foreach (IEnumerable<DimensionEntity> item in JsonSerializer.Deserialize<IEnumerable<IEnumerable<DimensionEntity>>>(ref reader, (JsonSerializerOptions)null))
				{
					List<Tuple<IQuantityDimension, int>> list = new List<Tuple<IQuantityDimension, int>>();
					foreach (DimensionEntity dimension in item)
					{
						list.Add(new Tuple<IQuantityDimension, int>(QuantityDimension.GetValues().First((IQuantityDimension qd) => qd.Name == dimension.Name), dimension.Exponent));
					}
					RuntimeUnitType.GetOrCreate(new DimensionsDefinition(list.ToArray()), session);
				}
				break;
			}
		}

	{

			writer.WriteNumber("Reference", (int)value.Ambient.Reference);
			writer.WriteEndObject();
		}

					writer.WriteNumber("Exponent", byDimension.Key[nonZeroDimension]);
					writer.WriteEndObject();
				}
				writer.WriteEndArray();
			}
			writer.WriteEndArray();
		}
		writer.WriteEndObject();

	{
		if (q.UnitType.Dimensions != UnitTypes.Temperature.Dimensions)
		{
			throw new ArgumentException($"Cannot specify Temperature from quantity with dimensions of {q.UnitType.Dimensions}");
		}
		return new Temperature(q.ValueIn(UnitTypes.Temperature.InnateUnit), UnitTypes.Temperature.InnateUnit);
	}

	{
		return new Temperature(value, UnitTypes.Temperature.InnateUnit);
	}

	{
		return Offset(uom) + uom.Factor * (_value - Offset(UnitTypes.Temperature.InnateUnit)) / UnitTypes.Temperature.InnateUnit.Factor;
	}

	protected static double Offset(TemperatureUnit unit, Session session = null)
	{
		session = session ?? Session.Instance;
		if (unit == Units.K || unit == Units.R)
		{
			return 0.0;
		}
		if (unit == Units.C)
		{
			return -273.15;
		}
		if (unit == Units.F)
		{
			return -459.67;
		}
		if (session.RuntimeUnitsOfMeasure.TryGetValue(unit.Symbol, out var value))
		{
			return value.Offset;
		}
		throw new ArgumentException("Cannot find offset value for Temperature unit of " + unit.Symbol);
	}

	protected override double GenericValueIn(IUnitOfMeasure uom)
	{
		if (!(uom is TemperatureUnit))
		{
			throw new ArgumentException($"Cannot return value from {UnitType.Name} quantity in {uom.Symbol} units");
		}
		return ValueIn((TemperatureUnit)uom);
	}

	public static Temperature operator +(Temperature lhs, TemperatureDifference rhs)
	{
		TemperatureUnit innateUnit = UnitTypes.Temperature.InnateUnit;
		return new Temperature(lhs._value + rhs.ValueIn(innateUnit), innateUnit);
	}

	{
		return rhs + lhs;
	}

	public static TemperatureDifference operator -(Temperature lhs, Temperature rhs)
	{
		return new TemperatureDifference(lhs._value - rhs._value, UnitTypes.Temperature.InnateUnit);
	}

	{
		return lhs + -rhs;
	}

public abstract class UnitOfMeasure : IUnitOfMeasure
{
	public readonly DimensionsDefinition Dimensions;

	public readonly string Symbol;

	public readonly double Factor;

	public readonly string Name;

	IUnitType IUnitOfMeasure.UnitType => UnitType;

	IDimensionsDefinition IUnitOfMeasure.Dimensions => Dimensions;

	string IUnitOfMeasure.Symbol => Symbol;

	double IUnitOfMeasure.Factor => Factor;

	string IUnitOfMeasure.Name => Name;

	public virtual BaseUnitType UnitType
	{
		get
		{
			if (!UnitTypes.ByDimensions.TryGetValue(Dimensions, out var value))
			{
				return null;
			}
			return value;
		}
	}

	protected UnitOfMeasure(DimensionsDefinition dimensions, string symbol, string name, double factor, bool exposed, IDictionary<IUnitOfMeasure, int> constituents = null)
	{
		Dimensions = dimensions;
		Symbol = symbol;
		Name = name;
		if (constituents == null)
		{
			Factor = factor;
			return;
		}
		foreach (KeyValuePair<IUnitOfMeasure, int> constituent in constituents)
		{
			factor *= Math.Pow(constituent.Key.Factor, constituent.Value);
		}
		Factor = factor;
	}

	public override int GetHashCode()
	{
		return Dimensions.GetHashCode() ^ Symbol.GetHashCode() ^ Factor.GetHashCode();
	}

	public override string ToString()
	{
		return $"{UnitType} unit, {Symbol} with factor {Factor}";
	}
}

public class UnitSet : UnitTypeDictionary<UnitOfMeasure>
{
	public T UnitOfMeasure<T>(UnitType<T> unitType, int? enumValue = null) where T : UnitOfMeasure
	{
		return (T)base[unitType, enumValue];
	}

	protected override UnitOfMeasure DefaultValue(UnitTypeKey unitTypeKey)
	{
		return (UnitOfMeasure)unitTypeKey.UnitType.GenericInnateUnit;
	}

	protected override void SetValue(UnitTypeKey key, UnitOfMeasure value)
	{
		if (value.UnitType != key.UnitType)
		{
			throw new ArgumentException($"Cannot assign {value.Symbol} as display unit for {key.UnitType} quantities");
		}
		base.SetValue(key, value);
	}
}

public class UnitType<T> : BaseUnitType where T : UnitOfMeasure
{
	public readonly T InnateUnit;

	protected readonly List<T> _unitsOfMeasure;

	internal override IUnitOfMeasure GenericInnateUnit => InnateUnit;

	public virtual IEnumerable<T> UnitsOfMeasure => _unitsOfMeasure;

	protected override IEnumerable<IUnitOfMeasure> GenericUnitsOfMeasure => ((IEnumerable<IUnitOfMeasure>)UnitsOfMeasure).ToList();

	internal UnitType(string name, DimensionsDefinition dimensions, T innateUnit, IEnumerable<T> units = null, double relativeErrorThreshold = 1E-12, double zeroThreshold = 1E-12)
		: base(name, dimensions, relativeErrorThreshold, zeroThreshold)
	{
		InnateUnit = innateUnit;
		_unitsOfMeasure = new List<T>();
		_unitsOfMeasure.Add(InnateUnit);
		if (units != null)
		{
			_unitsOfMeasure.AddRange(units);
		}
	}
}
public class UnitTypeConverter : JsonConverter<IUnitType>
{
	public override IUnitType Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
	{
		UnitTypeEntity unitTypeEntity = JsonSerializer.Deserialize<UnitTypeEntity>(ref reader, (JsonSerializerOptions)null);
		UnitTypes.ByName.TryGetValue(unitTypeEntity.Name, out var value);
		return value;
	}

	public override void Write(Utf8JsonWriter writer, IUnitType value, JsonSerializerOptions options)
	{
		UnitTypeEntity unitTypeEntity = new UnitTypeEntity();
		unitTypeEntity.Name = value.Name;
		unitTypeEntity.ZeroThreshold = value.ZeroThreshold;
		unitTypeEntity.RelativeErrorThreshold = value.RelativeErrorThreshold;
		UnitOfMeasureEntity unitOfMeasureEntity = new UnitOfMeasureEntity();
		unitOfMeasureEntity.Factor = value.InnateUnit.Factor;
		unitOfMeasureEntity.Symbol = value.InnateUnit.Symbol;
		unitOfMeasureEntity.Name = value.InnateUnit.Name;
		unitTypeEntity.InnateUnit = unitOfMeasureEntity;
		foreach (IUnitOfMeasure item3 in value.UnitsOfMeasure)
		{
			UnitOfMeasureEntity unitOfMeasureEntity2 = new UnitOfMeasureEntity
			{
				Factor = item3.Factor,
				Symbol = item3.Symbol,
				Name = item3.Name
			};
			foreach (IQuantityDimension nonZeroDimension in item3.Dimensions.NonZeroDimensions)
			{
				DimensionEntity item = new DimensionEntity
				{
					Name = nonZeroDimension.Name,
					Exponent = value.Dimensions[nonZeroDimension]
				};
				unitOfMeasureEntity2.Dimensions.Add(item);
			}
			unitTypeEntity.UnitOfMeasures.Add(unitOfMeasureEntity2);
		}
		foreach (IQuantityDimension nonZeroDimension2 in value.Dimensions.NonZeroDimensions)
		{
			DimensionEntity item2 = new DimensionEntity
			{
				Name = nonZeroDimension2.Name,
				Exponent = value.Dimensions[nonZeroDimension2]
			};
			unitTypeEntity.Dimensions.Add(item2);
		}
		JsonSerializer.Serialize<UnitTypeEntity>(writer, unitTypeEntity, (JsonSerializerOptions)null);
	}
}

public abstract class UnitTypeDictionary<T>
{
	protected struct UnitTypeKey
	{
		internal readonly DimensionsDefinition Dimensions;

		internal readonly int? EnumValue;

		internal readonly BaseUnitType UnitType;

		internal UnitTypeKey(BaseUnitType unitType, int? enumValue)
		{
			UnitType = unitType;
			Dimensions = unitType.Dimensions;
			EnumValue = enumValue;
		}

		public override int GetHashCode()
		{
			return Dimensions.GetHashCode() ^ (EnumValue.HasValue ? (EnumValue.Value * 23) : 347);
		}
	}

	protected readonly Dictionary<UnitTypeKey, T> Dictionary = new Dictionary<UnitTypeKey, T>();

	public string Name { get; set; }

	public T this[BaseUnitType unitType, int? enumValue = null]
	{
		get
		{
			return GetValue(new UnitTypeKey(unitType, enumValue));
		}
		set
		{
			SetValue(new UnitTypeKey(unitType, enumValue), value);
		}
	}

	protected UnitTypeDictionary()
	{
	}

	protected UnitTypeDictionary(UnitTypeDictionary<T> unitTypeDictionary)
	{
		Dictionary = new Dictionary<UnitTypeKey, T>(unitTypeDictionary.Dictionary);
	}

	protected abstract T DefaultValue(UnitTypeKey unitTypeKey);

	private T GetValue(UnitTypeKey unitTypeKey)
	{
		if (Dictionary.TryGetValue(unitTypeKey, out var value))
		{
			return value;
		}
		if (unitTypeKey.EnumValue.HasValue && Dictionary.TryGetValue(new UnitTypeKey(unitTypeKey.UnitType, null), out value))
		{
			return value;
		}
		return DefaultValue(unitTypeKey);
	}

	protected virtual void SetValue(UnitTypeKey key, T value)
	{
		if (Dictionary.ContainsKey(key))
		{
			Dictionary[key] = value;
		}
		else
		{
			Dictionary.Add(key, value);
		}
	}

	public bool HasSpecification(BaseUnitType unitType, int? enumValue)
	{
		return Dictionary.ContainsKey(new UnitTypeKey(unitType, enumValue));
	}

	public bool RemoveSpecification(BaseUnitType unitType, int? enumValue)
	{
		if (HasSpecification(unitType, enumValue))
		{
			return Dictionary.Remove(new UnitTypeKey(unitType, enumValue));
		}
		return false;
	}
}
