public class Rule
{
	private RuleItem[] _rules;

	private List<RuleItem> _exactRules;

	private List<RuleItem> _xxRules;

	public bool HasExactRules => _exactRules.Count > 0;

	public void Parse(string fileName)
	{
		using (StreamReader streamReader = new StreamReader(fileName))
		{
			JavaScriptSerializer javaScriptSerializer = new JavaScriptSerializer();
			string input = streamReader.ReadToEnd();
			_rules = javaScriptSerializer.Deserialize<RuleItem[]>(input);
		}
		_exactRules = new List<RuleItem>();
		_xxRules = new List<RuleItem>();
		bool flag = false;
		RuleItem[] rules = _rules;
		foreach (RuleItem ruleItem in rules)
		{
			if (ruleItem.Type == "~~")
			{
				flag = true;
			}
			else if (ruleItem.Type == "==" || ruleItem.Type == "!=")
			{
				_exactRules.Add(ruleItem);
				ruleItem.ExactRule = true;
			}
			else if (ruleItem.Type == "xx")
			{
				_xxRules.Add(ruleItem);
			}
		}
		if (flag)
		{
			ArrayList arrayList = new ArrayList();
			rules = _rules;
			foreach (RuleItem value in rules)
			{
				arrayList.Add(value);
			}
			IComparer comparer = new ItemComparer();
			arrayList.Sort(comparer);
			for (int j = 0; j < arrayList.Count; j++)
			{
				_rules[j] = (RuleItem)arrayList[j];
			}
		}
	}

	public void MatchFileName(string path)
	{
		string fileName = Path.GetFileName(path);
		for (int i = 0; i < _rules.Length; i++)
		{
			if (new Regex(_rules[i].FileName).IsMatch(fileName))
			{
				_rules[i].Enabled = true;
				_rules[i].PassCount = 0;
				_rules[i].FailCount = 0;
			}
		}
	}

	public bool MatchXXNode(string nodeName)
	{
		for (int i = 0; i < _xxRules.Count; i++)
		{
			if (_xxRules[i].Enabled)
			{
				string text = _xxRules[i].Node;
				if (_xxRules[i].Detail.Contains("\\w"))
				{
					text += _xxRules[i].Detail;
				}
				if (new Regex(text).IsMatch(nodeName))
				{
					_xxRules[i].PassCount++;
					return true;
				}
			}
		}
		return false;
	}

	public bool MatchNode(string nodeName, string aValue, string bValue, XmlNode node)
	{
		for (int i = 0; i < _rules.Length; i++)
		{
			if (!_rules[i].Enabled || _rules[i].ExactRule)
			{
				continue;
			}
			string text = _rules[i].Node;
			if (_rules[i].Type == "xx" && _rules[i].Detail.Contains("\\w"))
			{
				text += _rules[i].Detail;
			}
			if (new Regex(text).IsMatch(nodeName))
			{
				if (node.Attributes["note"] != null)
				{
					node.Attributes["note"].Value = _rules[i].Note;
				}
				bool num = MatchValue(_rules[i].Type, _rules[i].Detail, aValue, bValue, node);
				if (num)
				{
					_rules[i].PassCount++;
					return num;
				}
				_rules[i].FailCount++;
				return num;
			}
		}
		return false;
	}

	public bool MatchExactRule(string nodeName, string aValue, string bValue, XmlNode node)
	{
		foreach (RuleItem exactRule in _exactRules)
		{
			if (new Regex(exactRule.Node).IsMatch(nodeName))
			{
				if (node.Attributes["note"] != null)
				{
					node.Attributes["note"].Value = exactRule.Note;
				}
				bool flag = MatchValue(exactRule.Type, exactRule.Detail, "", bValue, node);
				if (flag)
				{
					exactRule.PassCount++;
				}
				else
				{
					exactRule.FailCount++;
				}
				if (exactRule.Type == "!=" && aValue == bValue)
				{
					flag = true;
				}
				return flag;
			}
		}
		return true;
	}

	private bool MatchValue(string matchType, string detail, string aValue, string bValue, XmlNode node)
	{
		switch (matchType)
		{
		case "==":
		case "!=":
			if (matchType == "==")
			{
				return bValue == detail;
			}
			return bValue != detail;
		case "~=":
		case "~~":
		case "~A=":
		case "~A<=":
		case "~A>=":
		case "0=":
		{
			if (string.IsNullOrEmpty(aValue) || string.IsNullOrEmpty(bValue))
			{
				return false;
			}
			if (!double.TryParse(aValue, NumberStyles.Number | NumberStyles.AllowExponent, CultureInfo.InvariantCulture.NumberFormat, out var result) || !double.TryParse(bValue, NumberStyles.Number | NumberStyles.AllowExponent, CultureInfo.InvariantCulture.NumberFormat, out var result2))
			{
				return false;
			}
			double num = double.Parse(detail, CultureInfo.InvariantCulture.NumberFormat);
			double num2;
			bool flag;
			switch (matchType)
			{
			case "0=":
				if (result < num)
				{
					result = 0.0;
				}
				if (result2 < num)
				{
					result2 = 0.0;
				}
				return result == result2;
			case "~=":
			case "~~":
				num2 = Math.Abs(result - result2) / Math.Abs(result);
				flag = num2 < num;
				break;
			case "~A=":
				num2 = Math.Abs(result - result2) / (Math.Abs(result + result2) / 2.0);
				flag = num2 < num;
				break;
			default:
				num2 = (result - result2) / (Math.Abs(result + result2) / 2.0);
				flag = ((matchType == "~A<=") ? (num2 < num) : (num2 > num));
				break;
			}
			if (!flag && node.Attributes["note"] != null)
			{
				string value3 = node.Attributes["note"].Value;
				if (value3.Contains("%DIFF%"))
				{
					node.Attributes["note"].Value = value3.Replace("%DIFF%", (num2 * 100.0).ToString("G3") + "%");
				}
			}
			return flag;
		}
		case "xx":
			return true;
		case "x=":
		{
			Regex regex = new Regex(detail);
			string value = regex.Match(aValue).Value;
			string value2 = regex.Match(bValue).Value;
			return value == value2;
		}
		default:
			return false;
		}
	}

	public void SaveStatistics(XmlDocument doc)
	{
		XmlElement xmlElement = doc.CreateElement("RuleStatistics");
		for (int i = 0; i < _rules.Length; i++)
		{
			if (_rules[i].Enabled && (_rules[i].PassCount > 0 || _rules[i].FailCount > 0))
			{
				XmlElement xmlElement2 = doc.CreateElement("Rule");
				XmlAttribute xmlAttribute = doc.CreateAttribute("node");
				xmlAttribute.Value = _rules[i].Node;
				xmlElement2.Attributes.Append(xmlAttribute);
				xmlAttribute = doc.CreateAttribute("passed");
				xmlAttribute.Value = _rules[i].PassCount.ToString();
				xmlElement2.Attributes.Append(xmlAttribute);
				xmlAttribute = doc.CreateAttribute("failed");
				xmlAttribute.Value = _rules[i].FailCount.ToString();
				xmlElement2.Attributes.Append(xmlAttribute);
				xmlAttribute = doc.CreateAttribute("note");
				xmlAttribute.Value = _rules[i].Note;
				xmlElement2.Attributes.Append(xmlAttribute);
				xmlElement.AppendChild(xmlElement2);
			}
		}
		doc.DocumentElement.AppendChild(xmlElement);
	}

		}
