		public CompareCtrl(int columns, string ruleFile, out bool ruleParseError, bool keepIdenticalNodes = false)
		{
			InitializeComponent();
			if (columns > 5)
			{
				columns = 5;
			}
			_columns = columns;
			_files = new string[columns];
			_comparer = new CompareAds.Comparer(columns, ruleFile, out ruleParseError, keepIdenticalNodes);
			_textBlocks = new List<TextBlock>();
			_gridItems = new ArrayList();
		}

		public void Clear()
		{
			Document = null;
			_comparer.Clear();
			_comparer = null;
			_textBlocks.Clear();
			_gridItems.Clear();
		}

		public void CompareFiles(string[] files)
		{
			for (int i = 0; i < _columns; i++)
			{
				_files[i] = files[i];
			}
			_identical = _comparer.CompareFiles(interactive: true, files, out var error, ref _scanned, ref _processed);
			if (string.IsNullOrEmpty(error))
			{
				Document = _comparer.Document;
				(base.Resources["Variables"] as XmlDataProvider).Document = Document;
			}
		}

		public void LoadGrid()
		{
			if (_gridItems.Count == 0 && Document != null)
			{
				MakeGridData(Document);
			}
		}

		public void ShowFolders()
		{
			if (VersionPanel.Children.Count <= 0)

				VersionPanel.Children.Add(element);
				for (int i = 0; i < _columns; i++)

					string directoryName = Path.GetDirectoryName(_files[i]);
					int num = directoryName.LastIndexOf('\\');

						Text = text,
						Style = (base.Resources["VersionText"] as Style)

					VersionPanel.Children.Add(element);
				}
			}
		}

		{
			for (int i = 0; i < _columns; i++)

			Document = new XmlDocument();

			(base.Resources["Variables"] as XmlDataProvider).Document = Document;
			MakeGridData(Document);
		}

		public void CopyGrid()
		{
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("Path\t");
			for (int i = 0; i < _columns; i++)
			{
				string directoryName = Path.GetDirectoryName(_files[i]);
				int num = directoryName.LastIndexOf('\\');
				stringBuilder.Append(directoryName.Substring(num + 1));
				stringBuilder.Append("\t");
			}
			stringBuilder.Append("Note\r\n");
			CollectionView collectionView = (CollectionView)CollectionViewSource.GetDefaultView(Grid.ItemsSource);
			foreach (GridItem gridItem in _gridItems)
			{
				if (!collectionView.Contains(gridItem))
				{
					continue;
				}
				StringBuilder stringBuilder2 = new StringBuilder(gridItem.Path);
				stringBuilder2.Append("\t");
				for (int j = 0; j < _columns; j++)
				{
					string value = gridItem.Value;
					switch (j)
					{
					case 1:
						value = gridItem.Value2;
						break;
					case 2:
						value = gridItem.Value3;
						break;
					case 3:
						value = gridItem.Value4;
						break;
					case 4:
						value = gridItem.Value5;
						break;
					}
					stringBuilder2.Append(value);
					stringBuilder2.Append("\t");
				}
				stringBuilder2.Append(gridItem.Note);
				stringBuilder2.Append("\r\n");
				stringBuilder.Append(stringBuilder2.ToString());
			}
			System.Windows.DataObject dataObject = new System.Windows.DataObject();
			dataObject.SetText(stringBuilder.ToString());
			System.Windows.Clipboard.SetDataObject(dataObject, copy: true);
		}

		private void MakeGridData(XmlDocument doc)
		{
			_gridItems.Clear();
			foreach (XmlNode childNode in doc.DocumentElement.ChildNodes)
			{
				if (!(childNode.Name == "RuleStatistics"))
				{
					AddGridItem(childNode);
				}
			}
			Grid.ItemsSource = _gridItems;
			((CollectionView)CollectionViewSource.GetDefaultView(Grid.ItemsSource)).Filter = GridItemFilter;
		}

		private bool GridItemFilter(object item)
		{
			if (_filterText == "")
			{
				return true;
			}
			if (((GridItem)item).Path.ToLower().Contains(_filterText.ToLower()))
			{
				return true;
			}
			return false;
		}

		private void AddGridItem(XmlNode root)
		{
			if (root.HasChildNodes)
			{
				foreach (XmlNode childNode in root.ChildNodes)
				{
					AddGridItem(childNode);
				}
				return;
			}
			if (root.Attributes["type"].Value != "l" && root.Attributes["path"] != null)
			{
				GridItem gridItem = new GridItem(root.Attributes["path"].Value.Substring(1));
				for (int i = 0; i < _columns; i++)
				{
					string name = ((i == 0) ? "value" : ("value" + (i + 1)));
					string value = ((root.Attributes[name] != null) ? SaveValue(root.Attributes[name].Value) : "");
					gridItem.AddValue(i, value);
				}
				string value2 = root.Attributes["note"].Value;
				gridItem.AddNote(value2);
				_gridItems.Add(gridItem);
			}
		}

		private string SaveValue(string value)
		{
			if (value.Length > 1000 && value.StartsWith("<") && value.EndsWith(">"))
			{
				return "[Xml Data : Length = " + value.Length + "]";
			}
			return value;
		}

			}
		}

		private void AlignTree()
		{
			try
			{
				double num = 0.0;
				foreach (TextBlock textBlock in _textBlocks)
				{
					Point point = textBlock.TransformToAncestor(Tree).Transform(new Point(0.0, 0.0));
					if (point.X + textBlock.ActualWidth > num)
					{
						num = point.X + textBlock.ActualWidth;
					}
				}
				foreach (TextBlock textBlock2 in _textBlocks)
				{
					textBlock2.Width = num - textBlock2.TransformToAncestor(Tree).Transform(new Point(0.0, 0.0)).X;
				}
				_maxTextWidth = num;
			}
			catch
			{
			}
		}

		private void Grid_Loaded(object sender, RoutedEventArgs e)
		{
			GridView gridView = (GridView)((System.Windows.Controls.ListView)sender).View;
			if (gridView.Columns.Count > _columns + 1)
			{
				for (int num = gridView.Columns.Count - 1 - 1; num > _columns; num--)
				{
					gridView.Columns.RemoveAt(num);
				}
			}
			for (int i = 0; i < _columns; i++)
			{
				string directoryName = Path.GetDirectoryName(_files[i]);
				int num2 = directoryName.LastIndexOf('\\');
				gridView.Columns[i + 1].Header = directoryName.Substring(num2 + 1);
			}
		}

		private void TextBlock_Loaded(object sender, RoutedEventArgs e)
		{
			TextBlock textBlock = (TextBlock)sender;
			if (textBlock.Text == "" || textBlock.Tag != null)
			{
				return;
			}
			if (textBlock.Name != "Note" && !_textBlocks.Contains(textBlock))
			{
				_textBlocks.Add(textBlock);
			}
			double pixelsPerDip = VisualTreeHelper.GetDpi(this).PixelsPerDip;
			FormattedText formattedText = new FormattedText(textBlock.Text, CultureInfo.CurrentUICulture, System.Windows.FlowDirection.LeftToRight, new Typeface(textBlock.FontFamily, textBlock.FontStyle, textBlock.FontWeight, textBlock.FontStretch), textBlock.FontSize, Brushes.Black, pixelsPerDip);
			if (_maxTextWidth != 0.0 && textBlock.Name != "Note")
			{
				Point point = textBlock.TransformToAncestor(Tree).Transform(new Point(0.0, 0.0));
				double num = _maxTextWidth - point.X;
				if (num < formattedText.Width + 10.0)
				{
					_maxTextWidth = point.X + formattedText.Width + 10.0;
					num = _maxTextWidth - point.X;
				}
				textBlock.Width = num;
			}
			else if (formattedText.Width > 110.0)
			{
				textBlock.Width = formattedText.Width + 10.0;
			}
			textBlock.Tag = true;
		}

		private void Tree_MouseEnter(object sender, System.Windows.Input.MouseEventArgs e)
		{
			if (!_aligned)
			{
				AlignTree();
				_aligned = true;
			}
		}

		private void SortType_SelectionChanged(object sender, SelectionChangedEventArgs e)
		{
			System.Windows.Controls.ComboBox comboBox = (System.Windows.Controls.ComboBox)sender;
			if (comboBox.SelectedIndex >= 0 && Grid != null)
			{
				CollectionView obj = (CollectionView)CollectionViewSource.GetDefaultView(Grid.ItemsSource);
				obj.Filter = null;
				Grid.ItemsSource = null;
				if (comboBox.SelectedIndex == 0)
				{
					MakeGridData(Document);
				}
				else
				{
					IComparer comparer = new GridItemComparer(comboBox.SelectedIndex);
					_gridItems.Sort(comparer);
				}
				Grid.ItemsSource = _gridItems;
				obj.Filter = GridItemFilter;
			}
		}

		private void FilterText_KeyUp(object sender, System.Windows.Input.KeyEventArgs e)
		{
			System.Windows.Controls.TextBox textBox = (System.Windows.Controls.TextBox)sender;
			string text = textBox.Text.Replace("'", "").Trim();
			if (text.Length > 0)
			{
				_filterText = text.ToLower();
			}
			else
			{
				textBox.Text = "";
				_filterText = "";
			}
			CollectionViewSource.GetDefaultView(Grid.ItemsSource).Refresh();
		}

	public class GridItemComparer : IComparer
	{
		private int _direction;

		public GridItemComparer(int direction)
		{
			_direction = direction;
		}

		int IComparer.Compare(object ci1, object ci2)
		{
			string path = ((GridItem)ci1).Path;
			string path2 = ((GridItem)ci2).Path;
			if (_direction == 1)
			{
				return path.CompareTo(path2);
			}
			return path2.CompareTo(path);
		}
	}

	public class VariableTemplateSelector : DataTemplateSelector
	{
		public override DataTemplate SelectTemplate(object item, DependencyObject container)
		{
			FrameworkElement frameworkElement = container as FrameworkElement;
			XmlNode xmlNode = (XmlNode)item;
			DataTemplate result = null;
			if (xmlNode.Name == "Variable")
			{
				switch (xmlNode.Attributes["type"].Value)
				{
				case "a":
				case "f":
					result = frameworkElement.FindResource("ArrayTemplate") as DataTemplate;
					break;
				case "b":
					result = frameworkElement.FindResource("BooleanTemplate") as DataTemplate;
					break;
				case "l":
					result = frameworkElement.FindResource("LabelTemplate") as DataTemplate;
					break;
				default:
					result = frameworkElement.FindResource("VariableTemplate") as DataTemplate;
					break;
				}
			}
			return result;
		}
	}

		private void RunWorker()
		{
			BackgroundWorker backgroundWorker = new BackgroundWorker();
			backgroundWorker.DoWork += Worker0_DoWork;
			backgroundWorker.RunWorkerCompleted += Worker0_RunWorkerCompleted;
			backgroundWorker.RunWorkerAsync();
		}

		private void Worker0_RunWorkerCompleted(object sender, RunWorkerCompletedEventArgs e)
		{
			for (int i = _startIndex; i < _endIndex; i++)
			{
				string fileName = Path.GetFileName(_runners[i].files[0]);
				List.Items.Add(fileName);
			}
			if (_endIndex == _totalFiles)
			{
				Mouse.OverrideCursor = null;
				return;
			}
			_startIndex = _endIndex;
			_endIndex = _startIndex + _maxThreads;
			if (_endIndex > _totalFiles)
			{
				_endIndex = _totalFiles;
			}
			_pastProgress = (double)_startIndex * 100.0 / (double)_totalFiles;
			_currentProgress = (double)(_endIndex - _startIndex) * 100.0 / (double)_totalFiles;
			RunWorker();
		}

		private void Worker0_DoWork(object sender, DoWorkEventArgs e)
		{
			BackgroundWorker backgroundWorker = new BackgroundWorker();
			backgroundWorker.DoWork += Worker2_DoWork;
			backgroundWorker.RunWorkerCompleted += Worker2_RunWorkerCompleted;
			BackgroundWorker backgroundWorker2 = new BackgroundWorker
			{
				WorkerReportsProgress = true
			};
			backgroundWorker2.DoWork += Worker_DoWork;
			backgroundWorker2.ProgressChanged += Worker_ProgressChanged;
			backgroundWorker2.RunWorkerCompleted += Worker_RunWorkerCompleted;
			for (int i = _startIndex; i < _endIndex; i++)
			{
				CompareFiles2(_runners[i]);
			}
			_running = true;
			backgroundWorker.RunWorkerAsync();
			backgroundWorker2.RunWorkerAsync();
			while (_running)
			{
				Thread.Sleep(100);
			}
		}

		private void Worker2_RunWorkerCompleted(object sender, RunWorkerCompletedEventArgs e)
		{
			_running = false;
		}

		private void Worker2_DoWork(object sender, DoWorkEventArgs e)
		{
			while (true)
			{
				int num = 0;
				foreach (Thread thread in _threads)
				{
					if (thread.ThreadState == System.Threading.ThreadState.Running || thread.ThreadState == System.Threading.ThreadState.Suspended)
					{
						num++;
					}
				}
				if (num != 0)
				{
					Thread.Sleep(100);
					continue;
				}
				break;
			}
		}

		private void Worker_RunWorkerCompleted(object sender, RunWorkerCompletedEventArgs e)
		{
			StringBuilder stringBuilder = new StringBuilder();
			for (int i = _startIndex; i < _endIndex; i++)
			{
				RunnerArg runnerArg = _runners[i];
				string fileName = Path.GetFileName(runnerArg.files[0]);
				stringBuilder.Append(fileName + " : ");
				string text = ((runnerArg.comparer.Comparer.ResultCode == 1) ? "Identical" : "Different");
				runnerArg.fileset.Status = text;
				runnerArg.fileset.Width = 200.0;
				stringBuilder.Append(text);
				stringBuilder.AppendLine();
			}
			_summaryText = stringBuilder.ToString();
			_threads.Clear();
		}

		private void Worker_DoWork(object sender, DoWorkEventArgs e)
		{
			int num = 0;
			do
			{
				int num2 = 0;
				int num3 = 0;
				for (int i = _startIndex; i < _endIndex; i++)
				{
					RunnerArg runnerArg = _runners[i];
					if (runnerArg.comparer.Scanned > 0)
					{
						num2 += runnerArg.comparer.Scanned;
						num3 += runnerArg.comparer.Processed;
						runnerArg.fileset.Width = (double)runnerArg.comparer.Processed * 200.0 / (double)runnerArg.comparer.Scanned;
					}
				}
				int num4 = (int)((double)num3 * 100.0 / (double)num2);
				if (num4 > 0 && num4 > num)
				{
					(sender as BackgroundWorker).ReportProgress(num4);
					num = num4;
				}
				Thread.Sleep(100);
			}
			while (_running);
		}

		private void Worker_ProgressChanged(object sender, ProgressChangedEventArgs e)
		{
			_totalProgress.Width = _pastProgress + (double)e.ProgressPercentage / 100.0 * _currentProgress;
		}

		private void ViewResult(object sender, RoutedEventArgs e)
		{
			OpenFileDialog openFileDialog = new OpenFileDialog
			{
				Filter = "xml files (*.xml) |*.xml|All files (*.*)|*.*",
				FilterIndex = 1,
				RestoreDirectory = true
			};

			{
				return;
			}
			Clear(null, null);
			XmlDocument xmlDocument = new XmlDocument();
			xmlDocument.Load(openFileDialog.FileName);
			int count = xmlDocument.DocumentElement.ChildNodes.Count;
			for (int i = 0; i < count; i++)
			{
				XmlNode xmlNode = xmlDocument.DocumentElement.ChildNodes[i];
				if (xmlNode.Name == "Summary")
				{
					string[] array = xmlNode.InnerText.Replace("\r\n", "\n").Split('\n');
					foreach (string text in array)
					{
						FileSet fileSet = new FileSet();
						if (!string.IsNullOrEmpty(text))
						{
							string[] array2 = text.Split(':');
							fileSet.FileName = array2[0];
							fileSet.Status = array2[1];
							fileSet.Width = 0.0;
						}
						Summary.Items.Add(fileSet);
					}
					continue;
				}
				List<string> list = new List<string>();
				int num = 0;
				while (true)
				{
					int j = ++num;
					string name = "File" + j;
					if (xmlNode.Attributes[name] == null)
					{
						break;
					}
					list.Add(xmlNode.Attributes[name].Value);
				}
				string fileName = Path.GetFileName(list[0]);
				bool ruleParseError;
				CompareCtrl compareCtrl = new CompareCtrl(list.Count, "", out ruleParseError);
				_compareCtrls.Add(compareCtrl);
				List.Items.Add(fileName);
				compareCtrl.LoadResult(list, xmlNode.OuterXml);
			}
		}
