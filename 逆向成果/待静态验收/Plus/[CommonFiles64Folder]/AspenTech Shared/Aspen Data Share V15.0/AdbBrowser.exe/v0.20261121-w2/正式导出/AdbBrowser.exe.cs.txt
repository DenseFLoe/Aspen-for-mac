	public class UnitConv
	{
		public static char[] sep = new char[1] { ' ' };

		public string Name { get; set; }

		public double A { get; set; }

		public double B { get; set; }

		public UnitConv(string line)
		{
			A = 1.0;
			B = 0.0;
			string[] array = line.Split(sep, StringSplitOptions.RemoveEmptyEntries);
			Name = array[0];
			if (array.Length > 1)
			{
				A = double.Parse(array[1], CultureInfo.InvariantCulture);
			}
			if (array.Length > 2)
			{
				B = double.Parse(array[2], CultureInfo.InvariantCulture);
			}
		}
	}
	public class UnitRow
	{
		public int RowNumber { get; set; }

		public string Name { get; set; }

		public Dictionary<string, UnitConv> Conversions { get; set; }

		public UnitConv[] Columns { get; set; }

		public UnitRow(string line, StreamReader sr)
		{
			string[] array = line.Split(UnitConv.sep, StringSplitOptions.RemoveEmptyEntries);
			RowNumber = int.Parse(array[0]);
			int num = int.Parse(array[1]);
			Name = array[2];
			Columns = new UnitConv[num];
			Conversions = new Dictionary<string, UnitConv>();
			for (int i = 0; i < num; i++)
			{
				line = sr.ReadLine();
				Columns[i] = new UnitConv(line);
				Conversions[Columns[i].Name] = Columns[i];
			}
		}
	}

	public class UnitConverter : IMultiValueConverter
	{
		public object Convert(object[] values, Type targetType, object parameter, CultureInfo culture)
		{
			string result = null;
			if (values != null && values[0] != DependencyProperty.UnsetValue)
			{
				result = (string)values[0];
			}
			if (values == null || values.Length != 3 || values[0] == DependencyProperty.UnsetValue || values[1] == DependencyProperty.UnsetValue || values[2] == DependencyProperty.UnsetValue)
			{
				return result;
			}
			if (double.TryParse((string)values[0], NumberStyles.Any, CultureInfo.CurrentCulture, out var result2) && int.TryParse((string)values[1], out var result3) && int.TryParse((string)values[2], out var result4))
			{
				UnitConv unitConv = UnitsTable.Rows[result4].Columns[result3];
				result = ((result2 - unitConv.B) / unitConv.A).ToString();
			}
			return result;
		}

		public object[] ConvertBack(object value, Type[] targetTypes, object parameter, CultureInfo culture)
		{
			object[] array = new object[3];
			if (value != null && value != DependencyProperty.UnsetValue && double.TryParse((string)value, out var result))
			{
				array[0] = result.ToString();
			}
			return array;
		}

		private void UpdateChildren()
		{
			if (base.Children.Count != 1 || base.Children[0].Name != null)
			{
				return;
			}
			base.Children.Clear();
			for (int i = 0; i < Count; i++)
			{
				object obj = _record.ExecutePath(new object[2]
				{
					Array.ToString(),
					AdbArrayMethodGuids.ValueAtLocation
				}, new object[1] { i });
				object obj2 = _record.ExecutePath(new object[2]
				{
					Array.ToString(),
					AdbArrayMethodGuids.IdAtLocation
				}, new object[1] { i });
				string text = $"#{i}";
				if (obj2 != null && IdType != typeof(AdbNoId))
				{
					if (IdType.IsSubclassOf(typeof(AdbBaseRecord)))
					{
						text = (obj2 as AdbBaseRecord).MyGuid.ToString();
						Type type = obj2.GetType();
						try
						{
							object obj3 = type.InvokeMember("ID", BindingFlags.Instance | BindingFlags.Public | BindingFlags.FlattenHierarchy | BindingFlags.GetField, null, obj2, new object[0]);
							if (obj3 != null)
							{
								text = obj3.ToString();
							}
						}
						catch (Exception)
						{
						}
					}
					else
					{
						text = obj2.ToString();
					}
				}
				if (ValueType == typeof(AdbPropertyHolder))
				{
					base.Children.Add(new AdbPropertyHolderNode((AdbPropertyHolder)((obj is AdbPropertyHolder) ? obj : null), text, ValueType));
					continue;
				}
				if (ValueType.IsSubclassOf(typeof(AdbBaseRecord)))
				{
					base.Children.Add(new AdbBrowserRecordNode(obj as AdbBaseRecord, text, ValueType));
					continue;
				}
				int? unitRow = null;
				object obj4 = _record.ExecutePath(new object[2]
				{
					Array.ToString(),
					AdbArrayMethodGuids.GetUnitRow
				}, new object[1] { i });
				if (obj4 != null)
				{
					unitRow = (int)obj4;
				}
				base.Children.Add(new AdbBrowserNode
				{
					Name = text,
					NodeName = text,
					Value = ((obj == null) ? "null" : obj.ToString()),
					UnitRow = unitRow
				});
			}
		}
