public sealed class AssemblyResolveHelper : IDisposable
{
	internal readonly string assemblySearchLocation;

	private bool disposedValue;

	public AssemblyResolveHelper(string location)
	{
		assemblySearchLocation = location;
		AppDomain.CurrentDomain.AssemblyResolve += ResolveAssemblyInSearchLocation;
	}

	private Assembly ResolveAssemblyInSearchLocation(object sender, ResolveEventArgs args)
	{
		string path = args.Name.Split(',').First() + ".dll";
		string text = Path.Combine(assemblySearchLocation, path);
		if (File.Exists(text) && AssemblyName.GetAssemblyName(text).FullName == args.Name)
		{
			return Assembly.LoadFile(text);
		}
		return null;
	}

	private void Dispose(bool disposing)
	{
		if (!disposedValue)
		{
			if (disposing)
			{
				AppDomain.CurrentDomain.AssemblyResolve -= ResolveAssemblyInSearchLocation;
			}
			disposedValue = true;
		}
	}

	public void Dispose()
	{
		Dispose(disposing: true);
	}
}

}
[Guid("6F08E9B6-1436-49E6-A1E5-756EB0D00360")]
[ComVisible(true)]
public class ExtnLoader : IExtnLoader
{
	[return: MarshalAs(UnmanagedType.IUnknown)]
	public object CreateObjectWithGuidInDLL([In][MarshalAs(UnmanagedType.BStr)] string guid, [In][MarshalAs(UnmanagedType.BStr)] string path)
	{
		if (string.IsNullOrEmpty(path) || string.IsNullOrEmpty(guid))
		{
			return null;
		}
		if (!Path.HasExtension(path) || !File.Exists(path))
		{
			return null;
		}
		if (!Guid.TryParse(guid, out var result))
		{
			return null;
		}
		using (new AssemblyResolveHelper(Path.GetDirectoryName(path)))
		{
			Type[] exportedTypes = Assembly.LoadFile(path).GetExportedTypes();
			foreach (Type type in exportedTypes)
			{
				if (!(result != type.GUID))
				{
					try
					{
						return Activator.CreateComInstanceFrom(path, type.FullName).Unwrap();
					}
					catch
					{
						return null;
					}
				}
			}
		}
		return null;
	}

	[return: MarshalAs(UnmanagedType.IUnknown)]
	public object CreateObjectWithProgIdInDLL([In][MarshalAs(UnmanagedType.BStr)] string progId, [In][MarshalAs(UnmanagedType.BStr)] string path)
	{
		if (string.IsNullOrEmpty(path) || string.IsNullOrEmpty(progId))
		{
			return null;
		}
		if (!Path.HasExtension(path) || !File.Exists(path))
		{
			return null;
		}
		using (new AssemblyResolveHelper(Path.GetDirectoryName(path)))
		{
			Type[] exportedTypes = Assembly.LoadFile(path).GetExportedTypes();
			foreach (Type type in exportedTypes)
			{
				IEnumerable<Attribute> customAttributes = type.GetCustomAttributes(typeof(ProgIdAttribute));
				string text = ((!customAttributes.Any()) ? type.FullName : (customAttributes.First() as ProgIdAttribute).Value);
				if (!(text != progId))
				{
					try
					{
						return Activator.CreateComInstanceFrom(path, type.FullName).Unwrap();
					}
					catch
					{
						return null;
					}
				}
			}
		}
		return null;
	}
}
