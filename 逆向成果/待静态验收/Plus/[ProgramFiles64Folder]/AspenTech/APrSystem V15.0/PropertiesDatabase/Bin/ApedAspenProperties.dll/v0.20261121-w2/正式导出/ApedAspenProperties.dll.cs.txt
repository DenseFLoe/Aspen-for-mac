	public void SetupCitation(IDatabank iDatabank, string strParaName, string strCompI, string strCompJ)
	{
		//IL_003a: Unknown result type (might be due to invalid IL or missing references)
		//IL_0040: Expected O, but got Unknown
		//IL_0122: Unknown result type (might be due to invalid IL or missing references)
		//IL_0129: Expected O, but got Unknown
		if (citationNameList != null && citationNameList.Count > 0)
		{
			citationNameList.Clear();
		}
		if (citationNameList == null)
		{
			citationNameList = new ArrayList();
		}
		Database val = (Database)iDatabank.Database;
		if (val == null)
		{
			return;
		}
		ICitationManager citationManager = val.CitationManager;
		ICompoundManager compoundManager = val.CompoundManager;
		IIdentifierValue val2 = null;
		if (citationManager == null || compoundManager == null)
		{
			return;
		}
		int num = 0;
		int num2 = 0;
		int num3 = -1;
		bool flag = false;
		compoundManager.GetCompoundIdentifier(strCompI, "NAME", "CompoundIndex", ref val2);
		if (val2 == null)
		{
			compoundManager.GetCompoundIdentifier(strCompI, "ALIAS", "CompoundIndex", ref val2);
			if (val2 == null)
			{
				return;
			}
		}
		num = val2.Compound.Id;
		compoundManager.GetCompoundIdentifier(strCompJ, "NAME", "CompoundIndex", ref val2);
		if (val2 == null)
		{
			compoundManager.GetCompoundIdentifier(strCompJ, "ALIAS", "CompoundIndex", ref val2);
			if (val2 == null)
			{
				return;
			}
		}
		num2 = val2.Compound.Id;
		compoundManager.GetCompoundDuoId(num, num2, false, ref num3, ref flag);
		ArrayList arrayList = new ArrayList();
		citationManager.GetCitation(iDatabank, num3.ToString(), "CompoundIndex", strParaName, ref arrayList);
		if (arrayList == null)
		{
			return;
		}
		int num4 = 0;
		foreach (ICitation item in arrayList)
		{
			ICitation val3 = item;
			string value = "[" + (num4 + 1) + "]: " + val3.Authors.ToString();
			num4++;
			citationNameList.Add(value);
		}
	}

	public int CitationSize()
	{
		return citationNameList.Count;
	}

	public string GetCitation(int i)
	{
		if (i < citationNameList.Count)
		{
			return (string)citationNameList[i];
		}
		return null;
	}
}
