		public void OnResize(object sender, EventArgs e)
		{

			//IL_0119: Expected O, but got Unknown
			Rectangle bounds = base.Bounds;
			if (bounds.Height == m_prevBounds.Height && bounds.Width == m_prevBounds.Width)
			{
				return;
			}
			if (bounds.Height < m_initBounds.Height)
			{
				SetBounds(base.Bounds.Left, base.Bounds.Top, base.Bounds.Width, m_initBounds.Height, BoundsSpecified.Height);
			}
			if (bounds.Width < m_initBounds.Width)

				SetBounds(base.Bounds.Left, base.Bounds.Top, m_initBounds.Width, base.Bounds.Height, BoundsSpecified.Width);
			}
			m_prevBounds = base.Bounds;
