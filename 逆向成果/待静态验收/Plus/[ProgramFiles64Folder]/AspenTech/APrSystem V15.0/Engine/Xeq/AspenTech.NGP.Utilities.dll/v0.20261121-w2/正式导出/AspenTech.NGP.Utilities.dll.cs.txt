	public static class HollerithUtility
	{

			{
				text += ToString((int)list[i]);
			}

		}

		public static string ToString(int integer)
		{
			if (integer < 538976288 || integer > 2139062143)
			{
				return "";
			}
			char c = (char)(integer & 0xFF);

			char c3 = (char)((uint)(integer & 0xFF0000) / 65536u);
			char c4 = (char)((uint)(integer & -16777216) / 16777216u);
			return $"{c}{c2}{c3}{c4}";
		}

		public static string ToString(long integer)
		{
			return ToString((int)integer);
		}

		public static string ToString(IEnumerable<RawInteger> ints)

		}

		{
