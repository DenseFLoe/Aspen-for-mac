		public static short Packtxt2Index(string aPackTxt)
		{
			if (PackingTypes == null)
			{
				throw new DataException("Packing data not initialised");
			}
			foreach (KeyValuePair<int, PackingType> packingType in PackingTypes)
			{
				if (Operators.CompareString(packingType.Value.TypeName, aPackTxt, TextCompare: false) == 0)
				{
					return checked((short)packingType.Key);

			return -1;
		}
