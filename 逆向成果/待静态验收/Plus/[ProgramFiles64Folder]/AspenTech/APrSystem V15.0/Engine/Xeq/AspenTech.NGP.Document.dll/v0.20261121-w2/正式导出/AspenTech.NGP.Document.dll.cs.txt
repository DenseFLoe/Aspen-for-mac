		public void SetPropertyPackage(string sJson)
		{
			PropertyPackageDto val = JsonSerializer.Deserialize<PropertyPackageDto>(sJson, (JsonSerializerOptions)null);
			PostDS(val.datasources.ToList());
			PostCL(val.compaliases.ToList());
			PostOpsets(val.propertymethods.ToList());
		}

		public void PostCL(List<string> compAliases)
		{
			if (compAliases != null && compAliases.Count != 0)
			{
				List<IComponent> components = new List<IComponent>();
				compAliases.ForEach(delegate(string item)
				{
					components.Add(Factories.CreateComponentByAlias(item));
				});
				ComponentManager.Components.AddRange((IEnumerable<IComponent>)components);
			}
		}

		public void PostDS(List<DatasourcesDto> dataSources)
		{
			_ = string.Empty;
			List<IDataSource> list = new List<IDataSource>();
			foreach (DatasourcesDto dataSource in dataSources)
			{
				list.Add(Factories.CreateDataSource(dataSource.databaseName, dataSource.databankName));
			}
			ParameterManager.DataSources.AddRange((IEnumerable<IDataSource>)list);
		}

		public void PostOpsets(List<string> optionSetNames)
		{
			if (optionSetNames != null && optionSetNames.Count != 0)
			{
				List<IOptionSet> options = new List<IOptionSet>();
				optionSetNames.ForEach(delegate(string item)
				{
					options.Add(Factories.OptionSet.GetOrCreate(item));
				});
				((IObservableList<IOptionSet>)(object)OptionSetManager.OptionSets).AddRange((IEnumerable<IOptionSet>)options);
				OptionSetManager[optionSetNames[0]].SelectAs((OptionSetChoice)1);
				OptionSetManager[optionSetNames[0]].SelectAs((OptionSetChoice)1);
				if (OptionSetManager[optionSetNames.Last()].IsFreeWaterMethod)
				{
					OptionSetManager[optionSetNames.Last()].SelectAs((OptionSetChoice)2);
				}
			}
		}
