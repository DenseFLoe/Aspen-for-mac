		{
			if (Components.Count == 0 || string.IsNullOrEmpty(ParameterName) || Values.Count == 0)

			if (IsPure)

				if (IJValues.Count == 0)

					return "BPVAL " + text3 + " " + text2 + " " + ValuesToAspenPlusBinaryInputString(JIValues);
				}
				if (JIValues.Count == 0)

					return "BPVAL " + text2 + " " + text3 + " " + ValuesToAspenPlusBinaryInputString(IJValues);
				}
				string text4 = "BPVAL " + text2 + " " + text3 + " " + ValuesToAspenPlusBinaryInputString(IJValues) + Environment.NewLine;
				return text4 + "BPVAL " + text3 + " " + text2 + " " + ValuesToAspenPlusBinaryInputString(JIValues);
			}
			if (IsPair)

				string text6 = ((Components[1] != null) ? Components[1].Descriptor : string.Empty);
