		internal class DocumentBandInfo : List<DocumentHeaderInfo>
		{
			private class GroupSortComparer : IComparer<UltraGridGroup>
			{
				int IComparer<UltraGridGroup>.Compare(UltraGridGroup x, UltraGridGroup y)
				{
					return x.Header.VisiblePosition.CompareTo(y.Header.VisiblePosition);
				}
			}

			private int[] colDims;

			private int[] headerRowDims;

			private UltraGridBand band;

			private int bandWidth;

			private List<UltraGridGroup> groupList;

			private GroupSortComparer groupSortComparer;

			private int? groupByConnectorsExtent = null;

			private PlaceholderType placeholderType;

			private Dictionary<int, List<DocumentHeaderInfo>> logicalRows;

			private int groupByColumnCount = -1;

			public int[] ColDims => colDims;

			public int[] HeaderRowDims => headerRowDims;

			public UltraGridBand Band => band;

			public int BandWidth => bandWidth;

			public int GroupByColumnCount
			{
				get
				{
					if (groupByColumnCount == -1)
					{
						groupByColumnCount = 0;
						foreach (UltraGridColumn item in (IEnumerable)band.SortedColumns)
						{
							if (item.IsGroupByColumn)
							{
								groupByColumnCount++;
							}
						}
					}
					return groupByColumnCount;
				}
			}

			public bool HasGroups
			{
				get
				{
					if (groupList != null)
					{
						return groupList.Count > 0;
					}
					return false;
				}
			}

			public int GroupByConnectorsExtent
			{
				get
				{
					if (!groupByConnectorsExtent.HasValue)
					{
						if (placeholderType == PlaceholderType.ColumnHeaders && band.HeaderPlacementResolved != HeaderPlacement.OncePerGroupedRowIsland)
						{
							groupByConnectorsExtent = band.CalcGroupByConnectorsExtent();
						}
						else
						{
							groupByConnectorsExtent = 0;
						}
					}
					return groupByConnectorsExtent.Value;
				}
			}

			public List<UltraGridGroup> GroupList
			{
				get
				{
					return groupList;
				}
				set
				{
					groupList = value;
				}
			}

			private GroupSortComparer MyGroupSortComparer
			{
				get
				{
					if (groupSortComparer == null)
					{
						groupSortComparer = new GroupSortComparer();
					}
					return groupSortComparer;
				}
			}

			public DocumentBandInfo(UltraGridBand band, PlaceholderType placeholderType)
			{
				this.band = band;
				this.placeholderType = placeholderType;
			}

			public int GetColumnWidth(DocumentHeaderInfo documentColumnInfo)
			{
				if (documentColumnInfo == null)
				{
					return 0;
				}
				int num = colDims[documentColumnInfo.OriginX];
				int num2 = colDims[documentColumnInfo.OriginX + documentColumnInfo.SpanX];
				return num2 - num;
			}

			public List<DocumentHeaderInfo> GetItemsInLogicalRow(int logicalRow)
			{
				if (logicalRows == null)
				{
					logicalRows = new Dictionary<int, List<DocumentHeaderInfo>>();
				}
				if (logicalRows.TryGetValue(logicalRow, out var value))
				{
					return value;
				}
				value = new List<DocumentHeaderInfo>();
				using (Enumerator enumerator = GetEnumerator())
				{
					while (enumerator.MoveNext())
					{
						DocumentHeaderInfo current = enumerator.Current;
						int num = current.OriginY + current.SpanY;
						if (current.OriginY <= logicalRow && num > logicalRow)
						{
							value.Add(current);
						}
					}
				}
				value.Sort(CompareOriginX);
				logicalRows[logicalRow] = value;
				return value;
			}

			private int CompareOriginX(DocumentHeaderInfo documentHeaderInfoX, DocumentHeaderInfo documentHeaderInfoY)
			{
				return documentHeaderInfoX.OriginX.CompareTo(documentHeaderInfoY.OriginX);
			}

			public void InitializeDims(int[] colDims, int[] headerRowDims)
			{
				this.colDims = colDims;
				this.headerRowDims = headerRowDims;
				CalculateBandWidth();
				Sort();
				BuildGroupList();
			}

			private void BuildGroupList()
			{
				groupList = new List<UltraGridGroup>(band.Groups.Count);
				foreach (UltraGridGroup group in band.Groups)
				{
					if (!group.Hidden)
					{
						groupList.Add(group);
					}
				}
				groupList.Sort(MyGroupSortComparer);
			}

			private void CalculateBandWidth()
			{
				bandWidth = band.GetExtent(BandOrigin.RowCellArea);
				bandWidth += GroupByConnectorsExtent;
			}
		}

		{

			private int spanX;

			private int spanY;

			private DocumentHeaderInfoType documentColumnInfoType;

			private ILayoutItem layoutItem;

			private HeaderBase header;

			private Rectangle relativeRect;

			public int OriginX => originX;

			public int OriginY => originY;

			public int SpanX => spanX;

			public int SpanY => spanY;

			public Rectangle RelativeRect => relativeRect;

			public UltraGridColumn Column
			{
				get
				{
					if (header == null)
					{
						return null;
					}
					return header.Column;
				}
			}

			public DocumentHeaderInfoType DocumentHeaderInfoType => documentColumnInfoType;

			public ILayoutItem LayoutItem => layoutItem;

			public HeaderBase Header => header;

			public DocumentHeaderInfo(HeaderBase header, int originX, int originY, int spanX, int spanY, DocumentHeaderInfoType documentColumnInfoType, ILayoutItem layoutItem)
			{
				this.originX = originX;
				this.originY = originY;
				this.spanX = spanX;
				this.spanY = spanY;
				this.documentColumnInfoType = documentColumnInfoType;
				this.layoutItem = layoutItem;
				this.header = header;
				relativeRect = new Rectangle(this.originX, this.originY, this.spanX, this.spanY);
			}

			public int CompareTo(DocumentHeaderInfo other)
			{
				if (OriginY != other.OriginY)
				{
					return OriginY.CompareTo(other.OriginY);
				}
				if (OriginX != other.originX)
				{
					return OriginX.CompareTo(other.originX);
				}
				if (header == null || other.header == null)
				{
					return 0;
				}
				return Header.VisiblePosition.CompareTo(other.Header.VisiblePosition);
			}
		}

		private class GroupLevelColumnList : List<GroupLevelColumnInfo>
		{
			private UltraGridBand band;

			private int level;

			internal GroupLevelColumnList(UltraGridBand band, int level)
			{
				this.band = band;
				this.level = level;
				BuildList();
			}

			private void BuildList()
			{
				int num = 0;
				UltraGridColumn firstVisibleCol = band.GetFirstVisibleCol(null, includeNonActivateable: true);
				UltraGridGroup ultraGridGroup = firstVisibleCol.Group;
				ultraGridGroup = ultraGridGroup.GetRelatedVisibleGroup(VisibleRelation.First);
				while (ultraGridGroup != null)
				{
					bool flag = false;
					firstVisibleCol = band.GetFirstVisibleCol(null, includeNonActivateable: true);
					int num2 = ultraGridGroup.Width;
					while (firstVisibleCol != null)
					{
						UltraGridColumn relatedVisibleColumn = firstVisibleCol.GetRelatedVisibleColumn(VisibleRelation.Next);
						if (firstVisibleCol.Group == ultraGridGroup && firstVisibleCol.Level == level)
						{
							int width = firstVisibleCol.CellSizeResolved.Width;
							if (relatedVisibleColumn != null && relatedVisibleColumn.Group == ultraGridGroup && relatedVisibleColumn.Level == firstVisibleCol.Level)
							{
								num2 -= width;
								num += width;
							}
							else
							{
								num += num2;
							}
							flag = true;
							Add(new GroupLevelColumnInfo(firstVisibleCol, num));
						}
						firstVisibleCol = relatedVisibleColumn;
					}
					if (!flag)
					{
						num += num2;
						Add(new GroupLevelColumnInfo(null, num));
					}
					ultraGridGroup = ultraGridGroup.GetRelatedVisibleGroup(VisibleRelation.Next);
					while (ultraGridGroup != null && ultraGridGroup.HiddenResolved)
					{
						ultraGridGroup = ultraGridGroup.GetRelatedVisibleGroup(VisibleRelation.Next);
					}
				}
				Sort();
			}
		}

		public WorksheetCellRangeInfo GetWorksheetCells(List<UltraGridCell> cells)
		{
			bool flag = true;
			UltraGridColumn ultraGridColumn = null;
			List<WorksheetCellInfo> list = new List<WorksheetCellInfo>();
			foreach (UltraGridCell cell in cells)
			{
				WorksheetCellInfo worksheetCellInfo = GetWorksheetCellInfo(cell);
				list.Add(worksheetCellInfo);
				if (flag)
				{
					if (ultraGridColumn == null)
					{
						ultraGridColumn = cell.Column;
					}
					else if (ultraGridColumn != cell.Column)
					{
						flag = false;
					}
				}
			}
			if (flag && list.Count > 0)
			{
				list.Sort(WorksheetCellComparer);
				int rowIndex = list[0].RowIndex;
				for (int i = 1; i < list.Count; i++)
				{
					WorksheetCellInfo worksheetCellInfo2 = list[i];
					if (worksheetCellInfo2.RowIndex != rowIndex + 1)
					{
						flag = false;
						break;
					}
					rowIndex = worksheetCellInfo2.RowIndex;
				}
			}
			WorksheetCellRangeInfo worksheetCellRangeInfo;
			if (flag)
			{
				worksheetCellRangeInfo = new WorksheetCellRangeInfo(cells.Count);
				foreach (WorksheetCellInfo item in list)
				{
					worksheetCellRangeInfo.AddWorksheetCellInfo(item);
				}
			}
			else
			{
				worksheetCellRangeInfo = HiddenWorksheetManager.GetHiddenWorksheetCells(ultraGridColumn, list);
			}
			return worksheetCellRangeInfo;
		}

	internal class WorksheetCellRangeInfo
	{
		private List<WorksheetCellInfo> worksheetCellInfos;

		private bool hasCellsInMultipleWorksheets;

		private bool hasCellsInMultipleColumns;

		private Worksheet worksheet;

		private int columnIndex = -1;

		public bool IsContiguous
		{
			get
			{
				if (hasCellsInMultipleColumns)
				{
					return false;
				}
				if (worksheetCellInfos != null && worksheetCellInfos.Count > 0)
				{
					worksheetCellInfos.Sort(MappingManager.WorksheetCellComparer);
					int rowIndex = worksheetCellInfos[0].RowIndex;
					for (int i = 1; i < worksheetCellInfos.Count; i++)
					{
						WorksheetCellInfo worksheetCellInfo = worksheetCellInfos[i];
						if (worksheetCellInfo.RowIndex != rowIndex + 1)
						{
							return false;
						}
						rowIndex = worksheetCellInfo.RowIndex;
					}
				}
				return true;
			}
		}

		public WorksheetCellRangeInfo()
			: this(0)
		{
		}

			{
				worksheetCellInfos = new List<WorksheetCellInfo>(initialCapacity);
			}
		}

		public void AddWorksheetCellInfo(WorksheetCellInfo worksheetCellInfo)
		{
			if (!hasCellsInMultipleWorksheets)
			{
				if (worksheet == null)
				{
					worksheet = worksheetCellInfo.Worksheet;
				}
				else if (worksheetCellInfo.Worksheet != Worksheet)
				{
					hasCellsInMultipleWorksheets = true;
					hasCellsInMultipleColumns = true;
				}
			}
			if (!hasCellsInMultipleColumns)
			{
				if (columnIndex == -1)
				{
					columnIndex = worksheetCellInfo.ColumnIndex;
				}
				else if (worksheetCellInfo.ColumnIndex != columnIndex)
				{
					hasCellsInMultipleColumns = true;
				}
			}
			WorksheetCellInfos.Add(worksheetCellInfo);
		}

		{
			return WorksheetCellInfos.ToArray();
		}

		public WorksheetRegion GetWorksheetRegion()
		{
			if (!IsContiguous)
			{
				throw new InvalidOperationException("Cannot create a WorksheetRegion for non-contiguous cells");
			}
			if (WorksheetCellInfos.Count == 0)
			{
				return null;
			}
			WorksheetCellInfos.Sort(MappingManager.WorksheetCellComparer);
			WorksheetCellInfo worksheetCellInfo = WorksheetCellInfos[0];
			WorksheetCellInfo worksheetCellInfo2 = WorksheetCellInfos[WorksheetCellInfos.Count - 1];
			return new WorksheetRegion(Worksheet, worksheetCellInfo.RowIndex, worksheetCellInfo.ColumnIndex, worksheetCellInfo2.RowIndex, worksheetCellInfo2.ColumnIndex);
		}

	internal class GridRowComparer : IComparer<UltraGridRow>
	{
		private FormulaRowIndexSource formulaRowIndexSource = FormulaRowIndexSource.VisibleIndex;

		public FormulaRowIndexSource FormulaRowIndexSource
		{
			get
			{
				return formulaRowIndexSource;
			}
			set
			{
				formulaRowIndexSource = value;
			}
		}

		public int Compare(UltraGridRow x, UltraGridRow y)
		{
			FormulaRowIndexSource formulaRowIndexSource = FormulaRowIndexSource;
			while (x.ParentRow != y.ParentRow)
			{
				formulaRowIndexSource = FormulaRowIndexSource.VisibleIndex;
				x = x.ParentRow;
				y = y.ParentRow;
			}
			int num;
			int value;
			switch (formulaRowIndexSource)
			{
			case FormulaRowIndexSource.VisibleIndex:
				num = x.VisibleIndex;
				value = y.VisibleIndex;
				break;
			case FormulaRowIndexSource.RowIndex:
				num = x.Index;
				value = y.Index;
				break;
			case FormulaRowIndexSource.ListIndex:
				num = x.ListIndex;
				value = y.ListIndex;
				break;
			case FormulaRowIndexSource.Default:
				num = x.VisibleIndex;
				value = y.VisibleIndex;
				break;
			default:
				num = x.VisibleIndex;
				value = y.VisibleIndex;
				break;
			}
			return num.CompareTo(value);
		}
	}

					gridRowComparer = new GridRowComparer();
				}
				return gridRowComparer;
			}
		}

		private Worksheet HiddenWorkSheet => hiddenWorkSheet;

		public HiddenWorksheetManager(MappingManager mappingManager, Workbook workbook)
		{
			this.mappingManager = mappingManager;
			this.workbook = workbook;
		}

		private HiddenWorksheetColumnInfo GetHiddenWorksheetColumnInfo(UltraGridColumn column)
		{
			if (ColumnToHiddenWorksheetInfo.TryGetValue(column, out var value))
			{
				return value;
			}
			value = BuildHiddenWorksheetColumnInfo(column);
			ColumnToHiddenWorksheetInfo[column] = value;
			return value;
		}

		private HiddenWorksheetColumnInfo BuildHiddenWorksheetColumnInfo(UltraGridColumn column)
		{
			if (hiddenWorkSheet == null)
			{
				string hiddenWorkSheetName = GetHiddenWorkSheetName();
				hiddenWorkSheet = workbook.Worksheets.Add(hiddenWorkSheetName);
				HiddenWorkSheet.DisplayOptions.Visibility = WorksheetVisibility.Hidden;
			}
			UltraGridBand band = column.Band;
			GridRowComparer.FormulaRowIndexSource = band.FormulaRowIndexSourceResolved;
			UltraGridLayout layout = band.Layout;
			lastUsedColumn++;
			HiddenWorksheetColumnInfo hiddenWorksheetColumnInfo = new HiddenWorksheetColumnInfo();
			List<UltraGridRow> list = new List<UltraGridRow>();
			int num = 0;
			UltraGridRow ultraGridRow = null;
			foreach (UltraGridRow item in layout.Rows.GetRowEnumerator(GridRowType.DataRow, band, band))
			{
				UltraGridRow parentRow = item.ParentRow;
				if (num != 0 && ultraGridRow != parentRow)
				{
					if (list != null && list.Count > 0)
					{
						list.Sort(GridRowComparer);
						BuildHiddenWorksheetColumnInfo(column, hiddenWorksheetColumnInfo, list, num);
					}
					list = new List<UltraGridRow>();
				}
				if (mappingManager.WasRowExported(item))
				{
					list.Add(item);
				}
				ultraGridRow = parentRow;
			}
			if (list != null && list.Count > 0)
			{
				list.Sort(GridRowComparer);
				BuildHiddenWorksheetColumnInfo(column, hiddenWorksheetColumnInfo, list, num);
			}
			return hiddenWorksheetColumnInfo;
		}

		private int BuildHiddenWorksheetColumnInfo(UltraGridColumn column, HiddenWorksheetColumnInfo hiddenWorksheetColumnInfo, List<UltraGridRow> rows, int startRow)
		{
			int num = startRow + rows.Count;
			for (int i = startRow; i < num; i++)
			{
				UltraGridRow ultraGridRow = rows[i - startRow];
				WorksheetCellInfo worksheetCellInfo = new WorksheetCellInfo(hiddenWorkSheet.Rows[i], lastUsedColumn);
				if (!column.Hidden)
				{
					UltraGridCell cell = ultraGridRow.Cells[column];
					WorksheetCellInfo worksheetCellInfo2 = mappingManager.GetWorksheetCellInfo(cell);
					string formulaString = string.Format("=IF(IsBlank({0}), \"\", {0})", worksheetCellInfo2.GetCellAddressString(includeWorksheetName: true));
					worksheetCellInfo.ApplyFormula(formulaString);
					hiddenWorksheetColumnInfo.MapWorkSheetCell(worksheetCellInfo2, worksheetCellInfo);
					continue;
				}
				CellInfo cellInfo = new CellInfo(column, ultraGridRow);
				if (column.Formula != null && column.Formula.Length > 0)
				{
					mappingManager.FormulaTargetCells.Add(cellInfo);
				}
				worksheetCellInfo.SetCellValue(ultraGridRow.GetExportValue(column));
				hiddenWorksheetColumnInfo.MapWorkSheetCell(cellInfo, worksheetCellInfo);
			}
			return num;
		}

		private string GetHiddenWorkSheetName()
		{
			string text = "Calc_Helper";
			int num = 0;
			while (workbook.Worksheets.Exists(text))
			{
				num++;
				text = string.Format("{0}_{1}", "Calc_Helper", num.ToString());
			}
			return text;
		}

		internal WorksheetCellInfo GetHiddenWorksheetCellInfo(CellInfo cellInfo)
		{
			HiddenWorksheetColumnInfo hiddenWorksheetColumnInfo = GetHiddenWorksheetColumnInfo(cellInfo.GridColumn);
			return hiddenWorksheetColumnInfo.GetHiddenWorksheetCellInfo(cellInfo);
		}

		public WorksheetCellRangeInfo GetHiddenWorksheetCells(UltraGridColumn column, List<WorksheetCellInfo> realworkSheetCellInfos)
		{
			_ = realworkSheetCellInfos[0].Worksheet;
			HiddenWorksheetColumnInfo hiddenWorksheetColumnInfo = GetHiddenWorksheetColumnInfo(column);
			WorksheetCellRangeInfo worksheetCellRangeInfo = new WorksheetCellRangeInfo(realworkSheetCellInfos.Count);
			foreach (WorksheetCellInfo realworkSheetCellInfo in realworkSheetCellInfos)
			{
				WorksheetCellInfo hiddenWorksheetCellInfo = hiddenWorksheetColumnInfo.GetHiddenWorksheetCellInfo(realworkSheetCellInfo);
				worksheetCellRangeInfo.AddWorksheetCellInfo(hiddenWorksheetCellInfo);
			}
			return worksheetCellRangeInfo;
		}

	internal class ArgumentInfo
	{
		private string text;

		private UltraCalcOperatorFunction? functionOperator;

		private int? orderOfPrecedence;

		public UltraCalcOperatorFunction? FunctionOperator => functionOperator;

		public int OrderOfPrecedence
		{
			get
			{
				if (!orderOfPrecedence.HasValue)
				{
					orderOfPrecedence = GetOrderOfPrecedence(FunctionOperator);
				}
				return orderOfPrecedence.Value;
			}

		public string Text => text;

		public ArgumentInfo(string text, UltraCalcOperatorFunction? functionOperator)
		{
			this.text = text;
			this.functionOperator = functionOperator;
		}

		public string GetArgumentText(UltraCalcOperatorFunction? nextFunctionOperator)
		{
			return GetArgumentText(nextFunctionOperator, 1);
		}

		public string GetArgumentText(UltraCalcOperatorFunction? nextFunctionOperator, int argumentIndex)
		{
			string text = Text;
			if (text == "DBNULL()" || text == "NULL()")
			{
				switch (nextFunctionOperator)
				{
				case UltraCalcOperatorFunction.Add:
				case UltraCalcOperatorFunction.Subtract:
				case UltraCalcOperatorFunction.Multiply:
				case UltraCalcOperatorFunction.Divide:
				case UltraCalcOperatorFunction.Equal:
				case UltraCalcOperatorFunction.NotEqual:
				case UltraCalcOperatorFunction.GreaterThanOrEqual:
				case UltraCalcOperatorFunction.GreaterThan:
				case UltraCalcOperatorFunction.LessThanOrEqual:
				case UltraCalcOperatorFunction.LessThan:
				case UltraCalcOperatorFunction.Exponent:
				case UltraCalcOperatorFunction.Percent:
				case UltraCalcOperatorFunction.UnaryMinus:
				case UltraCalcOperatorFunction.UnaryPlus:
					text = "0";
					break;
				case UltraCalcOperatorFunction.Concatenate:
					text = "\"\"";
					break;
				default:
					return string.Empty;
				}
			}
			if (HasLowerOrderOfPrecedence(nextFunctionOperator, argumentIndex))
			{
				return $"({text})";
			}
			return text;
		}

		public static int GetOrderOfPrecedence(UltraCalcOperatorFunction? functionOperator)
		{
			switch (functionOperator)
			{
			case UltraCalcOperatorFunction.Percent:
			case UltraCalcOperatorFunction.UnaryMinus:
			case UltraCalcOperatorFunction.UnaryPlus:
				return 6;
			case UltraCalcOperatorFunction.Exponent:
				return 5;
			case UltraCalcOperatorFunction.Multiply:
			case UltraCalcOperatorFunction.Divide:
				return 4;
			case UltraCalcOperatorFunction.Add:
			case UltraCalcOperatorFunction.Subtract:
				return 3;
			case UltraCalcOperatorFunction.Concatenate:
				return 2;
			case UltraCalcOperatorFunction.Equal:
			case UltraCalcOperatorFunction.NotEqual:
			case UltraCalcOperatorFunction.GreaterThanOrEqual:
			case UltraCalcOperatorFunction.GreaterThan:
			case UltraCalcOperatorFunction.LessThanOrEqual:
			case UltraCalcOperatorFunction.LessThan:
				return 1;
			default:
				return int.MaxValue;
			}
		}

		public bool HasLowerOrderOfPrecedence(UltraCalcOperatorFunction? operatorFunction, int argumentIndex)
		{
			int num = GetOrderOfPrecedence(operatorFunction);
			if (OrderOfPrecedence < num)
			{
				return true;
			}
			if (argumentIndex == 2 && OrderOfPrecedence == num)
			{
				return true;
			}
			return false;
		}

		public override string ToString()
		{
			return Text;
		}
	}
	internal struct WorksheetCellInfo
	{
		public static readonly WorksheetCellInfo Empty = default(WorksheetCellInfo);

		private readonly short columnIndex;

		private readonly WorksheetRow worksheetRow;

		public short ColumnIndex => columnIndex;

		public WorksheetRow WorksheetRow => worksheetRow;

		public int RowIndex
		{
			get
			{
				if (worksheetRow == null)
				{
					return -1;
				}
				return worksheetRow.Index;
			}
		}

		public Worksheet Worksheet
		{
			get
			{
				if (worksheetRow == null)
				{
					return null;
				}
				return worksheetRow.Worksheet;
			}
		}

		public bool IsEmpty => worksheetRow == null;

		public WorksheetCellInfo(WorksheetRow worksheetRow, short columnIndex)
		{
			this.columnIndex = columnIndex;
			this.worksheetRow = worksheetRow;
		}

		public WorksheetCellInfo(WorksheetCell workSheetCell)
			: this(workSheetCell.Worksheet.Rows[workSheetCell.RowIndex], (short)workSheetCell.ColumnIndex)
		{
		}

		public void ApplyFormula(string formulaString)
		{
			if (!string.IsNullOrEmpty(formulaString))
			{
				Formula formula = Formula.Parse(formulaString, CellReferenceMode.A1, CultureInfo.InvariantCulture);
				formula.ApplyTo(worksheetRow.Cells[columnIndex]);
			}
		}

		internal string GetCellAddressString(bool includeWorksheetName)
		{
			return WorksheetCell.GetCellAddressString(worksheetRow, columnIndex, CellReferenceMode.A1, includeWorksheetName);
		}

		internal string GetCellAddressString(bool includeWorksheetName, bool useRelativeColumn, bool useRelativeRow)
		{
			return WorksheetCell.GetCellAddressString(worksheetRow, columnIndex, CellReferenceMode.A1, includeWorksheetName, useRelativeColumn, useRelativeRow);
		}

		public void SetCellValue(object value)
		{
			worksheetRow.SetCellValue(columnIndex, value);
		}

		public override bool Equals(object obj)
		{
			if (!(obj is WorksheetCellInfo worksheetCellInfo))
			{
				return false;
			}
			if (columnIndex == worksheetCellInfo.columnIndex)
			{
				return worksheetRow == worksheetCellInfo.worksheetRow;
			}
			return false;
		}

		public override int GetHashCode()
		{
			int num = ColumnIndex.GetHashCode();
			if (worksheetRow != null)
			{
				num |= worksheetRow.GetHashCode() << 16;
			}
			return num;
		}
	}

		public UltraGridExcelExporter()
		{
			try
			{
				license = LicenseManager.Validate(typeof(UltraGridExcelExporter), this) as UltraLicense;
			}
			catch (FileNotFoundException)
			{
			}
			Graphics cachedGraphics = DrawUtility.GetCachedGraphics((Control)null);
			try
			{
				twipsInPixelY = 1440f / cachedGraphics.DpiY;
				IntPtr hdc = cachedGraphics.GetHdc();
				TEXTMETRIC textmetric;
				try
				{
					UnsafeGetTextMetrics(hdc, out textmetric);
				}
				catch
				{
					textmetric = new TEXTMETRIC
					{
						tmAveCharWidth = 7
					};
				}
				cachedGraphics.ReleaseHdc(hdc);
				if (textmetric.tmAveCharWidth > 0)
				{
					columnWidthUnitsInPixelX = 256f / (float)textmetric.tmAveCharWidth;
				}
				else
				{
					columnWidthUnitsInPixelX = 36.57143f;
				}
			}
			finally
			{
				DrawUtility.ReleaseCachedGraphics(cachedGraphics);
			}
		}

		private void ExportHelper(UltraGrid grid, Worksheet worksheet, int startRow, int startColumn, bool asynch)
		{
			if (grid == null)
			{
				throw new ArgumentNullException("grid");
			}
			if (worksheet == null)
			{
				throw new ArgumentNullException("worksheet");
			}
			if (currentWorkbook != null)
			{
				throw new InvalidOperationException("An export operation is currently in progress. A new export operation cannot begin until the current operation completes.");
			}
			if (startRow < 0 || startRow >= Workbook.GetMaxRowCount(worksheet.Workbook.CurrentFormat))
			{
				throw new ArgumentOutOfRangeException("row", startRow, Infragistics.Shared.SR.GetString("LER_ArgumentOutOfRangeException_1"));
			}
			if (startColumn < 0 || startColumn >= Workbook.GetMaxColumnCount(worksheet.Workbook.CurrentFormat))
			{
				throw new ArgumentOutOfRangeException("column", startColumn, Infragistics.Shared.SR.GetString("LER_ArgumentOutOfRangeException_2"));
			}
			UltraGridExporterHelper exporter = new UltraGridExporterHelper(this, worksheet, startRow, startColumn);
			currentWorkbook = worksheet.Workbook;
			this.asynch = asynch;
			if (asynch)
			{
				grid.ExportAsync(exporter);
			}
			else
			{
				grid.Export(exporter);
			}
		}

		internal void BeginExportInternal(UltraGridExporterHelper exportHelper, UltraGridLayout exportLayout, RowsCollection rows)
		{
			GridExportCache.ClearGridBagLayoutManagerCache();
			currentExportFormulas = ExportFormulas;
			currentExportValueLists = ExportValueLists;
			exportHelper.columnExcelFormats = new Hashtable(64);
			exportHelper.summaryExcelFormats = new Dictionary<SummarySettings, string>();
			exportHelper.lastBand = exportLayout.Bands[0];
			OnExportStarted(new ExportStartedEventArgs(exportHelper, exportLayout, rows));
			OnBeginExport(new BeginExportEventArgs(exportHelper, exportLayout, rows));
			if (exportLayout.ViewStyleBand == ViewStyleBand.Horizontal)
			{
				exportLayout.ViewStyleBand = ViewStyleBand.Vertical;
			}
			foreach (UltraGridBand band in exportLayout.Bands)
			{
				foreach (UltraGridColumn column in band.Columns)
				{
					column.Header.ExclusiveColScrollRegion = null;
				}
			}
			GridExportCache.Layout = exportLayout;
			GridExportCache.BuildReportInfoCache(exportLayout);
			if (rows.VisibleRowCount == 0 && exportLayout.Bands[0].GetFirstVisibleCol(null, includeNonActivateable: true) != null)
			{
				SetCurrentGridObjectSpan(exportHelper, exportLayout.Bands[0], null, GridExportCache.PlaceholderType.ColumnHeaders);
				ProcessHeaderRows(exportHelper, null, exportLayout.Bands[0]);
				if (exportLayout.Bands[0].Summaries.Count > 0)
				{
					SetCurrentGridObjectSpan(exportHelper, exportLayout.Bands[0], null, GridExportCache.PlaceholderType.Summaries);
					ProcessSummaryRows(exportHelper, null, exportLayout.Bands[0], rows);
				}
			}
		}

		internal void ProcessGridRowInternal(UltraGridExporterHelper exportHelper, UltraGridRow row, ProcessRowParams processRowParams)
		{
			AdjustPositionForRowsCollection(exportHelper, row);
			exportHelper.lastBand = row.Band;
			if (!GridExportCache.PlacedHeadersOnTop)
			{
				UltraGridLayout layout = GridExportCache.Layout;
				if (layout != null)
				{
					BandsCollection bands = layout.Bands;
					foreach (UltraGridBand item in bands)
					{
						if (!item.HiddenResolved && item.HeaderPlacementResolved == HeaderPlacement.FixedOnTop && item.GetFirstVisibleCol(null, includeNonActivateable: true) != null)
						{
							SetCurrentGridObjectSpan(exportHelper, item, null, GridExportCache.PlaceholderType.ColumnHeaders);
							exportHelper.currentPos.X += GetIndentationLevel(item);
							ProcessHeaderRows(exportHelper, null, item);
						}
					}
					GridExportCache.PlacedHeadersOnTop = true;
				}
				GridExportCache.PlacedHeadersOnTop = true;
			}
			HeaderPlacement headerPlacement = HeaderPlacement.Default;
			if (row != null)
			{
				UltraGridBand band = row.Band;
				if (band != null)
				{
					headerPlacement = band.HeaderPlacementResolved;
				}
			}
			bool flag = false;
			if (GridExportCache.HasVisibleGroupsOrColumnHeadersOnTop(row.Band))
			{
				UltraGridRow sibling = row.GetSibling(SiblingRow.Previous, spanBands: false, excludeCardBands: false, includeSpecialRows: true);
				switch (headerPlacement)
				{
				case HeaderPlacement.RepeatOnBreak:
				{
					bool flag2 = sibling == null;
					if (!flag2 && row.IsDataRow && !row.HiddenResolved)
					{
						UltraGridRow ultraGridRow = sibling;
						if (sibling.HiddenResolved)
						{
							while (ultraGridRow != null && sibling.HiddenResolved)
							{
								ultraGridRow = ultraGridRow.GetSibling(SiblingRow.Previous, spanBands: false, excludeCardBands: false, includeSpecialRows: true);
							}
						}
						if (ultraGridRow != null && HasChildHelper(sibling))
						{
							flag2 = true;
						}
					}
					flag = flag2;
					break;
				}
				case HeaderPlacement.OncePerGroupedRowIsland:
					if (sibling == null && !(row.ParentRow is UltraGridGroupByRow))
					{
						flag = true;
					}
					break;
				case HeaderPlacement.OncePerRowIsland:
					if (sibling == null)
					{
						flag = true;
					}
					break;
				case HeaderPlacement.FixedOnTop:
				{
					if (GridExportCache.PlacedHeadersOnTop)
					{
						break;
					}
					UltraGridLayout layout2 = GridExportCache.Layout;
					if (layout2 == null)
					{
						break;
					}
					BandsCollection bands2 = layout2.Bands;
					foreach (UltraGridBand item2 in bands2)
					{
						if (!item2.HiddenResolved)
						{
							SetCurrentGridObjectSpan(exportHelper, row.Band, null, GridExportCache.PlaceholderType.ColumnHeaders);
							exportHelper.currentPos.X = GetIndentationLevel(item2);
							ProcessHeaderRows(exportHelper, null, item2);
						}
					}
					GridExportCache.PlacedHeadersOnTop = true;
					break;
				}
				case HeaderPlacement.Default:
					if (sibling == null)
					{
						flag = true;
					}
					break;
				default:
					if (sibling == null)
					{
						flag = true;
					}
					break;
				}
			}
			if (!row.IsGroupByRow && flag && row.ParentCollection.VisibleRowCount > 0)
			{
				if (row != null && row.ParentCollection.Count > 0 && row.IsSummaryRow)
				{
					UltraGridGroupByRow row2 = row.ParentCollection[0] as UltraGridGroupByRow;
					int groupByDescendantLevelCount = GetGroupByDescendantLevelCount(row2);
					exportHelper.currentPos.X += groupByDescendantLevelCount;
				}
				SetCurrentGridObjectSpan(exportHelper, row.Band, null, GridExportCache.PlaceholderType.ColumnHeaders);
				ProcessHeaderRows(exportHelper, row, row.Band);
			}
			if (row is UltraGridGroupByRow)
			{
				UltraGridBand band2 = row.Band;
				UltraGridLayout layout3 = GridExportCache.Layout;
				if (layout3 != null)
				{
					UltraGridBand ultraGridBand = layout3.Bands[0];
					if (headerPlacement == HeaderPlacement.OncePerGroupedRowIsland && band2.GetFirstVisibleCol(null, includeNonActivateable: true) != null && (!GridExportCache.BandZeroHeaderExportedForGroupIsland || band2 != ultraGridBand))
					{
						SetCurrentGridObjectSpan(exportHelper, band2, null, GridExportCache.PlaceholderType.ColumnHeaders);
						exportHelper.currentPos.X++;
						ProcessHeaderRows(exportHelper, null, band2);
						GridExportCache.BandZeroHeaderExportedForGroupIsland = true;
					}
				}
				exportHelper.extentOfCurrentGridObject.Height = 1;
				exportHelper.extentOfCurrentGridObject.Width = 1;
			}
			else
			{
				SetCurrentGridObjectSpan(exportHelper, row.Band, row, (!row.IsSummaryRow) ? GridExportCache.PlaceholderType.Values : GridExportCache.PlaceholderType.Summaries);
			}
			if (row.IsSummaryRow)
			{
				ProcessSummaryRows(exportHelper, row, row.Band, row.ParentCollection);
				return;
			}
			ExcelExportInitializeRowEventArgs e = new ExcelExportInitializeRowEventArgs(exportHelper, processRowParams, row);
			if (row.HiddenResolved)
			{
				e.SkipRow = true;
				e.SkipDescendants = true;
			}
			OnInitializeRow(e);
			if (exportHelper.currentPos.Y >= Workbook.GetMaxRowCount(exportHelper.CurrentWorksheet.Workbook.CurrentFormat))
			{
				if (fileLimitBehaviour != FileLimitBehaviour.TruncateData)
				{
					throw new Exception(Infragistics.Shared.SR.GetString("LER_MaxExcelRowCountLimit"));
				}
			}
			else if (!e.TerminateExport)
			{
				if (!e.SkipRow)
				{
					ProcessSingleRow(exportHelper, row);
				}
				else
				{
					processRowParams.SkipRow = true;
				}
			}
		}

		internal void EndExportInternal(UltraGridExporterHelper exportHelper, bool canceled)
		{
			if (!canceled)
			{
				PositionWorksheetImages();
				ProcessFormulas();
			}
			valueListManager = null;
			OnExportEnding(new ExportEndingEventArgs(exportHelper, canceled));
			OnEndExport(new EndExportEventArgs(exportHelper, canceled));
			bool flag = true;
			if (currentWorkbook != null && currentFileName != null)
			{
				if (asynch)
				{
					try
					{
						currentWorkbook.Save(currentFileName);
					}
					catch (Exception ex)
					{
						AsynchronousExportErrorEventArgs e = new AsynchronousExportErrorEventArgs(currentWorkbook, currentFileName, ex);
						OnAsynchronousExportError(e);
						if (e.DisplayErrorMessage)
						{
							string text = Infragistics.Shared.SR.GetString("AsynchronousExportError_Dialog_Title");
							StringBuilder stringBuilder = new StringBuilder();
							stringBuilder.AppendLine(ex.Message);
							stringBuilder.AppendLine();
							stringBuilder.AppendLine(ex.StackTrace);
							UltraMessageBoxManager.Show((IWin32Window)GridExportCache.Layout.Grid, stringBuilder.ToString(), text, MessageBoxButtons.OK, MessageBoxIcon.Hand);
						}

					}
					finally
					{
						CleanUp();
					}
				}
			}
			CleanUp();

				OnExportEnded(new ExportEndedEventArgs(exportHelper, canceled));
			}
		}

		internal bool HasChildHelper(UltraGridRow row)
		{
			if (row.ChildBands != null)
			{
				foreach (UltraGridChildBand item in (IEnumerable)row.ChildBands)
				{
					if (!item.Band.HiddenResolved && item.Rows.VisibleRowCount > 0)
					{
						return true;
					}
				}
			}
			return false;
		}

		internal void AdjustPositionForRowsCollection(UltraGridExporterHelper exportHelper, UltraGridRow row)
		{
			int hierarchyLevel = GetHierarchyLevel(row);
			exportHelper.CurrentIndentationLevel = GetIndentationLevel(row);
			if (hierarchyLevel != exportHelper.HierarchyLevel)
			{
				exportHelper.currentPos.X = exportHelper.gridStartPos.X + exportHelper.CurrentIndentationLevel;
				while (exportHelper.HierarchyLevel != hierarchyLevel)
				{
					if (exportHelper.HierarchyLevel < hierarchyLevel)
					{
						exportHelper.HierarchyLevel++;
					}
					else
					{
						exportHelper.HierarchyLevel--;
					}
					SetCollapsedRow(exportHelper.CurrentWorksheet, exportHelper.currentPos.Y, exportHelper.HierarchyLevel);
					if (BandSpacing == BandSpacing.SingleRow)
					{
						IncreaseCurrentRow(exportHelper, 1);
					}
				}
			}
			else if (row.Band != exportHelper.lastBand)
			{
				SetCollapsedRow(exportHelper.CurrentWorksheet, exportHelper.currentPos.Y, exportHelper.HierarchyLevel);
				if (BandSpacing == BandSpacing.SingleRow)
				{
					IncreaseCurrentRow(exportHelper, 1);
				}
			}
		}

		private int GetMaxHierarchyLevel(UltraGridLayout layout)
		{
			int num = 0;
			foreach (UltraGridBand band in layout.Bands)
			{
				num += 1 + band.Columns.Count;
			}
			return num;
		}

		private int GetHierarchyLevel(UltraGridRow row)
		{
			if (row.ParentRow == null)
			{
				return 0;
			}
			return GetHierarchyLevel(row.ParentRow) + 1;
		}

		private int GetIndentationLevel(UltraGridRow row)
		{
			int num = 0;
			while (row != null)
			{
				UltraGridRow parentRow = row.ParentRow;
				if (parentRow != null)
				{
					if (row.IsDataRow || row.IsSummaryRow)
					{
						int indentation = row.Band.Indentation;
						num = ((!parentRow.IsGroupByRow || indentation > 0) ? (num + ((parentRow.Band.Indentation != 0) ? 1 : 0)) : (num + ((parentRow.Band.IndentationGroupByRowResolved != 0) ? 1 : 0)));
					}
					else if (row.IsGroupByRow)
					{
						num += ((row.Band.IndentationGroupByRowResolved != 0) ? 1 : 0);
					}
				}
				row = parentRow;
			}
			return num;
		}

		private int GetIndentationLevel(UltraGridBand band)
		{
			int num = 0;
			while (band != null)
			{
				if (band.IndentationGroupByRow != 0)
				{
					foreach (UltraGridColumn item in (IEnumerable)band.SortedColumns)
					{
						if (item.IsGroupByColumn)
						{
							num++;
						}
					}
				}
				UltraGridBand parentBand = band.ParentBand;
				if (parentBand != null && band.Indentation != 0)
				{
					num++;
				}
				band = parentBand;
			}
			return num;
		}

		internal void IncreaseCurrentRow(UltraGridExporterHelper exportHelper, int amount)
		{
			exportHelper.currentPos.Y += amount;
			exportHelper.currentPos.X = exportHelper.gridStartPos.X + exportHelper.CurrentIndentationLevel;
		}

		internal void SetCollapsedRow(Worksheet worksheet, int row, int hierarchyLevel)
		{
			if (row >= Workbook.GetMaxRowCount(worksheet.Workbook.CurrentFormat))
			{
				if (fileLimitBehaviour != FileLimitBehaviour.TruncateData)
				{
					throw new Exception(Infragistics.Shared.SR.GetString("LER_MaxExcelRowCountLimit"));
				}
				return;
			}
			WorksheetRow worksheetRow = worksheet.Rows[row];
			if (outliningStyle != OutliningStyle.None)
			{
				if (outliningStyle == OutliningStyle.IgnoreDepthOverflow && hierarchyLevel > 7)
				{
					hierarchyLevel = 7;
				}
				worksheetRow.OutlineLevel = hierarchyLevel;
				if (hierarchyLevel != 0)
				{
					worksheetRow.Hidden = true;
				}
				else
				{
					worksheetRow.Hidden = false;
				}
			}
		}

		private void ProcessHeaderRows(UltraGridExporterHelper exportHelper, UltraGridRow row, UltraGridBand band)
		{
			if (band.HeaderVisible)
			{
				ExportBandHeaderRow(exportHelper, row, band);
			}
			if (band.UseRowLayout && band.RowLayoutLabelStyle == RowLayoutLabelStyle.WithCellData)
			{
				return;
			}
			if (!GridExportCache.HasVisibleGroupsOrColumnHeadersOnTop(band))
			{
				if (row == null)
				{
					exportHelper.currentPos.X = exportHelper.gridStartPos.X;
				}
				return;
			}
			HeaderRowExportingEventArgs e = new HeaderRowExportingEventArgs(exportHelper, band, row, HeaderTypes.ColumnHeader);
			OnHeaderRowExporting(e);
			if (!e.Cancel)
			{
				exportHelper.extentOfCurrentGridObject.Location = exportHelper.currentPos;
				ExportColumnHeaderCells(exportHelper, row, band, setRowHeights: true);
				SetCollapsedRows(exportHelper.CurrentWorksheet, exportHelper.extentOfCurrentGridObject.Y, exportHelper.extentOfCurrentGridObject.Bottom - 1, exportHelper.HierarchyLevel);
				exportHelper.currentPos = exportHelper.extentOfCurrentGridObject.Location;
				IncreaseCurrentRow(exportHelper, exportHelper.extentOfCurrentGridObject.Height);
				HeaderRowExportedEventArgs e2 = new HeaderRowExportedEventArgs(exportHelper, band, row, HeaderTypes.ColumnHeader);
				OnHeaderRowExported(e2);
			}
		}

		private void ExportBandHeaderRow(UltraGridExporterHelper exportHelper, UltraGridRow row, UltraGridBand band)
		{
			HeaderRowExportingEventArgs e = new HeaderRowExportingEventArgs(exportHelper, band, row, HeaderTypes.BandHeader);
			OnHeaderRowExporting(e);
			if (e.Cancel)
			{
				return;
			}
			HeaderCellExportingEventArgs e2 = new HeaderCellExportingEventArgs(exportHelper, band.Header, row, HeaderTypes.BandHeader);
			OnHeaderCellExporting(e2);
			if (exportHelper.currentPos.Y >= Workbook.GetMaxRowCount(exportHelper.CurrentWorksheet.Workbook.CurrentFormat))
			{
				if (fileLimitBehaviour != FileLimitBehaviour.TruncateData)
				{
					throw new Exception(Infragistics.Shared.SR.GetString("LER_MaxExcelRowCountLimit"));
				}
				e2.Cancel = true;
			}
			if (exportHelper.currentPos.X + exportHelper.extentOfCurrentGridObject.Width - 1 >= Workbook.GetMaxColumnCount(exportHelper.CurrentWorksheet.Workbook.CurrentFormat))
			{
				if (fileLimitBehaviour != FileLimitBehaviour.TruncateData)
				{
					throw new Exception(Infragistics.Shared.SR.GetString("LER_MaxExcelColumnCountLimit"));
				}
				e2.Cancel = true;
			}
			if (!e2.Cancel)
			{
				WorksheetCell worksheetCell = exportHelper.CurrentWorksheet.Rows[exportHelper.currentPos.Y].Cells[exportHelper.currentPos.X];
				worksheetCell.Value = band.Header.Caption;
				if (exportHelper.extentOfCurrentGridObject.Width > 0)
				{
					WorksheetMergedCellsRegion worksheetMergedCellsRegion = exportHelper.CurrentWorksheet.MergedCellsRegions.Add(exportHelper.currentPos.Y, exportHelper.currentPos.X, exportHelper.currentPos.Y, exportHelper.currentPos.X + exportHelper.extentOfCurrentGridObject.Width - 1);
					worksheetMergedCellsRegion.CellFormat.SetFormatting(ResolvedHeader(exportHelper, band.Header));
				}
				HeaderCellExportedEventArgs e3 = new HeaderCellExportedEventArgs(exportHelper, band.Header, row, HeaderTypes.BandHeader);
				OnHeaderCellExported(e3);
			}
			SetCollapsedRow(exportHelper.CurrentWorksheet, exportHelper.currentPos.Y, exportHelper.HierarchyLevel);
			int x = exportHelper.currentPos.X;
			IncreaseCurrentRow(exportHelper, 1);
			exportHelper.currentPos.X = x;
			HeaderRowExportedEventArgs e4 = new HeaderRowExportedEventArgs(exportHelper, band, row, HeaderTypes.BandHeader);
			OnHeaderRowExported(e4);
		}

		private void ExportColumnHeaderCells(UltraGridExporterHelper exportHelper, UltraGridRow row, UltraGridBand band, bool setRowHeights)
		{
			GridExportCache.DocumentBandInfo documentBandInfo = GridExportCache.GetDocumentBandInfo(band, GridExportCache.PlaceholderType.ColumnHeaders);
			if (band.GroupHeadersVisible && documentBandInfo.HasGroups && !band.UseRowLayout)
			{
				SetColumnWidthsAndRowHeights(exportHelper, row, null, new int[1] { documentBandInfo.GroupList[0].Header.Height });
				int num = 0;
				foreach (UltraGridGroup group in documentBandInfo.GroupList)
				{
					int num2 = 0;
					foreach (GridExportCache.DocumentHeaderInfo item in documentBandInfo)
					{
						if (item.DocumentHeaderInfoType == GridExportCache.DocumentHeaderInfoType.Header)
						{
							UltraGridColumn column = item.Column;
							if (column != null && column.Group == group)
							{
								num2 = Math.Max(num2, item.OriginX + item.SpanX);
							}
						}
					}
					int spanX = num2 - num;
					GridExportCache.DocumentHeaderInfo documentColumnInfo = new GridExportCache.DocumentHeaderInfo(group.Header, num, 0, spanX, 1, GridExportCache.DocumentHeaderInfoType.Header, null);
					ExportHeaderCell(exportHelper, row, documentColumnInfo);

		private WorksheetCell SetRegionRelativeToOrigin(UltraGridExporterHelper exportHelper, Rectangle rect, object val, IWorksheetCellFormat cellFormat)

		private bool ColumnSummariesExist(UltraGridSummaryRow summaryRowContext, SummaryValuesCollection summaries, GridExportCache.DocumentBandInfo documentBandInfo, int level, Rectangle? intersectingRect)
		{
			foreach (GridExportCache.DocumentHeaderInfo item in documentBandInfo)
			{
				if (intersectingRect.HasValue && !new Rectangle(item.OriginX, item.OriginY, item.OriginX + item.SpanX, item.OriginY + item.SpanY).IntersectsWith(intersectingRect.Value))
				{
					continue;
				}
				switch (item.DocumentHeaderInfoType)
				{
				case GridExportCache.DocumentHeaderInfoType.Header:
				case GridExportCache.DocumentHeaderInfoType.Summary:
				{
					UltraGridColumn column = item.Column;
					if (column != null)
					{
						SummaryValue summaryValueFromPosition = summaries.GetSummaryValueFromPosition(SummaryPosition.UseSummaryPositionColumn, column, level, summaryRowContext);
						if (summaryValueFromPosition != null)
						{
							return true;
						}
					}
					break;
				}
				}
			}
			return false;
		}

		}

		private void ProcessFormula<T>(T formulaTargetInfo, object context, string gridFormula, WorksheetCellInfo workSheetCellInfo) where T : IFormulaTargetInfo
		{
			if (MappingManager.GetIsFormulaCancelled(formulaTargetInfo))
			{
				return;
			}
			FormulaExportingEventArgs e = OnFormulaExporting(context, gridFormula, workSheetCellInfo, FormulaExportAction.ExportFormula);
			switch (e.Action)
			{
			case FormulaExportAction.ExportFormula:
			{
				string text = null;
				try
				{
					text = ConvertGridFormulaToExcel(formulaTargetInfo, workSheetCellInfo);
					if (text == null)
					{
						break;
					}
					workSheetCellInfo.ApplyFormula(text);
				}
				catch (Exception ex)
				{
					if (ex is FormulaExportException ex2)
					{
						text = ex2.InvalidFormula;
					}
					FormulaExportErrorEventArgs e2 = OnFormulaExportError(context, gridFormula, workSheetCellInfo, FormulaExportErrorAction.WriteErrorMessageToWorksheetCell, ex, text);
					text = e2.ExcelFormula;
					if (e2.ApplyInvalidFormulaIfPossible)
					{
						bool flag = true;
						try
						{
							workSheetCellInfo.ApplyFormula(text);
						}
						catch
						{
							flag = false;
						}
						if (flag)
						{
							OnFormulaExported(context, gridFormula, workSheetCellInfo);
							break;
						}
					}
					switch (e2.Action)
					{
					case FormulaExportErrorAction.WriteErrorMessageToWorksheetCell:
						workSheetCellInfo.SetCellValue(e2.ErrorText);
						break;
					case FormulaExportErrorAction.CancelAll:
						MappingManager.SetIsFormulaCancelled(formulaTargetInfo);
						break;
					case FormulaExportErrorAction.CancelOne:
						break;
					}
					break;
				}
				OnFormulaExported(context, gridFormula, workSheetCellInfo);
				break;
			}
			case FormulaExportAction.CancelAll:
				MappingManager.SetIsFormulaCancelled(formulaTargetInfo);
				break;
			case FormulaExportAction.CancelOne:
				break;
			}
		}

		private string ConvertGridFormulaToExcel(WorksheetCellInfo targetWorksheetCellInfo, IUltraCalcFormula formula, IUltraCalcReference targetReference)
		{
			Exception ex = null;
			IUltraCalcFormulaToken[] formulaTokens = MappingManager.GetFormulaTokens(formula);
			Stack<ArgumentInfo> stack = new Stack<ArgumentInfo>();
			IUltraCalcFormulaToken[] array = formulaTokens;
			ArgumentInfo item;
			foreach (IUltraCalcFormulaToken ultraCalcFormulaToken in array)
			{
				switch (ultraCalcFormulaToken.Type)
				{
				case UltraCalcFormulaTokenType.Value:
					if (!(ultraCalcFormulaToken is IUltraCalcValueToken { Value: var ultraCalcValue }))
					{
						return null;
					}
					if (ultraCalcValue.IsReference)
					{
						ultraCalcValue = new UltraCalcValue(targetReference.ResolveReference(ultraCalcValue.ToReference(), ResolveReferenceType.RightHandSide));
					}
					item = new ArgumentInfo(GetArgumentString(ultraCalcValue, targetWorksheetCellInfo), null);
					stack.Push(item);
					break;
				case UltraCalcFormulaTokenType.Function:
				{
					if (!(ultraCalcFormulaToken is IUltraCalcFunctionToken ultraCalcFunctionToken))
					{
						return null;
					}
					if (ex == null && !IsValidFunctionName(currentWorkbook, ultraCalcFunctionToken))
					{
						ex = new FormulaExportException(FormulaExportErrorType.UnrecognizedFunction, Infragistics.Shared.SR.GetString("FormulaError_UnrecognizedFunction", ultraCalcFunctionToken.FunctionName), null);
					}
					string text = BuildFunctionString(ultraCalcFunctionToken, stack, targetWorksheetCellInfo, out ex);
					item = new ArgumentInfo(text, ultraCalcFunctionToken.FunctionOperator);
					stack.Push(item);
					break;
				}
				default:
					return null;
				}
			}
			if (stack.Count != 1)
			{
				return null;
			}
			item = stack.Pop();
			string text2 = item.Text;
			if (text2 == null || text2.Length == 0 || text2 == "DBNULL()" || text2 == "NULL()")
			{
				return null;
			}
			string text3 = $"={item.Text}";
			if (ex == null)
			{
				return text3;
			}
			if (ex is FormulaExportException ex2)
			{
				ex2.InvalidFormula = text3;
			}
			throw ex;
		}

		private string BuildFunctionString(IUltraCalcFunctionToken functionToken, Stack<ArgumentInfo> argumentStack, WorksheetCellInfo targetWorksheetCellInfo, out Exception ex)
		{
			ex = null;
			int argumentCount = functionToken.ArgumentCount;
			if (argumentStack.Count < argumentCount)
			{
				return "#REF!";
			}
			string functionName = functionToken.FunctionName.ToUpper();
			UltraCalcOperatorFunction? functionOperator = functionToken.FunctionOperator;
			CoerceFunction(argumentStack, ref argumentCount, ref functionName, ref functionOperator);
			switch (functionOperator)
			{
			case UltraCalcOperatorFunction.Add:
			case UltraCalcOperatorFunction.Subtract:
			case UltraCalcOperatorFunction.Multiply:
			case UltraCalcOperatorFunction.Divide:
			case UltraCalcOperatorFunction.Equal:
			case UltraCalcOperatorFunction.NotEqual:
			case UltraCalcOperatorFunction.GreaterThanOrEqual:
			case UltraCalcOperatorFunction.GreaterThan:
			case UltraCalcOperatorFunction.LessThanOrEqual:
			case UltraCalcOperatorFunction.LessThan:
			case UltraCalcOperatorFunction.Concatenate:
			case UltraCalcOperatorFunction.Exponent:
			{
				ArgumentInfo argumentInfo2 = argumentStack.Pop();
				ArgumentInfo argumentInfo3 = argumentStack.Pop();
				string argumentText2 = argumentInfo2.GetArgumentText(functionOperator, 2);
				string argumentText3 = argumentInfo3.GetArgumentText(functionOperator, 1);
				return $"{argumentText3} {functionName} {argumentText2}";
			}
			case UltraCalcOperatorFunction.UnaryMinus:
			case UltraCalcOperatorFunction.UnaryPlus:
			{
				ArgumentInfo argumentInfo = argumentStack.Pop();
				string argumentText = argumentInfo.GetArgumentText(functionOperator);
				functionName = functionName.Substring(0, 1);
				return $"{functionName}{argumentText}";
			}
			case UltraCalcOperatorFunction.Percent:
			{
				ArgumentInfo argumentInfo = argumentStack.Pop();
				string argumentText = argumentInfo.GetArgumentText(functionOperator);
				return $"{argumentText}{functionName}";
			}
			default:
			{
				if (CoerceFunction(argumentStack, functionToken, out var functionString, out ex))
				{
					return functionString;
				}
				return BuildFunctionString(argumentStack, argumentCount, functionName, functionOperator);
			}
			}
		}

		private static string BuildFunctionString(Stack<ArgumentInfo> argumentStack, IUltraCalcFunctionToken functionToken)
		{
			int argumentCount = functionToken.ArgumentCount;
			string functionName = functionToken.FunctionName.ToUpper();
			UltraCalcOperatorFunction? functionOperator = functionToken.FunctionOperator;
			return BuildFunctionString(argumentStack, argumentCount, functionName, functionOperator);
		}

		private static string BuildFunctionString(Stack<ArgumentInfo> argumentStack, int argumentCount, string functionName, UltraCalcOperatorFunction? functionOperator)
		{
			string text = null;
			for (int i = 0; i < argumentCount; i++)
			{
				if (text != null)
				{
					text = $", {text}";
				}
				ArgumentInfo argumentInfo = argumentStack.Pop();
				string argumentText = argumentInfo.GetArgumentText(functionOperator);
				text = $"{argumentText}{text}";
			}
			return $"{functionName}({text})";
		}

		private string GetArgumentString(UltraCalcValue calcValue, WorksheetCellInfo targetWorksheetCellInfo)
		{
			if (calcValue.IsReference)
			{
				if (calcValue.IsError)
				{
					UltraCalcErrorValue ultraCalcErrorValue = calcValue.ToErrorValue();
					if (ultraCalcErrorValue.Code == UltraCalcErrorCode.Reference)
					{
						return "#REF!";
					}
				}
				IUltraCalcReference ultraCalcReference = calcValue.ToReference();
				if (ultraCalcReference.Context is UltraGridCell cell)
				{
					WorksheetCellInfo worksheetCellInfo = MappingManager.GetWorksheetCellInfo(cell);
					bool includeWorksheetName = worksheetCellInfo.Worksheet != targetWorksheetCellInfo.Worksheet;
					return worksheetCellInfo.GetCellAddressString(includeWorksheetName, useRelativeColumn: true, useRelativeRow: true);
				}
				if (ultraCalcReference.Context is UltraGridColumn)
				{
					if (!ultraCalcReference.IsEnumerable)
					{
						return null;
					}
					WorksheetCellRangeInfo worksheetCells = MappingManager.GetWorksheetCells(ultraCalcReference);
					WorksheetRegion worksheetRegion = worksheetCells.GetWorksheetRegion();
					if (worksheetRegion != null)
					{
						bool flag = worksheetCells.Worksheet != targetWorksheetCellInfo.Worksheet;
						return worksheetRegion.ToString(CellReferenceMode.A1, flag, useRelativeColumn: true, useRelativeRow: true);
					}
					return "#REF!";
				}
				if (ultraCalcReference.Context is SummaryValue summaryValue)
				{
					WorksheetCellInfo worksheetCellInfo2 = MappingManager.GetWorksheetCellInfo(summaryValue);
					bool includeWorksheetName2 = worksheetCellInfo2.Worksheet != targetWorksheetCellInfo.Worksheet;
					return worksheetCellInfo2.GetCellAddressString(includeWorksheetName2, useRelativeColumn: true, useRelativeRow: true);
				}
				return null;
			}
			if (calcValue.Value is string)
			{
				return $"\"{calcValue.ToString()}\"";
			}
			return calcValue.ToString();
		}

		private static void CoerceFunction(Stack<ArgumentInfo> argumentStack, ref int argumentCount, ref string functionName, ref UltraCalcOperatorFunction? functionOperator)
		{
			switch (functionName)
			{
			case "CEILING":
			case "FLOOR":
				if (argumentCount == 1)
				{
					argumentCount = 2;
					argumentStack.Push(new ArgumentInfo("1", null));
				}
				break;
			case "MID":
				if (argumentCount == 1)
				{
					argumentCount = 3;
					argumentStack.Push(new ArgumentInfo("1", null));
					argumentStack.Push(new ArgumentInfo("0", null));
				}
				else if (argumentCount == 2)
				{
					argumentCount = 3;
					argumentStack.Push(new ArgumentInfo("0", null));
				}
				break;
			case "ROUND":
				if (argumentCount == 3)
				{
					argumentCount = 2;
					argumentStack.Pop();
				}
				break;
			case "ERRORTYPE":
				functionName = "ERROR.TYPE";
				break;
			case "ISDBNULL":
			case "ISNULL":
				functionName = "ISBLANK";
				break;
			}
		}

		private bool CoerceFunction(Stack<ArgumentInfo> argumentStack, IUltraCalcFunctionToken functionToken, out string functionString, out Exception ex)
		{
			string text = functionToken.FunctionName.ToUpper();
			ex = null;
			switch (text)
			{
			case "DATEADD":
			{
				ArgumentInfo argumentInfo4 = argumentStack.Pop();
				ArgumentInfo argumentInfo5 = argumentStack.Pop();
				ArgumentInfo argumentInfo3 = argumentStack.Pop();
				string text2 = argumentInfo3.Text.Replace("\"", string.Empty).Replace("'", string.Empty).ToLower();
				string[] array = new string[7]
				{
					argumentInfo4.ToString(),
					string.Empty,
					string.Empty,
					string.Empty,
					string.Empty,
					string.Empty,
					string.Empty
				};
				switch (text2)
				{
				case "yyyy":
					array[1] = $" + {argumentInfo5.Text}";
					break;
				case "m":
					array[2] = $" + {argumentInfo5.Text}";
					break;
				case "q":
					array[2] = $" + ({argumentInfo5.Text} * 3)";
					break;
				case "d":
				case "y":
					array[3] = $" + {argumentInfo5.Text}";
					break;
				case "w":
				case "ww":
					array[3] = $" + ({argumentInfo5.Text} * 7)";
					break;
				case "h":
					array[4] = $" + {argumentInfo5.Text}";
					break;
				case "n":
					array[5] = $" + {argumentInfo5.Text}";
					break;
				case "s":
					array[6] = $" + {argumentInfo5.Text}";
					break;
				default:
					functionString = null;
					return true;
				}
				functionString = string.Format("DATE(YEAR({0}){1},MONTH({0}){2},DAY({0}){3})+TIME(HOUR({0}){4},MINUTE({0}){5},SECOND({0}){6})", array);
				return true;
			}
			case "DATEDIFF":
			{
				ArgumentInfo argumentInfo = argumentStack.Pop();
				ArgumentInfo argumentInfo2 = argumentStack.Pop();
				ArgumentInfo argumentInfo3 = argumentStack.Pop();
				string text2 = argumentInfo3.Text.Replace("\"", string.Empty).Replace("'", string.Empty).ToLower();
				string text3;
				switch (text2)
				{
				case "d":
				case "y":
					text3 = string.Empty;
					break;
				case "w":
				case "ww":
					text3 = MappingManager.GetNamedReference(NamedReferenceConstant.DaysPerWeek).Name;
					break;
				case "h":
					text3 = MappingManager.GetNamedReference(NamedReferenceConstant.HoursPerDay).Name;
					break;
				case "n":
					text3 = MappingManager.GetNamedReference(NamedReferenceConstant.MinutesPerDay).Name;
					break;
				case "s":
					text3 = MappingManager.GetNamedReference(NamedReferenceConstant.SecondsPerDay).Name;
					break;
				default:
					argumentStack.Push(argumentInfo3);
					argumentStack.Push(argumentInfo2);
					argumentStack.Push(argumentInfo);
					functionString = BuildFunctionString(argumentStack, functionToken);
					ex = new FormulaExportException(FormulaExportErrorType.UnsupportedFunction, Infragistics.Shared.SR.GetString("FormulaError_UnsupportedFunction_DateDiff", text2), functionString);
					return true;
				}
				string arg = ((text3 == null || text3.Length == 0) ? string.Empty : $" * {text3.ToString()}");
				functionString = $"({argumentInfo.Text} - {argumentInfo2.Text}){arg}";
				return true;
			}
			default:
				functionString = null;
				return false;
			}
		}

		private static bool IsValidFunctionName(Workbook workbook, IUltraCalcFunctionToken functionToken)
		{
			if (functionToken.FunctionOperator.HasValue)
			{
				return true;
			}
			string functionName = functionToken.FunctionName;
			if (workbook.IsValidFunctionName(functionName))
			{
				return true;
			}
			switch (functionName.ToUpper())
			{
			case "DATEADD":
			case "DATEDIFF":
				return true;
			default:
				return false;
			}
		}

namespace Infragistics.Win.UltraWinGrid.ExcelExport.ValueListExporting
{
	internal class ValueListManager
	{
		public const string HIDDEN_WORKSHEET_BASE_NAME = "ValueList_Helper";

		private Workbook workbook;

		private Dictionary<IValueList, ListDataValidationRule> listValidationRules;

		private Worksheet hiddenWorkSheet;

		private int lastUsedColumn = -1;

		private Worksheet HiddenWorkSheet
		{
			get
			{
				if (hiddenWorkSheet == null)
				{
					string hiddenWorkSheetName = GetHiddenWorkSheetName();
					hiddenWorkSheet = workbook.Worksheets.Add(hiddenWorkSheetName);
					hiddenWorkSheet.DisplayOptions.Visibility = WorksheetVisibility.Hidden;
				}
				return hiddenWorkSheet;
			}
		}

		private Dictionary<IValueList, ListDataValidationRule> ListValidationRules
		{
			get
			{
				if (listValidationRules == null)
				{
					listValidationRules = new Dictionary<IValueList, ListDataValidationRule>();
				}
				return listValidationRules;
			}
		}

		internal ValueListManager(Workbook workbook)
		{
			this.workbook = workbook;
		}

		public ListDataValidationRule GetListDataValidationRule(IValueList valueList)
		{
			if (valueList == null || valueList.ItemCount <= 0)
			{
				return null;
			}
			if (!ListValidationRules.TryGetValue(valueList, out var value))
			{
				value = BuildListDataValidationRule(valueList);
				ListValidationRules[valueList] = value;
			}
			return value;
		}

		private void BumpLastColumnUsed()
		{
			lastUsedColumn++;
			if (lastUsedColumn >= workbook.MaxColumnCount)
			{
				lastUsedColumn = 0;
				hiddenWorkSheet = null;
			}
		}

		private ListDataValidationRule BuildListDataValidationRule(IValueList valueList)
		{
			BumpLastColumnUsed();
			int num = lastUsedColumn;
			for (int i = 0; i < valueList.ItemCount; i++)
			{
				HiddenWorkSheet.Rows[i].SetCellValue(num, valueList.GetText(i));
			}
			ListDataValidationRule listDataValidationRule = new ListDataValidationRule();
			WorksheetRegion worksheetRegion = new WorksheetRegion(hiddenWorkSheet, 0, num, valueList.ItemCount - 1, num);
			string arg = worksheetRegion.ToString(workbook.CellReferenceMode, includeWorksheetName: true);
			string valuesFormula = $"={arg}";
			listDataValidationRule.SetValuesFormula(valuesFormula, null);
			listDataValidationRule.ShowErrorMessageForInvalidValue = false;
			return listDataValidationRule;
		}

		private string GetHiddenWorkSheetName()
		{
			string text = "ValueList_Helper";
			int num = 0;
			while (workbook.Worksheets.Exists(text))
			{
				num++;
				text = string.Format("{0}_{1}", "ValueList_Helper", num.ToString());
			}
			return text;
		}
	}
