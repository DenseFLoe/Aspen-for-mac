			{
				this.band = band;
				this.propertyDescriptor = propertyDescriptor;
				this.changeType = changeType;

					deletedColumn = ((propertyDescriptor is UltraDataColumnPropertyDescriptor) ? ((UltraDataColumnPropertyDescriptor)propertyDescriptor).Column : null);

				}

			{
				if (rows != null && rows.Band == band)

					try
					{
						rows.FireBindingList_DataStructureChanged(changeType, propertyDescriptor);
						dataStructureFiredCount++;
					}
					catch (Exception)
					{
					}
					if (deletedChildBand != null)
					{
						rows.OnBandRemoved(deletedChildBand);
					}
					if (deletedColumn != null)
					{
						rows.OnColumnRemoved(deletedColumn);
					}
				}
				return true;
			}

		private object GetCellValue(UltraDataColumn column, bool throwInvalidColumnException, bool fireCellDataRequestedEvent, bool validateColumn)
		{
			if (parentCollection == null || (validateColumn && !Band.ValidateColumn(column, throwInvalidColumnException)))
			{
				return null;
			}
			object cellValue = ParentCollection.GetCellValue(this, column);
			if (cellValue != null)
			{
				return cellValue;
			}
			if (fireCellDataRequestedEvent)
			{
				UltraDataSource dataSource = DataSource;
				if (dataSource != null)
				{
					CellDataRequestedEventArgs e = new CellDataRequestedEventArgs(this, column);
					dataSource.FireEvent(DataSourceEventIds.CellDataRequested, e);
					if (e.CacheData)
					{
						if (e.Data != null)
						{
							ParentCollection.SetCellValue(this, column, e.Data);
						}
						else
						{
							ClearCellValue(column);
						}
					}
					if (e.Data != null)
					{
						return e.Data;
					}
				}
			}
			return column.DefaultValue;
		}

		internal void SetCellValue(UltraDataColumn column, object value, bool throwInvalidColumnException, bool fireCellUpdateEvents)
		{
			if (!Band.ValidateColumn(column, throwInvalidColumnException))
			{
				return;
			}
			object obj = column.ConvertToDataType(value);
			object cellValue = GetCellValue(column, throwInvalidColumnException: false, fireCellDataRequestedEvent: false, validateColumn: false);
			if (cellValue != null && obj != null && cellValue.GetType() == obj.GetType() && cellValue.Equals(obj) && IsCellValueCached(column))
			{
				return;
			}
			bool flag = true;
			if (fireCellUpdateEvents && DataSource != null)
			{
				CellDataUpdatingEventArgs e = new CellDataUpdatingEventArgs(this, column, obj);
				DataSource.FireEvent(DataSourceEventIds.CellDataUpdating, e);
				if (e.Cancel)
				{
					return;
				}
				obj = e.NewValue;
				flag = e.CacheData;
			}
			if (flag && obj != null)
			{
				ParentCollection.SetCellValue(this, column, obj);
			}
			else
			{
				ClearCellValue(column);
			}
			if (fireCellUpdateEvents && DataSource != null)
			{
				CellDataUpdatedEventArgs e2 = new CellDataUpdatedEventArgs(this, column, obj);
				DataSource.FireEvent(DataSourceEventIds.CellDataUpdated, e2);
			}
			if (!ItemChangedNotificationSuspended)
			{
				ParentCollection.FireBindingList_ItemChanged(this, column);
			}
		}

		void IEditableObject.BeginEdit()
		{
			if (IsInEdit)
			{
				return;
			}
			try
			{
				UltraDataSource dataSource = DataSource;
				if (dataSource != null && dataSource.dummyRowsCollection != parentCollection)
				{
					dataSource.FireEvent(DataSourceEventIds.RowBeginEdit, new RowBeginEditEventArgs(this));
				}
			}
			finally
			{
				IsInEdit = true;
				if (ParentCollection != null)
				{
					ParentCollection.BackupCellValues(this);
				}
			}
		}

		void IEditableObject.CancelEdit()
		{
			if (!IsInEdit)
			{
				return;
			}
			bool isAddRow = IsAddRow;
			IsInEdit = false;
			if (ParentCollection != null)
			{
				ParentCollection.RestoreBackupCellValues(this);
				ParentCollection.ClearBackupCellValues(this);
			}
			UltraDataSource dataSource = DataSource;
			if (dataSource != null && dataSource.dummyRowsCollection != parentCollection)
			{
				dataSource.FireEvent(DataSourceEventIds.RowCancelEdit, new RowCancelEditEventArgs(this));
			}
			if (!isAddRow)
			{
				return;
			}
			if (ParentCollection.Count <= 1 && Band != null)
			{
				foreach (UltraDataBand item in (IEnumerable)Band.ChildBands)
				{
					UltraDataRowsCollection childRows = GetChildRows(item, allocateIfNecessary: false, throwException: false);
					if (childRows != null)
					{
						item.tempNotificationRowsCollection = new WeakReference(childRows);
					}
				}
			}
			((IList)parentCollection).Remove((object)this);
		}

		void IEditableObject.EndEdit()
		{
			if (IsInEdit)
			{
				IsInEdit = false;
				if (ParentCollection != null)
				{
					ParentCollection.ClearBackupCellValues(this);
				}
				if (ParentCollection != null)
				{
					ParentCollection.ClearIfAddRow(this);
				}
				UltraDataSource dataSource = DataSource;
				if (dataSource != null && dataSource.dummyRowsCollection != parentCollection)
				{
					dataSource.FireEvent(DataSourceEventIds.RowEndEdit, new RowEndEditEventArgs(this));
				}
			}
		}

			foreach (UltraDataColumn item in (IEnumerable)Band.Columns)

			foreach (UltraDataBand item2 in (IEnumerable)Band.ChildBands)

				Utils.SerializeProperty(info, "Child_Band_" + item2.Key, GetChildRows(item2.Key));
			}
			if (ShouldSerializeTag())

				Utils.SerializeObjectProperty(info, "Tag", tag);
			}
			GetObjectData(info, context);
		}

				}
			}
			dataRowDeserializationData.rowData = arrayList.ToArray();

		private UltraDataRow InternalInsert(int index, bool raiseAddEvents, object[] cellValues, bool bypassEditState)
		{
			if (cellValues != null && cellValues.Length > Band.Columns.Count)
			{
				throw new ArgumentException("Specified cell values array contains more elements than the number of columns.", "cellValues");
			}
			RowAddingEventArgs e = null;
			if (raiseAddEvents && DataSource != null)
			{
				e = new RowAddingEventArgs(this, index);
				DataSource.FireEvent(DataSourceEventIds.RowAdding, e);
				if (e.Cancel)
				{
					return null;
				}
				index = e.Index;
			}
			if (index < 0 || index > Count)
			{
				throw new ArgumentOutOfRangeException("index", index, Infragistics.Shared.SR.GetString("LER_ArgumentOutOfRangeException_1"));
			}
			UltraDataRow ultraDataRow = new UltraDataRow(this);
			arrayList.Insert(index, ultraDataRow);
			if (cellValues != null)
			{
				UltraDataColumnsCollection columns = Band.Columns;
				for (int i = 0; i < cellValues.Length; i++)
				{
					object obj = cellValues[i];
					if (obj != null)
					{
						UltraDataColumn ultraDataColumn = columns[i];
						SetCellValue(ultraDataRow, ultraDataColumn, ultraDataColumn.ConvertToDataType(obj));
					}
				}
			}
			bool itemChangedNotificationSuspended = ultraDataRow.ItemChangedNotificationSuspended;
			ultraDataRow.ItemChangedNotificationSuspended = true;
			try
			{
				try
				{
					ultraDataRow.FireInitializeDataRow();
					if (raiseAddEvents && DataSource != null)
					{
						DataSource.FireEvent(DataSourceEventIds.RowAdded, new RowAddedEventArgs(ultraDataRow));
					}
				}
				finally
				{
					if (!bypassEditState)
					{
						lastAddedRow = ultraDataRow;
						((IEditableObject)ultraDataRow).BeginEdit();
					}
				}
				bool isInEdit = ultraDataRow.IsInEdit;
				FireBindingList_ItemAdded(index);
				if (bypassEditState && !isInEdit && ultraDataRow.IsInEdit)
				{
					((IEditableObject)ultraDataRow).EndEdit();
				}
			}
			finally
			{
				ultraDataRow.ItemChangedNotificationSuspended = itemChangedNotificationSuspended;
			}
			return ultraDataRow;
		}

		public void SetCount(int newCount, bool forceNotifyListReset)
		{
			if (newCount == Count)
			{
				return;
			}
			if (newCount < Count)
			{
				if (newCount == 0)
				{
					cellDataHolder = null;
					childRowsHolder = null;
				}
				else
				{
					for (int i = newCount; i < Count; i++)
					{
						UltraDataRow rowAt = GetRowAt(i, allocateIfNecessary: false);
						if (rowAt != null)
						{
							OnRowRemoved(rowAt);
						}
					}
				}
				arrayList.RemoveRange(newCount, Count - newCount);
			}
			else
			{
				arrayList.Expand(newCount);
			}
			if (lastAddedRow != null && !arrayList.Contains(lastAddedRow))
			{
				lastAddedRow = null;
			}
			FireBindingList_Reset();
		}

		internal void SyncChildRowsHolder()
		{
			UltraDataBandsCollection ultraDataBandsCollection = ((Band != null) ? Band.ChildBands : null);
			if (ultraDataBandsCollection == null)
			{
				return;
			}
			Hashtable hashtable = new Hashtable();
			if (childRowsHolder != null)
			{
				for (int i = 0; i < childRowsHolder.Length; i++)
				{
					UltraDataBand ultraDataBand = childRowsHolder[i].band;
					if (ultraDataBand != null && ultraDataBand.Index >= 0)
					{
						hashtable[ultraDataBand] = childRowsHolder[i];
					}
				}
			}
			childRowsHolder = new BandTablePair[ultraDataBandsCollection.Count];
			for (int i = 0; i < ultraDataBandsCollection.Count; i++)
			{
				UltraDataBand key = ultraDataBandsCollection[i];
				if (hashtable.ContainsKey(key))
				{
					ref BandTablePair reference = ref childRowsHolder[i];
					reference = (BandTablePair)hashtable[key];
				}
				else
				{
					ref BandTablePair reference2 = ref childRowsHolder[i];
					reference2 = new BandTablePair(key);
				}
			}
		}

			if (tp.backupValuesTable != null && tp.backupValuesTable.ContainsKey(row))
			{

		internal void OnRowRemoved(UltraDataRow row)
		{
			if (row == null)
			{
				return;

			ClearCellValues(row);
			if (childRowsHolder == null)
			{
				return;
			}
			for (int i = 0; i < childRowsHolder.Length; i++)
			{
				if (childRowsHolder[i].table != null && childRowsHolder[i].table.ContainsKey(row))
				{
					childRowsHolder[i].table.Remove(row);
					if (childRowsHolder[i].table.Count <= 0)
					{
						childRowsHolder[i].table = null;
					}
				}
			}
		}

		{
			int tableForChildBand = GetTableForChildBand(childBand, allocateIfNecessary);
			if (tableForChildBand < 0)
			{
				return null;
			}
			UltraDataRowsCollection ultraDataRowsCollection = (UltraDataRowsCollection)childRowsHolder[tableForChildBand].Table[row];
			if (ultraDataRowsCollection == null && allocateIfNecessary)

			return ultraDataRowsCollection;
		}

		private void SyncCellDataHolder()
		{
			UltraDataColumnsCollection ultraDataColumnsCollection = ((Band != null) ? Band.Columns : null);
			if (ultraDataColumnsCollection == null)
			{
				return;
			}
			Hashtable hashtable = new Hashtable();
			if (cellDataHolder != null)
			{
				for (int i = 0; i < cellDataHolder.Length; i++)
				{
					UltraDataColumn column = cellDataHolder[i].column;
					if (column != null && column.Index >= 0)
					{
						hashtable[column] = cellDataHolder[i];
					}
				}
			}
			cellDataHolder = new ColumnTablePair[ultraDataColumnsCollection.Count];
			for (int i = 0; i < ultraDataColumnsCollection.Count; i++)
			{
				UltraDataColumn ultraDataColumn = ultraDataColumnsCollection[i];
				if (hashtable.ContainsKey(ultraDataColumn))
				{
					ref ColumnTablePair reference = ref cellDataHolder[i];
					reference = (ColumnTablePair)hashtable[ultraDataColumn];
				}
				else
				{
					ref ColumnTablePair reference2 = ref cellDataHolder[i];
					reference2 = new ColumnTablePair(ultraDataColumn);
				}
			}
		}

		private int GetTableForColumn(UltraDataColumn column, bool allocateIfNecessary)
		{
			int index = column.Index;
			if (index < 0 || (!allocateIfNecessary && cellDataHolder == null))
			{
				return -1;
			}
			if (cellDataHolder == null || index >= cellDataHolder.Length || column != cellDataHolder[index].column)
			{
				SyncCellDataHolder();
			}
			return index;
		}

			if (childRowsHolder == null || index >= childRowsHolder.Length || childBand != childRowsHolder[index].band)
			{
				SyncChildRowsHolder();
			}
			return index;
		}

		internal bool TraverseRowsCollections(IRowsCollectionCallback callback, UltraDataBand band)
		{
			if (Band == band)
			{
				return callback.ProcessRowsCollection(this);
			}
			if (band == null && !callback.ProcessRowsCollection(this))
			{
				return false;
			}
			if (Band.HasChildBands)
			{
				for (int i = 0; i < Count; i++)
				{
					UltraDataRow rowAt = GetRowAt(i, allocateIfNecessary: false);
					if (rowAt == null)
					{
						continue;
					}
					UltraDataBandsCollection childBands = rowAt.Band.ChildBands;
					for (int j = 0; j < childBands.Count; j++)
					{
						UltraDataRowsCollection childRows = rowAt.GetChildRows(childBands[j], allocateIfNecessary: false, throwException: false);
						if (childRows != null && !childRows.TraverseRowsCollections(callback, band))
						{
							return false;
						}
					}
				}
			}
			return true;
		}

		PropertyDescriptorCollection ITypedList.GetItemProperties(PropertyDescriptor[] listAccessors)
		{
			UltraDataBand ultraDataBand = Band;
			bool flag = false;
			if (listAccessors != null && listAccessors.Length > 0 && listAccessors[listAccessors.Length - 1] is UltraDataBandPropertyDescriptor ultraDataBandPropertyDescriptor)
			{
				ultraDataBandPropertyDescriptor.EnsureValidBand();
				ultraDataBand = ultraDataBandPropertyDescriptor.Band;
				flag = true;
			}
			ArrayList arrayList = new ArrayList();
			if (!flag && ultraDataBand != null && ultraDataBand.IsTopMostBand && DataSource.dummyRowsCollection == this)
			{
				arrayList.Add(ultraDataBand.PropertyDescriptor);
			}
			else
			{
				UltraDataColumnsCollection ultraDataColumnsCollection = ((ultraDataBand != null && ultraDataBand.HasColumns) ? ultraDataBand.Columns : null);
				UltraDataBandsCollection ultraDataBandsCollection = ((ultraDataBand != null && ultraDataBand.HasChildBands) ? ultraDataBand.ChildBands : null);
				if (ultraDataColumnsCollection != null)
				{
					for (int i = 0; i < ultraDataColumnsCollection.Count; i++)
					{
						arrayList.Add(ultraDataColumnsCollection[i].PropertyDescriptor);
					}
				}
				if (ultraDataBandsCollection != null)
				{
					for (int i = 0; i < ultraDataBandsCollection.Count; i++)
					{
						arrayList.Add(ultraDataBandsCollection[i].PropertyDescriptor);
					}
				}
			}
			return new PropertyDescriptorCollection((PropertyDescriptor[])arrayList.ToArray(typeof(PropertyDescriptor)));
		}

		string ITypedList.GetListName(PropertyDescriptor[] listAccessors)
		{
			return Band.Key;
		}

		[SecurityPermission(SecurityAction.LinkDemand, Flags = SecurityPermissionFlag.SerializationFormatter)]
		void ISerializable.GetObjectData(SerializationInfo info, StreamingContext context)
		{
			for (int i = 0; i < Count; i++)
			{
				Utils.SerializeProperty(info, i.ToString(), this[i]);
			}
			if (ShouldSerializeTag())
			{
				Utils.SerializeObjectProperty(info, "Tag", tag);
			}
		}

		protected UltraDataRowsCollection(SerializationInfo info, StreamingContext context)
		{
			CreateArray();
			SerializationInfoEnumerator enumerator = info.GetEnumerator();
			while (enumerator.MoveNext())
			{
				SerializationEntry current = enumerator.Current;
				if (current.Value is UltraDataRow)
				{
					UltraDataRow ultraDataRow = (UltraDataRow)Utils.DeserializeProperty(current, typeof(UltraDataRow), null);
					if (ultraDataRow != null)
					{
						arrayList.Add(ultraDataRow);
					}
				}
				else
				{
					switch (current.Name)
					{
					case "Tag":
						tag = Utils.DeserializeObjectProperty(current);
						break;
					}
				}
			}
		}

		internal class UDSBindingSource : BindingSource
		{
			private UltraDataBand band;

			private bool isRoot;

			public UDSBindingSource(UltraDataBand band, object dataSource, string dataMember, bool isRoot)
				: base(dataSource, dataMember)
			{
				this.band = band;
				this.isRoot = isRoot;
			}

			internal void InternalSetBand(UltraDataBand band, Dictionary<string, BindingSource> oldBindingSources)
			{
				this.band = band;
				SetBindingSources(this.band, oldBindingSources);
				InternalSetBandHelper(this, this.band);
			}

			private void InternalSetBandHelper(BindingSource bindingSource, UltraDataBand band)
			{
				band.bindingSource = bindingSource;
				if (bindingSource is UDSBindingSource uDSBindingSource)
				{
					uDSBindingSource.band = band;
				}
			}

			internal Dictionary<string, BindingSource> GetOldBindingSources()
			{
				Dictionary<string, BindingSource> dictionary = new Dictionary<string, BindingSource>();
				GetBindingSources(band, dictionary);
				return dictionary;
			}

			private void GetBindingSources(UltraDataBand band, Dictionary<string, BindingSource> bindingSources)
			{
				if (band.bindingSource != null)
				{
					string bandPath = GetBandPath(band);
					if (!bindingSources.ContainsKey(bandPath))
					{
						bindingSources.Add(bandPath, band.bindingSource);
					}
				}
				if (!band.HasChildBands)
				{
					return;
				}
				foreach (UltraDataBand item in (IEnumerable)band.ChildBands)
				{
					GetBindingSources(item, bindingSources);
				}
			}

			private void SetBindingSources(UltraDataBand band, Dictionary<string, BindingSource> bindingSources)
			{
				if (band.bindingSource == null)
				{
					string bandPath = GetBandPath(band);
					if (bindingSources != null && bindingSources.ContainsKey(bandPath))
					{
						InternalSetBandHelper(bindingSources[bandPath], band);
						bindingSources.Remove(bandPath);
					}
				}
				if (!band.HasChildBands)
				{
					return;
				}
				foreach (UltraDataBand item in (IEnumerable)band.ChildBands)
				{
					SetBindingSources(item, bindingSources);
				}
			}

			private string GetBandPath(UltraDataBand band)
			{
				string bindingContextDataMember = band.GetBindingContextDataMember();
				if (bindingContextDataMember == null)
				{
					return string.Empty;
				}
				return bindingContextDataMember;
			}

			public override CurrencyManager GetRelatedCurrencyManager(string dataMember)
			{
				if (string.IsNullOrEmpty(dataMember))
				{
					return base.GetRelatedCurrencyManager(dataMember);
				}
				UltraDataBand ultraDataBand = ((!isRoot) ? (band.ChildBands.Exists(dataMember) ? band.ChildBands[dataMember] : null) : ((dataMember == band.Key) ? band : null));
				if (ultraDataBand == null)
				{
					throw new ArgumentException("Invalid data member.", "dataMember");
				}
				if (ultraDataBand.bindingSource == null || ultraDataBand.bindingSource.DataMember != dataMember)
				{
					ultraDataBand.bindingSource = new UDSBindingSource(ultraDataBand, this, dataMember, isRoot: false);
				}
				return ultraDataBand.bindingSource.CurrencyManager;
			}
		}

		public void SuspendBindingNotifications()
		{
			if (!BindingNotificationsSuspended)
			{
				notificationsSuspended_RowCollections = new Hashtable();
			}
			bindingNotificationsSuspendedCount++;
		}

		public void ResumeBindingNotifications()
		{
			ResumeBindingNotifications(fireNotifications: true);
		}

		public void ResumeBindingNotifications(bool fireNotifications)
		{
			if (BindingNotificationsSuspended)
			{
				bindingNotificationsSuspendedCount--;
			}
			if (BindingNotificationsSuspended || notificationsSuspended_RowCollections == null)
			{
				return;
			}
			if (fireNotifications)
			{
				IDictionaryEnumerator enumerator = notificationsSuspended_RowCollections.GetEnumerator();
				while (enumerator.MoveNext())
				{
					UltraDataRowsCollection ultraDataRowsCollection = (UltraDataRowsCollection)enumerator.Key;
					if (ultraDataRowsCollection != null && ultraDataRowsCollection.IsStillValid)
					{
						ultraDataRowsCollection.FireBindingList_Reset();
					}
				}
			}
			notificationsSuspended_RowCollections.Clear();
			notificationsSuspended_RowCollections = null;
		}

		internal void FireEvent(DataSourceEventIds id, EventArgs e)
		{
			if (!EventManager.CanFireEvent(id))
			{
				return;
			}
			EventManager.IncrementInProgress(id);
			try
			{
				switch (id)
				{
				case DataSourceEventIds.CellDataRequested:
					OnCellDataRequested((CellDataRequestedEventArgs)e);
					break;
				case DataSourceEventIds.CellDataUpdating:
					OnCellDataUpdating((CellDataUpdatingEventArgs)e);
					break;
				case DataSourceEventIds.CellDataUpdated:
					OnCellDataUpdated((CellDataUpdatedEventArgs)e);
					break;
				case DataSourceEventIds.InitializeDataRow:
					OnInitializeDataRow((InitializeDataRowEventArgs)e);
					break;
				case DataSourceEventIds.InitializeRowsCollection:
					OnInitializeRowsCollection((InitializeRowsCollectionEventArgs)e);
					break;
				case DataSourceEventIds.RowAdding:
					OnRowAdding((RowAddingEventArgs)e);
					break;
				case DataSourceEventIds.RowAdded:
					OnRowAdded((RowAddedEventArgs)e);
					break;
				case DataSourceEventIds.RowDeleting:
					OnRowDeleting((RowDeletingEventArgs)e);
					break;
				case DataSourceEventIds.RowDeleted:
					OnRowDeleted((RowDeletedEventArgs)e);
					break;
				case DataSourceEventIds.RowBeginEdit:
					OnRowBeginEdit((RowBeginEditEventArgs)e);
					break;
				case DataSourceEventIds.RowCancelEdit:
					OnRowCancelEdit((RowCancelEditEventArgs)e);
					break;
				case DataSourceEventIds.RowEndEdit:
					OnRowEndEdit((RowEndEditEventArgs)e);
					break;
				}
			}
			finally
			{
				EventManager.DecrementInProgress(id);
			}
		}

		private UltraDataBand GetBandWithDataMember(UltraDataBand band, string dataMember)
		{
			if (band != null)
			{
				if (dataMember.Equals(band.GetBindingContextDataMember()))
				{
					return band;
				}
				if (band.HasChildBands)
				{
					for (int i = 0; i < band.ChildBands.Count; i++)
					{
						UltraDataBand bandWithDataMember = GetBandWithDataMember(band.ChildBands[i], dataMember);
						if (bandWithDataMember != null)
						{
							return bandWithDataMember;
						}
					}
				}
			}
			return null;
		}

		private void Save(Stream stream, bool xml, bool includeData)
		{
			if (stream == null)
			{
				throw new ArgumentNullException(Infragistics.Shared.SR.GetString(null, "LE_V2_Exception_39"), Infragistics.Shared.SR.GetString(null, "LE_V2_Exception_40"));
			}
			IFormatter formatter = CreateFormatter(xml);
			ObjectStreamer graph = CreateObjectStreamer(includeData);
			formatter.Serialize(stream, graph);
		}

		private void Load(Stream stream, bool xml, bool includeData)
		{
			if (stream == null)
			{
				throw new ArgumentNullException(Infragistics.Shared.SR.GetString(null, "LE_V2_Exception_39"), Infragistics.Shared.SR.GetString(null, "LE_V2_Exception_40"));
			}
			IFormatter formatter = CreateFormatter(xml);
			formatter.Binder = (SerializationBinder)(object)new Infragistics.Win.UltraWinDataSource.Serialization.Binder();
			ObjectStreamer streamer = formatter.Deserialize(stream) as ObjectStreamer;
			if (rows != null)
			{
				Rows.Clear();
			}
			LoadHelper(streamer, includeData);
		}

		private void LoadHelper(ObjectStreamer streamer, bool includeData)
		{
			try
			{
				isInitializing = true;
				InitializeFrom(streamer, includeData);
			}
			finally
			{
				isInitializing = false;
			}
		}

		private ObjectStreamer CreateObjectStreamer(bool includeData)
		{
			ObjectStreamer objectStreamer = new ObjectStreamer("Infragistics.Win.UltraDataSource", 4);
			IDictionary dictionary = objectStreamer.Dictionary;
			if (ShouldSerializeAllowAdd())
			{
				dictionary.Add("AllowAdd", AllowAdd);
			}
			if (ShouldSerializeAllowDelete())
			{
				dictionary.Add("AllowDelete", AllowDelete);
			}
			if (ShouldSerializeReadOnly())
			{
				dictionary.Add("ReadOnly", ReadOnly);
			}
			if (ShouldSerializeBand())
			{
				dictionary.Add("Band", Band);
			}
			if (includeData && ShouldSerializeRows())
			{
				dictionary.Add("Rows", Rows);
			}
			return objectStreamer;
		}

		private void InitializeFrom(ObjectStreamer streamer, bool includeData)
		{
			UltraDataRowsCollection deserializedRows = null;
			if (streamer.AssemblyName == null || streamer.AssemblyName.IndexOf("Infragistics.Win.UltraDataSource") < 0)
			{
				throw new InvalidCastException(Infragistics.Shared.SR.GetString(null, "LE_V2_Exception_41"));
			}
			Dictionary<string, BindingSource> oldBindingSources = null;
			if (bindingSource != null)
			{
				oldBindingSources = bindingSource.GetOldBindingSources();
			}
			Reset();
			foreach (DictionaryEntry item in streamer.Dictionary)
			{
				if (!(item.Value is SerializationEntry))
				{
					continue;
				}
				SerializationEntry entry = (SerializationEntry)item.Value;
				switch (entry.Name)
				{
				case "AllowAdd":
					allowAdd = (bool)Utils.DeserializeProperty(entry, typeof(bool), AllowAdd);
					break;
				case "AllowDelete":
					allowDelete = (bool)Utils.DeserializeProperty(entry, typeof(bool), AllowDelete);
					break;
				case "ReadOnly":
					readOnly = (bool)Utils.DeserializeProperty(entry, typeof(bool), ReadOnly);
					break;
				case "Rows":
					if (includeData)
					{
						deserializedRows = (UltraDataRowsCollection)Utils.DeserializeProperty(entry, typeof(UltraDataRowsCollection), Rows);
					}
					break;
				case "Band":
					band = (UltraDataRootBand)Utils.DeserializeProperty(entry, typeof(UltraDataRootBand), Band);
					break;
				}
			}
			OnDeserializationComplete(deserializedRows, oldBindingSources);
		}

		private IFormatter CreateFormatter(bool xml)
		{
			if (xml)
			{
				SoapFormatter soapFormatter = new SoapFormatter();
				soapFormatter.AssemblyFormat = FormatterAssemblyStyle.Simple;
				return soapFormatter;
			}
			BinaryFormatter binaryFormatter = new BinaryFormatter();
			binaryFormatter.AssemblyFormat = FormatterAssemblyStyle.Simple;
			return binaryFormatter;
		}

		private void OnDeserializationComplete(UltraDataRowsCollection deserializedRows, Dictionary<string, BindingSource> oldBindingSources)
		{
			Rows.FireInitializeRowsCollection();
			Band.OnDeserializationComplete(null, this);
			Rows.OnDeserializationComplete(Band, null, deserializedRows);
			if (dummyRowsCollection != null)
			{
				dummyRowsCollection.InternalSetBand(Band);
			}
			if (bindingSource != null)
			{
				bindingSource.InternalSetBand(Band, oldBindingSources);
			}
			bool flag = isInitializing;
			try
			{
				isInitializing = false;
				Rows.FireBindingList_Reset();
				Band.FireBindingList_DataStructureChanged(ListChangedType.PropertyDescriptorChanged, null);
				Rows.FireBindingList_Reset();
			}
			finally
			{
				isInitializing = flag;
			}
		}

			}
		}

		CurrencyManager ICurrencyManagerProvider.GetRelatedCurrencyManager(string dataMember)
		{
			if (UseBindingSource)
			{
				return DataBindingUtils.GetCurrencyManager((BindingContext)null, (object)BindingSource, dataMember);
			}
			if (bindingContext == null)
			{
				bindingContext = new BindingContext();
			}
			if (dataMember == null || dataMember.Length == 0)
			{
				return (CurrencyManager)bindingContext[((IListSource)this).GetList(), null];
			}
			if (dataMember == Band.Key)
			{
				return (CurrencyManager)bindingContext[((IListSource)this).GetList(), dataMember];
			}
			return null;
		}
