	{
	}

	public static IACMCustomForm Create(XmlNode src)
	{
		ACMExpCustomForm aCMExpCustomForm = new ACMExpCustomForm();
		XmlNode namedItem = src.Attributes.GetNamedItem("Tabbed");
		if (namedItem != null)
		{
			aCMExpCustomForm.Tabbed = Convert.ToBoolean(namedItem.Value, CultureInfo.InvariantCulture);
		}
		namedItem = src.Attributes.GetNamedItem("Type");
		if (namedItem != null)
		{
			aCMExpCustomForm.FormType = namedItem.Value;
		}
		XmlNamespaceManager xmlNamespaceManager = new XmlNamespaceManager(src.OwnerDocument.NameTable);
		xmlNamespaceManager.AddNamespace("acm", "urn:acmcustomforms");
		XmlNode xmlNode = src.SelectSingleNode("acm:Progid", xmlNamespaceManager);
		if (xmlNode != null)
		{
			aCMExpCustomForm.ProgId = xmlNode.InnerText;
		}
		XmlNode xmlNode2 = src.SelectSingleNode("acm:Clsid", xmlNamespaceManager);
		if (xmlNode2 != null)
		{
			aCMExpCustomForm.Clsid = new Guid(xmlNode2.InnerText);
		}
		return aCMExpCustomForm;
	}

public class ACMExpGridForm : IACMGridForm
{
	private List<Tuple<string, string>> entries;

	public string Id { get; private set; }

	public string Mode { get; private set; }

	public string GridType { get; private set; }

	public bool Default { get; private set; }

	public string View { get; private set; }

	public string Name { get; private set; }

	public bool Autosave { get; private set; }

	public string[] Keys => entries.Select((Tuple<string, string> t) => t.Item1).ToArray();

	public string[] Values => entries.Select((Tuple<string, string> t) => t.Item2).ToArray();

	private ACMExpGridForm()
	{
		entries = new List<Tuple<string, string>>();
	}

	public static IACMGridForm Create(XmlNode src)
	{
		ACMExpGridForm aCMExpGridForm = new ACMExpGridForm();
		XmlNode namedItem = src.Attributes.GetNamedItem("Id");
		if (namedItem != null)
		{
			aCMExpGridForm.Id = namedItem.Value;
		}
		namedItem = src.Attributes.GetNamedItem("Mode");
		if (namedItem != null)
		{
			aCMExpGridForm.Mode = namedItem.Value;
		}
		namedItem = src.Attributes.GetNamedItem("Type");
		if (namedItem != null)
		{
			aCMExpGridForm.GridType = namedItem.Value;
		}
		namedItem = src.Attributes.GetNamedItem("Default");
		if (namedItem != null)
		{
			aCMExpGridForm.Default = Convert.ToBoolean(namedItem.Value, CultureInfo.InvariantCulture);
		}
		namedItem = src.Attributes.GetNamedItem("View");
		if (namedItem != null)
		{
			aCMExpGridForm.View = namedItem.Value;
		}
		namedItem = src.Attributes.GetNamedItem("Name");
		if (namedItem != null)
		{
			aCMExpGridForm.Name = namedItem.Value;
		}
		namedItem = src.Attributes.GetNamedItem("Autosave");
		if (namedItem != null)
		{
			aCMExpGridForm.Autosave = Convert.ToBoolean(namedItem.Value, CultureInfo.InvariantCulture);
		}
		XmlNamespaceManager xmlNamespaceManager = new XmlNamespaceManager(src.OwnerDocument.NameTable);
		xmlNamespaceManager.AddNamespace("acm", "urn:acmsheets");
		foreach (XmlNode item in src.SelectNodes("acm:Entry", xmlNamespaceManager))
		{
			string text = item.SelectSingleNode("acm:Key", xmlNamespaceManager)?.InnerText;
			string text2 = item.SelectSingleNode("acm:Value", xmlNamespaceManager)?.InnerText;
			if (!string.IsNullOrEmpty(text) && !string.IsNullOrEmpty(text2))
			{
				aCMExpGridForm.entries.Add(new Tuple<string, string>(text, text2));
			}
		}
		return aCMExpGridForm;
	}

	public IACMExpModel CreateModel(string name, string version, bool overwrite)
	{
		if (string.IsNullOrEmpty(workingPath))
		{
			return null;
		}
		string text = Path.Combine(Path.Combine(workingPath, version), name);
		Tuple<string, string> tuple = Tuple.Create(name, version);
		if (!overwrite && (InstalledLibraries.ContainsKey(tuple) || WorkingLibraries.ContainsKey(tuple)))
		{
			return null;
		}
		if (WorkingLibraries.ContainsKey(tuple) && Directory.Exists(text))
		{
			try
			{
				Directory.Delete(text, recursive: true);
			}
			catch
			{
			}
		}
		if (!Directory.Exists(text))
		{
			try
			{
				Directory.CreateDirectory(text);
			}
			catch (IOException)
			{
				return null;
			}
		}
		if (!WorkingLibraries.ContainsKey(tuple))
		{
			WorkingLibraries.Add(tuple, text);
		}
		if (!OpenLibraries.Contains(tuple))
		{
			OpenLibraries.Add(tuple);
		}
		NewACMExpModel result = new NewACMExpModel(name, version, null);
		OnLibrariesChanged();
		return result;
	}

	public void Refresh()
	{
		allModels = new List<ACMExpModelInfo>();
		Dictionary<string, double> duplicateModelNames = new Dictionary<string, double>();
		InstalledLibraries = new Dictionary<Tuple<string, string>, string>();
		WorkingLibraries = new Dictionary<Tuple<string, string>, string>();
		string text = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "AspenTech\\AES\\AM Models");
		if (!Directory.Exists(text) && text.IndexOf("\\..") < 0 && text.IndexOf("..\\") < 0)
		{
			Directory.CreateDirectory(text);
		}
		decimal result = 0.0m;
		decimal.TryParse(CompatableWith, NumberStyles.Any, CultureInfo.InvariantCulture, out result);
		string[] files = Directory.GetFiles(text, "*.atmlz", SearchOption.AllDirectories);
		foreach (string text2 in files)
		{
			string fileNameWithoutExtension = Path.GetFileNameWithoutExtension(text2);
			string name = new DirectoryInfo(Path.GetDirectoryName(text2)).Name;
			decimal result2 = 0.0m;
			if (!decimal.TryParse(name, NumberStyles.Any, CultureInfo.InvariantCulture, out result2) || result2 < 37.0m || result2 > result)
			{
				continue;
			}
			Tuple<string, string> tuple = new Tuple<string, string>(fileNameWithoutExtension, name);
			if (string.Compare(tuple.Item1, "BatchSep", ignoreCase: true) == 0 || string.Compare(tuple.Item1, "ARTRXNCONTAINER", ignoreCase: true) == 0 || string.Compare(tuple.Item1, "Valve", ignoreCase: true) == 0 || string.Compare(tuple.Item1, "RCSTR3", ignoreCase: true) == 0 || string.Compare(tuple.Item1, "RplugPDE", ignoreCase: true) == 0)
			{
				if (!unexpectedBuiltin)
				{
					unexpectedBuiltin = true;
				}
				continue;
			}
			InstalledLibraries.Add(tuple, text2);
			WorkingLibraries.Add(tuple, text2);
			ACMExpModelInfo aCMExpModelInfo = new ACMExpModelInfo(ACMExEntryType.NEWACMBLOCK);
			if (!aCMExpModelInfo.InitFromPackage(text2))
			{
				continue;
			}
			allModels.Add(aCMExpModelInfo);
			double dVersion = aCMExpModelInfo.DVersion;
			if (duplicateModelNames.ContainsKey(aCMExpModelInfo.Name))
			{
				if (dVersion > duplicateModelNames[aCMExpModelInfo.Name])
				{
					duplicateModelNames[aCMExpModelInfo.Name] = dVersion;
				}
			}
			else
			{
				duplicateModelNames.Add(aCMExpModelInfo.Name, dVersion);
			}
		}
		string text3 = string.Empty;
		RegistryKey registryKey = Registry.LocalMachine.OpenSubKey("Software\\AspenTech\\AMSystem\\" + CompatableWith + "\\Dir");
		if (registryKey != null)
		{
			object value = registryKey.GetValue("Bin");
			if (value != null)
			{
				text3 = value.ToString() + "\\";
			}
		}
		ACMExpModelInfo aCMExpModelInfo2 = new ACMExpModelInfo(ACMExEntryType.ACMBLOCKBUILTIN);
		aCMExpModelInfo2.Name = "BatchSep";
		aCMExpModelInfo2.Version = CompatableWith;
		aCMExpModelInfo2.PackageFile = text3 + "BatchSep.dll";
		aCMExpModelInfo2.Category = "Batch Models";
		allModels.Add(aCMExpModelInfo2);
		string[] array = CompatableWith.Split('.');
		if (array.Length > 1)
		{
			string text4 = $"Modeler{array[0]}{array[1]}_{CompatableWithMinor}.dll";
			ACMExpModelInfo aCMExpModelInfo3 = new ACMExpModelInfo(ACMExEntryType.ACMBLOCKBUILTIN);
			aCMExpModelInfo3.Name = "ARTRXNCONTAINER";
			aCMExpModelInfo3.Version = CompatableWith;
			aCMExpModelInfo3.PackageFile = text3 + text4;
			aCMExpModelInfo3.Category = string.Empty;
			aCMExpModelInfo3.Show = false;
			allModels.Add(aCMExpModelInfo3);
		}
		foreach (ACMExpModelInfo item in allModels.Where((ACMExpModelInfo m) => duplicateModelNames.ContainsKey(m.Name) && m.DVersion < duplicateModelNames[m.Name]))
		{
			string arg = ((item.Name.Length > 24) ? item.Name.Substring(0, 24) : item.Name);
			item.Display = $"{arg} v{item.Version.Replace('.', '_')}";
		}
		initialized = true;
	}

	public IACMExpModelInfo FindModelInfo(string name)
	{
		if (!initialized)
		{
			Refresh();
		}
		if (allModels == null)
		{
			return null;
		}
		return allModels.FirstOrDefault((ACMExpModelInfo x) => string.Equals(x.Display, name, StringComparison.OrdinalIgnoreCase) && (x.ClassType == ACMExEntryType.ACMBLOCK || x.ClassType == ACMExEntryType.NEWACMBLOCK || x.ClassType == ACMExEntryType.NEWACMSTRUCTURE || x.ClassType == ACMExEntryType.ACMBLOCKBUILTIN));
	}

	public IACMExpModel FindModel(string name)
	{
		IACMExpModelInfo iACMExpModelInfo = FindModelInfo(name);
		if (iACMExpModelInfo == null || iACMExpModelInfo.Model == null)
		{
			return null;
		}
		(iACMExpModelInfo as ACMExpModelInfo).Locked = true;
		return iACMExpModelInfo.Model;
	}

	public void UnlockModel(string name)

		IACMExpModelInfo iACMExpModelInfo = FindModelInfo(name);
		if (iACMExpModelInfo != null && iACMExpModelInfo.Model != null)
		{
			(iACMExpModelInfo as ACMExpModelInfo).Locked = false;
		}
	}

	public IACMExpModelInfo FindModelInfo(string name, string version)
	{
		if (!initialized)
		{
			Refresh();
		}
		if (allModels == null)
		{
			return null;
		}
		return allModels.FirstOrDefault((ACMExpModelInfo x) => string.Equals(x.Name, name, StringComparison.OrdinalIgnoreCase) && x.Version == version && (x.ClassType == ACMExEntryType.ACMBLOCK || x.ClassType == ACMExEntryType.NEWACMBLOCK || x.ClassType == ACMExEntryType.NEWACMSTRUCTURE || x.ClassType == ACMExEntryType.ACMBLOCKBUILTIN));
	}

	public IACMExpModel FindModel(string name, string version)
	{
		IACMExpModelInfo iACMExpModelInfo = FindModelInfo(name, version);
		if (iACMExpModelInfo == null || iACMExpModelInfo.Model == null)
		{
			return null;
		}
		(iACMExpModelInfo as ACMExpModelInfo).Locked = true;
		return iACMExpModelInfo.Model;
	}

			switch (ClassType)
			{
			case ACMExEntryType.ACMBLOCK:
				_Model = new OldACMExpModel(Name, Version, PackageFile);
				break;
			case ACMExEntryType.NEWACMBLOCK:
				_Model = new NewACMExpModel(Name, Version, PackageFile);
				break;
			case ACMExEntryType.NEWACMSTRUCTURE:
				_Model = new NewACMExpModel(Name, Version, PackageFile);
				break;
			case ACMExEntryType.ACMBLOCKBUILTIN:
				_Model = new ACMBuiltIn(Name, Version, PackageFile, Category);
				break;
			case ACMExEntryType.ACMREACTION:
				_Model = new OldACMExpReaction(Name, Version, PackageFile);
				break;
			case ACMExEntryType.NEWACMREACTION:
				_Model = new NewACMExpModel(Name, Version, PackageFile);
				break;
			}
			if (_Model != null)
			{
				_Model.Show = Show;
			}
			return _Model;
		}
		internal set
		{
			_Model = value;
		}
	}

	public bool Locked
	{
		get
		{
			return _fsLock != null;
		}
		internal set
		{
			if (ClassType == ACMExEntryType.NEWACMBLOCK)
			{
				if (value && _fsLock == null)
				{
					_fsLock = File.Open(PackageFile, FileMode.Open, FileAccess.Read, FileShare.Read);
				}
				if (!value && _fsLock != null)
				{
					_fsLock.Dispose();
					_fsLock = null;
				}
			}
		}
	}

	public bool InitFromPackage(string sPkgFile)
	{
		//IL_0009: Unknown result type (might be due to invalid IL or missing references)
		//IL_000e: Unknown result type (might be due to invalid IL or missing references)
		//IL_0017: Unknown result type (might be due to invalid IL or missing references)
		bool result = false;
		PackageFile = sPkgFile;
		try
		{
			AspenPackage val = new AspenPackage();
			val.Open(sPkgFile, FileMode.Open);
			byte[] bytes = val.GetRootPart().SaveToData();
			val.Close();
			string xml = Encoding.ASCII.GetString(bytes);
			XmlDocument xmlDocument = new XmlDocument();
			xmlDocument.LoadXml(xml);
			XmlNode xmlNode = xmlDocument.SelectSingleNode("/ATMLZ");
			Name = xmlNode.Attributes["Name"].Value;
			Version = xmlNode.Attributes["Version"].Value;
			RevisionGuid = xmlNode.Attributes["GUID"].Value;
			Timestamp = xmlNode.Attributes["Timestamp"].Value;
			Category = xmlNode.Attributes["LibraryCategory"].Value;
			if (xmlNode.Attributes["Description"] != null)
			{
				Description = xmlNode.Attributes["Description"].Value;
			}
			if (xmlNode.Attributes["Compiled"] != null)
			{
				Compiled = xmlNode.Attributes["Compiled"].Value;
			}
			string value = xmlNode.Attributes["Class"].Value;
			int result2 = 0;
			if (int.TryParse(value, out result2))
			{
				ACMExEntryType aCMExEntryType = (ACMExEntryType)result2;
				if (aCMExEntryType == ACMExEntryType.NEWACMBLOCK || aCMExEntryType == ACMExEntryType.NEWACMREACTION || aCMExEntryType == ACMExEntryType.NEWACMSTRUCTURE)
				{
					ClassType = aCMExEntryType;
				}
			}
			result = true;
		}
		catch
		{
		}
		return result;
	}

	public NewACMExpModel(string _name, string _version, string installLoc)
	{
		ClassType = ACMExEntryType.NEWACMBLOCK;
		Name = _name;
		Display = _name;
		Version = _version;
		Show = true;
		double result = 0.0;
		double.TryParse(Version, NumberStyles.Any, CultureInfo.InvariantCulture, out result);
		VersionNumber = result;
		isOpen = false;
		string path = Path.Combine(Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "Temp\\ACM_Models"), Path.Combine(Version, Name));
		WorkingFolder = Path.Combine(path, Path.GetRandomFileName());
		try
		{
			while (Directory.Exists(WorkingFolder))
			{
				WorkingFolder = Path.Combine(path, Path.GetRandomFileName());
			}
			if (WorkingFolder.IndexOf("\\..") < 0 && WorkingFolder.IndexOf("..\\") < 0)
			{
				Directory.CreateDirectory(WorkingFolder);
			}
		}
		catch
		{
		}
		installedLocation = installLoc;
		Open();
	}

	public string CreationCommand(bool EO, bool fixedInlets, bool noCheck)
	{
		string text = "";
		if (EO)
		{
			text += "eomodel=yes,";
		}
		if (fixedInlets)
		{
			text += "fixconnectedinletports=yes,";
		}
		if (noCheck)
		{
			text += "checktype=no,";
		}
		int num = (int)(VersionNumber + 0.001) * 10;
		string text2 = $"acmotf{num}";
		string text3 = Path.Combine(WorkingFolder, Path.Combine("ModelDef", Name + "_Export.xml"));
		bool flag = File.Exists(text3);
		string[] procedures = Procedures;
		string text4 = "";
		if (procedures != null)
		{
			string[] array = procedures;
			foreach (string text5 in array)
			{
				if (string.Compare(text5.Substring(text5.LastIndexOf("\\") + 1), Name + ".dll", ignoreCase: true) == 0)
				{
					text4 = text5;
				}
			}
		}
		if (!string.IsNullOrEmpty(text4))
		{
			return text + "model=" + text4;
		}
		if (flag)
		{
			if (ClassType == ACMExEntryType.NEWACMSTRUCTURE)
			{
				return text + "library=" + text2 + ",model=acmotfst,cfg=" + text3 + ",create=" + Name;
			}
			return text + "model=" + text2 + ",cfg=" + text3 + ",create=" + Name;
		}
		return null;
	}

	public void Open()
	{
		//IL_002f: Unknown result type (might be due to invalid IL or missing references)
		//IL_0035: Expected O, but got Unknown
		if (Name == "TwoVarIf")
		{
			count++;
			_ = count;
		}
		if (installedLocation != null)
		{
			try
			{
				AspenPackage val = new AspenPackage();
				val.Open(installedLocation, FileMode.Open);
				IPart[] parts = val.GetParts();
				foreach (IPart val2 in parts)
				{
					if (!Directory.Exists(Path.Combine(WorkingFolder, val2.FilePath)))
					{
						Directory.CreateDirectory(Path.Combine(WorkingFolder, val2.FilePath));
					}
					val2.SaveToFile(Path.Combine(WorkingFolder, Path.Combine(val2.FilePath, val2.Name)));
					File.SetAttributes(Path.Combine(WorkingFolder, Path.Combine(val2.FilePath, val2.Name)), FileAttributes.Normal);
				}
				val.Close();
				string text = Path.Combine(WorkingFolder, "Info.xml");
				if (File.Exists(text))
				{
					XmlDocument xmlDocument = new XmlDocument();
					xmlDocument.Load(text);
					XmlNode xmlNode = xmlDocument.SelectSingleNode("/ATMLZ");
					sTimestamp = xmlNode.Attributes["Timestamp"].Value;
					sGuid = xmlNode.Attributes["GUID"].Value;
					if (xmlNode.Attributes["Class"] != null)
					{
						string value = xmlNode.Attributes["Class"].Value;
						int result = 0;
						if (int.TryParse(value, out result))
						{
							ACMExEntryType aCMExEntryType = (ACMExEntryType)result;
							if (aCMExEntryType == ACMExEntryType.NEWACMBLOCK || aCMExEntryType == ACMExEntryType.NEWACMREACTION || aCMExEntryType == ACMExEntryType.NEWACMSTRUCTURE)
							{
								ClassType = aCMExEntryType;
							}
						}
					}
					if (xmlNode.Attributes["LibraryCategory"] != null && xmlNode.Attributes["LibraryCategory"].Value != "")
					{
						sLibraryCategory = new string[1] { xmlNode.Attributes["LibraryCategory"].Value };
					}
					if (xmlNode.Attributes["Description"] != null && xmlNode.Attributes["Description"].Value != "")
					{
						_description = WebUtility.HtmlDecode(xmlNode.Attributes["Description"].Value);
					}
					if (xmlNode.Attributes["Compiled"] != null && xmlNode.Attributes["Compiled"].Value != "")
					{
						_compiled = WebUtility.HtmlDecode(xmlNode.Attributes["Compiled"].Value);
					}
				}
				isOpen = true;
				return;
			}
			catch
			{
				return;
			}
		}
		UpdateInfo();
	}

	public void Close()
	{
		if (isOpen)
		{
			if (Name == "TwoVarIf")
			{
				count--;
				_ = count;
			}
			try
			{
				Directory.Delete(WorkingFolder, recursive: true);
			}
			catch
			{
			}
			isOpen = false;
		}
	}

	public void AddModelNode(XmlDocument document, XmlNode parent)
	{
		if (exportedModels != null)

			}

		}

		xmlNode2.Value = Name;
		xmlNode.Attributes.SetNamedItem(xmlNode2);
		if (!string.IsNullOrEmpty(Display) && Display != Name)
		{

		}

		{

		}
		xmlNode.AppendChild(xmlNode3);

				}
				break;

				{

				}
				break;
			case 4:
			{
				XmlNode newChild3 = document.CreateElement("else");

				break;

			{

				break;
			}
			case 6:
				if (expressions.ContainsKey(tuple.Item3))
				{
					ACMExpExpressionWriter aCMExpExpressionWriter = expressions[tuple.Item3];
					XmlNode xmlNode5 = document.CreateElement("for");
					xmlNode2 = document.CreateAttribute("element");
					xmlNode2.Value = tuple.Item2;
					xmlNode5.Attributes.SetNamedItem(xmlNode2);
					XmlNode xmlNode6 = document.CreateElement("in");
					aCMExpExpressionWriter.Write(document, xmlNode6);
					xmlNode5.AppendChild(xmlNode6);

				}
				break;
			case 7:
			{

				break;
			}
			}

		XmlNode xmlNode8 = document.CreateElement("equations");
		for (int j = 0; j < equations.Count(); j++)
		{
			Tuple<int, string, string> tuple2 = equations[j];

			case 1:

				{

				break;
			case 2:

				{
					ACMExpExpressionWriter aCMExpExpressionWriter4 = expressions[tuple2.Item3];
					XmlNode xmlNode11 = document.CreateElement("stif");

					aCMExpExpressionWriter4.Write(document, xmlNode11);
					xmlNode8.AppendChild(xmlNode11);
				}
				break;
			case 3:
				if (expressions.ContainsKey(tuple2.Item3))
				{
					ACMExpExpressionWriter aCMExpExpressionWriter5 = expressions[tuple2.Item3];
					XmlNode xmlNode12 = document.CreateElement("rtif");
					xmlNode2 = document.CreateAttribute("name");
					xmlNode2.Value = tuple2.Item2;
					xmlNode12.Attributes.SetNamedItem(xmlNode2);
					aCMExpExpressionWriter5.Write(document, xmlNode12);
					xmlNode8.AppendChild(xmlNode12);
				}
				break;
			case 4:
			{
				XmlNode newChild6 = document.CreateElement("else");
				xmlNode8.AppendChild(newChild6);
				break;
			}
			case 5:
			{
				XmlNode newChild5 = document.CreateElement("endif");
				xmlNode8.AppendChild(newChild5);
				break;
			}
			case 6:
				if (expressions.ContainsKey(tuple2.Item3))
				{
					ACMExpExpressionWriter aCMExpExpressionWriter3 = expressions[tuple2.Item3];
					XmlNode xmlNode9 = document.CreateElement("for");
					xmlNode2 = document.CreateAttribute("element");
					xmlNode2.Value = tuple2.Item2;
					xmlNode9.Attributes.SetNamedItem(xmlNode2);
					XmlNode xmlNode10 = document.CreateElement("in");
					aCMExpExpressionWriter3.Write(document, xmlNode10);
					xmlNode9.AppendChild(xmlNode10);
					xmlNode8.AppendChild(xmlNode9);
				}
				break;
			case 7:
			{
				XmlNode newChild4 = document.CreateElement("endfor");
				xmlNode8.AppendChild(newChild4);
				break;
			}
			case 8:
				if (proccalls.ContainsKey(tuple2.Item2))
				{
					proccalls[tuple2.Item2].Write(document, xmlNode8);
				}
				break;
			}
		}
		xmlNode.AppendChild(xmlNode8);
		XmlNode xmlNode13 = document.CreateElement("references");
		foreach (string library in libraries)
		{
			XmlNode xmlNode14 = document.CreateElement("lib");
			xmlNode14.InnerText = library;
			xmlNode13.AppendChild(xmlNode14);
		}
		xmlNode.AppendChild(xmlNode13);
		XmlNode xmlNode15 = document.CreateElement("scripts");
		foreach (KeyValuePair<string, string> script in scripts)
		{
			XmlNode xmlNode16 = document.CreateElement("script");
			xmlNode2 = document.CreateAttribute("name");
			xmlNode2.Value = script.Key;
			xmlNode16.Attributes.SetNamedItem(xmlNode2);
			xmlNode16.InnerText = script.Value;
			xmlNode15.AppendChild(xmlNode16);
		}
		xmlNode.AppendChild(xmlNode15);
		if (acmf.Count > 0)
		{
			XmlNode xmlNode17 = document.CreateElement("sources");
			foreach (KeyValuePair<string, Tuple<int, string>> item in acmf)
			{
				XmlNode xmlNode18 = document.CreateElement("source");
				xmlNode2 = document.CreateAttribute("name");
				xmlNode2.Value = item.Key + " " + item.Value.Item1.ToString(CultureInfo.InvariantCulture);
				xmlNode18.Attributes.SetNamedItem(xmlNode2);
				XmlCDataSection newChild7 = document.CreateCDataSection(item.Value.Item2);
				xmlNode18.AppendChild(newChild7);
				xmlNode17.AppendChild(xmlNode18);
			}
			xmlNode.AppendChild(xmlNode17);
		}
		if (parent == null)
		{
			document.AppendChild(xmlNode);
			return;
		}
		parent.AppendChild(xmlNode);
		foreach (IACMExpModelWriter localmodel in localmodels)
		{
			localmodel.AddModelNode(document, parent);
		}
		foreach (IACMExpVariableType localtype in localtypes)
		{
			localtype.AddVarNode(document, parent);
		}
	}
