		private static CxsNode GetNodeByPath(string datapath, CxsCase currentcase)
		{
			if (datapath.ToLowerInvariant().StartsWith("top."))
			{
				datapath = datapath.Substring(4);
			}
			CxsNode topNode = currentcase.TopNode;
			if (topNode != null)
			{
				return topNode.ResolveNode(datapath);
			}
			return null;
		}

		private static List<CxsNode> GetChildren(string datapath, CxsCase currentcase)
		{
			CxsNode nodeByPath = GetNodeByPath(datapath, currentcase);
			if (nodeByPath != null)
			{
				return ((IEnumerable<CxsNode>)nodeByPath.ChildNodes).ToList();
			}
			return null;
		}

		public static List<CxsNodeInfo> BuildNodeInfo(string datapath, CxsCase currentcase)
		{
			List<CxsNode> children = GetChildren(datapath, currentcase);
			if (children == null)
			{
				return null;
			}
			List<CxsNodeInfo> list = new List<CxsNodeInfo>();
			if (children.Count == 0)
			{
				return list;
			}
			string[] array = CxsConverter.ValueToStringArray(currentcase.BatchGet(datapath, "child_ids"));
			if (array != null && array.Count() > 0 && array.Count() != children.Count)
			{
				for (int i = 0; i < array.Count(); i++)
				{
					string path = CxsPath.CombinePathAndId(datapath, array[i]);
					list.Add(new CxsNodeInfo
					{
						isleaf = true,
						path = path,
						id = array[i],
						type = GetChildDataType(currentcase, path),
						containertype = GetChildContainerType(currentcase, path)
					});
				}
			}
			else
			{
				foreach (CxsNode item in children.OrderBy((CxsNode x) => x.Id))
				{
					list.Add(new CxsNodeInfo
					{
						isleaf = item.IsLeaf,
						path = item.Path,
						type = item.NodeType,
						id = item.Id,
						containertype = item.ContainerType
					});
				}
			}
			return list;
		}

		public static Dictionary<CxsNodeInfo, StreamConnectionInfo> RetrieveStreamConnectionInfo(string datapath, CxsCase currentcase)
		{
			List<CxsNode> children = GetChildren(datapath, currentcase);
			Dictionary<CxsNodeInfo, StreamConnectionInfo> dictionary = new Dictionary<CxsNodeInfo, StreamConnectionInfo>();
			if (children != null)
			{
				StreamConnectionInfo value = default(StreamConnectionInfo);
				foreach (CxsNode item in children)
				{
					if (string.Compare("MaterialStreamObject", item.NodeType, ignoreCase: true) == 0)
					{
						CxsNodeInfo key = new CxsNodeInfo
						{
							path = item.Path,
							isleaf = item.IsLeaf,
							type = item.NodeType,
							id = item.Id,
							containertype = item.ContainerType
						};
						_ = item.Id;
						CxsNode val = item.ResolveNode("Upstream Unit Operations");
						if (val != null && val.HasChildren)
						{
							CxsNode val2 = ((IEnumerable<CxsNode>)val.ChildNodes).FirstOrDefault();
							value.upstreamUo = ((val2 != null) ? val2.Id : null) ?? string.Empty;
						}
						else
						{
							value.upstreamUo = string.Empty;
						}
						CxsNode val3 = item.ResolveNode("Downstream Unit Operations");
						if (val3 != null && val3.HasChildren)
						{
							CxsNode val4 = ((IEnumerable<CxsNode>)val3.ChildNodes).FirstOrDefault();
							value.downstreamUo = ((val4 != null) ? val4.Id : null) ?? string.Empty;
						}
						else
						{
							value.downstreamUo = string.Empty;
						}
						if (!dictionary.ContainsKey(key))
						{
							dictionary.Add(key, value);
						}
					}
				}
			}
			return dictionary;
		}

		{
			if (collection.TryGetValue(uoid, out var value))
			{
				if (!value.Contains(streamInfo))
				{
					value.Add(streamInfo);
				}
				collection[uoid] = value;
			}
			else
			{
				value = new List<CxsNodeInfo> { streamInfo };
				collection.Add(uoid, value);
			}
		}

	public abstract class BindableBase : INotifyPropertyChanged
	{
		private readonly Dictionary<string, PropertyObserver> _propertyObserverContainer = new Dictionary<string, PropertyObserver>();

		public event PropertyChangedEventHandler PropertyChanged;

		protected void OnPropertyChanged([CallerMemberName] string propertyName = null)
		{
			this.PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
		}

		protected virtual bool SetProperty<T>(ref T storage, T value, [CallerMemberName] string propertyName = null)
		{
			if (EqualityComparer<T>.Default.Equals(storage, value))
			{
				return false;
			}
			storage = value;
			OnPropertyChanged(propertyName);
			return true;
		}

		protected PropertyObserver Make(string propertyName)
		{
			if (!_propertyObserverContainer.ContainsKey(propertyName))
			{
				_propertyObserverContainer[propertyName] = new PropertyObserver(delegate
				{
					OnPropertyChanged(propertyName);
				});
			}
			return _propertyObserverContainer[propertyName];
		}
	}

				_uset = value;
			}
		}

		public static bool IsSameValue(double a, double b)
		{
			if (double.IsNaN(a) && double.IsNaN(b))
			{
				return true;
			}
			if (a == -32767.0 && double.IsNaN(b))
			{
				return true;
			}
			if (double.IsNaN(a) || double.IsNaN(b))
			{
				return false;
			}
			if (Math.Abs(a - b) < 1E-08)
			{
				return true;
			}
			return false;
		}
	}

				case "Rated (area)":

				case "Required":
				case "Design":
					return LineRatingFlowSpec.Required;
				case "Manual":
					return LineRatingFlowSpec.Manual;
				default:
					return LineRatingFlowSpec.RatedArea;
				}
			}
			return null;
		}
	}
	public class EnumTranslator<T> where T : struct
	{
		private readonly Dictionary<T, string> _enumStringPairs;

		private readonly HashSet<string> _expectedMissingItems;

		private readonly string _emptyText;

		private readonly bool _isLimited;

		public List<string> StringValues => _enumStringPairs.Values.ToList();

		public List<T> EnumValues => _enumStringPairs.Keys.ToList();

		public EnumTranslator(Dictionary<T, string> pairs, string emptyText, bool isLimited = true, IEnumerable<string> expectedMissingItems = null)
		{
			_enumStringPairs = pairs;
			_emptyText = emptyText;
			_isLimited = isLimited;
			if (expectedMissingItems != null)
			{
				_expectedMissingItems = new HashSet<string>(expectedMissingItems);
			}
		}

		public T? Enum(string s)
		{
			if (_enumStringPairs == null || !_enumStringPairs.ContainsValue(s))
			{
				if (_isLimited && _expectedMissingItems != null)
				{
					_expectedMissingItems.Contains(s);
				}
				return null;
			}
			return _enumStringPairs.SingleOrDefault((KeyValuePair<T, string> p) => p.Value == s).Key;
		}

		public string StringValue(T? e)
		{
			if (!e.HasValue)
			{
				return _emptyText;
			}
			if (!_enumStringPairs.TryGetValue(e.Value, out var value))
			{
				return _emptyText;
			}
			return value;
		}
	}
	public class EnumTranslatorOneWay<T> where T : struct
	{
		private readonly Dictionary<T, string> _enumStringPairs;

		private readonly string _emptyText;

		public EnumTranslatorOneWay(Dictionary<T, string> pairs, string emptyText)
		{
			_enumStringPairs = pairs ?? throw new ArgumentNullException("pairs");
			_emptyText = emptyText;
		}

		public string StringValue(T? e)
		{
			if (!e.HasValue)
			{
				return _emptyText;
			}
			if (!_enumStringPairs.TryGetValue(e.Value, out var value))
			{
				return _emptyText;
			}
			return value;
		}
	}

	public class PropertyObserver
	{
		private readonly HashSet<string> _observedPropertiesExpressions = new HashSet<string>();

		private readonly Action _action;

		public PropertyObserver(Action action)
		{
			_action = action;
		}

		public PropertyObserver ObservesProperty<T>(Expression<Func<T>> propertyExpression)
		{
			if (_observedPropertiesExpressions.Contains(propertyExpression.ToString()))
			{
				throw new ArgumentException(propertyExpression.ToString() + " is already being observed.", "propertyExpression");
			}
			_observedPropertiesExpressions.Add(propertyExpression.ToString());
			PropertyObserverInternal.Observes(propertyExpression, _action);
			return this;
		}
	}
	internal class PropertyObserverInternal
	{
		private readonly Action _action;

		private PropertyObserverInternal(System.Linq.Expressions.Expression propertyExpression, Action action)
		{
			_action = action;
			SubscribeListeners(propertyExpression);
		}

		private void SubscribeListeners(System.Linq.Expressions.Expression propertyExpression)
		{
			Stack<string> stack = new Stack<string>();
			while (propertyExpression is MemberExpression memberExpression)
			{
				propertyExpression = memberExpression.Expression;
				stack.Push(memberExpression.Member.Name);
			}
			if (!(propertyExpression is ConstantExpression constantExpression))
			{
				throw new NotSupportedException("Operation not supported for the given expression type. Only MemberExpression and ConstanteExpression are currently supported.");
			}
			PropertyObserverNodeInternal propertyObserverNodeInternal = new PropertyObserverNodeInternal(stack.Pop(), _action);
			PropertyObserverNodeInternal propertyObserverNodeInternal2 = propertyObserverNodeInternal;
			using (Stack<string>.Enumerator enumerator = stack.GetEnumerator())
			{
				while (enumerator.MoveNext())
				{
					PropertyObserverNodeInternal propertyObserverNodeInternal3 = (propertyObserverNodeInternal2.Next = new PropertyObserverNodeInternal(enumerator.Current, _action));
					propertyObserverNodeInternal2 = propertyObserverNodeInternal3;
				}
			}
			if (!(constantExpression.Value is INotifyPropertyChanged inpcObject))
			{
				throw new InvalidOperationException("Trying to subscribe PropertyChanged listener in object that owns '" + propertyObserverNodeInternal.PropertyName + "' property, but the object does not implements INotifyPropertyChanged.");
			}
			propertyObserverNodeInternal.SubscribeListenerFor(inpcObject);
		}

		internal static PropertyObserverInternal Observes<T>(Expression<Func<T>> propertyExpression, Action action)
		{
			return new PropertyObserverInternal(propertyExpression.Body, action);
		}
	}
	internal class PropertyObserverNodeInternal
	{
		private readonly Action _action;

		private INotifyPropertyChanged _inpcObject;

		public string PropertyName { get; }

		public PropertyObserverNodeInternal Next { get; set; }

		public PropertyObserverNodeInternal(string propertyName, Action action)
		{
			PropertyObserverNodeInternal propertyObserverNodeInternal = this;
			PropertyName = propertyName;
			_action = delegate
			{
				action?.Invoke();
				if (propertyObserverNodeInternal.Next != null)
				{
					propertyObserverNodeInternal.Next.UnsubscribeListener();
					propertyObserverNodeInternal.GenerateNextNode();
				}
			};
		}

		public void SubscribeListenerFor(INotifyPropertyChanged inpcObject)
		{
			_inpcObject = inpcObject;
			_inpcObject.PropertyChanged += OnPropertyChanged;
			if (Next != null)
			{
				GenerateNextNode();
			}
		}

		private void GenerateNextNode()
		{
			Type type = _inpcObject.GetType();
			object value = (type.GetRuntimeProperty(PropertyName) ?? type.GetProperty(PropertyName, BindingFlags.Instance | BindingFlags.NonPublic)).GetValue(_inpcObject);
			if (value != null)
			{
				if (!(value is INotifyPropertyChanged inpcObject))
				{
					throw new InvalidOperationException("Trying to subscribe PropertyChanged listener in object that owns '" + Next.PropertyName + "' property, but the object does not implements INotifyPropertyChanged.");
				}
				Next.SubscribeListenerFor(inpcObject);
			}
		}

		private void UnsubscribeListener()
		{
			if (_inpcObject != null)
			{
				_inpcObject.PropertyChanged -= OnPropertyChanged;
			}
			Next?.UnsubscribeListener();
		}

		private void OnPropertyChanged(object sender, PropertyChangedEventArgs e)
		{
			if (e?.PropertyName == PropertyName || e == null || e.PropertyName == null)
			{
				_action?.Invoke();
			}
		}
	}

	public abstract class PsvKey
	{
		protected eRecordType _recordType;

		protected int _value = -1;

		public eRecordType RecordType => _recordType;

		public int Value => _value;

		internal PsvKey(eRecordType recordType, int newKey)
		{
			_recordType = recordType;
			_value = newKey;
		}

		public PsvKey()
		{
		}

		public override bool Equals(object obj)
		{
			PsvKey psvKey = obj as PsvKey;
			if (psvKey == null)
			{
				return false;
			}
			if (psvKey._recordType != _recordType)
			{
				return false;
			}
			if (psvKey._value != _value)
			{
				return false;
			}
			return true;
		}

		public static bool operator ==(PsvKey lhs, PsvKey rhs)
		{
			return lhs?.Equals(rhs) ?? ((object)lhs == rhs);
		}

		public static bool operator !=(PsvKey lhs, PsvKey rhs)
		{
			return !(lhs == rhs);
		}

		public override int GetHashCode()
		{
			return _recordType.GetHashCode() ^ _value.GetHashCode();
		}

		public PsvModelKey AsPsvModelKey()
		{
			if (this is PsvModelKey)
			{
				return this as PsvModelKey;
			}
			return new PsvModelKey(_recordType, _value);
		}
	}
