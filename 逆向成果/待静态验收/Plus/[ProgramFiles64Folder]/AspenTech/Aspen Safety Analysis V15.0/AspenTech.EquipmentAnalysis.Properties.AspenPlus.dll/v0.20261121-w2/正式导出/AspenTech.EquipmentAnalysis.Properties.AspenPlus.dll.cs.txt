		public bool SetLiquidVaporRatio(double VF)
		{
			if (_aprOverall == null)
			{
				return false;
			}
			if (_aprVapor == null)
			{
				return false;
			}
			if (_aprLiquid == null)
			{
				return false;
			}
			int[] array = new int[1];
			int[] array2 = new int[1];
			int[] array3 = new int[1];
			try
			{
				_aprOverall.GetIntegerPhaseVariable(cPHASEVARIABLE.cpvNumComp, array, 1);
				if (array[0] == 99999999)
				{
					return false;
				}
				_aprVapor.GetIntegerPhaseVariable(cPHASEVARIABLE.cpvNumComp, array3, 1);
				if (array3[0] == 99999999)
				{
					return false;
				}
				_aprLiquid.GetIntegerPhaseVariable(cPHASEVARIABLE.cpvNumComp, array2, 1);
				if (array2[0] == 99999999)
				{
					return false;
				}
				if (array[0] != array3[0] || array[0] != array2[0])
				{
					return false;
				}
				double[] array4 = new double[array3[0]];
				double[] array5 = new double[array2[0]];
				_aprVapor.GetRealPhaseVariable(cPHASEVARIABLE.cpvMoleFracs, array4, array3[0]);
				_aprLiquid.GetRealPhaseVariable(cPHASEVARIABLE.cpvMoleFracs, array5, array2[0]);
				switch (PSVDictionaries.GetPhaseFromVaporFraction(VF))
				{
				case FluidPhaseRegime.Vapor:
					_aprOverall.SetRealPhaseVariable(cPHASEVARIABLE.cpvMoleFracs, array4, array3[0]);
					break;
				case FluidPhaseRegime.Liquid:
					_aprOverall.SetRealPhaseVariable(cPHASEVARIABLE.cpvMoleFracs, array5, array2[0]);
					break;
				default:
				{
					double[] array6 = new double[array[0]];
					for (int i = 0; i < array[0]; i++)
					{
						array6[i] = array4[i] * VF + array5[i] * (1.0 - VF);
					}
					_aprOverall.SetRealPhaseVariable(cPHASEVARIABLE.cpvMoleFracs, array6, array[0]);
					break;
				}
				}
			}
			catch
			{
				return false;
			}
			return true;
		}
