		private void _Initialize(string path)
		{
			_path = path;
			string text = path;
			if (text.EndsWith("Fluid"))
			{
				text = path.Substring(0, path.Length - 6);
			}
			CxvActionResult val = CxsCaseHelper.MyDataProvider.InvokeAction(text, "GetDispatch", false, (object[])null);
			if (val.Result is PSVEmergScenario2 refScenario)
			{
				_refScenario = refScenario;
				_refFluid = (dynamic)_refScenario.WorkingFluid;
				return;
			}
			if (val.Result is PSVStorageTank refTank)
			{
				_refTank = refTank;
				_refFluid = (dynamic)_refTank.WorkingFluid;
				return;
			}
			throw new InvalidOperationException("Internal error: Dispatch object from path could not be converted to a PSVEmergScenario or PSVStorageTank.");
		}

		{
			if (!_lastFlashSucceeded)
			{
				return false;
			}
			if (Pressure() == pout)
			{
				return true;
			}
			_lastFlashSucceeded = false;
			if (double.IsNaN(pout))
			{
				return false;
			}
			bool num = _refFluid.IsenthalpicFlash(pout);
			if (num)
			{
				_lastFlashSucceeded = true;
			}
			return num;
		}
