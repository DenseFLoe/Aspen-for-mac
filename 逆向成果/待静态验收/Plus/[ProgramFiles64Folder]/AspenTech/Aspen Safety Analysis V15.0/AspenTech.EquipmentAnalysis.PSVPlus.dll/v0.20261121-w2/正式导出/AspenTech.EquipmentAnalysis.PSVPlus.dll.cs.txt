		public bool RenameCase(string oldPathName, string newPathName)
		{
			if (!Cases.Keys.Contains(oldPathName) | Cases.Keys.Contains(newPathName))
			{
				return false;
			}
			IPsvCase value = Cases[oldPathName];
			bool flag = Cases.TryRemove(oldPathName, out value);
			if (flag)
			{
				flag = Cases.TryAdd(newPathName, value);
			}
			return flag;
		}
	}
