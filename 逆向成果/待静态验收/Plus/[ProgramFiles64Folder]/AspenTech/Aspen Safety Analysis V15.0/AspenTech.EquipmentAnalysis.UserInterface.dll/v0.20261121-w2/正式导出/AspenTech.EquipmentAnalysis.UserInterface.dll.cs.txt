		public override void InitializeScreen(IUnityContainer container, ScreenSubject screenSubjectOpaque, ScreenState screenStateOpaque)
		{
			((ScreenContentControl)this).InitializeScreen(container, screenSubjectOpaque, screenStateOpaque);
			((ScreenContentControl)this).AllowFloating = false;
			((ScreenContentControl)this).AllowCloning = false;
			PSVApplication pSVApplication = UnityContainerExtensions.Resolve<PSVApplication>(((ScreenContentControl)this).UnityContainer, Array.Empty<ResolverOverride>());
			if (pSVApplication.ActiveCase == null)
			{
				IPSVMessager iPSVMessager = UnityContainerExtensions.Resolve<IPSVMessager>(container, Array.Empty<ResolverOverride>());
				if (iPSVMessager != null && iPSVMessager.PSVHostType == "ASPENPLUS")
				{
					iPSVMessager.InitializePSVPlusIfNot();
				}
			}
			_mycase = pSVApplication.ActiveCase;
			PSVScreenSubject pSVScreenSubject = screenSubjectOpaque as PSVScreenSubject;
			((ScreenContentControl)this).Header = pSVScreenSubject.TabLabel;
			InitializeFormContent(pSVScreenSubject);
			DoInitialization();
			((ScreenContentControl)this).ShouldApplyZoomToTransform = false;
			((ScreenContentControl)this).NavigationGroupKey = "PSVPlus";
		}

		private void InitializeFormContent(PSVScreenSubject formSubject)
		{
			RegisterEvents();
			_myControl = (BaseUserControl)_mycase.GetUserControl(formSubject.Id, formSubject.ModelKey);
			if (_myControl is ErrorControl)
			{
				((ContentControl)this).Content = _myControl;
				return;
			}
			if (formSubject.Id == PSVctrlIDs.ctrlVapRelief)
			{
				PSVScenarioControl obj = _myControl as PSVScenarioControl;
				ScenarioScreenSubject scenarioScreenSubject = formSubject as ScenarioScreenSubject;
				obj.ModelKey = scenarioScreenSubject.ModelKey;
				obj.NodeKey = scenarioScreenSubject.NodeKey;
			}
			else if (formSubject.Id == PSVctrlIDs.ctrlPSVSummaryForm)
			{
				PRDSummary obj2 = _myControl as PRDSummary;
				PSVSummaryScreenSubject pSVSummaryScreenSubject = formSubject as PSVSummaryScreenSubject;
				obj2.NodeKey = pSVSummaryScreenSubject.NodeKey;
				obj2.PSVkey = pSVSummaryScreenSubject.ModelKey;
				obj2.PSVName = pSVSummaryScreenSubject.PSVName;
			}
			else if (formSubject.Id == PSVctrlIDs.ctrlTankVent)
			{
				PSVTank pSVTank = _myControl as PSVTank;
				PSVTankScreenSubject pSVTankScreenSubject = formSubject as PSVTankScreenSubject;
				if (pSVTankScreenSubject.NodePath == null || pSVTankScreenSubject.NodeKey.Equals(pSVTankScreenSubject.NodePath))
				{
					pSVTank.NodeKey = pSVTankScreenSubject.NodeKey;
					pSVTank.TankKey = pSVTankScreenSubject.ModelKey;
				}
				else
				{
					_myControl = null;
				}
			}
			else
			{
				_ = formSubject.Id;
				_ = 26;
			}
			((ContentControl)this).Content = _myControl;
		}

		private void OnOwnerStreamChanged(PSVOwnerStreamChangedEventArgs args)
		{
			PSVScreenSubject pSVScreenSubject = ((ScreenContentControl)this).ScreenSubject as PSVScreenSubject;
			if (pSVScreenSubject.Id == PSVctrlIDs.ctrlPSVSummaryForm)
			{
				if (_myControl is PRDSummary)
				{
					PRDSummary pRDSummary = _myControl as PRDSummary;
					if (pRDSummary.PSVkey == args.PSVKey && string.Compare(pRDSummary.NodeKey, args.NewCxsPath, ignoreCase: true) != 0)
					{
						pRDSummary.UpdateNodeKey(args.NewCxsPath);
					}
				}
			}
			else if (pSVScreenSubject.Id == PSVctrlIDs.ctrlVapRelief && _myControl is PSVScenarioControl)
			{
				PSVScenarioControl pSVScenarioControl = _myControl as PSVScenarioControl;
				if (string.Compare(pSVScenarioControl.NodeKey, args.OldCxsPath, ignoreCase: true) == 0)
				{
					pSVScenarioControl.UpdateNodeKey(args.NewCxsPath);
				}
			}
		}

		private void OnLineSizingSettingsChanged(LineSizingSettingsChangedEventArgs args)
		{
			if (!((ScreenContentControl)(object)this).IsActiveScreen() || !(((ScreenContentControl)this).ScreenSubject is PSVScreenSubject pSVScreenSubject))
			{
				return;
			}
			if (pSVScreenSubject.Id == PSVctrlIDs.ctrlVapRelief)
			{
				if (_myControl is PSVScenarioControl)
				{
					(_myControl as PSVScenarioControl).OnLineSizingSettingsChanged();
				}
			}
			else if (pSVScreenSubject.Id == PSVctrlIDs.ctrlPSVSummaryForm && _myControl is PRDSummary pRDSummary)
			{
				pRDSummary.UpdateRatingTabSelOriCombo();
			}
		}

			}
		}

		private void OnNameChanged(RenameEventArgs args)
		{
			if (!((ScreenContentControl)(object)this).IsActiveScreen() || !(((ScreenContentControl)this).ScreenSubject is PSVScreenSubject pSVScreenSubject))
			{
				return;
			}
			IPSVMessager iPSVMessager = UnityContainerExtensions.Resolve<IPSVMessager>(((ScreenContentControl)this).UnityContainer, Array.Empty<ResolverOverride>());
			if (pSVScreenSubject.Id == PSVctrlIDs.ctrlVapRelief)
			{
				if (!(pSVScreenSubject is ScenarioScreenSubject scenarioScreenSubject))
				{
					return;
				}
				string header = ((ScreenContentControl)this).Header;
				string text = null;
				bool flag = false;
				if (string.Compare(args.ObjectType, "ScenarioName", ignoreCase: false) == 0 && object.Equals(pSVScreenSubject.ModelKey, args.Key))
				{
					text = Regex.Replace(header, "-" + args.OldName + "$", "-" + args.NewName);
					scenarioScreenSubject.ScenarioName = args.NewName;
					scenarioScreenSubject.TabLabel = text;
					if (iPSVMessager != null && iPSVMessager.PSVHostType != "ASPENPLUS")
					{
						pSVScreenSubject.NodePath = NavigatorNodeHelper.NodesCache[args.Key.Value.ToString()][1];
					}
					flag = true;
				}
				else if (string.Compare(args.ObjectType, "ValveName", ignoreCase: true) == 0 && args.Scenarios.Contains(pSVScreenSubject.ModelKey))
				{
					text = ((iPSVMessager == null || !iPSVMessager.PSVHostType.Equals("ASPENPLUS")) ? $"{args.NewName}-{scenarioScreenSubject.ScenarioName}" : (args.NewName + "-" + scenarioScreenSubject.ScenarioName));
					flag = true;
				}
				if (flag && string.Compare(header, text, ignoreCase: true) != 0)
				{
					((ScreenContentControl)this).Header = text;
					pSVScreenSubject.TabLabel = text;
				}
			}
			if (pSVScreenSubject.Id == PSVctrlIDs.ctrlPSVSummaryForm && string.Compare(args.ObjectType, "ValveName", ignoreCase: true) == 0 && object.Equals(pSVScreenSubject.ModelKey, args.Key))
			{
				string newName = args.NewName;
				if (string.Compare(((ScreenContentControl)this).Header, newName, ignoreCase: true) != 0)
				{
					((ScreenContentControl)this).Header = newName;
					pSVScreenSubject.TabLabel = newName;
					((PSVSummaryScreenSubject)pSVScreenSubject).PSVName = newName;
				}
				if (_myControl is PRDSummary pRDSummary)
				{
					pRDSummary.ValveNameChanged();
				}
				if (iPSVMessager != null && iPSVMessager.PSVHostType.Equals("ASPENPLUS") && pSVScreenSubject is PSVSummaryScreenSubject pSVSummaryScreenSubject && !string.IsNullOrEmpty(args.Path))
				{
					int num = args.Path.IndexOf(".PSValve.");
					if (num >= 0)
					{
						pSVSummaryScreenSubject.NodeKey = args.Path.Substring(0, num + 8) + "." + newName;
						pSVSummaryScreenSubject.TabLabel = newName;
						pSVSummaryScreenSubject.PSVName = newName;
					}
				}
			}
			if (pSVScreenSubject.Id == PSVctrlIDs.ctrlTankVent && string.Compare(args.ObjectType, "TankName", ignoreCase: true) == 0 && object.Equals(pSVScreenSubject.ModelKey, args.Key) && pSVScreenSubject is PSVTankScreenSubject pSVTankScreenSubject && pSVTankScreenSubject.ModelKey == args.Key)
			{
				pSVTankScreenSubject.TankName = args.NewName;
				((ScreenContentControl)this).Header = args.NewName;
				pSVScreenSubject.NodePath = args.Path;
				pSVScreenSubject.NodeKey = args.Path;
				pSVScreenSubject.TabLabel = args.NewName;
			}
		}

		private static bool IsScreenInvalid(IUnityContainer container, ScreenSubject screenSubject, ScreenState screenState)
		{
			try
			{
				if (screenState != null && screenState.IsRestoreFromSavedLayout)
				{
					IPSVMessager iPSVMessager = UnityContainerExtensions.Resolve<IPSVMessager>(container, Array.Empty<ResolverOverride>());
					if (iPSVMessager != null)
					{
						if (iPSVMessager.PSVHostType == "ASPENPLUS")
						{
							string text = null;
							ScenarioScreenSubject scenarioScreenSubject = screenSubject as ScenarioScreenSubject;
							PSVSummaryScreenSubject pSVSummaryScreenSubject = screenSubject as PSVSummaryScreenSubject;
							if (scenarioScreenSubject != null)
							{
								text = scenarioScreenSubject.NodeKey;
								if (!string.IsNullOrEmpty(text) && !string.IsNullOrEmpty(scenarioScreenSubject.ScenarioName))
								{
									int num = text.IndexOf(".PSValve.");
									Collection<string> collection = CxsConverter.ValueToStringCollection(CxsCaseHelper.MyDataProvider.BatchGet(text.Substring(0, num) + ".PSVScenario", "child_ids", (string)null));
									if (collection != null && collection.Contains(scenarioScreenSubject.ScenarioName))
									{
										string text2 = text.Substring(num + 9);
										if (num > 0 && !string.IsNullOrEmpty(text2))
										{
											Collection<string> collection2 = CxsConverter.ValueToStringCollection(CxsCaseHelper.MyDataProvider.BatchGet(text.Substring(0, num) + ".PSValve", "child_ids", (string)null));
											if (collection2 == null || !collection2.Contains(text2))
											{
												return true;
											}
										}
										return false;
									}
									return true;
								}
							}
							else if (pSVSummaryScreenSubject != null)
							{
								text = pSVSummaryScreenSubject.NodeKey;
							}
							if (!string.IsNullOrEmpty(text))
							{
								int num2 = text.IndexOf(".PSValve.");
								string text3 = text.Substring(num2 + 9);
								if (num2 > 0 && !string.IsNullOrEmpty(text3))
								{
									Collection<string> collection3 = CxsConverter.ValueToStringCollection(CxsCaseHelper.MyDataProvider.BatchGet(text.Substring(0, num2) + ".PSValve", "child_ids", (string)null));
									if (collection3 == null || !collection3.Contains(text3))
									{
										return true;
									}
								}
							}
						}
						else
						{
							IPsvCase activeCase = UnityContainerExtensions.Resolve<PSVApplication>(container, Array.Empty<ResolverOverride>()).ActiveCase;
							PSVScreenSubject pSVScreenSubject = screenSubject as PSVScreenSubject;
							if (activeCase != null && pSVScreenSubject?.ModelKey != null)
							{
								switch (pSVScreenSubject.ModelKey.RecordType)
								{
								case eRecordType.PRDSystem:
									if (activeCase.GetPRDSystemModel(pSVScreenSubject.ModelKey) == null)
									{
										return true;
									}
									break;
								case eRecordType.OrificeCalc:
									if (activeCase.GetPSVScenarioModel(pSVScreenSubject.ModelKey) == null)
									{
										return true;
									}
									break;
								}
							}
						}
					}
				}
				return false;
			}
			catch (IndexOutOfRangeException)
			{
				return false;
			}
			catch (Exception)
			{
				return false;
			}
		}

		public static bool IsActiveScreen(this ScreenContentControl screen)
		{
			if (screen == null)
			{
				throw new ArgumentNullException("screen");
			}
			if (UnityContainerExtensions.Resolve<IShellApplication>(screen.UnityContainer, Array.Empty<ResolverOverride>()).ActiveWindow.FindScreen((Func<ScreenContentControl, bool>)((ScreenContentControl s) => ((object)s).Equals((object)screen))) != null)
			{

		}

		public static string GenerateVisualNodePath(this PSVRootNode node)
		{
			string text = ((SimpleNavigatorTreeNode)node).Text.Trim();
			while (node.Parent != null)
			{
				text = ((SimpleNavigatorTreeNode)node.Parent).Text + "." + text;
				node = node.Parent;
			}
			return text;
		}

		public static bool IsEmpty(this double val)
		{
			if (val != -32767.0)
			{
				return false;
			}
			return true;
		}

		public static OrificeSizingMethod GetPhase(this IPPPFluid package)
		{
			OrificeSizingMethod result = OrificeSizingMethod.Mixed_Direct;
			double num = package.VaporFraction();
			if (!num.IsEmpty())
			{
				if (num >= 0.999999)
				{
					result = OrificeSizingMethod.Vapor;
				}
				else if (num <= 1E-06)
				{
					result = OrificeSizingMethod.Liquid;
				}
			}
			return result;
		}
	}

				AppendContextMenuItem(targetNavigatorView, targetNode, menu, NavPaneCommand_AddScenario);
				AppendContextMenuItem(targetNavigatorView, targetNode, menu, NavPaneCommand_Copy);
				AppendContextMenuItem(targetNavigatorView, targetNode, menu, NavPaneCommand_Delete, insertSeparatorFirst: true);

			case "EmScenario":

				break;

				menu.Items.Clear();
				break;
			case "Subflowsheet":
				break;
			case "ColSubflowsheet":
				break;
			}
		}

		private static void AppendCreatePSVMenuItem(PSVRootNode targetNode, ContextMenu menu)
		{
			if (targetNode == null || string.IsNullOrEmpty(targetNode.DataPath))
			{
				return;
			}
			for (int i = 0; i < 2; i++)
			{
				MenuItem menuItem = new MenuItem();
				if (i == 0)
				{
					menuItem.Header = FormResource.ContextMenu_AddNewPSV;
				}
				else
				{
					menuItem.Header = FormResource.ContextMenu_AddRD;
				}
				foreach (CxsNodeInfo item in (targetNode as UONode).Products.OrderBy((CxsNodeInfo p) => p.id))
				{
					if (!string.IsNullOrEmpty(item.path))
					{
						if (i == 0)
						{
							menuItem.Items.Add(CreatePSVSceondLevelMenuItem(item, targetNode, bpsv: true));
						}
						else
						{
							menuItem.Items.Add(CreatePSVSceondLevelMenuItem(item, targetNode, bpsv: false));
						}
					}
				}
				menu.Items.Add(menuItem);
			}
		}

		private static void AppendPasteMenuItem(PSVRootNode node, ContextMenu menu)
		{
			if (node == null || string.IsNullOrEmpty(node.DataPath))
			{
				return;
			}
			MenuItem menuItem = new MenuItem();
			menuItem.Header = FormResource.ContextMenu_Paste;
			if (!NavPaneCommand_Paste_CanExecute(node))
			{
				menuItem.IsEnabled = false;
			}
			else
			{
				List<CxsNodeInfo> outlets = (node as UONode).Products;
				if (outlets.Count > 1)
				{
					foreach (CxsNodeInfo outlet in outlets.OrderBy((CxsNodeInfo p) => p.id))
					{
						if (!string.IsNullOrEmpty(outlet.path))
						{
							MenuItem menuItem2 = new MenuItem();
							menuItem2.Header = outlet.id;
							menuItem2.Click += delegate
							{
								PasteNode(NavigatorNodeHelper.CopiedPRDNode, outlet.path);
							};
							menuItem.Items.Add(menuItem2);
						}
					}
				}
				else
				{
					menuItem.Click += delegate
					{
						PasteNode(NavigatorNodeHelper.CopiedPRDNode, outlets[0].path);
					};
				}
			}
			menu.Items.Add(menuItem);
		}

	public class PSVRootNode : EquipmentAnalysisNode
	{
		private bool _isExpanded;

		public PSVRootNode Parent { get; set; }

		public string DataPath { get; set; }

		public bool? IsLeaf { get; set; }

		public string NodeType { get; set; }

		public Dictionary<string, List<CxsNodeInfo>> uoInletsMap { get; set; }

		public Dictionary<string, List<CxsNodeInfo>> uoOutletsMap { get; set; }

		public List<CxsNodeInfo> subunattached { get; set; }

		private List<CxsNodeInfo> subUnitOps { get; set; }

		protected override eRecordType RecordType => eRecordType.PRDSystem;

		public override bool IsExpanded
		{
			get
			{
				return _isExpanded;
			}
			set
			{
				if (value != _isExpanded)
				{
					_isExpanded = value;
					((SimpleNavigatorTreeNode)this).OnIsExpandedChanged();
				}
			}
		}

		public PSVPlusNavigatorViewModel NavigatorVM { get; set; }

		public static event EventHandler IsExpandedChanged;

		protected override void OnIsExpandedChanged()
		{
			RefreshChildren();
			if (PSVRootNode.IsExpandedChanged != null)
			{
				PSVRootNode.IsExpandedChanged(this, EventArgs.Empty);
			}
			((SimpleNavigatorTreeNode)this).OnIsExpandedChanged();
		}

		public void NotifyChilrenChanged()
		{
			((SimpleNavigatorTreeNode)this).OnPropertyChanged("Children");
		}

		public virtual void RefreshChildren()
		{
			((Collection<NavigatorTreeNode>)(object)((NavigatorTreeNode)this).Children).Clear();
			if (!IsLeaf.HasValue || !IsLeaf.Value)
			{
				AppendChildrenToParent(this);
			}
		}

		public virtual void InitializeNode(CxsNodeInfo info)
		{
			DataPath = info.path;
			((SimpleNavigatorTreeNode)this).Text = info.id;
			IsLeaf = info.isleaf;
			((SimpleNavigatorTreeNode)this).UnityContainer = ((SimpleNavigatorTreeNode)Parent).UnityContainer;
			if (NodeType == "PRD" && !NavigatorNodeHelper.NodesCache.ContainsKey(info.id))
			{
				NavigatorNodeHelper.NodesCache.Add(info.id, new List<string>
				{
					info.path,
					this.GenerateVisualNodePath()
				});
			}
		}

		private void AppendChildrenToParent(PSVRootNode father)
		{
			if (father.NodeType == "TopLevelUOs")
			{
				List<CxsNodeInfo> list = AspenCaseDataRequest.BuildNodeInfo("appModel.FlowSheetObjects.UnitOps", CxsCaseHelper.MyCase);
				if (list != null && list.Count > 0)
				{
					CreateUOChild(father, list, father.NavigatorVM.uoFeedDic, father.NavigatorVM.uoOutletDic);
				}
			}
			else if (father.NodeType == "UO")
			{
				AppendPSVToUo((UONode)father);
			}
			else if (father.NodeType == "PRD")
			{
				AppendScenarioToPSV((PSVNode)father);
			}
			else if (father.NodeType == "TopUnattachedStreams")
			{
				List<CxsNodeInfo> standaloneStreams = father.NavigatorVM.standaloneStreams;
				if (standaloneStreams != null && standaloneStreams.Count > 0)
				{
					CreateStreamChild(father, standaloneStreams, "TopUnattachedStreams", bcreateRoot: false, bcreatechild: true);
				}
			}
			else if (father.NodeType == "UnattachedStreams")
			{
				AppendPSVToStream(father);
			}
			else if (father.NodeType == "Subflowsheet")
			{
				(uoInletsMap, uoOutletsMap, subunattached) = CxsNodeInfoHelper.CollectUoAndStream(CxsCaseHelper.MyCase, $"{father.DataPath}.Streams");
				List<CxsNodeInfo> list2 = (subUnitOps = AspenCaseDataRequest.BuildNodeInfo($"{father.DataPath}.UnitOps", CxsCaseHelper.MyCase));
				if (list2 != null && list2.Count > 0)
				{
					CreateUOChild(father, list2, uoInletsMap, uoOutletsMap, bcreateRoot: true);
				}
				if (subunattached != null && subunattached.Count > 0)
				{
					CreateStreamChild(father, subunattached, "SubflowsheetUnattachedStreams", bcreateRoot: true, bcreatechild: false);
				}
				AppendPSVToUo((UONode)father);
			}
			else if (father.NodeType == "ColSubflowsheet")
			{
				(uoInletsMap, uoOutletsMap, subunattached) = CxsNodeInfoHelper.CollectUoAndStream(CxsCaseHelper.MyCase, $"{father.DataPath}.My SubFlowsheet.Streams");
				List<CxsNodeInfo> list4 = (subUnitOps = AspenCaseDataRequest.BuildNodeInfo($"{father.DataPath}.My SubFlowsheet.UnitOps", CxsCaseHelper.MyCase));
				if (list4 != null && list4.Count > 0)
				{
					CreateUOChild(father, list4, uoInletsMap, uoOutletsMap, bcreateRoot: true);
				}
				if (subunattached != null && subunattached.Count > 0)
				{
					CreateStreamChild(father, subunattached, "ColSubflowsheetUnattachedStreams", bcreateRoot: true, bcreatechild: false);
				}

			}

			{
				CreateStreamChild(father, father.Parent.subunattached, "UnattachedStreams", bcreateRoot: false, bcreatechild: true);
			}
		}

		private void CreateUOChild(PSVRootNode parent, List<CxsNodeInfo> uoNodes, Dictionary<string, List<CxsNodeInfo>> inletMap, Dictionary<string, List<CxsNodeInfo>> outletMap, bool bcreateRoot = false)
		{
			PSVRootNode pSVRootNode = parent;
			if ((parent.NodeType == "Subflowsheet" || parent.NodeType == "ColSubflowsheet") && bcreateRoot)
			{
				pSVRootNode = UnityContainerExtensions.Resolve<PSVRootNode>(((SimpleNavigatorTreeNode)this).UnityContainer, Array.Empty<ResolverOverride>());

				pSVRootNode.Parent = parent;
			}
			for (int i = 0; i < uoNodes.Count; i++)
			{
				CxsNodeInfo info = uoNodes[i];
				string type = info.type;
				string id = info.id;
				if (string.IsNullOrEmpty(type) || string.IsNullOrEmpty(id) || ShouldNoUONode(type))
				{
					continue;
				}

				{
					((Collection<NavigatorTreeNode>)(object)((NavigatorTreeNode)pSVRootNode).Children).Add((NavigatorTreeNode)(object)uONode);
					uONode.Parent = pSVRootNode;
					if (IsFlowSheetType(type))
					{
						uONode.NodeType = "Subflowsheet";
					}
					else if (IsColFlowSheetType(type))
					{
						uONode.NodeType = "ColSubflowsheet";
					}
					else
					{
						uONode.NodeType = "UO";
					}
					if (outletMap.TryGetValue(id, out var value))
					{
						uONode.Products = value;
					}
					if (inletMap.TryGetValue(id, out var value2))
					{
						uONode.Feeds = value2;
					}
					uONode.InitializeNode(info);
					if (uONode.NodeType == "Subflowsheet" || uONode.NodeType == "ColSubflowsheet")
					{
						((NavigatorTreeNode)uONode).IsExpanded = true;
					}
				}
			}
		}

				int num = pSVTreeNode.PsvPath.LastIndexOf('.');
				string text = pSVTreeNode.PsvPath.Substring(0, num);
				object[] array = new object[1] { pSVTreeNode.PsvPath.Substring(num + 1) };
				CxsCaseHelper.MyDataProvider.InvokeAction(text, "CopyChildren", false, array);
			}
		}

		[DllImport("user32.dll")]
		public static extern IntPtr GetClipboardOwner();

		[DllImport("user32.dll", SetLastError = true)]
		private static extern uint GetWindowThreadProcessId(IntPtr hWnd, out uint lpdwProcessId);

		public static void CanExecutePasteCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			if (_copiedPSVTreeNode != null && UnityContainerExtensions.Resolve<PSVApplication>(((SimpleNavigatorTreeNode)_copiedPSVTreeNode).UnityContainer, Array.Empty<ResolverOverride>()).GetModelKeyByName(_copiedPSVTreeNode.PsvPath) != null && Clipboard.ContainsData("Aspen Plus Selection") && Clipboard.ContainsData("Aspen Plus Selection Type"))
			{
				GetWindowThreadProcessId(GetClipboardOwner(), out var lpdwProcessId);
				if (Process.GetCurrentProcess().Id == lpdwProcessId)
				{
					IDataObject dataObject = Clipboard.GetDataObject();
					if (dataObject != null && dataObject.GetData("Aspen Plus Selection Type", autoConvert: false) is MemoryStream stream)
					{
						string text = new StreamReader(stream).ReadToEnd();
						if (text.Equals("PSVALVE") || text.StartsWith("PSVALVE "))
						{
							e.CanExecute = true;
							return;
						}
					}
				}
			}
			e.CanExecute = false;
		}

		public static void ExecutePasteCommand(object sender, ExecutedRoutedEventArgs e)
		{
			string text = e.Parameter as string;
			if (string.IsNullOrEmpty(text))
			{
				return;
			}
			TreeNavigatorViewTreeViewItem val = (TreeNavigatorViewTreeViewItem)((sender is TreeNavigatorViewTreeViewItem) ? sender : null);
			if (val == null)
			{
				return;
			}
			CommonTreeNode cNode = null;
			bool prdTypeIsPSV = false;
			if (IsPRDAttached(val, text, out cNode, out prdTypeIsPSV))
			{
				if (cNode != null && cNode.Messager != null)
				{
					string applicationName = cNode.Messager.ApplicationName;
					cNode.Messager.ShowMessageBox(Window.GetWindow((DependencyObject)(object)val), prdTypeIsPSV ? FormResource.PsvAlreadyAttachedToStream : FormResource.RDAlreadyAttachedToStream, applicationName, MessageBoxButton.OK, MessageBoxImage.Exclamation, MessageBoxResult.OK, MessageBoxOptions.None);
				}
				return;
			}
			if (cNode != null)
			{
				cNode.IsAddingNodeFromNavTree = true;
			}
			int length = text.LastIndexOf(".Streams.");
			string text2 = text.Substring(0, length) + ".PSValve";
			if (!CxsCaseHelper.MyDataProvider.InvokeAction(text2, "DoPaste", false, Array.Empty<object>()).Succeeded)
			{
				return;
			}
			PSVTreeNode copiedPSVTreeNode = _copiedPSVTreeNode;
			PsvModelKey modelKeyByName = UnityContainerExtensions.Resolve<PSVApplication>(((SimpleNavigatorTreeNode)copiedPSVTreeNode).UnityContainer, Array.Empty<ResolverOverride>()).GetModelKeyByName(copiedPSVTreeNode.PsvPath);
			if (!(modelKeyByName == null))
			{
				if (copiedPSVTreeNode.PRDType == ReliefDeviceType.Psv)
				{
					CxsCaseHelper.MyDataProvider.InvokeAction(text, "SendMessage", false, new object[1] { $"CopyPSVSizing {modelKeyByName.Value}" });
				}
				else
				{
					CxsCaseHelper.MyDataProvider.InvokeAction(text, "SendMessage", false, new object[1] { $"CopyRuptureDisk {modelKeyByName.Value}" });
				}
			}
		}
	}
