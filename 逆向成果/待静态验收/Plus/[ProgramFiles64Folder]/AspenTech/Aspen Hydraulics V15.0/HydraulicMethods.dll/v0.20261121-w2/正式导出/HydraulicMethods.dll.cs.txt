		public static double ChurchillFFF(double Re, double relRoughness)
		{
			double y = 16.0;
			double num = 12.0;
			try
			{
				double num2 = Math.Pow(7.0 / Re, 0.9) + 0.27 * relRoughness;
				double num3 = Math.Pow(37530.0 / Re, y);
				double num4 = Math.Pow(2.457 * Math.Log(1.0 / num2), y);
				double x = Math.Pow(8.0 / Re, num) + Math.Pow(num4 + num3, -1.5);
				return 2.0 * Math.Pow(x, 1.0 / num);
			}
			catch (Exception innerException)
			{
				throw new Exception(Resources.ChurchillFFFFailed, innerException);
			}
		}

		public static double aDPDZcorrection(double velocity1, double velocity2, double density, double pressure)
		{
			double num = velocity1 * velocity2 * density / pressure;
			if (num >= 1.0 || num < 0.0)
			{
				return RealValue.FEMPTY;
			}
			return 1.0 - num;
		}

		public static void GetSupVelocities(IFlowStream stream, double diameter, ref double vsg, ref double vsl, ref double vm)
		{
			IPhase phaseByType = stream.GetPhaseByType(PhaseType.Vapour);
			IPhase combinedLiquidPhase = stream.CombinedLiquidPhase;
			double num = stream.MassFlowrate.Value / HTFSUtility.FlowArea(diameter);
			vsg = num * phaseByType.MassFraction.Value / phaseByType.Density.Value;
			vsl = num * combinedLiquidPhase.MassFraction.Value / combinedLiquidPhase.Density.Value;
			vm = vsg + vsl;
		}

		public static double BeggsAndBrillFrictionFactor(double Re, double laml, double vf, double relRoughness)
		{
			double num = -0.0523;
			double num2 = 3.182;
			double num3 = -0.8725;
			double num4 = 0.01853;
			double num5 = 1.0 - vf;
			if (num5 <= 0.0)
			{
				throw new Exception(Resources.BeggsBrillPgNegativeHoldup);
			}
			double num6 = laml / (num5 * num5);
			double num7 = 0.0;
			if (num6 >= 1.0 && num6 <= 1.2)
			{
				num7 = Math.Log(2.2 * num6 - 1.2);
			}
			else
			{
				double num8 = Math.Log(num6);
				num7 = num8 / (num + num2 * num8 + num3 * num8 * num8 + num4 * Math.Pow(num8, 4.0));
			}
			return Math.Exp(num7) * ChurchillFFF(Re, relRoughness);
		}

		public double CalculateVoidFraction(IFlowStream Stream, double diameter, double inclinationRads, double Roughness, VoidFractionMethodEnum vfcor)
		{
			try
			{
				IPhase phaseByType = Stream.GetPhaseByType(PhaseType.Overall);
				IPhase combinedLiquidPhase = Stream.CombinedLiquidPhase;
				IPhase phaseByType2 = Stream.GetPhaseByType(PhaseType.Vapour);
				double num = Roughness / diameter;
				double slipRatio = 1.0;
				double vsg = 0.0;
				double vsl = 0.0;
				double vm = 0.0;
				switch (vfcor)
				{
				case VoidFractionMethodEnum.Homogeneous:
					mVF.Value = TwoPhaseFlow.VoidFractionInPipe(combinedLiquidPhase.Density.Value, phaseByType2.Density.Value, phaseByType2.MassFraction.Value, slipRatio);
					break;
				case VoidFractionMethodEnum.HTFS:
				{
					double angle = 180.0 * inclinationRads / Math.PI;
					double area = HTFSUtility.FlowArea(diameter);
					double value = HTFSUtility.CalculateMassFlux(Stream.MassFlowrate.Value, area);
					slipRatio = TwoPhaseFlow.SlipRatioInPipe(combinedLiquidPhase.Density.Value, phaseByType2.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType2.Viscosity.Value, phaseByType2.MassFraction.Value, Math.Abs(value), angle);
					mVF.Value = TwoPhaseFlow.VoidFractionInPipe(combinedLiquidPhase.Density.Value, phaseByType2.Density.Value, phaseByType2.MassFraction.Value, slipRatio);
					break;
				}
				case VoidFractionMethodEnum.BeggsAndBrill:
				{
					PipeFlow.GetSupVelocities(Stream, diameter, ref vsg, ref vsl, ref vm);
					double laml = vsl / vm;
					double frn = vm * vm / (9.80665 * diameter);
					mVF.Value = PipeFlow.BeggsAndBrillVoidFraction(combinedLiquidPhase, inclinationRads, laml, frn, vsl);
					break;
				}
				default:
					return RealValue.FEMPTY;
				}
				mVFUpdated = true;
				return mVF.Value;
			}
			catch (Exception innerException)
			{
				throw new Exception(string.Format(Resources.MethodFailed, "CalculateVoidFraction"), innerException);
			}
		}

		public double CalculateDP_DLInPipe(IFlowStream Stream, double diameter, double inclinationRads, double Roughness, VoidFractionMethodEnum vfcor, PressureGradientMethodEnum pgcor, ref bool isChoked)
		{
			isChoked = false;
			IPhase phaseByType = Stream.GetPhaseByType(PhaseType.Overall);
			IPhase combinedLiquidPhase = Stream.CombinedLiquidPhase;
			IPhase phaseByType2 = Stream.GetPhaseByType(PhaseType.Vapour);
			double num = Roughness / diameter;
			double area = HTFSUtility.FlowArea(diameter);
			double num2 = HTFSUtility.CalculateMassFlux(Stream.MassFlowrate.Value, area);
			double fEMPTY = RealValue.FEMPTY;
			double fEMPTY2 = RealValue.FEMPTY;
			double fEMPTY3 = RealValue.FEMPTY;
			fEMPTY3 = ((!mVFUpdated) ? CalculateVoidFraction(Stream, diameter, inclinationRads, num, vfcor) : mVF.Value);
			try
			{
				double num3 = 1.0;
				switch (Stream.NumberOfPhases)
				{
				case 1:
				{
					phaseByType = Stream.GetPhaseByID(1);
					double num9 = num2 / phaseByType.Density.Value;
					double re = num2 * diameter / phaseByType.Viscosity.Value;
					fEMPTY = -2.0 * phaseByType.Density.Value * num9 * num9 * PipeFlow.ChurchillFFF(re, num) / diameter;
					fEMPTY2 = -9.80665 * Math.Sin(inclinationRads) * phaseByType.Density.Value;
					num3 = PipeFlow.aDPDZcorrection(num9, num9, phaseByType.Density.Value, phaseByType.Pressure.Value);
					break;
				}
				case 2:
				{
					phaseByType = Stream.GetPhaseByType(PhaseType.Overall);
					phaseByType2 = Stream.GetPhaseByType(PhaseType.Vapour);
					combinedLiquidPhase = Stream.CombinedLiquidPhase;
					double vsg = 0.0;
					double vsl = 0.0;
					double vm = 0.0;
					PipeFlow.GetSupVelocities(Stream, diameter, ref vsg, ref vsl, ref vm);
					double num4 = vsl / vm;
					double num5 = num4 * combinedLiquidPhase.Viscosity.Value + (1.0 - num4) * phaseByType2.Viscosity.Value;
					double num6 = num4 * combinedLiquidPhase.Density.Value + (1.0 - num4) * phaseByType2.Density.Value;
					switch (pgcor)
					{
					case PressureGradientMethodEnum.Homogeneous:
					{
						double num8 = num6 * vm * diameter / num5;
						if (num8 <= 0.0)
						{
							throw new Exception(Resources.HomogeneousPGInvalidRe);
						}
						fEMPTY = -2.0 * PipeFlow.ChurchillFFF(num8, num) * num6 * vm * vm / diameter;
						fEMPTY2 = -9.80665 * Math.Sin(inclinationRads) * num6;
						break;
					}
					case PressureGradientMethodEnum.HTFS:
						fEMPTY = TwoPhaseFlow.FrictionalPressureGradientInPipe(Stream, num2, diameter, num);
						fEMPTY2 = -9.80665 * Math.Sin(inclinationRads) * num6;
						break;
					case PressureGradientMethodEnum.BeggsAndBrill:
					{
						double num7 = num6 * vm * diameter / num5;
						if (num7 <= 0.0)
						{
							throw new Exception(Resources.BeggsBrillPGInvalidRe);
						}
						fEMPTY = -2.0 * PipeFlow.BeggsAndBrillFrictionFactor(num7, num4, fEMPTY3, num) * num6 * vm * vm / diameter;
						fEMPTY2 = -9.80665 * Math.Sin(inclinationRads) * (phaseByType2.Density.Value * fEMPTY3 + combinedLiquidPhase.Density.Value * (1.0 - fEMPTY3));
						break;
					}
					default:
						return RealValue.FEMPTY;
					}
					num3 = PipeFlow.aDPDZcorrection(vm, vsg, phaseByType.Density.Value, phaseByType.Pressure.Value);
					break;
				}
				default:
					return RealValue.FEMPTY;
				}
				if (num3 == RealValue.FEMPTY)
				{
					isChoked = true;
					return RealValue.FEMPTY;
				}
				mPGUpdated = true;
				mPG.Value = (fEMPTY + fEMPTY2) / num3;
				return mPG.Value;
			}
			catch (Exception innerException)
			{
				throw new Exception(string.Format(Resources.MethodFailed, "CalculateDP_DLInPipe"), innerException);
			}
		}
