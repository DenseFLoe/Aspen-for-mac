[ComVisible(false)]
public interface IBallValve : IValve, IPipeFittings
{
	double OpenAngle { get; set; }

	double BlindAngle { get; set; }
}
[ComVisible(false)]
public class BallValve : Valve, IBallValve, IValve, IPipeFittings
{
	private double openAngle;

	private double blindAngle;

	public double OpenAngle
	{
		get
		{
			return openAngle;
		}
		set
		{
			openAngle = value;
			equivThroatDiam = calculateEquivalentThroatDiameter();
		}
	}

	public double BlindAngle
	{
		get
		{
			return blindAngle;
		}
		set
		{
			blindAngle = value;
			equivThroatDiam = calculateEquivalentThroatDiameter();
		}
	}

	public BallValve()
	{
		InitialiseBallValve();
	}

	public BallValve(double Diameter, IFlowStream StreamIn, IFlowStream StreamOut)
		: base(Diameter, StreamIn, StreamOut)
	{
		InitialiseBallValve();
	}

	public BallValve(double Diameter, double ThroatDiam, double InletAngle, IFlowStream StreamIn, IFlowStream StreamOut)
		: base(Diameter, ThroatDiam, InletAngle, StreamIn, StreamOut)
	{
		InitialiseBallValve();
	}

	public BallValve(double Diameter, double ThroatDiam, double InletAngle, double OpenAngle, IFlowStream StreamIn, IFlowStream StreamOut)
		: base(Diameter, ThroatDiam, InletAngle, StreamIn, StreamOut)
	{
		InitialiseBallValve();
		openAngle = OpenAngle;
		equivThroatDiam = calculateEquivalentThroatDiameter();
	}

	private void InitialiseBallValve()
	{
		openAngle = 90.0;
		blindAngle = 10.0;
		equivThroatDiam = calculateEquivalentThroatDiameter();
	}

	protected override bool CheckGeometry()
	{
		bool flag = true;
		if ((calculationMode == CalculationMode.SizingFactorMode || calculationMode == CalculationMode.SizingMode) && sizingWithDefault)
		{
			if (openAngle != 90.0)
			{
				openAngle = 90.0;
				LogMessage(MessageLogType.Note, Resources.OpeningAngleSetTo90);
			}
		}
		else if (methodType == MethodType.HTFSMethod)
		{
			if (openAngle < blindAngle || openAngle > 90.0)
			{
				LogStatus("ValveBadOpenAngleBlindRange", 500);
				LogMessage(MessageLogType.Fatal, Resources.OpeningAngleNotBetweenBlindAnd90);
				throw new Exception(Resources.OpeningAngleNotBetweenBlindAnd90B);
			}
		}
		else if (openAngle != 90.0)
		{
			LogStatus("ValvePartiallyOpenUnsupported", 500);
			LogMessage(MessageLogType.Fatal, Resources.PartiallyOpenedNotSupported);
			throw new Exception(Resources.PartiallyOpenedNotSupportedBall);
		}
		if (flag && (blindAngle < 5.0 || blindAngle > 45.0))
		{
			LogStatus("ValveBadBlindAngleRange", 500);
			LogMessage(MessageLogType.Fatal, Resources.BlindAngleNotBetween5And45);
			throw new Exception(Resources.BlindAngleNotBetween5And45B);
		}
		if (flag)
		{
			flag = base.CheckGeometry();
		}
		return flag;
	}

	protected override double calculateKFactorWithHTFSMethod(IFlowStream stream0, double MassFlux)
	{
		if (stream0.NumberOfPhases < 2)
		{
			return SinglePhaseFlow.KFactorForBallValve(Ports[0].Diameter.Value, throatDiam, MassFlux, openAngle, blindAngle, stream0.GetPhaseByID(1).Viscosity.Value);
		}
		if (stream0.GetPhaseByType(PhaseType.Vapour) == null)
		{
			return SinglePhaseFlow.KFactorForBallValve(Ports[0].Diameter.Value, throatDiam, MassFlux, openAngle, blindAngle, stream0.CombinedLiquidPhase.Viscosity.Value);
		}
		double num = SinglePhaseFlow.KFactorForBallValve(Ports[0].Diameter.Value, throatDiam, MassFlux, openAngle, blindAngle, stream0.CombinedLiquidPhase.Viscosity.Value);
		double num2 = TwoPhaseFlow.ValvePlateMultiplier(stream0.CombinedLiquidPhase.Density.Value, stream0.GetPhaseByType(PhaseType.Vapour).Density.Value, stream0.GetPhaseByType(PhaseType.Vapour).MassFraction.Value);
		return num * num2;
	}

	protected override double calculateKFactorWithCraneMethod()
	{
		double num = Crane.FrictionFactorInTurbulenceZone(Ports[0].Diameter.Value);
		double num2 = 3.0 * num;
		if (diameterRatio == 1.0)
		{
			return num2;
		}
		if (inletAngle < 45.0)
		{
			return Crane.KFactorFormula5(diameterRatio, num2, inletAngle);
		}
		return Crane.KFactorFormula6(diameterRatio, num2, inletAngle);
	}

	public override double calculateEquivalentThroatDiameter()
	{
		if (openAngle == 90.0)
		{
			return throatDiam;
		}
		double num = 0.5 * throatDiam / Math.Sin(HTFSUtility.DegreesToRads(0.5 * (90.0 - blindAngle)));
		double num2 = num * Math.Sin(HTFSUtility.DegreesToRads(0.5 * (openAngle - blindAngle))) * Math.Cos(HTFSUtility.DegreesToRads(90.0 - openAngle));
		double num3 = 4.0 * Math.Asin(Math.Sqrt(num2 / throatDiam));
		double num4 = num2 / Math.Tan(0.25 * num3);
		double num5 = 0.5 * num3 * Math.Pow(0.5 * throatDiam, 2.0) - num4 * Math.Sqrt(Math.Pow(0.5 * throatDiam, 2.0) - num4 * num4);
		double num6 = 2.0 * num5;
		return Math.Sqrt(4.0 * num6 / Math.PI);
	}

	public override ICorrelationManager CreateCorrelation(MethodType lMethod, IFlowStream stream)
	{
		ICorrelationManager result = null;
		ILiquidHoldUpCorrelation liquidHoldUpCorrelation = null;
		switch (lMethod)
		{
		case MethodType.HTFSMethod:
			liquidHoldUpCorrelation = new HTFSLiquidHoldUpCorrelation(stream, base.PipeDiameter, 0.0, 0.0, HTFSLiquidHoldUpCorrelation.FittingType.Valve);
			result = new BallValveHTFSFrictionCorrelation(stream, base.PipeDiameter, base.ThroatDiameter, base.InletAngle, OpenAngle, BlindAngle, liquidHoldUpCorrelation);
			break;
		case MethodType.Crane_HTFSMethod:
			liquidHoldUpCorrelation = new HTFSLiquidHoldUpCorrelation(stream, base.PipeDiameter, 0.0, 0.0, HTFSLiquidHoldUpCorrelation.FittingType.Valve);
			result = new BallValveCraneFrictionCorrelation(stream, base.PipeDiameter, base.ThroatDiameter, base.InletAngle, OpenAngle, BlindAngle, liquidHoldUpCorrelation);
			break;
		case MethodType.CraneMethod:
			liquidHoldUpCorrelation = new HomogeneousLiquidHoldUpCorrelation(stream, base.PipeDiameter, 0.0, 0.0);
			result = new BallValveCraneFrictionCorrelation(stream, base.PipeDiameter, base.ThroatDiameter, base.InletAngle, OpenAngle, BlindAngle, liquidHoldUpCorrelation);
			break;
		case MethodType.TotalPressureMethod:
			result = new UserFrictionFactorCorrelation(stream, 0.0, base.PipeDiameter, base.PipeDiameter);
			break;
		}
		return result;
	}

	public override string ToString()
	{
		return Resources.FittingBallValve;
	}
}
[ComVisible(false)]
public interface IBend : IPipeFittings
{
	double Angle { get; set; }

	double KFactor { get; set; }

	double Radius { get; set; }

	double LowestPressure { get; set; }
}
[ComVisible(false)]
public abstract class Bend : PipeFitting, IBend, IPipeFittings
{
	protected double angle;

	protected double radius;

	protected double K;

	private double lowestPressure;

	public double Angle
	{
		get
		{
			return angle;
		}
		set
		{
			angle = value;
		}
	}

	public double Radius
	{
		get
		{
			return radius;
		}
		set
		{
			radius = value;
		}
	}

	public double KFactor
	{
		get
		{
			return K;
		}
		set
		{
			K = value;
		}
	}

	public double LowestPressure
	{
		get
		{
			return lowestPressure;
		}
		set
		{
			lowestPressure = value;
		}
	}

	public Bend()
		: base(2)
	{
		InitialiseBend();
	}

	public Bend(double Diameter, IFlowStream StreamIn, IFlowStream StreamOut)
		: base(2)
	{
		Ports[0].Diameter.Value = Diameter;
		Ports[0].Stream = StreamIn;
		Ports[1].Diameter.Value = Diameter;
		Ports[1].Stream = StreamOut;
		InitialiseBend();
	}

	public Bend(double Diameter, double Angle, double Radius, IFlowStream StreamIn, IFlowStream StreamOut)
		: base(2)
	{
		Ports[0].Diameter.Value = Diameter;
		Ports[0].Stream = StreamIn;
		Ports[1].Diameter.Value = Diameter;
		Ports[1].Stream = StreamOut;
		InitialiseBend();
		angle = Angle;
		radius = Radius;
	}

	private void InitialiseBend()
	{
		angle = 90.0;
		radius = Ports[0].Diameter.Value * 3.0;
		K = 0.0;
		lowestPressure = 10000.0;
	}

	protected override bool CheckGeometry()
	{
		bool flag = true;
		if (radius < 0.0)
		{
			LogStatus("BendNegativeRadius", 500);
			LogMessage(MessageLogType.Fatal, Resources.NegativeRadiusWarning);
			throw new Exception(Resources.NegativeRadiusWarningB);
		}
		if (angle < 0.0 || angle > 180.0)
		{
			LogStatus("BendTooLargeAngle", 500);
			LogMessage(MessageLogType.Fatal, Resources.AngleAbove180Warning);
			throw new Exception(Resources.AngleAbove180WarningB);
		}
		return base.CheckGeometry();
	}

	public override double CalculatePressureDrop()
	{
		bool flag = true;
		if (calculationMode == CalculationMode.UnknownMode || calculationMode == CalculationMode.PressureDropMode)
		{
			flag = CheckingForDPCalculation();
		}
		if (flag)
		{
			DPz = 0.0;
			double result = 0.0;
			DH = 0.0;
			if (methodType == MethodType.StaticPressureMethod)
			{
				result = 0.0;
			}
			else
			{
				IFlowStream stream = Ports[StartPortNo].Stream;
				PipeFlowPressureSolver pipeFlowPressureSolver = CreateIncompressibleSolver(methodType, stream);
				if (pipeFlowPressureSolver != null)
				{
					IFlowStream stream2 = Ports[EndPortNo].Stream;
					double length = 1.0;
					if (pipeFlowPressureSolver.Correlation is IEquivalentLengthCorrelation equivalentLengthCorrelation)
					{
						length = equivalentLengthCorrelation.CalculateEquivalentLength();
					}
					if (pipeFlowPressureSolver.CalculateUnknownStream(stream2, stream, length, StepwiseFactor, FlowDirectionFactor))
					{
						result = stream2.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
						DH = stream2.GetPhaseByID(0).Enthalpy.Value - stream.GetPhaseByID(0).Enthalpy.Value;
					}
				}
			}
			if (calculationMode == CalculationMode.PressureDropMode)
			{
				if (base.IsChoked)
				{
					if (StepwiseFactor > 0.0)
					{
						PipeFlowFunctions.getChokingCondition(pIPPP, Ports[1].Stream, Ports[0].Stream, Ports[0].Diameter.Value, getMinimumDiameter(), (int)methodType, requiredAccuracy, StepwiseFactor);
						LogMessage(MessageLogType.Fatal, Resources.FlowChokedUpstreamEstimated);
						if (task != TaskType.Design)
						{
							return -1.0;
						}
						throw new Exception(Resources.FatalErrorChoked);
					}
					LogStatus("ChokedFlowInletSpec", 500);
					throw new Exception(Resources.ChokedFlowInletPSpec);
				}
				updateOtherPort(CalculatedResult.PressureDrop, result);
			}
			return result;
		}
		return RealValue.FEMPTY;
	}

	protected override PipeFlowPressureSolver CreateIncompressibleSolver(MethodType lMethod, IFlowStream stream)
	{
		PipeFlowPressureSolver pipeFlowPressureSolver = null;
		if (lMethod == MethodType.HTFSMethod)
		{
			pipeFlowPressureSolver = new PipeFlowPressureSolverPressureMethod();
			pipeFlowPressureSolver.MomentumEnergyDelegate = HTFSUtility.MomentumEnergy;
			pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseAverage;
		}
		else
		{
			pipeFlowPressureSolver = new PipeFlowPressureSolverLengthMethod();
			if (StepwiseFactor == 1.0)
			{
				pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseUnknownStream;
			}
			else
			{
				pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseKnownStream;
			}
		}
		pipeFlowPressureSolver.HeatFlux = 0.0;
		pipeFlowPressureSolver.PhysicalPropertyPackage = pIPPP;
		pipeFlowPressureSolver.Accuracy = base.RequiredAccuracy;
		double value = Ports[0].Diameter.Value;
		pipeFlowPressureSolver.SetGeometricalParameters(value, value, 0.0);
		pipeFlowPressureSolver.IncludeMomentumEnergy = base.IncludeKineticEnergy;
		pipeFlowPressureSolver.Correlation = CreateCorrelation(lMethod, stream);
		if (base.LogMessages is MessageLogger)
		{
			pipeFlowPressureSolver.Log = (MessageLogger)base.LogMessages;
		}
		return pipeFlowPressureSolver;
	}

	public abstract void CalculateKAndf(IFlowStream stream, double MassFlux, out double K, out double f);

	protected double CalculateDPWithCraneMethod()
	{
		IPort port = Ports[StartPortNo];
		IFlowStream flowStream = (IFlowStream)port.Stream.Clone();
		double area = HTFSUtility.FlowArea(port.Diameter.Value);
		double num = HTFSUtility.CalculateMassFlux(flowStream.MassFlowrate.Value, area);
		CalculateKAndf(flowStream, num, out var K, out var _);
		double num2 = Crane.CalculateHomogeneousEquivalentDensity(flowStream);
		DPf = 0.5 * K * num * num / num2 * StepwiseFactor * FlowDirectionFactor;
		DPa = CalculateDPa(flowStream, num, DPf);
		if (!base.IsChoked)
		{
			return DPf + DPa;
		}
		return RealValue.FEMPTY;
	}

	protected double CalculateDPWithCrane_HTFSMethod()
	{
		IPort port = Ports[StartPortNo];
		IFlowStream flowStream = (IFlowStream)port.Stream.Clone();
		double area = HTFSUtility.FlowArea(port.Diameter.Value);
		double num = HTFSUtility.CalculateMassFlux(flowStream.MassFlowrate.Value, area);
		CalculateKAndf(flowStream, num, out var K, out var _);
		double num2 = HTFSUtility.CalculateEquivalentDensity(flowStream, num, 0.0);
		if (flowStream.NumberOfPhases < 2)
		{
			DPf = 0.5 * K * num * num / num2 * StepwiseFactor * FlowDirectionFactor;
		}
		else if (flowStream.GetPhaseByType(PhaseType.Vapour) == null)
		{
			DPf = 0.5 * K * num * num / num2 * StepwiseFactor * FlowDirectionFactor;
		}
		else
		{
			double num3 = num * (1.0 - flowStream.GetPhaseByType(PhaseType.Vapour).MassFraction.Value);
			double num4 = num * flowStream.GetPhaseByType(PhaseType.Vapour).MassFraction.Value;
			double num5 = 0.5 * K * num3 * num3 / flowStream.CombinedLiquidPhase.Density.Value;
			double num6 = 0.5 * K * num4 * num4 / flowStream.GetPhaseByType(PhaseType.Vapour).Density.Value;
			double num7 = TwoPhaseFlow.CoefficeCFor90DegreeBend(flowStream.CombinedLiquidPhase.Density.Value, flowStream.GetPhaseByType(PhaseType.Vapour).Density.Value, radius / port.Diameter.Value);
			DPf = (num5 + num7 * Math.Sqrt(num5 * num6) + num6) * StepwiseFactor * FlowDirectionFactor;
		}
		DPa = CalculateDPa(flowStream, num, DPf);
		if (!base.IsChoked)
		{
			return DPf + DPa;
		}
		return RealValue.FEMPTY;
	}

	protected double CalculateDPWithHTFSMethod()
	{
		double num = 0.05;
		IPort port = Ports[StartPortNo];
		double value = port.Diameter.Value;
		IFlowStream flowStream = (IFlowStream)port.Stream.Clone();
		double area = HTFSUtility.FlowArea(port.Diameter.Value);
		double num2 = HTFSUtility.CalculateMassFlux(flowStream.MassFlowrate.Value, area);
		double num3 = 0.0;
		double K = 0.0;
		double f = 0.0;
		CalculateKAndf(flowStream, num2, out K, out f);
		num3 = 0.25 * K * port.Diameter.Value / f;
		IFlowStream flowStream2 = (IFlowStream)flowStream.Clone();
		IFlowStream flowStream3 = (IFlowStream)flowStream2.Clone();
		double num4 = 0.0;
		double num5 = 0.0;
		double num6 = 0.0;
		double num7 = 0.0;
		double num8 = 0.0;
		double num9 = 0.0;
		double num10 = 0.0;
		double num11 = 0.0;
		double num12 = 0.0;
		int num13 = 0;
		int num14 = 1;
		while (num4 < num3 && !base.IsChoked)
		{
			double num15 = 0.0;
			num13++;
			if (num4 < 1E-06)
			{
				num6 = HTFSUtility.CalculateFrictionalPressureGradientInBend(flowStream2, port.Diameter.Value, num2, f, K, radius) * FlowDirectionFactor;
				num11 = HTFSUtility.CalculateEquivalentDensity(flowStream2, num2, angle);
			}
			else
			{
				flowStream2 = (IFlowStream)flowStream3.Clone();
				num6 = num7;
				num11 = num12;
			}
			double num16 = 1.0;
			num5 = 0.0;
			if (num12 <= 0.0)
			{
				num12 = num11;
			}
			int num17 = 0;
			while (num16 > requiredAccuracy * num3 && num17 < 10)
			{
				num17++;
				num9 = flowStream2.GetPhaseByID(0).Pressure.Value * (1.0 + StepwiseFactor * num);
				num14 = Math.Sign(num6 * (num5 + 1E-05) + num15);
				num9 = flowStream2.GetPhaseByID(0).Pressure.Value * (1.0 - StepwiseFactor * (double)num14 * num);
				if (num9 < lowestPressure)
				{
					base.IsChoked = true;
					chokedLocation = ChokedLocationEnum.Outlet;
					break;
				}
				num8 = (flowStream2.GetPhaseByID(0).Pressure.Value - num9) * (0.0 - StepwiseFactor);
				num10 = TwoPhaseFlow.Enthalpy(flowStream2.GetPhaseByID(0).Enthalpy.Value, 0.0, 0.0, num2, port.Diameter.Value, num5, flowStream3.MassFlowrate.Value, num11, num12, 0.0);
				pIPPP.ReflashStreamWithSameComposition(flowStream3, FlashType.PHFlash, num9, num10);
				CalculateKAndf(flowStream3, num2, out K, out f);
				num7 = HTFSUtility.CalculateFrictionalPressureGradientInBend(flowStream3, port.Diameter.Value, num2, f, K, radius) * FlowDirectionFactor;
				num12 = HTFSUtility.CalculateEquivalentDensity(flowStream3, num2, angle);
				num15 = 0.5 * num2 * num2 * (1.0 / num12 - 1.0 / num11) * (0.0 - StepwiseFactor);
				double num18 = Math.Abs((num8 - num15) / (0.5 * (num6 + num7)));
				num16 = Math.Abs(num18 - num5);
				if (num18 > num3 && num >= 0.05)
				{
					num *= 0.75;
				}
				else if (num18 < num3 / 100.0 || num18 / value < 1.0)
				{
					num *= 1.25;
				}
				if (num18 > 0.0)
				{
					num5 = num18;
				}
			}
			num4 += num5;
			if (!checkChoking)
			{
				continue;
			}
			base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, flowStream3, area);
			if (base.IsChoked)
			{
				if (StepwiseFactor > 0.0)
				{
					chokedLocation = ChokedLocationEnum.Inlet;
				}
				else
				{
					chokedLocation = ChokedLocationEnum.Outlet;
				}
			}
		}
		if (!base.IsChoked)
		{
			if (num4 - num3 > requiredAccuracy * num3)
			{
				num8 *= (num3 - (num4 - num5)) / num5;
				num9 = flowStream2.GetPhaseByID(0).Pressure.Value + StepwiseFactor * num8;
				num10 = flowStream2.GetPhaseByID(0).Enthalpy.Value + (flowStream3.GetPhaseByID(0).Enthalpy.Value - flowStream2.GetPhaseByID(0).Enthalpy.Value) * (num3 - (num4 - num5)) / num5;
				pIPPP.ReflashStreamWithSameComposition(flowStream3, FlashType.PHFlash, num9, num10);
			}
			double num19 = flowStream3.GetPhaseByID(0).Pressure.Value - flowStream.GetPhaseByID(0).Pressure.Value;
			double num20 = HTFSUtility.CalculateEquivalentDensity(flowStream, num2, angle);
			double num21 = HTFSUtility.CalculateEquivalentDensity(flowStream3, num2, angle);
			DPa = -0.5 * num2 * num2 * (1.0 / num21 - 1.0 / num20);
			DPf = num19 - DPa - DPz;
			DH = flowStream3.GetPhaseByID(0).Enthalpy.Value - flowStream.GetPhaseByID(0).Enthalpy.Value;
			return num19;
		}
		return RealValue.FEMPTY;
	}

	public override string ToString()
	{
		return Resources.FittingBend;
	}

				return SinglePhaseFlow.KFactorForBordaInlet(Ports[1].Diameter.Value, massFlux, stream.GetPhaseByID(1).Viscosity.Value);
			}
			return SinglePhaseFlow.KFactorForBordaInlet(Ports[1].Diameter.Value, massFlux, stream.CombinedLiquidPhase.Viscosity.Value);
		}
		return 1.0;
	}

	}

[ComVisible(false)]
public class ButterflyValve : Valve, IButterflyValve, IValve, IPipeFittings
{
	private double openAngle;

	private double discToPipeDR;

	public double OpenAngle

	public ButterflyValve(double Diameter, IFlowStream StreamIn, IFlowStream StreamOut)

		discToPipeDR = DiscToPipeDR;
		equivThroatDiam = calculateEquivalentThroatDiameter();

	private void InitialiseButterflyValve()
	{
		openAngle = 90.0;
		discToPipeDR = 1.0;
	}

	protected override bool CheckGeometry()
	{
		bool flag = true;
		if ((calculationMode == CalculationMode.SizingFactorMode || calculationMode == CalculationMode.SizingMode) && sizingWithDefault)
		{
			if (openAngle != 90.0)
			{
				openAngle = 90.0;
				LogMessage(MessageLogType.Note, Resources.OpeningAngleSetTo90);
			}
		}
		else if (methodType == MethodType.HTFSMethod)
		{
			if (openAngle < 10.0 || openAngle > 90.0)
			{
				LogStatus("ValveBadOpenAngleRange", 500);
				LogMessage(MessageLogType.Fatal, Resources.OpeningAngleNotBetween10And90);
				throw new Exception(Resources.OpeningAngleNotBetween10And90B);
			}
		}
		else if (openAngle != 90.0)
		{
			LogStatus("ValvePartiallyOpenUnsupported", 500);
			LogMessage(MessageLogType.Fatal, Resources.PartiallyOpenedNotSupported);
			throw new Exception(Resources.PartiallyOpenedNotSupportedButterfly);
		}
		if (methodType == MethodType.HTFSMethod)
		{
			if (discToPipeDR < 0.98 || discToPipeDR > 1.0)
			{
				LogStatus("ValveBadDiscPipeRatioRange", 200);
				LogMessage(MessageLogType.Fatal, Resources.DiscToPipeRatioSetTo1);
				if (task == TaskType.Design)
				{
					throw new Exception(Resources.InvalidDiscToPipeRatio);
				}
				discToPipeDR = 1.0;
			}
		}
		else if (discToPipeDR != 1.0)
		{
			LogMessage(MessageLogType.Fatal, Resources.DefaultDiscToPipeRatio1);
			discToPipeDR = 1.0;
		}
		if (flag)
		{
			flag = base.CheckGeometry();
		}
		return flag;
	}

	protected override double calculateKFactorWithHTFSMethod(IFlowStream stream0, double MassFlux)
	{
		if (stream0.NumberOfPhases < 2)
		{
			return SinglePhaseFlow.KFactorForButterflyValve(Ports[0].Diameter.Value, throatDiam, MassFlux, openAngle, stream0.GetPhaseByID(1).Viscosity.Value);
		}
		if (stream0.GetPhaseByType(PhaseType.Vapour) == null)
		{
			return SinglePhaseFlow.KFactorForButterflyValve(Ports[0].Diameter.Value, throatDiam, MassFlux, openAngle, stream0.CombinedLiquidPhase.Viscosity.Value);
		}
		double num = SinglePhaseFlow.KFactorForButterflyValve(Ports[0].Diameter.Value, throatDiam, MassFlux, openAngle, stream0.CombinedLiquidPhase.Viscosity.Value);
		double num2 = TwoPhaseFlow.ValvePlateMultiplier(stream0.CombinedLiquidPhase.Density.Value, stream0.GetPhaseByType(PhaseType.Vapour).Density.Value, stream0.GetPhaseByType(PhaseType.Vapour).MassFraction.Value);
		return num * num2;
	}

	protected override double calculateKFactorWithCraneMethod()
	{
		double value = Ports[0].Diameter.Value;
		double num = Crane.FrictionFactorInTurbulenceZone(value);
		if (value < 0.2)
		{
			K = 45.0 * num;
		}
		else if (value < 0.4)
		{
			K = 35.0 * num;
		}
		else
		{
			K = 25.0 * num;
		}
		return K;
	}

	public override double calculateEquivalentThroatDiameter()
	{
		double d = 1.0 - discToPipeDR * discToPipeDR * Math.Cos(HTFSUtility.DegreesToRads(openAngle));
		return throatDiam * Math.Sqrt(d);
	}

	public override ICorrelationManager CreateCorrelation(MethodType lMethod, IFlowStream stream)
	{
		ICorrelationManager result = null;
		ILiquidHoldUpCorrelation liquidHoldUpCorrelation = null;
		switch (lMethod)
		{
		case MethodType.HTFSMethod:
			liquidHoldUpCorrelation = new HTFSLiquidHoldUpCorrelation(stream, base.PipeDiameter, 0.0, 0.0, HTFSLiquidHoldUpCorrelation.FittingType.Valve);
			result = new ButterflyValveHTFSFrictionCorrelation(stream, base.PipeDiameter, base.InletAngle, OpenAngle, DiscToPipeDiameterRatio, liquidHoldUpCorrelation);
			break;
		case MethodType.Crane_HTFSMethod:
			liquidHoldUpCorrelation = new HTFSLiquidHoldUpCorrelation(stream, base.PipeDiameter, 0.0, 0.0, HTFSLiquidHoldUpCorrelation.FittingType.Valve);
			result = new ButterflyValveCraneFrictionCorrelation(stream, base.PipeDiameter, base.InletAngle, OpenAngle, DiscToPipeDiameterRatio, liquidHoldUpCorrelation);
			break;
		case MethodType.CraneMethod:
			liquidHoldUpCorrelation = new HomogeneousLiquidHoldUpCorrelation(stream, base.PipeDiameter, 0.0, 0.0);
			result = new ButterflyValveCraneFrictionCorrelation(stream, base.PipeDiameter, base.InletAngle, OpenAngle, DiscToPipeDiameterRatio, liquidHoldUpCorrelation);
			break;
		case MethodType.TotalPressureMethod:
			result = new UserFrictionFactorCorrelation(stream, 0.0, base.PipeDiameter, base.PipeDiameter);
			break;
		}
		return result;
	}

	public override string ToString()
	{
		return Resources.FittingButterflyValve;
	}

			break;
		}
		return result;
	}

[ComVisible(false)]
public interface IContraction : ISwage, IPipeFittings
{
}
[ComVisible(false)]
public abstract class Contraction : Swage, IContraction, ISwage, IPipeFittings
{
	public Contraction()
	{
	}

	public Contraction(double UpstreamDiam, double DownstreamDiam, IFlowStream streamIn, IFlowStream streamOut)
	{
		Ports[0].Diameter.Value = UpstreamDiam;
		Ports[0].Stream = streamIn;
		Ports[1].Diameter.Value = DownstreamDiam;
		Ports[1].Stream = streamOut;
		diameterRatio = DownstreamDiam / UpstreamDiam;
	}

	protected override bool CheckGeometry()
	{
		bool flag = true;
		if (Ports[1].Diameter.Value > Ports[0].Diameter.Value)
		{
			LogStatus("ContractionBadDiameters", 500);
			LogMessage(MessageLogType.Fatal, "The outlet diameter must be smaller than the inlet diameter.");
			throw new Exception("A contraction must have a smaller outlet diameter than inlet diameter.");
		}
		if (!Ports[0].Diameter.IsUnknown && !Ports[1].Diameter.IsUnknown)
		{
			diameterRatio = Ports[1].Diameter.Value / Ports[0].Diameter.Value;
		}
		else if (diameterRatio > 0.0 && diameterRatio <= 1.0)
		{
			if (!Ports[0].Diameter.IsUnknown)
			{
				Ports[1].Diameter.Value = Ports[0].Diameter.Value * diameterRatio;
			}
			else
			{
				Ports[0].Diameter.Value = Ports[1].Diameter.Value / diameterRatio;
			}
		}
		else
		{
			flag = false;
		}
		if (flag)
		{
			flag = base.CheckGeometry();
		}
		return flag;
	}

	protected override double getBodyDiameter()
	{
		if (Ports[0].Diameter.IsUnknown)
		{
			if (Ports[1].Diameter.IsUnknown)
			{
				LogStatus("ThrottleUnknownDiameter", 500);
				LogMessage(MessageLogType.Fatal, Resources.InOrOutDiamNeeded);
				throw new Exception(Resources.InOrOutDiamNeededContraction);
			}
			if (diameterRatio > 0.0)
			{
				Ports[0].Diameter.Value = Ports[1].Diameter.Value / diameterRatio;
			}
			else
			{
				Ports[0].Diameter.Value = Ports[1].Diameter.Value;
			}
		}
		return Ports[0].Diameter.Value;
	}

	protected override void updatePortDiameter(double D)
	{
		Ports[0].Diameter.Value = D;
		Ports[1].Diameter.Value = D * diameterRatio;
	}

	protected override double getKFactorForStaticPressureDrop(double Kt)
	{
		return Kt + 1.0 - Math.Pow(diameterRatio, 4.0);
	}

	protected override double calculateKFactorWithCraneMethod()
	{
		if (angle <= 45.0)
		{
			K = Crane.KFactorFormula1(diameterRatio, 0.0, angle);
		}
		else
		{
			K = Crane.KFactorFormula2(diameterRatio, 0.0, angle);
		}
		return K;
	}

	protected override double calculateMultiplierWithCrane_HTFSMethod(IFlowStream stream, double massFlux)
	{
		IPhase combinedLiquidPhase = stream.CombinedLiquidPhase;
		IPhase phaseByType = stream.GetPhaseByType(PhaseType.Vapour);
		if (combinedLiquidPhase.Viscosity.Value / phaseByType.Viscosity.Value < 1000.0)
		{
			return TwoPhaseFlow.MultiplierOfFriedel(massFlux, Ports[0].Diameter.Value, combinedLiquidPhase, phaseByType);
		}
		if (massFlux > 100.0)
		{
			double y = Math.Sqrt(combinedLiquidPhase.Density.Value / phaseByType.Density.Value);
			return TwoPhaseFlow.MultiplierOfChisholm(y, massFlux, phaseByType.MassFraction.Value, 0.0);
		}
		double constC = TwoPhaseFlow.CoefficeCByFittingsName("SuddenContraction", combinedLiquidPhase.Density.Value, phaseByType.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType.Viscosity.Value, phaseByType.MassFraction.Value, massFlux);
		double xtt = TwoPhaseFlow.XLMParameter(Ports[0].Diameter.Value, massFlux, 0.0, phaseByType.MassFraction.Value, combinedLiquidPhase.Density.Value, phaseByType.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType.Viscosity.Value);
		return TwoPhaseFlow.MultiplierOfMartinelli(constC, xtt) * HTFSUtility.Square(combinedLiquidPhase.MassFraction.Value);
	}

	protected override double calculateMultiplierWithHTFSMethod(IFlowStream stream, double massFlux)
	{
		IPhase combinedLiquidPhase = stream.CombinedLiquidPhase;
		IPhase phaseByType = stream.GetPhaseByType(PhaseType.Vapour);
		return TwoPhaseFlow.MultiplierOfMartinelliAtL0(combinedLiquidPhase.Density.Value, phaseByType.Density.Value, phaseByType.MassFraction.Value);
	}

	protected override double calculateDPWithHTFSMethod(IFlowStream stream, double massFlux)
	{
		double num = RealValue.FEMPTY;
		double fEMPTY = RealValue.FEMPTY;
		double fEMPTY2 = RealValue.FEMPTY;
		bool flag = false;
		double Pc_P = RealValue.FEMPTY;
		double Mc = RealValue.FEMPTY;
		double num2 = massFlux / (diameterRatio * diameterRatio);
		if (checkChoking)
		{
			base.IsChoked = HTFSUtility.CheckChokingForIdealFlowInContraction(stream, massFlux, diameterRatio);
		}
		if (!base.IsChoked)
		{
			double num3 = HTFSUtility.CalculateStagnationPressure(stream, massFlux);
			bool flag2 = false;
			if (stream.NumberOfPhases == 1 && stream.GetPhaseByID(1).PhaseType == PhaseType.Liquid)
			{
				flag2 = true;
			}
			double density = getDensity(stream);
			K = getKFactor(stream, num2);
			if (stream.NumberOfPhases > 1)
			{
				double kFactorForStaticPressureDrop = getKFactorForStaticPressureDrop(K);
				double num4 = calculateMultiplierWithHTFSMethod(stream, num2);
				kFactorForStaticPressureDrop *= num4;
				num = 0.5 * kFactorForStaticPressureDrop * num2 * num2 / density;
			}
			else
			{
				fEMPTY = 0.5 * K * num2 * num2 / density;
				fEMPTY2 = 0.5 * (num2 * num2 / density - massFlux * massFlux / density);
				num = fEMPTY + fEMPTY2;
			}
			if (Math.Abs(num / stream.GetPhaseByID(0).Pressure.Value) > 0.15)
			{
				double num5 = diameterRatio / Math.Sqrt(1.0 + Math.Sqrt(K));
				double num6 = massFlux / (num5 * num5);
				flag = HTFSUtility.CalculateCriticalPressureRatioAndMassFlux(stream, pIPPP, num5, out Pc_P, out Mc);
				if (Mc == RealValue.FEMPTY)
				{
					Mc = 1E+20;
				}
				if (num6 > Mc)
				{
					base.IsChoked = true;
				}
				else if (massFlux > 0.75 * Mc && !flag2)
				{
					double num7 = ((stream.NumberOfPhases >= 2) ? TwoPhaseFlow.ThroatToUpstreamPressureRatioForContraction(stream, diameterRatio, massFlux, Pc_P) : SinglePhaseFlow.ThroatToUpstreamPressureRatioForContraction(stream, num5, massFlux, Pc_P));
					if (num7 <= 0.0)
					{
						LogMessage(MessageLogType.Fatal, Resources.CantFindPThroat);
						base.IsChoked = true;
					}
					else
					{
						double num8 = num7 * stream.GetPhaseByID(0).Pressure.Value;
						IFlowStream flowStream = (IFlowStream)stream.Clone();
						pIPPP.ReflashStreamWithSameComposition(flowStream, FlashType.PSFlash, num8, flowStream.GetPhaseByID(0).Entropy.Value);
						double num9 = num5 / diameterRatio;
						density = HTFSUtility.CalculateEquivalentDensity(flowStream, num6, 0.0);
						if (num8 + 0.5 * num6 * num6 / density > num3)
						{
							base.IsChoked = true;
						}
						else
						{
							double num10 = ((flowStream.NumberOfPhases <= 1) ? SinglePhaseFlow.DownstreamPressureForCompressibleEnlargement(Ports[1].Diameter.Value, Ports[1].Diameter.Value * num9, num6, flowStream.GetPhaseByID(0).Pressure.Value, flowStream.GetPhaseByType(PhaseType.Vapour).Density.Value, flowStream.GetPhaseByType(PhaseType.Vapour).Cp.Value / flowStream.GetPhaseByType(PhaseType.Vapour).Cv.Value) : TwoPhaseFlow.DownstreamPressureForCompressibleEnlargement(flowStream, pIPPP, num6, num9));
							num = stream.GetPhaseByID(0).Pressure.Value - num10;
						}
					}
				}
			}
		}
		if (base.IsChoked || (flag && Math.Abs(num) > stream.GetPhaseByID(0).Pressure.Value * (1.0 - Pc_P)) || (!flag && Math.Abs(num) > stream.GetPhaseByID(0).Pressure.Value * 0.6))
		{
			base.IsChoked = true;
			chokedLocation = ChokedLocationEnum.Outlet;
			num = RealValue.FEMPTY;
		}
		if (!base.IsChoked)
		{
			calculateDH(stream, massFlux, num);
			IFlowStream flowStream2 = (IFlowStream)stream.Clone();
			pIPPP.ReflashStreamWithSameComposition(flowStream2, FlashType.PHFlash, flowStream2.GetPhaseByID(0).Pressure.Value - num, flowStream2.GetPhaseByID(0).Enthalpy.Value + DH * (0.0 - StepwiseFactor));
			double num11 = HTFSUtility.CalculateEquivalentDensity(stream, massFlux, 0.0);
			double num12 = HTFSUtility.CalculateEquivalentDensity(flowStream2, num2, 0.0);
			fEMPTY2 = 0.5 * (num2 * num2 / num12 - massFlux * massFlux / num11);
			fEMPTY = num - fEMPTY2;
			if (fEMPTY <= 0.0)
			{
				fEMPTY = 0.0;
			}
			DPa = fEMPTY2 * StepwiseFactor;
			DPf = fEMPTY * StepwiseFactor;
		}
		return num;
	}

	protected override double calculateDPWithCraneMethod(IFlowStream stream, double massFlux)
	{
		double result = RealValue.FEMPTY;
		double massFlux2 = massFlux / (diameterRatio * diameterRatio);
		if (checkChoking)
		{
			base.IsChoked = HTFSUtility.CheckChokingForIdealFlowInContraction(stream, massFlux, diameterRatio);
		}
		if (!base.IsChoked)
		{
			K = getKFactor(stream, massFlux2);
			double density = getDensity(stream);
			double num = 0.5 * K * massFlux * massFlux / density;
			DPf = num * StepwiseFactor;
			double num2 = Math.Abs(CalculateDPa(stream, massFlux, DPf));
			result = num + num2;
		}
		return result;
	}

	public override ICorrelationManager CreateCorrelation(MethodType lMethod, IFlowStream stream)
	{
		ICorrelationManager result = null;
		ILiquidHoldUpCorrelation liquidHoldUpCorrelation = null;
		switch (lMethod)
		{
		case MethodType.HTFSMethod:
			liquidHoldUpCorrelation = new HTFSLiquidHoldUpCorrelation(stream, base.UpstreamPipeDiameter, 0.0, 0.0, HTFSLiquidHoldUpCorrelation.FittingType.Contraction);
			result = new ContractionHTFSFrictionCorrelation(stream, base.UpstreamPipeDiameter, base.DownstreamPipeDiameter, base.Angle, liquidHoldUpCorrelation);
			break;
		case MethodType.Crane_HTFSMethod:
			liquidHoldUpCorrelation = new HTFSLiquidHoldUpCorrelation(stream, base.UpstreamPipeDiameter, 0.0, 0.0, HTFSLiquidHoldUpCorrelation.FittingType.Contraction);
			result = new ContractionCraneFrictionCorrelation(stream, base.UpstreamPipeDiameter, base.DownstreamPipeDiameter, base.Angle, liquidHoldUpCorrelation);
			break;
		case MethodType.CraneMethod:
			liquidHoldUpCorrelation = new HomogeneousLiquidHoldUpCorrelation(stream, base.UpstreamPipeDiameter, 0.0, 0.0);
			result = new ContractionCraneFrictionCorrelation(stream, base.UpstreamPipeDiameter, base.DownstreamPipeDiameter, base.Angle, liquidHoldUpCorrelation);
			break;
		case MethodType.TotalPressureMethod:
			result = new UserFrictionFactorCorrelation(stream, 0.0, base.UpstreamPipeDiameter, base.DownstreamPipeDiameter);
			break;
		}
		return result;
	}

	public Enlargement(double UpstreamDiam, double DownstreamDiam, IFlowStream streamIn, IFlowStream streamOut)
	{
		Ports[0].Diameter.Value = UpstreamDiam;
		Ports[0].Stream = streamIn;
		Ports[1].Diameter.Value = DownstreamDiam;
		Ports[1].Stream = streamOut;
		diameterRatio = UpstreamDiam / DownstreamDiam;
	}

	protected override bool CheckGeometry()
	{
		bool flag = true;
		if (Ports[0].Diameter.Value > Ports[1].Diameter.Value)
		{
			LogStatus("EnlargementBadDiameters", 500);
			LogMessage(MessageLogType.Fatal, Resources.OutletSmallerError);
			throw new Exception(Resources.OutletSmallerErrorEnlargement);
		}
		if (!Ports[0].Diameter.IsUnknown && !Ports[1].Diameter.IsUnknown)
		{
			diameterRatio = Ports[0].Diameter.Value / Ports[1].Diameter.Value;
		}
		else if (diameterRatio > 0.0 && diameterRatio <= 1.0)
		{
			if (!Ports[0].Diameter.IsUnknown)
			{
				Ports[1].Diameter.Value = Ports[0].Diameter.Value / diameterRatio;
			}
			else
			{
				Ports[0].Diameter.Value = Ports[1].Diameter.Value * diameterRatio;
			}
		}
		else
		{
			flag = false;
		}
		if (flag)
		{
			flag = base.CheckGeometry();
		}
		return flag;
	}

	protected override double getBodyDiameter()
	{
		if (Ports[0].Diameter.IsUnknown)
		{
			if (Ports[1].Diameter.IsUnknown)
			{
				LogStatus("ThrottleUnknownDiameter", 500);
				LogMessage(MessageLogType.Fatal, Resources.InOrOutDiamNeeded);
				throw new Exception(Resources.InOrOutDiamNeededEnlargement);
			}
			if (diameterRatio > 0.0)
			{
				Ports[0].Diameter.Value = Ports[1].Diameter.Value * diameterRatio;
			}
			else
			{
				Ports[0].Diameter.Value = Ports[1].Diameter.Value;
			}
		}
		return Ports[0].Diameter.Value;
	}

	protected override void updatePortDiameter(double D)
	{
		Ports[0].Diameter.Value = D;
		Ports[1].Diameter.Value = D / diameterRatio;
	}

	protected override double getKFactorForStaticPressureDrop(double Kt)
	{
		return Kt - 1.0 + Math.Pow(diameterRatio, 4.0);
	}

	protected override double calculateKFactorWithCraneMethod()
	{
		if (angle <= 45.0)
		{
			K = Crane.KFactorFormula3(diameterRatio, 0.0, angle);
		}
		else
		{
			K = Crane.KFactorFormula4(diameterRatio, 0.0, angle);
		}
		return K * Math.Pow(diameterRatio, 4.0);
	}

	protected override double calculateMultiplierWithCrane_HTFSMethod(IFlowStream stream, double massFlux)
	{
		IPhase combinedLiquidPhase = stream.CombinedLiquidPhase;
		IPhase phaseByType = stream.GetPhaseByType(PhaseType.Vapour);
		if (combinedLiquidPhase.Viscosity.Value / phaseByType.Viscosity.Value < 1000.0)
		{
			return TwoPhaseFlow.MultiplierOfFriedel(massFlux, Ports[0].Diameter.Value, combinedLiquidPhase, phaseByType);
		}
		if (massFlux > 100.0)
		{
			double y = Math.Sqrt(combinedLiquidPhase.Density.Value / phaseByType.Density.Value);
			return TwoPhaseFlow.MultiplierOfChisholm(y, massFlux, phaseByType.MassFraction.Value, 0.0);
		}
		double constC = TwoPhaseFlow.CoefficeCByFittingsName("SuddenEnlargement", combinedLiquidPhase.Density.Value, phaseByType.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType.Viscosity.Value, phaseByType.MassFraction.Value, massFlux);
		double xtt = TwoPhaseFlow.XLMParameter(Ports[0].Diameter.Value, massFlux, 0.0, phaseByType.MassFraction.Value, combinedLiquidPhase.Density.Value, phaseByType.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType.Viscosity.Value);
		return TwoPhaseFlow.MultiplierOfMartinelli(constC, xtt) * HTFSUtility.Square(combinedLiquidPhase.MassFraction.Value);
	}

	protected override double calculateMultiplierWithHTFSMethod(IFlowStream stream, double massFlux)
	{
		IPhase combinedLiquidPhase = stream.CombinedLiquidPhase;
		IPhase phaseByType = stream.GetPhaseByType(PhaseType.Vapour);
		return TwoPhaseFlow.EnlargementTPMultiplier(combinedLiquidPhase.Density.Value, phaseByType.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType.Viscosity.Value, phaseByType.MassFraction.Value, massFlux, 0.0);
	}

	protected override double calculateDPWithHTFSMethod(IFlowStream stream, double massFlux)
	{
		try
		{
			double fEMPTY = RealValue.FEMPTY;
			double fEMPTY2 = RealValue.FEMPTY;
			double fEMPTY3 = RealValue.FEMPTY;
			double num = massFlux * diameterRatio * diameterRatio;
			double density = getDensity(stream);
			if (stream.NumberOfPhases < 2)
			{
				K = getKFactor(stream, massFlux);
				fEMPTY2 = 0.5 * K * massFlux * massFlux / density;
				fEMPTY3 = 0.5 * (num * num / density - massFlux * massFlux / density);
				fEMPTY = fEMPTY2 + fEMPTY3;
			}
			else
			{
				double kFactorForStaticPressureDrop = getKFactorForStaticPressureDrop(stream, massFlux);
				fEMPTY = 0.5 * kFactorForStaticPressureDrop * massFlux * massFlux / density;
			}
			if (Math.Abs(fEMPTY / stream.GetPhaseByID(0).Pressure.Value) > 0.15)
			{
				double num2;
				if (stream.NumberOfPhases < 2)
				{
					IPhase phaseByID = stream.GetPhaseByID(1);
					num2 = SinglePhaseFlow.DownstreamPressureForCompressibleEnlargement(Ports[1].Diameter.Value, Ports[0].Diameter.Value, massFlux, phaseByID.Pressure.Value, phaseByID.Density.Value, phaseByID.Cp.Value / phaseByID.Cv.Value);
				}
				else
				{
					num2 = TwoPhaseFlow.DownstreamPressureForCompressibleEnlargement(stream, pIPPP, massFlux, diameterRatio);
				}
				fEMPTY = stream.GetPhaseByID(0).Pressure.Value - num2;
			}
			if (!base.IsChoked)
			{
				calculateDH(stream, massFlux, fEMPTY);
				IFlowStream flowStream = (IFlowStream)stream.Clone();
				pIPPP.ReflashStreamWithSameComposition(flowStream, FlashType.PHFlash, flowStream.GetPhaseByID(0).Pressure.Value - fEMPTY, flowStream.GetPhaseByID(0).Enthalpy.Value + DH);
				double num3 = HTFSUtility.CalculateEquivalentDensity(stream, massFlux, 0.0);
				double num4 = HTFSUtility.CalculateEquivalentDensity(flowStream, num, 0.0);
				fEMPTY3 = 0.5 * (num * num / num4 - massFlux * massFlux / num3);
				fEMPTY2 = fEMPTY - fEMPTY3;
				if (fEMPTY2 <= 0.0)
				{
					fEMPTY2 = 0.0;
				}
				DPf = fEMPTY2 * StepwiseFactor;
				DPa = fEMPTY3 * StepwiseFactor;
			}
			return fEMPTY;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "Enlargement.calculateDPWithHTFSMethod"), innerException);
		}
	}

	protected override double calculateDPWithCraneMethod(IFlowStream stream, double massFlux)
	{
		K = getKFactor(stream, massFlux);
		double density = getDensity(stream);
		double num = 0.5 * K * massFlux * massFlux / density;
		double num2 = 0.0 - Math.Abs(CalculateDPa(stream, massFlux, num * StepwiseFactor));
		double result = num + num2;
		DPf = num * StepwiseFactor;
		DPa = num2 * StepwiseFactor;
		return result;
	}

	public override ICorrelationManager CreateCorrelation(MethodType lMethod, IFlowStream stream)
	{
		ICorrelationManager result = null;
		switch (lMethod)
		{
		case MethodType.HTFSMethod:
		{
			ILiquidHoldUpCorrelation liqHoldUpCorr = new HTFSLiquidHoldUpCorrelation(stream, base.UpstreamPipeDiameter, 0.0, 0.0, HTFSLiquidHoldUpCorrelation.FittingType.Expansion);
			result = new EnlargementHTFSFrictionCorrelation(stream, base.UpstreamPipeDiameter, base.DownstreamPipeDiameter, base.Angle, liqHoldUpCorr);
			break;
		}
		case MethodType.Crane_HTFSMethod:
		{
			ILiquidHoldUpCorrelation liqHoldUpCorr = new HTFSLiquidHoldUpCorrelation(stream, base.UpstreamPipeDiameter, 0.0, 0.0, HTFSLiquidHoldUpCorrelation.FittingType.Expansion);
			result = new EnlargementCraneFrictionCorrelation(stream, base.UpstreamPipeDiameter, base.DownstreamPipeDiameter, base.Angle, liqHoldUpCorr);
			break;
		}
		case MethodType.CraneMethod:
		{
			ILiquidHoldUpCorrelation liqHoldUpCorr = new HomogeneousLiquidHoldUpCorrelation(stream, base.UpstreamPipeDiameter, 0.0, 0.0);
			result = new EnlargementCraneFrictionCorrelation(stream, base.UpstreamPipeDiameter, base.DownstreamPipeDiameter, base.Angle, liqHoldUpCorr);
			break;
		}
		case MethodType.TotalPressureMethod:
			result = new UserFrictionFactorCorrelation(stream, 0.0, base.UpstreamPipeDiameter, base.DownstreamPipeDiameter);
			break;
		}
		return result;
	}

	public override string ToString()
	{
		return Resources.FittingEnlargement;
	}

	protected override bool CheckGeometry()
	{
		try
		{
			int num = 0;
			int num2 = 0;
			for (int i = 0; i < NoOfPorts; i++)
			{
				if (Ports[i].Diameter.IsUnknown)
				{
					if (calculationMode == CalculationMode.UnknownMode || calculationMode == CalculationMode.SizingMode)
					{
						return false;
					}
					string pleaseSetDiam = Resources.PleaseSetDiam;
					LogMessage(MessageLogType.Fatal, pleaseSetDiam);
					throw new Exception(pleaseSetDiam);
				}
				if (Ports[i].Type == PortType.In)
				{
					num++;
				}
				else if (Ports[i].Type == PortType.Out)
				{
					num2++;
				}
			}
			if (num < 1 || num2 < 1)
			{
				string need1In1Out = Resources.Need1In1Out;
				LogMessage(MessageLogType.Fatal, need1In1Out);
				if (task == TaskType.Design)
				{
					throw new Exception(need1In1Out);
				}
			}
			return true;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "CheckGeometry"), innerException);
		}
	}

	public override double CalculatePressureDrop()
	{
		bool flag = true;
		double result = 0.0;
		if (calculationMode == CalculationMode.UnknownMode || calculationMode == CalculationMode.PressureDropMode)
		{
			flag = CheckingForDPCalculation();
		}
		if (flag)
		{
			if (FlowDirectionFactor < 0.0)
			{
				LogMessage(MessageLogType.Warning, Resources.RevFlowFlareTip);
			}
			DPz = 0.0;
			DPf = 0.0;
			DPa = 0.0;
			IFlowStream stream = Ports[StartPortNo].Stream;
			double value = Ports[0].Diameter.Value;
			double area = HTFSUtility.FlowArea(value);
			if (checkChoking)
			{
				base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, stream, area);
			}
			if (!base.IsChoked)
			{
				IFlowStream flowStream = (IFlowStream)stream.Clone();
				double area2 = HTFSUtility.FlowArea(Ports[0].Diameter.Value);
				double massFlux = HTFSUtility.CalculateMassFlux(flowStream.MassFlowrate.Value, area2);
				result = calculateDP(flowStream, massFlux);
			}
			if (calculationMode == CalculationMode.PressureDropMode)
			{
				if (base.IsChoked)
				{
					if (!(StepwiseFactor > 0.0))
					{
						LogStatus("ChokedFlowInletSpec", 500);
						throw new Exception(Resources.ChokedFlowInletPSpec);
					}
					PipeFlowFunctions.getChokingCondition(pIPPP, Ports[1].Stream, Ports[0].Stream, Ports[0].Diameter.Value, getMinimumDiameter(), (int)methodType, requiredAccuracy, StepwiseFactor);
					LogMessage(MessageLogType.Fatal, Resources.FlowChokedUpstreamEstimated);
				}
				else
				{
					updateOtherPort(CalculatedResult.PressureDrop, result);
				}
			}
		}
		return result;
	}

	protected override double calculateDP(IFlowStream stream, double MassFlux)
	{
		double result = 0.0;
		if (methodType == MethodType.PressureCurveMethod || methodType == MethodType.StaticFrictionFactorMethod || methodType == MethodType.TotalFrictionFactorMethod)
		{
			PipeFlowPressureSolver pipeFlowPressureSolver = CreateIncompressibleSolver(methodType, stream);
			if (pipeFlowPressureSolver != null)
			{
				IFlowStream stream2 = Ports[EndPortNo].Stream;
				if (pipeFlowPressureSolver.CalculateUnknownStream(stream2, stream, 1.0, StepwiseFactor, FlowDirectionFactor))
				{
					result = stream2.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
					DH = stream2.GetPhaseByID(0).Enthalpy.Value - stream.GetPhaseByID(0).Enthalpy.Value;
				}
			}
		}
		return result;
	}

	public override ICorrelationManager CreateCorrelation(MethodType lMethod, IFlowStream stream)
	{
		ICorrelationManager result = null;
		switch (lMethod)
		{
		case MethodType.PressureCurveMethod:
		{
			PressureDropCurveCalculation pressureDropCurveCalculation = new PressureDropCurveCalculation(stream, PressureDropCurves);
			pressureDropCurveCalculation.ReferencePressure = OutletPressure();
			pressureDropCurveCalculation.PressureCorrection = CurvePressureCorrection;
			pressureDropCurveCalculation.DensityCorrectionMethod = PressureDropCurveCalculation.DensityMethod.IdealGas;
			result = pressureDropCurveCalculation;
			break;
		}
		case MethodType.StaticFrictionFactorMethod:
		case MethodType.TotalFrictionFactorMethod:
			result = new UserFrictionFactorCorrelation(stream, UserFrictionFactor, TipDiameter, 1000.0 * TipDiameter);
			break;
		}
		return result;
	}

	protected override PipeFlowPressureSolver CreateIncompressibleSolver(MethodType lMethod, IFlowStream stream)
	{
		PipeFlowPressureSolver pipeFlowPressureSolver = new PipeFlowPressureSolverLengthMethod();
		pipeFlowPressureSolver.HeatFlux = 0.0;
		pipeFlowPressureSolver.PhysicalPropertyPackage = pIPPP;
		pipeFlowPressureSolver.Accuracy = base.RequiredAccuracy;
		pipeFlowPressureSolver.SetGeometricalParameters(TipDiameter, 1000.0 * TipDiameter, 0.0);
		pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseUnknownStream;
		pipeFlowPressureSolver.Correlation = CreateCorrelation(lMethod, stream);
		if (lMethod == MethodType.StaticFrictionFactorMethod || lMethod == MethodType.PressureCurveMethod)
		{
			pipeFlowPressureSolver.IncludeAccelerationPressureDrop = AccelerationPressureMethodEnum.Ignore;
		}
		pipeFlowPressureSolver.IncludeMomentumEnergy = base.IncludeKineticEnergy;
		pipeFlowPressureSolver.RatedFlowRatio = base.RatedFlowRatio;
		if (base.LogMessages is MessageLogger)
		{
			pipeFlowPressureSolver.Log = (MessageLogger)base.LogMessages;
		}
		return pipeFlowPressureSolver;
	}

	internal double OutletPressure()
	{
		return Ports[1].Stream.GetPhaseByID(0).Pressure.Value;
	}

			result = new GateValveCraneFrictionCorrelation(stream, base.PipeDiameter, base.ThroatDiameter, base.InletAngle, Openness, liquidHoldUpCorrelation);
			break;
		case MethodType.TotalPressureMethod:
			result = new UserFrictionFactorCorrelation(stream, 0.0, base.PipeDiameter, base.PipeDiameter);
			break;

		return result;

	protected override bool CheckGeometry()
	{
		bool flag = true;
		if (nCuts > 1)
		{
			if (radius <= 0.0)
			{
				LogStatus("BendNegativeRadius", 500);
				LogMessage(MessageLogType.Fatal, Resources.NegativeRadiusWarning);
				throw new Exception(Resources.NegativeRadiusWarningB);
			}
			if (angle != 90.0)
			{
				LogMessage(MessageLogType.Note, Resources.BendAngle90MoreThan1Cut);
				angle = 90.0;
			}
		}
		if (flag)
		{
			flag = base.CheckGeometry();
		}
		return flag;
	}

	private void InitialiseMitreBend()
	{
		nCuts = 1;
		nGuideVanes = 0;
	}

	public override void CalculateKAndf(IFlowStream stream, double MassFlux, out double K, out double f)
	{
		if (StartPortNo < 0)
		{
			StartPortNo = 0;
		}
		double value = Ports[StartPortNo].Diameter.Value;
		if (methodType == MethodType.CraneMethod || methodType == MethodType.Crane_HTFSMethod)
		{
			K = Crane.KFactorForMitreBend(value, angle);
			f = RealValue.FEMPTY;
			return;
		}
		double viscosity = ((stream.NumberOfPhases >= 2) ? stream.CombinedLiquidPhase.Viscosity.Value : stream.GetPhaseByID(1).Viscosity.Value);
		K = SinglePhaseFlow.KFactorForMitreBend(value, radius, nCuts, MassFlux, angle, viscosity);
		double re = SinglePhaseFlow.ReynoldsNumber(MassFlux, value, viscosity);
		f = SinglePhaseFlow.FrictionFactor(re, 0.0);
	}

	public override ICorrelationManager CreateCorrelation(MethodType lMethod, IFlowStream stream)
	{
		ICorrelationManager result = null;
		ILiquidHoldUpCorrelation liquidHoldUpCorrelation = null;
		double value = Ports[0].Diameter.Value;
		switch (lMethod)
		{
		case MethodType.HTFSMethod:
			liquidHoldUpCorrelation = new HTFSLiquidHoldUpCorrelation(stream, value, 0.0, 0.0);
			result = new MitreBendHTFSFrictionCorrelation(stream, value, base.Radius, base.Angle, NumberOfCuts, NumberOfGuideVanes, liquidHoldUpCorrelation);
			break;
		case MethodType.Crane_HTFSMethod:
			liquidHoldUpCorrelation = new CraneFittingLiquidHoldUpCorrelation(stream, value, "SuddenContraction");
			result = new MitreBendCraneFrictionCorrelation(stream, value, base.Radius, base.Angle, NumberOfCuts, NumberOfGuideVanes, liquidHoldUpCorrelation);
			break;
		case MethodType.CraneMethod:
			liquidHoldUpCorrelation = new HomogeneousLiquidHoldUpCorrelation(stream, value, 0.0, 0.0);
			result = new MitreBendCraneFrictionCorrelation(stream, value, base.Radius, base.Angle, NumberOfCuts, NumberOfGuideVanes, liquidHoldUpCorrelation);
			break;
		case MethodType.TotalPressureMethod:
			result = new UserFrictionFactorCorrelation(stream, 0.0, value, value);
			break;
		}
		return result;
	}

		plateThickness = PlateThickness;
	}

	public override double CalculatePressureDrop()
	{
		bool flag = true;
		double num = 0.0;
		if (calculationMode == CalculationMode.UnknownMode || calculationMode == CalculationMode.PressureDropMode)
		{
			flag = CheckingForDPCalculation();
		}
		if (flag)
		{
			if (FlowDirectionFactor < 0.0)
			{
				LogMessage(MessageLogType.Warning, Resources.NonSymmetricOrifice);
			}
			DPz = 0.0;
			DPf = 0.0;
			DPa = 0.0;
			IFlowStream flowStream = (IFlowStream)Ports[StartPortNo].Stream.Clone();
			double area = HTFSUtility.FlowArea(throatDiam);
			if (checkChoking && StepwiseFactor < 0.0)
			{
				base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, flowStream, area);
			}
			if (!base.IsChoked)
			{
				double area2 = HTFSUtility.FlowArea(Ports[0].Diameter.Value);
				double massFlux = HTFSUtility.CalculateMassFlux(flowStream.MassFlowrate.Value, area2);
				num = 0.0;
				if (methodType == MethodType.StaticPressureMethod)
				{
					num = 0.0;
				}
				else if (methodType == MethodType.ContractionExpansionMethod)
				{
					IFlowStream flowStream2 = (IFlowStream)flowStream.Clone();
					(bool, double, double, double, double, bool) tuple = SwageCalculation(flowStream2, base.ThroatDiameter, flowStream, Ports[StartPortNo].Diameter.Value, SwageCalculationMethod, base.CompressibleTransitionFactor, StepwiseFactor);
					if (tuple.Item1)
					{
						double num2 = flowStream2.GetPhaseByID(0).Pressure.Value - flowStream.GetPhaseByID(0).Pressure.Value;
						num += num2;
						DPf += tuple.Item2;
						DPa += tuple.Item3;
						DPz += tuple.Item4;
						if (tuple.Item6)
						{
							base.IsChoked = true;
						}
						(bool, double, double, double, double, bool) tuple2 = SwageCalculation(Ports[EndPortNo].Stream, Ports[EndPortNo].Diameter.Value, flowStream2, base.ThroatDiameter, SwageCalculationMethod, base.CompressibleTransitionFactor, StepwiseFactor);
						if (tuple2.Item1)
						{
							double num3 = Ports[EndPortNo].Stream.GetPhaseByID(0).Pressure.Value - flowStream2.GetPhaseByID(0).Pressure.Value;
							num += num3;
							DPf += tuple2.Item2;
							DPa += tuple2.Item3;
							DPz += tuple2.Item4;
							if (tuple2.Item6)
							{
								base.IsChoked = true;
							}
						}
					}
				}
				else
				{
					num = calculateDP(flowStream, massFlux);
					if (base.CheckChoking && PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, flowStream, area))
					{
						LogStatus("PossibleChokeOnOrifice", 100);
					}
				}
			}
			if (calculationMode == CalculationMode.PressureDropMode)
			{
				if (base.CheckChoking && base.IsChoked)
				{
					if (StepwiseFactor > 0.0)
					{
						LogMessage(MessageLogType.Fatal, Resources.FlowChokedUpstreamEstimated);
						if (task != TaskType.Design)
						{
							return RealValue.FEMPTY;
						}
						throw new Exception(Resources.FatalErrorChoked);
					}
					LogStatus("ChokedFlowInletSpec", 500);
					throw new Exception(Resources.ChokedFlowInletPSpec);
				}
				updateOtherPort(CalculatedResult.PressureDrop, num);
			}
		}
		return num;
	}

	protected override double calculateDP(IFlowStream stream, double MassFlux)
	{
		double num = 0.0;
		bool flag = methodType == MethodType.HTFSMethod && base.CompressibleTransitionFactor == 0.0;
		bool flag2 = false;
		PipeFlowPressureSolver pipeFlowPressureSolver = CreateIncompressibleSolver(methodType, stream);
		if (pipeFlowPressureSolver != null)
		{
			IFlowStream stream2 = Ports[EndPortNo].Stream;
			if (pipeFlowPressureSolver.CalculateUnknownStream(stream2, stream, 1.0, StepwiseFactor, FlowDirectionFactor))
			{
				num = stream2.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
				DH = stream2.GetPhaseByID(0).Enthalpy.Value - stream.GetPhaseByID(0).Enthalpy.Value;
				flag2 = stream.GetPhaseByType(PhaseType.Vapour) == null || stream2.GetPhaseByType(PhaseType.Vapour) == null;
			}
		}
		if (!flag2)
		{
			bool flag3 = methodType == MethodType.HTFSMethod && base.CompressibleTransitionFactor >= 0.0 && base.CompressibleTransitionFactor < 1.0;
			double value = stream.GetPhaseByID(0).Pressure.Value;
			if (flag3 && Math.Abs(num / value) >= base.CompressibleTransitionFactor)
			{
				IFlowStream flowStream = (IFlowStream)stream.Clone();
				if (CompressiblePressureCalculation(flowStream, stream, 1.0, StepwiseFactor, FlowDirectionFactor))
				{
					num = flowStream.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
					DH = flowStream.GetPhaseByID(0).Enthalpy.Value - stream.GetPhaseByID(0).Enthalpy.Value;
				}
			}
		}
		return num;
	}

			result = new OrificeHTFSFrictionCorrelation(stream, base.PipeDiameter, base.ThroatDiameter, PlateThickness, liquidHoldUpCorrelation);
			break;
		case MethodType.CraneMethod:
			liquidHoldUpCorrelation = new HomogeneousLiquidHoldUpCorrelation(stream, base.PipeDiameter, 0.0, 0.0);
			result = new OrificeCraneFrictionCorrelation(stream, base.PipeDiameter, base.ThroatDiameter, PlateThickness, liquidHoldUpCorrelation);
			break;
		case MethodType.Crane_HTFSMethod:
			liquidHoldUpCorrelation = new HTFSLiquidHoldUpCorrelation(stream, base.PipeDiameter, 0.0, 0.0, HTFSLiquidHoldUpCorrelation.FittingType.Orifice);

		case MethodType.TotalPressureMethod:
			result = new UserFrictionFactorCorrelation(stream, 0.0, base.PipeDiameter, base.PipeDiameter);
			break;

	protected override PipeFlowPressureSolver CreateIncompressibleSolver(MethodType lMethod, IFlowStream stream)

		else
		{
			pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseKnownStream;
		}

		pipeFlowPressureSolver.Correlation = CreateCorrelation(lMethod, stream);
		if (base.LogMessages is MessageLogger)
		{
			pipeFlowPressureSolver.Log = (MessageLogger)base.LogMessages;

			}

	protected override double calculateKFactorWithHTFSMethod(IFlowStream stream0, double MassFlux)
	{
		K = SinglePhaseFlow.KFactorForOrifice(Viscosity: (stream0.NumberOfPhases >= 2) ? stream0.CombinedLiquidPhase.Viscosity.Value : stream0.GetPhaseByID(1).Viscosity.Value, PipeDiam: Ports[0].Diameter.Value, ThroatDiam: throatDiam, PlateThickness: plateThickness, MassFlux: MassFlux);
		double x = throatDiam / Ports[0].Diameter.Value;
		K *= Math.Pow(x, 4.0);
		if (stream0.NumberOfPhases > 1)
		{
			double num = TwoPhaseFlow.ValvePlateMultiplier(stream0.CombinedLiquidPhase.Density.Value, stream0.GetPhaseByType(PhaseType.Vapour).Density.Value, stream0.GetPhaseByType(PhaseType.Vapour).MassFraction.Value);
			K *= num;
		}
		return K;
	}

	{
		double num = 0.0;
		num = Crane.KFactorForOrifice(Ports[0].Diameter.Value, throatDiam, plateThickness);
		double x = throatDiam / Ports[0].Diameter.Value;
		return num * Math.Pow(x, 4.0);
	}

	protected override double calculateKFactorWithCrane_HTFSMethod(IFlowStream stream, double MassFlux)
	{
		double num = calculateKFactorWithCraneMethod();
		if (stream.NumberOfPhases < 2)
		{
			return num;
		}
		IPhase combinedLiquidPhase = stream.CombinedLiquidPhase;
		IPhase phaseByType = stream.GetPhaseByType(PhaseType.Vapour);
		double num2;
		if (combinedLiquidPhase.Viscosity.Value / phaseByType.Viscosity.Value < 1000.0)
		{
			num2 = TwoPhaseFlow.MultiplierOfFriedel(MassFlux, Ports[0].Diameter.Value, combinedLiquidPhase, phaseByType);
		}
		else if (MassFlux > 100.0)
		{
			double y = Math.Sqrt(combinedLiquidPhase.Density.Value / phaseByType.Density.Value);
			num2 = TwoPhaseFlow.MultiplierOfChisholm(y, MassFlux, phaseByType.MassFraction.Value, 0.0);
		}
		else
		{
			double constC = TwoPhaseFlow.CoefficeCByFittingsName("ThinPlate", combinedLiquidPhase.Density.Value, phaseByType.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType.Viscosity.Value, phaseByType.MassFraction.Value, MassFlux);
			double xtt = TwoPhaseFlow.XLMParameter(Ports[0].Diameter.Value, MassFlux, 0.0, phaseByType.MassFraction.Value, combinedLiquidPhase.Density.Value, phaseByType.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType.Viscosity.Value);
			num2 = TwoPhaseFlow.MultiplierOfMartinelli(constC, xtt) * HTFSUtility.Square(combinedLiquidPhase.MassFraction.Value);
		}
		return num * num2;
	}

	public override double CalculatePressureDrop()
	{
		bool flag = true;
		double result = RealValue.FEMPTY;
		try
		{
			if (calculationMode == CalculationMode.UnknownMode || calculationMode == CalculationMode.PressureDropMode)
			{
				flag = CheckingForDPCalculation();
			}
			if (flag)
			{
				double value = Ports[StartPortNo].Diameter.Value;
				double area = HTFSUtility.FlowArea(value);
				IFlowStream stream = Ports[StartPortNo].Stream;
				IFlowStream stream2 = Ports[EndPortNo].Stream;
				if (checkChoking)
				{
					base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, stream, area);
				}
				if (!base.IsChoked)
				{
					result = 0.0;
					DH = 0.0;
					if (methodType != MethodType.AdiabaticGasMethod && methodType != MethodType.IsothermalGasMethod)
					{
						PipeFlowPressureSolver pipeFlowPressureSolver = CreateIncompressibleSolver(methodType, stream);
						if (pipeFlowPressureSolver != null && pipeFlowPressureSolver.CalculateUnknownStream(stream2, stream, Length.Value, StepwiseFactor, FlowDirectionFactor))
						{
							if (checkChoking)
							{
								base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, stream2, area);
							}
							result = stream2.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
							DH = stream2.GetPhaseByID(0).Enthalpy.Value - stream.GetPhaseByID(0).Enthalpy.Value;
							FlowRegimeEnum flowRegimeEnum = (pipeFlowPressureSolver.Correlation as PipeCorrelationManager).EstimateFlowPattern();
							stream.FlowRegime = flowRegimeEnum.ToString();
							stream2.FlowRegime = flowRegimeEnum.ToString();
							FlowRegimeEnum flowRegimeEnum2 = (pipeFlowPressureSolver.Correlation as PipeCorrelationManager).EstimateWaterFlowPattern();
							stream.WaterFlowRegime = flowRegimeEnum2.ToString();
							stream2.WaterFlowRegime = flowRegimeEnum2.ToString();
							mLiquidHoldup = (pipeFlowPressureSolver.Correlation as PipeCorrelationManager).CalculateLiquidHoldUp();
							mWaterHoldup = (pipeFlowPressureSolver.Correlation as PipeCorrelationManager).CalculateWaterHoldUp();
							DPf = pipeFlowPressureSolver.FrictionPressureDrop;
							DPz = pipeFlowPressureSolver.GravityPressureDrop;
							DPa = pipeFlowPressureSolver.AccelerationPressureDrop;
							mEquivLength = (pipeFlowPressureSolver.Correlation as PipeCorrelationManager).EquivalentPipeLength;
							frictionFactor = (pipeFlowPressureSolver.Correlation as PipeCorrelationManager).UsedFrictionFactor;
							mRe = (pipeFlowPressureSolver.Correlation as PipeCorrelationManager).UsedReynoldsNumber;
						}
					}
					else if (CompressiblePressureCalculation(stream2, stream, Length.Value, StepwiseFactor, FlowDirectionFactor))
					{
						result = stream2.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
						DH = stream2.GetPhaseByID(0).Enthalpy.Value - stream.GetPhaseByID(0).Enthalpy.Value;
						DPf = RealValue.FEMPTY;
						DPz = RealValue.FEMPTY;
						DPa = RealValue.FEMPTY;
					}
				}
				if (calculationMode == CalculationMode.PressureDropMode)
				{
					if (base.IsChoked)
					{
						if (StepwiseFactor > 0.0)
						{
							PipeFlowFunctions.getChokingCondition(pIPPP, Ports[1].Stream, Ports[0].Stream, Ports[0].Diameter.Value, getMinimumDiameter(), (int)methodType, requiredAccuracy, StepwiseFactor);
							LogMessage(MessageLogType.Fatal, Resources.FlowChokedUpstreamEstimated);
							if (task != TaskType.Design || StepwiseFactor > 0.0)
							{
								return -1.0;
							}
							throw new Exception(Resources.ChokedFlowTxt);
						}
						LogStatus("ChokedFlowInletSpec", 500);
						throw new Exception(Resources.ChokedFlowInletPSpec);
					}
					updateOtherPort(CalculatedResult.PressureDrop, result);
				}
				return result;
			}
			return 0.0;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "PipeCell.CalculatePressureDrop"), innerException);
		}
	}

	protected override bool CompressiblePressureCalculation(IFlowStream unknownStream, IFlowStream knownStream, double fittingLength, double stepFactor, double flowFactor)
	{
		bool flag = false;
		if (methodType == MethodType.IsothermalGasMethod || methodType == MethodType.AdiabaticGasMethod)
		{
			GasPressureSolver gasPressureSolver = null;
			if (methodType == MethodType.IsothermalGasMethod)
			{
				gasPressureSolver = new IsothermalGasPressureSolver();
			}
			else if (methodType == MethodType.AdiabaticGasMethod)
			{
				gasPressureSolver = new AdiabaticGasPressureSolver();
			}
			if (gasPressureSolver != null)
			{
				gasPressureSolver.PipeRoughness = Roughness;
				gasPressureSolver.Accuracy = base.RequiredAccuracy;
				double value = Ports[StartPortNo].Diameter.Value;
				gasPressureSolver.SetGeometricalParameters(value, value, Angle);
				gasPressureSolver.PhysicalPropertyPackage = pIPPP;
				gasPressureSolver.FrictionFactorMethod = mFrictionFactorMethod;
				gasPressureSolver.IncludeMomentumEnergy = base.IncludeKineticEnergy;
				gasPressureSolver.FittingCoeffA = FittingCoeffA;
				gasPressureSolver.FittingCoeffB = FittingCoeffB;
				gasPressureSolver.FittingRoughness = FittingRoughness;
				gasPressureSolver.PipeLengthMultiplier = LengthMultiplier;
				gasPressureSolver.RatedFlowRatio = base.RatedFlowRatio;
				flag = gasPressureSolver.CalculateUnknownStream(unknownStream, knownStream, fittingLength, stepFactor, flowFactor);
				if (flag)
				{
					mEquivLength = gasPressureSolver.EquivalentPipeLength;
					frictionFactor = gasPressureSolver.UsedFrictionFactor;
					mRe = gasPressureSolver.UsedReynoldsNumber;
					knownStream.FlowRegime = FlowRegimeEnum.Vapour.ToString();
					unknownStream.FlowRegime = FlowRegimeEnum.Vapour.ToString();
				}
			}
		}
		return flag;
	}

	public override ICorrelationManager CreateCorrelation(MethodType lMethod, IFlowStream stream)
	{
		PipeCorrelationManager pipeCorrelationManager = null;
		IPressureGradientCorrelation pressureGradientCorrelation;
		ILiquidHoldUpCorrelation lhMethod;
		IFlowRegimeCorrelation frMethod;
		switch (methodType)
		{
		case MethodType.CraneMethod:
			pressureGradientCorrelation = new CranePressureGradientCorrelation();
			lhMethod = new HomogeneousLiquidHoldUpCorrelation();
			frMethod = null;
			break;
		case MethodType.DunsRosMethod:
			pressureGradientCorrelation = new DunsAndRosPressureGradientCorrelation();
			lhMethod = new DunsAndRosLiquidHoldUpCorrelation();
			frMethod = new DunsAndRosFlowRegimeCorrelation();
			break;
		case MethodType.AzizGovierFogarasiMethod:
			pressureGradientCorrelation = new AGFPressureGradientCorrelation();
			lhMethod = new AGFLiquidHoldUpCorrelation();
			frMethod = new AGFFlowRegimeCorrelation();
			break;
		case MethodType.GregoryAzizMandhaneMethod:
			pressureGradientCorrelation = new GAMPressureGradientCorrelation();
			lhMethod = new HomogeneousLiquidHoldUpCorrelation();
			frMethod = new MandhaneFlowRegimeCorrelation();
			break;
		case MethodType.OrkiszewskiMethod:
			pressureGradientCorrelation = new ORKPressureGradientCorrelation();
			lhMethod = new ORKLiquidHoldUpCorrelation();
			frMethod = new ORKFlowRegimeCorrelation();
			break;
		case MethodType.LockhartMartinelliMethod:
			pressureGradientCorrelation = new LockhartMartinelliPressureGradientCorrelation();
			lhMethod = new HomogeneousLiquidHoldUpCorrelation();
			frMethod = null;
			break;
		case MethodType.HagedornBrownMethod:
			pressureGradientCorrelation = new HagedornBrownPressureGradientCorrelation();
			lhMethod = new HagedornBrownLiquidHoldUpCorrelation();
			frMethod = null;
			break;
		case MethodType.DuklerMethod:
			pressureGradientCorrelation = new DuklerPressureGradientCorrelation();
			lhMethod = new DuklerLiquidHoldUpCorrelation();
			frMethod = null;
			break;
		case MethodType.BeggsBrillMethod:
			pressureGradientCorrelation = new BeggsBrillPressureGradientCorrelation();
			lhMethod = new BeggsBrillLiquidHoldUpCorrelation();
			frMethod = new BeggsBrillFlowRegimeCorrelation();
			break;
		case MethodType.TulsaUnifiedModel2P:
			pressureGradientCorrelation = new TulsaUnifiedCorrelation();
			(pressureGradientCorrelation as TulsaUnifiedCorrelation).TwoPhaseVersion = true;
			lhMethod = pressureGradientCorrelation as TulsaUnifiedCorrelation;
			frMethod = pressureGradientCorrelation as TulsaUnifiedCorrelation;
			break;
		case MethodType.TulsaUnifiedModel3P:
			pressureGradientCorrelation = new TulsaUnifiedCorrelation();
			(pressureGradientCorrelation as TulsaUnifiedCorrelation).TwoPhaseVersion = false;
			lhMethod = pressureGradientCorrelation as TulsaUnifiedCorrelation;
			frMethod = pressureGradientCorrelation as TulsaUnifiedCorrelation;
			break;
		default:
			pressureGradientCorrelation = new HTFSPressureGradientCorrelation();
			lhMethod = new HTFSLiquidHoldUpCorrelation();
			frMethod = null;
			break;
		}
		double value = Ports[StartPortNo].Diameter.Value;
		pipeCorrelationManager = new PipeCorrelationManager(pressureGradientCorrelation, lhMethod, frMethod, stream, value, Length.Value, Angle, Roughness);
		pipeCorrelationManager.FrictionFactorMethod = FrictionFactorMethod;
		pipeCorrelationManager.FittingCoeffA = FittingCoeffA;
		pipeCorrelationManager.FittingCoeffB = FittingCoeffB;
		pipeCorrelationManager.FittingRoughness = FittingRoughness;
		pipeCorrelationManager.PipeLengthMultiplier = LengthMultiplier;
		pipeCorrelationManager.UseAGAHead = mUseAGAHead;
		return pipeCorrelationManager;
	}

	protected override PipeFlowPressureSolver CreateIncompressibleSolver(MethodType lMethod, IFlowStream stream)
	{
		MethodType methodType = base.methodType;
		if ((uint)methodType > 1u)
		{
			_ = methodType - 4;
			_ = 7;
		}
		PipeFlowPressureSolver pipeFlowPressureSolver = new PipeFlowPressureSolverAveragePropertiesMethod();
		(pipeFlowPressureSolver as PipeFlowPressureSolverAveragePropertiesMethod).DoInnerLoop = false;
		if (isAdiabatic)
		{
			pipeFlowPressureSolver.HeatFlux = 0.0;
		}
		else
		{
			pipeFlowPressureSolver.HeatFlux = HeatFlux;
			pipeFlowPressureSolver.ExternalTemperature = externalTemp;
		}
		pipeFlowPressureSolver.PhysicalPropertyPackage = pIPPP;
		pipeFlowPressureSolver.Accuracy = base.RequiredAccuracy;
		double value = Ports[StartPortNo].Diameter.Value;
		pipeFlowPressureSolver.SetGeometricalParameters(value, value, Angle);
		pipeFlowPressureSolver.IncludeMomentumEnergy = base.IncludeKineticEnergy;
		pipeFlowPressureSolver.StaticHeadMethod = StaticHeadMethod;
		pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseAverage;
		pipeFlowPressureSolver.IncludeAccelerationPressureDrop = AccelerationMethod;
		pipeFlowPressureSolver.CalculateEnthalpyChange = false;
		MethodType methodType2 = base.methodType;
		if (methodType2 == MethodType.DuklerMethod)
		{
			pipeFlowPressureSolver.IgnoreDeccelerationPressureDrop = true;
		}
		else
		{
			pipeFlowPressureSolver.IgnoreDeccelerationPressureDrop = false;
		}
		pipeFlowPressureSolver.RatedFlowRatio = base.RatedFlowRatio;
		pipeFlowPressureSolver.Correlation = CreateCorrelation(lMethod, stream);
		if (base.LogMessages is MessageLogger)
		{
			pipeFlowPressureSolver.Log = (MessageLogger)base.LogMessages;
		}
		return pipeFlowPressureSolver;
	}

	protected override bool CheckGeometry()

		{
			LogStatus("PipeBadInclinationAngleRange", 500);
			LogMessage(MessageLogType.Fatal, Resources.PipeInclineTooLarge1);
			throw new Exception(Resources.PipeInclineTooLarge2);
		}
		return base.CheckGeometry();
	}

			throw new Exception(string.Format(Resources.MethodFailed, "PipeCell.MakeEnergyBalance"), innerException);
		}
	}

	public virtual double CalculateMassFlowrate()
	{
		double num = 0.0;
		double num2 = 100.0;
		calculationMode = CalculationMode.MassFlowrateMode;
		if (BCType == BoundaryConditionType.UnknownBC)
		{
			CheckBoundaryCondition();
		}
		CheckPressureDrop();
		CheckGeometry();
		switch (BCType)
		{
		case BoundaryConditionType.InletBC:
		case BoundaryConditionType.OutletBC:
		{
			num = getInitialMassFlowrate(Ports[StartPortNo]);
			double num3 = 5.0 * num;
			double num4 = 0.0;
			double num5 = 1000.0 * num;
			while (num2 > requiredAccuracy || (num3 - num4) / num < 0.2 * requiredAccuracy)
			{
				double num6 = CalculatePressureDrop();
				num2 = Math.Abs(1.0 - num6 / DP);
				if (IsChoked)
				{
					num5 = num;
					if (Math.Abs(num6) < Math.Abs(DP))
					{
						if (task != TaskType.Design)
						{
							return -1.0;
						}
						LogMessage(MessageLogType.Fatal, Resources.InputDPTooLarge);
						throw new Exception(Resources.InputDPTooLargeFE);
					}
				}
				if (!(num2 < requiredAccuracy))
				{
					if (Math.Abs(num6) > Math.Abs(DP) || IsChoked)
					{
						num3 = num;
						num = 0.5 * (num3 + num4);
						IsChoked = false;
					}
					else
					{
						num4 = num;
						double num7 = Math.Min(0.25 * num, Math.Min(0.5 * Math.Abs(num3 - num), 0.5 * Math.Abs(num5 - num)));
						num += num7;
					}
				}
				IRealValue massFlowrate = Ports[0].Stream.MassFlowrate;
				double value = (Ports[1].Stream.MassFlowrate.Value = num);
				massFlowrate.Value = value;
			}
			updateOtherPort(CalculatedResult.MassFlowrate, num);
			return num;
		}
		case BoundaryConditionType.PartialBC:
			if (task != TaskType.Design)
			{
				return -1.0;
			}
			LogMessage(MessageLogType.Fatal, Resources.PartialBCNotSupported);
			throw new Exception(Resources.PartialBCNotSupportedFE);
		default:
			LogMessage(MessageLogType.Fatal, Resources.CheckingMassFlow);
			throw new Exception(Resources.CheckingMassFlowFE);

	public override double CalculatePressureDrop()
	{
		bool flag = true;
		if (calculationMode == CalculationMode.UnknownMode || calculationMode == CalculationMode.PressureDropMode)
		{
			flag = CheckingForDPCalculation();
		}
		if (flag)
		{
			double num = 0.0;
			switch (methodType)
			{
			case MethodType.DirectDeltaPMethod:
				num = ((!(StepwiseFactor * FlowDirectionFactor < 0.0)) ? (0.0 - deltaP) : deltaP);
				break;
			case MethodType.PressureRatioMethod:
				num = ((!(StepwiseFactor * FlowDirectionFactor < 0.0)) ? ((0.0 - (1.0 - 1.0 / pRatio)) * Ports[StartPortNo].Stream.GetPhaseByID(0).Pressure.Value) : ((pRatio - 1.0) * Ports[StartPortNo].Stream.GetPhaseByID(0).Pressure.Value));
				break;
			case MethodType.PolynomialCurveMethod:
			{
				double num2 = 0.0;
				switch (mPolyCurveFlowBasis)
				{
				case FlowBasisType.MolarFlow:
					num2 = Ports[StartPortNo].Stream.MassFlowrate.Value / Ports[StartPortNo].Stream.GetPhaseByID(0).MolecularWeight.Value;
					break;
				case FlowBasisType.MassFlow:
					num2 = Ports[StartPortNo].Stream.MassFlowrate.Value;
					break;
				case FlowBasisType.ActualVolumeFlow:
					num2 = Ports[StartPortNo].Stream.MassFlowrate.Value / Ports[StartPortNo].Stream.GetPhaseByID(0).Density.Value;
					break;
				}
				double num3 = num2 * num2;
				double num4 = num2 * num3;
				num = Ports[StartPortNo].Stream.GetPhaseByID(0).Density.Value * (coeffA + coeffB * num2 + coeffC * num3 + coeffD * num4 + coeffE * num4 * num2 + coeffF * num4 * num3);
				if (StepwiseFactor * FlowDirectionFactor > 0.0)
				{
					num = 0.0 - num;
				}
				break;
			}
			}
			if (calculationMode == CalculationMode.PressureDropMode)
			{
				updateOtherPort(CalculatedResult.PressureDrop, num);
			}
			return num;
		}
		return RealValue.FEMPTY;
	}

	protected override PipeFlowPressureSolver CreateIncompressibleSolver(MethodType lMethod, IFlowStream stream)

	public override double CalculatePressureDrop()
	{
		try
		{
			bool flag = true;
			double result = 0.0;
			if (calculationMode == CalculationMode.UnknownMode || calculationMode == CalculationMode.PressureDropMode)
			{
				flag = CheckingForDPCalculation();
			}
			if (FlowDirectionFactor < 0.0)
			{
				LogMessage(MessageLogType.Fatal, Resources.CreateExpansion);
				if (task == TaskType.Design)
				{
					throw new Exception(Resources.CreateExpansionFE);
				}
				flag = false;
			}
			if (flag)
			{
				DPz = 0.0;
				DPf = 0.0;
				DPa = 0.0;
				K = 0.0;
				if (Math.Abs(diameterRatio - 1.0) < requiredAccuracy)
				{
					result = 0.0;
				}
				else
				{
					IFlowStream flowStream = (IFlowStream)Ports[StartPortNo].Stream.Clone();
					double diameter = Math.Min(Ports[0].Diameter.Value, Ports[1].Diameter.Value);
					double area = HTFSUtility.FlowArea(diameter);
					if (checkChoking)
					{
						base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, flowStream, area);
					}
					if (!base.IsChoked)
					{
						if (methodType == MethodType.StaticPressureMethod)
						{
							result = 0.0;
						}
						else
						{
							double area2 = HTFSUtility.FlowArea(Ports[0].Diameter.Value);
							double massFlux = HTFSUtility.CalculateMassFlux(flowStream.MassFlowrate.Value, area2);
							result = calculateDP(flowStream, massFlux);
							if (checkChoking)
							{
								base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, flowStream, area2);
							}
						}
					}
				}
				if (calculationMode == CalculationMode.PressureDropMode)
				{
					if (base.IsChoked)
					{
						chokedLocation = ChokedLocationEnum.Throat;
						if (StepwiseFactor > 0.0)
						{
							PipeFlowFunctions.getChokingCondition(pIPPP, Ports[1].Stream, Ports[0].Stream, Ports[0].Diameter.Value, getMinimumDiameter(), (int)methodType, requiredAccuracy, StepwiseFactor);
							LogMessage(MessageLogType.Fatal, Resources.FlowChokedUpstreamEstimated);
							if (task != TaskType.Design)
							{
								return RealValue.FEMPTY;
							}
							throw new Exception(Resources.FatalErrorChoked);
						}
						LogStatus("ChokedFlowInletSpec", 500);
						throw new Exception(Resources.ChokedFlowInletPSpec);
					}
					updateOtherPort(CalculatedResult.PressureDrop, result);
				}
				return result;
			}
			return RealValue.FEMPTY;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "Swage.CalculatePressureDrop"), innerException);
		}
	}

	protected abstract double calculateDPWithHTFSMethod(IFlowStream stream, double massFlux);

	protected abstract double calculateDPWithCraneMethod(IFlowStream stream, double massFlux);

	protected abstract double calculateKFactorWithHTFSMethod(IFlowStream stream0, double MassFlux);

	protected abstract double calculateKFactorWithCraneMethod();

	protected override double calculateDP(IFlowStream stream, double massFlux)
	{
		try
		{
			double num = RealValue.FEMPTY;
			bool flag = methodType == MethodType.HTFSMethod && base.CompressibleTransitionFactor == 0.0;
			if (!flag)
			{
				PipeFlowPressureSolver pipeFlowPressureSolver = CreateIncompressibleSolver(methodType, stream);
				if (pipeFlowPressureSolver != null)
				{
					IFlowStream stream2 = Ports[EndPortNo].Stream;
					if (pipeFlowPressureSolver.CalculateUnknownStream(stream2, stream, 1.0, StepwiseFactor, FlowDirectionFactor))
					{
						num = stream2.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
						DH = stream2.GetPhaseByID(0).Enthalpy.Value - stream.GetPhaseByID(0).Enthalpy.Value;
						DPf = pipeFlowPressureSolver.FrictionPressureDrop;
						DPz = pipeFlowPressureSolver.GravityPressureDrop;
						DPa = pipeFlowPressureSolver.AccelerationPressureDrop;
						if (pipeFlowPressureSolver.Correlation is SwageFrictionFactorCorrelation swageFrictionFactorCorrelation)
						{
							K = swageFrictionFactorCorrelation.CalculatedKFactor;
						}
						else
						{
							K = 0.0;
						}
					}
				}
			}
			bool flag2 = methodType == MethodType.HTFSMethod && base.CompressibleTransitionFactor > 0.0 && base.CompressibleTransitionFactor < 1.0;
			double value = stream.GetPhaseByID(0).Pressure.Value;
			if (flag || (flag2 && Math.Abs(num / value) >= base.CompressibleTransitionFactor))
			{
				IFlowStream flowStream = (IFlowStream)stream.Clone();
				if (CompressiblePressureCalculation(flowStream, stream, 1.0, StepwiseFactor, FlowDirectionFactor))
				{
					num = flowStream.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
					DH = flowStream.GetPhaseByID(0).Enthalpy.Value - stream.GetPhaseByID(0).Enthalpy.Value;
				}
				else
				{
					LogStatus("AhCompSwageFail", 200);
				}
			}
			return num;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "Swage.calculateDP"), innerException);
		}
	}

	protected override PipeFlowPressureSolver CreateIncompressibleSolver(MethodType lMethod, IFlowStream stream)
	{
		PipeFlowPressureSolver pipeFlowPressureSolver = new PipeFlowPressureSolverLengthMethod();
		pipeFlowPressureSolver.HeatFlux = 0.0;
		pipeFlowPressureSolver.PhysicalPropertyPackage = pIPPP;
		pipeFlowPressureSolver.Accuracy = base.RequiredAccuracy;
		pipeFlowPressureSolver.SetGeometricalParameters(UpstreamPipeDiameter, DownstreamPipeDiameter, 0.0);
		if (lMethod == MethodType.HTFSMethod)
		{
			pipeFlowPressureSolver.MomentumEnergyDelegate = HTFSUtility.MomentumEnergy;
		}
		pipeFlowPressureSolver.IncludeMomentumEnergy = base.IncludeKineticEnergy;
		if (StepwiseFactor == 1.0)
		{
			pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseUnknownStream;
		}
		else
		{
			pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseKnownStream;
		}
		pipeFlowPressureSolver.RatedFlowRatio = base.RatedFlowRatio;
		pipeFlowPressureSolver.Correlation = CreateCorrelation(lMethod, stream);
		if (base.LogMessages is MessageLogger)
		{
			pipeFlowPressureSolver.Log = (MessageLogger)base.LogMessages;
		}
		return pipeFlowPressureSolver;
	}

	protected override bool CompressiblePressureCalculation(IFlowStream unknownStream, IFlowStream knownStream, double fittingLength, double stepFactor, double flowFactor)
	{
		CompressibleHTFSPressureSolver compressibleHTFSPressureSolver = new CompressibleHTFSPressureSolver();
		compressibleHTFSPressureSolver.HeatFlux = 0.0;
		compressibleHTFSPressureSolver.PhysicalPropertyPackage = pIPPP;
		compressibleHTFSPressureSolver.Accuracy = base.RequiredAccuracy;
		compressibleHTFSPressureSolver.SetGeometricalParameters(UpstreamPipeDiameter, DownstreamPipeDiameter, 0.0);
		compressibleHTFSPressureSolver.MomentumEnergyDelegate = HTFSUtility.MomentumEnergy;
		compressibleHTFSPressureSolver.UseIdealGasApproximation = false;
		if (Angle < 90.0)
		{
			compressibleHTFSPressureSolver.UseVenaContracta = false;
		}
		else
		{

			}
			IPort port = null;

			{
				port = Ports[PortIndex2];
				port2 = Ports[PortIndex1];
				num2 = PortIndex2;
				num3 = PortIndex1;
			}
			if (port == null || port2 == null)
			{
				return result;
			}
			double num4 = 1.0;
			if (Ports[num2].Type == PortType.Out && Ports[num3].Type == PortType.In)

			}
			result = 0.0;
			DPf = 0.0;
			DPa = 0.0;
			DPz = 0.0;

				(bool, double, double, double, double, bool) tuple = SwageCalculation(flowStream, num5, stream, port2.Diameter.Value, SwageCalculationMethod, SwageTransitionFactor, num4);
				if (tuple.Item1)
				{
					double num7 = flowStream.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
					result += num7 * num4;
					DPf += tuple.Item2;
					DPa += tuple.Item3;
					DPz += tuple.Item4;

	internal (bool isSolved, double DPf, double DPa, double DPz, double K) TeeCalculation(int iTargetPort, IFlowStream targetStream, double targetDiam, int iSourcePort, IFlowStream sourceStream, double sourceDiam, double stepFactor)
	{
		double num = 0.0;
		bool flag = false;

			pipeFlowPressureSolverLengthMethod = new PipeFlowPressureSolverLengthMethod();
			pipeFlowPressureSolverLengthMethod.IncludeAccelerationPressureDrop = AccelerationPressureMethodEnum.Include;
			break;
		case MethodTypeForTee.MillerCharts:
		case MethodTypeForTee.Gardel:
		case MethodTypeForTee.Simple:
			if (iTargetPort == 2 || iSourcePort == 2)
			{
				if (teeMethod == MethodTypeForTee.MillerCharts)
				{
					correlationManager = new MillerMainSideDPCorrelation();
					(correlationManager as MillerMainSideDPCorrelation).AreaLimited = MillerChartAreaLimited;
					(correlationManager as MillerMainSideDPCorrelation).ExtrapolationMethod = MillerChartExtrapolation;
				}
				else if (teeMethod == MethodTypeForTee.Gardel)
				{
					correlationManager = new GardelMainSideDPCorrelation();
				}
				else if (teeMethod == MethodTypeForTee.Simple)
				{
					correlationManager = new SimpleMainSideDPCorrelation();
				}
			}
			else if (teeMethod == MethodTypeForTee.MillerCharts)
			{
				correlationManager = new MillerMainMainDPCorrelation();
				(correlationManager as MillerMainMainDPCorrelation).AreaLimited = MillerChartAreaLimited;
				(correlationManager as MillerMainMainDPCorrelation).ExtrapolationMethod = MillerChartExtrapolation;
			}
			else if (teeMethod == MethodTypeForTee.Gardel)
			{
				correlationManager = new GardelMainMainDPCorrelation();
			}
			else if (teeMethod == MethodTypeForTee.Simple)
			{
				correlationManager = new SimpleMainMainDPCorrelation();
			}
			if (correlationManager is TeePressureDropCorrelation teePressureDropCorrelation)
			{
				int num2 = ((Ports[0].Type != PortType.In) ? 1 : 0);
				teePressureDropCorrelation.MainInletStream = Ports[num2].Stream;
				teePressureDropCorrelation.MainOutletStream = Ports[1 - num2].Stream;
				teePressureDropCorrelation.BranchStream = Ports[2].Stream;
				teePressureDropCorrelation.BodyDiameter = getBodyDiameter();
				teePressureDropCorrelation.BranchDiameter = Ports[2].Diameter.Value;
				teePressureDropCorrelation.SideAngle = getTeeAngle(num2);
				teePressureDropCorrelation.TypeOfTee = teeType;
			}
			pipeFlowPressureSolverLengthMethod = new PipeFlowPressureSolverLengthMethod();
			pipeFlowPressureSolverLengthMethod.IncludeAccelerationPressureDrop = AccelerationPressureMethodEnum.Include;
			break;
		default:
			throw new Exception(Resources.UnrecognizedTeeMethod);
		}
		double num3 = 0.0;
		double num4 = 0.0;
		double num5 = 0.0;
		double num6 = 0.0;
		if (stepFactor == 1.0)
		{
			num3 = targetDiam;
			num4 = sourceDiam;
			num5 = targetStream.MassFlowrate.Value;
			num6 = sourceStream.MassFlowrate.Value;
		}
		else
		{
			num3 = sourceDiam;
			num4 = targetDiam;
			num5 = sourceStream.MassFlowrate.Value;
			num6 = targetStream.MassFlowrate.Value;
		}
		double item = 0.0;
		double item2 = 0.0;
		double item3 = 0.0;
		double item4 = 0.0;
		if (pipeFlowPressureSolverLengthMethod != null)

			pipeFlowPressureSolverLengthMethod.Accuracy = base.RequiredAccuracy;
			pipeFlowPressureSolverLengthMethod.SetGeometricalParameters(num3, num4, 0.0);

			pipeFlowPressureSolverLengthMethod.GradientEstimate = GradientEstimateMethod.UseUnknownStream;

			flag = pipeFlowPressureSolverLengthMethod.CalculateUnknownStream(targetStream, sourceStream, 1.0, stepFactor, FlowDirectionFactor);

			{
				item = pipeFlowPressureSolverLengthMethod.FrictionPressureDrop;
				item2 = pipeFlowPressureSolverLengthMethod.AccelerationPressureDrop;
				item3 = pipeFlowPressureSolverLengthMethod.GravityPressureDrop;
				item4 = ((!(pipeFlowPressureSolverLengthMethod.Correlation is TeePressureDropCorrelation teePressureDropCorrelation2)) ? 0.0 : teePressureDropCorrelation2.CalculatedKFactor);

			double energy = targetStream.GetPhaseByID(0).Enthalpy.Value * value;
			SetPortStateByEnergy(iTargetPort, energy);
			flag = true;
		}
		return (isSolved: flag, DPf: item, DPa: item2, DPz: item3, K: item4);
	}

			{
				if (value != getPartialBodyDiameter(portIndex))
				{
					result = true;
				}
			}
			else if (teeMethod == MethodTypeForTee.BalanceStaticPressure || teeMethod == MethodTypeForTee.BalanceTotalPressure)
			{
				result = false;
			}
			else if (value != getBodyDiameter())
			{
				result = true;
				if ((teeMethod == MethodTypeForTee.MillerCharts || teeMethod == MethodTypeForTee.Gardel) && portIndex == 2)
				{
					result = false;
				}
			}
		}
		return result;
	}

[ComVisible(false)]
public abstract class ThrottleElement : PipeFitting, IThrottleElement, IPipeFittings
{
	protected double K;

	protected double throatDiam;

	protected double diameterRatio;

	public double PipeDiameter
	{
		get
		{
			return Ports[0].Diameter.Value;
		}
		set
		{
			Ports[0].Diameter.Value = value;
			Ports[1].Diameter.Value = value;
			if (value > 0.0 && throatDiam > 0.0)
			{
				diameterRatio = throatDiam / value;
			}
		}
	}

	public double ThroatDiameter
	{
		get
		{
			return throatDiam;
		}
		set
		{
			throatDiam = value;
			if (value > 0.0 && PipeDiameter > 0.0)
			{
				diameterRatio = value / PipeDiameter;
			}
		}
	}

	public double ThroatToPipeDiameterRatio => diameterRatio;

	public ThrottleElement()
		: base(2)
	{
		InitialiseThrottleElement();
	}

	public ThrottleElement(double PipeDiam, double ThroatDiam, IFlowStream streamIn, IFlowStream streamOut)
		: base(2)
	{
		Ports[0].Diameter.Value = PipeDiam;
		Ports[0].Stream = streamIn;
		Ports[1].Diameter.Value = PipeDiam;
		Ports[1].Stream = streamOut;
		throatDiam = ThroatDiam;
		diameterRatio = ThroatDiam / PipeDiam;
	}

	private void InitialiseThrottleElement()
	{
		K = RealValue.FEMPTY;
		throatDiam = Ports[0].Diameter.Value;
		diameterRatio = 1.0;
	}

	protected override double calculateDP(IFlowStream stream, double MassFlux)
	{
		if (checkChoking)
		{
			base.IsChoked = HTFSUtility.CheckChokingForIdealFlowInContraction(stream, MassFlux, diameterRatio);
		}
		double num = 0.0;
		bool flag = false;
		double Pc_P = RealValue.FEMPTY;
		double Mc = RealValue.FEMPTY;
		if (!base.IsChoked)
		{
			K = getKFactor(stream, MassFlux);
			double num2 = diameterRatio / Math.Sqrt(1.0 + Math.Sqrt(K));
			double num3 = MassFlux / (num2 * num2);
			double density = getDensity(stream);
			num = 0.5 * K * MassFlux * MassFlux / density;
			if (num / stream.GetPhaseByID(0).Pressure.Value > 0.15 && (stream.NumberOfPhases != 1 || stream.GetPhaseByID(1).PhaseType != PhaseType.Liquid))
			{
				flag = HTFSUtility.CalculateCriticalPressureRatioAndMassFlux(stream, pIPPP, num2, out Pc_P, out Mc);
				if (Mc == RealValue.FEMPTY)
				{
					Mc = 1E+20;
				}
				if (num3 > 0.99 * Mc)
				{
					base.IsChoked = true;
				}
				if (!base.IsChoked)
				{
					double num4 = 0.0;
					if (stream.NumberOfPhases < 2)
					{
						if (stream.GetPhaseByID(1).PhaseType == PhaseType.Vapour)
						{
							num4 = SinglePhaseFlow.ThroatToUpstreamPressureRatioForContraction(stream, num2, MassFlux, Pc_P);
						}
					}
					else
					{
						num4 = TwoPhaseFlow.ThroatToUpstreamPressureRatioForContraction(stream, num2, MassFlux, Pc_P);
					}
					if (num4 <= 0.0)
					{
						LogMessage(MessageLogType.Fatal, Resources.CantFindPThroat);
						base.IsChoked = true;
					}
					else
					{
						double num5 = num4 * stream.GetPhaseByID(0).Pressure.Value;
						IFlowStream flowStream = (IFlowStream)stream.Clone();
						pIPPP.ReflashStreamWithSameComposition(flowStream, FlashType.PSFlash, num5, stream.GetPhaseByID(1).Entropy.Value);
						double area = HTFSUtility.FlowArea(throatDiam);
						base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, flowStream, area);
						if (!base.IsChoked)
						{
							double num6;
							if (flowStream.NumberOfPhases < 2)
							{
								IPhase phaseByID = flowStream.GetPhaseByID(1);
								num6 = ((phaseByID.PhaseType != PhaseType.Vapour) ? (num5 + 0.5 * (num3 * num3 - MassFlux * MassFlux) / phaseByID.Density.Value) : SinglePhaseFlow.DownstreamPressureForCompressibleEnlargement(PipeDiameter, throatDiam, num3, phaseByID.Pressure.Value, phaseByID.Density.Value, phaseByID.Cp.Value / phaseByID.Cv.Value));
							}
							else
							{
								num6 = TwoPhaseFlow.DownstreamPressureForCompressibleEnlargement(flowStream, pIPPP, num3, num2);
							}
							num = stream.GetPhaseByID(0).Pressure.Value - num5 + (num5 - num6);
						}
					}
				}
			}
		}
		if (base.IsChoked || (flag && Math.Abs(num) > stream.GetPhaseByID(0).Pressure.Value * (1.0 - Pc_P)) || (!flag && Math.Abs(num) > stream.GetPhaseByID(0).Pressure.Value * 0.6))
		{
			base.IsChoked = true;
			chokedLocation = ChokedLocationEnum.Throat;
			num = RealValue.FEMPTY;
		}
		if (!base.IsChoked)
		{
			calculateDH(stream, MassFlux, num);
			DPa = CalculateDPa(stream, MassFlux, num);
			DPf = num * StepwiseFactor * FlowDirectionFactor - DPa;
		}
		return num;
	}

	protected override double CalculateDPa(IFlowStream streamIn, double MIn, double dp)
	{
		double data = streamIn.GetPhaseByID(0).Pressure.Value - dp;
		double data2 = streamIn.GetPhaseByID(0).Enthalpy.Value + DH;
		double num = MIn * Math.Pow(Ports[0].Diameter.Value / Ports[1].Diameter.Value, 2.0);
		IFlowStream stream = (IFlowStream)streamIn.Clone();
		double num2 = ((methodType != MethodType.HTFSMethod) ? Crane.CalculateHomogeneousEquivalentDensity(streamIn) : HTFSUtility.CalculateEquivalentDensity(streamIn, MIn, 0.0));
		pIPPP.ReflashStreamWithSameComposition(stream, FlashType.PHFlash, data, data2);
		double num3 = ((methodType != MethodType.HTFSMethod) ? Crane.CalculateHomogeneousEquivalentDensity(stream) : HTFSUtility.CalculateEquivalentDensity(stream, num, 0.0));
		DPa = 0.5 * (num * num / num3 - MIn * MIn / num2) * StepwiseFactor * FlowDirectionFactor;
		return DPa;
	}

	protected override bool CheckGeometry()
	{
		bool flag = true;
		if (throatDiam > Ports[0].Diameter.Value || throatDiam < 0.0)
		{
			LogStatus("ThrottleBadThroatDiam", 500);
			LogMessage(MessageLogType.Fatal, Resources.ThrottleBadThroatDiam1);
			throw new Exception(Resources.ThrottleBadThroatDiam2);
		}
		if (diameterRatio != throatDiam / Ports[0].Diameter.Value)
		{
			diameterRatio = throatDiam / Ports[0].Diameter.Value;
		}
		if (flag)
		{
			flag = base.CheckGeometry();
		}
		return flag;
	}

	public override double CalculatePressureDrop()
	{
		bool flag = true;
		double result = 0.0;
		if (calculationMode == CalculationMode.UnknownMode || calculationMode == CalculationMode.PressureDropMode)
		{
			flag = CheckingForDPCalculation();
		}
		if (flag)
		{
			if (FlowDirectionFactor < 0.0)
			{
				LogMessage(MessageLogType.Warning, Resources.NonSymmetricOrifice);
			}
			DPz = 0.0;
			DPf = 0.0;
			DPa = 0.0;
			IFlowStream flowStream = (IFlowStream)Ports[StartPortNo].Stream.Clone();
			double area = HTFSUtility.FlowArea(throatDiam);
			if (checkChoking)
			{
				base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, flowStream, area);
			}
			if (!base.IsChoked)
			{
				double area2 = HTFSUtility.FlowArea(Ports[0].Diameter.Value);
				double massFlux = HTFSUtility.CalculateMassFlux(flowStream.MassFlowrate.Value, area2);
				result = ((StartPortNo >= 1) ? calculateDP(flowStream, massFlux) : calculateDP(flowStream, massFlux));
			}
			if (calculationMode == CalculationMode.PressureDropMode)
			{
				if (base.IsChoked)
				{
					chokedLocation = ChokedLocationEnum.Throat;
					if (StepwiseFactor > 0.0)
					{
						PipeFlowFunctions.getChokingUpstreamCondition(pIPPP, Ports[1].Stream, Ports[0].Stream, Ports[0].Diameter.Value, getMinimumDiameter(), (int)methodType, requiredAccuracy);
						LogMessage(MessageLogType.Fatal, Resources.FlowChokedUpstreamEstimated);
						if (task != TaskType.Design)
						{
							return -1.0;
						}
						throw new Exception(Resources.FatalErrorChoked);
					}
					LogStatus("ChokedFlowInletSpec", 500);
					throw new Exception(Resources.ChokedFlowInletPSpec);
				}
				updateOtherPort(CalculatedResult.PressureDrop, result);
			}
		}
		return result;
	}

	protected override double getBodyDiameter()
	{
		if (Ports[0].Diameter.IsUnknown)
		{
			if (Ports[1].Diameter.IsUnknown)
			{
				LogStatus("ThrottleUnknownDiameter", 500);
				LogMessage(MessageLogType.Fatal, Resources.InOrOutDiamNeeded);
				throw new Exception(Resources.InOrOutDiamNeededThrottle);
			}
			Ports[0].Diameter.Value = Ports[1].Diameter.Value;
		}
		return Ports[0].Diameter.Value;
	}

	protected override void updatePortDiameter(double D)
	{
		Ports[0].Diameter.Value = D;
		Ports[1].Diameter.Value = D;
		throatDiam = D * diameterRatio;
	}

	protected abstract double calculateKFactorWithHTFSMethod(IFlowStream stream, double MassFlux);

	protected abstract double calculateKFactorWithCraneMethod();

	protected abstract double calculateKFactorWithCrane_HTFSMethod(IFlowStream stream, double MassFlux);

	public double getKFactor(IFlowStream stream, double MassFlux)
	{
		switch (methodType)
		{
		case MethodType.CraneMethod:
			K = calculateKFactorWithCraneMethod();
			break;
		case MethodType.Crane_HTFSMethod:
			K = calculateKFactorWithCrane_HTFSMethod(stream, MassFlux);
			break;
		default:
			K = calculateKFactorWithHTFSMethod(stream, MassFlux);
			break;
		}
		return K;
	}

	protected double getDensity(IFlowStream stream)
	{
		if (stream.NumberOfPhases < 2)
		{
			return stream.GetPhaseByID(1).Density.Value;
		}
		return stream.CombinedLiquidPhase.Density.Value;
	}

	protected override double getMinimumDiameter()
	{
		try
		{
			return throatDiam;
		}
		catch (Exception ex)
		{
			throw ex;
		}
	}

	public override string ToString()
	{
		return Resources.FittingThrottleElement;
	}
}
[ComVisible(false)]
public interface IValve : IPipeFittings
{
	double PipeDiameter { get; set; }

	double ThroatDiameter { get; set; }

	double KFactor { get; set; }

	double InletAngle { get; set; }
}
[ComVisible(false)]
public abstract class Valve : PipeFitting, IValve, IPipeFittings
{
	protected double throatDiam;

	protected double equivThroatDiam;

	protected double inletAngle;

	protected double K;

	protected double dh;

	protected double diameterRatio;

	public double InletAngle
	{
		get
		{
			return inletAngle;
		}
		set
		{
			inletAngle = value;
			equivThroatDiam = calculateEquivalentThroatDiameter();
		}
	}

	public double PipeDiameter
	{
		get
		{
			return Ports[0].Diameter.Value;
		}
		set
		{
			Ports[0].Diameter.Value = value;
			Ports[1].Diameter.Value = value;
			if (throatDiam <= 0.0)
			{
				throatDiam = value;
			}
			if (value > 0.0)
			{
				diameterRatio = throatDiam / value;
			}
			if (diameterRatio == 1.0)
			{
				inletAngle = 0.0;
			}
			equivThroatDiam = calculateEquivalentThroatDiameter();
		}
	}

	public double ThroatDiameter
	{
		get
		{
			return throatDiam;
		}
		set
		{
			throatDiam = value;
			if (value > 0.0)
			{
				diameterRatio = value / Ports[0].Diameter.Value;
			}
			if (diameterRatio == 1.0)
			{
				inletAngle = 0.0;
			}
			equivThroatDiam = calculateEquivalentThroatDiameter();
		}
	}

	public double KFactor
	{
		get
		{
			return K;
		}
		set
		{
			K = value;
		}
	}

	public Valve()
		: base(2)
	{
		InitialiseValve();
	}

	public Valve(double Diameter, IFlowStream StreamIn, IFlowStream StreamOut)
		: base(2)
	{
		Ports[0].Diameter.Value = Diameter;
		Ports[0].Stream = StreamIn;
		Ports[1].Diameter.Value = Diameter;
		Ports[1].Stream = StreamOut;
		InitialiseValve();
	}

	public Valve(double Diameter, double ThroatDiam, double InletAngle, IFlowStream StreamIn, IFlowStream StreamOut)
		: base(2)
	{
		Ports[0].Diameter.Value = Diameter;
		Ports[0].Stream = StreamIn;
		Ports[1].Diameter.Value = Diameter;
		Ports[1].Stream = StreamOut;
		InitialiseValve();
		throatDiam = ThroatDiam;
		inletAngle = InletAngle;
		diameterRatio = throatDiam / Diameter;
		equivThroatDiam = calculateEquivalentThroatDiameter();
	}

	private void InitialiseValve()
	{
		inletAngle = 0.0;
		K = 0.0;
		dh = 0.0;
		throatDiam = Ports[0].Diameter.Value;
		diameterRatio = 1.0;
		equivThroatDiam = throatDiam;
	}

	public double getKFactor(IFlowStream stream, double MassFlux)
	{
		switch (methodType)
		{
		case MethodType.CraneMethod:
			K = calculateKFactorWithCraneMethod();
			break;
		case MethodType.Crane_HTFSMethod:
			K = calculateKFactorWithCrane_HTFSMethod(stream, MassFlux);
			break;
		default:
			K = calculateKFactorWithHTFSMethod(stream, MassFlux);
			break;
		}
		return K;
	}

	private double getDensity(IFlowStream stream)
	{
		if (stream.NumberOfPhases < 2)
		{
			return stream.GetPhaseByID(1).Density.Value;
		}
		return methodType switch
		{
			MethodType.CraneMethod => Crane.CalculateHomogeneousEquivalentDensity(stream), 
			_ => stream.CombinedLiquidPhase.Density.Value, 
		};
	}

	protected override double calculateDP(IFlowStream stream, double MassFlux)
	{
		double result = 0.0;
		PipeFlowPressureSolver pipeFlowPressureSolver = CreateIncompressibleSolver(methodType, stream);
		if (pipeFlowPressureSolver != null)
		{
			IFlowStream stream2 = Ports[EndPortNo].Stream;
			if (pipeFlowPressureSolver.CalculateUnknownStream(stream2, stream, 1.0, StepwiseFactor, FlowDirectionFactor))
			{
				result = stream2.GetPhaseByID(0).Pressure.Value - stream.GetPhaseByID(0).Pressure.Value;
				DH = stream2.GetPhaseByID(0).Enthalpy.Value - stream.GetPhaseByID(0).Enthalpy.Value;
			}
		}
		if (!base.IsChoked)
		{
			return result;
		}
		return RealValue.FEMPTY;
	}

	protected override PipeFlowPressureSolver CreateIncompressibleSolver(MethodType lMethod, IFlowStream stream)
	{
		PipeFlowPressureSolver pipeFlowPressureSolver = new PipeFlowPressureSolverLengthMethod();
		pipeFlowPressureSolver.HeatFlux = 0.0;
		pipeFlowPressureSolver.PhysicalPropertyPackage = pIPPP;
		pipeFlowPressureSolver.Accuracy = base.RequiredAccuracy;
		pipeFlowPressureSolver.SetGeometricalParameters(PipeDiameter, PipeDiameter, 0.0);
		if (lMethod == MethodType.HTFSMethod)
		{
			pipeFlowPressureSolver.MomentumEnergyDelegate = HTFSUtility.MomentumEnergy;
		}
		pipeFlowPressureSolver.IncludeMomentumEnergy = base.IncludeKineticEnergy;
		if (StepwiseFactor == 1.0)
		{
			pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseUnknownStream;
		}
		else
		{
			pipeFlowPressureSolver.GradientEstimate = GradientEstimateMethod.UseKnownStream;
		}
		pipeFlowPressureSolver.Correlation = CreateCorrelation(lMethod, stream);
		if (base.LogMessages is MessageLogger)
		{
			pipeFlowPressureSolver.Log = (MessageLogger)base.LogMessages;
		}
		return pipeFlowPressureSolver;
	}

	private bool checkChokingForValve(IFlowStream stream, double massFlux, double dp)
	{
		bool flag = false;
		try
		{
			double num = calculateEquivalentThroatDiameter();
			double area = HTFSUtility.FlowArea(num);
			if (checkChoking)
			{
				flag = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, stream, area);
			}
			if (!flag)
			{
				double num2 = num / Ports[0].Diameter.Value;
				if (num2 > 0.99)
				{
					num2 = 0.99;
				}
				double num3 = massFlux / (num2 * num2);
				HTFSUtility.CalculateCriticalPressureRatioAndMassFlux(stream, pIPPP, num2, out var Pc_P, out var Mc);
				if (Mc == RealValue.FEMPTY)
				{
					Mc = 1E+20;
				}
				if (num3 > 0.99 * Mc)
				{
					flag = true;
				}
				else
				{
					double num4 = 1.0 - dp / stream.GetPhaseByID(0).Pressure.Value;
					if (Pc_P != RealValue.FEMPTY && num4 < Pc_P)
					{
						flag = true;
					}
				}
			}
			return flag;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "checkChokingForValve"), innerException);
		}
	}

	protected abstract double calculateKFactorWithHTFSMethod(IFlowStream stream0, double MassFlux);

	protected abstract double calculateKFactorWithCraneMethod();

	protected double calculateKFactorWithCrane_HTFSMethod(IFlowStream stream, double MassFlux)
	{
		double num = calculateKFactorWithCraneMethod();
		if (stream.NumberOfPhases < 2)
		{
			return num;
		}
		IPhase combinedLiquidPhase = stream.CombinedLiquidPhase;
		IPhase phaseByType = stream.GetPhaseByType(PhaseType.Vapour);
		double num2;
		if (combinedLiquidPhase.Viscosity.Value / phaseByType.Viscosity.Value < 1000.0)
		{
			num2 = TwoPhaseFlow.MultiplierOfFriedel(MassFlux, Ports[0].Diameter.Value, combinedLiquidPhase, phaseByType);
		}
		else if (MassFlux > 100.0)
		{
			double y = Math.Sqrt(combinedLiquidPhase.Density.Value / phaseByType.Density.Value);
			num2 = TwoPhaseFlow.MultiplierOfChisholm(y, MassFlux, phaseByType.MassFraction.Value, 0.0);
		}
		else
		{
			double constantC = getConstantC(combinedLiquidPhase.Density.Value, phaseByType.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType.Viscosity.Value, phaseByType.MassFraction.Value, MassFlux);
			double xtt = TwoPhaseFlow.XLMParameter(Ports[0].Diameter.Value, MassFlux, 0.0, phaseByType.MassFraction.Value, combinedLiquidPhase.Density.Value, phaseByType.Density.Value, combinedLiquidPhase.Viscosity.Value, phaseByType.Viscosity.Value);
			num2 = TwoPhaseFlow.MultiplierOfMartinelli(constantC, xtt) * HTFSUtility.Square(combinedLiquidPhase.MassFraction.Value);
		}
		return num * num2;
	}

	public abstract double calculateEquivalentThroatDiameter();

	protected virtual double getConstantC(double DensityL, double DensityG, double VisL, double VisG, double Quality, double MassFlux)
	{
		return TwoPhaseFlow.CoefficeCByFittingsName("SuddenContraction", DensityL, DensityG, VisL, VisG, Quality, MassFlux);
	}

	protected override double getMinimumDiameter()
	{
		double num = calculateEquivalentThroatDiameter();
		double d = SinglePhaseFlow.ContractionCoeff(Ports[0].Diameter.Value, num);
		return num * Math.Sqrt(d);
	}

	protected override bool CheckGeometry()
	{
		bool flag = true;
		if ((calculationMode == CalculationMode.SizingFactorMode || calculationMode == CalculationMode.SizingMode) && sizingWithDefault)
		{
			if (throatDiam != Ports[0].Diameter.Value)
			{
				throatDiam = Ports[0].Diameter.Value;
				LogMessage(MessageLogType.Note, Resources.ValveThroatSetToUpstream);
			}
			if (inletAngle != 0.0)
			{
				inletAngle = 0.0;
				LogMessage(MessageLogType.Note, Resources.ValveInletAngleSetToZero);
			}
		}
		else
		{
			if (throatDiam < 0.0 || throatDiam > Math.Min(Ports[0].Diameter.Value, Ports[1].Diameter.Value))
			{
				LogStatus("ValveBadThroatDiamRange", 500);
				LogMessage(MessageLogType.Fatal, Resources.ValveBadThroatDiam1);
				throw new Exception(Resources.ValveBadThroatDiam2);
			}
			if (inletAngle < 0.0 || inletAngle > 180.0)
			{
				LogStatus("ValveInletAngle", 500);
				LogMessage(MessageLogType.Fatal, Resources.ValveBadInletAngle1);
				throw new Exception(Resources.ValveBadInletAngle2);
			}
		}
		if (flag)
		{
			flag = base.CheckGeometry();
		}
		return flag;
	}

	public override double CalculatePressureDrop()
	{
		bool flag = true;
		double num = 0.0;
		if (calculationMode == CalculationMode.UnknownMode || calculationMode == CalculationMode.PressureDropMode)
		{
			flag = CheckingForDPCalculation();
		}
		if (flag)
		{
			DPz = 0.0;
			DPf = 0.0;
			DPa = 0.0;
			IFlowStream flowStream = (IFlowStream)Ports[StartPortNo].Stream.Clone();
			double num2 = calculateEquivalentThroatDiameter();
			double num3 = SinglePhaseFlow.ContractionCoeff(Ports[0].Diameter.Value, num2);
			double num4 = HTFSUtility.FlowArea(num2);
			if (checkChoking && StepwiseFactor < 0.0)
			{
				base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, flowStream, num4 * num3);
			}
			if (!base.IsChoked)
			{
				if (methodType == MethodType.StaticPressureMethod)
				{
					num = 0.0;
				}
				else
				{
					double area = HTFSUtility.FlowArea(Ports[0].Diameter.Value);
					double massFlux = HTFSUtility.CalculateMassFlux(flowStream.MassFlowrate.Value, area);
					num = calculateDP(flowStream, massFlux);
				}
			}
			if (calculationMode == CalculationMode.PressureDropMode)
			{
				if (checkChoking && StepwiseFactor > 0.0)
				{
					IFlowStream stream = (IFlowStream)Ports[EndPortNo].Stream.Clone();
					base.IsChoked = PipeFlowFunctions.isExcessSpeedOfSound(pIPPP, stream, num4 * num3);
				}
				if (base.IsChoked)
				{
					chokedLocation = ChokedLocationEnum.Throat;
					if (StepwiseFactor > 0.0)
					{
						PipeFlowFunctions.getChokingCondition(pIPPP, Ports[1].Stream, Ports[0].Stream, Ports[0].Diameter.Value, getMinimumDiameter(), (int)methodType, requiredAccuracy, StepwiseFactor);
						LogMessage(MessageLogType.Fatal, Resources.FlowChokedUpstreamEstimated);
						if (task != TaskType.Design || StepwiseFactor > 0.0)
						{
							return -1.0;
						}
						throw new Exception(Resources.FatalErrorChoked);
					}
					LogStatus("ChokedFlowInletSpec", 500);
					throw new Exception(Resources.ChokedFlowInletPSpec);
				}
				updateOtherPort(CalculatedResult.PressureDrop, num);
			}
			if (Math.Abs(num) < 1E-06)
			{
				num = 0.0;
			}
			return num;
		}
		return RealValue.FEMPTY;
	}

	public override string ToString()
	{
		return Resources.FittingValve;
	}
}
