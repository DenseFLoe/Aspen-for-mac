		private void EditOver(object sender, KeyPressEventArgs e)
		{
			if (e.KeyChar == '\r')
			{
				m_ValueEditor.Hide();
				if (UpdateSolver(m_EditedItem.SubItems[0].Text, m_ValueEditor.Text) == AMOSretcode.AMSuccess)
				{
					m_EditedItem.SubItems[1].Text = m_ValueEditor.Text;
					StatusMessage(Color.Green, "Parameter " + m_EditedItem.SubItems[0].Text + " successfully changed to " + m_ValueEditor.Text);
				}
			}
			if (e.KeyChar == '\u001b')
			{
				m_ValueEditor.Hide();
				StatusMessage(Color.Black, "Cancelled");
			}
		}

		private AMOSretcode UpdateSolver(string paramName, string paramValue)
		{
			AMOSretcode aMOSretcode = AMOSretcode.AMError;
			ArrayList arrayList = new ArrayList();
			m_table.GetParameterInfo(paramName, arrayList);
			try
			{
				if (arrayList[1] is int)
				{
					int num = Convert.ToInt32(paramValue);
					aMOSretcode = m_table.SetParameter(paramName, num);
				}
				else if (arrayList[1] is double)
				{
					double num2 = Convert.ToDouble(paramValue);
					aMOSretcode = m_table.SetParameter(paramName, num2);
				}
				else if (arrayList[1] is string)
				{
					aMOSretcode = m_table.SetParameter(paramName, paramValue);
				}
				switch (aMOSretcode)
				{
				case AMOSretcode.AMSuccess:
					statuslabel.Text = "Param val successfully changed :-)";
					break;
				case AMOSretcode.AMParamOutOfBounds:
					statuslabel.ForeColor = Color.Red;
					if (arrayList[1] is int)
					{
						statuslabel.Text = $"Int param out of range [{arrayList[2]}-{arrayList[3]}]";
					}
					else if (arrayList[2] is double)
					{
						statuslabel.Text = $"Dbl param out of range [{arrayList[2]}-{arrayList[3]}]";
					}
					else
					{
						statuslabel.Text = $"String arg invalid; valid values: {arrayList[2]}";
					}
					break;
				default:
					statuslabel.Text = "Unexpected failure setting param - AOS didn't like it :-(";
					break;
				}
			}
			catch (FormatException)
			{
				StatusMessage(Color.Red, "Data entered in incorrect format - ignored");
			}
			return aMOSretcode;
		}

	public sealed class StopWatch
	{
		private long m_StartTime;

		private long m_Frequency;

		private long StartTime
		{
			get
			{
				return m_StartTime;
			}
			set
			{
				m_StartTime = value;
			}
		}

		private long Frequency
		{
			get
			{
				return m_Frequency;
			}
			set
			{
				m_Frequency = value;
			}
		}

		[DllImport("kernel32.dll")]
		private static extern int QueryPerformanceCounter(ref long x);

		[DllImport("kernel32.dll")]
		private static extern int QueryPerformanceFrequency(ref long x);

		public StopWatch()
		{
			Frequency = GetFrequency();
			Reset();
		}

		public void Reset()
		{
			StartTime = GetValue();
		}

		public long Peek()
		{
			return (long)((double)(GetValue() - StartTime) / (double)Frequency * 10000.0);
		}

		public double cpuSecs()
		{
			return (double)Peek() / 10000.0;
		}

		private long GetValue()
		{
			long x = 0L;
			if (QueryPerformanceCounter(ref x) == 0)
			{
				throw new NotSupportedException("Error while querying the high-resolution performance counter.");
			}
			return x;
		}

		private long GetFrequency()
		{
			long x = 0L;
			if (QueryPerformanceFrequency(ref x) == 0)
			{
				throw new NotSupportedException("Error while querying the performance counter frequency.");
			}
			return x;
		}
	}
	public class CRandMT
	{
		private static int MT_N = 624;

		private static int MT_M = 397;

		private static uint MAT_A = 2567483615u;

		private static uint UMASK = 2147483648u;

		private static uint LMASK = 2147483647u;

		private static double dFac = 2.3283064365386963E-10;

		private uint[] state;

		private uint[] mag01;

		private uint counter;

		private uint seed;

		private double dInHand;

		private bool bInHand;

		private double dInHand64;

		private bool bInHand64;

		public uint Seed
		{
			get
			{
				return seed;
			}
			set
			{
				seed = value;
				reSeed();
			}
		}

		public CRandMT(uint seed)
		{
			state = new uint[MT_N];
			mag01 = new uint[2];
			if (seed == 0)
			{
				this.seed = (uint)DateTime.Now.Ticks;
			}
			else
			{
				this.seed = seed;
			}
			reSeed();
		}

		public uint lRand()
		{
			uint num;
			if (counter >= MT_N)
			{
				int i;
				for (i = 0; i < MT_N - MT_M; i++)
				{
					num = (state[i] & UMASK) | (state[i + 1] & LMASK);
					state[i] = state[i + MT_M] ^ (num >> 1) ^ mag01[num & 1];
				}
				for (; i < MT_N - 1; i++)
				{
					num = (state[i] & UMASK) | (state[i + 1] & LMASK);
					state[i] = state[i + (MT_M - MT_N)] ^ (num >> 1) ^ mag01[num & 1];
				}
				num = (state[MT_N - 1] & UMASK) | (state[0] & LMASK);
				state[MT_N - 1] = state[MT_M - 1] ^ (num >> 1) ^ mag01[num & 1];
				counter = 0u;
			}
			num = state[counter++];
			num ^= num >> 11;
			num ^= (num << 7) & 0x9D2C5680u;
			num ^= (num << 15) & 0xEFC60000u;
			return num ^ (num >> 18);
		}

		public int iRand(int lo, int hi)
		{
			if (lo == 0 && hi == 0)
			{
				return (int)lRand();
			}
			return (int)Math.Floor((double)(hi - lo + 1) * dRand64()) + lo;
		}

		public double dRand()
		{
			return dFac * (double)lRand();
		}

		public double dRand64()
		{
			uint num = lRand() >> 5;
			uint num2 = lRand() >> 6;
			return ((double)num * 67108864.0 + (double)num2) * 1.1102230246251565E-16;
		}

		public double nRand()
		{
			if (bInHand)
			{
				bInHand = false;
				return dInHand;
			}
			double num;
			double num2;
			double num3;
			do
			{
				num = 2.0 * dFac * (double)lRand() - 1.0;
				num2 = 2.0 * dFac * (double)lRand() - 1.0;
				num3 = num * num + num2 * num2;
			}
			while (num3 >= 1.0 || num3 == 0.0);
			double num4 = -2.0 * Math.Log(num3) / num3;
			dInHand = num * num4;
			bInHand = true;
			return num2 * num4;
		}

		public double nRand64()
		{
			if (bInHand64)
			{
				bInHand64 = false;
				return dInHand64;
			}
			double num;
			double num2;
			double num3;
			do
			{
				num = 2.0 * dRand64() - 1.0;
				num2 = 2.0 * dRand64() - 1.0;
				num3 = num * num + num2 * num2;
			}
			while (num3 >= 1.0 || num3 == 0.0);
			double num4 = Math.Sqrt(-2.0 * Math.Log(num3) / num3);
			dInHand64 = num * num4;
			bInHand64 = true;
			return num2 * num4;
		}

		public double eRand()
		{
			uint num;
			do
			{
				num = lRand();
			}
			while (num == 0);
			return 0.0 - Math.Log(dFac * (double)num);
		}

		private void reSeed()
		{
			state[0] = seed & 0xFFFFFFFFu;
			for (counter = 1u; counter < MT_N; counter++)
			{
				state[counter] = 1812433253 * (state[counter - 1] ^ (state[counter - 1] >> 30)) + counter;
			}
			mag01[0] = 0u;
			mag01[1] = MAT_A;
			bInHand = false;
			bInHand64 = false;
		}
	}

		public static double TestLinearSolverSolution(int[] irn, int[] jcn, double[] vals, double[] x, double[] rhs, out double rms)
		{
			for (int i = 0; i < irn.Length; i++)
			{
				rhs[irn[i] - 1] -= vals[i] * x[jcn[i] - 1];
			}
			double num = 0.0;
			rms = 0.0;
			for (int j = 0; j < rhs.Length; j++)
			{
				double num2 = Math.Abs(rhs[j]);
				rms += num2 * num2;
				if (num2 > num)
				{
					num = num2;
				}
			}
			rms /= rhs.Length;
			rms = Math.Sqrt(rms);
			return num;
		}

	[ComVisible(false)]
	public interface IManagedObjStore
	{
		int Count { get; }

		int AddObject(object obj);

		object RetrieveObject(int at_index);

		void DeleteObject(int at_index);
	}
	internal class UManagedObjStore : IManagedObjStore
	{
		private ArrayList mList;

		private int mObjCount;

		private static UManagedObjStore gInstance;

		public int Count => mObjCount;

		private UManagedObjStore()
		{
			mList = new ArrayList();
			mObjCount = 0;
		}

		public static IManagedObjStore GetInstance()
		{
			if (gInstance == null)
			{
				gInstance = new UManagedObjStore();
			}
			return gInstance;
		}

		public int AddObject(object obj)
		{
			int result = mList.Add(obj);
			mObjCount++;
			return result;
		}

		public object RetrieveObject(int at_index)
		{
			return mList[at_index];
		}

		public void DeleteObject(int at_index)
		{
			mList[at_index] = 0;
			mObjCount--;
			if (mObjCount == 0)
			{
				mList.Clear();
			}
		}
	}
	internal class SManagedObjStore : IManagedObjStore
	{
		private ArrayList mList;

		private int mObjCount;

		private static SManagedObjStore gInstance;

		public int Count => mObjCount;

		private SManagedObjStore()
		{
			lock (this)
			{
				mList = new ArrayList();
				mObjCount = 0;
			}
		}

		public static IManagedObjStore GetInstance()
		{
			if (gInstance == null)
			{
				gInstance = new SManagedObjStore();
			}
			return gInstance;
		}

		public int AddObject(object obj)
		{
			lock (this)
			{
				int result = mList.Add(obj);
				mObjCount++;
				return result;
			}
		}

		public object RetrieveObject(int at_index)
		{
			return mList[at_index];
		}

		public void DeleteObject(int at_index)
		{
			lock (this)
			{
				mList[at_index] = 0;
				mObjCount--;
				if (mObjCount == 0)
				{
					mList.Clear();
				}
			}
		}
	}
