[ComVisible(false)]
internal readonly struct CEstimateVariable
{
	internal enum VarType
	{
		Unknown = -1,
		MassFlow,
		Pressure,
		Temperature,
		Enthalpy,
		MolWt,
		MassFrac
	}

	public string Description { get; }

	public CNetworkPort Port { get; }

	public VarType Type { get; }

	public int ComponentIndex { get; }

	public IRealValue Variable { get; }

	public double Value
	{
		get
		{
			return Variable.Value;
		}
		set
		{
			Variable.Value = value;
		}
	}

	internal CEstimateVariable(CNetworkPort Port, VarType Type, string Description, IRealValue Variable, int ComponentIndex = 0)
	{
		if (Port == null)
		{
			throw new ArgumentNullException("Port");
		}
		if (Variable == null)
		{
			throw new ArgumentNullException("Variable");
		}
		this.Description = Description;
		this.Port = Port;
		this.Type = Type;
		this.Variable = Variable;
		this.ComponentIndex = ComponentIndex;
	}

	public static bool operator ==(CEstimateVariable x, CEstimateVariable y)
	{
		if (x.Description != y.Description)
		{
			return false;
		}
		if (x.Port != y.Port)
		{
			return false;
		}
		if (x.Type != y.Type)
		{
			return false;
		}
		if (x.Value != y.Value)
		{
			return false;
		}
		return true;
	}

	public static bool operator !=(CEstimateVariable x, CEstimateVariable y)
	{
		return !(x == y);
	}

	public override bool Equals(object obj)
	{
		if (obj is CEstimateVariable cEstimateVariable)
		{
			return cEstimateVariable == this;
		}
		return false;
	}

	public override int GetHashCode()
	{
		return Description.GetHashCode() ^ Port.GetHashCode() ^ Variable.GetHashCode();
	}
}

	internal void CreateVariables()
	{
		int num = -32767;
		int num2 = -32767;
		IPhase phase = null;
		IFlowStream flowStream = null;
		try
		{
			flowStream = Port.FlowStream;
			phase = flowStream.GetPhaseByID(0);
			num2 = phase.NumberOfCompounds;
			mVariables = new CEstimateVariable[5 + num2];
			mVariables[0] = new CEstimateVariable(this, CEstimateVariable.VarType.MassFlow, "Mass Flow", flowStream.MassFlowrate);
			mVariables[1] = new CEstimateVariable(this, CEstimateVariable.VarType.Pressure, "Pressure", phase.Pressure);
			mVariables[2] = new CEstimateVariable(this, CEstimateVariable.VarType.Pressure, "Temperature", phase.Temperature);
			mVariables[3] = new CEstimateVariable(this, CEstimateVariable.VarType.Enthalpy, "Enthalpy", phase.Enthalpy);
			mVariables[4] = new CEstimateVariable(this, CEstimateVariable.VarType.MolWt, "Mol. Wt.", phase.MolecularWeight);
			for (num = 0; num < num2; num++)
			{
				mVariables[5 + num] = new CEstimateVariable(this, CEstimateVariable.VarType.MassFrac, "Cmp Mass Frac.", phase.ComponentMassFractions[num], num);
			}
		}
		catch (Exception innerException)
		{
			throw new Exception(Resources.FailedToCreateVars, innerException);
		}
	}

	internal double NewPressureFromHistory(double Proposed)
	{
		CNetworkPortError cNetworkPortError = null;
		double num = -32767.0;
		try
		{
			if (IsChokedBoundary)
			{
				return PressureSpec;
			}
			if (mErrorHistory.Count < 3)
			{
				num = Proposed;
			}
			else
			{
				cNetworkPortError = (CNetworkPortError)mErrorHistory[mErrorHistory.Count - 2];
				num = cNetworkPortError.Pressure + mSolver.InnerDamper * (Proposed - cNetworkPortError.Pressure);
			}
			return num;
		}
		catch (Exception ex)
		{
			throw ex;
		}
	}

									cNetworkPort.Port.FlowStream.MassFlowrate.Erase();

					cLinearEquationCoefficient = (CLinearEquationCoefficient)cLinearEquation.Coeffs[num2];
					if (cLinearEquationCoefficient.Value != 0.0)
					{
						num6++;
					}
				}
			}
			array = new int[num6];
			array2 = new int[num6];
			array6 = new double[num6];
			num6 = 0;
			num3 = mEquations.Count;
			for (num = 0; num < num3; num++)
			{
				cLinearEquation = (CLinearEquation)mEquations[num];
				num4 = cLinearEquation.Coeffs.Count;
				for (num2 = 0; num2 < num4; num2++)
				{
					cLinearEquationCoefficient = (CLinearEquationCoefficient)cLinearEquation.Coeffs[num2];
					if (cLinearEquationCoefficient.Value != 0.0)
					{
						array[num6] = num + 1;
						array2[num6] = cLinearEquationCoefficient.VarIndex + 1;
						array6[num6] = cLinearEquationCoefficient.Value;
						array5[cLinearEquationCoefficient.VarIndex] = true;
						num6++;
					}
				}
			}
			num7 = mEquations.Count;
			num5 = mVariables.GetUpperBound(0) + 1;
			if (ma48.SetPattern(num7, num5, num6, array, array2) != AMOSretcode.AMSuccess)
			{
				throw new Exception(string.Format(Resources.MethodFailed, "SetPattern"));
			}
			if (ma48.SetMatrixVals(array6) != AMOSretcode.AMSuccess)
			{
				throw new Exception(string.Format(Resources.MethodFailed, "SetMatrixVals"));
			}
			array3 = new double[num7];
			num3 = num7;
			for (num = 0; num < num3; num++)
			{
				cLinearEquation = (CLinearEquation)mEquations[num];
				array3[num] = cLinearEquation.RHS;
			}
			if (ma48.SetRHS(array3) != AMOSretcode.AMSuccess)
			{
				throw new Exception(string.Format(Resources.MethodFailed, "SetRHS"));
			}
			array4 = new double[num5];
			if (ma48.GetSolution(array4) != AMOSretcode.AMSuccess)
			{
				throw new Exception(string.Format(Resources.MethodFailed, "GetSolution"));
			}
			num3 = mVariables.GetUpperBound(0) + 1;
			for (num = 0; num < num3; num++)
			{
				CEstimateVariable cEstimateVariable = mVariables[num];
				if (!array5[num])
				{
					continue;
				}
				if (!mInverse)
				{
					cEstimateVariable.Value = array4[num];
					continue;
				}
				if (array4[num] == 0.0)
				{
					throw new Exception(Resources.BadSolution);
				}
				cEstimateVariable.Value = 1.0 / array4[num];
			}
		}
		catch (Exception innerException)
		{
			throw new Exception("CLinearEquationSet Solve Failure", innerException);
		}
	}

	internal void Solve_Sparse()
	{
		IAMOSNonLinearEquationSolverFactory iAMOSNonLinearEquationSolverFactory = null;
		int iterations = -32767;
		IMessageLog messageLog = null;
		AMOSretcode aMOSretcode = AMOSretcode.AMError;
		try
		{
			if (mLog == null)
			{
				messageLog = new MessageLogger();
			}
			else
			{
				messageLog = mLog;
			}
			iAMOSNonLinearEquationSolverFactory = new NonLinearSolverFactory("aos_sparse.dll");
			myTable = iAMOSNonLinearEquationSolverFactory as IAMOSParameterTable;
			myTable.SetParameter("PrintLevel", 0);
			myTable.SetParameter("SeriouslyVerbose", 0);
			myTable.SetParameter("RelTol", mVarTolerance);
			myTable.SetParameter("ResTol", mResidTolerance);
			myTable.SetParameter("DogLeg", 1);
			myTable.SetParameter("CreepIter", mMaxIter);
			myTable.SetParameter("CreepSize", mDamper);
			myTable.SetParameter("MA48.ScaleMatrix", 1);
			switch (mSolver.NonlinearSolver)
			{
			case NonLinearSolverMethod.Newton:
				myTable.SetParameter("NewtonMethod", "Newton");
				break;
			case NonLinearSolverMethod.NewtonHybrid:
				myTable.SetParameter("NewtonMethod", "Hybrid");
				break;
			case NonLinearSolverMethod.FastNewton:
				myTable.SetParameter("NewtonMethod", "Fast Newton");
				break;
			}
			myTable.SetParameter("ConvTest", 3);
			mAMOSlog = new AMOSMessagesHandler();
			aMOSretcode = iAMOSNonLinearEquationSolverFactory.CreateSolver(this, mAMOSlog, out amosSolver);
			myExtTable = amosSolver as IAMOSExtendedParameterTable;
			mOuterIters = 0;
			mInnerStatus = SSNetworkSolver.InnerStatus.OK;
			bool flag = true;
			while (flag && mOuterIters <= mMaxIter)
			{
				aMOSretcode = amosSolver.SolveStep(AMOSSolverDirective.do_first_step, out iterations);
				if (mInnerStatus == SSNetworkSolver.InnerStatus.Recycle)
				{
					flag = true;
					mDamper *= 0.1;
					myTable.SetParameter("CreepIter", 1);
					myTable.SetParameter("CreepSize", mDamper);
					if (mDamper < 1E-05)
					{
						string solvedAtMinLocal = Resources.SolvedAtMinLocal;
						DiagnosticMessage(solvedAtMinLocal, SolverMessageTypeEnum.Note, "Loop", addTrace: true, addMsgLog: false);
						flag = false;
					}
				}
				else
				{
					flag = false;
				}
				while (aMOSretcode == AMOSretcode.AMSolving)
				{
					aMOSretcode = amosSolver.SolveStep(AMOSSolverDirective.do_solve_step, out iterations);
				}
			}
		}
		catch (Exception innerException)
		{
			throw new Exception("CNonLinearEquationSet Solve Failure", innerException);
		}
	}

	internal bool NextValue(double Old, double step, double mstep, double Mmax, double Mmin, ref double NewVal)
	{
		bool result = false;
		if (mstep != RealValue.FEMPTY)
		{
			if (Math.Abs(step) > mstep)
			{
				if (step > 0.0)
				{
					NewVal = Old + mstep;
				}
				else
				{
					NewVal = Old - mstep;
				}
				result = true;
			}
			else
			{
				NewVal = Old + step;
			}
		}
		else
		{
			NewVal = Old + step;
		}
		if (Mmax != RealValue.FEMPTY && NewVal > Mmax)
		{
			NewVal = Mmax;
			result = true;
		}
		if (Mmin != RealValue.FEMPTY && NewVal < Mmin)
		{
			NewVal = Mmin;
			result = true;
		}
