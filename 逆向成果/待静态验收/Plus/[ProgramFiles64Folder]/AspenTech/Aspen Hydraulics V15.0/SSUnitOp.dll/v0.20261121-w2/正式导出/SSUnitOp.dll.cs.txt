	private void Initialise(UnitOp UnitOp)
	{
		AspenTech.Hydraulics.Framework.Valve valve = null;
		try
		{
			if (UnitOp == null)
			{
				throw new ArgumentNullException();
			}
			valve = UnitOp as AspenTech.Hydraulics.Framework.Valve;
			FittingProperties properties = ValveFactory.GetProperties(mValve.TypeOfValve);
			FittingProperty commonByName = properties.GetCommonByName("PipeDiam");
			FittingProperty commonByName2 = properties.GetCommonByName("ThroatDiam");
			FittingProperty commonByName3 = properties.GetCommonByName("InletAngle");
			commonByName.Value = valve.PortCollection.Item(0).Diameter.Value;
			commonByName2.Value = valve.ThroatDiameter.Value;
			commonByName3.Value = 90.0;
			Properties specificProps = mValve.SpecificProps;
			for (int i = 0; i < specificProps.Count; i++)
			{

				if (specificItem != null)
				{
					if (specificProps[i].Type == AspenTech.Hydraulics.Data.PropertyType.Real)
					{
						specificItem.Value = (specificProps[i] as RealProperty).Value;
					}
					else if (specificProps[i].Type == AspenTech.Hydraulics.Data.PropertyType.Integer)
					{
						specificItem.Value = (specificProps[i] as IntegerProperty).Value;
					}
				}
			}
			StreamCollection streamCollection = new StreamCollection();
			streamCollection.Add(valve.PortCollection.Item(0).FlowStream);
			streamCollection.Add(valve.PortCollection.Item(1).FlowStream);
			mFitting = ValveFactory.CreateFitting(mValve.TypeOfValve, properties, streamCollection);
