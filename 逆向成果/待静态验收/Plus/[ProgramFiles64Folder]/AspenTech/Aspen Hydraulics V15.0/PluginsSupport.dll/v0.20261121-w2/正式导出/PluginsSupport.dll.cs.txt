[Guid("F447C52F-30E8-473F-BEAE-FA92E2E006CD")]
public class PluginManager : IPluginManager
{
	private StringCollection _locations;

	public StringCollection Locations
	{
		get
		{
			return _locations;
		}
		set
		{
			_locations = value;
		}
	}

	public PluginManager()
	{
		Locations = new StringCollection();
	}

	public IPlugins QueryInterface(string InterfaceName)
	{
		StringCollection stringCollection = new StringCollection();
		if (Locations.Count == 0)
		{
			Assembly assembly = Assembly.GetAssembly(GetType());
			Locations.Add(Path.GetDirectoryName(assembly.Location));
		}
		string[] array = new string[Locations.Count];
		Locations.CopyTo(array, 0);
		for (int i = 0; i < Locations.Count; i++)
		{
			if (Directory.Exists(array[i]))
			{
				stringCollection.AddRange(Directory.GetFiles(array[i], "*.dll"));
			}
		}
		string[] array2 = new string[stringCollection.Count];
		stringCollection.CopyTo(array2, 0);
		return new Plugins(array2, InterfaceName);
	}

	public IPlugins QueryInterface(string AssemblyName, string InterfaceName)
	{
		return new Plugins(AssemblyName, InterfaceName);
	}

	public void AddLocation(string location)
	{
		_locations.Add(location);
	}

	public void ClearLocations()
	{
		_locations.Clear();
	}
}

[Guid("103BC927-893D-4720-ADEE-3FA1FD07AAB3")]
public class Plugins : CollectionBase, IPlugins, IEnumerable
{
	[Guid("7C55586B-A25E-4DBE-B3E6-43E035BA1035")]
	public class PluginEnumerator : IEnumerator
	{
		private int _position;

		private readonly Plugins _pluginCollection;

		public Plugin Current => (Plugin)_pluginCollection.Item(_position);

		object IEnumerator.Current => _pluginCollection.Item(_position);

		public PluginEnumerator(Plugins PluginCollection)
		{
			_pluginCollection = PluginCollection;
			_position = -1;
		}

		public bool MoveNext()
		{
			if (_position < _pluginCollection.Count - 1)
			{
				_position++;
				return true;
			}
			return false;
		}

		public void Reset()
		{
			_position = -1;
		}
	}

	private static readonly List<string> _notAssembly = new List<string>();

	public int PlugInCount => base.List.Count;

	public Plugins(string[] AssemblyNames, string QueriedInterface)
	{
		foreach (string assemblyName in AssemblyNames)
		{
			ExtractPlugins(assemblyName, QueriedInterface);
		}
	}

	public Plugins(string AssemblyName, string QueriedInterface)
	{
		ExtractPlugins(AssemblyName, QueriedInterface);
	}

	public new PluginEnumerator GetEnumerator()
	{
		return new PluginEnumerator(this);
	}

	IEnumerator IEnumerable.GetEnumerator()
	{
		return new PluginEnumerator(this);
	}

	public void Add(IPlugin ThePlugin)
	{
		base.List.Add(ThePlugin);
	}

	public void Remove(int Index)
	{
		if (Index > base.List.Count - 1 || Index < 0)
		{
			throw new IndexOutOfRangeException();
		}
		base.List.RemoveAt(Index);
	}

	public static Plugins operator +(Plugins LHS, Plugins RHS)
	{
		for (int i = 0; i < RHS.Count; i++)
		{
			LHS.Add(RHS.Item(i));
		}
		return LHS;
	}

	public IPlugin Item(int Index)
	{
		Plugin plugin = null;
		if (Index > base.List.Count - 1 || Index < 0)
		{
			return null;
		}
		return (Plugin)base.List[Index];
	}

	private void ExtractPlugins(string AssemblyName, string QueriedInterface)
	{
		if (_notAssembly.Contains(AssemblyName))
		{
			return;
		}
		try
		{
			Assembly assembly;
			try
			{
				assembly = Assembly.LoadFrom(AssemblyName);
			}
			catch (BadImageFormatException)
			{
				_notAssembly.Add(AssemblyName);
				return;
			}
			Type[] types = assembly.GetTypes();
			Type[] array = types;
			foreach (Type type in array)
			{
				if (null != type.GetInterface(QueriedInterface) && !type.IsAbstract)
				{
					Plugin value = new Plugin(assembly.Location, type.FullName, QueriedInterface);
					base.List.Add(value);
				}
			}
		}
		catch (Exception ex2)
		{
			string text = ex2.ToString();
		}
	}
}
