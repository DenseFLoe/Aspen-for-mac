	{
		try
		{
			if (Re == 0.0)
			{
				return 0.0;
			}

			double x2 = 5.68E-09 * Re * Re * 0.25;

	public static FlowRegimeEnum FlowRegimeOfBeggsAndBrill(IFlowStream stream, double diam, out double volumeFractionL, out double Fr, out double vm)
	{
		FlowRegimeEnum result = FlowRegimeEnum.Liquid;
		double num = 0.0;
		double num2 = 0.0;
		IPhase phase = null;
		IPhase phase2 = null;
		Fr = 0.0;
		volumeFractionL = 0.0;
		vm = 0.0;
		try
		{
			if (stream.NumberOfPhases < 2)
			{
				result = ((stream.GetPhaseByID(1).PhaseType != PhaseType.Vapour) ? FlowRegimeEnum.Liquid : FlowRegimeEnum.Vapour);
			}
			else
			{
				phase = stream.CombinedLiquidPhase;
				phase2 = stream.GetPhaseByType(PhaseType.Vapour);
				num = stream.MassFlowrate.Value * (1.0 - phase2.MassFraction.Value) / phase.Density.Value;
				num2 = stream.MassFlowrate.Value * phase2.MassFraction.Value / phase2.Density.Value;
				volumeFractionL = num / (num + num2);
				vm = (num + num2) / (Math.PI * diam * diam * 0.25);
				Fr = vm * vm / (9.80665 * diam);
				LConstantsOfBeggsAndBrill(volumeFractionL, out var L, out var L2, out var L3, out var L4);
				if ((volumeFractionL < 0.01 && Fr < L) || (volumeFractionL >= 0.01 && Fr < L2))
				{
					result = FlowRegimeEnum.Segregated;
				}
				else if (volumeFractionL >= 0.01 && L2 <= Fr && Fr <= L3)
				{
					result = FlowRegimeEnum.Transition;
				}
				else if ((0.01 <= volumeFractionL && volumeFractionL < 0.4 && L3 < Fr && Fr <= L) || (volumeFractionL >= 0.4 && L3 < Fr && Fr <= L4))
				{
					result = FlowRegimeEnum.Intermittent;
				}
				else if ((volumeFractionL < 0.4 && Fr >= L) || (volumeFractionL >= 0.4 && Fr > L4))
				{
					result = FlowRegimeEnum.Distributed;
				}
			}
			return result;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "FlowRegimeOfBeggsAndBrill"), innerException);
		}
	}

	internal static void LConstantsOfBeggsAndBrill(double volumeFractionL, out double L1, out double L2, out double L3, out double L4)
	{
		L1 = 316.0 * Math.Pow(volumeFractionL, 0.302);
		L2 = 0.000925 * Math.Pow(volumeFractionL, -2.468);
		L3 = 0.1 * Math.Pow(volumeFractionL, -1.452);
		L4 = 0.5 * Math.Pow(volumeFractionL, -6.738);
	}

	internal static double CConstanOfBeggsAndBrill(double volumeFractionL, double Nlv, double Fr, FlowRegimeEnum pattern, double angle)
	{
		double num = 0.0;
		try
		{
			if (angle >= 0.0)
			{
				switch (pattern)
				{
				case FlowRegimeEnum.Stratified:
				case FlowRegimeEnum.Wavy:
				case FlowRegimeEnum.Annular:
				case FlowRegimeEnum.Segregated:
					num = (1.0 - volumeFractionL) * Math.Log(0.011 * Math.Pow(volumeFractionL, -3.768) * Math.Pow(Nlv, 3.539) * Math.Pow(Fr, -1.614));
					break;
				case FlowRegimeEnum.Plug:
				case FlowRegimeEnum.Slug:
				case FlowRegimeEnum.Intermittent:
					num = (1.0 - volumeFractionL) * Math.Log(2.96 * Math.Pow(volumeFractionL, 0.305) * Math.Pow(Nlv, -0.4473) * Math.Pow(Fr, 0.0978));
					break;
				default:
					num = 0.0;
					break;
				}
			}
			else
			{
				num = (1.0 - volumeFractionL) * Math.Log(4.7 * Math.Pow(volumeFractionL, -0.3692) * Math.Pow(Nlv, 0.1244) * Math.Pow(Fr, -0.5056));
			}
			if (num < 0.0)
			{
				num = 0.0;
			}
			return num;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "CConstantOfBeggsAndBrill"), innerException);
		}
	}

	internal static double HoldupOfBeggsAndBrillForHorizontalPipe(double volumeFractionL, double Fr, FlowRegimeEnum pattern)

			case FlowRegimeEnum.Wavy:
			case FlowRegimeEnum.Annular:

			case FlowRegimeEnum.Plug:
			case FlowRegimeEnum.Slug:
			case FlowRegimeEnum.Intermittent:
				return 0.845 * Math.Pow(volumeFractionL, 0.5351) / Math.Pow(Fr, 0.0173);
			default:

	public static double VoidFractionOfBeggsAndBrill(IFlowStream stream, double diam, double angle)
	{
		IPhase phase = null;
		IPhase phase2 = null;
		double num = 0.0;
		double num2 = 0.0;
		double num3 = 0.0;
		double num4 = 0.0;
		double num5 = 0.0;
		try
		{
			if (stream.NumberOfPhases < 2)
			{
				if (stream.GetPhaseByID(1).PhaseType == PhaseType.Vapour)
				{
					return 1.0;
				}
				return 0.0;
			}
			phase = stream.CombinedLiquidPhase;
			phase2 = stream.GetPhaseByType(PhaseType.Vapour);
			num5 = stream.MassFlowrate.Value * (1.0 - phase2.MassFraction.Value) / phase.Density.Value / (Math.PI / 4.0 * diam * diam);
			num4 = num5 * Math.Pow(phase.Density.Value / (9.80665 * phase.SurfaceTension.Value), 0.25);
			double volumeFractionL;
			double Fr;
			double vm;
			FlowRegimeEnum flowRegimeEnum = FlowRegimeOfBeggsAndBrill(stream, diam, out volumeFractionL, out Fr, out vm);
			LConstantsOfBeggsAndBrill(volumeFractionL, out var _, out var L2, out var L3, out var _);
			double num6 = (L3 - Fr) / (L3 - L2);
			if (flowRegimeEnum == FlowRegimeEnum.Transition)
			{
				double num7 = CConstanOfBeggsAndBrill(volumeFractionL, num4, Fr, FlowRegimeEnum.Segregated, angle);
				double num8 = HoldupOfBeggsAndBrillForHorizontalPipe(volumeFractionL, Fr, FlowRegimeEnum.Segregated);
				double num9 = CConstanOfBeggsAndBrill(volumeFractionL, num4, Fr, FlowRegimeEnum.Intermittent, angle);
				double num10 = HoldupOfBeggsAndBrillForHorizontalPipe(volumeFractionL, Fr, FlowRegimeEnum.Intermittent);
				num3 = num6 * num8 * (1.0 + num7 * (Math.Sin(HTFSUtility.DegreesToRads(1.8 * angle)) - 0.333 * Math.Pow(Math.Sin(HTFSUtility.DegreesToRads(1.8 * angle)), 3.0))) + (1.0 - num6) * num10 * (1.0 + num9 * (Math.Sin(HTFSUtility.DegreesToRads(1.8 * angle)) - 0.333 * Math.Pow(Math.Sin(HTFSUtility.DegreesToRads(1.8 * angle)), 3.0)));
			}
			else
			{
				num2 = HoldupOfBeggsAndBrillForHorizontalPipe(volumeFractionL, Fr, flowRegimeEnum);
				double num11 = CConstanOfBeggsAndBrill(volumeFractionL, num4, Fr, flowRegimeEnum, angle);
				num3 = num2 * (1.0 + num11 * (Math.Sin(HTFSUtility.DegreesToRads(1.8 * angle)) - 0.333 * Math.Pow(Math.Sin(HTFSUtility.DegreesToRads(1.8 * angle)), 3.0)));
			}
			if (angle > 0.0)
			{
				num3 *= 0.924;
				if (num3 < volumeFractionL)
				{
					num3 = volumeFractionL;
				}
			}
			else if (angle < 0.0)
			{
				num3 *= 0.685;
			}
			return 1.0 - num3;
		}
		catch (Exception innerException)
		{
			throw new Exception(string.Format(Resources.MethodFailed, "VoidFractionOfBeggsAndBrill"), innerException);
		}
	}

	public static double PressreGradientOfBeggsAndBrill(IFlowStream stream, double diam, double angle, double roughness)

		IPhase phase2 = null;
		double num2 = 0.0;

		double num7 = 0.0;
		double num8 = 0.0;
		double num9 = 0.0;
		double num10 = 0.0;
		double num11 = 0.0;
		double num12 = 0.0;
		double num13 = 0.0;
		try
		{
			double volumeFractionL;
			double Fr;
			double vm;
			FlowRegimeEnum flowRegimeEnum = FlowRegimeOfBeggsAndBrill(stream, diam, out volumeFractionL, out Fr, out vm);
			num = 1.0 - VoidFractionOfBeggsAndBrill(stream, diam, angle);
			num4 = volumeFractionL / (num * num);
			if (num4 == 1.0)
			{
				num9 = 0.0;
			}
			else if (num4 > 1.0 && num4 < 1.2)
			{
				num9 = Math.Log(2.2 * num4 - 1.2);
			}
			else
			{
				double num14 = Math.Log(num4);
				num9 = num14 / (-0.0523 + 3.182 * num14 - 0.8725 * num14 * num14 + 0.01853 * Math.Pow(num14, 4.0));
			}
			num8 = Math.Exp(num9);
			phase = stream.CombinedLiquidPhase;
			phase2 = stream.GetPhaseByType(PhaseType.Vapour);
			num2 = phase.Density.Value * volumeFractionL + phase2.Density.Value * (1.0 - volumeFractionL);
			num3 = phase.Density.Value * num + phase2.Density.Value * (1.0 - num);
			num11 = phase.Viscosity.Value * volumeFractionL + phase2.Viscosity.Value * (1.0 - volumeFractionL);
			num5 = num2 * vm * diam / num11;
			num7 = FrictionFactorOfMoody(num5, roughness / diam);
			num6 = num7 * num8;
			num12 = stream.MassFlowrate.Value * phase2.MassFraction.Value / (phase2.Density.Value * 0.25 * Math.PI * diam * diam);
			num10 = vm * num12 * num2 / phase.Pressure.Value;

	public static double FrictionfactorOfChurchill(double Re, double ed)
	{
		double y = 16.0;

		try
		{
			double num2 = Math.Pow(7.0 / Re, 0.9) + 0.27 * ed;
			double num3 = Math.Pow(37530.0 / Re, y);
			double num4 = Math.Pow(2.457 * Math.Log(1.0 / num2), y);
			double x = Math.Pow(8.0 / Re, num) + Math.Pow(num4 + num3, -1.5);
			return 2.0 * Math.Pow(x, 1.0 / num);
		}
		catch (Exception innerException)
		{
			throw new Exception(Resources.ChurchillCorrFailed, innerException);
		}
	}

	public static void GetSupVelocities(IFlowStream stream, double diameter, ref double vsg, ref double vsl, ref double vm)
	{

		IPhase combinedLiquidPhase = stream.CombinedLiquidPhase;
		double num = stream.MassFlowrate.Value / HTFSUtility.FlowArea(diameter);
		vsg = num * phaseByType.MassFraction.Value / phaseByType.Density.Value;
		vsl = num * combinedLiquidPhase.MassFraction.Value / combinedLiquidPhase.Density.Value;
		vm = vsg + vsl;
	}

	public static double GetWallisBubbleVelocity(IFlowStream stream, double diameter)
	{
		double fEMPTY = RealValue.FEMPTY;
		IPhase combinedLiquidPhase = stream.CombinedLiquidPhase;
		IPhase phaseByType = stream.GetPhaseByType(PhaseType.Vapour);
		if (combinedLiquidPhase == null || phaseByType == null || diameter <= 0.0)
		{
			return fEMPTY;
		}
		double value = combinedLiquidPhase.Density.Value;
		double value2 = phaseByType.Density.Value;
		double value3 = combinedLiquidPhase.SurfaceTension.Value;
		double value4 = combinedLiquidPhase.Viscosity.Value;
		double num = 9.80665 * diameter * diameter * (value - value2) / value3;
		double num2 = Math.Sqrt(9.80665 * value * (value - value2) * diameter * diameter * diameter) / value4;
		double num3 = 1.0;
		if (num2 <= 18.0)
		{
			num3 = 2.5;
		}
		else if (num2 > 18.0 && num2 < 250.0)
		{
			num3 = 69.0 / Math.Pow(num2, 0.35);
		}
		else if (num2 >= 250.0)
		{
			num3 = 10.0;
		}
		double num4 = 0.345 * (1.0 - Math.Exp(-0.029 * num2)) * (1.0 - Math.Exp((3.37 - num) / num3));
		return num4 * 9.80665 * diameter * (value - value2) / value;
	}
