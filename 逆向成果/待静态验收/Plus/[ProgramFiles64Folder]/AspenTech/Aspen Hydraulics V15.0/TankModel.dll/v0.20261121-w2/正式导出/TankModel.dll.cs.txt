	public double InitWettedArea()
	{
		try
		{
			if (initLiqHeight > vesselHeight)
			{
				throw new ArgumentOutOfRangeException(string.Format(Resources.InitialLiquidHeightTooLarge, initLiqHeight.ToString()));
			}
			return Math.PI * vesselDiameter * initLiqHeight;
		}
		catch (Exception innerException)
		{
			throw new Exception(Resources.WettedAreaCalcFailed, innerException);
		}
	}

	public void ConfigureVessel(IFlowStream myStream, IPhysicalPropertyPackage myPPP)
	{
		vesselVolume = Math.PI * (vesselDiameter / 2.0) * (vesselDiameter / 2.0) * vesselHeight;
		vesselArea = Math.PI * vesselDiameter * vesselHeight;
		vesselTopArea = Math.PI * (vesselDiameter / 2.0) * (vesselDiameter / 2.0);
		vesselBottomArea = Math.PI * (vesselDiameter / 2.0) * (vesselDiameter / 2.0);
		vesselTotalArea = vesselArea + vesselTopArea + vesselBottomArea;
		posValve = vesselHeight;
		innerArea = vesselArea;
		outerArea = Math.PI * (vesselDiameter + 2.0 * (metalThk + insulationThk)) * vesselHeight;
		middleArea = Math.PI * (vesselDiameter + 2.0 * metalThk) * vesselHeight;
		deltaVapMoles = 0.0;
		deltaLiqMoles = 0.0;
		double num = 0.0;
		double num2 = 0.0;
		double num3 = 0.0;
		double num4 = 0.0;
		deltaP = 0.0;
		deltaT = 0.0;
		double value = myStream.GetPhaseByID(0).Pressure.Value;
		double value2 = myStream.GetPhaseByID(0).Temperature.Value;
		myPPP.FlashStream(myStream, FlashType.PTFlash, value, value2);
		int numberOfPhases = myStream.NumberOfPhases;
		if (numberOfPhases == 1)
		{
			if (myStream.GetPhaseByID(1).PhaseType == PhaseType.Vapour)
			{
				if (currentTime == 0.0)
				{
					initLiqHeight = 0.0;
					liqHeight = 0.0;
					isInitLiqHeightSpecified = true;
					InitLiqVol = 0.0;
					liqVol = 0.0;
					vapVol = vesselVolume;
					vesselDensity = myStream.GetPhaseByType(PhaseType.Vapour).Density.Value;
					vesselVapRho = myStream.GetPhaseByType(PhaseType.Vapour).Density.Value;
					vesselVapMW = myStream.GetPhaseByType(PhaseType.Vapour).MolecularWeight.Value;
					vesselLiqRho = 0.0;
					vesselLiqMW = 0.0;
					vesselTotalMass = vesselDensity * vesselVolume;
					vesselVapMass = vesselTotalMass;
					vesselLiqMass = 0.0;
					vesselTotalMoles = vesselTotalMass / myStream.GetPhaseByType(PhaseType.Vapour).MolecularWeight.Value;
					vesselVapMoles = vesselTotalMoles;
					vesselLiqMoles = 0.0;
				}
			}
			else if (currentTime == 0.0)
			{
				initLiqHeight = vesselHeight;
				isInitLiqHeightSpecified = true;
				liqHeight = vesselHeight;
				liqVol = vesselVolume;
				vapVol = 0.0;
				vesselDensity = myStream.GetPhaseByType(PhaseType.Liquid).Density.Value;
				vesselLiqRho = myStream.GetPhaseByType(PhaseType.Liquid).Density.Value;
				vesselLiqMW = myStream.GetPhaseByType(PhaseType.Liquid).MolecularWeight.Value;
				vesselVapRho = 0.0;
				vesselVapMW = 0.0;
				vesselTotalMass = vesselDensity * vesselVolume;
				vesselLiqMass = vesselTotalMass;
				vesselVapMass = 0.0;
				vesselTotalMoles = vesselTotalMass / myStream.GetPhaseByType(PhaseType.Liquid).MolecularWeight.Value;
				vesselLiqMoles = vesselTotalMoles;
				vesselVapMoles = 0.0;
			}
		}
		if (numberOfPhases == 2)
