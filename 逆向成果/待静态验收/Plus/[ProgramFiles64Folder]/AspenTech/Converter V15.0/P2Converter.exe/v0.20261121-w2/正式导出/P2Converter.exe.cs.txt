namespace P2Converter
{
	public class App : Application
	{
		private bool _contentLoaded;

		protected override void OnStartup(StartupEventArgs e)
		{
			string[] args = e.Args;
			foreach (string text in args)
			{
				if (text.StartsWith("/AspenPlus"))
				{
					ApplicationName.appName = "Aspen Plus";
				}
				else if (text.StartsWith("/HYSYS"))
				{
					ApplicationName.appName = "HYSYS";
				}
				else if (text.StartsWith("/V"))
				{
					HYSYSVersionNumber.versionNumber = text.Substring(1);
				}
			}
			base.OnStartup(e);
		}

		[DebuggerNonUserCode]
		[GeneratedCode("PresentationBuildTasks", "4.0.0.0")]
		public void InitializeComponent()
		{
			if (!_contentLoaded)
			{
				_contentLoaded = true;
				base.StartupUri = new Uri("Window1.xaml", UriKind.Relative);
				Uri resourceLocator = new Uri("/P2Converter;component/app.xaml", UriKind.Relative);
				Application.LoadComponent(this, resourceLocator);
			}
		}

		[STAThread]
		[DebuggerNonUserCode]
		[GeneratedCode("PresentationBuildTasks", "4.0.0.0")]
		public static void Main()
		{
			App app = new App();
			app.InitializeComponent();
			app.Run();
		}
	}
	public class Window1 : Window, IComponentConnector
	{
		private ApplicationV150 pgui = new ApplicationV150();

		internal Window1 mainwindow;

		private bool _contentLoaded;

		public Window1()
		{
			InitializeComponent();
			if (string.IsNullOrEmpty(ApplicationName.appName) || (ApplicationName.appName == "HYSYS" && string.IsNullOrEmpty(HYSYSVersionNumber.versionNumber)))
			{
				MessageBox.Show("Converter need know the target the model should be converted to, that is, Aspen Plus or Aspen HYSYS and its version number, those information should be passed to the converter if you start it from the shortcut. Please close it and try to start from the shortcut.", P2Globalization.GetString("P2Convert"), MessageBoxButton.OK, MessageBoxImage.Asterisk);
			}
			pgui.OnClose += pgui_OnClose;
			ConvertType type = ConvertType.All;
			if (ApplicationName.appName == "Aspen Plus")
			{
				type = ConvertType.AspenPlus;
			}
			else if (ApplicationName.appName == "HYSYS")
			{
				type = ConvertType.HYSYS;
			}
			pgui.Init(type);
		}

		private void pgui_OnClose()
		{
			Close();
		}

		[DebuggerNonUserCode]
		[GeneratedCode("PresentationBuildTasks", "4.0.0.0")]
		public void InitializeComponent()
		{
			if (!_contentLoaded)
			{
				_contentLoaded = true;
				Uri resourceLocator = new Uri("/P2Converter;component/window1.xaml", UriKind.Relative);
				Application.LoadComponent(this, resourceLocator);
			}
		}

		[DebuggerNonUserCode]
		[GeneratedCode("PresentationBuildTasks", "4.0.0.0")]
		[EditorBrowsable(EditorBrowsableState.Never)]
		void IComponentConnector.Connect(int connectionId, object target)
		{
			if (connectionId == 1)
			{
				mainwindow = (Window1)target;
			}
			else
			{
				_contentLoaded = true;
			}
		}
	}
