		private void btnOpen_Click(object sender, RoutedEventArgs e)
		{
			System.Windows.Forms.OpenFileDialog openFileDialog = new System.Windows.Forms.OpenFileDialog();
			openFileDialog.Multiselect = true;
			openFileDialog.Filter = P2Globalization.GetString("FileFilter");
			if (string.IsNullOrEmpty(ApplicationName.appName))
			{
				ApplicationName.appName = cmbType.Text;
			}
			if (openFileDialog.ShowDialog() != System.Windows.Forms.DialogResult.OK)
			{
				return;
			}
			DebugPrint.Clear(LeaveParse: false);
			btnRun.IsEnabled = true;
			edtOutputHYSYS.Document.Blocks.Clear();
			edtOutputPlus.Document.Blocks.Clear();
			if (cmbType.SelectedIndex == 0)
			{
				DebugPrint.OutputText = edtOutputHYSYS;
			}
			else
			{
				DebugPrint.OutputText = edtOutputPlus;
			}
			tabOutput.SelectedIndex = cmbType.SelectedIndex;
			IP2ConvertCore iP2ConvertCore = null;
			try
			{
				iP2ConvertCore = Activator.CreateInstance(Type.GetTypeFromProgID("P2ConvertCore.P2ConvertCoreV150", throwOnError: true)) as IP2ConvertCore;
			}
			catch (Exception ex)
			{
				System.Windows.MessageBox.Show(ex.Message);
				return;
			}
			m_P2Objects = iP2ConvertCore.Parse(openFileDialog.FileName, out var visual);
			P2Component p2Component = null;
			foreach (P2BaseObject child in m_P2Objects.Children)
			{
				if (child is P2Component)
				{
					p2Component = child as P2Component;
					break;
				}
			}
			m_oclObjects.Clear();
			m_oclObjects.Add(visual);
			tvwPro2File.ItemsSource = m_oclObjects;
			AppTitle = P2Globalization.GetString("AppTitle") + openFileDialog.FileName;
			iP2ConvertCore = null;
			ExpandAll(tvwPro2File);
			Options.IsEosHYSYSProp = true;
			if (p2Component != null && p2Component.SolidComponentExists && ConvertType != ConvertType.AspenPlus)
			{
				System.Windows.MessageBox.Show("Model includes solid component should be converted to Aspen Plus instead of Aspen HYSYS. Please start \"PROII to Aspen Plus Converter\" application.", P2Globalization.GetString("P2Convert"), MessageBoxButton.OK, MessageBoxImage.Asterisk);
				btnRun.IsEnabled = false;
			}
			else
			{
				dlgFind.CurrentObject = visual;
				tvwPro2File.Focus();
			}
		}

		private void ExpandAll(ItemsControl control)
		{
			if (control == null)
			{
				return;
			}
			foreach (object item in (IEnumerable)control.Items)
			{
				if (control.ItemContainerGenerator.ContainerFromItem(item) is TreeViewItem { HasItems: not false } treeViewItem)
				{
					treeViewItem.IsExpanded = true;
					treeViewItem.UpdateLayout();
					ExpandAll(treeViewItem);
				}
			}
		}

		private void btnRun_Click(object sender, RoutedEventArgs e)
		{
			if (m_oclObjects.Count == 0)
			{
				System.Windows.MessageBox.Show(P2Globalization.GetString("OpenFirst"), AppTitle, MessageBoxButton.OK, MessageBoxImage.Hand);
				return;
			}
			base.Cursor = System.Windows.Input.Cursors.Wait;
			edtOutputHYSYS.Document.Blocks.Clear();
			edtOutputPlus.Document.Blocks.Clear();
			DebugPrint.Clear(LeaveParse: true);
			if (cmbType.SelectedIndex == 0)
			{
				(Activator.CreateInstance(Type.GetTypeFromProgID("P2ConvertCore.P2ConvertCoreV150")) as IP2ConvertCore).ConvertHYSYS2(m_P2Objects);
				SetEOSPanel();
			}
			else
			{
				string text = P2Globalization.GetString("AppTitle");
				Convert2Plus(AppTitle.Substring(text.Length, AppTitle.Length - text.Length), delegate(string filePath)
				{
					PrintErrorMessage(filePath);
				});
			}
			base.Cursor = System.Windows.Input.Cursors.Arrow;
		}

		private void PrintErrorMessage(string filePath)
		{
			try
			{
				edtOutputPlus.Document.Blocks.Clear();
				if (File.Exists(filePath))
				{
					string[] array = File.ReadAllLines(filePath);
					Paragraph paragraph = new Paragraph();
					string[] array2 = array;
					foreach (string text in array2)
					{
						Run run = new Run(text + "\r\n");
						if (text.Contains("*ERROR*"))
						{
							run.Foreground = new SolidColorBrush(Colors.Red);
						}
						else if (text.Contains("*WARNING*"))
						{
							run.Foreground = new SolidColorBrush(Colors.Blue);
						}
						else if (text.Contains("Version:"))
						{
							continue;
						}
						paragraph.Inlines.Add(run);
					}
					edtOutputPlus.Document.Blocks.Add(paragraph);
				}
				else
				{
					edtOutputPlus.AppendText("Conversion Completed.");
				}
			}
			catch
			{
			}
		}

		private bool ConvertVersionNumberToDouble(string version, out double result)
		{
			if (string.IsNullOrEmpty(version))
			{
				result = 12.0;
				return true;
			}
			if (Regex.IsMatch(version, "^[V,v]{1}[0-9]{1,2}\\.[0-9]"))
			{
				double.TryParse(version.Substring(1), out result);
				return true;
			}
			result = 0.0;
			return false;
		}

		private void btnHelp_Click(object sender, RoutedEventArgs e)
		{
			RegistryKey rootKey = RetrieveKey();
			if (ApplicationName.appName == "Aspen Plus")
			{
				showWebHelpForAspenPlus(rootKey);
			}
			else
			{
				if (!ConvertVersionNumberToDouble(HYSYSVersionNumber.versionNumber, out var result))
				{
					return;
				}
				if (!(result >= 11.0))
				{
					try
					{
						Process process = new Process();
						process.StartInfo.FileName = $"{Environment.CurrentDirectory}\\HYSYS PROII Converter_V1.5.pdf";
						process.Start();
						return;
					}
					catch (Exception)
					{
						System.Windows.MessageBox.Show(P2Globalization.GetString("PDFNotInstalled"), AppTitle, MessageBoxButton.OK, MessageBoxImage.Exclamation);
						return;
					}
				}
				if (htmlHelpMain == null)
				{
					InitializeHtmlHelpComponent(rootKey);
				}
				ShowHelpPage();
			}
		}

		private RegistryKey RetrieveKey()
		{
			return RegistryKey.OpenBaseKey(RegistryHive.LocalMachine, Environment.Is64BitOperatingSystem ? RegistryView.Registry64 : RegistryView.Registry32);
		}

		private void showWebHelpForAspenPlus(RegistryKey rootKey)
		{
			RegistryKey registryKey = rootKey.OpenSubKey("Software\\AspenTech\\Aspen Plus\\" + HYSYSVersionNumber.InternalVersion + "\\mm");
			if (registryKey == null)
			{
				registryKey = rootKey.OpenSubKey("Software\\Wow6432Node\\AspenTech\\Aspen Plus\\" + HYSYSVersionNumber.InternalVersion + "\\mm");
			}
			string uriString = string.Empty;
			if (registryKey != null)
			{
				List<string> list = new List<string>();
				list.AddRange(registryKey.GetValueNames());
				if (list.Contains("HtmlHelpRoot"))
				{
					uriString = registryKey.GetValue("HtmlHelpRoot").ToString();
					uriString = uriString.Replace("apwn.htm", "Subsystems/userguide1/Content/html/converting_pro_ii_input_keyword_files.htm");
				}
				registryKey.Close();
			}
			System.Windows.Controls.WebBrowser webBrowser = new System.Windows.Controls.WebBrowser();
			Uri uri = new Uri(uriString);
			webBrowser.Navigating += wb_Navigating;
			if (!string.IsNullOrEmpty(uri.LocalPath) && File.Exists(uri.LocalPath))
			{
				webBrowser.Navigate(uri);
				if (aplusHelpWin == null)
				{
					aplusHelpWin = new Window();
					aplusHelpWin.Closed += aplusHelpWin_Closed;
					aplusHelpWin.Owner = this;
					aplusHelpWin.Title = "PRO/II Converter";
					aplusHelpWin.Content = webBrowser;
				}
				aplusHelpWin.Show();
			}
		}

		{
			FieldInfo field = typeof(System.Windows.Controls.WebBrowser).GetField("_axIWebBrowser2", BindingFlags.Instance | BindingFlags.NonPublic);

				object value = field.GetValue(wb);

			}
		}

		private void InitializeHtmlHelpComponent(RegistryKey rootKey)
		{
			if (htmlHelpMain == null)
			{
				htmlHelpMain = new HtmlHelpStrongNameInvoker();
				string htmlHelpPath = GetHtmlHelpPath(rootKey);
				if (!string.IsNullOrEmpty(htmlHelpPath))
				{
					uriHtmlHelpSite = htmlHelpPath;
					htmlHelpSelfExtractor = GetHtmlHelpExtractor(rootKey);
					htmlHelpMain.HtmlHelpInitWrap("PROII Converter", HYSYSVersionNumber.versionNumber, HYSYSVersionNumber.InternalVersion, htmlHelpPath, "PROII Converter", htmlHelpSelfExtractor, bNoSelfExtract: false);
				}
			}
		}

		private static string GetHtmlHelpPath(RegistryKey rootKey)
		{
			string name = "SOFTWARE\\AspenTech\\PROII Converter\\" + HYSYSVersionNumber.InternalVersion;
			RegistryKey registryKey = rootKey.OpenSubKey(name, writable: false);
			if (registryKey == null)
			{
				return null;
			}
			string result = null;
			try
			{
				object value = registryKey.GetValue("HtmlHelpRoot");
				if (value != null)
				{
					result = value.ToString();
				}
			}
			catch (Exception)
			{
			}
			registryKey.Close();
			return result;
		}

		private bool FindItem2Select(ItemsControl ctrl, string str)
		{
			foreach (object item in (IEnumerable)ctrl.Items)
			{
				P2BaseObject p2BaseObject = item as P2BaseObject;
				TreeViewItem treeViewItem = (TreeViewItem)ctrl.ItemContainerGenerator.ContainerFromItem(p2BaseObject);
				if (p2BaseObject.Name.ToUpper().IndexOf(str.ToUpper()) == -1)
				{
					if (treeViewItem != null && treeViewItem.HasItems && FindItem2Select(treeViewItem, str))
					{
						return true;
					}
					continue;
				}
				treeViewItem.IsSelected = true;
				return true;
			}
			return false;
		}
