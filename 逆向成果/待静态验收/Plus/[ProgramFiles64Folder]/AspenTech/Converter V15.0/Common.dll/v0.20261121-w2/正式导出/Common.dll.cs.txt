public class CleanResource
{
	public static void ReleaseComObjectHelper(object obj, bool bFinalRelease)
	{
		try
		{
			if (obj != null)
			{
				if (bFinalRelease)
				{
					Marshal.FinalReleaseComObject(obj);
				}
				else
				{
					Marshal.ReleaseComObject(obj);
				}
				obj = null;
			}
		}
		catch (Exception)
		{
		}
	}
}

public static class CurveInterpolation
{
	public static double GetInterpolationDatum(double[] ExperimentalX, double[] ExperimentalY, double x0)
	{
		double[] array = new double[3];
		double[] array2 = new double[3];
		if (ExperimentalX.Length > 2)
		{
			if (x0 <= ExperimentalX[2])
			{
				for (int i = 0; i <= 2; i++)
				{
					array[i] = ExperimentalX[i];
					array2[i] = ExperimentalY[i];
				}
			}
			else if (x0 >= ExperimentalX[ExperimentalX.Length - 3])
			{
				for (int j = 0; j <= 2; j++)
				{
					array[j] = ExperimentalX[ExperimentalX.Length - 3 + j];
					array2[j] = ExperimentalY[ExperimentalX.Length - 3 + j];
				}
			}
			else
			{
				int num = 0;
				for (int k = 0; k <= ExperimentalX.Length - 1; k++)
				{
					if (x0 >= ExperimentalX[k])
					{
						num = k;
					}
				}
				for (int l = 0; l <= 2; l++)
				{
					array[l] = ExperimentalX[num + l];
					array2[l] = ExperimentalY[num + l];
				}
			}
			return LagrangeInterpolation(array, array2, x0, 2);
		}
		if (ExperimentalX.Length == 2)
		{
			return LagrangeInterpolation(ExperimentalX, ExperimentalY, x0, 1);
		}
		DebugPrint.OutputDebugString(DebugLevel.BuildHYSYSObject, WarningLevel.Warning, "No Enough Points To Interpolate");
		return 0.0;
	}

	public static double LagrangeInterpolation(double[] x, double[] y, double x0, int order)
	{
		double num = 0.0;
		for (int i = 0; i <= order; i++)
		{
			double num2 = 1.0;
			for (int j = 0; j <= order; j++)
			{
				if (i != j)
				{
					num2 = num2 * (x0 - x[j]) / (x[i] - x[j]);
				}
			}
			num += y[i] * num2;
		}
		return num;
	}

	{

		{
			int num = 1;
			int num2 = dtOutput.Rows.Count;
			for (int i = 0; i < num2; i++)
			{
				DataRow dataRow = dtOutput.Rows[i];
				if (((int)dataRow["DebugLevel"] & num) == 0)
				{
					dtOutput.Rows.Remove(dataRow);
					i--;
					num2--;
				}
			}
			Refresh();
		}
		else
		{
			dtOutput.Rows.Clear();
		}

	private static void Add2DataTable(DebugLevel level, WarningLevel warning, string strMessage)

		dataRow["Message"] = strMessage;
		dtOutput.Rows.Add(dataRow);
	}

	public static void Refresh()
	{
		if (OutputText == null)
		{
			return;
		}
		OutputText.Document.Blocks.Clear();
		int showLevel = (int)ShowLevel;
		foreach (DataRow row in dtOutput.Rows)
		{
			_ = (DebugLevel)row["DebugLevel"];
			WarningLevel warningLevel = (WarningLevel)row["WarningLevel"];
			string text = row["Message"].ToString();
			int num = (int)warningLevel;
			if ((showLevel & num) != 0)
			{
				System.Windows.Controls.Image image = new System.Windows.Controls.Image();
				Paragraph paragraph = new Paragraph();
				switch (warningLevel)
				{
				case WarningLevel.Notes:
					image.Source = biNotes;
					paragraph.Foreground = System.Windows.Media.Brushes.Black;
					break;
				case WarningLevel.Warning:
					image.Source = biWarning;
					paragraph.Foreground = System.Windows.Media.Brushes.Blue;
					break;
				case WarningLevel.Error:
					image.Source = biError;
					paragraph.Foreground = System.Windows.Media.Brushes.Red;
					break;
				}
				image.Width = 10.0;
				image.Margin = new Thickness(0.0, 0.0, 3.0, 0.0);
				paragraph.Inlines.Add(image);
				paragraph.Inlines.Add(text);
				OutputText.Document.Blocks.Add(paragraph);
				OutputText.ScrollToEnd();
			}
		}
	}

	public DEFINE(string DefineExpress, string BlockName, string ID)
	{
		MainObjID = ID;

		int num = DefineExpress.IndexOf(" AS ");
		if (num > 0)
		{
			string Type = string.Empty;
			string ID2 = string.Empty;
			string Prop = string.Empty;
			bool bWetOrDry = false;
			MainProperty = DefineExpress.Substring(0, num).Trim();
			string text = DefineExpress.Substring(num + 4).Trim();
			OperatorName = GetOperator(text);
			if (!string.IsNullOrEmpty(OperatorName))
			{
				string[] array = text.Split(new string[1] { OperatorName }, StringSplitOptions.RemoveEmptyEntries);
				if (array.Length == 2)
				{
					SetValue(array[0].ToString().Trim(), ref Type, ref ID2, ref Prop, ref bWetOrDry, bFirst: true);
					FirstRefType = Type;
					FirstRefID = ID2;
					FirstRefProperty = Prop;
					if (!bDryRate)

						bDryRate = bWetOrDry;
					}
					Type = string.Empty;

					Prop = string.Empty;
					if (array[1].StartsWith(","))
					{
						array[1] = array[1].Remove(0, 1);

					SetValue(array[1].ToString().Trim(), ref Type, ref ID2, ref Prop, ref bWetOrDry, bFirst: false);
					SecondRefType = Type;
					SecondRefID = ID2;

					if (!bDryRate)

		}

	private void DefineLinktoCalculator()
	{
		if ((!string.IsNullOrEmpty(FirstRefType.ToUpper().Trim()) && FirstRefType.ToUpper().Trim().StartsWith("CALC")) || (!string.IsNullOrEmpty(SecondRefType.ToUpper().Trim()) && SecondRefType.ToUpper().Trim().StartsWith("CALC")))

		}

	private void SetValue(string strExpress, ref string Type, ref string ID, ref string Prop, ref bool bWetOrDry, bool bFirst)
	{
		if (strExpress.ToUpper().Contains("COMP="))
		{
			strExpress = RemoveComponentDef(strExpress, bFirst);
		}
		string[] array = ParseUtilty.CutBySpaceOrComma(strExpress);
		if (array.Length == 2)
		{
			string text = array[0].ToString().Trim();
			string text2 = array[1].ToString().Trim();
			if (text.Contains("="))
			{
				string[] array2 = ParseUtilty.CutByEqual(text);
				Type = array2[0].ToString().Trim();
				ID = array2[1].ToString().Trim();
				Prop = text2;
			}
			else
			{
				string[] array2 = ParseUtilty.CutByEqual(text2);
				Type = array2[0].ToString().Trim();
				ID = array2[1].ToString().Trim();
				Prop = text;
			}
			return;
		}
		if (array.Length == 1)
		{
			Prop = array[0].ToString().Trim();
			return;
		}
		string[] array3 = array;
		foreach (string text3 in array3)
		{
			if (text3.Contains("="))
			{
				if (text3.ToUpper().StartsWith("PHASE"))
				{
					PHASE = ParseUtilty.GetStringValue(text3);
					continue;
				}
				if (text3.ToUpper().StartsWith("TRAY"))
				{
					TrayNumber = int.Parse(ParseUtilty.GetStringValue(text3));
					continue;
				}
				string[] array2 = ParseUtilty.CutByEqual(text3);
				Type = array2[0].ToString().Trim();
				ID = array2[1].ToString().Trim();
			}
			else if (text3.ToUpper() == "WET")
			{
				bWetOrDry = false;
			}
			else if (text3.ToUpper() == "DRY")
			{
				bWetOrDry = true;
			}
			else
			{
				Prop = text3;
			}
		}
	}

	public static string ConvertName(string strName, bool bIsStream)
	{
		if (ChangeName2HYSYS)
		{
			if (strName.Length >= 8)
			{
				if (bIsStream)
				{
					if (dictStream.ContainsKey(strName))
					{
						return dictStream[strName];
					}
				}
				else if (dictEquipment.ContainsKey(strName))
				{
					return dictEquipment[strName];
				}
				string text = "";
				while (true)
				{
					int num = (bIsStream ? nStreamIndex.ToString().Length : nEquipmentIndex.ToString().Length);
					text = strName.Substring(0, 8 - num) + (bIsStream ? nStreamIndex.ToString() : nEquipmentIndex.ToString());
					if (!(bIsStream ? dictStream.ContainsKey(text) : dictStream.ContainsKey(text)))
					{
						break;
					}
					if (bIsStream)
					{
						nStreamIndex++;
					}
					else
					{
						nEquipmentIndex++;
					}
				}
				if (bIsStream)
				{
					dictStream.Add(strName, text);
					nStreamIndex++;
				}
				else
				{
					dictEquipment.Add(strName, text);
					nEquipmentIndex++;
				}
				return text;
			}
			return strName;
		}
		return strName;
	}

	private string RemoveComponentDef(string strExpress, bool bPrimary)
	{
		_ = string.Empty;
		MatchCollection matchCollection = new Regex("COMP\\s*=\\s*(\\d+),\\s*(\\d+)", RegexOptions.IgnoreCase).Matches(strExpress);
		Match match = null;
		if (matchCollection.Count == 1)
		{
			match = matchCollection[0];
			return setComponentData(match, strExpress, bPrimary);
		}
		matchCollection = new Regex("COMPONENT\\s*=\\s*(\\d+),\\s*(\\d+)", RegexOptions.IgnoreCase).Matches(strExpress);
		if (matchCollection.Count == 1)
		{
			match = matchCollection[0];
			return setComponentData(match, strExpress, bPrimary);
		}
		matchCollection = new Regex("COMP\\s*=\\s*(\\d+)", RegexOptions.IgnoreCase).Matches(strExpress);
		if (matchCollection.Count == 1)
		{
			match = matchCollection[0];
			return setComponentData(match, strExpress, bPrimary).Trim();
		}
		matchCollection = new Regex("COMPONENT\\s*=\\s*(\\d+)", RegexOptions.IgnoreCase).Matches(strExpress);
		if (matchCollection.Count == 1)
		{
			match = matchCollection[0];
			return setComponentData(match, strExpress, bPrimary);
		}
		return strExpress;
	}

	private string setComponentData(Match match, string strExpress, bool bPrimary)
	{
		if (bPrimary)
		{
			PrimaryStartComponent = int.Parse(match.Groups[1].Value);
			if (match.Groups.Count == 3)
			{
				PrimaryEndComponent = int.Parse(match.Groups[2].Value);
			}
			else if (match.Groups.Count == 2)
			{
				PrimaryEndComponent = PrimaryStartComponent;
			}
		}
		else
		{
			ReferenceStartComponent = int.Parse(match.Groups[1].Value);
			if (match.Groups.Count == 3)
			{
				ReferenceEndComponent = int.Parse(match.Groups[2].Value);
			}
			else if (match.Groups.Count == 2)
			{
				ReferenceEndComponent = ReferenceStartComponent;
			}
		}
		int num = strExpress.IndexOf(match.Groups[0].Value);
		int num2 = num + match.Groups[0].Value.Length;
		if (num2 < strExpress.Length)
		{
			num2++;
		}
		return strExpress.Remove(num, num2 - num);
	}

public class RangeValue
{
	public string Name { get; set; }

	public string StartValue { get; set; }

	public string EndValue { get; set; }

	public string Qualifier { get; set; }

	public int QualifierToInt => GetQualifierAsInt();

	public RangeValue()
	{
	}

	public RangeValue(string qual, string start, string end)
	{
		Qualifier = qual;
		StartValue = start;
		EndValue = end;
	}

	public void GetValueAsDouble(ref double start, ref double end)
	{
		if (!double.TryParse(StartValue, out start) || !double.TryParse(EndValue, out end))
		{
			throw new Exception("Failed ot parse Value of " + StartValue + "or" + EndValue + " to double");
		}
	}

	public void GetValueAsInt(ref int start, ref int end)
	{
		if (!int.TryParse(StartValue, out start) || !int.TryParse(EndValue, out end))
		{
			throw new Exception("Failed ot parse Value of " + StartValue + "or" + EndValue + " to integer");
		}
	}

	private int GetQualifierAsInt()
	{
		if (int.TryParse(Qualifier, out var result))
		{
			return result;
		}
		throw new Exception("Failed ot parse Value of " + Qualifier + " to integer");
	}
}

public class UnitType
{
	public IDictionary<string, Unit> Units = new Dictionary<string, Unit>();

	public string Name { get; set; }

	public string Key { get; set; }

	public Unit DefaultUnit { get; set; }

	public override string ToString()
	{
		return Name;
	}

	public UnitType()
	{
	}

	public UnitType(string strName, string strKey)
	{
		Name = strName;
		Key = strKey;
	}

	public Unit Get(string strKey)
	{
		return Units[strKey];
	}

	public void SetDefaultUnit(string strUnit)
	{
		Unit unit = Units[strUnit.ToUpper()];
		if (unit != null)
		{
			if (DefaultUnit != null)
			{
				DefaultUnit = null;
			}
			DefaultUnit = unit;
		}
		else if (DefaultUnit != null)
		{
			DefaultUnit = null;
		}
	}

	public IDictionary<string, Unit> GetUnits()
	{
		return Units;
	}
}
public class UnitTypes
{
	private IDictionary<string, UnitType> EnglishTypes = new Dictionary<string, UnitType>();

	private IDictionary<string, UnitType> MetricTypes = new Dictionary<string, UnitType>();

	private IDictionary<string, UnitType> SITypes = new Dictionary<string, UnitType>();

	public void Add(Dimensional dim, UnitType type)
	{
		switch (dim)
		{
		case Dimensional.English:
			EnglishTypes.Add(type.Key.ToUpper(), type);
			break;
		case Dimensional.Metric:
			MetricTypes.Add(type.Key.ToUpper(), type);
			break;
		case Dimensional.SI:
			SITypes.Add(type.Key.ToUpper(), type);
			break;
		}
	}

	public UnitType GetUnitType(Dimensional dim, string strName)
	{
		switch (dim)
		{
		case Dimensional.English:
			if (!EnglishTypes.ContainsKey(strName))
			{
				return null;
			}
			return EnglishTypes[strName];
		case Dimensional.Metric:
			if (!MetricTypes.ContainsKey(strName))
			{
				return null;
			}
			return MetricTypes[strName];
		case Dimensional.SI:
			if (!SITypes.ContainsKey(strName))
			{
				return null;
			}
			return SITypes[strName];
		default:
			return null;
		}
	}
}
