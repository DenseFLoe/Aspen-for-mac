	public void PickupColumnMainFeedsAndProducts(IList<HYSYSStream> stmColl)
	{
		foreach (string outletStream in base.OutletStreams)
		{
			bool flag = true;
			foreach (HYSYSColumn sIDESTRIPPER in SIDESTRIPPERS)
			{
				if (sIDESTRIPPER.InletStreams.Contains(outletStream))
				{
					flag = false;
					break;
				}
			}
			if (flag)
			{
				foreach (HYSYSColumn sIDERECTIFIER in SIDERECTIFIERS)
				{
					if (sIDERECTIFIER.InletStreams.Contains(outletStream))
					{
						flag = false;
						break;
					}
				}
			}
			if (flag)
			{
				HYSYSMainFlowsheetProducts.Add(outletStream);
			}
		}
		foreach (string inletStream in base.InletStreams)
		{
			bool flag2 = true;
			foreach (HYSYSColumn sIDESTRIPPER2 in SIDESTRIPPERS)
			{
				if (sIDESTRIPPER2.OutletStreams.Contains(inletStream))
				{
					flag2 = false;
					break;
				}
			}
			if (flag2)
			{
				using List<HYSYSColumn>.Enumerator enumerator2 = SIDERECTIFIERS.GetEnumerator();
				if (enumerator2.MoveNext())
				{
					_ = enumerator2.Current;
					flag2 = false;
				}
			}
			if (flag2)
			{
				HYSYSMainFlowsheetFeeds.Add(inletStream);
			}
		}
		changeStmInfoOfSubCols(stmColl);
	}

	private void changeStmInfoOfSubCols(IList<HYSYSStream> stmColl)
	{
		foreach (HYSYSColumn sIDESTRIPPER in SIDESTRIPPERS)
		{
			foreach (string outletStream in sIDESTRIPPER.OutletStreams)
			{
				if (base.InletStreams.Contains(outletStream))
				{
					continue;
				}
				if (!HYSYSMainFlowsheetProducts.Contains(outletStream))
				{
					HYSYSMainFlowsheetProducts.Add(outletStream);
				}
				foreach (HYSYSStream item in stmColl)
				{
					if (item.Stream == outletStream && item.Source.UID == sIDESTRIPPER.UID)
					{
						item.Source = this;
						break;
					}
				}
			}
			foreach (string inletStream in sIDESTRIPPER.InletStreams)
			{
				if (base.OutletStreams.Contains(inletStream))
				{
					continue;
				}
				if (!HYSYSMainFlowsheetFeeds.Contains(inletStream))
				{
					HYSYSMainFlowsheetFeeds.Add(inletStream);
				}
				foreach (HYSYSStream item2 in stmColl)
				{
					if (item2.Stream == inletStream && item2.Destination.UID == sIDESTRIPPER.UID)
					{
						item2.Destination = this;
						break;
					}
				}
			}
		}
		foreach (HYSYSColumn sIDERECTIFIER in SIDERECTIFIERS)
		{
			foreach (string outletStream2 in sIDERECTIFIER.OutletStreams)
			{
				if (base.InletStreams.Contains(outletStream2))
				{
					continue;
				}
				if (!HYSYSMainFlowsheetProducts.Contains(outletStream2))
				{
					HYSYSMainFlowsheetProducts.Add(outletStream2);
				}
				foreach (HYSYSStream item3 in stmColl)
				{
					if (item3.Stream == outletStream2 && item3.Source.UID == sIDERECTIFIER.UID)
					{
						item3.Source = this;
						break;
					}
				}
			}
			foreach (string inletStream2 in sIDERECTIFIER.InletStreams)
			{
				if (base.OutletStreams.Contains(inletStream2))
				{
					continue;
				}
				foreach (HYSYSStream item4 in stmColl)
				{
					if (!HYSYSMainFlowsheetFeeds.Contains(inletStream2))
					{
						HYSYSMainFlowsheetFeeds.Add(inletStream2);
					}
					if (item4.Stream == inletStream2 && item4.Destination.UID == sIDERECTIFIER.UID)
					{
						item4.Destination = this;
						break;
					}
				}
			}
		}
	}

	public void SeparateGlobalAndLocalSpecs()
	{
		AnalysisSpecification(this);
		SetLocalSpecs(this);
		foreach (HYSYSColumn sIDESTRIPPER in SIDESTRIPPERS)
		{
			AnalysisSpecification(sIDESTRIPPER);
			SetLocalSpecs(sIDESTRIPPER);
		}
		foreach (HYSYSColumn sIDERECTIFIER in SIDERECTIFIERS)
		{
			AnalysisSpecification(sIDERECTIFIER);
			SetLocalSpecs(sIDERECTIFIER);
		}
	}

	public void SeparateGlobalAndLocalDefines()
	{
		AnalysisDefine(this);
		SetLocalDefines(this);
		foreach (HYSYSColumn sIDESTRIPPER in SIDESTRIPPERS)
		{
			AnalysisDefine(sIDESTRIPPER);
			SetLocalDefines(sIDESTRIPPER);
		}
		foreach (HYSYSColumn sIDERECTIFIER in SIDERECTIFIERS)
		{
			AnalysisDefine(sIDERECTIFIER);
			SetLocalDefines(sIDERECTIFIER);
		}
	}

	private void AnalysisSpecification(HYSYSColumn columnData)
	{
		if (columnData.GlobalSpec == null || columnData.GlobalSpec.Count == 0)
		{
			return;
		}
		List<string> allObjNames = GetAllObjNames();
		foreach (Specification item in columnData.GlobalSpec)
		{
			item.SetGlobalProperty(columnData.UID, columnData.InletStreams, columnData.OutletStreams, allObjNames, HYSYSMainFlowsheetProducts, SubColumnProducts());
		}
	}

	private void AnalysisDefine(HYSYSColumn columnData)
	{
		if (columnData.GlobalDefine == null || columnData.GlobalDefine.Count == 0)
		{
			return;
		}
		List<string> allObjNames = GetAllObjNames();
		foreach (DEFINE item in columnData.GlobalDefine)
		{
			item.SetGlobalProperty(columnData.UID, columnData.InletStreams, columnData.OutletStreams, allObjNames, HYSYSMainFlowsheetProducts);
		}
	}

	private List<string> GetAllObjNames()
	{
		List<string> list = new List<string>();
		list.Add(base.UID);
		foreach (HYSYSColumn sIDESTRIPPER in SIDESTRIPPERS)
		{
			list.Add(sIDESTRIPPER.UID);
		}
		foreach (HYSYSColumn sIDERECTIFIER in SIDERECTIFIERS)
		{
			list.Add(sIDERECTIFIER.UID);
		}
		return list;
	}

	public void RemoveInternalStreams(List<string> internalStreamNames)
	{
		foreach (string internalStreamName in internalStreamNames)
		{
			if (base.InletStreams.Contains(internalStreamName))
			{
				base.InletStreams.Remove(internalStreamName);
			}
			else if (base.OutletStreams.Contains(internalStreamName))
			{
				base.OutletStreams.Remove(internalStreamName);
			}
		}
	}

	private List<string> SubColumnProducts()
	{
		List<string> list = new List<string>();
		foreach (HYSYSColumn sIDESTRIPPER in SIDESTRIPPERS)
		{
			foreach (string outletStream in sIDESTRIPPER.OutletStreams)
			{
				list.Add(outletStream);
			}
		}
		foreach (HYSYSColumn sIDERECTIFIER in SIDERECTIFIERS)
		{
			foreach (string outletStream2 in sIDERECTIFIER.OutletStreams)
			{
				list.Add(outletStream2);
			}
		}
		return list;
	}

	private void SetLocalSpecs(HYSYSColumn columnData)
	{
		if (columnData.GlobalSpec == null || columnData.GlobalSpec.Count == 0)
		{
			return;
		}
		foreach (Specification item in columnData.GlobalSpec)
		{
			if (!item.bGlobalSpec)
			{
				columnData.LocalSpec.Add(item);
			}
		}
	}

	private void SetLocalDefines(HYSYSColumn columnData)
	{
		if (columnData.GlobalDefine == null || columnData.GlobalDefine.Count == 0)
		{
			return;
		}
		foreach (DEFINE item in columnData.GlobalDefine)
		{
			if (!item.bGlobalDefine)
			{
				columnData.LocalDefine.Add(item);
			}
		}
	}

	public static string GetMethodName(string p2Set)
	{
		string text = string.Empty;
		if (HYSYSBaseObject.g_Map[strFluidPackagesInDic] is HYSYSFluidPackages hYSYSFluidPackages)
		{
			foreach (HYSYSFluidPackage eOSFluidPackage in hYSYSFluidPackages.EOSFluidPackages)
			{
				if (eOSFluidPackage.FluidPackageName == p2Set)
				{
					text = p2Set;
					break;
				}
			}
			if (string.IsNullOrEmpty(text))
			{
				foreach (HYSYSFluidPackage activityFluidPackage in hYSYSFluidPackages.ActivityFluidPackages)
				{
					if (activityFluidPackage.FluidPackageName == p2Set)
					{
						text = p2Set;
						break;
					}
				}
			}
		}
		return text;
	}

	public static string GetThermoKeyByPackageName(string packageName)
	{
		HYSYSFluidPackages hYSYSFluidPackages = HYSYSBaseObject.g_Map[strFluidPackagesInDic] as HYSYSFluidPackages;
		string result = string.Empty;
		if (hYSYSFluidPackages == null)
		{
			return string.Empty;
		}
		foreach (HYSYSFluidPackage eOSFluidPackage in hYSYSFluidPackages.EOSFluidPackages)
		{
			if (eOSFluidPackage.FluidPackageName == packageName)
			{
				result = eOSFluidPackage.ThomoKey;
				break;
			}
		}
		return result;
	}

	public void LoadStreams()
	{
		if (ExchangerType == HysysHeatExchangerType.Cooler)
		{
			foreach (string inletStream in base.InletStreams)
			{
				HotSideInletStreams.Add(inletStream.Trim());
			}
			{
				foreach (string outletStream in base.OutletStreams)
				{
					HotSideOutletStreams.Add(outletStream.Trim());
				}
				return;
			}
		}
		if (ExchangerType != HysysHeatExchangerType.Heater)
		{
			return;
		}
		foreach (string inletStream2 in base.InletStreams)
		{
			ColdSideInletStreams.Add(inletStream2.Trim());
		}
		foreach (string outletStream2 in base.OutletStreams)
		{
			ColdSideOutletStreams.Add(outletStream2.Trim());
		}
	}

	public void SetInletAndOutletStreams()
	{
		foreach (string hotSideInletStream in HotSideInletStreams)
		{
			base.InletStreams.Add(hotSideInletStream);
		}
		foreach (string coldSideInletStream in ColdSideInletStreams)
		{
			base.InletStreams.Add(coldSideInletStream);
		}
		foreach (string hotSideOutletStream in HotSideOutletStreams)
		{
			base.OutletStreams.Add(hotSideOutletStream);
		}
		foreach (string coldSideOutletStream in ColdSideOutletStreams)
		{
			base.OutletStreams.Add(coldSideOutletStream);
		}
	}

public class HYSYSUnitOperationBase : HYSYSBaseObject
{
	public string UID { get; set; }

	public string FluidPackage { get; set; }

	public List<string> InletStreams { get; set; }

	public List<string> OutletStreams { get; set; }

	public List<string> ConvertNotes { get; set; }

	public HYSYSUnitOperationBase()
	{
		InletStreams = new List<string>();
		OutletStreams = new List<string>();
		ConvertNotes = new List<string>();
		base.GlobalDefine = new List<DEFINE>();
		base.LocalDefine = new List<DEFINE>();
		base.GlobalSpec = new List<Specification>();
		base.LocalSpec = new List<Specification>();
		if (BaseObject._Report != null)
		{
			Report report = BaseObject._Report;
			int hYSYSEquipmentCount = report.HYSYSEquipmentCount + 1;
			report.HYSYSEquipmentCount = hYSYSEquipmentCount;
		}
	}

	public void SetCreateStreams(List<HYSYSBaseObject> objectList)
	{
		new List<HYSYSBaseObject>();
		foreach (string inletStream in InletStreams)
		{
			HYSYSStream hYSYSStream = HYSYSStream.GetStreamByName(inletStream);
			if (hYSYSStream == null)
			{
				foreach (HYSYSBaseObject @object in objectList)
				{
					if (@object.GetType() == typeof(HYSYSStream))
					{
						HYSYSStream hYSYSStream2 = @object as HYSYSStream;
						if (hYSYSStream2.Stream == inletStream)
						{
							hYSYSStream = hYSYSStream2;
							break;
						}
					}
				}
				if (hYSYSStream == null)
				{
					hYSYSStream = new HYSYSStream();
					hYSYSStream.Stream = inletStream;
					objectList.Add(hYSYSStream);
				}
			}
			hYSYSStream.Destination = this;
		}
		foreach (string outletStream in OutletStreams)
		{
			HYSYSStream hYSYSStream3 = HYSYSStream.GetStreamByName(outletStream);
			if (hYSYSStream3 == null)
			{
				foreach (HYSYSBaseObject object2 in objectList)
				{
					if (object2.GetType() == typeof(HYSYSStream))
					{
						HYSYSStream hYSYSStream4 = object2 as HYSYSStream;
						if (hYSYSStream4.Stream == outletStream)
						{
							hYSYSStream3 = hYSYSStream4;
							break;
						}
					}
				}
				if (hYSYSStream3 == null)
				{
					hYSYSStream3 = new HYSYSStream();
					hYSYSStream3.Stream = outletStream;
					objectList.Add(hYSYSStream3);
				}
			}
			hYSYSStream3.Source = this;
		}
	}

	public void SeparateDefines()
	{
		foreach (DEFINE item in base.GlobalDefine)
		{
			item.SetGlobalProperty(UID, InletStreams, OutletStreams);
			if (!item.bGlobalDefine)
			{
				base.LocalDefine.Add(item);
			}
		}
	}

	public void SeparateSpecifications()
	{
		foreach (Specification item in base.GlobalSpec)
		{
			item.SetGlobalProperty(UID, InletStreams, OutletStreams);
			if (!item.bGlobalSpec)
			{
				base.LocalSpec.Add(item);
			}
		}
	}
