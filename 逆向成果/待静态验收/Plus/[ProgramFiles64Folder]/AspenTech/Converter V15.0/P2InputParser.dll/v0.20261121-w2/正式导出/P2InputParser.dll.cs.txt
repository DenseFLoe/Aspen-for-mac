	private FileStream m_fs;

	private StreamReader m_sr;

	private ArrayList m_arlParse;

	private ParseNode m_pnCurrent;

	private bool IsProcedure;

	public bool IsSupportVersion { get; set; }

	public static string TrimEx(string str)
	{
		str = str.Replace('\t', ' ');
		str = str.Replace("\0", "");
		return str.Trim();
	}

	public P2Parser(string strFilePath)
	{
		m_arlParse = ParseNodesLoader.LoadParse();
		m_fs = new FileStream(strFilePath, FileMode.Open, FileAccess.Read);
		m_sr = new StreamReader(m_fs, Encoding.Default);
	}

	~P2Parser()
	{
		if (m_sr != null)
		{
			m_sr.Close();
			m_sr = null;
		}
		if (m_fs != null)
		{
			m_fs.Close();
			m_fs = null;
		}
	}

	private P2BaseObject BuildP2Object(P2BaseObject obj)
	{
		if (obj.Name.ToUpper().StartsWith(Categories.Title))
		{
			return new P2Title(obj);
		}
		if (obj.Name.ToUpper().StartsWith(Categories.Component))
		{
			return new P2Component(obj);
		}
		if (obj.Name.ToUpper().StartsWith(Categories.Thermodynamic))
		{
			return new P2ThermoDynamic(obj);
		}
		if (obj.Name.ToUpper().StartsWith(Categories.Stream))
		{
			return new P2Stream(obj);
		}
		if (obj.Name.ToUpper().StartsWith(Categories.RX))
		{
			return new P2ReactionData(obj);
		}
		if (obj.Name.ToUpper().StartsWith(Categories.Unit))
		{
			return new P2UnitOperation(obj);
		}
		if (obj.Name.ToUpper().StartsWith(Categories.Recy))
		{
			if (obj is VisualBaseNode visualBaseNode)
			{
				visualBaseNode._ConvertResult = VisualBaseNode.ConvertResult.CR_IGNORED;
				visualBaseNode.Message = obj.Name + P2Globalization.GetString("CR_IGNORED");
				if (string.Compare(ApplicationName.appName.Trim(), "HYSYS", ignoreCase: true) == 0)
				{
					DebugPrint.OutputDebugString(DebugLevel.Parse, WarningLevel.Warning, visualBaseNode.Message);
				}
			}
		}
		else
		{
			obj.Name.ToUpper().StartsWith(Categories.Case);
		}
		return null;
	}

	private bool IsBreakLine(ref string strLine)
	{
		for (int i = 0; i < strLine.Length; i++)
		{
			switch (strLine[i])
			{
			case '&':
				strLine = strLine.Substring(0, i).Trim();
				return true;
			case '$':
				strLine = strLine.Substring(0, i).Trim();
				return false;
			case '*':
				if (i < strLine.Length - 1)
				{
					int length = i;
					for (i++; i < strLine.Length; i++)
					{
						switch (strLine[i])
						{
						case '$':
							strLine = strLine.Substring(0, length);
							return true;
						case '\t':
						case ' ':
							continue;
						}
						if (!IsProcedure)
						{
							strLine = strLine.Substring(0, length);
							return true;
						}
					}
					break;
				}
				strLine = strLine.Substring(0, i);
				return true;
			}
		}
		return false;
	}

	private int GetParseLevel(ArrayList arlParses, string strHead, int nLevel, bool bGoChildren)
	{
		for (int i = 0; i < arlParses.Count; i++)
		{
			ParseNode parseNode = arlParses[i] as ParseNode;
			if (strHead.ToUpper().StartsWith(parseNode.Key))
			{
				m_pnCurrent = parseNode;
				return nLevel;
			}
		}
		if (bGoChildren)
		{
			for (int i = 0; i < arlParses.Count; i++)
			{
				ParseNode parseNode2 = arlParses[i] as ParseNode;
				int parseLevel = GetParseLevel(parseNode2.Children, strHead, nLevel + 1, bGoChildren: false);
				if (parseLevel != -1)
				{
					return parseLevel;
				}
			}
		}
		return -1;
	}

	private int PreParse(string strHead)
	{
		int num;
		if (m_pnCurrent == null)
		{
			num = GetParseLevel(m_arlParse, strHead, 0, bGoChildren: true);
		}
		else
		{
			num = GetParseLevel(m_pnCurrent.Children, strHead, m_nLevel + 1, bGoChildren: true);
			ParseNode parent = m_pnCurrent.Parent;
			int num2 = m_nLevel + 1;
			while (num == -1)
			{
				if (parent == null)
				{
					num = GetParseLevel(m_arlParse, strHead, 0, bGoChildren: true);
					break;
				}
				num2--;
				num = GetParseLevel(parent.Children, strHead, num2, bGoChildren: true);
				parent = parent.Parent;
				if (num == -1 && parent == null)
				{
					num = GetParseLevel(m_arlParse, strHead, 0, bGoChildren: false);
					if (num == -1)
					{
						break;
					}
				}
			}
		}
		if (num == -1)
		{
			num = m_nLevel;
		}
		return num;
	}

	private void PerParse(ref string str)
	{
		if (str.ToUpper().Trim().StartsWith("COMP") && str.ToUpper().Trim().EndsWith("DATA"))
		{
			str = "COMPONENT DATA";
		}
		if (str.ToUpper().TrimStart().StartsWith("CMPR "))
		{
			str = str.Replace("CMPR", "COMPRESSOR");
		}
		if (str.ToUpper().Trim().StartsWith("SIDE") && !str.ToUpper().Trim().StartsWith("SIDE(") && !str.ToUpper().Trim().StartsWith("SIDESTRIPPER"))
		{
			string text = str.Substring(4).Trim();
			if (str.ToUpper().Trim().EndsWith("RECTIFIER"))
			{
				str = "SIDERECTIFIER " + text;
			}
			else
			{
				str = "SIDESTRIPPER " + text;
			}
		}
		if (str.StartsWith("PROCEDURE") && !IsProcedure)
		{
			IsProcedure = true;
		}
	}

	private void Parsed(string str)
	{
		if (str.StartsWith("RETURN") && IsProcedure)
		{
			IsProcedure = false;
		}
	}

	private bool ParseNode(ref VisualBaseNode p2bObject)
	{
		bool result = false;
		string text;
		while (true)
		{
			text = m_sr.ReadLine();
			if (text == null)
			{
				return false;
			}
			text = text.ToUpper();
			text = TrimEx(text);
			PerParse(ref text);
			while (text.StartsWith("$") || TrimEx(text).Length < 1)
			{
				RetrieveVersionNumber(text);
				text = m_sr.ReadLine();
				if (text == null)
				{
					return false;
				}
				text = text.ToUpper();
				text = TrimEx(text);
				PerParse(ref text);
			}
			if (!IsBreakLine(ref text))
			{
				break;
			}
			m_strCurrent += text;
		}
		m_strCurrent += text;
		if (m_strCurrent.ToUpper() == "END")
		{
			m_strCurrent = "";
		}
		int num = m_strCurrent.IndexOf(' ');
		if (num == -1)
		{
			num = m_strCurrent.IndexOf('\t');
		}
		string text2 = ((num == -1) ? m_strCurrent : m_strCurrent.Substring(0, num));
		Parsed(text2);
		int num2 = PreParse(text2);
		if (num2 == m_nLevel)
		{
			if (!string.IsNullOrEmpty(m_strCurrent))
			{
				VisualBaseNode visualBaseNode = new VisualBaseNode();
				visualBaseNode.Name = m_strCurrent;
				visualBaseNode.Parent = p2bObject;
				p2bObject.Children.Add(visualBaseNode);
			}
			m_strCurrent = "";
			ParseNode(ref p2bObject);
		}
		else if (num2 > m_nLevel)
		{
			VisualBaseNode visualBaseNode2 = new VisualBaseNode();
			visualBaseNode2.Name = m_strCurrent;
			if (p2bObject.Children.Count > 0)
			{
				VisualBaseNode p2bObject2 = (VisualBaseNode)(visualBaseNode2.Parent = (VisualBaseNode)p2bObject.Children[p2bObject.Children.Count - 1]);
				p2bObject2.Children.Add(visualBaseNode2);
				m_nLevel = num2;
				m_strCurrent = "";
				ParseNode(ref p2bObject2);
			}
		}
		else if (num2 < m_nLevel)
		{
			for (int i = num2; i < m_nLevel; i++)
			{
				p2bObject = (VisualBaseNode)p2bObject.Parent;
			}
			if (!string.IsNullOrEmpty(m_strCurrent))
			{
				VisualBaseNode visualBaseNode3 = new VisualBaseNode();
				visualBaseNode3.Name = m_strCurrent;
				visualBaseNode3.Parent = p2bObject;
				p2bObject.Children.Add(visualBaseNode3);
			}
			m_nLevel = num2;
			m_strCurrent = "";
			ParseNode(ref p2bObject);
		}
		return result;
	}

	private void RetrieveVersionNumber(string strm)
	{
		if (!strm.StartsWith("$") || strm.ToUpper().IndexOf("PRO/II Keyword Generation System".ToUpper()) == -1)
		{
			return;
		}
		int num = strm.ToUpper().IndexOf("<VERSION");
		int num2 = strm.IndexOf('>');
		if (num == -1 || num2 == -1)
		{
			return;
		}
		m_strVersionNumber = strm.Substring(num + 8).Trim();
		m_strVersionNumber = m_strVersionNumber.Substring(0, m_strVersionNumber.Length - 1);
		if (m_strVersionNumber.Length > 3)
		{
			m_strVersionNumber = m_strVersionNumber.Substring(0, 3);
		}
		m_strVersionNumber = m_strVersionNumber.Replace(',', '.');
		if (double.TryParse(m_strVersionNumber, NumberStyles.Float | NumberStyles.AllowThousands, CultureInfo.InvariantCulture, out var result))
		{
			if (result >= 8.3)
			{
				IsSupportVersion = true;
			}
			else
			{
				IsSupportVersion = false;
			}
		}
	}

	public P2BaseObject Parse(out VisualBaseNode txtTree)
	{
		txtTree = new VisualBaseNode();
		txtTree.Name = "PRO/II Input File";
		ParseNode(ref txtTree);
		P2BaseObject p2BaseObject = new P2BaseObject();
		p2BaseObject.Name = "PRO/II Input File";
		if (!IsSupportVersion && !string.IsNullOrEmpty(m_strVersionNumber))
		{
			DebugPrint.OutputDebugString(DebugLevel.Parse, WarningLevel.Warning, P2Globalization.GetString("OldVersionWarning"));
		}
		foreach (VisualBaseNode child in txtTree.Children)
		{
			P2BaseObject p2BaseObject2 = BuildP2Object(child);
			if (p2BaseObject2 != null)
			{
				p2BaseObject2.VisualNode = child;
				p2BaseObject.Children.Add(p2BaseObject2);
			}
		}
		return p2BaseObject;
	}
}

	public ArrayList Children => children;

	public ParseNode Parent { get; set; }

	public override string ToString()

		return Key;
	}
}
