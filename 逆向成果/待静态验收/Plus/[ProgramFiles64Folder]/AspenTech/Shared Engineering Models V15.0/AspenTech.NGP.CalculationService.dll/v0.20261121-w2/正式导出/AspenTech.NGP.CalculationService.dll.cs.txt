		public string OutputDirectory
		{
			get
			{
				if (string.IsNullOrWhiteSpace(_OutputDirectory))
				{
					_OutputDirectory = Directory.GetCurrentDirectory();
				}
				return _OutputDirectory;
			}
			set
			{
				if (Status == BeadStatus.Uninitialized)
				{
					try
					{
						Directory.CreateDirectory(value);
						_OutputDirectory = value;
						return;
					}
					catch (Exception)
					{
						throw;
					}
				}
				throw new InvalidOperationException("Cannot edit OutputDirectory after initialization.");
			}
		}

		public string RunId
		{
			get
			{
				if (string.IsNullOrWhiteSpace(_RunId))
				{
					_RunId = DateTime.Now.ToString("yyyy.MM.dd.HHmmssfff");
				}
				return _RunId;
			}
			set
			{
				if (Status == BeadStatus.Uninitialized)
				{
					if (value.Length <= 255)

					}
					throw new ArgumentOutOfRangeException($"RunId must be <= 255 characters: {value} ({value.Length} )");
				}
				throw new InvalidOperationException("Cannot edit RunId after initialization.");
			}
		}

		public CalculationService(int fileNumber, IDBConfiguration dbConfiguration = null)
		{
			dBConfiguration = dbConfiguration;
			FileNumber = fileNumber;
			string text = string.Empty;
			string text2 = string.Empty;
			string text3 = string.Empty;
			string text4 = string.Empty;
			bool num = Environment.OSVersion.Platform == PlatformID.Win32NT;
			if (num)
			{
				text3 = Helper.IFortDirectoryPath();
				text = Helper.AprSysDirectoryPath(GlobalSpecs.MM);
				text2 = Helper.OomfDirectoryPath(GlobalSpecs.MM);
				if (!Directory.Exists(text))
				{
					throw new DirectoryNotFoundException("Required directory not found: " + text);
				}
				if (!Directory.Exists(text2))
				{
					throw new DirectoryNotFoundException("Required directory not found: " + text2);
				}
				if (!Directory.Exists(text3))
				{
					throw new DirectoryNotFoundException("Required directory not found: " + text3);
				}
			}
			else
			{
				text4 = Path.GetDirectoryName(Assembly.GetExecutingAssembly().Location);
			}
			if (num)
			{
				if (Directory.Exists(text) && Directory.Exists(text3) && Directory.Exists(text2))
				{
					M2UnmBridgeAPI_Init(text, text2, text3);
				}
			}
			else if (Directory.Exists(text4))
			{
				M2UnmBridgeAPI_Init(text4, text4, text4);
			}
			AspenPlusEngineSettings = new LegacyAspenSettings();
			PlexDocument = DocumentFactory.Instance.GetDocument(FileNumber, dbConfiguration);
			_FileReporter = new FileReporter(fileNumber);
		}

		public CalculationService(int fileNumber, ILogger ngpLogger = null, IDBConfiguration dbConfiguration = null)
		{
			dBConfiguration = dbConfiguration;
			if (ngpLogger != null)
			{
				NGPLogger = ngpLogger;
			}
			else
			{
				NGPLogger = NullLogger.Instance;
			}
			NGPLogger.LogInformation("Start Base CalculationService Constructor for file number {fileNumber}", fileNumber.ToString());
			FileNumber = fileNumber;
			string text = string.Empty;
			string text2 = string.Empty;
			string text3 = string.Empty;
			string text4 = string.Empty;
			bool flag = Environment.OSVersion.Platform == PlatformID.Win32NT;
			if (flag)
			{
				text3 = Helper.IFortDirectoryPath();
				NGPLogger.LogInformation("IFORT Path set to: \n      { ifortPath }", text3);
				text = Helper.AprSysDirectoryPath(GlobalSpecs.MM);
				NGPLogger.LogInformation("APRSYS Path set to: \n       { aprsysPath }", text);
				text2 = Helper.OomfDirectoryPath(GlobalSpecs.MM);
				NGPLogger.LogInformation("OOMF Path set to: \n      { oomfPath }", text2);
				if (!Directory.Exists(text))
				{
					NGPLogger.LogCritical("Required directory not found: \n       " + text);
					throw new DirectoryNotFoundException("Required directory not found: " + text);
				}
				if (!Directory.Exists(text2))
				{
					throw new DirectoryNotFoundException("Required directory not found: " + text2);
				}
				if (!Directory.Exists(text3))
				{
					NGPLogger.LogCritical("Required directory not found: \n       " + text3);
					throw new DirectoryNotFoundException("Required directory not found: " + text3);
				}
			}
			else
			{
				text4 = Path.GetDirectoryName(Assembly.GetExecutingAssembly().Location);
			}
			if (flag)
			{
				if (Directory.Exists(text) && Directory.Exists(text3) && Directory.Exists(text2))
				{
					int num = M2UnmBridgeAPI_Init(text, text2, text3);
					NGPLogger.LogInformation("Load ManagedToUnManagedBridge with status: " + num);
					Console.WriteLine("The last Win32 Error was: " + Marshal.GetLastWin32Error());
				}
			}
			else if (Directory.Exists(text4))
			{
				M2UnmBridgeAPI_Init(text4, text4, text4);
			}
			NGPLogger.LogInformation("Load Settings");
			AspenPlusEngineSettings = new LegacyAspenSettings();
			NGPLogger.LogInformation("Load Document");
			PlexDocument = DocumentFactory.Instance.GetDocument(FileNumber);
			NGPLogger.LogInformation("Load FileReporter");
			_FileReporter = new FileReporter(fileNumber);
			NGPLogger.LogInformation("End Base CalculationService Constructor");
		}

		public void Initialize()
		{
			long num = M2UnmBridgeAPI_GetNCC();
			Console.WriteLine("M2UnmBridgeAPI_GetNCC:" + num);
			if ((BeadStatus.Modified == Status || num > 0) && Status != BeadStatus.Uninitialized && Status != BeadStatus.Ready)
			{
				Console.WriteLine("old engine release from Initialize");
				M2UnmBridgeAPI_ClosePDF();
			}
			if (Status != BeadStatus.Ready)
			{
				_FileReporter.Initialize(RunId, OutputDirectory);
				PrintComponentsTable(PrintToFile.ReportAndHistory);
				Console.WriteLine("before Generate Beads");
				PlexDocument.GenerateBeads();
				Console.WriteLine("After Generate Beads");
				M2UnmBridgeAPI_SetNCC(ComponentManager.Count);
				Console.WriteLine("After 2UnmBridgeAPI_SetNCC");
				M2UnmBridgeAPI_GetRetentionSizes(out var NRETN, out var NIRETN, out var MRL, out var MRS, out var MRD);
				Console.WriteLine("GetRetentionSizes" + NRETN + NIRETN + MRL + MRS);
				FlashRetenSize = new FlashRetentionSize(NRETN, NIRETN, MRL, MRS, MRD);
				Console.WriteLine("Before M2UnmBridgeAPI_Initialize");
				M2UnmBridgeAPI_Initialize(RunId, FileReporter.HistoryFile, FileReporter.ReportFile, -_FileReporter.HistoryChannel, -_FileReporter.ReportChannel);
				Console.WriteLine("After M2UnmBridgeAPI_Initialize");
			}
		}

		internal void CheckStatus()
		{
			if (BeadStatus.Ready != Status)
			{
				if (Status != BeadStatus.Uninitialized && BeadStatus.Modified != Status)
				{
					throw new NotImplementedException($"Unknown managed PLEX status {Status}.");
				}
				Initialize();
			}
		}

		protected virtual void Dispose(bool disposing)
		{
			if (!AlreadyDisposed)
			{
				ReleaseUnmanagedResources();
				if (disposing)
				{
					ReleaseManagedResources();
				}
				AlreadyDisposed = true;
			}
		}

		private void CheckDimension(IList<int> componentIDX, params IList<double>[] composition)
		{
			CheckDimension(componentIDX);
			for (int i = 0; i < composition.Length; i++)
			{
				if (composition[i].Count != componentIDX.Count)
				{
					NGPLogger.LogWarning("Composition and IDX must have equal dimensions.");
				}
			}
		}

		private void CheckDimension(IList<int> componentIDX)
		{
			if (ComponentManager.Count < componentIDX.Count)
			{
				NGPLogger.LogWarning("IDX size cannot be larger than the number of components.");
			}
		}

		private KeySinglePhase KeyPhase(ValidPhases validPhases)
		{
			if (ValidPhases.VaporLiquid == validPhases || ValidPhases.VaporLiquidLiquid == validPhases || ValidPhases.VaporLiquidFreeWater == validPhases || ValidPhases.VaporLiquidDirtyWater == validPhases)
			{
				return KeySinglePhase.Ignore;
			}
			if (ValidPhases.LiquidLiquid == validPhases)
			{
				return KeySinglePhase.NoVapor;
			}
			if (ValidPhases.LiquidOnly == validPhases)
			{
				return KeySinglePhase.Liquid;
			}
			if (ValidPhases.VaporOnly == validPhases)
			{
				return KeySinglePhase.Vapor;
			}
			throw new ArgumentException($"Unknown value for {"ValidPhases"}: {validPhases}?");
		}

		private int PhaseNumber(ValidPhases validPhases)
		{
			if (ValidPhases.VaporLiquid == validPhases || ValidPhases.LiquidLiquid == validPhases)
			{
				return 2;
			}
			if (ValidPhases.VaporLiquidLiquid == validPhases)
			{
				return 3;
			}
			if (ValidPhases.LiquidOnly == validPhases || ValidPhases.VaporOnly == validPhases)
			{
				return 1;
			}
			if (ValidPhases.VaporLiquidFreeWater == validPhases)
			{
				return 12;
			}
			if (ValidPhases.VaporLiquidDirtyWater == validPhases)
			{
				return 22;
			}
			throw new ArgumentException("PhaseNumber", $"Unknown valid phase definition = {validPhases}");
		}

		private void PrintComponentsTable(PrintToFile printTo)
		{
			string text = string.Format("{0,-11}{1,-15}{2,-15}{3,-35}", "ID", "Type", "Alias", "Name");
			FileReporter.WriteLine(printTo, Environment.NewLine + " " + text);
			foreach (IComponent component in ComponentManager.Components)
			{
				text = $"{component.EngineId,-11}{component.ComponentType,-15}{component.Descriptor,-15}{component.Ids.Name,-35}";
				FileReporter.WriteLine(printTo, " " + text);
			}
			FileReporter.WriteLine(printTo, new string('_', FileReporter.PageWidth));
		}

		public void SetDataSources(List<string> sources)
		{
			List<IDataSource> list = new List<IDataSource>();
			foreach (string source in sources)
			{
				string[] array = source.Trim().ToUpper().Split(' ');
				if (array.Length == 2)
				{
					list.Add(Factories.CreateDataSource(array[0], array[1]));
				}
			}
			if (list.Count > 0)
			{
				ParameterManager.DataSources.AddRange(list);
			}
		}

		public void SetComponents(List<string> components)
		{
			List<IComponent> list = new List<IComponent>();
			foreach (string component in components)
			{
				string[] array = component.Trim().ToUpper().Split(' ');
				string text = array[0];
				string text2 = ((array.Length == 2) ? array[1] : text);
				list.Add(Factories.CreateComponentByAlias(text2, text));
			}
			if (list.Count > 0)
			{
				ComponentManager.Components.AddRange(list);
			}
		}

						break;
					}
				}
			}
			if (list.Count > 0)
			{
				ComponentManager.CreateHenryList(henryid, list);
			}
		}

		private void Add_MoleDensity(ref IFlashResultPhaseProps results)
		{
			results.RhoV = RMISS;
			results.RhoL = RMISS;
			results.RhoL1 = RMISS;
			results.RhoL2 = RMISS;
			if (results.VMXV > 0.0 && results.VMXV != RMISS)
			{
				results.RhoV = 1.0 / results.VMXV;
			}
			if (results.VMXL > 0.0 && results.VMXL != RMISS)
			{
				results.RhoL = 1.0 / results.VMXL;
			}
			if (results.Beta < 1.0)
			{
				if (results.VMXL1 > 0.0 && results.VMXL1 != RMISS)
				{
					results.RhoL1 = 1.0 / results.VMXL1;
				}
				if (results.VMXL2 > 0.0 && results.VMXL2 != RMISS)
				{
					results.RhoL2 = 1.0 / results.VMXL2;
				}
			}
		}

		private void Add_MassDensity(ref IFlashResultPhaseProps results)
		{
			results.MassRhoV = RMISS;
			results.MassRhoL = RMISS;
			results.MassRhoL1 = RMISS;
			results.MassRhoL2 = RMISS;
			if (results.VMXV > 0.0 && results.VMXV != RMISS)
			{
				results.MassRhoV = results.MWMXV / results.VMXV;
			}
			if (results.VMXL > 0.0 && results.VMXL != RMISS)
			{
				results.MassRhoL = results.MWMXL / results.VMXL;
			}
			if (results.Beta < 1.0)
			{
				if (results.VMXL1 > 0.0 && results.VMXL1 != RMISS)
				{
					results.MassRhoL1 = results.MWMXL1 / results.VMXL1;
				}
				if (results.VMXL2 > 0.0 && results.VMXL2 != RMISS)
				{
					results.MassRhoL2 = results.MWMXL2 / results.VMXL2;
				}
			}
		}

		private void Add_SpecificHeat(ref IFlashResultPhaseProps results)
		{
			EnthalpyVapor(results, out var propertyValue, out var temperatureDerivative);
			results.CpV = temperatureDerivative;
			EnthalpyLiquid(results, out propertyValue, out temperatureDerivative);
			results.CpL = temperatureDerivative;
			if (results.Beta < 1.0)
			{
				EnthalpyLiquid1(results, out propertyValue, out temperatureDerivative);
				results.CpL1 = temperatureDerivative;
				EnthalpyLiquid2(results, out propertyValue, out temperatureDerivative);
				results.CpL2 = temperatureDerivative;
			}
		}

	public static class CalculationServiceFactory
	{
		private static readonly object _lockingObject = new object();

		private static readonly Dictionary<int, ICalculationService> CalcServiceById = new Dictionary<int, ICalculationService>();

		private static CalculationService Instantiater(int id, IDBConfiguration dbConfiguration = null)
		{
			return new CalculationService(id, dbConfiguration);
		}

		private static CalculationService Instantiater(int id, ILogger ngpLogger, IDBConfiguration dbConfiguration = null)
		{
			return new CalculationService(id, ngpLogger, dbConfiguration);
		}

		public static ICalculationService GetOrCreate(int fileNumber, IDBConfiguration dbConfiguration = null)
		{
			lock (_lockingObject)
			{
				if (CalcServiceById.TryGetValue(fileNumber, out var value))
				{
					return value;
				}
				CalcServiceById.Add(fileNumber, Instantiater(fileNumber, dbConfiguration));
				return CalcServiceById[fileNumber];
			}
		}

		public static ICalculationService CreateOrOverwrite(int fileNumber, IDBConfiguration dbConfiguration = null)
		{
			lock (_lockingObject)
			{
				if (CalcServiceById.TryGetValue(fileNumber, out var _))
				{
					CalcServiceById[fileNumber] = Instantiater(fileNumber, dbConfiguration);
					return CalcServiceById[fileNumber];
				}
				CalcServiceById.Add(fileNumber, Instantiater(fileNumber, dbConfiguration));
				return CalcServiceById[fileNumber];
			}
		}

		public static ICalculationService GetOrCreate(int fileNumber, ILogger ngpLogger, IDBConfiguration dbConfiguration = null)
		{
			lock (_lockingObject)
			{
				if (CalcServiceById.TryGetValue(fileNumber, out var value))
				{
					return value;
				}
				CalcServiceById.Add(fileNumber, Instantiater(fileNumber, ngpLogger, dbConfiguration));
				return CalcServiceById[fileNumber];
			}
		}

		internal static void Remove(int fileNumber)
		{
			CalcServiceById.Remove(fileNumber);
		}

		public static void Reset()
		{
			lock (_lockingObject)
			{
				CalcServiceById.Clear();
			}
		}

		public void WriteLine(PrintToFile printTo, string message)
		{
			CheckFiles();
			List<int> list = new List<int>();
			List<string> list2 = new List<string>();
			switch (printTo)
			{
			case PrintToFile.HistoryOnly:
				list.Add(HistoryChannel);
				list2.Add(HistoryFile);
				break;
			case PrintToFile.ReportOnly:
				list.Add(ReportChannel);
				list2.Add(ReportFile);
				break;
			case PrintToFile.ReportAndHistory:
				list.Add(ReportChannel);
				list.Add(HistoryChannel);
				list2.Add(ReportFile);
				list2.Add(HistoryFile);
				break;
			default:
				throw new NotImplementedException($"Unhandled print-to-file option {printTo}.");
			case PrintToFile.None:
				break;
			}
			for (int i = 0; i < list.Count; i++)
			{
				try
				{
					using StreamWriter streamWriter = new StreamWriter(list2[i], append: true);
					streamWriter.WriteLine(message);
					streamWriter.Flush();
					streamWriter.Close();
				}
				catch
				{
					try
					{
						M2UnmBridgeAPI_WriteLineToFile(list[i], message);
					}
					catch (Exception)
					{
						throw;
					}
				}
			}
		}

		internal void Initialize(string runId, string outputDirectory)
		{
			RunId = runId;
			HistoryChannel = 9000 + (FileNumber - 1) * 2;
			ReportChannel = HistoryChannel + 1;
			try
			{
				Directory.CreateDirectory(outputDirectory);
				OutputDirectory = outputDirectory;
				HistoryFile = Path.Combine(OutputDirectory, RunId + ".his");
				ReportFile = Path.Combine(OutputDirectory, RunId + ".rep");
				File.Create(HistoryFile).Close();
				File.Create(ReportFile).Close();
			}
			catch (IOException)
			{
			}
			catch (Exception)
			{
				throw;
			}
			Initialized = true;
			PrintFileHeader(PrintToFile.ReportAndHistory);
		}

		private void CheckFiles()
		{
			if (!Initialized)
			{
				throw new Exception("You must initialize before writing to a file.");
			}
		}

		private void PrintFileHeader(PrintToFile printTo)
		{
			string text = "Aspen Properties Component Service (PCS)";
			int num = (int)Math.Round((double)(80 - text.Length) * 0.5, MidpointRounding.AwayFromZero);
			int count = 80 - num - text.Length - 2;
			string message = "+" + new string('=', 78) + "+";
			string message2 = "|" + new string(' ', 78) + "|";
			string message3 = "|" + new string(' ', num) + text + new string(' ', count) + "|";
			WriteLine(printTo, message);
			WriteLine(printTo, message2);
			WriteLine(printTo, message3);
			WriteLine(printTo, message2);
			WriteLine(printTo, message);
		}
