			LogOptionSets();
		}

		public void ReadFile(string fullName, IDatabaseManagementLog dblogger = null)
		{
			if (dblogger == null)
			{
				dblogger.Logger = NullLogger.Instance;
			}
			else
			{
				DBLogger = dblogger;
			}
			Reset();
			Logger.LogInformation("Begin reading input file at { Time }\n" + C_TAB + fullName, DateTime.UtcNow);
			if (!File.Exists(fullName))
			{
				Logger.LogCritical(new FileNotFoundException(fullName), "Unable to read file at {Time}", DateTime.UtcNow);
				return;
			}
			if (Path.GetExtension(fullName).ToLower() == ".inp")
			{
				Lines = File.ReadAllLines(fullName);
			}
			else
			{
				if (!(Path.GetExtension(fullName).ToLower() == ".cfg"))
				{
					Logger.LogCritical(new FileLoadException("Expecting *.INP file type.", fullName), "Invalid file type at {Time}", DateTime.UtcNow);
					return;
				}
				Lines = File.ReadAllLines(fullName);
				aprinpFile = GetAPRINPFile(fullName);
				Lines = File.ReadAllLines(aprinpFile);
			}
			ReadUnitSystem();
			ReadDataSources();
			ReadComponents();
			ReadOptionSets();
			ReadMaterialStreams();
		}

		public void Run(ICalculationService API)
		{
			if (API == null)
			{
				Logger.LogError("Cannot calculate streams because API is null at { Time }", DateTime.UtcNow);
				return;
			}
			foreach (MaterialStream materialStream in MaterialStreams)
			{
				materialStream.Run(API);
				string text = "";
				if (materialStream.FlashResults == null)
				{
					continue;
				}
				text += $"\n{C_TAB}TEMP = {materialStream.FlashResults.Temperature} K";
				if (materialStream.Temperature != null)
				{
					text += " (!)";
				}
				text += $"\n{C_TAB}PRES = {materialStream.FlashResults.Pressure} Pa";
				if (materialStream.Pressure != null)
				{
					text += " (!)";
				}
				text += $"\n{C_TAB}VFRAC = {materialStream.FlashResults.VaporFraction}";
				if (materialStream.VaporFraction != null)
				{
					text += " (!)";
				}
				text = text + "\n" + C_TAB + "Feed composition:";
				for (int i = 0; i < materialStream.Components.Count; i++)
				{
					text += $"\n{C_TAB}  {materialStream.Components[i].UserSuppliedName} = {materialStream.Composition[i]}";
				}
				if (materialStream.FlashResults.Beta == 1.0)
				{
					text = text + "\n" + C_TAB + "Liquid phase:";
					for (int j = 0; j < materialStream.Components.Count; j++)
					{
						text += $"\n{C_TAB}  {materialStream.Components[j].UserSuppliedName} = {materialStream.FlashResults.X[j]}";
					}
				}
				else
				{
					text = text + "\n" + C_TAB + "Liquid phase 1:";
					for (int k = 0; k < materialStream.Components.Count; k++)
					{
						text += $"\n{C_TAB}  {materialStream.Components[k].UserSuppliedName} = {materialStream.FlashResults.X1[k]}";
					}
					text = text + "\n" + C_TAB + "Liquid phase 2s:";
					for (int l = 0; l < materialStream.Components.Count; l++)
					{
						text += $"\n{C_TAB}  {materialStream.Components[l].UserSuppliedName} = {materialStream.FlashResults.X2[l]}";
					}
				}
				if (materialStream.FlashResults.VaporFraction < 1.0)
				{
					text = text + "\n" + C_TAB + "Vapor phase:";
					for (int m = 0; m < materialStream.Components.Count; m++)
					{
						text += $"\n{C_TAB}  {materialStream.Components[m].UserSuppliedName} = {materialStream.FlashResults.Y[m]}";
					}
				}
				text = text + "\n" + C_TAB + "HMX = " + materialStream.FlashResults.Enthalpy + " J/kmol";
				Logger.LogInformation("Stream results for " + materialStream.Name + " at { Time }" + text, DateTime.UtcNow);
			}
		}

		private string GetSectionBuffer(string section)
		{
			for (int i = 0; i < Lines.Length; i++)
			{
				string text = Lines[i];
				if (!text.StartsWith(section, StringComparison.OrdinalIgnoreCase))
				{
					continue;
				}
				string text2 = text;
				for (int j = ++i; j < Lines.Length; j++)
				{
					text = Lines[j];
					if (text.StartsWith(";") || string.IsNullOrWhiteSpace(text))
					{
						break;
					}
					text2 += text;
				}
				RegexOptions options = RegexOptions.None;
				Regex regex = new Regex("[ ]{2,}", options);
				text2 = text2.Substring(section.Length, text2.Length - section.Length).Replace('&', ' ').Replace('\'', ' ');
				return regex.Replace(text2, " ").Trim();
			}
			return string.Empty;
		}

		private void ReadComponents()
		{
			foreach (string item in GetSectionBuffer("COMPONENTS ").Split(new string[1] { " / " }, StringSplitOptions.RemoveEmptyEntries).ToList())
			{
				string text = "";
				string text2 = "";
				switch (item.NumberOfWords())
				{
				case 1:
					text = item.Split()[0];
					Components.Add(Factories.CreateHypoComponent(text));
					break;
				case 2:
				case 3:
					text = item.Split()[0];
					text2 = item.Split()[1];
					Components.Add(Factories.CreateComponentByAlias(text2, text));
					break;
				}
			}
		}

		private void ReadDataSources()
		{
			foreach (string item in GetSectionBuffer("PROP-SOURCES ").Split(new string[1] { " / " }, StringSplitOptions.RemoveEmptyEntries).ToList())
			{
				IDatabank databank = null;
				switch (item.NumberOfWords())
				{
				case 1:
					databank = Helper.FindDatabank(item.Split()[0]);
					DBLogger.Logger.LogTrace("Found databank {databankName} (Id: {Id}) within database {databaseName}. \n{ROW2}{databaseVersion}\n{ROW3}{databaseSource}", databank.Name, databank.Id.ToString(), databank.Database.Alias, C_TAB, databank.Database.DataVersion, C_TAB, databank.Database.DataSource);
					break;
				case 2:
					databank = Helper.FindDatabank(item.Split()[1]);
					DBLogger.Logger.LogTrace("Found databank {databankName} (Id: {Id}) within database {databaseName}. \n{ROW2}{databaseVersion}\n{ROW3}{databaseSource}", databank.Name, databank.Id.ToString(), databank.Database.Alias, C_TAB, databank.Database.DataVersion, C_TAB, databank.Database.DataSource);
					break;
				default:
					DBLogger.Logger.LogError("Unable to parse databank at { Time }\n" + C_TAB + item, DateTime.UtcNow);
					break;
				}
				if (databank != null)
				{
					DataSources.Add(Factories.CreateDataSource(databank.Database.Alias, databank.Name));
					continue;
				}
				Logger.LogError("Unknown databank at { Time }\n" + C_TAB + item, DateTime.UtcNow);
			}
			if (DataSources.Count != 0)
			{
				return;
			}
			List<string> list = new List<string>
			{
				"PURE" + GlobalSpecs.MMPack,
				"AQUEOUS",
				"SOLIDS",
				"INORGANIC",
				"AP-EOS",
				"NIST-TRC"
			};
			DBLogger.Logger.LogWarning("No databanks found, will look for default databanks {defaultDBs}", list.ToString());
			foreach (string item2 in list)
			{
				IDatabank databank2 = Helper.FindDatabank(item2);
				if (databank2 != null)
				{
					DBLogger.Logger.LogTrace("Found databank {databankName} (Id: {Id}) within database {databaseName}. \n{ROW2}{databaseVersion}\n{ROW3}{databaseSource}", databank2.Name, databank2.Id.ToString(), databank2.Database.Alias, C_TAB, databank2.Database.DataVersion, C_TAB, databank2.Database.DataSource);
					DataSources.Add(Factories.CreateDataSource(databank2.Database.Alias, databank2.Name));
				}
				else
				{
					Logger.LogError("Unknown databank at { Time }\n" + C_TAB + item2, DateTime.UtcNow);
				}
			}
		}

		private void ReadOptionSets()
		{
			foreach (string item in GetSectionBuffer("PROPERTIES ").Replace("PROPERTIES ", " ").Replace(" / ", " ").Split(new string[1] { " " }, StringSplitOptions.RemoveEmptyEntries)
				.ToList())
			{
				try
				{
					IOptionSet orCreate = Factories.OptionSet.GetOrCreate(item);
					if (orCreate != null)
					{
						OptionSets.Add(orCreate);
					}
				}
				catch
				{
					Logger.LogError("Unknown option set at { Time }\n" + C_TAB + item, DateTime.UtcNow);
				}
			}
		}

		private void ReadMaterialStreams()
		{
			for (int i = 0; i < Lines.Length; i++)
			{
				string text = Lines[i].Replace("  ", " ").Trim();
				string text2 = text;
				if (text2.StartsWith("FLOWSHEET"))
				{
					do
					{
						text = Lines[++i].Trim();
					}
					while (!string.IsNullOrWhiteSpace(text));
					continue;
				}
				if (text2.StartsWith("HIERARCHY "))
				{
					Logger.LogWarning("\n***  INFORMATION IN HIERARCHIES IS IGNORED AT THIS TIME  ***\n");
					break;
				}
				if (!text.StartsWith("STREAM ", StringComparison.OrdinalIgnoreCase))
				{
					continue;
				}
				do
				{
					text = Lines[++i].Replace("  ", " ").Replace(" <", "=").Replace("> ", " ")
						.Trim();
					if (text.StartsWith(";"))
					{
						break;
					}
					text2 = text2 + " " + text;
				}
				while (!string.IsNullOrWhiteSpace(text));
				string[] array = text2.Replace("/", " ").Split(new string[1] { " " }, StringSplitOptions.RemoveEmptyEntries);
				try
				{
					string text3 = array[1];
					_ = array[3];
					string[] array2 = array[4].Split('=');
					string[] array3 = array[5].Split('=');
					string compositionBasis = array[6];
					Dictionary<IComponent, double> dictionary = new Dictionary<IComponent, double>();
					if (array.Length > 6 && (array.Length - 5) % 2 == 0)
					{
						for (int j = 7; j < array.Length; j++)
						{
							foreach (IComponent component in Components)
							{
								if (string.Equals(array[j], component.UserSuppliedName, StringComparison.OrdinalIgnoreCase) && double.TryParse(array[++j], out var result))
								{
									dictionary.Add(component, result);
								}
							}
						}
					}
					IQuantity quantity = null;
					IQuantity quantity2 = null;
					if (array2.Length == 2)
					{
						string quantityName = array2[0];
						double value = double.Parse(array2[1]);
						TryGetQuantity(quantityName, value, out quantity);
					}
					else if (array2.Length == 3)
					{
						string quantityName2 = array2[0];
						double value2 = double.Parse(array2[1]);
						string unitSymbol = array2[2];
						TryGetQuantity(quantityName2, value2, unitSymbol, out quantity);
					}
					if (array3.Length == 2)
					{
						string quantityName3 = array3[0];
						double value3 = double.Parse(array3[1]);
						TryGetQuantity(quantityName3, value3, out quantity2);
					}
					else if (array3.Length == 3)
					{
						string quantityName4 = array3[0];
						double value4 = double.Parse(array3[1]);
						string unitSymbol2 = array3[2];
						TryGetQuantity(quantityName4, value4, unitSymbol2, out quantity2);
					}
					if (quantity != null && quantity2 != null)
					{
						MaterialStreams.Add(new MaterialStream(text3, quantity, quantity2, compositionBasis, dictionary));
						continue;
					}
					Logger.LogError("Incomplete stream specs at { Time }\n" + C_TAB + "STREAM " + text3, DateTime.UtcNow);
				}
				catch (Exception ex)
				{
					Logger.LogCritical("Unknown exception reading streams at { Time }\n" + C_TAB + ex.Message, DateTime.UtcNow);
					throw;
				}
			}
		}

		private QuantityInfo GetQuantityInfo(string quantityName)
		{
			CheckQuantityName(ref quantityName);
			return ((IEnumerable<QuantityInfo>)Quantity.Infos).FirstOrDefault((Func<QuantityInfo, bool>)((QuantityInfo info) => info.Name.Equals(quantityName, StringComparison.OrdinalIgnoreCase)));
		}

		private bool TryGetQuantity(string quantityName, double value, out IQuantity quantity)
		{
			//IL_0047: Unknown result type (might be due to invalid IL or missing references)
			//IL_0013: Unknown result type (might be due to invalid IL or missing references)
			quantity = null;
			QuantityInfo quantityInfo = GetQuantityInfo(quantityName);
			if (quantityInfo == null)
			{
				return false;
			}
			try
			{
				quantity = Quantity.From(QuantityValue.op_Implicit(value), UnitByQuantityName[quantityInfo.ValueType.Name]);
			}
			catch (Exception ex)
			{
				if (!quantityName.Equals("VFRAC", StringComparison.OrdinalIgnoreCase))
				{
					Logger.LogCritical("Unknown exception getting quantity at { Time }\n" + C_TAB + ex.Message, DateTime.UtcNow);
					throw;
				}
				quantity = Quantity.From(QuantityValue.op_Implicit(value), (Enum)(object)(ScalarUnit)1);
			}
			return quantity != null;
		}

		private bool TryGetQuantity(string quantityName, double value, string unitSymbol, out IQuantity quantity)
		{
			quantity = null;
			QuantityInfo quantityInfo = GetQuantityInfo(quantityName);
			if (quantityInfo == null)
			{
				return false;
			}
			Enum obj = UnitsNetSetup.Default.UnitParser.Parse(unitSymbol, (quantityInfo != null) ? quantityInfo.UnitType : null, (IFormatProvider)null);
			Quantity.TryFrom(value, obj, ref quantity);
			return quantity != null;
		}

		private void ReadUnitSystem()
		{
			//IL_0169: Unknown result type (might be due to invalid IL or missing references)
			//IL_016f: Expected O, but got Unknown
			//IL_0222: Unknown result type (might be due to invalid IL or missing references)
			//IL_022c: Expected O, but got Unknown
			//IL_01b8: Unknown result type (might be due to invalid IL or missing references)
			//IL_01be: Expected O, but got Unknown
			//IL_0204: Unknown result type (might be due to invalid IL or missing references)
			//IL_020a: Expected O, but got Unknown
			List<string> list = GetSectionBuffer("IN-UNITS ").Replace("= ", "=").Split(new string[1] { " " }, StringSplitOptions.RemoveEmptyEntries).ToList();
			BaseUnitSet = "SI";
			StringDictionary stringDictionary = new StringDictionary();
			if (list.Count > 0)
			{
				BaseUnitSet = list[0].ToUpper();
				for (int i = 1; i < list.Count; i++)
				{
					string[] array = list[i].Split('=');
					if (array.Length == 2)
					{
						string key = array[0];
						string value = array[1];
						stringDictionary.Add(key, value);
					}
				}
			}
			BaseUnits val = null;
			UnitByQuantityName = new Dictionary<string, Enum>(StringComparer.OrdinalIgnoreCase);
			CultureInfo cultureInfo = new CultureInfo("en-US");
			UnitsNetSetup.Default.UnitAbbreviations.MapUnitToDefaultAbbreviation<TemperatureUnit>((TemperatureUnit)3, (IFormatProvider)cultureInfo, "F");
			UnitsNetSetup.Default.UnitAbbreviations.MapUnitToDefaultAbbreviation<TemperatureUnit>((TemperatureUnit)1, (IFormatProvider)cultureInfo, "C");
			UnitsNetSetup.Default.UnitAbbreviations.MapUnitToDefaultAbbreviation<TemperatureUnit>((TemperatureUnit)5, (IFormatProvider)cultureInfo, "R");
			switch (BaseUnitSet)
			{
			case "SI":
				val = new BaseUnits((LengthUnit?)(LengthUnit)21, (MassUnit?)(MassUnit)8, (DurationUnit?)(DurationUnit)9, (ElectricCurrentUnit?)(ElectricCurrentUnit)1, (TemperatureUnit?)(TemperatureUnit)8, (AmountOfSubstanceUnit?)(AmountOfSubstanceUnit)5, (LuminousIntensityUnit?)(LuminousIntensityUnit)1);
				UnitByQuantityName.Add("pressure", (Enum)(object)(PressureUnit)36);
				break;
			case "ENG":
				val = new BaseUnits((LengthUnit?)(LengthUnit)11, (MassUnit?)(MassUnit)19, (DurationUnit?)(DurationUnit)2, (ElectricCurrentUnit?)(ElectricCurrentUnit)6, (TemperatureUnit?)(TemperatureUnit)3, (AmountOfSubstanceUnit?)(AmountOfSubstanceUnit)15, (LuminousIntensityUnit?)(LuminousIntensityUnit)1);
				UnitByQuantityName.Add("pressure", (Enum)(object)(PressureUnit)40);
				break;
			case "MET":
				val = new BaseUnits((LengthUnit?)(LengthUnit)11, (MassUnit?)(MassUnit)19, (DurationUnit?)(DurationUnit)2, (ElectricCurrentUnit?)(ElectricCurrentUnit)6, (TemperatureUnit?)(TemperatureUnit)3, (AmountOfSubstanceUnit?)(AmountOfSubstanceUnit)15, (LuminousIntensityUnit?)(LuminousIntensityUnit)1);
				UnitByQuantityName.Add("pressure", (Enum)(object)(PressureUnit)1);
				break;
			}
			UnitSystem = new UnitSystem(val);
			UnitByQuantityName.Add("length", (Enum)(object)val.Length);
			UnitByQuantityName.Add("mass", (Enum)(object)val.Mass);
			UnitByQuantityName.Add("time", (Enum)(object)val.Time);
			UnitByQuantityName.Add("current", (Enum)(object)val.Current);
			UnitByQuantityName.Add("temperature", (Enum)(object)val.Temperature);
			UnitByQuantityName.Add("moles", (Enum)(object)val.Amount);
			foreach (DictionaryEntry item in stringDictionary)
			{
				string text = (string)item.Key;
				string text2 = (string)item.Value;
				if (TryParseUnit(text, text2, out var unitValue))
				{
					if (!UnitByQuantityName.ContainsKey(text))
					{
						UnitByQuantityName.Add(text, unitValue);
					}
					else
					{
						UnitByQuantityName[text] = unitValue;
					}
					continue;
				}
				Logger.LogWarning("Unknown quantity-unit pair at { Time }" + $"\n{C_TAB}{text}: {text2}", DateTime.UtcNow);
			}
		}

		private bool TryParseUnit(string quantityName, string unit, out Enum unitValue)
		{
			unitValue = null;
			CheckQuantityName(ref quantityName);
			QuantityInfo? obj = ((IEnumerable<QuantityInfo>)Quantity.Infos).FirstOrDefault((Func<QuantityInfo, bool>)((QuantityInfo info) => info.Name.Equals(quantityName, StringComparison.OrdinalIgnoreCase)));
			Type type = ((obj != null) ? obj.UnitType : null);
			if (type != null)
			{
				unitValue = UnitsNetSetup.Default.UnitParser.Parse(unit, type, (IFormatProvider)null);
				return true;
			}
			return false;
		}

		private void CheckQuantityName(ref string quantityName)
		{
			quantityName.Trim().ToUpper();
			if (quantityName.Equals("TEMP", StringComparison.OrdinalIgnoreCase))
			{
				quantityName = "Temperature";
			}
			else if (quantityName.Equals("PRES", StringComparison.OrdinalIgnoreCase))
			{
				quantityName = "Pressure";
			}
			else if (quantityName.Equals("VFRAC", StringComparison.OrdinalIgnoreCase))
			{
				quantityName = "Scalar";
			}
		}

		private string GetAPRINPFile(string FullName)
		{
			aprinpFile = null;
			string sectionBuffer = GetSectionBuffer("APRINP ");
			sectionBuffer = sectionBuffer.Trim('"');
			if (sectionBuffer == null)
			{
				return aprinpFile;
			}
			string directoryName = Path.GetDirectoryName(FullName);
			aprinpFile = Path.Combine(directoryName, sectionBuffer);
			return aprinpFile;
		}

		private void Reset()

			Lines = null;
			BaseUnitSet = string.Empty;
			UnitSystem = null;
			UnitByQuantityName = null;

		private void LogComponents()
		{
			if (Components.Count > 0)
			{
				List<string> values = Components.Select((IComponent c) => c.ToString()).ToList();
				LogValues("COMPONENTS cid [cname]", values);
			}
		}

		private void LogDataSources(bool defaultSources = false)
		{
			if (DataSources.Count > 0)
			{
				List<string> values = DataSources.Select((IDataSource c) => c.ToString()).ToList();
				if (!defaultSources)
				{
					LogValues("PROP-SOURCES dsname", values);
				}
				else
				{
					LogValues("Input contains no PROP-SOURCES, use default values", values, LogLevel.Warning);
				}
			}
		}

		private void LogOptionSets()
		{
			if (OptionSets.Count > 0)
			{
				List<string> values = OptionSets.Select((IOptionSet c) => c?.ToString()).ToList();
				LogValues("PROPERTIES opsetname", values);
			}
		}

		private void LogUnitSystem()
		{
			string text = "";
			foreach (KeyValuePair<string, Enum> item in UnitByQuantityName)
			{
				text += $"\n{C_TAB}{item.Key}: {item.Value}";
			}
			Logger.LogInformation("IN-UNITS " + BaseUnitSet + text);
		}

		private void LogValues(string message, List<string> values, LogLevel logLevel = LogLevel.Information)
		{
			int num = 0;
			for (int i = 0; i < values.Count; i++)
			{
				num = i + 1;
				message += $"\n{C_TAB}[{num}] {values[i]}";
			}
			switch (logLevel)
			{
			case LogLevel.Information:
				Logger.LogInformation(message);
				break;
			case LogLevel.Warning:
				Logger.LogWarning(message);
				break;
			default:
				Logger.LogError("[LogLevel] value not understood by method.");
				break;
			}
		}
