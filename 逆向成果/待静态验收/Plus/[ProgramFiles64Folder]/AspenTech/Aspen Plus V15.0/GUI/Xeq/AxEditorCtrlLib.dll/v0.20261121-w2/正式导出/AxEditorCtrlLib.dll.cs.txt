[ClassInterface(ClassInterfaceType.None)]
public class AxEditorCtrlEventMulticaster : _DEditorCtrlEvents
{
	private AxEditorCtrl parent;

	public AxEditorCtrlEventMulticaster(AxEditorCtrl parent)
	{
		this.parent = parent;
	}

	public virtual void ContextMenuOpening()
	{
		EventArgs e = new EventArgs();
		parent.RaiseOnContextMenuOpening(parent, e);
	}
