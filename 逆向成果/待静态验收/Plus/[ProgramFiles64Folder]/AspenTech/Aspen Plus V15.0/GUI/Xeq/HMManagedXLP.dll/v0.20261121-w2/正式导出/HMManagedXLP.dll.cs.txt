public class XlpEquationEnumerator : IEnumerator<XLPEquation>, IEnumerator, IDisposable
{
	private readonly XLPManager _manager;

	private readonly XLPEqnCollection _pEqnList;

	private readonly int _equationCount;

	private readonly byte[] _bEqn;

	private XLPEquation _current;

	private int _currentNumber;

	public XLPEquation Current => _current;

	object IEnumerator.Current => _current;

	public XlpEquationEnumerator(XLPManager manager)
	{
		_manager = manager;
		_pEqnList = manager.GetEquationCollectionIntf();
		_equationCount = _pEqnList.NumberOfEquations();
		_bEqn = new byte[128];
		Reset();
	}

	public bool MoveNext()
	{
		if (_currentNumber >= _equationCount)
		{
			return false;
		}
		StringBuilder strname = new StringBuilder(_bEqn.Length);
		if (_current == null)
		{
			XLPEqnCollection.FrstEquationName(_pEqnList.Ptr, _bEqn);
		}
		else
		{
			XLPEqnCollection.NextEquationName(_pEqnList.Ptr, _bEqn);
		}
		_pEqnList.convert2String(_bEqn, strname, out var stringItem);
		_current = _manager.FindEquationObject(stringItem);
		_currentNumber++;
		return true;
	}

	public void Dispose()
	{
	}

	public void Reset()
	{
		_current = null;
		_currentNumber = 0;
	}
}

	public XLPExpressionTreeNode Constant(double constant)
	{
		return manager.MakeExpressionTreeConstant(this, constant);
	}

	public XLPExpressionTreeNode Variable(string variableName)
	{
		return manager.MakeExpressionTreeVariable(this, variableName);
	}

	public XLPExpressionTreeNode UnaryNode(ExpTreeNodeType type, XLPExpressionTreeNode operand)
	{
		return manager.MakeExpressionTreeUnaryOperator(this, (short)type, operand);

	{
		return manager.MakeExpressionTreeSwitchOperator(this, condition, constant, trueOperand, falseOperand);
	}

	{

		}
		set
		{
			xlpMgrPtr = value;
		}
	}

	public XLPManager(string modelName)
	{
		XLPcreate();
		GetXLPBuilder();
		SetXLPModelName(modelName);
	}

	[DllImport("hmxlpxmps", CallingConvention = CallingConvention.Cdecl)]
	private static extern void NameScreen(IntPtr xlpPtr, string name, byte[] newName);

	public void NameScreen(ref string name)
	{
		byte[] array = new byte[256];
		NameScreen(xlpMgrPtr, name, array);
		if (array != null)
		{
			name = StringGet(array);
		}
		name = name.ToUpper();
	}

	public void ProcessXLPNames()
	{
		XLPVarCollection variableCollectionIntf = GetVariableCollectionIntf();
		int num = variableCollectionIntf.NumberOfVariables();
		byte[] array = new byte[128];
		StringBuilder strname = new StringBuilder(array.Length);
		for (int i = 0; i < num; i++)
		{
			if (i == 0)
			{
				XLPVarCollection.FrstVariableName(variableCollectionIntf.Ptr, array);
			}
			else
			{
				XLPVarCollection.NextVariableName(variableCollectionIntf.Ptr, array);
			}
			variableCollectionIntf.convert2String(array, strname, out var stringItem);
			XLPVariable xLPVariable = FindVariableObject(stringItem);
			if (xLPVariable != null)
			{
				NameScreen(ref stringItem);
				UpdateVariableName(xLPVariable, stringItem);
			}
		}
		XLPEqnCollection equationCollectionIntf = GetEquationCollectionIntf();
		int num2 = equationCollectionIntf.NumberOfEquations();
		byte[] array2 = new byte[128];
		StringBuilder strname2 = new StringBuilder(array2.Length);
		for (int j = 0; j < num2; j++)
		{
			if (j == 0)
			{
				XLPEqnCollection.FrstEquationName(equationCollectionIntf.Ptr, array2);
			}
			else
			{
				XLPEqnCollection.NextEquationName(equationCollectionIntf.Ptr, array2);
			}
			equationCollectionIntf.convert2String(array2, strname2, out var stringItem2);
			XLPEquation xLPEquation = FindEquationObject(stringItem2);
			if (xLPEquation != null)
			{
				NameScreen(ref stringItem2);
				UpdateEquationName(xLPEquation, stringItem2);
			}
		}
	}

	public bool LoadSolutionFile(string fullModelPath, string fileName, XLPManager scdXLP, ArrayList messages)
	{
		CultureInfo currentCulture = Thread.CurrentThread.CurrentCulture;
		try
		{
			CultureInfo currentCulture2 = new CultureInfo("en-US");
			Thread.CurrentThread.CurrentCulture = currentCulture2;
			double num = 1E-05;
			bool flag = false;
			string path = fullModelPath + fileName;
			ArrayList arrayList = new ArrayList();
			if (!File.Exists(path))
			{
				return false;
			}
			using (StreamReader streamReader = new StreamReader(path, Encoding.Default))
			{
				string text = streamReader.ReadLine();
				text = streamReader.ReadLine();
				while ((text = streamReader.ReadLine()) != null && text != "")
				{
					if (text[0] == '*' || text[0] == '$')
					{
						continue;
					}
					int num2 = text.LastIndexOf(' ');
					if (num2 <= 0)
					{
						continue;
					}
					string text2 = text.Substring(0, num2).Trim();
					if (!flag && text2.StartsWith("INVT"))
					{
						flag = true;
					}
					double num3 = Convert.ToDouble(text.Substring(num2 + 1));
					if (Math.Abs(num3) < 1E-06)
					{
						num3 = 0.0;
					}
					XLPVariable xLPVariable = FindVariableObject(text2);
					if (xLPVariable == null && scdXLP != null)
					{
						xLPVariable = scdXLP.FindVariableObject(text2);
						if (xLPVariable != null)
						{
							xLPVariable.Value = num3;
						}
					}
					else
					{
						if (xLPVariable == null)
						{
							continue;
						}
						xLPVariable.Value = num3;
						if (xLPVariable.Value < xLPVariable.LowerBnd)
						{
							if (xLPVariable.Value < xLPVariable.LowerBnd - num && Math.Abs(xLPVariable.Value - xLPVariable.LowerBnd) / (1.0 + Math.Abs(xLPVariable.Value)) > num)
							{
								string value = "Variable " + xLPVariable.Name + " value " + xLPVariable.Value.ToString("G6") + " is below the lower bound of " + xLPVariable.LowerBnd.ToString("G6");
								messages.Add(value);
							}
							xLPVariable.Value = xLPVariable.LowerBnd;
						}
						else if (xLPVariable.Value > xLPVariable.UpperBnd)
						{
							if (xLPVariable.Value > xLPVariable.UpperBnd + num && Math.Abs(xLPVariable.Value - xLPVariable.UpperBnd) / (1.0 + Math.Abs(xLPVariable.Value)) > num)
							{
								string value = "Variable " + xLPVariable.Name + " value " + xLPVariable.Value.ToString("G6") + " is above the upper bound of " + xLPVariable.UpperBnd.ToString("G6");
								messages.Add(value);
							}
							xLPVariable.Value = xLPVariable.UpperBnd;
						}
						if (xLPVariable.Spec == 3 && Math.Abs(xLPVariable.UpperBnd - xLPVariable.LowerBnd) > num)
						{
							string value = "Variable " + xLPVariable.Name + " is fixed but the bounds are open " + xLPVariable.UpperBnd.ToString("G6") + " and the solution file value " + num3.ToString("G6") + " is ignored.";
							messages.Add(value);
						}
						if (num3 == 99999.0)
						{
							xLPVariable.Value = 0.0;
						}
						if (text2.StartsWith("IEXC"))
						{
							arrayList.Add(xLPVariable);
						}
						if (scdXLP == null)
						{
							continue;
						}
						xLPVariable = scdXLP.FindVariableObject(text2);
						if (xLPVariable == null)
						{
							continue;
						}
						xLPVariable.Value = num3;
						if (xLPVariable.Value < xLPVariable.LowerBnd)
						{
							if (xLPVariable.Value < xLPVariable.LowerBnd - num && Math.Abs(xLPVariable.Value - xLPVariable.LowerBnd) / (1.0 + Math.Abs(xLPVariable.Value)) > num)
							{
								string value = "*** Warning for variable " + xLPVariable.Name + ": Initial value " + xLPVariable.Value.ToString("G6") + " is below the lower bound of  " + xLPVariable.LowerBnd.ToString("G6");
								messages.Add(value);
							}
							xLPVariable.Value = xLPVariable.LowerBnd;
						}
						else if (xLPVariable.Value > xLPVariable.UpperBnd)
						{
							if (xLPVariable.Value > xLPVariable.UpperBnd + num && Math.Abs(xLPVariable.Value - xLPVariable.UpperBnd) / (1.0 + Math.Abs(xLPVariable.Value)) > num)
							{
								string value = "*** Warning for variable " + xLPVariable.Name + ": Initial value " + xLPVariable.Value.ToString("G6") + " is above the upper bound of  " + xLPVariable.UpperBnd.ToString("G6");
								messages.Add(value);
							}
							xLPVariable.Value = xLPVariable.UpperBnd;
						}
						if (xLPVariable.Spec == 3 && Math.Abs(xLPVariable.UpperBnd - xLPVariable.LowerBnd) > num)
						{
							string value = "Variable " + xLPVariable.Name + " is fixed but the bounds are open " + xLPVariable.UpperBnd.ToString("G6") + " and the solution file value " + num3.ToString("G6") + " is ignored.";
							messages.Add(value);
						}
					}
				}
			}
			if (!flag && arrayList.Count > 0)
			{
				foreach (XLPVariable item in arrayList)
				{
					double value2 = item.Value;
					string vname = "IDEF" + item.Name.Substring(4);

	public bool FixSolution(XLPManager pXLP, string fullModelPath, string fileName)
	{
		string path = fullModelPath + fileName;
		if (File.Exists(path))
		{
			using (StreamReader streamReader = File.OpenText(path))
			{
				string text = streamReader.ReadLine();
				text = streamReader.ReadLine();
				while ((text = streamReader.ReadLine()) != null && text != "")
				{
					if (text[0] == '*' || text[0] == '$')
					{
						continue;
					}
					int num = text.IndexOf(' ');
					if (num < 0)
					{
						continue;
					}
					string vname = text.Substring(0, num);
					double num2 = Convert.ToDouble(text.Substring(num + 1));
					if (Math.Abs(num2) < 1E-06)
					{
						num2 = 0.0;
					}
					XLPVariable xLPVariable = pXLP.FindVariableObject(vname);
					if (xLPVariable != null && num2 != 99999.0)
					{
						if (num2 >= 0.0)
						{
							xLPVariable.LowerBnd = 0.99 * num2;
							xLPVariable.UpperBnd = 1.01 * num2;
						}
						else
						{
							xLPVariable.LowerBnd = 1.01 * num2;
							xLPVariable.UpperBnd = 0.99 * num2;
						}
					}
				}
			}
			return true;
		}
		return false;
	}

		_pVarList = manager.GetVariableCollectionIntf();
		_variableCount = _pVarList.NumberOfVariables();
		_bVar = new byte[128];
		Reset();
	}

	public bool MoveNext()
	{
		if (_currentNumber >= _variableCount)
		{
			return false;
		}
		StringBuilder strname = new StringBuilder(_bVar.Length);
		if (_current == null)
		{
			XLPVarCollection.FrstVariableName(_pVarList.Ptr, _bVar);
		}
		else
		{
			XLPVarCollection.NextVariableName(_pVarList.Ptr, _bVar);
		}
		_pVarList.convert2String(_bVar, strname, out var stringItem);
		_current = _manager.FindVariableObject(stringItem);
		_currentNumber++;
		return true;
	}

	public void Dispose()
	{
	}

	public void Reset()
	{
		_current = null;
		_currentNumber = 0;
	}
}
