	public class OomfKernelIndexBaseScope
	{
		internal int _refCount;

		internal int _originalBase;

		private readonly IOomfKernel _kernel;

		private readonly int _newBase;

		private readonly object _lock;

		public OomfKernelIndexBaseScope(IOomfKernel kernel, int newBase)
		{
			_kernel = kernel;
			_newBase = newBase;
			_lock = new object();
		}

		public int Increment()
		{
			lock (_lock)
			{
				if (_refCount == 0)
				{
					_originalBase = _kernel.GetVarIndexBase();
					_kernel.SetVarIndexBase(_newBase);
				}
				_refCount++;
			}
			return _refCount;
		}

		public int Decrement()
		{
			lock (_lock)
			{
				if (_refCount == 1)
				{
					_kernel.SetVarIndexBase(_originalBase);
				}
				_refCount--;
			}
			return _refCount;
		}
	}
	public class OomfKernelIndexBaseScopeReference : IDisposable
	{
		private bool _disposed;

		private readonly OomfKernelIndexBaseScope _scope;

		public OomfKernelIndexBaseScopeReference(OomfKernelIndexBaseScope scope)
		{
			_scope = scope;
			_scope.Increment();
		}

		protected virtual void Dispose(bool disposing)
		{
			if (!_disposed)
			{
				if (disposing)
				{
					_scope.Decrement();
				}
				_disposed = true;
			}
		}

		public void Dispose()
		{
			Dispose(disposing: true);
			GC.SuppressFinalize(this);
		}
	}

		public string GetBlockVariableName(Variable variable, IDictionary<Variable, string> oomfNameMap)
		{
			string text = (oomfNameMap.ContainsKey(variable) ? oomfNameMap[variable] : variable.Write().ToUpper());
			return BlockVariablePrefix + text;
		}

		public string RemoveBlockName(string blockVariableName)
		{
			if (blockVariableName.StartsWith(BlockVariablePrefix))
			{
				return blockVariableName.Substring(BlockVariablePrefix.Length);
			}
			return blockVariableName;
		}

		public void ValidateSystem(BlockInfo blockInfo, IEnumerable<VariableInfo> varInfos, ICollection<double> knownValues)
		{
			using (new OomfKernelIndexBaseScopeReference(_indexBaseScope))
			{
				int equationCount = blockInfo.EquationCount;
				int variableCount = blockInfo.VariableCount;
				int num = varInfos.Count((VariableInfo info) => info.Index >= blockInfo.VariableStartIndex && info.Index <= blockInfo.VariableEndIndex && info.Spec == VariableSpec.Constant);
				int num2 = variableCount - num - knownValues.Count - equationCount;
				if (num2 < 0)
				{
					throw new OomfSolverException(OomfErrors.OomfSolveFailed, ModelErrors.TooManySpecifications, string.Format(Resources.Overspecified, variableCount, equationCount, knownValues.Count));
				}
				if (num2 > 0)
				{
					throw new OomfSolverException(OomfErrors.OomfSolveFailed, ModelErrors.TooFewSpecifications, string.Format(Resources.Underspecified, variableCount, equationCount, knownValues.Count));
				}
				if (knownValues.Any((double v) => double.IsNaN(v) || double.IsInfinity(v)))
				{
					throw new OomfSolverException(OomfErrors.OomfSolveFailed, ModelErrors.InvalidVariableValue, Resources.InvalidVariableValue);
				}
			}
		}

		public IReadOnlyDictionary<Variable, double> Solve(IDictionary<Variable, double> knowns, IDictionary<Variable, double> initialUnknownGuesses, IDictionary<Variable, string> oomfNameMap)
		{
			using (new OomfKernelIndexBaseScopeReference(_indexBaseScope))
			{
				IDictionary<int, (VariableSpec, double)> dictionary = null;
				try
				{
					BlockInfo blkInfo = _kernel.GetBlockInfo(BlockName);
					IEnumerable<VariableInfo> variables = _kernel.GetVariables();
					ValidateSystem(blkInfo, variables, knowns.Values);
					IDictionary<string, int> varIndices = variables.Where((VariableInfo info) => info.Index >= blkInfo.VariableStartIndex && info.Index <= blkInfo.VariableEndIndex).ToDictionary((VariableInfo info) => info.Name, (VariableInfo info) => info.Index);
					dictionary = GetVariablesState(blkInfo);
					foreach (KeyValuePair<Variable, double> known in knowns)
					{
						string blockVariableName = GetBlockVariableName(known.Key, oomfNameMap);
						int validVariableIndex = GetValidVariableIndex(varIndices, blockVariableName);
						_kernel.SetVariableValue(validVariableIndex, known.Value);
						_kernel.SetVariableSpec(validVariableIndex, VariableSpec.Constant);
					}
					IDictionary<Variable, int> dictionary2 = new Dictionary<Variable, int>();
					foreach (KeyValuePair<Variable, double> initialUnknownGuess in initialUnknownGuesses)
					{
						string blockVariableName2 = GetBlockVariableName(initialUnknownGuess.Key, oomfNameMap);
						int validVariableIndex2 = GetValidVariableIndex(varIndices, blockVariableName2);
						dictionary2.Add(initialUnknownGuess.Key, validVariableIndex2);
						_kernel.SetVariableValue(validVariableIndex2, initialUnknownGuess.Value);
						_kernel.SetVariableSpec(validVariableIndex2, VariableSpec.Calc);
					}
					var (solveStatus, message) = _kernel.Solve();
					if (solveStatus != SolveStatus.NormalConvergence)
					{
						throw new OomfSolverException(OomfSolverException.GetOomfError(solveStatus), ModelErrors.SolveFailed, message);
					}
					Dictionary<Variable, double> dictionary3 = new Dictionary<Variable, double>();
					foreach (KeyValuePair<Variable, int> item in dictionary2)
					{
						dictionary3.Add(item.Key, _kernel.GetVariableValue(item.Value));
					}
					return dictionary3;
				}
				catch (OomfHostException ex)
				{
					throw new OomfSolverException(OomfErrors.OomfHostError, ModelErrors.SolveFailed, string.Format(Resources.OomfHostError, ex.Message), ex);
				}
				catch (OomfSolverException)
				{
					throw;
				}
				catch (Exception inner)
				{
					throw new OomfSolverException(OomfErrors.OomfSolveFailed, ModelErrors.SolveFailed, Resources.GeneralSolveError, inner);
				}
				finally
				{
					if (_kernel != null && dictionary != null)
					{
						SetVariablesState(dictionary);
					}
				}
			}
		}

		internal IDictionary<int, (VariableSpec Spec, double Value)> GetVariablesState(BlockInfo blkInfo)
		{
			using (new OomfKernelIndexBaseScopeReference(_indexBaseScope))
			{
				IDictionary<int, (VariableSpec, double)> dictionary = new Dictionary<int, (VariableSpec, double)>();
				for (int i = blkInfo.VariableStartIndex; i <= blkInfo.VariableEndIndex; i++)
				{
					dictionary[i] = (_kernel.GetVariableSpec(i), _kernel.GetVariableValue(i));
				}
				return dictionary;
			}
		}

		internal IDictionary<int, (VariableSpec Spec, double Value)> PrepareForDirectOomfEvaluation(double[] variableValues, BlockInfo blkInfo)
		{
			using (new OomfKernelIndexBaseScopeReference(_indexBaseScope))
			{
				if (variableValues.Length != blkInfo.VariableCount)
				{
					throw new OomfSolverException(OomfErrors.IncorrectVariableCount, ModelErrors.UnrecognizedVariable, string.Format(Resources.WrongNumberVariables, blkInfo.VariableCount, variableValues.Length));
				}
				IDictionary<int, (VariableSpec, double)> variablesState = GetVariablesState(blkInfo);
				int num = blkInfo.VariableStartIndex;
				int num2 = 0;
				while (num <= blkInfo.VariableEndIndex)
				{
					_kernel.SetVariableValue(num, variableValues[num2]);
					num++;
					num2++;
				}
				return variablesState;
			}
		}

		internal void SetVariablesState(IDictionary<int, (VariableSpec Spec, double Value)> states)
		{
			using (new OomfKernelIndexBaseScopeReference(_indexBaseScope))
			{
				foreach (KeyValuePair<int, (VariableSpec, double)> state in states)
				{
					_kernel.SetVariableSpec(state.Key, state.Value.Item1);
					_kernel.SetVariableValue(state.Key, state.Value.Item2);
				}
			}
		}

		internal double[] GetCurrentXlpVariableValues()
		{
			using (new OomfKernelIndexBaseScopeReference(_indexBaseScope))
			{
				BlockInfo blockInfo = _kernel.GetBlockInfo(BlockName);
				return (from index in Enumerable.Range(blockInfo.VariableStartIndex, blockInfo.VariableEndIndex - blockInfo.VariableStartIndex + 1)
					select _kernel.GetVariableValue(index)).ToArray();
			}
		}

		public IEnumerable<IXlpJacobianNonZero> EvaluateJacobian(double[] variableValues)
		{
			using (new OomfKernelIndexBaseScopeReference(_indexBaseScope))
			{
				IDictionary<int, (VariableSpec, double)> dictionary = new Dictionary<int, (VariableSpec, double)>();
				try
				{
					BlockInfo blockInfo = _kernel.GetBlockInfo(BlockName);
					dictionary = PrepareForDirectOomfEvaluation(variableValues, blockInfo);
					_kernel.EvaluateJacobian();
					IList<IXlpJacobianNonZero> list = new List<IXlpJacobianNonZero>();
					JacobianStructure jacobianStructure = _kernel.GetJacobianStructure();
					int num = 1;
					foreach (JacobianNonZero nonZero in jacobianStructure.NonZeros)
					{
						if (nonZero.EqnIndex >= blockInfo.EquationStartIndex && nonZero.EqnIndex <= blockInfo.EquationStopIndex && nonZero.VarIndex >= blockInfo.VariableStartIndex && nonZero.VarIndex <= blockInfo.VariableEndIndex)
						{
							list.Add(new XlpJacobianNonZero
							{
								EquationNumber = nonZero.EqnIndex - blockInfo.EquationStartIndex,
								VariableNumber = nonZero.VarIndex - blockInfo.VariableStartIndex,
								Value = _kernel.GetJacobianNonZero(num)
							});
						}
						num++;
					}
					return list;
				}
				finally
				{
					if (_kernel != null && dictionary != null)
					{
						SetVariablesState(dictionary);
					}
				}
			}
		}

		public double[] EvaluateResiduals(double[] variableValues)
		{
			using (new OomfKernelIndexBaseScopeReference(_indexBaseScope))
			{
				IDictionary<int, (VariableSpec, double)> dictionary = new Dictionary<int, (VariableSpec, double)>();
				try
				{
					BlockInfo blockInfo = _kernel.GetBlockInfo(BlockName);
					dictionary = PrepareForDirectOomfEvaluation(variableValues, blockInfo);
					_kernel.EvaluateEquations();
					double[] array = new double[blockInfo.EquationCount];
					int num = blockInfo.EquationStartIndex;
					int num2 = 0;
					while (num <= blockInfo.EquationStopIndex)
					{
						array[num2] = _kernel.GetEquationResidual(num);
						num++;
						num2++;
					}
					return array;
				}
				finally
				{
					if (_kernel != null && dictionary != null)
					{
						SetVariablesState(dictionary);
					}
				}
			}
		}

		internal int GetValidVariableIndex(IDictionary<string, int> varIndices, string varName)
		{
			if (!varIndices.ContainsKey(varName))
			{
				throw new OomfSolverException(OomfErrors.UnknownVariableName, ModelErrors.UnrecognizedVariable, string.Format(Resources.UnknownInputVariable, varName));
			}
			return varIndices[varName];
		}

		private void Dispose(bool disposing)
		{
			if (disposedValue)
			{
				return;
			}
			if (disposing && _kernel != null)
			{
				_kernel.Dispose();
			}
			string[] array = extensions;
			foreach (string text in array)
			{
				string path = Path.Combine(Path.GetTempPath(), _simulationId + text);
				if (File.Exists(path))
				{
					try
					{
						File.Delete(path);
					}
					catch (Exception)
					{
					}
				}
			}
			disposedValue = true;
		}

				SolveStatus.MaxIterationsExceeded => OomfErrors.MaxIterationsExceeded,

		public static bool EnsureOomfIsOnPath()
		{
			if (_isOomfOnThePath)
			{
				return true;
			}
			string text = null;
			try
			{
				using RegistryKey registryKey = Registry.LocalMachine.OpenSubKey(OomfVersioning.OomfVersionRegistryKey);
				text = (string)registryKey?.GetValue("oomftop");
			}
			catch (Exception)
			{
				return false;
			}
			if (string.IsNullOrEmpty(text))
			{
				return false;
			}
			text = Path.Combine(NativeMethods.GetShortName(text), "xeq");
			string environmentVariable = Environment.GetEnvironmentVariable("PATH");
			if (!environmentVariable.Split(Path.PathSeparator).Contains(text))
			{
				Environment.SetEnvironmentVariable("PATH", $"{text}{Path.PathSeparator}{environmentVariable}");
			}
			string value = text + ";" + Environment.GetEnvironmentVariable("PATH");
			Environment.SetEnvironmentVariable("PATH", value);
			_isOomfOnThePath = true;
			return _isOomfOnThePath;
		}

		private static IntPtr LoadOomfHostDll()
		{
			if (!EnsureOomfIsOnPath())
			{
				throw new OomfHostException("Failed to add OOMF to PATH environment variable");
			}
			IntPtr intPtr = NativeMethods.LoadLibrary("OOMFHOST.DLL");
			if (intPtr == IntPtr.Zero)
			{
				throw new OomfHostException("Failed to load OOMFHOST DLL");
			}
			return intPtr;
		}

		private static SolveStatus ToSolveStatus(int value)
		{
			return value switch
			{
				1 => SolveStatus.NormalConvergence, 
				2 => SolveStatus.MaxIterationsExceeded, 
				3 => SolveStatus.UnboundedProblem, 
				4 => SolveStatus.ProgrammingFailure, 
				5 => SolveStatus.ResidualTooLarge, 
				6 => SolveStatus.BadInput, 
				7 => SolveStatus.BadBounds, 
				8 => SolveStatus.InternalError, 
				9 => SolveStatus.SparseSolverError, 
				10 => SolveStatus.UserBreak, 
				11 => SolveStatus.Singularity, 
				_ => SolveStatus.Unknown, 
			};
		}

		public event EventHandler<OomfKernelEventArgs> OnOomfEvent
		{
			add
			{
				if (_onOomfEventHandler == null)
				{
					_onOomfEventHandler = value;
					ConnectEventCallback();
				}
				else
				{
					_onOomfEventHandler = (EventHandler<OomfKernelEventArgs>)Delegate.Combine(_onOomfEventHandler, value);
				}
			}
			remove
			{
				if (_onOomfEventHandler != null)
				{
					_onOomfEventHandler = (EventHandler<OomfKernelEventArgs>)Delegate.Remove(_onOomfEventHandler, value);
					if (_onOomfEventHandler == null)
					{
						DisconnectEventCallback();
					}
				}
			}
		}

		public static string MakeValidKernelName(string name)
		{
			StringBuilder stringBuilder = new StringBuilder(name.Length);
			HashSet<char> hashSet = Path.GetInvalidFileNameChars().ToHashSet();
			hashSet.Add(' ');
			bool flag = false;
			string text = name;
			foreach (char c in text)
			{
				if (hashSet.Contains(c))
				{
					flag = true;
					stringBuilder.Append('_');
				}
				else
				{
					stringBuilder.Append(c);
				}
			}
			if (flag)
			{
				name = stringBuilder.ToString();
			}
			if (name.Length > 64)
			{
				name = stringBuilder.ToString(0, 64);
			}
			return name;
		}

	public sealed class OomfMessageSource : IOomfMessageSource, IDisposable
	{
		private static readonly Lazy<IOomfMessageSource> s_singleton = new Lazy<IOomfMessageSource>(() => new OomfMessageSource());

		private SendMessageFn _sendMessageDelegate;

		private EventHandler<string> _onMessageHandler;

		public static IOomfMessageSource Instance => s_singleton.Value;

		public event EventHandler<string> OnMessage
		{
			add
			{
				if (_onMessageHandler == null)
				{
					_onMessageHandler = value;
					ConnectSendMessage();
				}
				else
				{
					_onMessageHandler = (EventHandler<string>)Delegate.Combine(_onMessageHandler, value);
				}
			}
			remove
			{
				if (_onMessageHandler != null)
				{
					_onMessageHandler = (EventHandler<string>)Delegate.Remove(_onMessageHandler, value);
					if (_onMessageHandler == null)
					{
						DisconnectSendMessage();
					}
				}
			}
		}

		public void Dispose()
		{
			DisconnectSendMessage();
		}

		private void ConnectSendMessage()
		{
			if (_sendMessageDelegate == null)
			{
				_sendMessageDelegate = ProcessSendMessage;
				OomfHostApi.SetMessageCallback(_sendMessageDelegate);
			}
		}

		private void DisconnectSendMessage()
		{
			SendMessageFn sendMessageDelegate = _sendMessageDelegate;
			_sendMessageDelegate = null;
			if (sendMessageDelegate != null)
			{
				OomfHostApi.SetMessageCallback(null);
			}
		}

		private void ProcessSendMessage(string msg)
		{
			_onMessageHandler(this, msg);
		}
	}
	public static class OomfVersioning
	{
		internal static string _oomfVersion;

		public static string OomfRegistryKey => Path.Combine("SOFTWARE", "AspenTech", "OOMF");

		public static string OomfVersionRegistryKey => Path.Combine(OomfRegistryKey, DetermineOomfVersion());

		public static string DetermineOomfVersion()
		{
			if (_oomfVersion == null)
			{
				FileVersionInfo fileVersionInfo = Process.GetCurrentProcess().MainModule.FileVersionInfo;
				string text = $"{fileVersionInfo.ProductMajorPart}.{fileVersionInfo.ProductMinorPart}";
				string[] array = null;
				using (RegistryKey registryKey = Registry.LocalMachine.OpenSubKey(OomfRegistryKey))
				{
					if (registryKey != null)
					{
						array = registryKey.GetSubKeyNames();
					}
				}
				if (array == null)
				{
					throw new OomfHostException("Could not find an OOMF installation");
				}
				if (array.Contains(text))
				{
					_oomfVersion = text;
				}
				else
				{
					string currentOomfVersion = GetCurrentOomfVersion();
					if (!string.IsNullOrEmpty(currentOomfVersion) && array.Contains(currentOomfVersion))
					{
						_oomfVersion = currentOomfVersion;
					}
					else
					{
						_oomfVersion = array.LastOrDefault((string versionString) => double.TryParse(versionString, out var _));
					}
				}
			}
			return _oomfVersion;
		}

		public static string GetCurrentOomfVersion()
		{
			using (RegistryKey registryKey = Registry.LocalMachine.OpenSubKey(Path.Combine(OomfRegistryKey, "CurVer")))
			{
				if (registryKey != null)
				{
					return registryKey.GetValue(string.Empty)?.ToString();
				}
			}
			return null;
		}
