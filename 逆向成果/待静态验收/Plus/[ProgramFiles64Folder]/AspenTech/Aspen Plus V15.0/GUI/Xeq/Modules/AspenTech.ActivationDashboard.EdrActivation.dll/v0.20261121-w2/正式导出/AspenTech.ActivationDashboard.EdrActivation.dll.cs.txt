		private void column_SizeChanged(object sender, SizeChangedEventArgs e)
		{
			if (e.PreviousSize.Width != 0.0)
			{
				if (double.IsNaN(datagrid.Width) && datagrid.ActualWidth != 0.0)
				{
					datagrid.Width = datagrid.ActualWidth;
				}
				if (_columnTag != (sender as FrameworkElement).Tag || _newWidth != e.NewSize.Width)
				{
					datagrid.Width += e.NewSize.Width - e.PreviousSize.Width;
					_columnTag = (sender as FrameworkElement).Tag.ToString();
					_newWidth = e.NewSize.Width;
				}
			}
		}

		private void datagrid_Sorting(object sender, DataGridSortingEventArgs e)
		{
			if (datagrid == null || datagrid.Items == null || datagrid.Items.Count == 0 || !(datagrid.Items[0] is ActivationSummaryViewModel.Exchanger exchanger))
			{
				return;
			}
			switch (e.Column.DisplayIndex)
			{
			case 0:
				return;
			case 1:
			{
				string text = exchanger.ContainerName;
				for (int j = 1; j < datagrid.Items.Count; j++)
				{
					if (text != (datagrid.Items[j] as ActivationSummaryViewModel.Exchanger).ContainerName.ToString())
					{
						return;
					}
				}
				break;
			}
			case 2:
			{
				string text = exchanger.StatusSortIndex.ToString();
				for (int k = 1; k < datagrid.Items.Count; k++)
				{
					if (text != (datagrid.Items[k] as ActivationSummaryViewModel.Exchanger).StatusSortIndex.ToString())
					{
						return;
					}
				}
				break;
			}
			case 3:
			{
				string text = exchanger.RiskSortIndex.ToString();
				for (int i = 1; i < datagrid.Items.Count; i++)
				{
					if (text != (datagrid.Items[i] as ActivationSummaryViewModel.Exchanger).RiskSortIndex.ToString())
					{
						return;
					}
				}
				break;
			}
			}
			if (e.Column.SortDirection == ListSortDirection.Ascending)
			{
				e.Column.SortDirection = ListSortDirection.Descending;
			}
			else
			{
				e.Column.SortDirection = ListSortDirection.Ascending;
			}
			e.Handled = true;
		}

		private bool CheckforCalcErrors(IEDRActivatedExchanger exch)
		{
			if (_statusExchangers == null)
			{
				GetStatusExchangers(exch.ContainerName);
			}
			string text = "";
			if (exch.ContainerName.Length > 0)
			{
				text = text + exch.ContainerName + ".";
			}
			text += exch.ExchangerName;
			if (exch.IsContainer)
			{
				for (int i = 0; i < _statusExchangers.Count; i++)
				{
					if (_statusExchangers.ElementAt(i).Key.StartsWith(text) && _statusExchangers.ElementAt(i).Value.CalcError)
					{
						exch.CalcError = true;
						return exch.CalcError;
					}
				}
			}
			else
			{
				for (int j = 0; j < _statusExchangers.Count; j++)
				{
					if (_statusExchangers.ElementAt(j).Key.Equals(text))
					{
						exch.CalcError = _statusExchangers.ElementAt(j).Value.CalcError;
						return exch.CalcError;
					}
				}
			}
			return exch.CalcError;
		}

		{
			if (_statusExchangers == null)
			{
				_statusExchangers = new SortedList<string, ModelStatusExchanger>();
			}
			if (string.IsNullOrEmpty(containerName))
			{
				foreach (KeyValuePair<string, IEDRActivatedExchanger> exchanger in GenericActivationCollection._exchangers)
				{
					if (_statusExchangers.Keys.Contains(exchanger.Key))
					{
						_statusExchangers.Remove(exchanger.Key);
					}
					if (exchanger.Value.EDRApp != null && exchanger.Value.IsRigorous)
					{
						_statusExchangers.Add(exchanger.Key, new ModelStatusExchanger(exchanger.Value.EDRApp));
					}
				}
			}
			else
			{
				IDictionary<string, IEDRActivatedExchanger> subExchangers = GetSubExchangers(containerName);
				if (subExchangers != null)
				{
					foreach (KeyValuePair<string, IEDRActivatedExchanger> item in subExchangers)
					{
						if (item.Value.EDRApp != null && item.Value.IsRigorous)
						{
							if (_statusExchangers.Keys.Contains(item.Key))
							{
								_statusExchangers.Remove(item.Key);
							}
							_statusExchangers.Add(item.Key, new ModelStatusExchanger(item.Value.EDRApp));
						}
					}
				}
			}
			return _statusExchangers;
		}

		{

			if (RoughlyEqual(requiredArrangeSize, arrangedResult))

			}
			if (_requiredSizeAfterRender.HasValue && RoughlyEqual(_requiredSizeAfterRender.Value, requiredArrangeSize))
			{
				return false;
			}

			return true;

		private System.Windows.Size GetRequiredArrangeSize(System.Windows.Size arrangedResult)
		{
			Queue<Tuple<FrameworkElement, FrameworkElement>> queue = new Queue<Tuple<FrameworkElement, FrameworkElement>>();

			System.Windows.Size? sizeOverflow = GetSizeOverflow(queue);
			if (sizeOverflow.HasValue)
			{
				return new System.Windows.Size(arrangedResult.Width + sizeOverflow.Value.Width, arrangedResult.Height + sizeOverflow.Value.Height);
			}
			return arrangedResult;

		private static System.Windows.Size? GetSizeOverflow(Queue<Tuple<FrameworkElement, FrameworkElement>> queque)

				FrameworkElement item2 = tuple.Item2;
				if (item.Visibility == Visibility.Visible && !ImpossibleSizeOverflow(item))
				{
					System.Windows.Size rightBottom = GetRightBottom(item, item2, item.Margin);
					if (rightBottom.Width > item2.ActualWidth || rightBottom.Height > item2.ActualHeight)
					{
						return new System.Windows.Size(Math.Max(0.0, rightBottom.Width - item2.ActualWidth), Math.Max(0.0, rightBottom.Height - item2.ActualHeight));
					}
					int childrenCount = VisualTreeHelper.GetChildrenCount(item);
					for (int i = 0; i < childrenCount; i++)
					{
						FrameworkElement item3 = VisualTreeHelper.GetChild(item, i) as FrameworkElement;

			{
				return ActivationState.Error;
			}
			if (state1 == ActivationState.Warning || state2 == ActivationState.Warning)
			{
				return ActivationState.Warning;
			}
			if (state1 == ActivationState.Advise || state2 == ActivationState.Advise)
			{
				return ActivationState.Advise;
			}
			if (state1 == ActivationState.BuiltIn && state2 == ActivationState.BuiltIn)
			{
				return ActivationState.BuiltIn;

		private ActivationState MaxState(ActivationState state1, ActivationState state2)
		{
			if (state1 == ActivationState.Error || state2 == ActivationState.Error)
			{
				return ActivationState.Error;
			}
			if (state1 == ActivationState.Warning || state2 == ActivationState.Warning)
			{
				return ActivationState.Warning;
			}
			if (state1 == ActivationState.Advise || state2 == ActivationState.Advise)
			{
				return ActivationState.Advise;
			}
			if (state1 == ActivationState.BuiltIn && state2 == ActivationState.BuiltIn)
			{
				return ActivationState.BuiltIn;
			}
			return ActivationState.OK;
		}

				for (int k = 1; k < datagrid.Items.Count; k++)

					if (text != (datagrid.Items[k] as ActivationSummaryViewModel.Exchanger).StatusSortIndex.ToString())
					{
						return;
					}
				}
				break;
			}
			case 3:
			{

				for (int i = 1; i < datagrid.Items.Count; i++)
				{
					if (text != (datagrid.Items[i] as ActivationSummaryViewModel.Exchanger).RiskSortIndex.ToString())
					{
						return;
					}
				}
				break;
			}
			}
			if (e.Column.SortDirection == ListSortDirection.Ascending)
			{
				e.Column.SortDirection = ListSortDirection.Descending;
			}
			else
			{
				e.Column.SortDirection = ListSortDirection.Ascending;

		public ICommand CheckedCommand => (ICommand)new RelayActionCommand<object>((Action<object>)Check);

		public ICommand OKCommand => (ICommand)new RelayActionCommand((Action)SaveFilter, (Func<bool>)CanSaveFilter);

		public ICommand CancelCommand => (ICommand)new RelayActionCommand((Action)CancelFilter);

		public FilterPanel()
		{
			InitializeComponent();
			listbox.Items.Clear();
			base.DataContext = this;
		}

		private void Check(object args)
		{
			if (!(args is System.Windows.Controls.CheckBox checkBox))
			{
				return;
			}
			if (checkBox.Content.ToString() == Resource.Filter_SelectAll)
			{
				for (int i = 1; i < _dataSource.Count; i++)
				{
					DataSource[i].IsChecked = checkBox.IsChecked.Value;
				}
				return;
			}
			if (!checkBox.IsChecked.Value)
			{
				DataSource[0].IsChecked = false;
				return;
			}
			for (int j = 1; j < _dataSource.Count; j++)
			{
				if (!_dataSource[j].IsChecked.Value)
				{
					return;
				}
			}
			DataSource[0].IsChecked = true;
		}

		private void SaveFilter()
		{
			_filter = new List<object>();
			if (!_dataSource[0].IsChecked.Value)
			{
				foreach (FilterItem item in _dataSource)
				{
					if (item.IsChecked.Value)
					{
						_filter.Add(item.ItemCallback);
					}
				}
			}
			FilterCallback();
		}

		private bool CanSaveFilter()
		{
			for (int i = 0; i < _dataSource.Count; i++)
			{
				if (_dataSource[i].IsChecked.Value)
				{
					return true;
				}
			}
			return false;
		}

		private void CancelFilter()
		{
			(base.Parent as Popup).IsOpen = false;
		}

		public void FilterCallback()
		{
			(base.Parent as Popup).IsOpen = false;
			if (this.FilterCallbackCalled != null)
			{
				this.FilterCallbackCalled(new object[2] { _columnName, _filter }, new EventArgs());
			}
		}

	public class PieChart : System.Windows.Controls.UserControl, IComponentConnector
	{
		private double _fontSize = 9.0;

		private double _radius = 30.0;

		private double _rectSize = 5.0;

		private string _title;

		private ObservableCollection<PieSegment> _dataSource;

		internal TextBlock title;

		internal Canvas canvPie;

		internal Grid gridLegend;

		private bool _contentLoaded;

		public string Title
		{
			get
			{
				return _title;
			}
			set
			{
				_title = value;
				title.Text = _title;
			}
		}

		public ObservableCollection<PieSegment> DataSource
		{
			get
			{
				return _dataSource;
			}
			set
			{
				_dataSource = value;
				if (_title == Resource.pieChartTitle_Risk && _dataSource.Last().Value == GetTotal() && base.Tag != "Sub")
				{
					base.IsEnabled = false;
					canvPie.Children.Clear();
					gridLegend.Children.Clear();
				}
				else
				{
					base.IsEnabled = base.Tag != "Sub";
					ResetPie();
				}
			}
		}

		public PieChart()
		{
			InitializeComponent();
		}

		private void CreatePie()
		{
			canvPie.Children.Clear();
			if (DataSource == null)
			{
				return;
			}
			double total = GetTotal();
			if (!(total > 0.0))
			{
				return;
			}
			double num = -Math.PI / 2.0;
			foreach (PieSegment item in DataSource)
			{
				System.Windows.Shapes.Path path = new System.Windows.Shapes.Path();
				Geometry geometry;
				if (item.Value == total)
				{
					geometry = new EllipseGeometry(new System.Windows.Point(_radius / 2.0, _radius / 2.0), _radius / 2.0, _radius / 2.0);
				}
				else
				{
					geometry = new PathGeometry();
					double x = Math.Cos(num) * _radius / 2.0 + _radius / 2.0;
					double y = Math.Sin(num) * _radius / 2.0 + _radius / 2.0;
					LineSegment lineSegment = new LineSegment(new System.Windows.Point(x, y), isStroked: true);
					double num2 = item.Value / total * 360.0;
					num += Math.PI * num2 / 180.0;
					x = Math.Cos(num) * _radius / 2.0 + _radius / 2.0;
					y = Math.Sin(num) * _radius / 2.0 + _radius / 2.0;
					ArcSegment arcSegment = new ArcSegment(new System.Windows.Point(x, y), new System.Windows.Size(_radius / 2.0, _radius / 2.0), num2, num2 > 180.0, SweepDirection.Clockwise, isStroked: false);
					LineSegment lineSegment2 = new LineSegment(new System.Windows.Point(_radius / 2.0, _radius / 2.0), isStroked: true);
					PathFigure value = new PathFigure(new System.Windows.Point(_radius / 2.0, _radius / 2.0), new PathSegment[3] { lineSegment, arcSegment, lineSegment2 }, closed: false);
					((PathGeometry)geometry).Figures.Add(value);
				}
				SolidColorBrush solidColorBrush = new SolidColorBrush();
				solidColorBrush.Color = item.Color;
				path.Fill = solidColorBrush;
				path.Data = geometry;
				canvPie.Children.Add(path);
			}
		}

		private void CreateLegend()
		{
			gridLegend.Children.Clear();
			if (DataSource == null)
			{
				return;
			}
			int count = DataSource.Count;
			if (count > 0)
			{
				for (int i = 0; i < count; i++)
				{
					RowDefinition rowDefinition = new RowDefinition();
					rowDefinition.Height = GridLength.Auto;
					gridLegend.RowDefinitions.Add(rowDefinition);
					System.Windows.Shapes.Rectangle rectangle = new System.Windows.Shapes.Rectangle();
					SolidColorBrush solidColorBrush = new SolidColorBrush();
					solidColorBrush.Color = DataSource[i].Color;
					rectangle.Width = _rectSize;
					rectangle.Height = _rectSize;
					rectangle.Fill = solidColorBrush;
					rectangle.VerticalAlignment = VerticalAlignment.Center;
					rectangle.Margin = new Thickness(2.0, 0.0, 0.0, 0.0);
					Grid.SetRow(rectangle, i);
					Grid.SetColumn(rectangle, 0);
					gridLegend.Children.Add(rectangle);
					TextBlock textBlock = new TextBlock();
					textBlock.Text = DataSource[i].Label;
					textBlock.FontSize = _fontSize;
					textBlock.Margin = new Thickness(2.0, 0.0, 0.0, 0.0);
					Grid.SetRow(textBlock, i);
					Grid.SetColumn(textBlock, 1);
					gridLegend.Children.Add(textBlock);
					TextBlock textBlock2 = new TextBlock();
					textBlock2.Text = DataSource[i].Value.ToString();
					textBlock2.FontSize = _fontSize;

					ItemContent = Resource.Filter_SelectAll
				});
				if (_dicFilter[FilterType.Risk] == null || _dicFilter[FilterType.Risk].Count == 0)
				{
					foreach (Exchanger item in _cvExchangerSummary)
					{
						if ((item.OperationalRisk == null || !observableCollection.Select((FilterItem a) => a.ItemCallback).Contains(item.OperationalRisk)) && (!string.IsNullOrEmpty(item.OperationalRisk) || !observableCollection.Select((FilterItem a) => a.ItemContent).Contains(Resource.Filter_Blanks)))
						{
							FilterItem filterItem = new FilterItem(base.Container);
							filterItem.IsChecked = true;
							if (string.IsNullOrEmpty(item.OperationalRisk))
							{
								filterItem.ItemContent = Resource.Filter_Blanks;
							}
							else
							{
								Ellipse ellipse = new Ellipse();
								ellipse.Style = (Style)_filterButton.TryFindResource(item.OperationalRisk);
								filterItem.ItemContent = ellipse;
							}
							filterItem.ItemCallback = item.OperationalRisk;
							observableCollection.Add(filterItem);
						}
					}
				}
				else
				{
					foreach (Exchanger item2 in _exchangerSummary)
					{
						if ((string.IsNullOrEmpty(item2.OperationalRisk) || !observableCollection.Select((FilterItem a) => a.ItemCallback).Contains(item2.OperationalRisk)) && (!string.IsNullOrEmpty(item2.OperationalRisk) || !observableCollection.Select((FilterItem a) => a.ItemContent).Contains(Resource.Filter_Blanks)))
						{
							FilterItem filterItem2 = new FilterItem(base.Container);
							filterItem2.IsChecked = _dicFilter[FilterType.Risk].Contains(item2.OperationalRisk);
							if (!filterItem2.IsChecked.Value)
							{
								observableCollection[0].IsChecked = false;
							}
							if (item2.OperationalRisk == null)
							{
								filterItem2.ItemContent = Resource.Filter_Blanks;
							}
							else
							{
								Ellipse ellipse2 = new Ellipse();
								ellipse2.Style = (Style)_filterButton.TryFindResource(item2.OperationalRisk);
								filterItem2.ItemContent = ellipse2;
							}
							filterItem2.ItemCallback = item2.OperationalRisk;
							observableCollection.Add(filterItem2);
						}
					}
				}
				FilterPanel filterPanel = new FilterPanel();
				filterPanel.DataSource = observableCollection;
				filterPanel.ColumnName = GetFilterType(_filterButton.Tag.ToString());
				_riskFilterPopup = new Popup();
				_riskFilterPopup.Child = filterPanel;
				_riskFilterPopup.PlacementTarget = _filterButton;
				_riskFilterPopup.Placement = PlacementMode.Left;

				_riskFilterPopup.VerticalOffset = _filterButton.ActualHeight;
				_riskFilterPopup.IsOpen = true;
				_riskFilterPopup.StaysOpen = false;
				_riskFilterPopup.Closed += delegate
				{
					if (!_filterButton.IsPressed)
					{
						_filterButton.IsChecked = false;
					}
				};
				filterPanel.FilterCallbackCalled += FilterCallbackCalled;
			}
		}

		private FilterType GetFilterType(string controlTag)
		{
			FilterType result = FilterType.Container;
			switch (controlTag)
			{
			case "Container":
				result = FilterType.Container;
				break;
			case "Status":
				result = FilterType.Status;
				break;
			case "Risk":
				result = FilterType.Risk;
				break;
			}
			return result;
		}

		private void FilterCallbackCalled(object sender, EventArgs e)
		{
			object[] array = sender as object[];
			FilterType key = (FilterType)array[0];
			_dicFilter[key] = array[1] as List<object>;
			FilterData();
			_filterButton.Style = (Style)_filterButton.TryFindResource((_dicFilter[key].Count == 0) ? "unfilterStyle" : "filterStyle");
		}

		private void FilterData()
		{
			CVExchangerSummary.Filter = delegate(object a)
			{
				foreach (KeyValuePair<FilterType, List<object>> item in _dicFilter)
				{
					if (item.Value != null && item.Value.Count != 0)
					{
						switch (item.Key)
						{
						case FilterType.Container:
							if (!item.Value.Contains((a as Exchanger).ContainerName))
							{
								return false;
							}
							break;
						case FilterType.Status:
							if (!item.Value.Contains((a as Exchanger).Status))
							{
								return false;
							}
							break;
						case FilterType.Risk:
							if (!item.Value.Contains((a as Exchanger).OperationalRisk))
							{
								return false;
							}
							break;
						}
					}
				}
				return true;
			};
		}
