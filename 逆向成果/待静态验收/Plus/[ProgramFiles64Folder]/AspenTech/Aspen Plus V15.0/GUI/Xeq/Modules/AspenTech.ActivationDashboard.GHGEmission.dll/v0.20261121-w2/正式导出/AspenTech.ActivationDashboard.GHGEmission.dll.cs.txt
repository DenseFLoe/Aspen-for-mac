		public event PropertyChangedEventHandler PropertyChanged;

		private static void OnOffStateDisplayChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
		{
			(d as DashboardItem).RefreshUIDisplay();
		}

		private void RefreshUIDisplay()
		{
			UpdateOnOffButtonStatusAndDoRelatedActions();
		}

		protected void NotifyPropertyChanged(string propertyName)
		{
			this.PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
		}

		public DashboardItem(IUnityContainer container)
		{
			_container = container;
			InitializeComponent();
			_agent = UnityContainerExtensions.Resolve<IGHGEmissionAgent>(_container, Array.Empty<ResolverOverride>());
			_agent.UpdateDataEvent += RefreshDataEventHandler;
			_agent.ResetDashboardEvent += ResetDashboardEvent;
			SetPropertyBinding();
		}

		private void SetPropertyBinding()
		{
			((FrameworkElement)this).SetBinding(OnOffStateDisplayProperty, (BindingBase)new Binding("IsActive")
			{
				Source = _agent,
				Mode = BindingMode.TwoWay
			});
		}

		private void ResetDashboardEvent(object sender, EventArgs e)
		{
			OnOffStateDisplay = true;
		}

		private void RefreshDataEventHandler(object sender, EventArgs e)
		{
			if (OnOffStateDisplay)
			{
				GetEngineValuesForUIDisplay();
			}
		}

		private void GetEngineValuesForUIDisplay()
		{
			CO2Scope1 = _agent.GetNetStreamCO2();
			CO2Scope2 = _agent.GetUtilityCO2();
			CarbonTax = _agent.GetCarbonTax();
			CO2Scope1Unit = _agent.GetNetStreamCO2Unit();
			CO2Scope2Unit = _agent.GetUtilityCO2Unit();
			CarbonTaxUnit = _agent.GetCarbonTaxUnit();
		}

		protected override void OnClicked(object originalSource)
		{
			_agent.ShowCO2EmissionScreen();
			((ContentDashboardItem)this).OnClicked(originalSource);
		}

		private void UpdateOnOffButtonStatusAndDoRelatedActions()
		{
			if (OnOffStateDisplay)
			{
				OnOffButtonControlToolTip = Resources.GHGE_DashBoard_ONOFFButton_ON;
				GetEngineValuesForUIDisplay();
			}
			else
			{
				OnOffButtonControlToolTip = Resources.GHGE_DashBoard_ONOFFButton_OFF;
				ClearUIDisplay();
			}
		}

		private void ClearUIDisplay()
		{
			CO2Scope1 = null;
			CO2Scope2 = null;
			CarbonTax = null;
			CO2Scope1Unit = string.Empty;
			CO2Scope2Unit = string.Empty;
			CarbonTaxUnit = string.Empty;
		}

		protected override string SaveStateOverride()
		{
			return new XDocument(new XComment("This is the current state of GHGEmission Dashboard"), new XElement("Root", new XElement("OnOffState", new XAttribute("Value", OnOffStateDisplay)))).ToString();
		}

		protected override void RestoreStateOverride(string state)
		{
			if (string.IsNullOrEmpty(state))
			{
				_agent.IsActive = true;
				((ContentDashboardItem)this).RestoreStateOverride(state);
				return;
			}
			XDocument xDocument = XDocument.Parse(state);
			if (xDocument != null && UnityContainerExtensions.Resolve<IDashboardPanelAgent>(_container, Array.Empty<ResolverOverride>()) != null)
			{
				XElement xElement = xDocument.Root.Element("OnOffState");
				if (xElement != null)
				{
					XAttribute xAttribute = xElement.Attribute("Value");
					if (string.Equals(xAttribute.Value, "true", StringComparison.InvariantCultureIgnoreCase) || string.Equals(xAttribute.Value, "On", StringComparison.InvariantCultureIgnoreCase))
					{
						OnOffStateDisplay = true;
					}
					else
					{
						OnOffStateDisplay = false;
					}
				}
			}
			((ContentDashboardItem)this).RestoreStateOverride(state);
		}

		[DebuggerNonUserCode]
		[GeneratedCode("PresentationBuildTasks", "4.0.0.0")]
		public void InitializeComponent()
		{
			if (!_contentLoaded)
			{
				_contentLoaded = true;
				Uri resourceLocator = new Uri("/AspenTech.ActivationDashboard.GHGEmission;component/dashboarditem.xaml", UriKind.Relative);
				Application.LoadComponent(this, resourceLocator);
			}
		}

		[DebuggerNonUserCode]
		[GeneratedCode("PresentationBuildTasks", "4.0.0.0")]
		[EditorBrowsable(EditorBrowsableState.Never)]
		void IComponentConnector.Connect(int connectionId, object target)
		{
			if (connectionId == 1)
			{
				DBItem = (DashboardItem)target;
			}
			else
			{
				_contentLoaded = true;
			}
		}
	}
