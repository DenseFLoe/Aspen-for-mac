		public static string[] ValueToStringArray(object value)
		{
			if (value == null || value == DBNull.Value)
			{
				return null;
			}
			if (value.GetType() != typeof(string))
			{
				return null;
			}
			string text = (string)value;
			if (string.IsNullOrEmpty(text))
			{
				return null;
			}
			if (text[text.Length - 1] == '\n')
			{
				text = text.Remove(text.Length - 1);
			}
			return text.Split('\n');
		}

		private void Display()
		{
			string cellName = CellName;
			int num = 0;
			for (int i = 0; i < _prompts.Length; i++)
			{
				IHNode iHNode = _mainInput.FindNode(_prompts[i].Variable + "\\" + cellName);
				if (iHNode == null)
				{
					continue;
				}
				bool flag = (dynamic)iHNode.get_AttributeValue((short)7, Type.Missing) != 1;
				if (!(bool)(!flag || (_prompts[i].Text == "Units:" && (dynamic)iHNode.get_Value(Type.Missing) != null)))
				{
					continue;
				}
				_labelStepVisibilities[num] = Visibility.Visible;
				_labelStepCaptions[num] = _prompts[i].Text;
				if (iHNode.get_AttributeType((short)5) == 0)
				{
					_comboStepVisibilities[num] = Visibility.Collapsed;
					_textStepVisibilities[num] = Visibility.Visible;
					if (iHNode.ValueType == 3 && iHNode.get_AttributeType((short)8) != 0 && (dynamic)iHNode.get_AttributeValue((short)8, Type.Missing) != null)
					{
						_textMaxLentgh[num] = (dynamic)iHNode.get_AttributeValue((short)8, Type.Missing);
					}
					else
					{
						_textMaxLentgh[num] = 10;
					}
				}
				else
				{
					_comboStepVisibilities[num] = Visibility.Visible;
					_textStepVisibilities[num] = Visibility.Collapsed;
				}
				if (flag || _prompts[i].Text == "Sentence:")
				{
					_isComboStepEnabled[num] = true;
					_isTextStepEnabled[num] = true;
					_comboStepVisibilities[num] = Visibility.Collapsed;
					_textStepVisibilities[num] = Visibility.Visible;
				}
				else
				{
					_isComboStepEnabled[num] = false;
					_isTextStepEnabled[num] = false;
				}
				_stepVariables[num] = _prompts[i].Variable;
				if (_comboStepVisibilities[num] == Visibility.Visible)
				{
					IEnumerable<string> optionsList = GetOptionsList(_prompts[i].Variable, cellName, _prompts[i].ConvertToMixedCase);
					if (!optionsList.SequenceEqual(ComboStepSources[num]))
					{
						ComboStepSources[num].Clear();
						foreach (string item in optionsList)
						{
							ComboStepSources[num].Add(item);
						}
					}
				}
				OnPropertyChanged("StepComboValue" + num);
				num++;
				if (num > 7)
				{
					break;
				}
			}
			for (int j = num; j < 8; j++)
			{
				_labelStepVisibilities[j] = Visibility.Collapsed;
				_comboStepVisibilities[j] = Visibility.Collapsed;
				_textStepVisibilities[j] = Visibility.Collapsed;
			}
			OnPropertyChanged("LabelStepVisibilities");
			OnPropertyChanged("LabelStepCaptions");
			OnPropertyChanged("ComboStepVisibilities");
			OnPropertyChanged("IsComboStepEnabled");
			OnPropertyChanged("TextStepVisibilities");
			OnPropertyChanged("IsTextStepEnabled");
			OnPropertyChanged("TextMaxLentgh");
			OnPropertyChanged("ComboStepSources");
			GetAllPrompts();
		}

		private void OptInformationFlowChanged()
		{
			if (!AreAllOptionsUpdated)
			{
				return;
			}
			switch (_oldIndex)
			{
			case -1:
				if (IsImportVar)
				{
					SetRWVar(0, _readCollection, value: true);
				}
				else if (IsExportVar)
				{
					SetRWVar(1, _writeCollection, value: true);
				}
				else if (IsTearVar)
				{
					SetRWVar(1, _writeCollection, value: true);
					SetTearVar(value: true);
				}
				break;
			case 0:
				if (!IsImportVar)
				{
					if (IsExportVar)
					{
						SetRWVar(0, _readCollection, value: false);
						SetRWVar(1, _writeCollection, value: true);
					}
					else if (IsTearVar)
					{
						SetRWVar(0, _readCollection, value: false);
						SetRWVar(1, _writeCollection, value: true);
						SetTearVar(value: true);
					}
					else
					{
						SetRWVar(0, _readCollection, value: false);
						SetRWVar(1, _writeCollection, value: false);
						SetTearVar(value: false);
					}
				}
				break;
			case 1:
				if (IsImportVar)
				{
					SetRWVar(1, _writeCollection, value: false);
					SetRWVar(0, _readCollection, value: true);
				}
				else if (!IsExportVar)
				{
					if (IsTearVar)
					{
						SetTearVar(value: true);
						break;
					}
					SetRWVar(0, _readCollection, value: false);
					SetRWVar(1, _writeCollection, value: false);
					SetTearVar(value: false);
				}
				break;
			case 2:
				if (IsImportVar)
				{
					SetTearVar(value: false);
					SetRWVar(1, _writeCollection, value: false);
					SetRWVar(0, _readCollection, value: true);
				}
				else if (IsExportVar)
				{
					SetTearVar(value: false);
					SetRWVar(0, _readCollection, value: false);
					SetRWVar(1, _writeCollection, value: true);
				}
				else if (!IsTearVar)
				{
					SetRWVar(0, _readCollection, value: false);
					SetRWVar(1, _writeCollection, value: false);
					SetTearVar(value: false);
				}
				break;
			}
			if (IsImportVar)
			{
				_oldIndex = 0;
			}
			else if (IsExportVar)
			{
				_oldIndex = 1;
			}
			else if (IsTearVar)
			{
				_oldIndex = 2;
			}
			else
			{
				_oldIndex = -1;
			}
		}
