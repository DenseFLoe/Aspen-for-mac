		{
			if (data == null)

			}
			IEnumerable<UtilityDataProperty> enumerable = data.Where((UtilityDataProperty d) => d != null);
			if (enumerable == null)

				return null;
			}
			int num = enumerable.Count();
			if (num == 0)
			{
				return null;
			}
			try
			{
				StringBuilder stringBuilder = new StringBuilder();
				for (int num2 = 0; num2 < num; num2++)
				{
					UtilityDataProperty val = enumerable.ElementAt(num2);
					if (!string.IsNullOrEmpty(val.Name))
					{
						stringBuilder.Append(val.Name);
					}
					stringBuilder.Append('\t');

					{
						stringBuilder.Append(val.Value.ToString());
					}
					stringBuilder.Append('\t');
					if (!string.IsNullOrEmpty(val.Units))
					{
						stringBuilder.Append(val.Units);
					}
					if (num2 != num - 1)
					{

					}
				}
				return stringBuilder.ToString();
