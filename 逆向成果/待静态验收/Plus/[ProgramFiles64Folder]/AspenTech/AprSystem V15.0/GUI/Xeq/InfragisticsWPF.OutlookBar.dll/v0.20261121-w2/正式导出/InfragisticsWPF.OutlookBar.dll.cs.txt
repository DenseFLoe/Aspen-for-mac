	[DesignTimeVisible(false)]
	public class GroupAreaSplitter : Thumb
	{
		private double _dragIngrementUp;

		private double _dragIngrementDn;

		private RowDefinition _contentRow;

		private RowDefinition _groupsRow;

		private GroupsPresenter _navigationArea;

		internal double DragIngrementDn
		{
			get
			{
				return _dragIngrementDn;
			}
			set
			{
				_dragIngrementDn = value;
			}
		}

		internal double DragIngrementUp
		{
			get
			{
				return _dragIngrementUp;
			}
			set
			{
				_dragIngrementUp = value;
			}
		}

		static GroupAreaSplitter()
		{
			FrameworkElement.DefaultStyleKeyProperty.OverrideMetadata(typeof(GroupAreaSplitter), new FrameworkPropertyMetadata(typeof(GroupAreaSplitter)));
		}

		public GroupAreaSplitter()
		{
			base.DragDelta += GroupAreaSplitter_DragDelta;
		}

		protected override AutomationPeer OnCreateAutomationPeer()
		{
			return new GroupAreaSplitterAutomationPeer(this);
		}

		public override void OnApplyTemplate()
		{
			base.OnApplyTemplate();
			if (!(GetValue(XamOutlookBar.OutlookBarProperty) is XamOutlookBar) || !(VisualTreeHelper.GetParent(this) is Grid grid))
			{
				return;
			}
			int num = (int)GetValue(Grid.RowProperty);
			if (grid.RowDefinitions.Count <= num + 1 || num < 1)
			{
				return;
			}
			_contentRow = grid.RowDefinitions[num - 1];
			_groupsRow = grid.RowDefinitions[num + 1];
			foreach (FrameworkElement child in grid.Children)
			{
				_navigationArea = child as GroupsPresenter;
				if (_navigationArea != null)
				{
					break;
				}
			}
		}

		internal bool ShrinkNavigationArea()
		{
			if (_groupsRow != null)
			{
				double num = _groupsRow.ActualHeight - DragIngrementDn;
				if (num < 0.0)
				{
					num = 0.0;
				}
				GridLength height = new GridLength(num);
				_groupsRow.Height = height;
				(_groupsRow.Parent as Grid).UpdateLayout();
				if (_navigationArea != null)
				{
					_navigationArea.Height = height.Value;
				}
				return true;
			}
			return false;
		}

		internal bool ExtendNavigationArea()
		{
			if (_contentRow != null)
			{
				if (DragIngrementUp < double.Epsilon)
				{
					return false;
				}
				_ = _contentRow.ActualHeight;
				_ = _contentRow.MinHeight;
				GridLength height = new GridLength(_groupsRow.ActualHeight + DragIngrementUp);
				double num = _contentRow.ActualHeight - DragIngrementUp;
				if (num < 0.0)
				{
					num = 0.0;
				}
				if (new GridLength(num).Value < _contentRow.MinHeight)
				{
					return false;
				}
				_groupsRow.Height = height;
				_ = _groupsRow.Parent;
				if (_navigationArea != null)
				{
					_navigationArea.Height = height.Value;
					_navigationArea.UpdateLayout();
					if (Math.Abs(_groupsRow.ActualHeight - height.Value) > 0.1)
					{
						_navigationArea.Height = height.Value;
						_navigationArea.UpdateLayout();
					}
				}
				return true;
			}
			return false;
		}

		internal bool CanSetNavigationAreaSize(double newHeight)
		{
			if (_contentRow != null)
			{
				double num = newHeight - _groupsRow.ActualHeight;
				double num2 = _contentRow.ActualHeight - num;
				if (num2 < 0.0)
				{
					num2 = 0.0;
				}
				if (num2 < _contentRow.MinHeight)
				{
					return false;
				}
				return true;
			}
			return false;
		}

		internal bool SetNavigationAreaSize(double newHeight)
		{
			if (_contentRow == null)
			{
				OnApplyTemplate();
			}
			if (_contentRow != null)
			{
				if (!CanSetNavigationAreaSize(newHeight))
				{
					return false;
				}
				_groupsRow.Height = new GridLength(newHeight);
				Grid obj = _groupsRow.Parent as Grid;
				obj.InvalidateMeasure();
				obj.UpdateLayout();
				if (_navigationArea != null)
				{
					_navigationArea.Height = newHeight;
					_navigationArea.UpdateLayout();
					if (Math.Abs(_groupsRow.ActualHeight - newHeight) > 0.1)
					{
						_navigationArea.Height = newHeight;
						_navigationArea.UpdateLayout();
					}
				}
				return true;
			}
			return false;
		}

		internal void PerformMove(double x, double y)
		{
			if (GetValue(XamOutlookBar.OutlookBarProperty) is XamOutlookBar { VisibleGroupsCount: var visibleGroupsCount } xamOutlookBar && visibleGroupsCount != 0)
			{
				double num = y - Utilities.PointToScreenSafe(this, new Point(0.0, 0.0)).Y;
				double num2 = 24.0;
				if (xamOutlookBar.NavigationAreaGroups.Count > 0)
				{
					num2 = xamOutlookBar.NavigationAreaGroups[0].ActualHeight;
				}
				else if (xamOutlookBar.OverflowAreaGroups.Count > 0)
				{
					num2 = xamOutlookBar.OverflowAreaGroups[0].ActualHeight;
				}
				int num3 = (int)(num / num2);
				int num4 = xamOutlookBar.NavigationAreaGroups.Count - num3;
				if (num4 < 0)
				{
					num4 = 0;
				}
				if (num4 > visibleGroupsCount)
				{
					num4 = visibleGroupsCount;
				}
				xamOutlookBar.NavigationAreaMaxGroups = num4;
				xamOutlookBar.NavigationAreaMaxGroups = xamOutlookBar.NavigationAreaGroups.Count;
			}
		}

		private void GroupAreaSplitter_DragDelta(object sender, DragDeltaEventArgs e)
		{
			if (!(GetValue(XamOutlookBar.OutlookBarProperty) is XamOutlookBar xamOutlookBar))
			{
				return;
			}
			if (e.VerticalChange >= 0.0)
			{
				if (!(DragIngrementDn < 0.01) && !(e.VerticalChange < DragIngrementDn) && xamOutlookBar.NavigationAreaGroups.Count > 0)
				{
					xamOutlookBar.NavigationAreaMaxGroups = xamOutlookBar.NavigationAreaGroups.Count - 1;
				}
			}
			else if (!(DragIngrementUp < 0.01) && !(Math.Abs(e.VerticalChange) < DragIngrementUp))
			{
				xamOutlookBar.NavigationAreaMaxGroups = xamOutlookBar.NavigationAreaGroups.Count + 1;
				xamOutlookBar.NavigationAreaMaxGroups = xamOutlookBar.NavigationAreaGroups.Count;
			}
		}

		private bool CanExtendNavigationArea()
		{
			if (!(GetValue(XamOutlookBar.OutlookBarProperty) is XamOutlookBar xamOutlookBar))
			{
				return false;
			}
			if (_contentRow == null)
			{
				return false;
			}
			if (DragIngrementUp < double.Epsilon)
			{
				return false;
			}
			if (xamOutlookBar.NavigationAreaGroups.Count >= xamOutlookBar.Groups.Count)
			{
				return false;
			}
			new GridLength(_groupsRow.ActualHeight + DragIngrementUp);
			double num = _contentRow.ActualHeight - DragIngrementUp;
			if (num < 0.0)
			{
				num = 0.0;
			}
			if (new GridLength(num).Value < _contentRow.MinHeight)
			{
				return false;
			}
			return true;
		}
	}
	[DesignTimeVisible(false)]
	public class GroupsPresenter : ItemsControl
	{
		private XamOutlookBar _xamBar;

		public static readonly DependencyProperty OrientationProperty;

		[Bindable(true)]
		public Orientation Orientation
		{
			get
			{
				return (Orientation)GetValue(OrientationProperty);
			}
			set
			{
				SetValue(OrientationProperty, value);
			}
		}

		internal GroupsPanel GroupItemsPanel
		{
			get
			{
				if (VisualTreeHelper.GetChildrenCount(this) <= 0)
				{
					return null;
				}
				DependencyObject child = VisualTreeHelper.GetChild(this, 0);
				while (VisualTreeHelper.GetChildrenCount(child) > 0 && !(child is GroupsPanel))
				{
					child = VisualTreeHelper.GetChild(child, 0);
				}
				return child as GroupsPanel;
			}
		}

		internal double NextAreaItemHeight
		{
			get
			{
				if (OverflowItemsWithGroups.Count > 0)
				{
					return OverflowItemsWithGroups[0].DesiredNavigationHeight;
				}
				return 0.0;
			}
		}

		internal double NextAreaItemWidth
		{
			get
			{
				if (OverflowItemsWithGroups.Count > 0)
				{
					return OverflowItemsWithGroups[0].DesiredOverflowWidth;
				}
				return 0.0;
			}
		}

		internal double NextAreaSize
		{
			get
			{
				if (Orientation == Orientation.Vertical)
				{
					return NextAreaItemHeight;
				}
				return NextAreaItemWidth;
			}
		}

		internal int MaxVerticalItems
		{
			get
			{
				if (Orientation != Orientation.Vertical || _xamBar == null)
				{
					return int.MaxValue;
				}
				return _xamBar.NavigationAreaMaxGroupsResolved;
			}
		}

		internal OutlookBarGroupLocation DisplayedLocation
		{
			get
			{
				if (Orientation == Orientation.Vertical)
				{
					return OutlookBarGroupLocation.NavigationGroupArea;
				}
				return OutlookBarGroupLocation.OverflowArea;
			}
		}

		internal OutlookBarGroupCollection DisplayedItems
		{
			get
			{
				if (Orientation == Orientation.Vertical)
				{
					return _xamBar._navigationAreaGroups;
				}
				return _xamBar._overflowAreaGroups;
			}
		}

		internal OutlookBarGroupCollection OverflowItems
		{
			get
			{
				if (_xamBar == null)
				{
					_xamBar = GetValue(XamOutlookBar.OutlookBarProperty) as XamOutlookBar;
				}
				if (Orientation == Orientation.Vertical)
				{
					return _xamBar._overflowAreaGroups;
				}
				return _xamBar._contextMenuGroups;
			}
		}

		internal OutlookBarGroupCollection OverflowItemsWithGroups
		{
			get
			{
				if (OverflowItems.Count > 0)
				{
					return OverflowItems;
				}
				return _xamBar._contextMenuGroups;
			}
		}

		internal OutlookBarGroupLocation OverflowLocation
		{
			get
			{
				if (Orientation == Orientation.Vertical)
				{
					return OutlookBarGroupLocation.OverflowArea;
				}
				return OutlookBarGroupLocation.OverflowContextMenu;
			}
		}

		internal bool MustRemoveItems
		{
			get
			{
				GroupsPanel groupItemsPanel = GroupItemsPanel;
				if (groupItemsPanel == null)
				{
					return false;
				}
				if (groupItemsPanel.Overflow || base.Items.Count > MaxVerticalItems)
				{
					return base.Items.Count > 0;
				}
				return false;
			}
		}

		static GroupsPresenter()
		{
			OrientationProperty = DependencyProperty.Register("Orientation", typeof(Orientation), typeof(GroupsPresenter), new PropertyMetadata(Orientation.Vertical));
			ItemsPanelTemplate defaultValue = new ItemsPanelTemplate(new FrameworkElementFactory(typeof(GroupsPanel)));
			ItemsControl.ItemsPanelProperty.OverrideMetadata(typeof(GroupsPresenter), new FrameworkPropertyMetadata(defaultValue));
		}

		protected override Size ArrangeOverride(Size constraint)
		{
			Size result = base.ArrangeOverride(constraint);
			if (_xamBar != null)
			{
				GroupsPanel groupItemsPanel = GroupItemsPanel;
				if (groupItemsPanel != null && Orientation == Orientation.Vertical)
				{
					base.Height = groupItemsPanel.DesiredSize.Height;
				}
				_xamBar.SetSplitterIncrement();
			}
			return result;
		}

		protected override Size MeasureOverride(Size constraint)
		{
			Size result = base.MeasureOverride(constraint);
			GroupsPanel groupItemsPanel = GroupItemsPanel;
			if (groupItemsPanel == null || _xamBar == null)
			{
				return result;
			}
			bool mustRemoveItems = MustRemoveItems;
			groupItemsPanel.Measure(constraint);
			while (MustRemoveItems)
			{
				DropLastItem();
				groupItemsPanel.Measure(constraint);
				result = groupItemsPanel.DesiredSize;
			}
			while (groupItemsPanel.HasFreeSpace && !mustRemoveItems && OverflowItemsWithGroups.Count != 0 && !(groupItemsPanel.DeltaSize + 0.1 < NextAreaSize) && GetItemFromOverflow())
			{
				groupItemsPanel.Measure(constraint);
				if (MustRemoveItems)
				{
					DropLastItem();
					groupItemsPanel.Measure(constraint);
					break;
				}
				result = groupItemsPanel.DesiredSize;
			}
			return base.MeasureOverride(constraint);
		}

		public override void OnApplyTemplate()
		{
			_xamBar = GetValue(XamOutlookBar.OutlookBarProperty) as XamOutlookBar;
			GroupsPanel groupItemsPanel = GroupItemsPanel;
			if (groupItemsPanel != null)
			{
				groupItemsPanel.IsItemsHost = true;
			}
		}

		private void DropLastItem()
		{
			int index = base.Items.Count - 1;
			OutlookBarGroup outlookBarGroup = DisplayedItems[index];
			DisplayedItems.RemoveAt(index);
			outlookBarGroup.SetLocation(OverflowLocation);
			OverflowItems.Insert(0, outlookBarGroup);
			outlookBarGroup.InvalidateMeasure();
		}

		private bool GetItemFromOverflow()
		{
			OutlookBarGroup outlookBarGroup = OverflowItemsWithGroups[0];
			OverflowItemsWithGroups.RemoveAt(0);
			outlookBarGroup.SetLocation(DisplayedLocation);
			DisplayedItems.Add(outlookBarGroup);
			outlookBarGroup.InvalidateMeasure();
			return true;
		}

		private void SetOrientation(Orientation o)
		{
			GroupsPanel groupItemsPanel = GroupItemsPanel;
			if (groupItemsPanel == null)
			{
				FrameworkElementFactory visualTree = base.ItemsPanel.VisualTree;
				if (o == Orientation.Vertical)
				{
					visualTree.SetValue(FrameworkElement.HorizontalAlignmentProperty, HorizontalAlignment.Stretch);
				}
				else
				{
					visualTree.SetValue(FrameworkElement.HorizontalAlignmentProperty, HorizontalAlignment.Right);
				}
				visualTree.SetValue(StackPanel.OrientationProperty, o);
			}
			else
			{
				groupItemsPanel.SetValue(StackPanel.OrientationProperty, o);
				if (o == Orientation.Vertical)
				{
					groupItemsPanel.SetValue(FrameworkElement.HorizontalAlignmentProperty, HorizontalAlignment.Stretch);
				}
				else
				{
					groupItemsPanel.SetValue(FrameworkElement.HorizontalAlignmentProperty, HorizontalAlignment.Right);
				}
			}
		}
	}

	[TemplatePart(Name = "PART_OverflowMenu", Type = typeof(MenuItem))]
	[TemplatePart(Name = "PART_AddRemoveMenu", Type = typeof(MenuItem))]
	[DesignTimeVisible(false)]
	public class GroupOverflowArea : Control
	{
		private MenuItem _overflowMenu;

		private MenuItem _addRemoveMenu;

		private ObservableCollection<OutlookBarGroupProxy> _groupsAddRemove;

		static GroupOverflowArea()
		{
			FrameworkElement.DefaultStyleKeyProperty.OverrideMetadata(typeof(GroupOverflowArea), new FrameworkPropertyMetadata(typeof(GroupOverflowArea)));
		}

		public override void OnApplyTemplate()
		{
			base.OnApplyTemplate();
			if (_overflowMenu != null)
			{
				_overflowMenu.SubmenuOpened -= OverflowMenu_SubmenuOpened;
			}
			if (_addRemoveMenu != null)
			{
				_addRemoveMenu.SubmenuOpened -= AddRemoveMenu_SubmenuOpened;
				_addRemoveMenu.Loaded -= AddRemoveMenu_Loaded;
			}
			_overflowMenu = GetTemplateChild("PART_OverflowMenu") as MenuItem;
			_addRemoveMenu = GetTemplateChild("MenuAddRemove") as MenuItem;
			if (_addRemoveMenu == null)
			{
				_addRemoveMenu = GetTemplateChild("PART_AddRemoveMenu") as MenuItem;
			}
			if (_overflowMenu != null)
			{
				_overflowMenu.SubmenuOpened += OverflowMenu_SubmenuOpened;
			}
			if (_addRemoveMenu != null)
			{
				_addRemoveMenu.SubmenuOpened += AddRemoveMenu_SubmenuOpened;
				_addRemoveMenu.Loaded += AddRemoveMenu_Loaded;
			}
			RebuildAddRemoveMenu();
		}

		private void AddRemoveMenu_SubmenuOpened(object sender, RoutedEventArgs e)
		{
			if (e.Source == _addRemoveMenu)
			{
				RebuildAddRemoveMenu();
			}
		}

		private void AddRemoveMenu_Loaded(object sender, RoutedEventArgs e)
		{
			if (GetValue(XamOutlookBar.OutlookBarProperty) is XamOutlookBar xamOutlookBar && _addRemoveMenu.IsEnabled != xamOutlookBar.Groups.Count > 0)
			{
				RebuildAddRemoveMenu();
			}
		}

		private void RebuildAddRemoveMenu()
		{
			XamOutlookBar xamOutlookBar = GetValue(XamOutlookBar.OutlookBarProperty) as XamOutlookBar;
			if (_addRemoveMenu == null || xamOutlookBar == null)
			{
				return;
			}
			_groupsAddRemove = new ObservableCollection<OutlookBarGroupProxy>();
			foreach (OutlookBarGroup group in xamOutlookBar.Groups)
			{
				_groupsAddRemove.Add(new OutlookBarGroupProxy(group));
			}
			List<OutlookBarGroupProxy> list = new List<OutlookBarGroupProxy>(_groupsAddRemove);
			list.Sort(new GroupsOrderComparer());
			_groupsAddRemove = new ObservableCollection<OutlookBarGroupProxy>(list);
			_addRemoveMenu.IsEnabled = _groupsAddRemove.Count > 0;
			_addRemoveMenu.Items.Clear();
			for (int i = 0; i < _groupsAddRemove.Count; i++)
			{
				OutlookBarGroupProxy outlookBarGroupProxy = _groupsAddRemove[i];
				MenuItem menuItem = new MenuItem();
				Image image = new Image();
				image.Source = outlookBarGroupProxy.Group.SmallImageResolved;
				menuItem.Header = outlookBarGroupProxy.Group.Header;
				menuItem.IsChecked = outlookBarGroupProxy.IsSelected;
				menuItem.IsCheckable = true;
				menuItem.Icon = image;
				Binding binding = new Binding("Visibility");
				binding.Source = outlookBarGroupProxy.Group;
				binding.Mode = BindingMode.TwoWay;
				binding.Converter = new VisibilityToBooleanConverter();
				menuItem.SetBinding(MenuItem.IsCheckedProperty, binding);
				menuItem.SetResourceReference(FrameworkElement.StyleProperty, XamOutlookBar.OverflowMenuItemStyleKey);
				_addRemoveMenu.Items.Add(menuItem);
			}
		}

		private void OverflowMenu_SubmenuOpened(object sender, RoutedEventArgs e)
		{
			if (e.Source == _overflowMenu)
			{
				RebuildOverflowMenu();
			}
		}

		private void RebuildOverflowMenu()
		{
			ClearMenuGroups();
			AddMenuGroups();
			if (GetValue(XamOutlookBar.OutlookBarProperty) is XamOutlookBar xamOutlookBar)
			{
				xamOutlookBar.SaveGroupOrderAndVisibility();
			}
		}

		private void AddMenuGroups()
		{
			XamOutlookBar xamOutlookBar = GetValue(XamOutlookBar.OutlookBarProperty) as XamOutlookBar;
			if (_overflowMenu == null || xamOutlookBar == null)
			{
				return;
			}
			for (int i = 0; i < xamOutlookBar._contextMenuGroups.Count; i++)
			{
				OutlookBarGroup outlookBarGroup = xamOutlookBar._contextMenuGroups[i];
				if (outlookBarGroup.Visibility != Visibility.Collapsed)
				{
					XamOutlookBar.SetOutlookBar(outlookBarGroup, xamOutlookBar);
					MenuItem menuItem = new MenuItem();
					Image image = new Image();
					image.Source = outlookBarGroup.SmallImageResolved;
					menuItem.Header = outlookBarGroup.Header;
					menuItem.Icon = image;
					menuItem.IsCheckable = false;
					menuItem.Command = OutlookBarCommands.SelectGroupCommand;
					menuItem.CommandParameter = outlookBarGroup;
					menuItem.CommandTarget = xamOutlookBar;
					menuItem.Tag = outlookBarGroup;
					menuItem.SetResourceReference(FrameworkElement.StyleProperty, XamOutlookBar.OverflowMenuItemStyleKey);
					_overflowMenu.Items.Add(menuItem);
				}
			}
		}

		private void ClearMenuGroups()
		{
			if (_overflowMenu == null)
			{
				return;
			}
			for (int i = 0; i < _overflowMenu.Items.Count; i++)
			{
				if (_overflowMenu.Items[i] is MenuItem menuItem && menuItem.Tag is OutlookBarGroup)
				{
					menuItem.Header = null;
					_overflowMenu.Items.RemoveAt(i);
					i--;
				}
			}
		}
	}
	[TemplatePart(Name = "PART_ListBox", Type = typeof(Grid))]
	[DesignTimeVisible(false)]
	public class NavigationPaneOptionsControl : Control
	{
		private ListBox _lbGroups;

		private XamOutlookBar _xamBar;

		private ObservableCollection<ListBoxItem> _listBoxGroupProxyItems;

		public XamOutlookBar OutlookBar
		{
			get
			{
				return _xamBar;
			}
			set
			{
				_xamBar = value;
				if (_xamBar != null && _xamBar.NavigationPaneOptionsControlStyle != null)
				{
					base.Style = _xamBar.NavigationPaneOptionsControlStyle;
				}
				if (base.IsLoaded)
				{
					LoadGroups();
				}
			}
		}

		static NavigationPaneOptionsControl()
		{
			FrameworkElement.DefaultStyleKeyProperty.OverrideMetadata(typeof(NavigationPaneOptionsControl), new FrameworkPropertyMetadata(typeof(NavigationPaneOptionsControl)));
			CommandBinding commandBinding = new CommandBinding(NavigationPaneOptionsControlCommands.CommitChangesAndCloseCommand, ExecutedCommitChangesAndCloseCommand, CanExecuteCommand);
			CommandManager.RegisterClassCommandBinding(typeof(NavigationPaneOptionsControl), commandBinding);
			CommandBinding commandBinding2 = new CommandBinding(NavigationPaneOptionsControlCommands.MoveSelectedDownCommand, ExecutedMoveSelectedDownCommand, CanExecuteCommand);
			CommandManager.RegisterClassCommandBinding(typeof(NavigationPaneOptionsControl), commandBinding2);
			CommandBinding commandBinding3 = new CommandBinding(NavigationPaneOptionsControlCommands.MoveSelectedUpCommand, ExecutedMoveSelectedUpCommand, CanExecuteCommand);
			CommandManager.RegisterClassCommandBinding(typeof(NavigationPaneOptionsControl), commandBinding3);
			CommandBinding commandBinding4 = new CommandBinding(NavigationPaneOptionsControlCommands.ResetGroupSequenceAndVisibilityCommand, ExecutedResetGroupSequenceAndVisibiltyCommand, CanExecuteCommand);
			CommandManager.RegisterClassCommandBinding(typeof(NavigationPaneOptionsControl), commandBinding4);
		}

		public NavigationPaneOptionsControl()
		{
			base.Loaded += delegate
			{
				FocusFirstGroup();
			};
		}

		public override void OnApplyTemplate()
		{
			base.OnApplyTemplate();
			_lbGroups = GetTemplateChild("PART_ListBox") as ListBox;
			LoadGroups();
		}

		protected override void OnPreviewKeyDown(KeyEventArgs e)
		{
			if (e.Key == Key.Space && Keyboard.FocusedElement is ListBoxItem && (Keyboard.FocusedElement as ListBoxItem).Content is OutlookBarGroupProxy outlookBarGroupProxy && !(Keyboard.FocusedElement is CheckBox))
			{
				outlookBarGroupProxy.IsSelected = !outlookBarGroupProxy.IsSelected;
			}
			base.OnPreviewKeyDown(e);
		}

		protected override void OnPreviewMouseDown(MouseButtonEventArgs e)
		{
			if (_lbGroups != null && ItemsControl.ContainerFromElement(_lbGroups, e.OriginalSource as DependencyObject) is ListBoxItem listBoxItem && listBoxItem.Content is OutlookBarGroupProxy)
			{
				_lbGroups.SelectedItem = listBoxItem;
			}
			base.OnPreviewMouseDown(e);
		}

		protected override void OnPreviewMouseDoubleClick(MouseButtonEventArgs e)
		{
			if (_lbGroups != null && ItemsControl.ContainerFromElement(_lbGroups, e.OriginalSource as DependencyObject) is ListBoxItem listBoxItem)
			{
				DependencyObject parent = listBoxItem.InputHitTest(e.GetPosition(listBoxItem)) as DependencyObject;
				if (!(FindParent(parent, typeof(CheckBox)) is CheckBox) && listBoxItem.Content is OutlookBarGroupProxy outlookBarGroupProxy)
				{
					outlookBarGroupProxy.IsSelected = !outlookBarGroupProxy.IsSelected;
				}
			}
			base.OnPreviewMouseDoubleClick(e);
		}

		private void LoadGroups()
		{
			if (_xamBar == null || _lbGroups == null)
			{
				return;
			}
			_listBoxGroupProxyItems = new ObservableCollection<ListBoxItem>();
			foreach (OutlookBarGroup group in _xamBar.Groups)
			{
				_listBoxGroupProxyItems.Add(new ListBoxItem
				{
					Content = new OutlookBarGroupProxy(group)
				});
			}
			Binding binding = new Binding();
			binding.Source = _listBoxGroupProxyItems;
			_lbGroups.SetBinding(ItemsControl.ItemsSourceProperty, binding);
			if (_lbGroups.Items.Count > 0)
			{
				_lbGroups.SelectedIndex = 0;
			}
			FocusFirstGroup();
		}

		private void FocusFirstGroup()
		{
			if (_lbGroups != null && _lbGroups.Items.CurrentItem != null)
			{
				ListBoxItem listBoxItem = (ListBoxItem)_lbGroups.Items.CurrentItem;
				if (listBoxItem != null)
				{
					Keyboard.Focus(listBoxItem);
					listBoxItem.Focus();
				}
			}
		}

		private void ResetToInitialState()
		{
			if (_xamBar == null)
			{
				return;
			}
			List<OutlookBarGroupProxy> list = new List<OutlookBarGroupProxy>();
			foreach (ListBoxItem listBoxGroupProxyItem in _listBoxGroupProxyItems)
			{
				OutlookBarGroupProxy item = listBoxGroupProxyItem.Content as OutlookBarGroupProxy;
				list.Add(item);
			}
			list.Sort(new GroupsOrderComparer());
			_listBoxGroupProxyItems.Clear();
			foreach (OutlookBarGroupProxy item2 in list)
			{
				item2.IsSelected = item2.Group.IsVisibleOnStart;
				_listBoxGroupProxyItems.Add(new ListBoxItem
				{
					Content = item2
				});
			}
			if (_listBoxGroupProxyItems.Count > 0)
			{
				_lbGroups.SelectedIndex = 0;
			}
			_xamBar.RaiseGroupsReset(new RoutedEventArgs());
		}

		private void UpdateXamOutlookBar()
		{
			try
			{
				if (_xamBar == null)
				{
					return;
				}
				_xamBar.Groups.CanEdit = true;
				_xamBar._contextMenuGroups.Clear();
				_xamBar._overflowAreaGroups.Clear();
				_xamBar._navigationAreaGroups.Clear();
				OutlookBarGroupCollection outlookBarGroupCollection = new OutlookBarGroupCollection(_xamBar);
				outlookBarGroupCollection.CanEdit = true;
				foreach (ListBoxItem listBoxGroupProxyItem in _listBoxGroupProxyItems)
				{
					OutlookBarGroupProxy outlookBarGroupProxy = listBoxGroupProxyItem.Content as OutlookBarGroupProxy;
					outlookBarGroupProxy.Group.Visibility = ((!outlookBarGroupProxy.IsSelected) ? Visibility.Collapsed : Visibility.Visible);
					outlookBarGroupCollection.Add(outlookBarGroupProxy.Group);
				}
				_xamBar.SetGroups(outlookBarGroupCollection);
				_xamBar.VerifySelectedGroup(forceSelectedGroup: true);
			}
			finally
			{
				_xamBar.Groups.CanEdit = false;
			}
		}

		private DependencyObject FindParent(DependencyObject parent, Type parentType)
		{
			while (parent != null)
			{
				if (parent.GetType() == parentType)
				{
					return parent;
				}
				parent = VisualTreeHelper.GetParent(parent);
			}
			return null;
		}

		private static void CanExecuteCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			NavigationPaneOptionsControl navigationPaneOptionsControl = sender as NavigationPaneOptionsControl;
			e.CanExecute = false;
			if (navigationPaneOptionsControl._xamBar != null && navigationPaneOptionsControl != null && navigationPaneOptionsControl._lbGroups != null)
			{
				if (e.Command == NavigationPaneOptionsControlCommands.CommitChangesAndCloseCommand)
				{
					e.CanExecute = true;
				}
				else if (e.Command == NavigationPaneOptionsControlCommands.ResetGroupSequenceAndVisibilityCommand)
				{
					e.CanExecute = true;
				}
				else if (e.Command == NavigationPaneOptionsControlCommands.MoveSelectedDownCommand)
				{
					e.CanExecute = navigationPaneOptionsControl._lbGroups.SelectedIndex >= 0 && navigationPaneOptionsControl._lbGroups.SelectedIndex < navigationPaneOptionsControl._lbGroups.Items.Count - 1;
				}
				else if (e.Command == NavigationPaneOptionsControlCommands.MoveSelectedUpCommand)
				{
					e.CanExecute = navigationPaneOptionsControl._lbGroups.SelectedIndex > 0;
				}
			}
		}

		private static void ExecutedCommitChangesAndCloseCommand(object sender, ExecutedRoutedEventArgs e)
		{
			if (sender is NavigationPaneOptionsControl navigationPaneOptionsControl)
			{
				navigationPaneOptionsControl.UpdateXamOutlookBar();
				ToolWindow toolWindow = ToolWindow.GetToolWindow(navigationPaneOptionsControl);
				if (toolWindow != null)
				{
					toolWindow.DialogResult = true;
				}
			}
		}

		private static void ExecutedMoveSelectedDownCommand(object sender, ExecutedRoutedEventArgs e)
		{
			if (sender is NavigationPaneOptionsControl { _lbGroups: not null } navigationPaneOptionsControl)
			{
				navigationPaneOptionsControl.SwapGroups(navigationPaneOptionsControl._lbGroups.SelectedIndex, navigationPaneOptionsControl._lbGroups.SelectedIndex + 1);
			}
		}

		private static void ExecutedMoveSelectedUpCommand(object sender, ExecutedRoutedEventArgs e)
		{
			if (sender is NavigationPaneOptionsControl { _lbGroups: not null } navigationPaneOptionsControl)
			{
				navigationPaneOptionsControl.SwapGroups(navigationPaneOptionsControl._lbGroups.SelectedIndex, navigationPaneOptionsControl._lbGroups.SelectedIndex - 1);
			}
		}

		private void SwapGroups(int selectedIndex, int nextIndex)
		{
			_lbGroups.SelectedItem = null;
			_listBoxGroupProxyItems.Move(selectedIndex, nextIndex);
			_lbGroups.SelectedIndex = nextIndex;
		}

		private static void ExecutedResetGroupSequenceAndVisibiltyCommand(object sender, ExecutedRoutedEventArgs e)
		{
			if (sender is NavigationPaneOptionsControl navigationPaneOptionsControl)
			{
				navigationPaneOptionsControl.ResetToInitialState();
			}
		}
	}

		protected override IEnumerator LogicalChildren
		{
			get
			{
				IEnumerator enumerator = base.LogicalChildren;
				if (enumerator != null && SuppressLogicalContent)
				{
					List<object> list = new List<object>();
					while (enumerator.MoveNext())
					{
						if (enumerator.Current != base.Content)
						{
							list.Add(enumerator.Current);
						}
					}
					enumerator = list.GetEnumerator();
				}
				return enumerator;
			}
		}

		protected override Size MeasureOverride(Size constraint)
		{
			bool num = _lastMeasuredLocation != Location;
			if (num && double.IsInfinity(constraint.Width) && OutlookBar != null)
			{
				double actualWidth = OutlookBar.ActualWidth;
				Size constraint2 = new Size(actualWidth, constraint.Height);
				constraint2 = base.MeasureOverride(constraint2);
			}
			Size result = base.MeasureOverride(constraint);
			if (Location == OutlookBarGroupLocation.NavigationGroupArea && ((_calculatedHeight < 1.0 && !_heightIsFromArrange) || _calculatedHeight < result.Height))
			{
				_calculatedHeight = result.Height;
			}
			if (Location == OutlookBarGroupLocation.OverflowArea && ((_calculatedWidth < 1.0 && !_widthIsFromArrange) || _calculatedWidth < result.Width))
			{
				_calculatedWidth = result.Width;
			}
			if (num)
			{
				if (_lastMeasuredLocation == OutlookBarGroupLocation.NavigationGroupArea)
				{
					_calculatedHeight = base.DesiredSize.Height;
				}
				if (_lastMeasuredLocation == OutlookBarGroupLocation.OverflowArea)
				{
					_calculatedWidth = base.DesiredSize.Width;
				}
			}
			_lastMeasuredLocation = Location;
			return result;
		}

		private static object OnCoerceIsSelectedCallback(DependencyObject obj, object value)
		{
			OutlookBarGroup outlookBarGroup = (OutlookBarGroup)obj;
			if (outlookBarGroup.OutlookBar == null)
			{
				return value;
			}
			if (!outlookBarGroup.OutlookBar.Groups.Contains(outlookBarGroup))
			{
				return false;
			}
			return outlookBarGroup.OutlookBar.SelectGroup(outlookBarGroup, (bool)value);
		}

		protected virtual void SetVisualState(bool useTransitions)
		{
			if (!base.IsEnabled)
			{
				VisualStateManager.GoToState(this, "Disabled", useTransitions);
			}
			else if (IsMouseOverGroup)
			{
				if (IsMouseLeftButtonPressed && !IsSelected)
				{
					VisualStateUtilities.GoToState(this, useTransitions, "Pressed", "MouseOver", "Normal");
				}
				else
				{
					VisualStateUtilities.GoToState(this, useTransitions, "MouseOver", "Normal");
				}
			}
			else
			{
				VisualStateManager.GoToState(this, "Normal", useTransitions);
			}
			switch (Location)
			{
			case OutlookBarGroupLocation.NavigationGroupArea:
				VisualStateManager.GoToState(this, "NavigationArea", useTransitions);
				break;
			case OutlookBarGroupLocation.OverflowArea:
			case OutlookBarGroupLocation.OverflowContextMenu:
				VisualStateManager.GoToState(this, "OverflowArea", useTransitions);
				break;
			}
			XamOutlookBar outlookBar = OutlookBar;
			if (outlookBar != null && outlookBar.IsMinimized)
			{
				VisualStateManager.GoToState(this, "Minimized", useTransitions);
			}
			else
			{
				VisualStateManager.GoToState(this, "NotMinimized", useTransitions);
			}
			if (IsSelected)
			{
				VisualStateManager.GoToState(this, "Selected", useTransitions);
			}
			else
			{
				VisualStateManager.GoToState(this, "Unselected", useTransitions);
			}
			if (outlookBar != null && outlookBar.VerticalSplitterLocation == VerticalSplitterLocation.Left)
			{
				VisualStateManager.GoToState(this, "Left", useTransitions);
			}
			else
			{
				VisualStateManager.GoToState(this, "Right", useTransitions);
			}
		}

		internal static void OnVisualStatePropertyChanged(DependencyObject target, DependencyPropertyChangedEventArgs e)
		{
			if (target is OutlookBarGroup outlookBarGroup)
			{
				outlookBarGroup.UpdateVisualStates();
			}
		}

		protected internal void UpdateVisualStates()
		{
			UpdateVisualStates(useTransitions: true);
		}

		protected internal void UpdateVisualStates(bool useTransitions)
		{
			if (_hasVisualStateGroups)
			{
				if (!base.IsLoaded)
				{
					useTransitions = false;
				}
				SetVisualState(useTransitions);
			}
		}

		private bool IsMouseOverThis(MouseEventArgs e)
		{
			Point position = e.GetPosition(this);
			DependencyObject dependencyObject = VisualTreeHelper.HitTest(this, position)?.VisualHit;
			while (dependencyObject != null && dependencyObject != this)
			{
				dependencyObject = VisualTreeHelper.GetParent(dependencyObject);
			}
			return dependencyObject != null;
		}

		private void OnIsMinimizedChanged()
		{
			UpdateVisualStates();
		}

		internal void ResolveSmallImage()
		{
			XamOutlookBar outlookBar = OutlookBar;
			if (outlookBar != null)
			{
				_isMinimizedTracker = new PropertyValueTracker(outlookBar, XamOutlookBar.IsMinimizedProperty, OnIsMinimizedChanged, callAsynchronously: false);
			}
			if (SmallImage == null)
			{
				SmallImageResolved = DefaultSmallImage;
			}
		}

		private void NotifyPropertyChanged(string info)
		{
			if (this.PropertyChanged != null)
			{
				this.PropertyChanged(this, new PropertyChangedEventArgs(info));
			}
		}

		protected override void OnPropertyChanged(DependencyPropertyChangedEventArgs e)
		{
			base.OnPropertyChanged(e);
			NotifyPropertyChanged(e.Property.Name);
		}

		{
			CheckIsUsingGroupsSource();
			base.RemoveItem(index);
		}

		public override void InsertRange(int index, IEnumerable<OutlookBarGroup> collection)
		{
			CheckIsUsingGroupsSource();
			base.InsertRange(index, collection);
		}

		protected override void MoveItem(int oldIndex, int newIndex)
		{
			CheckIsUsingGroupsSource();
			base.MoveItem(oldIndex, newIndex);
		}

		public override void RemoveRange(int index, int count)
		{
			CheckIsUsingGroupsSource();
			base.RemoveRange(index, count);
		}

		protected override void SetItem(int index, OutlookBarGroup item)
		{
			CheckIsUsingGroupsSource();
			base.SetItem(index, item);
		}

		protected override void OnItemPropertyChanged(ItemPropertyChangedEventArgs e)
		{
			if (e.PropertyName == "Key")
			{
				OutlookBarGroup outlookBarGroup = e.Item as OutlookBarGroup;
				foreach (KeyValuePair<string, OutlookBarGroup> item in _keysHashtable)
				{
					if (item.Value == outlookBarGroup)
					{
						_keysHashtable.Remove(item.Key);
						break;
					}
				}
				AddKey(outlookBarGroup);
			}
			base.OnItemPropertyChanged(e);
		}

		protected override void OnCollectionChanged(NotifyCollectionChangedEventArgs e)
		{
			switch (e.Action)
			{
			case NotifyCollectionChangedAction.Remove:
				if (e.OldItems == null)
				{
					break;
				}
				foreach (OutlookBarGroup oldItem in e.OldItems)
				{
					RemoveKey(oldItem);
				}
				break;
			case NotifyCollectionChangedAction.Add:
			case NotifyCollectionChangedAction.Replace:
				if (e.OldItems != null)
				{
					foreach (OutlookBarGroup oldItem2 in e.OldItems)
					{
						RemoveKey(oldItem2);
					}
				}
				if (e.NewItems == null)
				{
					break;
				}
				foreach (OutlookBarGroup newItem in e.NewItems)
				{
					AddKey(newItem);
				}
				break;
			case NotifyCollectionChangedAction.Reset:
				ResetKeysHashtable();
				break;
			}
			base.OnCollectionChanged(e);
		}

		public bool ContainsKey(string key)
		{
			return _keysHashtable.ContainsKey(key);
		}

		private void CheckIsUsingGroupsSource()
		{
			if (_xob != null && _xob.GroupsSource != null && !CanEdit)
			{
				throw new InvalidOperationException(XamOutlookBar.GetString("LE_CannotModifyBoundGroups"));
			}
		}

		private void AddKey(OutlookBarGroup gr)
		{
			CheckKey(gr);
			if (!string.IsNullOrEmpty(gr.Key))
			{
				_keysHashtable[gr.Key] = gr;
			}
		}

		private void RemoveKey(OutlookBarGroup gr)
		{
			if (!string.IsNullOrEmpty(gr.Key))
			{
				_keysHashtable.Remove(gr.Key);
			}
		}

		private void ResetKeysHashtable()
		{
			_keysHashtable.Clear();
			foreach (OutlookBarGroup item in base.Items)
			{
				AddKey(item);
			}
		}

		private void CheckKey(OutlookBarGroup gr)
		{
			if (IsDuplicateKey(gr))
			{
				OnDuplicateKey(gr);
			}
		}

		private void OnDuplicateKey(OutlookBarGroup gr)
		{
			string format = string.Format(XamOutlookBar.GetString("LE_DuplicateKey"), gr.Header, gr.Key);
			gr.Key = "";
			throw new Exception(string.Format(format));
		}

		private bool IsDuplicateKey(OutlookBarGroup gr)
		{
			if (string.IsNullOrEmpty(gr.Key))
			{
				return false;
			}
			if (!_keysHashtable.ContainsKey(gr.Key))
			{
				return false;
			}
			OutlookBarGroup outlookBarGroup = _keysHashtable[gr.Key];
			if (outlookBarGroup == null)
			{
				return false;
			}
			return !outlookBarGroup.Equals(gr);
		}

		internal void Refresh()
		{
			NotifyCollectionChangedEventArgs e = new NotifyCollectionChangedEventArgs(NotifyCollectionChangedAction.Reset);
			OnCollectionChanged(e);
		}
	}

		private class GroupsItemsSource : ItemsControl
		{
			private OutlookBarGroupCollection _groups;

			public GroupsItemsSource(OutlookBarGroupCollection groups)
			{
				_groups = groups;
			}

			protected override void OnItemsChanged(NotifyCollectionChangedEventArgs e)
			{
				try
				{
					_groups.CanEdit = true;
					switch (e.Action)
					{
					case NotifyCollectionChangedAction.Add:
					{
						for (int j = 0; j < e.NewItems.Count; j++)
						{
							object obj = e.NewItems[j];
							if (obj != null)
							{
								_groups.Insert(j + e.NewStartingIndex, CreateOutlookBarGroup(obj));
								RebuildAllGroups();
							}
						}
						break;
					}
					case NotifyCollectionChangedAction.Move:
						RebuildAllGroups();
						break;
					case NotifyCollectionChangedAction.Remove:
					{
						for (int i = 0; i < e.OldItems.Count; i++)
						{
							OutlookBarGroup outlookBarGroup = e.OldItems[i] as OutlookBarGroup;
							if (outlookBarGroup == null)
							{
								string key = (i + e.OldStartingIndex).ToString();
								outlookBarGroup = _groups[key];
							}
							if (outlookBarGroup != null)
							{
								_groups.Remove(outlookBarGroup);
							}
							RebuildAllGroups();
						}
						break;
					}
					case NotifyCollectionChangedAction.Replace:
						RebuildAllGroups();
						break;
					case NotifyCollectionChangedAction.Reset:
						RebuildAllGroups();
						break;
					}
				}
				finally
				{
					_groups.CanEdit = false;
				}
				base.OnItemsChanged(e);
			}

			protected override void OnItemsSourceChanged(IEnumerable oldValue, IEnumerable newValue)
			{
				base.OnItemsSourceChanged(oldValue, newValue);
			}

			internal void RebuildAllGroups()
			{
				bool flag = false;
				if (_groups.Count != base.Items.Count)
				{
					flag = true;
				}
				else
				{
					for (int i = 0; i < base.Items.Count; i++)
					{
						if (base.Items[i] != _groups[i].DataContext)
						{
							flag = true;
							break;
						}
					}
				}
				if (!flag)
				{
					return;
				}
				_groups.Clear();
				foreach (object item in (IEnumerable)base.Items)
				{
					if (item != null)
					{
						_groups.Add(CreateOutlookBarGroup(item));
					}
				}
			}

			private OutlookBarGroup CreateOutlookBarGroup(object o)
			{
				if (o is OutlookBarGroup)
				{
					return o as OutlookBarGroup;
				}
				return new OutlookBarGroup
				{
					DataContext = o,
					Key = Guid.NewGuid().ToString()
				};
			}
		}

		protected override Size MeasureOverride(Size constraint)
		{
			Size result = base.MeasureOverride(constraint);
			double minimizedWidth = _minimizedWidth;
			_minimizedWidth = GetMinimizedWidth();

		public override void OnApplyTemplate()
		{
			base.OnApplyTemplate();
			if (_verticalSplitter != null)
			{
				_verticalSplitter.DragDelta -= VerticalSplitter_DragDelta;
				_verticalSplitter.DragCompleted -= VerticalSplitter_DragCompleted;
				_verticalSplitter.DragStarted -= VerticalSplitter_DragStarted;
			}
			if (_grid != null)
			{
				_grid.SizeChanged -= Grid_SizeChanged;
			}
			_navigationAreaIC = GetTemplateChild("PART_NavigationArea") as GroupsPresenter;
			_overflowArea = GetTemplateChild("PART_OverflowArea") as GroupOverflowArea;
			_splitter = GetTemplateChild("PART_Splitter") as GroupAreaSplitter;
			_selectedGroupContent = GetTemplateChild("PART_SelectedGroupContent") as SelectedGroupContent;
			_selectedGroupHeader = GetTemplateChild("PART_SelectedGroupHeader") as SelectedGroupHeader;
			_verticalSplitter = GetTemplateChild("PART_VerticalSplitter") as Thumb;
			if (_verticalSplitter != null)
			{
				_verticalSplitter.DragDelta += VerticalSplitter_DragDelta;
				_verticalSplitter.DragCompleted += VerticalSplitter_DragCompleted;
				_verticalSplitter.DragStarted += VerticalSplitter_DragStarted;
			}
			InitializeGroups(moveToNavigationArea: true, fillNavAreaWithDesiredNumberOfGroups: false);
			if (base.IsLoaded)
			{
				_splitter.OnApplyTemplate();
				_splitter.InvalidateMeasure();
				GetMaxGroupsInTheNavigationArea(NavigationAreaMaxGroupsResolved, clearGroupsBefore: true);
			}
			UpdateSelectedGroupContent();
			_grid = GetTemplateChild("PART_Grid") as Grid;
			if (_grid != null)
			{
				_grid.SizeChanged += Grid_SizeChanged;
			}
			_hasVisualStateGroups = VisualStateUtilities.GetHasVisualStateGroups(this);
			UpdateVisualStates(useTransitions: false);
			if (_selectedGroupHeader != null)
			{
				_selectedGroupHeader.SetIsMinimized(IsMinimized);
			}
		}

		protected override void OnRenderSizeChanged(SizeChangedInfo sizeInfo)
		{
			base.OnRenderSizeChanged(sizeInfo);
			SetNavigationAreaSize(sizeInfo.NewSize.Height - sizeInfo.PreviousSize.Height);
			_selectionInitialized = Groups.Count == 0;
			if (sizeInfo.WidthChanged && _selectionInitialized)
			{
				setMinimizedByWidth(sizeInfo.NewSize.Width);
			}
		}

		private bool setMinimizedByWidth(double newWidth)
		{
			if (!_inferMinimizedStateFromWidth)
			{
				return false;
			}
			if (_animationInProcess)
			{
				return false;
			}
			bool isMinimized = IsMinimized;
			if ((double.IsNaN(newWidth) || newWidth > minExpandedWidth) && IsMinimized)
			{
				IsMinimized = false;
			}
			else if (newWidth < minExpandedWidth && !IsMinimized)
			{
				IsMinimized = newWidth > 0.0;
			}
			if (!IsMinimized)
			{
				_lastExpandedWidth = newWidth;
			}
			return isMinimized != IsMinimized;
		}

		{
			double num = ((VerticalSplitterLocation == VerticalSplitterLocation.Right && base.FlowDirection == FlowDirection.LeftToRight) ? e.HorizontalChange : (0.0 - e.HorizontalChange));
			double num2 = base.ActualWidth + num;
			if (IsMinimized && num2 >= minExpandedWidth)
			{
				bool useAnimations = _useAnimations;
				_useAnimations = VerticalSplitterResizeMode == SplitterResizeMode.Deferred;
				_lastExpandedWidth = num2;
				IsMinimized = false;
				_useAnimations = useAnimations;
			}
			if (num2 >= 1.0 && VerticalSplitterResizeMode == SplitterResizeMode.Deferred)
			{
				if (!_animationInProcess)
				{
					BeginAnimation(FrameworkElement.WidthProperty, null);
				}
				if (!IsMinimized && num2 < minExpandedWidth)
				{
					IsMinimized = true;
				}
				else
				{
					base.Width = num2;
				}
			}
			RemoveAdorner();
			_inferMinimizedStateFromWidth = false;
		}

		private void AddRemoveLogicalChildren(NotifyCollectionChangedEventArgs e)
		{
			switch (e.Action)
			{
			case NotifyCollectionChangedAction.Add:
			case NotifyCollectionChangedAction.Remove:
			case NotifyCollectionChangedAction.Replace:
			case NotifyCollectionChangedAction.Reset:
				if (e.OldItems != null)
				{
					foreach (OutlookBarGroup oldItem in e.OldItems)
					{
						RemoveFromLogicalTree(oldItem);
					}
				}
				if (e.NewItems == null)
				{
					break;
				}
				{
					foreach (OutlookBarGroup newItem in e.NewItems)
					{
						AddToLogicalTree(newItem);
					}
					break;
				}
			case NotifyCollectionChangedAction.Move:
				break;
			}
		}

		private DependencyObject FindRootInLogicalTree(DependencyObject obj)
		{
			DependencyObject dependencyObject = obj;
			while (obj != null)
			{
				dependencyObject = obj;
				if (dependencyObject == this)
				{
					return this;
				}
				obj = LogicalTreeHelper.GetParent(obj);
			}
			return dependencyObject;
		}

		private void AddToLogicalTree(DependencyObject obj)
		{
			obj = FindRootInLogicalTree(obj);
			if (obj != null && obj != this)
			{
				if (_logicalChildren == null)
				{
					_logicalChildren = new List<DependencyObject>();
				}
				else if (_logicalChildren.IndexOf(obj) >= 0)
				{
					return;
				}
				AddLogicalChild(obj);
				_logicalChildren.Add(obj);
			}
		}

		private void RemoveFromLogicalTree(DependencyObject obj)
		{
			if (obj != null)
			{
				if (_logicalChildren != null)
				{
					_logicalChildren.Remove(obj);
				}
				RemoveLogicalChild(obj);
			}
		}

		{

			_animationInProcess = true;
			_sbExpand.Completed += ExpandStoryboard_Completed;
			_sbExpand.Begin(this);
		}

		private int GetMaxGroupsInTheNavigationArea(int desiredCount, bool clearGroupsBefore)
		{
			if (clearGroupsBefore && desiredCount > 0)
			{
				GetMaxGroupsInTheNavigationArea(0, clearGroupsBefore: false);
			}
			int num = ((desiredCount < 0) ? Groups.Count : (desiredCount - _navigationAreaGroups.Count));
			if (base.IsLoaded && _splitter != null)
			{
				int num2 = ((desiredCount < 0) ? Groups.Count : Math.Min(desiredCount, Groups.Count));
				int count = Groups.Count;
				double navigationAreaSize = 0.0;
				double num3 = 0.0;
				for (int i = 0; i < count; i++)
				{
					if (num2 <= 0)
					{
						break;
					}
					if (Groups[i].Visibility != Visibility.Collapsed)
					{
						num2--;
						num3 += Groups[i].DesiredNavigationHeight;
						if (!_splitter.CanSetNavigationAreaSize(num3))
						{
							break;
						}
						navigationAreaSize = num3;
					}
				}
				if (_splitter.SetNavigationAreaSize(navigationAreaSize))
				{
					SetSplitterIncrement();
					return desiredCount;
				}
				return _navigationAreaGroups.Count;
			}
			if (num > 0)
			{
				for (int j = 0; j < num; j++)
				{
					if (!ExtendNavigationArea())
					{
						break;
					}
					UpdateLayout();
					SetSplitterIncrement();
				}
			}
			else if (num < 0)
			{
				num *= -1;
				for (int j = 0; j < num; j++)
				{
					if (!ShrinkNavigationArea())
					{
						break;
					}
					UpdateLayout();
					SetSplitterIncrement();
					if (_navigationAreaGroups.Count == desiredCount)
					{
						break;
					}
				}
			}
			SetSplitterIncrement();
			return _navigationAreaGroups.Count;
		}

		private bool ExecuteCommandImpl(RoutedCommand command, object parameter, object source, object originalSource)
		{
			if (command == null)
			{
				throw new ArgumentNullException("command");
			}
			ExecutingCommandEventArgs e = new ExecutingCommandEventArgs(command);
			if (!RaiseExecutingCommand(e))
			{
				return !e.ContinueKeyRouting;
			}
			bool flag = false;
			if (command == OutlookBarCommands.GroupMoveDownCommand)
			{
				flag = ExecuteGroupMoveDownCommand(source, originalSource, parameter);
			}
			else if (command == OutlookBarCommands.GroupMoveUpCommand)
			{
				flag = ExecuteGroupMoveUpCommand(source, originalSource, parameter);
			}
			else if (command == OutlookBarCommands.SelectGroupCommand)
			{
				flag = ExecuteSelectGroupCommand(source, originalSource, parameter);
			}
			else if (command == OutlookBarCommands.ShowFewerButtonsCommand)
			{
				flag = ExecuteShowFewerButtonsCommand(source, originalSource, parameter);
			}
			else if (command == OutlookBarCommands.ShowMoreButtonsCommand)
			{
				flag = ExecuteShowMoreButtonsCommand(source, originalSource, parameter);
			}
			else if (command == OutlookBarCommands.ShowOptionsCommand)
			{
				flag = ExecuteShowOptionsCommand(source, originalSource, parameter);
			}
			else if (command == OutlookBarCommands.ShowPopupCommand)
			{
				flag = ExecuteShowPopupCommand(source, originalSource, parameter);
			}
			if (flag)
			{
				RaiseExecutedCommand(new ExecutedCommandEventArgs(command));
			}
			return flag;
		}

		private bool ExecuteShowPopupCommand(object source, object originalSource, object parameter)
		{
			if (_selectedGroupContent == null)
			{
				return false;
			}
			OutlookBarCancelableRoutedEventArgs e = new OutlookBarCancelableRoutedEventArgs();
			RaiseSelectedGroupPopupOpening(e);
			if (!e.Cancel && _selectedGroupContent.GetTemplateChildHelper("PART_Popup") is Popup popup)
			{
				if (popup.IsOpen || SelectedGroup == null)
				{
					popup.IsOpen = false;
					return false;
				}
				PopupResizerDecorator popupResizerDecorator = _selectedGroupContent.GetTemplateChildHelper("popDec") as PopupResizerDecorator;
				if (popupResizerDecorator == null)
				{
					popupResizerDecorator = _selectedGroupContent.GetTemplateChildHelper("PART_PopupResizerDecorator") as PopupResizerDecorator;
				}
				FrameworkElement frameworkElement = _selectedGroupContent.GetTemplateChildHelper("SelectedGroupContentPresenter") as FrameworkElement;
				if (frameworkElement == null)
				{
					frameworkElement = _selectedGroupContent.GetTemplateChildHelper("PART_SelectedGroupContentPresenter") as FrameworkElement;
				}
				if (frameworkElement == null)
				{
					return false;
				}
				if (popupResizerDecorator != null)
				{
					double num = _selectedGroupContent.ActualHeight;
					if (_navigationAreaIC != null)
					{
						num += _navigationAreaIC.ActualHeight;
					}
					if (_splitter != null)
					{
						num += _splitter.ActualHeight;
					}
					popupResizerDecorator.MaxHeight = num;
					double d = _lastExpandedWidth;
					if (double.IsNaN(d))
					{
						_selectedGroupContent.Measure(new Size(double.PositiveInfinity, double.PositiveInfinity));
						d = _selectedGroupContent.DesiredSize.Width;
					}
					popupResizerDecorator.Width = ((!double.IsNaN(d)) ? _lastExpandedWidth : _lastExpandedSelectedGroupSize.Width);
					PopupResizerDecorator.Constraints constraints = PopupResizerDecorator.GetResizeConstraints(frameworkElement);
					if (constraints == null)
					{
						constraints = new PopupResizerDecorator.Constraints();
					}
					constraints.MinimumHeight = Math.Min(_selectedGroupContent.ActualHeight, popupResizerDecorator.MaxHeight);
					constraints.MaximumHeight = popupResizerDecorator.MaxHeight;
					constraints.MinimumWidth = 2.0 * minExpandedWidth;
					constraints.MaximumWidth = base.MaxWidth;
					PopupResizerDecorator.SetResizeConstraints(frameworkElement, constraints);
				}
				popup.Closed += Popup_Closed;
				OutlookBarGroup selectedGroup = SelectedGroup;
				try
				{
					if (selectedGroup != null)
					{
						selectedGroup.SuppressLogicalContent = true;
					}
					popup.IsOpen = true;
				}
				finally
				{
					if (selectedGroup != null)
					{
						selectedGroup.SuppressLogicalContent = false;
					}
				}

		private static void CanExecuteCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			XamOutlookBar xamOutlookBar = sender as XamOutlookBar;
			e.CanExecute = false;
			if (xamOutlookBar == null)
			{
				return;
			}
			if (e.Command == OutlookBarCommands.ShowMoreButtonsCommand)
			{
				e.CanExecute = xamOutlookBar._navigationAreaGroups.Count < xamOutlookBar.VisibleGroupsCount && xamOutlookBar._splitter != null;
			}
			else if (e.Command == OutlookBarCommands.ShowFewerButtonsCommand)
			{
				e.CanExecute = xamOutlookBar._navigationAreaGroups.Count > 0 && xamOutlookBar._splitter != null;
			}
			else if (e.Command == OutlookBarCommands.ShowPopupCommand && xamOutlookBar._selectedGroupContent != null)
			{
				if (xamOutlookBar._selectedGroupContent.GetTemplateChildHelper("PART_Popup") is Popup popup)
				{
					e.CanExecute = xamOutlookBar.IsMinimized && !popup.IsOpen && xamOutlookBar.SelectedGroup != null;
				}
				else
				{
					e.CanExecute = false;
				}
			}
			else if (e.Command == OutlookBarCommands.ShowOptionsCommand)
			{
				e.CanExecute = true;
			}
			else if (e.Command == OutlookBarCommands.GroupMoveDownCommand)
			{
				OutlookBarGroup groupForCommand = GetGroupForCommand(e.Source, e.OriginalSource, e.Parameter);
				if (groupForCommand == null)
				{
					e.CanExecute = false;
				}
				else
				{
					e.CanExecute = xamOutlookBar.Groups.IndexOf(groupForCommand) < xamOutlookBar.Groups.Count - 1;
				}
			}
			else if (e.Command == OutlookBarCommands.GroupMoveUpCommand)
			{
				OutlookBarGroup groupForCommand2 = GetGroupForCommand(e.Source, e.OriginalSource, e.Parameter);
				if (groupForCommand2 == null)
				{
					e.CanExecute = false;
				}
				else
				{
					e.CanExecute = xamOutlookBar.Groups.IndexOf(groupForCommand2) > 0;
				}
			}
			else if (e.Command == OutlookBarCommands.SelectGroupCommand)
			{
				OutlookBarGroup groupForCommand3 = GetGroupForCommand(e.Source, e.OriginalSource, e.Parameter);
				e.CanExecute = groupForCommand3 != null && !groupForCommand3.IsSelected;
			}
		}

		private static OutlookBarGroup GetGroupForCommand(object source, object originalSource, object parameter)
		{
			OutlookBarGroup outlookBarGroup = parameter as OutlookBarGroup;
			if (outlookBarGroup == null)
			{
				outlookBarGroup = source as OutlookBarGroup;
				if (outlookBarGroup == null)
				{
					outlookBarGroup = originalSource as OutlookBarGroup;
				}
			}
			return outlookBarGroup;
		}

		private bool ShouldUpdateLayoutOnThemeChange()
		{
			for (DependencyObject dependencyObject = this; dependencyObject != null; dependencyObject = Utilities.GetParent(dependencyObject, includeLogicalParent: false) ?? Utilities.GetParent(dependencyObject, includeLogicalParent: true))
			{
				if (dependencyObject is Page)
				{
					return true;
				}
				if (dependencyObject is Window)
				{
					return true;
				}
			}
			if (Window.GetWindow(this) != null)
			{
				return false;
			}
			return true;
		}

		private static void OnThemeChanged(DependencyObject target, DependencyPropertyChangedEventArgs e)
		{
			XamOutlookBar xamOutlookBar = target as XamOutlookBar;
			if (xamOutlookBar.IsInitialized)
			{
				xamOutlookBar.InvalidateMeasure();
				if (xamOutlookBar.ShouldUpdateLayoutOnThemeChange())
				{
					xamOutlookBar.UpdateLayout();
				}
			}
			xamOutlookBar.OnThemeChanged((string)e.OldValue, (string)e.NewValue);
		}

		protected virtual void OnThemeChanged(string previousValue, string currentValue)
		{
			RoutedPropertyChangedEventArgs<string> e = new RoutedPropertyChangedEventArgs<string>(previousValue, currentValue);
			e.RoutedEvent = ThemeChangedEvent;
			e.Source = this;
			RaiseEvent(e);
		}

	[DesignTimeVisible(false)]
	public class GroupsPanel : StackPanel
	{
		private bool _overflow;

		private double _deltaSize;

		private bool _hasFreeSpace;

		private double _maxHeightHorizontal;

		internal bool Overflow
		{
			get
			{
				return _overflow;
			}
			set
			{
				_overflow = value;
			}
		}

		internal double DeltaSize
		{
			get
			{
				return _deltaSize;
			}
			set
			{
				_deltaSize = value;
			}
		}

		internal bool HasFreeSpace
		{
			get
			{
				return _hasFreeSpace;
			}
			set
			{
				_hasFreeSpace = value;
			}
		}

		private GroupsPresenter GroupItemsPresenter
		{
			get
			{
				DependencyObject parent = VisualTreeHelper.GetParent(this);
				while (parent != null && !(parent is GroupsPresenter))
				{
					parent = VisualTreeHelper.GetParent(parent);
				}
				return parent as GroupsPresenter;
			}
		}

		public GroupsPanel()
		{
			base.CanVerticallyScroll = false;
			_maxHeightHorizontal = 0.0;
		}

		public override void EndInit()
		{
			if (GroupItemsPresenter != null)
			{
				base.Orientation = GroupItemsPresenter.Orientation;
				if (base.Orientation == Orientation.Horizontal)
				{
					base.HorizontalAlignment = HorizontalAlignment.Right;
				}
			}
			base.EndInit();
		}

		protected override Size MeasureOverride(Size constraint)
		{
			Size result = base.MeasureOverride(constraint);
			bool flag;
			if (base.Orientation == Orientation.Vertical)
			{
				_deltaSize = constraint.Height - result.Height;
				flag = !double.IsInfinity(constraint.Height);
			}
			else
			{
				_deltaSize = constraint.Width - result.Width;
				flag = !double.IsInfinity(constraint.Width);
				if (_maxHeightHorizontal > result.Height)
				{
					result.Height = _maxHeightHorizontal;
				}
				else
				{
					_maxHeightHorizontal = result.Height;
				}
			}
			_overflow = _deltaSize + 0.001 < 0.0;
			_hasFreeSpace = _deltaSize > 0.001 && flag;
			if (_overflow)
			{
				if (base.Orientation == Orientation.Vertical)
				{
					result.Height = constraint.Height;
				}
				else
				{
					result.Width = constraint.Width;
				}
				GroupItemsPresenter.InvalidateMeasure();
			}
			return result;
		}

		protected override void OnChildDesiredSizeChanged(UIElement child)
		{
			if (child is OutlookBarGroup { OutlookBar: not null, Visibility: not Visibility.Collapsed, DesiredNavigationHeight: var desiredNavigationHeight, DesiredSize: var desiredSize } outlookBarGroup)
			{
				bool reloadGroups = desiredNavigationHeight != desiredSize.Height;
				outlookBarGroup.DesiredNavigationHeight = outlookBarGroup.DesiredSize.Height;
				outlookBarGroup.OutlookBar.RefreshNavigationArea(reloadGroups);
			}
			base.OnChildDesiredSizeChanged(child);
		}
	}

		ExpandCollapseState IExpandCollapseProvider.ExpandCollapseState
		{
			get
			{
				XamOutlookBar xamOutlookBar = (XamOutlookBar)base.Owner;
				if (!xamOutlookBar.AllowMinimized)
				{
					return ExpandCollapseState.LeafNode;
				}
				if (xamOutlookBar.IsMinimized)
				{
					return ExpandCollapseState.Collapsed;
				}
				return ExpandCollapseState.Expanded;
			}
		}

			return "XamOutlookBar";
		}

		public override object GetPattern(PatternInterface patternInterface)
		{
			if (patternInterface == PatternInterface.ExpandCollapse || patternInterface == PatternInterface.Selection)
			{
				return this;
			}
			return base.GetPattern(patternInterface);
		}

		void IExpandCollapseProvider.Collapse()
		{
			if (!IsEnabled())
			{
				throw new ElementNotEnabledException();
			}
			if (((IExpandCollapseProvider)this).ExpandCollapseState == ExpandCollapseState.LeafNode)
			{
				throw new InvalidOperationException();
			}
			((XamOutlookBar)base.Owner).IsMinimized = true;
		}

		void IExpandCollapseProvider.Expand()
		{
			if (!IsEnabled())
			{
				throw new ElementNotEnabledException();
			}
			if (((IExpandCollapseProvider)this).ExpandCollapseState == ExpandCollapseState.LeafNode)
			{
				throw new InvalidOperationException();
			}
			((XamOutlookBar)base.Owner).IsMinimized = false;
		}

		IRawElementProviderSimple[] ISelectionProvider.GetSelection()
		{
			OutlookBarGroup selectedGroup = ((XamOutlookBar)base.Owner).SelectedGroup;
			if (selectedGroup != null)
			{
				List<IRawElementProviderSimple> list = new List<IRawElementProviderSimple>(1);
				if (selectedGroup != null)
				{
					AutomationPeer peer = UIElementAutomationPeer.CreatePeerForElement(selectedGroup);
					list.Add(ProviderFromPeer(peer));
				}
				return list.ToArray();
			}
			return null;
		}
