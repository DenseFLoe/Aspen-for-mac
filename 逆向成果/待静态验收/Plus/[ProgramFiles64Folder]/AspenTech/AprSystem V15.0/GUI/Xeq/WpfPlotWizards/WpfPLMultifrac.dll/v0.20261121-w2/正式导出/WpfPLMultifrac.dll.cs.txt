		void IRibbonPlotWizardType.Context(object oNode, ref object oPlotWizardComponent, ref int ierror)
		{
			if (oPlotWizardComponent == null || oNode == null)
			{
				ierror = 1;
				return;
			}
			_context = oNode as IHNode;
			_wizardPlot = oPlotWizardComponent as IHWizardPlot;
			GetVarName();
			if (_columnList.Count == 0)
			{
				_help.LoadColumn(_context, varname, _columnList);
				if (_columnList.Count <= 0)
				{
					ierror = 1;
					return;
				}
				_currentColumnId = _columnList[0];
			}
			string text = varname1 + "/" + _currentColumnId;
			_wizardPlot.SetIndependentVariable(_context, text + "!Label");
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, text);
			if (_isAddEnthalpyBox)
			{
				_wizardPlot.AddDependentVariable(_context, varname2 + "/" + _currentColumnId);
			}
			((PlotwizardHelper)_help).LoadUnitsList(_context.Application.Tree, _wizardPlot, _enthalpyUnitList);
			_wizardPlot.PlotVal.SwapAxises = true;
			if (_enthalpyUnitList.Count == 0)
			{
				_wizardPlot.set_ErrorStatus(2, true);
			}
		}

		List<string> IRibbonPlotWizardType.FieldDataOptionList(string sFieldName)
		{
			if (sFieldName.CompareTo(ColumnListKey) == 0)
			{
				return _columnList;
			}
			if (sFieldName.CompareTo(EnthalpyUnitListKey) == 0)
			{
				return _enthalpyUnitList;
			}
			if (sFieldName.CompareTo(StageOnKey) == 0)
			{
				return _help.StageOnList;
			}
			return new List<string>();
		}

		RibbonControlType IRibbonPlotWizardType.GetFieldCtrlType(string sFieldName)
		{
			if (sFieldName.CompareTo(EnthalpyUnitListKey) != 0 && sFieldName.CompareTo(ColumnListKey) != 0 && sFieldName.CompareTo(StageOnKey) != 0)
			{
				if (sFieldName.CompareTo(EnthalpyBoxKey) != 0)
				{
					return (RibbonControlType)0;
				}
				return (RibbonControlType)2;
			}
			return (RibbonControlType)1;
		}

		void IRibbonPlotWizardType.GetFieldData(string sFieldName, ref string sCurrentVal)
		{
			if (sFieldName.CompareTo(EnthalpyUnitListKey) == 0)
			{
				sCurrentVal = ((PlotwizardHelper)_help).GetUnitString(_context.Application.Tree, _wizardPlot);
			}
			else if (sFieldName.CompareTo(ColumnListKey) == 0)
			{
				sCurrentVal = _currentColumnId;
			}
			else if (sFieldName.CompareTo(EnthalpyBoxKey) == 0)
			{
				sCurrentVal = _isAddEnthalpyBox.ToString();
			}
		}

		void IRibbonPlotWizardType.GetFieldStyle(string sFieldName, ref string sCurrentVal)
		{
			if (sFieldName.CompareTo(StageOnKey) == 0)
			{
				sCurrentVal = _help.GetStageOn(_wizardPlot.PlotVal.SwapAxises);
			}
		}

		void IRibbonPlotWizardType.SetFieldData(string sFieldName, string sValue)
		{
			if (sFieldName.CompareTo(EnthalpyUnitListKey) == 0 && _currentEnthalpyUnit.CompareTo(sValue) != 0)
			{
				_currentEnthalpyUnit = sValue;
				((PlotwizardHelper)_help).SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentEnthalpyUnit);
			}
			else if (sFieldName.CompareTo(ColumnListKey) == 0)
			{
				_currentColumnId = sValue;

			else if (sFieldName.CompareTo(EnthalpyBoxKey) == 0)

				_isAddEnthalpyBox = string.Compare(sValue, bool.TrueString, ignoreCase: true) == 0;

				_wizardPlot.RemoveAllDependentVariable();
				_wizardPlot.AddDependentVariable(_context, text2);
				((IHPlotVal)_wizardPlot.PlotVal).set_Label(0, WpfPLMultifracResources.CGCCstgH_Legends0);

		void IRibbonPlotWizardType.Context(object oNode, ref object oPlotWizardComponent, ref int ierror)
		{
			if (oPlotWizardComponent == null || oNode == null)
			{
				ierror = 1;
				return;
			}
			_context = oNode as IHNode;
			_wizardPlot = oPlotWizardComponent as IHWizardPlot;
			GetVarName();
			_columnList = new List<string>();
			_help.LoadColumn(_context, "Y", _columnList);
			if (_columnList.Count == 0)
			{
				ierror = 1;
				return;
			}
			if (_columnList.Count > 0)
			{
				_currentColumnId = _columnList[0];
			}
			IHNode iHNode = null;
			bool flag = false;
			for (int i = 0; i < _varnames.GetLength(1); i++)
			{
				iHNode = _context.FindNode(_varnames[0, i]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					_rbArr[i].IsEnabled = true;
					flag = true;
				}
			}
			if (!flag)
			{
				ierror = 1;
			}
			else if (string.IsNullOrEmpty(_currentConfigValue))
			{
				_rbArr[0].IsChecked = true;
				IHNodeCol iHNodeCol = null;
				int basis = _wizardPlot.basis;
				for (int j = 0; j < _rbArr.Count; j++)
				{
					if (!_rbArr[j].IsEnabled)
					{
						continue;
					}
					iHNode = _context.FindNode(_varnames[basis - 1, j]);
					if (iHNode != null && iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null && iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements.Count > 0)
					{
						iHNodeCol = iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
						if (iHNodeCol != null && iHNodeCol.Count > 0)
						{
							_rbArr[j].IsChecked = true;
							break;
						}
					}
				}
				List<string> list = new List<string>();
				if (iHNodeCol != null && iHNodeCol.Count > 0)
				{
					int count = iHNodeCol.Count;
					for (int k = 0; k < count; k++)
					{
						iHNodeCol.get_Label(0, k, Type.Missing);
						list.Add(iHNodeCol.get_Label(0, k, Type.Missing));
					}
					CurveList.SetCurves(list.ToArray(), isKeepSelection: true);
				}
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		public void UpdateWizardData()
		{
			_wizardPlot.RemoveAllDependentVariable();
			int num = 0;
			if (RBVapor.IsChecked == true)
			{
				_wizardPlot.PlotVal.Title = PlotPWRResource.MULTIFRAC_Comp_PlotTitle;
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, PlotPWRResource.MULTIFRAC_Comp_YAxisTitle);
				num = 0;
			}
			else if (true == RBLiquid.IsChecked)
			{
				_wizardPlot.PlotVal.Title = WpfPLMultifracResources.Composition_Title_Liquid;
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLMultifracResources.Composition_YTitle_Liquid);
				num = 1;
			}
			else if (true == RBOrganic.IsChecked)
			{
				_wizardPlot.PlotVal.Title = WpfPLMultifracResources.Composition_Title_Organic;
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLMultifracResources.Composition_YTitle_Organic);
				num = 2;
			}
			else if (true == RBFreewater.IsChecked)
			{
				_wizardPlot.PlotVal.Title = WpfPLMultifracResources.Composition_Title_FreeWater;
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLMultifracResources.Composition_YTitle_FreeWater);
				num = 3;
			}
			_wizardPlot.PlotVal.Title = _wizardPlot.PlotVal.Title + string.Format(PlotPWRResource.Multifrac_Column_Title, _currentColumnId);
			string[] array = new string[3]
			{
				WpfPLMultifracResources.YTitle_Ymole,
				WpfPLMultifracResources.YTitle_Xmole,
				WpfPLMultifracResources.YTitle_X1mole
			};
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			if (CurveList.Curves.Count == 0 || list.Count == 0)
			{
				_wizardPlot.set_ErrorStatus(3, true);
				return;
			}
			int num2 = _wizardPlot.basis - 1;
			IHNode iHNode = _context.FindNode(_varnames[num2, num]);
			_wizardPlot.SetIndependentVariable(_context, _varnames[num2, num] + "/" + _currentColumnId + "!Label");
			if (iHNode != null)
			{
				num2 = list.Count;
				if (iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null)
				{
					int count = iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements.Count;
					_wizardPlot.PlotVal.NVariables = num2;
					for (int i = 0; i < list.Count; i++)
					{
						string text = list[i];
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(i, text);
						for (int j = 0; j < count; j++)
						{
							object obj = iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[j, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing);
							((IHPlotVal)_wizardPlot.PlotVal).set_YData(i, j, (short)0, obj);
						}
					}
					((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, array[num]);
				}
			}
			_currentConfigValue = SaveConfigValue();
		}

		void IRibbonPlotWizardType.Context(object oNode, ref object oPlotWizardComponent, ref int ierror)
		{
			if (oPlotWizardComponent == null || oNode == null)
			{
				ierror = 1;
				return;
			}
			_context = oNode as IHNode;
			_wizardPlot = oPlotWizardComponent as IHWizardPlot;
			GetVarName();
			if (_columnList.Count == 0)
			{
				_help.LoadColumn(_context, varname, _columnList);
				if (_columnList.Count == 0)
				{
					ierror = 1;
					return;
				}
				if (_columnList.Count > 0)
				{
					_currentColumnId = _columnList[0];
				}
			}
			_wizardPlot.SetIndependentVariable(_context, "Output/" + varname + "/" + _currentColumnId);
			if (_exergyUnitList == null || _exergyUnitList.Count == 0)
			{
				((PlotwizardHelper)_help).LoadXUnitsList(_context.Application.Tree, _wizardPlot, _exergyUnitList);
				if (_exergyUnitList.Count == 0)
				{
					ierror = 1;
					return;
				}
				CBExgergyUnit.ValueOptions = _exergyUnitList;
				CBExgergyUnit.Value = ((PlotwizardHelper)_help).GetXUnitString(_context.Application.Tree, _wizardPlot);
			}
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, "Output/B_TEMP4/" + _currentColumnId);
			if (_termperatureUnitList == null || _termperatureUnitList.Count == 0)
			{
				((PlotwizardHelper)_help).LoadUnitsList(_context.Application.Tree, _wizardPlot, _termperatureUnitList);
				CBTemperatureUnit.ValueOptions = _termperatureUnitList;
				CBTemperatureUnit.Value = ((PlotwizardHelper)_help).GetUnitString(_context.Application.Tree, _wizardPlot);
			}
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				((UIElement)(object)LblTemp).IsEnabled = false;
				((UIElement)(object)CBTemperatureUnit).IsEnabled = false;
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
			_wizardPlot.RemoveAllDependentVariable();
			if (RBStage.IsChecked == true)
			{
				_wizardPlot.AddDependentVariable(_context, "Output/B_TEMP4/" + _currentColumnId + "!Label");
			}
			else if (RBTemperature.IsChecked == true)
			{
				_wizardPlot.AddDependentVariable(_context, "Output/B_TEMP4/" + _currentColumnId);
			}
			else if (RBCarnotFactor.IsChecked == true)
			{
				_wizardPlot.AddDependentVariable(_context, "Output/CARNOT/" + _currentColumnId);
			}
		}

		public void UpdateWizardData()
		{
			string text = _currentColumnId.Trim();
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.SetIndependentVariable(_context, "Output/" + varname + "/" + text);
			if (RBStage.IsChecked == true)
			{
				_wizardPlot.AddDependentVariable(_context, "Output/B_TEMP4/" + _currentColumnId + "!Label");
			}
			else if (RBTemperature.IsChecked == true)
			{
				_wizardPlot.AddDependentVariable(_context, "Output/B_TEMP4/" + _currentColumnId);
			}
			else if (RBCarnotFactor.IsChecked == true)
			{
				_wizardPlot.AddDependentVariable(_context, "Output/CARNOT/" + _currentColumnId);
			}
			((PlotwizardHelper)_help).SaveXUnitsComboText(_context.Application.Tree, _wizardPlot, CBExgergyUnit.Value.ToString());
			if (RBTemperature.IsChecked == true)
			{
				((PlotwizardHelper)_help).SaveUnitsComboText(_context.Application.Tree, _wizardPlot, CBTemperatureUnit.Value.ToString());
			}
			if (RBStage.IsChecked == true)
			{
				_wizardPlot.PlotVal.Title = string.Format(WpfPLMultifracResources.Exergy_Stage_Title, _currentColumnId);
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLMultifracResources.Exergy_Stage_YTitle);
			}
			else if (RBTemperature.IsChecked == true)
			{
				_wizardPlot.PlotVal.Title = string.Format(WpfPLMultifracResources.Exergy_T_Title, _currentColumnId);
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLMultifracResources.Exergy_T_YTitle);
			}
			else if (RBCarnotFactor.IsChecked == true)
			{
				_wizardPlot.PlotVal.Title = string.Format(WpfPLMultifracResources.Exergy_CarnotFactor_Title, _currentColumnId);
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLMultifracResources.Exergy_CarnotFactor_YTitle);
			}
			_currentConfigValue = SaveConfigValue();
		}

			RBVapor.IsChecked = Convert.ToBoolean(value);
			RBLiquid.IsChecked = !RBVapor.IsChecked;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);

			value = array[num].Substring(array[num].IndexOf('=') + 1);

			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);

			foreach (string value2 in array2)
			{
				_CBArr[Convert.ToInt16(value2)].IsChecked = true;
			}
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			array2 = value.Split(',');
			foreach (string value3 in array2)

				((UIElement)(object)_CBArr[Convert.ToInt16(value3)]).IsEnabled = true;

		void IRibbonPlotWizardType.Context(object oNode, ref object oPlotWizardComponent, ref int ierror)
		{
			if (oPlotWizardComponent == null || oNode == null)
			{
				ierror = 1;
				return;
			}
			_context = oNode as IHNode;
			_wizardPlot = oPlotWizardComponent as IHWizardPlot;
			GetVarName();
			if (_columnList.Count == 0)
			{
				_help.LoadColumn(_context, "VAP_ACT", _columnList);
				if (_columnList.Count == 0)
				{
					ierror = 1;
					return;
				}
				if (_columnList.Count > 0)
				{
					_currentColumnId = _columnList[0];
				}
			}
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.SetIndependentVariable(_context, varnames[1] + "/" + _currentColumnId + "!Label");
			_wizardPlot.AddDependentVariable(_context, varnames[0] + "/" + _currentColumnId);
			_wizardPlot.AddDependentVariable(_context, varnames[1] + "/" + _currentColumnId);
			_wizardPlot.AddDependentVariable(_context, varnames[2] + "/" + _currentColumnId);
			if (_flowUnitList == null || _flowUnitList.Count == 0)
			{
				((PlotwizardHelper)_help).LoadUnitsList(_context.Application.Tree, _wizardPlot, _flowUnitList);
				if (_flowUnitList.Count == 0)
				{
					ierror = 1;
					return;
				}
			}
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				for (int i = 0; i < _CBArr.Length; i++)
				{
					IHNode iHNode = _context.FindNode(varnames[i]);
					_CBArr[i].IsChecked = false;
					((UIElement)(object)_CBArr[i]).IsEnabled = false;
					if (iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null && iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null)
					{
						((UIElement)(object)_CBArr[i]).IsEnabled = true;
					}
				}
				for (int j = 0; j < _CBArr.Length; j++)
				{
					if (((UIElement)(object)_CBArr[j]).IsEnabled && _wizardPlot.get_HasDependentVariable(_context, varnames[j] + "/" + _currentColumnId))
					{
						_CBArr[j].IsChecked = true;
					}
				}
			}
			else
			{
				for (int k = 0; k < _CBArr.Length; k++)
				{
					_CBArr[k].IsChecked = false;
					((UIElement)(object)_CBArr[k]).IsEnabled = false;
				}
				RestoreConfigValue(_currentConfigValue);
			}
		}

				varnames[1] = "Output/LIQ_ACT";
				varnames[2] = "Output/LIQ_MAX";
			}
			_wizardPlot.RemoveAllDependentVariable();
			string text = _currentColumnId.Trim();
			_wizardPlot.SetIndependentVariable(_context, varnames[1] + "/" + text + "!Label");
			int num = 0;
			for (int i = 0; i < _CBArr.Length; i++)
			{
				string text2 = varnames[i];
				if (_CBArr[i].IsChecked == true && !_wizardPlot.get_HasDependentVariable(_context, text2 + "/" + text))
				{
					_wizardPlot.AddDependentVariable(_context, text2 + "/" + text);
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(num, _CBArr[i].Content.ToString());
					num++;
				}
			}
			_wizardPlot.PlotVal.Title = string.Format(WpfPLMultifracResources.Hydraulic_Title_Mole, text);
			((PlotwizardHelper)_help).SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentFlowUnit);
			_currentConfigValue = SaveConfigValue();
		}

		public void UpdateWizardData()
		{
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			if (CurveList.Curves.Count == 0 || list.Count == 0)
			{
				_wizardPlot.set_ErrorStatus(3, true);
				return;
			}
			_wizardPlot.RemoveAllDependentVariable();
			int num = 0;
			if (RBVapor2Liquid.IsChecked == true)
			{
				_wizardPlot.PlotVal.Title = WpfPLMultifracResources.KValues_VL_Title;
				num = 0;
			}
			else if (true == RBVapor2Organic.IsChecked)
			{
				_wizardPlot.PlotVal.Title = WpfPLMultifracResources.KValues_VO_Title;
				num = 1;
			}
			else if (true == RBVapor2Freewater.IsChecked)
			{
				_wizardPlot.PlotVal.Title = WpfPLMultifracResources.KValues_VFW_Title;
				num = 2;
			}
			else if (true == RBOrganic2Freewater.IsChecked)
			{
				_wizardPlot.PlotVal.Title = WpfPLMultifracResources.KValues_OFW_Title;
				num = 3;
			}
			int num2 = _wizardPlot.basis - 1;
			IHNode iHNode = _context.FindNode(_varnames[num]);
			_wizardPlot.SetIndependentVariable(_context, _varnames[num] + "/" + _currentColumnId + "!Label");
			if (iHNode != null)
			{
				num2 = list.Count;
				if (iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null)
				{
					num = iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements.Count;
					_wizardPlot.PlotVal.NVariables = num2;
					for (int i = 0; i < list.Count; i++)
					{
						string text = list[i];
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(i, text);
						for (int j = 0; j < num; j++)
						{
							object obj = iHNode.Elements[_currentColumnId, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[j, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing);
							((IHPlotVal)_wizardPlot.PlotVal).set_YData(i, j, (short)0, obj);
						}
					}
				}
			}
			string arg = _currentColumnId.Trim();
			int num3 = _wizardPlot.PlotVal.Title.IndexOf("(");
			if (num3 < 0)
			{
				_wizardPlot.PlotVal.Title = _wizardPlot.PlotVal.Title + string.Format(PlotPWRResource.Multifrac_Column_Title, arg);
			}
			else
			{
				_wizardPlot.PlotVal.Title = _wizardPlot.PlotVal.Title.Substring(0, num3) + string.Format(PlotPWRResource.Multifrac_Column_Title, arg);
			}
			((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLMultifracResources.TooltipHeaderForRibbonItem_KValues);
			_currentConfigValue = SaveConfigValue();
		}
