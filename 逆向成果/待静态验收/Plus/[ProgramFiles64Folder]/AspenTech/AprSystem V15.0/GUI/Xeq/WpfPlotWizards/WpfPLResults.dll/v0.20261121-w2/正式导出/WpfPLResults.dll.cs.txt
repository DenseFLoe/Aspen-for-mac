	[ProgId("WpfPLResults.RibbonPlotWizardMgr")]
	public class RibbonPlotWizardMgr : IRibbonPlotWizardMgr
	{
		private static string[] supportedTypes = new string[6] { "API Gravity", "Specific Gravity", "UOPK", "Molecular Wt", "Petro Properties", "Viscosity" };

		private IHNode _hnode;

		private IHWizardPlot _wizardPlot;

		private static List<PWRPlotType> _lspwrTypes = new List<PWRPlotType>();

		private PWRParser _pwrParser = new PWRParser();

		private Assembly _asm;

		private Dictionary<string, IRibbonPlotWizardType> _dicTagIdToPlotType;

		private IHWizardPlot _tempWizard = (CHWizardPlot)Activator.CreateInstance(Marshal.GetTypeFromCLSID(new Guid("9E14C11F-6706-4067-8112-35CE9468F3A5")));

		public RibbonPlotWizardMgr()
		{
			//IL_0001: Unknown result type (might be due to invalid IL or missing references)
			//IL_000b: Expected O, but got Unknown
			Initialize();
		}

		long IRibbonPlotWizardMgr.SupportedPlotTypes(ref List<string> lsTagIds)
		{
			lsTagIds.Clear();
			for (int i = 0; i < _lspwrTypes.Count; i++)
			{
				PWRPlotType val = _lspwrTypes[i];
				IRibbonPlotWizardType iplotType = null;
				_tempWizard.Clear();
				_pwrParser.FillDefaultPlotVal(val, ref _hnode, ref _tempWizard);
				if (DoesPlotTypeHaveData(val.TagId, ref iplotType, ref _tempWizard))
				{
					lsTagIds.Add(val.TagId);
				}
			}
			return _lspwrTypes.Count;
		}

		long IRibbonPlotWizardMgr.GetPlotType(string sTagId, ref IRibbonPlotWizardType plwType)
		{
			if (_dicTagIdToPlotType.ContainsKey(sTagId))
			{
				plwType = _dicTagIdToPlotType[sTagId];
				if (plwType == null)
				{
					return 0L;
				}
				PWRPlotType val = null;
				_pwrParser.FindPlotType(sTagId, ref _lspwrTypes, ref val);
				_pwrParser.FillDefaultPlotVal(val, ref _hnode, ref _wizardPlot);
			}
			return 0L;
		}

		void IRibbonPlotWizardMgr.SetContext(object oNode, ref object oWizard)
		{
			_hnode = oNode as IHNode;
			_wizardPlot = oWizard as IHWizardPlot;
		}

		void IRibbonPlotWizardMgr.ShowInputDialog(string sPlotName)
		{
		}

		private void InitializeWizardPlot()
		{
		}

		private void Initialize()
		{
			_asm = Assembly.GetExecutingAssembly();
			StreamReader streamReader = new StreamReader(Assembly.GetExecutingAssembly().GetManifestResourceStream("WpfPLResults.Text.Resultspwr.txt"));
			_lspwrTypes.Clear();
			_pwrParser.ParsePwrFile(streamReader, ref _lspwrTypes);
			_dicTagIdToPlotType = new Dictionary<string, IRibbonPlotWizardType>();
			string[] array = supportedTypes;
			foreach (string text in array)
			{
				if (text == supportedTypes[0])
				{
					IRibbonPlotWizardType value = (IRibbonPlotWizardType)(object)new APIGravity();
					_dicTagIdToPlotType.Add(text, value);
				}
				else if (text == supportedTypes[1])
				{
					IRibbonPlotWizardType value2 = (IRibbonPlotWizardType)(object)new SpecificGravity();
					_dicTagIdToPlotType.Add(text, value2);
				}
				else if (text == supportedTypes[2])
				{
					IRibbonPlotWizardType value3 = (IRibbonPlotWizardType)(object)new UOPK();
					_dicTagIdToPlotType.Add(text, value3);
				}
				else if (text == supportedTypes[3])
				{
					IRibbonPlotWizardType value4 = (IRibbonPlotWizardType)(object)new MolecularWt();
					_dicTagIdToPlotType.Add(text, value4);
				}
				else if (text == supportedTypes[4])
				{
					IRibbonPlotWizardType value5 = (IRibbonPlotWizardType)(object)new Petroprop();
					_dicTagIdToPlotType.Add(text, value5);
				}
				else if (text == supportedTypes[5])
				{
					IRibbonPlotWizardType value6 = (IRibbonPlotWizardType)(object)new Viscosity();
					_dicTagIdToPlotType.Add(text, value6);
				}
			}
		}

		private bool DoesPlotTypeHaveData(string sTagId, ref IRibbonPlotWizardType iplotType, ref IHWizardPlot iWizardPlot)
		{
			int num = 0;
			if (_dicTagIdToPlotType.ContainsKey(sTagId))
			{
				iplotType = _dicTagIdToPlotType[sTagId];
				if (iplotType == null)
				{
					return false;
				}
				object obj = iWizardPlot;
				iplotType.Context((object)_hnode, ref obj, ref num);
				if (num == 0)
				{
					return true;
				}
			}
			return false;
		}
	}

			string path = "PetroCharacterization/Results/Output/Prop Data/PPPROP";
			IHNode iHNode = _context.FindNode(path);
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				ierror = 1;
				return;
			}
			_wizardPlot.RemoveAllDependentVariable();
			int num = iHNode.Elements.get_RowCount(0);
			iHNode.Elements.get_RowCount(1);
			List<string> list = new List<string>();
			if (num > 0)
			{
				for (int i = 0; i < num; i++)
				{
					string item = iHNode.Elements.get_Label(0, i, Type.Missing);
					if (!list.Contains(item))
					{
						list.Add(item);
					}
				}
			}
			CurvesSelector.SetCurves(list.ToArray());
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		public void UpdateWizardData()
		{
			_wizardPlot.RemoveAllDependentVariable();
			string path = "PetroCharacterization/Results/Output/Prop Data/PPPROP";
			IHNode iHNode = _context.FindNode(path);
			if (iHNode == null)
			{
				return;
			}
			List<string> list = new List<string>(CurvesSelector.GetSelectedCurves());
			int num = iHNode.Elements.get_RowCount(1);
			if (list.Count > 0)
			{
				_wizardPlot.PlotVal.NVariables = list.Count;
				for (int i = 0; i < list.Count; i++)
				{
					int location = iHNode.Elements.get_LabelLocation(list[i], 0);
					string text = iHNode.Elements.get_Label(0, location, Type.Missing);
					for (int j = 0; j < num; j++)
					{
						string loc_or_name = iHNode.Elements.get_Label(1, j, Type.Missing);
						try
						{
							((IHPlotVal)_wizardPlot.PlotVal).set_YData(i, j, (short)1, (object)iHNode.Elements[text, loc_or_name, Type.Missing, Type.Missing, Type.Missing]);
						}
						catch (Exception)
						{
						}
					}
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(i, text);
				}
			}
			if (!string.IsNullOrEmpty(_currentTemparetureUnit))
			{
				_help.SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentTemparetureUnit);
			}
			_currentConfigValue = SaveConfigValue();
		}

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			int num = CurvesSelector.GetSelectedCurves().Count();
			if (num == CurvesSelector.Curves.Count)
			{
				CurvesSelector.IsCheckAll = true;
			}
			else
			{
				string[] selectedCurves = CurvesSelector.GetSelectedCurves();
				CurvesSelector.IsCheckAll = false;
				CurvesSelector.SetSelectedCurves(selectedCurves);
			}
			e.CanExecute = num != 0;
		}

		private void SetCurves()
		{
			List<string> list = new List<string>();
			if (radioButtonAbsVis.IsChecked == true)
			{
				for (int i = 0; i < _absVisList.Count(); i++)
				{
					list.Add(_absVisList[i]);
				}
			}
			else
			{
				for (int j = 0; j < _kinVisList.Count(); j++)
				{
					list.Add(_kinVisList[j]);
				}
			}
			CurvesSelector.IsCheckAll = false;
			CurvesSelector.SetCurves(list.ToArray());
		}
