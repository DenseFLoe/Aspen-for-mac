		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot2;
			IHNode iHNode = _context.FindNode("Output/DUTY_HOT");
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				ierror = 1;
				return;
			}
			List<string> list = new List<string>();
			for (int i = 0; i < iHNode.Elements.Count; i++)
			{
				list.Add(iHNode.Elements.get_Label(0, i, Type.Missing));
			}
			CurveList.SetCurves(list.ToArray());
			CurveList.Header = ApwnResources.BrowserTQCurves;
			CurveList.IsCheckAll = true;
			_bHasInputDialog = true;
			if (list.Count == 1)
			{
				_bHasInputDialog = false;
			}
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				CurveList.IsCheckAll = false;
				RestoreConfigValue(_currentConfigValue);
				_bHasInputDialog = true;
			}
		}

		public void UpdateWizardData()
		{
			_wizardPlot.RemoveAllDependentVariable();
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			if (list.Count > 0)
			{
				int num = 0;
				for (int i = 0; i < list.Count; i++)
				{
					string text = list[i];
					_wizardPlot.SetIndependentVariable(_context, "Output/DUTY_HOT/" + text);
					_wizardPlot.AddDependentVariable(_context, "Output/TEMP_CLD/" + text);
					(_wizardPlot.PlotVal as IHPlotVal2).set_LineColor((short)num, (short)1);
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(num++, text + " " + PlotPWRResource.Heatx_TQCurve_ColdStream);
					_wizardPlot.AddDependentVariable(_context, "Output/TEMP_HOT/" + text);
					(_wizardPlot.PlotVal as IHPlotVal2).set_LineColor((short)num, (short)3);
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(num++, text + " " + PlotPWRResource.Heatx_TQCurve_HotStream);
				}
				_help.SaveUnitsComboText(_context.Application.Tree, (IHWizardPlot)_wizardPlot, _currentTempUnit);
				_help.SaveXUnitsComboText(_context.Application.Tree, (IHWizardPlot)_wizardPlot, _currentDutyUnit);
				_currentConfigValue = SaveConfigValue();
				_bHasInputDialog = true;
			}
		}

		private void ExecuteOKCommand(object sender, ExecutedRoutedEventArgs e)
		{
			UpdateWizardData();
			_help.CloseForm((UserControl)this, (bool?)true);
		}

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			int num = CurveList.GetSelectedCurves().Count();
			e.CanExecute = num != 0;
		}

		private void ExecuteCancelCommand(object sender, ExecutedRoutedEventArgs e)
		{
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				RestoreConfigValue(_currentConfigValue);
			}
			_help.CloseForm((UserControl)this, (bool?)false);
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("CurveSelection=");
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			int count = list.Count;
			for (int i = 0; i < count; i++)
			{
				if (i > 0)
				{
					stringBuilder.Append(",");
				}
				stringBuilder.Append(list[i]);
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CheckAll=");
			stringBuilder.Append(CurveList.IsCheckAll);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string text = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = text.Split(',');
			List<string> list = new List<string>();
			string[] array3 = array2;
			foreach (string item in array3)
			{
				list.Add(item);
			}
			CurveList.SetSelectedCurves(list.ToArray());
			num++;
			text = array[num].Substring(array[num].IndexOf('=') + 1);
			if (list.Count != CurveList.Curves.Count)
			{
				CurveList.IsCheckAll = false;
			}
			else
			{
				CurveList.IsCheckAll = Convert.ToBoolean(text);
			}
		}
