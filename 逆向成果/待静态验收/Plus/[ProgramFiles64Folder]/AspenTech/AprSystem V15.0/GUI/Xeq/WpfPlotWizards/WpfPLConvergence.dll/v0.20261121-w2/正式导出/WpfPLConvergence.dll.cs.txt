		private bool GetVarName()
		{
			bool flag = false;
			broAndNew.Clear();
			_baseType = _context.get_AttributeValue((short)6, Type.Missing) as string;
			if (!string.IsNullOrEmpty(_baseType) && _baseType.ToLower() == "secant")
			{
				_iHNode = _context.FindNode("Output/ERR_TOL3");
				if (_iHNode != null && _iHNode.Elements != null && _iHNode.Elements.Count > 0)
				{
					_currentXVariableName = "Output/ERR_TOL3/#";
					_currentYVariableName = "Output/ERR_TOL3/#";
					_xAxisTtitle = WpfPLConvergenceHistoryResources.ConvHistoryXAxisTitle;
					_isAllowDialogueBox = false;
					flag = true;
				}
			}
			if (!string.IsNullOrEmpty(_baseType) && (_baseType.ToLower() == "broyden" || _baseType.ToLower() == "newton"))
			{
				_potentialXVairable1 = "Output/ERR_TOL2";
				_potentialXVairable2 = "Output/ERR_TOL3/#";
				_iHNode = _context.FindNode("Output/VAR_VAL2");
				if (_iHNode != null && _iHNode.Elements != null && _iHNode.Elements.Count > 0)
				{
					if (IsVariableAvailable(_potentialXVairable1))
					{
						_currentXVariableName = _potentialXVairable1;
					}
					else
					{
						_currentXVariableName = _potentialXVairable2;
					}
					_currentYVariableName = "Output/VAR_VAL2";
					_xAxisTtitle = WpfPLConvergenceHistoryResources.ConvHistoryXAxisTitle;
					if (_isUsingDefaultValue)
					{
						IsToleranceMaxErrorChecked = true;
						IsToleranceMaxErrorEnabled = true;
						_isAllowDialogueBox = true;
					}
					else if (!IsToleranceMaxErrorEnabled)
					{
						IsToleranceMaxErrorEnabled = true;
					}
					if (!broAndNew.Contains(_currentYVariableName) && IsToleranceMaxErrorChecked && IsToleranceMaxErrorEnabled)
					{
						broAndNew.Add(_currentYVariableName);
					}
					flag = true;
					_isBroAndNew = true;
				}
				_iHNode = _context.FindNode("Output/ERR_TOL3");
				if (_iHNode != null && _iHNode.Elements != null && _iHNode.Elements.Count > 0)
				{
					if (IsVariableAvailable(_potentialXVairable1))
					{
						_currentXVariableName = _potentialXVairable1;
					}
					else
					{
						_currentXVariableName = _potentialXVairable2;
					}
					_currentYVariableName = "Output/ERR_TOL3/#";
					_xAxisTtitle = WpfPLConvergenceHistoryResources.ConvHistoryXAxisTitle;
					designSpecCombo.Items.Clear();
					if (!broAndNew.Contains(_currentYVariableName) && IsDesignSpecErrorChecked)
					{
						broAndNew.Add(_currentYVariableName);
					}
					foreach (object element in _iHNode.Elements)
					{
						designSpecCombo.Items.Add((element as IHNode).get_Name(Type.Missing));
					}
					if (_isUsingDefaultValue)
					{
						IsDesignSpecErrorChecked = true;
						IsDesignSpecErrorEnabled = true;
						designSpecCombo.IsEnabled = true;
						if (designSpecCombo.Items != null && designSpecCombo.Items.Count > 0 && designSpecCombo.IsEnabled)
						{
							designSpecCombo.SelectedIndex = 0;
						}
						_isAllowDialogueBox = true;
					}
					else
					{
						if (!IsDesignSpecErrorEnabled)
						{
							IsDesignSpecErrorEnabled = true;
						}
						designSpecCombo.IsEnabled = true;
						if (!string.IsNullOrEmpty(_selectedDesignSpec) && designSpecCombo.Items != null && designSpecCombo.Items.Count > 0 && designSpecCombo.IsEnabled)
						{
							int num = designSpecCombo.Items.IndexOf(_selectedDesignSpec);
							if (num == -1)
							{
								designSpecCombo.SelectedIndex = 0;
							}
							else
							{
								designSpecCombo.SelectedIndex = num;
							}
						}
					}
					flag = true;
					_isBroAndNew = true;
				}
				_iHNode = _context.FindNode("Output/ERR_TOL2");
				if (_iHNode != null && _iHNode.Elements != null && _iHNode.Elements.Count > 0)
				{
					if (IsVariableAvailable(_potentialXVairable1))
					{
						_currentXVariableName = _potentialXVairable1;
					}
					else
					{
						_currentXVariableName = _potentialXVairable2;
					}
					_currentYVariableName = "Output/ERR_TOL2";
					_xAxisTtitle = WpfPLConvergenceHistoryResources.ConvHistoryXAxisTitle;
					if (_isUsingDefaultValue)
					{
						IsTearMaxErrorChecked = true;
						IsTearMaxErrorEnabled = true;
						_isAllowDialogueBox = true;
					}
					else if (!IsTearMaxErrorEnabled)
					{
						IsTearMaxErrorEnabled = true;
					}
					if (!broAndNew.Contains(_currentYVariableName) && IsTearMaxErrorChecked)
					{
						broAndNew.Add(_currentYVariableName);
					}
					flag = true;
					_isBroAndNew = true;
				}
			}
			if (!string.IsNullOrEmpty(_baseType) && (_baseType.ToLower() == "wegstein" || _baseType.ToLower() == "direct"))
			{
				_iHNode = _context.FindNode("Output/ERR_TOL2");
				if (_iHNode != null && _iHNode.Elements != null && _iHNode.Elements.Count > 0)
				{
					_currentXVariableName = "Output/ERR_TOL2";
					_currentYVariableName = "Output/ERR_TOL2";
					_xAxisTtitle = WpfPLConvergenceHistoryResources.ConvHistoryXAxisTitle;
					if (_isUsingDefaultValue)
					{
						IsTearMaxErrorEnabled = true;
						IsTearMaxErrorChecked = true;
						IsDesignSpecErrorEnabled = false;
						IsDesignSpecErrorChecked = false;
						IsToleranceMaxErrorChecked = false;
						IsToleranceMaxErrorEnabled = false;
						_isAllowDialogueBox = true;
					}
					else
					{
						if (!IsTearMaxErrorEnabled)
						{
							IsTearMaxErrorEnabled = true;
						}
						if (IsDesignSpecErrorEnabled)
						{
							IsDesignSpecErrorEnabled = false;
						}
						if (IsToleranceMaxErrorEnabled)
						{
							IsToleranceMaxErrorEnabled = false;
						}
					}
					flag = true;
				}
			}
			if (!flag)
			{
				IsTearMaxErrorEnabled = false;
				IsDesignSpecErrorEnabled = false;
				IsToleranceMaxErrorEnabled = false;
				designSpecCombo.Items.Clear();
				designSpecCombo.IsEnabled = false;
			}
			if (!string.IsNullOrEmpty(_baseType) && _baseType.ToLower() == "sqp")
			{
				_isAllowDialogueBox = true;
			}
			return flag;
		}

		{
			if (sFieldName.CompareTo(_stageOrderFieldId) == 0)
			{
				if (_wizardPlot.PlotVal.ReverseXAxis)
				{
					sCurrentVal = _descending;
				}
				else
				{
					sCurrentVal = _ascending;
				}
			}
			else if (sFieldName.CompareTo(_stageOnFieldId) == 0)
			{
				if (_wizardPlot.PlotVal.SwapAxises)
				{
					sCurrentVal = _yAxis;
				}
				else
				{
					sCurrentVal = _xAxis;
				}
			}
		}

		}

			if (!string.IsNullOrEmpty(sValue))
			{
				string[] array = sValue.Split(',');
				if (array.Count() == 11)
				{
					IsToleranceMaxErrorEnabled = array[0].Equals(bool.TrueString);
					IsToleranceMaxErrorChecked = array[1].Equals(bool.TrueString);
					IsDesignSpecErrorEnabled = array[2].Equals(bool.TrueString);
					IsDesignSpecErrorChecked = array[3].Equals(bool.TrueString);
					IsComboEnabled = array[4].Equals(bool.TrueString);
					int.TryParse(array[5], out var result);
					designSpecCombo.SelectedIndex = result;
					IsTearMaxErrorEnabled = array[6].Equals(bool.TrueString);
					IsTearMaxErrorChecked = array[7].Equals(bool.TrueString);
					IsIncludeZeroLine = array[8].Equals(bool.TrueString);
					_isAllowDialogueBox = array[9].Equals(bool.TrueString);
					_selectedDesignSpec = (string.IsNullOrEmpty(array[10]) ? null : array[10]);
					DoUpdateTarget();
					_isUsingDefaultValue = false;
				}
			}
		}

		public void UpdateWizardData()
		{
			string yAxisTitle = GetYAxisTitle();
			if (broAndNew.Count > 0 && _isBroAndNew)
			{
				_wizardPlot.SetIndependentVariable(_context, _currentYVariableName + "!Label");
				_wizardPlot.RemoveAllDependentVariable();
				for (int i = 0; i < broAndNew.Count; i++)
				{
					_wizardPlot.PlotVal.NVariables = i + 1;
					if (broAndNew[i] == "Output/VAR_VAL2")
					{
						IHNode iHNode = _context.FindNode(broAndNew[i]);
						int num = iHNode.Elements.get_RowCount(0);
						for (int j = 0; j < i + 1; j++)
						{
							_wizardPlot.PlotVal.NPoints = num;
							for (int k = 0; k < num; k++)
							{
								try
								{
									double? num2 = iHNode.Elements[k, j, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing) as double?;
									if (num2.HasValue)
									{
										double num3 = Math.Log10(Math.Abs(num2.Value));
										((IHPlotVal)_wizardPlot.PlotVal).set_YData(0, k, (short)0, (object)num3);
									}
								}
								catch
								{
								}
							}
							((IHPlotVal)_wizardPlot.PlotVal).set_Label(i, WpfPLConvergenceHistoryResources.MaximumError);
							((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle((object)i, yAxisTitle);
							_wizardPlot.PlotVal.XAxisTitle = _xAxisTtitle;
						}
					}
					else if (broAndNew[i] == "Output/ERR_TOL3/#")
					{
						if (_isUsingDefaultValue || (!_isUsingDefaultValue && designSpecCombo.Items.Contains(_selectedDesignSpec)))
						{
							if (designSpecCombo.SelectedItem != null && !string.IsNullOrEmpty(designSpecCombo.SelectedItem.ToString()))
							{
								string vName = "Output/ERR_TOL3/" + designSpecCombo.SelectedItem.ToString() + "/";
								SetYData(vName, i);
							}
							else
							{
								SetYData(broAndNew[i], i);
							}
