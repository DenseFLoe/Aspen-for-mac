public class HtmlHelpCefLibMgr
{
	private static string strRenderPath;

	private static HtmlHelpCefLibMgr helpCefLibMgr;

	private HtmlHelpCefLibMgr()
	{
		string location = GetType().Assembly.Location;
		FileInfo fileInfo = new FileInfo(location);
		if (fileInfo != null)
		{
			location = fileInfo.DirectoryName;
			if (!string.IsNullOrWhiteSpace(location))
			{
				strRenderPath = Path.Combine(location, "CefSharp.BrowserSubprocess.exe");
			}
			else
			{
				strRenderPath = "CefSharp.BrowserSubprocess.exe";
			}
		}
	}

	public static void InitLib()
	{
		//IL_004d: Unknown result type (might be due to invalid IL or missing references)
		//IL_0052: Unknown result type (might be due to invalid IL or missing references)
		//IL_005a: Unknown result type (might be due to invalid IL or missing references)
		//IL_0065: Unknown result type (might be due to invalid IL or missing references)
		//IL_006c: Unknown result type (might be due to invalid IL or missing references)
		//IL_007d: Expected O, but got Unknown
		if (helpCefLibMgr == null)
		{
			helpCefLibMgr = new HtmlHelpCefLibMgr();
		}
		string text = Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData, Environment.SpecialFolderOption.Create) ?? "";
		string text2 = string.Empty;
		try
		{
			text2 = Directory.CreateDirectory(text + "\\HTMLHelp").FullName;
		}
		catch (Exception value)
		{
			Console.WriteLine(value);
		}
		try
		{
			CefSettings val = new CefSettings
			{
				LogSeverity = (LogSeverity)99,
				BrowserSubprocessPath = strRenderPath,
				IgnoreCertificateErrors = true,
				CachePath = (text2 ?? "")
			};
			((Dictionary<string, string>)(object)((CefSettingsBase)val).CefCommandLineArgs).Add("disable-gpu", "1");
			if (!Cef.IsInitialized)
			{
				Cef.Initialize((CefSettingsBase)(object)val, false, (IBrowserProcessHandler)null);
			}
		}
		catch (Exception)
		{
		}
	}

	public static void Shutdown()
	{
		Cef.Shutdown();
	}
}
