	private bool GetVisibleScrollAreaList()
	{
		foreach (IMMScrollAreaPlot scrollArea in _mmTabPage.ScrollAreas)
		{

			{
				_VisibleScrollAreaList.Add(scrollArea);
			}
		}
		bool result = false;
		if (_VisibleScrollAreaList.Count == 1)
		{
			string text = ((object)_VisibleScrollAreaList[0]).GetType().ToString();
			if (text.Contains(_sAdsDataGridType))

			{
				_sGridType = _sMMScrollAreaMSType;
				result = false;
			}
		}
		return result;

	private void BuildVarLists()

		foreach (IMMScrollAreaPlot item in scrollAreas)

			num++;

				continue;

			_Dicfields = item.GetVisibleFields();
			_variableLists.Add(item.GetName(), item.GetVisibleFields());
			_DicfieldsInfo = item.GetVisibleFieldInfo();

						array[i] = array[i].Trim();

							_DicTitle.Add(item2.Key, array[i]);
						}
						if (i == 1)
						{
							_DicPath.Add(item2.Key, array[i]);
						}
					}
				}
			}
			_variableInfoLists.Add(item.GetName(), item.GetVisibleFieldInfo());
			if (_preSelectedFields.Count > 0)
			{
				continue;
			}
			_preSelectedFields = item.GetSelectedFields();
			_DicpreSelectedFields = item.GetSelectedFieldsDic();
			int count = _preSelectedFields.Count;
			if (count > 0)
			{
				for (int j = 0; j < count; j++)
				{
					string text = _preSelectedFields[j];
					foreach (KeyValuePair<int, string> item3 in _DicTitle)
					{
						if (text.Contains(item3.Value))
						{
							_preSelectedFields[j] = item3.Value;
							break;
						}
					}
				}
			}
			_preSelectedFieldsList.Add(item.GetName(), item.GetSelectedFields());
		}
		int num3 = 0;
		foreach (KeyValuePair<string, Dictionary<int, string>> variableList in _variableLists)
		{
			if (variableList.Value == null)
			{
				continue;
			}
			num3++;
			Dictionary<int, string> value = variableList.Value;
			foreach (int item4 in value.Keys.OrderBy((int k) => k))
			{
				if (string.IsNullOrEmpty(value[item4]))
				{
					continue;
				}
				string text2 = value[item4].ToString();
				foreach (KeyValuePair<int, string> item5 in _DicTitle)
				{
					if (text2.Contains(item5.Value))
					{
						text2 = item5.Value;
						break;
					}
				}
				text2 = text2.Trim();
				if (_columnHeaderList.Contains(text2))
				{
					text2 = text2 + "__" + num3;
				}
				_columnHeaderList.Add(text2);
				_DicHeaders.Add(text2, item4);
			}

		{
			list2.Add(preSelectedField);
		}
		CurveList.SetSelectedCurves(list2.ToArray());

		ComboBoxParaVar.Items.Clear();

		{
			for (int i = 0; i < list.Count; i++)

				ComboBoxParaVar.Items.Add(list[i]);
			}
			ComboBoxParaVar.Visibility = Visibility.Visible;
			((UIElement)(object)lblParaVar).Visibility = Visibility.Visible;
		}
	}

	private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
	{
		int selectedIndex = ComboBoxXAxis.SelectedIndex;
		int selectedIndex2 = ComboBoxParaVar.SelectedIndex;
		int num = CurveList.GetSelectedCurves().Count();
		if (num == CurveList.Curves.Count)
		{
			CurveList.IsCheckAll = true;
		}
		else
		{
			string[] selectedCurves = CurveList.GetSelectedCurves();
			CurveList.IsCheckAll = false;
			CurveList.SetSelectedCurves(selectedCurves);
		}
		if (_bParametric)
		{
			if (selectedIndex >= 0 && selectedIndex2 >= 0 && num > 0)
			{
				e.CanExecute = true;
			}
			else
			{
				e.CanExecute = false;
			}
		}
		else if (selectedIndex >= 0 && num > 0)
		{
			e.CanExecute = true;
		}
		else
		{
			e.CanExecute = false;
		}
	}

	private void ExecuteCancelCommand(object sender, ExecutedRoutedEventArgs e)

		CloseForm(this, false);
	}

	private void BuildVarLists()
	{
		List<IMMScrollAreaPlot> scrollAreas = _mmTabPage.ScrollAreas;
		int num = -1;
		foreach (IMMScrollAreaPlot item in scrollAreas)
		{
			if ((item as FrameworkElement).Visibility != Visibility.Visible)
			{
				continue;
			}
			num++;
			if (item == null)
			{
				continue;
			}
			item.GetName();
			Dictionary<int, string> visibleFields = item.GetVisibleFields();
			Dictionary<int, VariableInstance> dictionary = new Dictionary<int, VariableInstance>();
			if (visibleFields != null)
			{
				foreach (KeyValuePair<int, string> item2 in visibleFields)
				{
					VariableInstance variableInstance = new VariableInstance();
					variableInstance.GridColumnName = item2.Value;
					variableInstance.DisplayName = TrunColumnToNormalDisplay(item2.Value);
					variableInstance.GridIndex = num;
					dictionary.Add(item2.Key, variableInstance);
				}
			}
			if (!_variableLists.Keys.Contains(item.GetName()))
			{
				_variableLists.Add(item.GetName(), dictionary);
			}
			else
			{
				_variableLists[item.GetName()] = dictionary;
			}
			if (_preSelectedFields.Count <= 0)
			{
				_preSelectedFields = item.GetSelectedFields();
				_scrollArea = item;
			}
		}
		foreach (KeyValuePair<string, Dictionary<int, VariableInstance>> variableList in _variableLists)
		{
			if (variableList.Value == null)
			{
				continue;
			}
			Dictionary<int, VariableInstance> value = variableList.Value;
			foreach (int item3 in value.Keys.OrderBy((int k) => k))
			{
				if (!string.IsNullOrEmpty(value[item3].DisplayName))
				{
					string text = value[item3].DisplayName.ToString();
					if (_columnHeaderList.Contains(text))
					{
						text = text + "__" + value[item3].GridIndex;
					}
					_columnHeaderList.Add(text);
				}
			}
		}
	}

				if (!flag || !flag2)

	private string TrunColumnToNormalDisplay(string columnValue)
	{
		string empty = string.Empty;
		empty = ((columnValue.IndexOf("OBJECT-") > 0) ? "\"OBJECTIVE FUNCTION\"" : ((columnValue.IndexOf("KUHN-") > 0) ? "\"KUHN-TUCKER ERROR\"" : ((columnValue.IndexOf("LAGRANG-") <= 0) ? columnValue.Replace("\n", " ") : "\"LAGRANGIAN FUNCTION\"")));
		if (empty.Contains("Unitless"))
		{
			empty = columnValue.Replace("Unitless", "").TrimEnd();
		}
		return empty;
	}

	private int GetSelectedField(string sHeader, string selectedSA)
	{
		int result = -1;
		if (sHeader.IndexOf("__") > 0)
		{
			sHeader.Remove(sHeader.IndexOf("__"));
		}
		foreach (KeyValuePair<string, Dictionary<int, VariableInstance>> variableList in _variableLists)
		{
			if (!(variableList.Key.ToString().ToLower() == selectedSA.ToLower()))
			{
				continue;
			}
			foreach (KeyValuePair<int, VariableInstance> item in variableList.Value)
			{
				if (!string.IsNullOrEmpty(item.Value.DisplayName))
				{
					string text = item.Value.DisplayName.ToString();
					if (sHeader.ToLower() == text.ToLower())
					{
						result = Convert.ToInt32(item.Key);
						break;
					}
				}
			}
		}
		return result;
	}
