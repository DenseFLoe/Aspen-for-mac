		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			string text = (dynamic)_context.FindNode("Input/STATESPEC").get_Value(Type.Missing);
			if (!string.IsNullOrEmpty(text) && text.Contains("GEN-"))
			{
				ierror = 1;
				return;
			}
			VarNames = new string[3] { "Input/CIDS", "Input/USE", "Input/VALUE" };
			if (!Build_CandidateList())
			{
				ierror = 1;
				return;
			}
			if (string.IsNullOrEmpty(_componentId) && ComponentList.Count > 0)
			{
				_componentId = ComponentList[0];
			}
			if (string.IsNullOrEmpty(_propertyId) && XPropName.Count > 0)
			{
				if (_wizardPlot.basis == 29)
				{
					if (XPropName.Count > 1)
					{
						_propertyId = XPropName[1];
					}
					else
					{
						_propertyId = XPropName[0];
					}
				}
				else
				{
					_propertyId = XPropName[0];
				}
			}
			if (_wizardPlot.basis != 29 || _wizardPlot.basis != 30)
			{
				_help.DetermineMoleType(_context, out MoleType);
			}
			Build_Node_Lookup();
			SetWindowTitle();
			SetAxisTitles();
		}

		{

			if (DataElement.Length == 0)
			{
				return false;
			}
			PlotType = _wizardPlot.basis;
			switch (PlotType)
			{
			case 1:
			case 2:
			case 3:
			case 4:
			case 5:
			case 6:
			case 7:
				ComponentList = _help.GetComponentList(_context, VarNames[0]);
				_showProperty = false;
				_showComponent = true;
				break;
			case 8:
				ComponentList = _help.GetComponentList(_context, VarNames[0]);
				_showProperty = true;
				_showComponent = true;
				break;
			case 9:
			case 29:
			case 30:
				_showProperty = true;
				_showComponent = false;
				break;
			}
			bool flag = true;
			return PlotType switch
			{
				1 => IsTempPresent() && IsXPresent() && IsYPresent(), 
				2 => IsPresPresent() && IsXPresent() && IsYPresent(), 
				3 => IsTempPresent() && IsXPresent(), 
				4 => IsPresPresent() && IsXPresent(), 
				5 => IsYPresent() && IsXPresent(), 
				6 => IsTempPresent() && CountFor("X1") == 2 && CountFor("X2") == 2,

				9 => IsTempPresent() && IsPropertyPresent() && IsMixture(),

				30 => IsPresPresent2() && IsPropertyPresent() && !IsMixture(),

		private bool IsPropertyPresent()
		{
			bool result = false;
			IHNode iHNode = _context.FindNode(VarNames[2]);
			int num = iHNode.Elements.get_RowCount(1);
			int num2 = 0;
			XPropName.Clear();
			XPropUnit.Clear();
			XPropc.Clear();
			for (int i = 0; i < num; i++)
			{
				string text = iHNode.Elements.get_Label(1, i, Type.Missing);
				string item = iHNode.Elements.get_Label(2, i, Type.Missing);
				if (text != null)
				{
					switch (text.Length)
					{
					case 5:
						switch (text[4])
						{
						case '1':
							if (text == "TEMP1")
							{
								continue;
							}
							break;
						case '2':
							if (text == "TEMP2")
							{
								continue;
							}
							break;
						}
						break;
					case 1:
					{
						char c = text[0];
						if (c == 'X' || c == 'Y')
						{
							continue;
						}
						break;
					}
					case 2:
						switch (text[1])
						{
						case '1':
							if (text == "X1")
							{
								continue;
							}
							break;
						case '2':
							if (text == "X2")
							{
								continue;
							}
							break;
						}
						break;
					case 11:
						if (text == "TEMPERATURE")
						{
							continue;
						}
						break;
					case 8:
						if (!(text == "PRESSURE"))
						{
							break;
						}
						if (_wizardPlot.basis != 9 && _wizardPlot.basis != 8 && _wizardPlot.basis != 30)
						{
							num2++;
							if (!XPropName.Contains(text))
							{
								XPropName.Add(text);
							}
							if (!XPropName.Contains(item))
							{
								XPropUnit.Add(item);
							}
							if (!XPropc.Contains(i))
							{
								XPropc.Add(i);
							}
						}
						continue;
					}
				}
				num2++;
				if (!XPropName.Contains(text))
				{
					XPropName.Add(text);
				}
				if (!XPropName.Contains(item))
				{
					XPropUnit.Add(item);
				}
				if (!XPropc.Contains(i))
				{
					XPropc.Add(i);
				}
			}
			if (num2 > 0)
			{
				result = true;
			}
			return result;
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			string text = (dynamic)_context.FindNode("Input/STATESPEC").get_Value(Type.Missing);
			if (!string.IsNullOrEmpty(text) && text.Contains("GEN-"))
			{
				ierror = 1;
				return;
			}
			VarNames = new string[3] { "Input/CIDS", "Input/USE", "Input/VALUE" };
			DataNum = _help.GetDataElementsArray(_context, VarNames[1]);
			if (DataNum.Length == 0)
			{
				ierror = 1;
				return;
			}
			IHNode iHNode = _context.FindNode(VarNames[2]);
			List<string> list = new List<string>();
			if (iHNode != null)
			{
				int num = iHNode.Elements.get_RowCount(1);
				for (int i = 0; i < num; i++)
				{
					string text2 = iHNode.Elements.get_Label(1, i, Type.Missing);
					string text3 = iHNode.Elements.get_Label(2, i, Type.Missing);
					list.Add(text2 + " " + text3);
				}
			}
			ComboBoxXAxis.Items.Clear();
			ComboBoxParaVar.Items.Clear();
			List<string> list2 = new List<string>();
			CurveList.IsCheckAll = false;
			for (int j = 0; j < list.Count; j++)
			{
				ComboBoxXAxis.Items.Add(list[j]);
				ComboBoxParaVar.Items.Add(list[j]);
				list2.Add(list[j]);
				CurveList.SetCurves(list2.ToArray());
			}
			chkParaVar.IsChecked = false;
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				ComboBoxXAxis.Text = ComboBoxXAxis.Items[0].ToString();
			}
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string text = array[num].Substring(array[num].IndexOf('=') + 1);
			ComboBoxXAxis.Text = text;
			num++;
			text = array[num].Substring(array[num].IndexOf('=') + 1);

		public string Join(List<string> xW, string sFill, bool bTrail)
		{
			string text = "";
			for (int i = 0; i < xW.Count; i++)
			{
				if (!string.IsNullOrEmpty(xW[i]))
				{
					text += xW[i].ToUpper();
					if (i != xW.Count - 1 || bTrail)
					{
						text += sFill;
					}
				}
				else if (i == xW.Count - 1 && bTrail && text.Length == 0)
				{
					text = sFill;
				}
			}
			return text;
		}
