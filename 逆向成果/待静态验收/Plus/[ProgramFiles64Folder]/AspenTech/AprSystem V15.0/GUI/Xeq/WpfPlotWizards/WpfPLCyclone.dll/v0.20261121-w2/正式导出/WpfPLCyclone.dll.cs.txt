		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot2;
			List<string> list = new List<string>();
			IHNode iHNode = _context.FindNode("Output/VORTEX_EFF");
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				ierror = 1;
				return;
			}
			int count = iHNode.Elements.Count;
			for (int i = 0; i < count; i++)
			{
				string item = iHNode.Elements.get_Label(0, i, Type.Missing);
				list.Add(item);
			}
			SubStreamList.SetCurves(list.ToArray());
			SubStreamList.Header = WpfPLDiscurvesResources.PSDSubstream;
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				List<string> list2 = new List<string>();
				list2.Add(list[0].ToString());
				SubStreamList.SetSelectedCurves(list2.ToArray());
			}
			if (list.Count == 1)
			{
				SubStreamList.IsCheckAll = true;
			}
			iHNode = iHNode.Elements[list[0].ToString(), Type.Missing, Type.Missing, Type.Missing, Type.Missing];
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				ierror = 1;
				return;
			}
			_bHasInputDialog = false;
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				SubStreamList.IsCheckAll = false;
				RestoreConfigValue(_currentConfigValue);
				_bHasInputDialog = true;
			}
			_wizardPlot.SetIndependentVariable(_context, "Output/PART_DIAM");
		}

		public void UpdateWizardData()
		{
			_wizardPlot.RemoveAllDependentVariable();
			IHNode iHNode = _context.FindNode("Output/PART_DIAM");
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				return;
			}
			List<string> list = new List<string>(SubStreamList.GetSelectedCurves());
			if (list.Count <= 0)
			{
				return;
			}
			_wizardPlot.RemoveAllIndependentVariable();
			string text = "Output/PART_DIAM/";
			string text2 = "Output/VORTEX_EFF/";
			try
			{
				for (int i = 0; i < list.Count; i++)
				{
					string indepvar = text + list[i];
					_wizardPlot.SetMultIndependentVariables(_context, indepvar);
					string depvar = text2 + list[i];
					_wizardPlot.AddDependentVariable(_context, depvar);
					string label = list[i];
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(i, label);
				}
				_help.SaveXUnitsComboText(_context.Application.Tree, (IHWizardPlot)_wizardPlot, _currentLengthUnit);
				string xUnitString = _help.GetXUnitString(_context.Application.Tree, (IHWizardPlot)_wizardPlot);
				string xAxisTitle = WpfPLDiscurvesResources.SepEff_XAxisTitle + " " + xUnitString;
				_wizardPlot.PlotVal.XAxisTitle = xAxisTitle;
			}
			catch
			{
			}
			_currentConfigValue = SaveConfigValue();
			_bHasInputDialog = true;
		}

		private void ExecuteOKCommand(object sender, ExecutedRoutedEventArgs e)
		{
			UpdateWizardData();
			SubStreamList.UpdateCurvesStatus();
			_CancelValue = _currentConfigValue;
			_help.CloseForm((UserControl)this, (bool?)true);
		}

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			int num = SubStreamList.GetSelectedCurves().Count();
			_CancelValue = _currentConfigValue;
			e.CanExecute = num != 0;
		}

		private void ExecuteCancelCommand(object sender, ExecutedRoutedEventArgs e)
		{
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				RestoreConfigValue(_currentConfigValue);
			}
			_help.CloseForm((UserControl)this, (bool?)false);
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("SubStreamSelection=");
			List<string> list = new List<string>(SubStreamList.GetSelectedCurves());
			int count = list.Count;
			for (int i = 0; i < count; i++)
			{
				if (i > 0)
				{
					stringBuilder.Append(",");
				}
				stringBuilder.Append(list[i]);
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CheckAll=");
			stringBuilder.Append(SubStreamList.IsCheckAll);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string text = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = text.Split(',');
			List<string> list = new List<string>();
			string[] array3 = array2;
			foreach (string item in array3)
			{
				list.Add(item);
			}
			SubStreamList.SetSelectedCurves(list.ToArray());
			num++;
			text = array[num].Substring(array[num].IndexOf('=') + 1);
			SubStreamList.IsCheckAll = Convert.ToBoolean(text);
		}
