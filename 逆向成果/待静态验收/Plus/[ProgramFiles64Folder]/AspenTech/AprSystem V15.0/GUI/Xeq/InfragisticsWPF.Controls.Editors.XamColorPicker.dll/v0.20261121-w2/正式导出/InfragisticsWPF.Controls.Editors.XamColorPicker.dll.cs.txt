			if (GetPattern(PatternInterface.Value) != null)
			{
				string oldValue2 = ((!oldValue.HasValue) ? null : oldValue.ToString());
				string newValue2 = ((!newValue.HasValue) ? null : newValue.ToString());
				RaisePropertyChangedEvent(ValuePatternIdentifiers.ValueProperty, oldValue2, newValue2);
			}
		}

		internal void RaiseExpandCollapseStatePropertyChangedEvent(ExpandCollapseState oldValue, ExpandCollapseState newValue)
		{
			if (oldValue != newValue && AutomationPeer.ListenerExists(AutomationEvents.PropertyChanged))
			{
				RaisePropertyChangedEvent(ExpandCollapsePatternIdentifiers.ExpandCollapseStateProperty, oldValue, newValue);
			}
		}

		protected virtual void Collapse()
		{
			OwningColorPicker.IsDropDownOpen = false;
		}

		protected virtual void Expand()
		{
			OwningColorPicker.IsDropDownOpen = true;
		}

		protected virtual void SetSelectedColor(string colorInARGB)
		{
			string message = "The expected string format is '#AARRGGBB', with each part being the HEX value for the argument.";
			if (string.IsNullOrEmpty(colorInARGB) || colorInARGB.Length != 9)
			{
				throw new ArgumentException(message);
			}
			string s = colorInARGB.Substring(1, 2);
			string s2 = colorInARGB.Substring(3, 2);
			string s3 = colorInARGB.Substring(5, 2);
			string s4 = colorInARGB.Substring(7, 2);
			Color value = Color.FromArgb(ConvertHexStringToByte(s), ConvertHexStringToByte(s2), ConvertHexStringToByte(s3), ConvertHexStringToByte(s4));
			OwningColorPicker.SelectedColor = value;
		}

		protected IRawElementProviderSimple[] GetSelection()
		{
			List<IRawElementProviderSimple> list = new List<IRawElementProviderSimple>();
			if (OwningColorPicker.SelectedColor.HasValue && OwningColorPicker.ColorStripManager != null && OwningColorPicker.ColorStripManager.SelectedColorItem != null)
			{
				ColorItemBox colorItemBox = OwningColorPicker.ColorStripManager.SelectedColorItem.ColorItemBox;
				if (colorItemBox != null)
				{
					AutomationPeer peer = UIElementAutomationPeer.FromElement(colorItemBox) ?? UIElementAutomationPeer.CreatePeerForElement(colorItemBox);
					list.Add(ProviderFromPeer(peer));
				}
			}
			return list.ToArray();
		}

		private byte ConvertHexStringToByte(string s)
		{
			s = s.ToUpper();
			char c = s[0];
			char c2 = s[1];
			int num = 0;
			if (char.IsDigit(c))
			{
				num = Convert.ToInt16(c.ToString()) * 16;
			}
			else
			{
				int num2 = c - 65;
				num = (10 + num2) * 16;
			}
			if (char.IsDigit(c2))
			{
				int num3 = Convert.ToInt16(c2.ToString());
				num += num3;
			}
			else
			{
				int num4 = c2 - 65;
				num += 10 + num4;
			}
			return Convert.ToByte(num);
		}

			Collapse();
		}

		void IExpandCollapseProvider.Expand()
		{
			Expand();
		}

		internal void HardSelectNewColor(Color? color)
		{
			SelectedColor = color;
			if (DropDownPopup != null && DropDownPopup.IsOpen)
			{
				IsDropDownOpen = false;
			}
			if (color.HasValue)
			{
				Color value = color.Value;
				if (!RecentColorPalette.Colors.Contains(value) && _allowRecentColorModification && !RecentColorPalette.Colors.Contains(value))
				{
					RecentColorPalette.Colors.Add(value);
				}
			}
		}

		internal void SoftSelectNewColor(Color? color)
		{
			_allowRecentColorModification = false;
			if (!color.HasValue)
			{
				SelectedColor = PreviouslySelectedColor;
				SelectedColorPreview = PreviouslySelectedColor;
			}
			else
			{
				SelectedColor = color;
				SelectedColorPreview = color;
			}
			_allowRecentColorModification = true;
		}

		private void EnsureVisualState()
		{
			base.LayoutUpdated -= XamColorPicker_LayoutUpdated;
			if (IsDropDownOpen)
			{
				VisualStateManager.GoToState(this, "Open", useTransitions: false);
				base.LayoutUpdated += XamColorPicker_LayoutUpdated;
			}
			else
			{
				VisualStateManager.GoToState(this, "Closed", useTransitions: false);
			}
			if (ColorPalettes.Count > 1)
			{
				VisualStateManager.GoToState(this, "Show", useTransitions: false);
			}
			else
			{
				VisualStateManager.GoToState(this, "Hide", useTransitions: false);
			}
			if (ShowAdvancedEditorButton)
			{
				VisualStateManager.GoToState(this, "ShowEditorButton", useTransitions: false);
			}
			else
			{
				VisualStateManager.GoToState(this, "HideEditorButton", useTransitions: false);
			}
			if (ShowRecentColorsPalette)
			{
				VisualStateManager.GoToState(this, "ShowRCPalette", useTransitions: false);
			}
			else
			{
				VisualStateManager.GoToState(this, "HideRCPalette", useTransitions: false);
			}
			if (ShowDerivedColorPalettes)
			{
				VisualStateManager.GoToState(this, "ShowDerivedPalettes", useTransitions: false);
			}
			else
			{
				VisualStateManager.GoToState(this, "HideDerivedPalettes", useTransitions: false);
			}
		}

		private void EnsureDerivedColorStrips()
		{
			if (_derivedColorStripsPanel == null)
			{
				return;
			}
			_derivedColorStripsPanel.Children.Clear();
			if (ColorStripManager != null)
			{
				foreach (ColorStrip derivedColorStrip in _derivedColorStrips)
				{
					ColorStripManager.ColorStrips.Remove(derivedColorStrip);
				}
			}
			_derivedColorStrips.Clear();
			double num = 100.0 / (double)(DerivedPalettesCount + 2);
			for (int i = 0; i < DerivedPalettesCount; i++)
			{
				ColorStrip colorStrip = new ColorStrip();
				colorStrip.ColorItemBoxStyle = DerivedPaletteColorItemBoxStyle;
				colorStrip.DarknessShift = Math.Min(num * (double)(i + 1), 100.0);
				if (ColorStripManager != null)
				{
					ColorStripManager.ColorStrips.Add(colorStrip);
				}
				_derivedColorStrips.Add(colorStrip);
				_derivedColorStripsPanel.Children.Add(colorStrip);
			}
		}

		private void SetCurrentColorPalette(ColorPalette cp)
		{
			if (_currentStrip != null)
			{
				_currentStrip.ColorPalette = cp;
			}
			EnsureDerivedColorStrips();
			if (DerivedPalettesCount > 0)
			{
				_ = 100.0 / (double)DerivedPalettesCount;
				for (int i = 0; i < DerivedPalettesCount; i++)
				{
					if (_derivedColorStrips.Count > 0)
					{
						SetDerivedColorPalettes(cp, _derivedColorStrips[i]);
					}
				}
			}
			if (ColorStripManager != null)
			{
				ColorStripManager.SetPreviouslySelectedColorInternal(SelectedColor);
			}
		}

		private void SetDerivedColorPalettes(ColorPalette cp, ColorStrip cs)
		{
			if (cp == null)
			{
				cs.ColorPalette = null;
				return;
			}
			ColorPalette colorPalette = new ColorPalette();
			foreach (ColorPatch color in cp.Colors)
			{
				HSL hSL = HSL.FromColor(color.Color);
				HSL hSL2 = hSL.Darker(cs.DarknessShift);
				if (hSL2.H == hSL.H && hSL2.L == hSL.L && hSL2.S == hSL.S)
				{
					hSL2 = hSL.Lighter(cs.DarknessShift);
				}
				colorPalette.Colors.Add(hSL2.ToColor());
			}
			cs.ColorPalette = colorPalette;
		}

		public override void OnApplyTemplate()
		{
			base.OnApplyTemplate();
			UIElement rootVisual = PlatformProxy.GetRootVisual(this);
			if (rootVisual != null)
			{
				rootVisual.RemoveHandler(UIElement.MouseLeftButtonDownEvent, new MouseButtonEventHandler(RootVisual_MouseLeftButtonDown));
				rootVisual.RemoveHandler(UIElement.KeyDownEvent, new KeyEventHandler(KeyDownHandler));
				rootVisual.RemoveHandler(UIElement.MouseLeftButtonUpEvent, new MouseButtonEventHandler(RootVisual_MouseLeftButtonUp));
			}
			if (GetTemplateChild("Root") is Grid grid)
			{
				ColorStripManager = grid.Resources["StripManager"] as ColorStripManager;
				if (ColorStripManager != null)
				{
					ColorStripManager.Parent = this;
					ColorStripManager.SelectedColorItemChanged += ColorStripManager_SelectedColorItemChanged;
					ColorStripManager.HoverColorItemChanged += ColorStripManager_HoverColorItemChanged;
				}
				DropDownPopup = GetTemplateChild("Popup") as Popup;
				if (DropDownPopup != null)
				{
					if (Utilities.GetAncestorFromType(this, typeof(Popup), allowSubclassOfType: false) is Popup)
					{
						DropDownPopup.StaysOpen = true;
					}
					else
					{
						DropDownPopup.StaysOpen = false;
					}
					DropDownPopup.AllowsTransparency = true;
					DropDownPopup.Placement = PlacementMode.Bottom;
					DropDownPopup.PlacementTarget = this;
					DropDownPopup.MouseDown -= DropDownPopup_MouseDown;
					DropDownPopup.MouseDown += DropDownPopup_MouseDown;
					DropDownPopup.Closed -= DropDownPopup_Closed;
					DropDownPopup.Closed += DropDownPopup_Closed;
				}
			}
			if (GetTemplateChild("ColorPalettesButton") is Button button)
			{
				button.Click += ColorPalettesDropDownButton_Click;
			}
			_colorPalettesCombo = GetTemplateChild("ColorPalettes") as ComboBox;
			if (_colorPalettesCombo != null)
			{
				_colorPalettesCombo.KeyDown += ColorPalettesCombo_KeyDown;
				_colorPalettesCombo.SelectionChanged += ColorPalettesCombo_SelectionChanged;
			}
			ColorPickerDialog = GetTemplateChild("ColorPickerDialog") as ColorPickerDialog;
			_rootPopupElement = GetTemplateChild("RootPopupElement") as FrameworkElement;
			_currentStrip = GetTemplateChild("CurrentColorStrip") as ColorStrip;
			_derivedColorStripsPanel = GetTemplateChild("DerivedPalettes") as Panel;
			EnsureDerivedColorStrips();
			if (CurrentPalette == null && ColorPalettes.Count > 0)
			{
				CurrentPalette = ColorPalettes[0];
			}
			else
			{
				SetCurrentColorPalette(CurrentPalette);
			}
			_recentColorsStrip = GetTemplateChild("RecentColorStrip") as ColorStrip;
			if (_recentColorsStrip != null)
			{
				_recentColorsStrip.ColorPalette = RecentColorPalette;
			}
			EnsureVisualState();
		}

		private void ColorPalettes_CollectionChanged(object sender, NotifyCollectionChangedEventArgs e)
		{
			if (ColorStripManager != null)
			{
				if (ColorPalettes == null || ColorPalettes.Count == 0)
				{
					ColorStripManager.ClearIsHoverSilent();
					ColorStripManager.ClearIsSelectedSilent();
					foreach (ColorStrip colorStrip in ColorStripManager.ColorStrips)
					{
						colorStrip.ColorPalette = null;
					}
					CurrentPalette = null;
					_recentColorPalette = null;
					SelectedColor = null;
					SelectedColorPreview = null;
				}
				else if (e.Action == NotifyCollectionChangedAction.Add && ColorPalettes.Count == 1)
				{
					if (CurrentPalette == null)
					{
						ColorPalette currentPalette = ColorPalettes[0];
						CurrentPalette = currentPalette;
					}
				}
				else if (e.Action == NotifyCollectionChangedAction.Remove && e.OldItems != null)
				{
					for (int i = 0; i < e.OldItems.Count; i++)
					{
						if ((ColorPalette)e.OldItems[i] == CurrentPalette)
						{
							if (ColorPalettes.Count > 0)
							{
								CurrentPalette = ColorPalettes[0];
							}
							else
							{
								CurrentPalette = null;
							}
							if (SelectedColor.HasValue && CurrentPalette == null)
							{
								SelectedColor = null;
								SelectedColorPreview = null;
							}
						}
					}
				}
			}
			EnsureVisualState();
		}

		{
			if (!_isMouseDownInside)
			{
				Color? previouslySelectedColor = PreviouslySelectedColor;
				IsDropDownOpen = false;
				if (!IsDropDownOpen && previouslySelectedColor.HasValue)
				{
					PreviouslySelectedColor = previouslySelectedColor;
					SelectedColor = previouslySelectedColor;
					SelectedColorPreview = previouslySelectedColor;

		}

		private void KeyDownHandler(object sender, KeyEventArgs e)
		{
			if (e.Handled)
			{
				return;
			}
			if (e.Key == Key.Escape)
			{
				if (PreviouslySelectedColor.HasValue)
				{
					SelectedColor = PreviouslySelectedColor;
					SelectedColorPreview = PreviouslySelectedColor;
				}
				bool handled = false;
				if (ColorPickerDialog != null && ColorPickerDialog.IsOpen)
				{
					handled = true;
					ColorPickerDialog.IsOpen = false;
				}
				if (IsDropDownOpen)
				{
					handled = true;
					IsDropDownOpen = false;
				}
				e.Handled = handled;
			}
			else if (e.Key == Key.Space || (e.Key == Key.Down && (Keyboard.Modifiers & ModifierKeys.Alt) != ModifierKeys.None))
			{
				IsDropDownOpen = true;
				e.Handled = true;
			}
		}

		private void RootVisual_MouseLeftButtonDown(object sender, MouseButtonEventArgs e)
		{
			bool isMouseDownInside = false;
			DependencyObject dependencyObject = e.OriginalSource as DependencyObject;
			if (dependencyObject is UIElement)
			{
				while (dependencyObject != null)
				{
					if (dependencyObject == this)
					{
						isMouseDownInside = true;
						break;
					}
					DependencyObject parent = VisualTreeHelper.GetParent(dependencyObject);
					if (parent == null && dependencyObject is FrameworkElement frameworkElement)
					{
						parent = frameworkElement.Parent;
					}
					dependencyObject = parent;
				}
			}
			_isMouseDownInside = isMouseDownInside;
		}

		private static void OnIsDropDownOpenChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
		{
			XamColorPicker xamColorPicker = (XamColorPicker)d;
			if ((!xamColorPicker.IsLoaded && (bool)e.NewValue) || (xamColorPicker.ColorPickerDialog != null && xamColorPicker.ColorPickerDialog.IsOpen && xamColorPicker.ColorPickerDialog.ActualHeight > 0.0))
			{
				xamColorPicker._ignoreIsDropDownChanging = true;
				xamColorPicker.IsDropDownOpen = false;
				xamColorPicker._ignoreIsDropDownChanging = false;
			}
			else
			{
				xamColorPicker.EnsureVisualState();
				if (!xamColorPicker._ignoreIsDropDownChanging)
				{
					if ((bool)e.NewValue)
					{
						if (!xamColorPicker.OnDropDownOpening())
						{
							UIElement rootVisual = PlatformProxy.GetRootVisual(xamColorPicker);
							if (rootVisual != null)
							{
								rootVisual.AddHandler(UIElement.MouseLeftButtonDownEvent, new MouseButtonEventHandler(xamColorPicker.RootVisual_MouseLeftButtonDown), handledEventsToo: true);
								rootVisual.AddHandler(UIElement.KeyDownEvent, new KeyEventHandler(xamColorPicker.KeyDownHandler), handledEventsToo: true);
								rootVisual.AddHandler(UIElement.MouseLeftButtonUpEvent, new MouseButtonEventHandler(xamColorPicker.RootVisual_MouseLeftButtonUp), handledEventsToo: true);
							}
							xamColorPicker.InvalidateDropDownPosition();
							if (xamColorPicker.CurrentPalette == null && xamColorPicker.ColorPalettes.Count > 0)
							{
								xamColorPicker.CurrentPalette = xamColorPicker.ColorPalettes[0];
							}
							xamColorPicker.OnDropDownOpened();
						}
						else
						{
							xamColorPicker._ignoreIsDropDownChanging = true;
							xamColorPicker.IsDropDownOpen = false;
							xamColorPicker._ignoreIsDropDownChanging = false;
						}
					}
					else if (!xamColorPicker.OnDropDownClosing())
					{
						UIElement rootVisual2 = PlatformProxy.GetRootVisual(xamColorPicker);
						if (rootVisual2 != null)
						{
							rootVisual2.RemoveHandler(UIElement.MouseLeftButtonDownEvent, new MouseButtonEventHandler(xamColorPicker.RootVisual_MouseLeftButtonDown));
							rootVisual2.RemoveHandler(UIElement.KeyDownEvent, new KeyEventHandler(xamColorPicker.KeyDownHandler));
							rootVisual2.RemoveHandler(UIElement.MouseLeftButtonUpEvent, new MouseButtonEventHandler(xamColorPicker.RootVisual_MouseLeftButtonUp));
						}
						xamColorPicker.OnDropDownClosed();
					}

		private void Header_MouseMove(object sender, MouseEventArgs e)
		{
			if (isInMouseMove)
			{
				return;
			}
			isInMouseMove = true;
			try
			{
				if (!_isMouseDown)
				{
					return;
				}
				e.Handled = true;
				if (!_isDragging)
				{
					_header.CaptureMouse();
					Point position = e.GetPosition(_header);
					_offsetPoint = new Point(_header.ActualWidth - position.X, _header.ActualHeight - position.Y);
					_isDragging = true;
					return;
				}
				Point position2 = e.GetPosition(ColorPicker);
				_popup.VerticalOffset = position2.Y - _offsetPoint.Y;
				if (SystemParameters.IsMenuDropRightAligned)
				{
					_popup.HorizontalOffset = position2.X + _offsetPoint.X;
				}
				else
				{
					_popup.HorizontalOffset = position2.X - _offsetPoint.X;
				}
			}
			finally
			{
				isInMouseMove = false;
			}
		}

		private void SetRGBValuesToCurrentColor()
		{
			Red = CurrentColor.R;
			Blue = CurrentColor.B;
			Green = CurrentColor.G;
			Alpha = CurrentColor.A;
		}

		private void SetHSLValuesToCurrentColor()
		{
			if (!_suspendHSLNotify)
			{
				HSL hSL = HSL.FromColor(CurrentColor);
				H = hSL.H;
				L = hSL.L;
				S = hSL.S;

		}

		private void SetCMYKValuesToCurrentColor()
		{
			CMYK cMYK = CMYK.FromColor(CurrentColor);
			C = cMYK.C;
			M = cMYK.M;
			Y = cMYK.Y;
			K = cMYK.K;
		}

		private void SetLightnessMarker(HSB color)
		{
			if (_shadingCanvas != null)
			{
				double num = color.S / 100.0;
				double num2 = color.B / 100.0;
				if (!double.IsNaN(_lightnessRectange.Height))
				{
					double num3 = _lightnessRectange.ActualWidth * num;
					double num4 = _lightnessRectange.ActualHeight - _lightnessRectange.ActualHeight * num2;
					_lightnessSelector.SetValue(Canvas.LeftProperty, num3 - _lightnessSelector.Height / 2.0);
					_lightnessSelector.SetValue(Canvas.TopProperty, num4 - _lightnessSelector.Height / 2.0);
				}
			}
		}

		private void SetLightnessBox(double huePosition)
		{
			double actualHeight = _hueRectangle.ActualHeight;
			double num = huePosition / actualHeight;
			if (_hueRectangle.Fill is LinearGradientBrush linearGradientBrush)
			{
				int num2 = (int)((double)(linearGradientBrush.GradientStops.Count - 1) * num * 255.0);
				byte b = (byte)(num2 % 255);
				byte b2 = (byte)(255 - b);
				byte a = byte.MaxValue;
				Color transparent = Colors.Transparent;
				if (num >= 1.0)
				{
					num2 = 0;
				}
				transparent = (num2 / 255) switch
				{
					0 => Color.FromArgb(a, byte.MaxValue, b, 0), 
					1 => Color.FromArgb(a, b2, byte.MaxValue, 0), 
					2 => Color.FromArgb(a, 0, byte.MaxValue, b), 
					3 => Color.FromArgb(a, 0, b2, byte.MaxValue), 
					4 => Color.FromArgb(a, b, 0, byte.MaxValue), 
					5 => Color.FromArgb(a, byte.MaxValue, 0, b2), 
					_ => Colors.Black, 
				};
				_lightnessRectange.Fill = new SolidColorBrush(transparent);
			}
		}

		private void SetLightnessBox(HSB color)
		{
			if (_hueRectangle != null)
			{
				double lightnessBox = color.H / 360.0 * _hueRectangle.Height;
				SetLightnessBox(lightnessBox);
			}
		}

		private void SetHueMarker(double hue)
		{
			if (_hueRectangle != null)
			{
				double num = (double.IsNaN(_hueRectangle.Height) ? 0.0 : _hueRectangle.Height);
				double num2 = (double.IsNaN(_hueSelector.Height) ? 0.0 : _hueSelector.Height);
				double num3 = hue / 360.0 * num;
				_hueSelector.SetValue(Canvas.TopProperty, num3 - num2 / 2.0);
			}
		}

		internal void UpdateDisplayBoxes()
		{
			if (_lightnessSelector != null)
			{
				_lightnessSelector.Visibility = Visibility.Collapsed;
				_lightnessSelector.Visibility = Visibility.Visible;
				HSB hSB = HSB.FromColor(CurrentColor);
				SetHueMarker(hSB.H);
				SetLightnessBox(hSB);
				SetLightnessMarker(hSB);
			}
		}

		private Color GenerateNewColor()
		{
			return Color.FromArgb(Alpha, Red, Green, Blue);
		}

		private HSL GenerateNewHSLColor()
		{
			return HSL.FromHSL(H, S, L, Alpha);
		}

		private CMYK GenerateNewCMYKColor()
		{
			return CMYK.FromCMYK(C, M, Y, K, Alpha);
		}

		private void EnsureVisualStates()
		{
			if (ColorSliderView == ColorSliderView.RGB)
			{
				VisualStateManager.GoToState(this, "RGB", useTransitions: false);
			}
			else if (ColorSliderView == ColorSliderView.HSL)
			{
				VisualStateManager.GoToState(this, "HSL", useTransitions: false);
			}
			else if (ColorSliderView == ColorSliderView.CMYK)
			{
				VisualStateManager.GoToState(this, "CMYK", useTransitions: false);
			}
		}

		private void SetShadingCanvasPointer(double x, double y)
		{
			int num = (int)x;
			int num2 = (int)y;
			_lightnessSelector.SetValue(Canvas.LeftProperty, (double)num - _lightnessSelector.Height / 2.0);
			_lightnessSelector.SetValue(Canvas.TopProperty, (double)num2 - _lightnessSelector.Height / 2.0);
		}

		private void UpdateColorSystem(ColorSliderView colorSystem)
		{
			switch (colorSystem)
			{
			case ColorSliderView.CMYK:
				SetHSLValuesToCurrentColor();
				SetRGBValuesToCurrentColor();
				break;
			case ColorSliderView.RGB:
				SetHSLValuesToCurrentColor();
				SetCMYKValuesToCurrentColor();
				break;
			case ColorSliderView.HSL:
				SetCMYKValuesToCurrentColor();
				SetRGBValuesToCurrentColor();
				break;
			}
		}

		private void RangeCheck(double lowValue, double highValue, string errorMsg, double currentValue)
		{
			if (currentValue < lowValue || currentValue > highValue)
			{
				throw new ArgumentException(GetString(errorMsg));
			}
		}

		private static void CurrentColorChanged(DependencyObject obj, DependencyPropertyChangedEventArgs e)
		{
			AdvancedColorShadePicker advancedColorShadePicker = (AdvancedColorShadePicker)obj;
			advancedColorShadePicker.OnPropertyChanged("CurrentColor");
			if (!advancedColorShadePicker._suspendNotifyPropertyChange)
			{
				advancedColorShadePicker._suspendNotifyPropertyChange = true;
				advancedColorShadePicker.SetCMYKValuesToCurrentColor();
				advancedColorShadePicker.SetHSLValuesToCurrentColor();
				advancedColorShadePicker.SetRGBValuesToCurrentColor();
				advancedColorShadePicker._suspendNotifyPropertyChange = false;
			}
		}

		private static void SelectedColorChanged(DependencyObject obj, DependencyPropertyChangedEventArgs e)
		{
			AdvancedColorShadePicker advancedColorShadePicker = (AdvancedColorShadePicker)obj;
			advancedColorShadePicker.OnPropertyChanged("SelectedColor");
			if (!advancedColorShadePicker.SelectedColor.HasValue)
			{
				advancedColorShadePicker.CurrentColor = Colors.Transparent;
			}
			else
			{
				advancedColorShadePicker.CurrentColor = advancedColorShadePicker.SelectedColor.Value;
			}
			HSL hSL = HSL.FromColor(advancedColorShadePicker.CurrentColor);
			advancedColorShadePicker.UpdateDisplayBoxes();
			advancedColorShadePicker.SetHueMarker(hSL.H);
		}

		private void ProcessHueCanvasMouseEvent(MouseEventArgs e)
		{
			Point position = e.GetPosition(_hueRectangle);
			int num = (int)((position.Y < 0.0) ? 0.0 : ((position.Y > _hueRectangle.ActualHeight) ? _hueRectangle.ActualHeight : position.Y));
			SetLightnessBox(num);
			_hueSelector.SetValue(Canvas.TopProperty, (double)num - _hueSelector.Height / 2.0);
			_suspendHSLNotify = true;
			H = (double)num / _hueRectangle.ActualHeight * 360.0;
			Color currentColor = new HSL
			{
				Alpha = Alpha,
				H = H,
				S = S,
				L = L
			}.ToColor();
			CurrentColor = currentColor;
			_suspendHSLNotify = false;
		}

		private void ProcessShadingCanvasMouseEvent(MouseEventArgs e)
		{
			if (_isProcessShadingCanvasMouseEvent)
			{
				return;
			}
			_isProcessShadingCanvasMouseEvent = true;
			try
			{
				ProcessShadingCanvasMouseEventImpl(e);
			}
			finally
			{
				_isProcessShadingCanvasMouseEvent = false;
			}
		}

		private void ProcessShadingCanvasMouseEventImpl(MouseEventArgs e)
		{
			Point position = e.GetPosition(_hueRectangle);
			if (!(position.Y < 0.0))
			{
				if (!(position.Y > _hueRectangle.ActualHeight))
				{
					_ = position.Y;
				}
				else
				{
					_ = _hueRectangle.ActualHeight;
				}
			}
			Point position2 = e.GetPosition(_shadingCanvas);
			double num = ((position2.X < 0.0) ? 0.0 : ((position2.X > _shadingCanvas.ActualWidth) ? _shadingCanvas.ActualWidth : position2.X));
			double num2 = ((position2.Y < 0.0) ? 0.0 : ((position2.Y > _shadingCanvas.ActualHeight) ? _shadingCanvas.ActualHeight : position2.Y));
			HSL hSL = new HSL
			{
				Alpha = CurrentColor.A,
				H = H
			};
			_suspendHSLNotify = true;
			double num3 = num / _shadingCanvas.ActualWidth;
			double num4 = 1.0 - num2 / _shadingCanvas.ActualHeight;
			double num5 = (2.0 - num3) * num4;
			double num6 = num3 * num4;
			double val = 0.0;
			double val2 = num5 / 2.0;
			if (num6 != 0.0)
			{
				val = num6 / ((num5 <= 1.0) ? num5 : (2.0 - num5));
			}
			double s = (hSL.S = Math.Min(1.0, Math.Max(0.0, val)) * 100.0);
			S = s;
			s = (hSL.L = Math.Min(1.0, Math.Max(0.0, val2)) * 100.0);
			L = s;
			_shadingCanvas.CaptureMouse();
			SetShadingCanvasPointer(num, num2);
			Color currentColor = hSL.ToColor();
			CurrentColor = currentColor;
			_suspendHSLNotify = false;
		}

				ProcessShadingCanvasMouseEvent(e);
				e.Handled = true;
			}
		}

			else if (ColorItem != null && ColorItem.WasSelected)
			{
				VisualStateManager.GoToState(this, "WasSelected", useTransitions: false);
			}
		}

			}

			EnsureState();
		}

		private void EnsurePanelChildrenSetup()
		{
			if (_colorPickerPanel == null)
			{
				return;
			}
			_colorBoxItems = new List<ColorItemBox>();
			int count = _colorPickerPanel.Children.Count;
			int i = 0;
			for (int j = 0; j < count; j++)
			{
				ColorItemBox colorItemBox = _colorPickerPanel.Children[j] as ColorItemBox;
				if (ColorItemBoxStyle != null)
				{
					colorItemBox.Style = ColorItemBoxStyle;
				}
				else if (colorItemBox.Style != null)
				{
					colorItemBox.ClearValue(FrameworkElement.StyleProperty);
				}
				if (colorItemBox != null)
				{
					_colorBoxItems.Add(colorItemBox);
					i++;
				}
				if (i >= VisibleItemCount)
				{
					break;
				}
			}
			for (; i < VisibleItemCount; i++)
			{
				ColorItemBox colorItemBox2 = new ColorItemBox();
				if (ColorItemBoxStyle != null)
				{
					colorItemBox2.Style = ColorItemBoxStyle;
				}
				_colorPickerPanel.Children.Add(colorItemBox2);
				_colorBoxItems.Add(colorItemBox2);
			}
		}

		private void PopulateChildren()
		{
			int i = 0;
			if (_colorBoxItems == null)
			{
				return;
			}
			if (ColorPalette != null)
			{
				if (ColorPalette.Colors != null)
				{
					for (; i < ColorPalette.Colors.Count && i < _colorItems.Count; i++)
					{
						_colorItems[i].Color = ColorPalette.Colors[i].Color;
						_colorBoxItems[i].ColorItem = _colorItems[i];
					}
				}
				new SolidColorBrush(ColorPalette.DefaultColor);
				for (; i < VisibleItemCount; i++)
				{
					_colorItems[i].Color = ColorPalette.DefaultColor;
					_colorBoxItems[i].ColorItem = _colorItems[i];
				}
			}
			else
			{
				new SolidColorBrush(Colors.Transparent);
				for (; i < VisibleItemCount; i++)
				{
					_colorItems[i].Color = Colors.Transparent;
					_colorBoxItems[i].ColorItem = _colorItems[i];
				}
			}
		}

		protected void EnsureVisualState()
		{
			if (_colorBoxItems == null)
			{
				return;
			}
			foreach (ColorItemBox colorBoxItem in _colorBoxItems)
			{
				colorBoxItem.EnsureState();
			}
		}

		protected internal void SetSelectedColorItem(ColorItem ci, bool fromClick)
		{
			if (_selectedColorItem == ci)
			{
				return;
			}
			ColorItem selectedColorItem = _selectedColorItem;
			if (_colorItems.Contains(ci) || ci == null)
			{
				if (_selectedColorItem != null)
				{
					_selectedColorItem.SetSelectedInternal(selected: false);
				}
				_selectedColorItem = ci;
				if (_selectedColorItem != null)
				{
					_selectedColorItem.SetSelectedInternal(selected: true);
				}
				OnSelectedColorItemChanged(selectedColorItem, ci);
			}
			EnsureVisualState();
		}

		protected internal void ClearIsHoverSilent()
		{
			foreach (ColorItem colorItem in _colorItems)
			{
				colorItem.SetHoverInternal(hover: false);
			}
			_hoverColorItem = null;
		}

		protected internal void ClearIsSelectedSilent()
		{
			foreach (ColorItem colorItem in _colorItems)
			{
				colorItem.SetSelectedInternal(selected: false);
				colorItem.SetWasSelected(wasSelected: false);
			}
			_selectedColorItem = null;
		}

		internal void ClearSelectedColorInternal()
		{
			_suppressSelectedColorItem = true;
			SelectedColorItem = null;
			_suppressSelectedColorItem = false;
		}

		internal ColorItem SetPreviouslySelectedColorInteral(Color color)
		{
			ColorItem colorItem = null;
			foreach (ColorItem colorItem2 in _colorItems)
			{
				colorItem2.SetHoverInternal(hover: false);
				if (colorItem2.Color == color && colorItem == null)
				{
					colorItem2.SetWasSelected(wasSelected: true);
					colorItem = colorItem2;
				}
				else
				{
					colorItem2.SetWasSelected(wasSelected: false);
				}
			}
			return colorItem;
		}

		protected virtual void OnSelectedColorItemChanged(ColorItem oldColorItem, ColorItem newColorItem)
		{
			if (this.SelectedColorItemChanged != null)
			{
				this.SelectedColorItemChanged(this, new SelectedColorItemChangedEventArgs
				{
					OriginalColorItem = oldColorItem,
					NewColorItem = newColorItem
				});
			}
		}

		protected virtual void OnHoverColorItemChanged(ColorItem oldColorItem, ColorItem newColorItem)
		{
			if (this.HoverColorItemChanged != null)
			{
				this.HoverColorItemChanged(this, new SelectedColorItemChangedEventArgs
				{
					OriginalColorItem = oldColorItem,
					NewColorItem = newColorItem
				});
			}
		}
	}
	public class ColorStripManager : DependencyObject
	{
		private ColorItem _selectedColorItem;

		private ColorItem _hoverColorItem;

		private ObservableCollection<ColorStrip> _colorStrips;

		private Color? _previouslySelectedColor;

		public static readonly DependencyProperty ManagerProperty = DependencyProperty.RegisterAttached("Manager", typeof(ColorStripManager), typeof(ColorStripManager), new PropertyMetadata(ManagerChanged));

		protected internal ColorItem SelectedColorItem
		{
			get
			{
				return _selectedColorItem;
			}
			set
			{
				ColorItem selectedColorItem = _selectedColorItem;
				if (_selectedColorItem != null)
				{
					_selectedColorItem.SetSelectedInternal(selected: false);
				}
				_selectedColorItem = value;
				OnSelectedColorItemChanged(selectedColorItem, value);
			}
		}

		protected internal ColorItem HoverColorItem
		{
			get
			{
				return _hoverColorItem;
			}
			set
			{
				if (_hoverColorItem != value)
				{
					ColorItem hoverColorItem = _hoverColorItem;
					if (_hoverColorItem != null)
					{
						_hoverColorItem.IsHover = false;
					}
					_hoverColorItem = value;
					OnHoverColorItemChanged(hoverColorItem, value);
				}
			}
		}

		protected internal XamColorPicker Parent { get; set; }

		public ObservableCollection<ColorStrip> ColorStrips
		{
			get
			{
				if (_colorStrips == null)
				{
					_colorStrips = new ObservableCollection<ColorStrip>();
					_colorStrips.CollectionChanged += ColorStrips_CollectionChanged;
				}
				return _colorStrips;
			}
			set
			{
				if (_colorStrips != value)
				{
					if (_colorStrips != null)
					{
						_colorStrips.CollectionChanged -= ColorStrips_CollectionChanged;
					}
					_colorStrips = value;
					if (_colorStrips != null)
					{
						_colorStrips.CollectionChanged += ColorStrips_CollectionChanged;
					}
				}
			}
		}

		protected internal event EventHandler<SelectedColorItemChangedEventArgs> HoverColorItemChanged;

		public event EventHandler<SelectedColorItemChangedEventArgs> SelectedColorItemChanged;

		private void AddSelectedItemChanged(ObservableCollection<ColorStrip> strips)
		{
			if (strips == null)
			{
				return;
			}
			foreach (ColorStrip strip in strips)
			{
				strip.SelectedColorItemChanged += ColorStrip_SelectedColorItemChanged;
			}
		}

		private void RemoveSelectedItemChanged(ObservableCollection<ColorStrip> strips)
		{
			if (strips == null)
			{
				return;
			}
			foreach (ColorStrip strip in strips)
			{
				strip.SelectedColorItemChanged -= ColorStrip_SelectedColorItemChanged;
			}
		}

		protected internal void SetPreviouslySelectedColorInternal(Color? color)
		{
			if (!color.HasValue && _selectedColorItem == null)
			{
				return;
			}
			bool flag = !color.HasValue;
			if (flag)
			{
				_previouslySelectedColor = null;
			}
			else
			{
				_previouslySelectedColor = color.Value;
			}
			foreach (ColorStrip colorStrip in _colorStrips)
			{
				if (flag)
				{
					colorStrip.ClearSelectedColorInternal();
					continue;
				}
				ColorItem colorItem = colorStrip.SetPreviouslySelectedColorInteral(color.Value);
				if (colorItem != null)
				{
					flag = true;
					_selectedColorItem = colorItem;
				}
			}
		}

		protected internal void ClearIsHoverSilent()
		{
			foreach (ColorStrip colorStrip in _colorStrips)
			{
				colorStrip.ClearIsHoverSilent();
			}
			_hoverColorItem = null;
		}

		protected internal void ClearIsSelectedSilent()
		{
			foreach (ColorStrip colorStrip in _colorStrips)
			{
				colorStrip.ClearIsSelectedSilent();
			}
			_selectedColorItem = null;
		}

		protected virtual void OnHoverColorItemChanged(ColorItem oldColorItem, ColorItem newColorItem)
		{
			if (this.HoverColorItemChanged != null)
			{
				this.HoverColorItemChanged(this, new SelectedColorItemChangedEventArgs
				{
					OriginalColorItem = oldColorItem,
					NewColorItem = newColorItem
				});
			}
		}

		protected virtual void OnSelectedColorItemChanged(ColorItem oldColorItem, ColorItem newColorItem)
		{
			if (this.SelectedColorItemChanged != null)
			{
				this.SelectedColorItemChanged(this, new SelectedColorItemChangedEventArgs
				{
					OriginalColorItem = oldColorItem,
					NewColorItem = newColorItem
				});
			}
		}

		private void ColorStrip_SelectedColorItemChanged(object sender, SelectedColorItemChangedEventArgs e)
		{
			SelectedColorItem = e.NewColorItem;
		}

		private void ColorStrips_CollectionChanged(object sender, NotifyCollectionChangedEventArgs e)
		{
			if (e.Action == NotifyCollectionChangedAction.Add && e.NewItems != null)
			{
				for (int i = 0; i < e.NewItems.Count; i++)
				{
					if (e.NewItems[i] is ColorStrip colorStrip)
					{
						colorStrip.SelectedColorItemChanged += ColorStrip_SelectedColorItemChanged;
						colorStrip.HoverColorItemChanged += ColorStrip_HoverColorItemChanged;
						colorStrip.Loaded += ColorStrip_Loaded;
					}
				}
			}
			if (e.Action != NotifyCollectionChangedAction.Remove || e.OldItems == null)
			{
				return;
			}
			for (int j = 0; j < e.OldItems.Count; j++)
			{
				if (e.OldItems[j] is ColorStrip colorStrip2)
				{
					colorStrip2.SelectedColorItemChanged -= ColorStrip_SelectedColorItemChanged;
					colorStrip2.HoverColorItemChanged -= ColorStrip_HoverColorItemChanged;
					colorStrip2.Loaded -= ColorStrip_Loaded;
				}
			}
		}

		private void ColorStrip_Loaded(object sender, RoutedEventArgs e)
		{
			SetPreviouslySelectedColorInternal(_previouslySelectedColor);
		}

		private void ColorStrip_HoverColorItemChanged(object sender, SelectedColorItemChangedEventArgs e)
		{
			HoverColorItem = e.NewColorItem;
		}

		public static void SetManager(DependencyObject obj, ColorStripManager propertyValue)
		{
			obj.SetValue(ManagerProperty, propertyValue);
		}

		public static ColorStripManager GetManager(DependencyObject obj)
		{
			return (ColorStripManager)obj.GetValue(ManagerProperty);
		}

		private static void ManagerChanged(object sender, DependencyPropertyChangedEventArgs args)
		{
			if (sender is ColorStrip item)
			{
				if (args.OldValue != null && args.OldValue is ColorStripManager colorStripManager)
				{
					colorStripManager.ColorStrips.Remove(item);
				}
				if (args.NewValue != null && args.NewValue is ColorStripManager colorStripManager2)
				{
					colorStripManager2.ColorStrips.Add(item);
				}
			}
		}
	}

		};

		{
			SnapsToDevicePixels = true,

		};

		{
			Width = 8.0,
			Height = 8.0,

			UseLayoutRounding = true
		};

		private readonly Rectangle _tile2 = new Rectangle
		{
			Width = 8.0,
			Height = 8.0,
			SnapsToDevicePixels = true,

		{
			get
			{
				return _squareSize;
			}
			set
			{
				if (_squareSize != value)
				{
					if (value <= 0)
					{
						throw new ArgumentException(SR.GetString("SquareSizeOutOfRange"));
					}
					_squareSize = value;
					Invalidate();
				}
			}
		}

		{
			FrameworkElement.DefaultStyleKeyProperty.OverrideMetadata(typeof(TransparencyBackgroundControl), new FrameworkPropertyMetadata(typeof(TransparencyBackgroundControl)));
		}

		{
			base.IsTabStop = false;
			_canvas = new Canvas();
			Canvas.SetLeft(_tile2, 8.0);
			Canvas.SetTop(_tile2, 8.0);
			_canvas.Children.Add(_tile1);
			_canvas.Children.Add(_tile2);
			_visualBrush.Visual = _canvas;
			base.Focusable = false;
			base.SnapsToDevicePixels = true;
			base.UseLayoutRounding = true;

			if (_root != null)

				_root.SizeChanged += RootSizeChanged;
			}

		{
			if (_root != null)

		{
			_canvas.Background = new SolidColorBrush(_color1);
			_tile1.Fill = new SolidColorBrush(_color2);
			_tile2.Fill = new SolidColorBrush(_color2);
			Rectangle tile = _tile1;
			Rectangle tile2 = _tile1;
			Rectangle tile3 = _tile2;
			double num = (_tile2.Height = _squareSize);
			double num3 = (tile3.Width = num);
			double width = (tile2.Height = num3);
			tile.Width = width;
			Canvas.SetLeft(_tile2, _squareSize);
			Canvas.SetTop(_tile2, _squareSize);
			VisualBrush visualBrush = _visualBrush;
			Rect viewport = (_visualBrush.Viewport = new Rect(0.0, 0.0, 2 * _squareSize, 2 * _squareSize));
			visualBrush.Viewport = viewport;
		}

			_canvas.Height = (int)Math.Ceiling(e.NewSize.Height);

	public class PreviousPaletteCommand : PickerCommandBase
	{
		public override void Execute(object parameter)
		{
			if (!(parameter is XamColorPicker xamColorPicker))
			{
				return;
			}
			if (xamColorPicker.CurrentPalette == null)
			{
				if (xamColorPicker.ColorPalettes.Count > 0)
				{
					xamColorPicker.CurrentPalette = xamColorPicker.ColorPalettes[0];
				}
				return;
			}
			int num = xamColorPicker.ColorPalettes.IndexOf(xamColorPicker.CurrentPalette);
			if (num > 0)
			{
				xamColorPicker.CurrentPalette = xamColorPicker.ColorPalettes[--num];
			}
		}
	}
	public class NextPaletteCommand : PickerCommandBase
	{
		public override void Execute(object parameter)
		{
			if (!(parameter is XamColorPicker xamColorPicker))
			{
				return;
			}
			if (xamColorPicker.CurrentPalette == null)
			{
				if (xamColorPicker.ColorPalettes.Count > 0)
				{
					xamColorPicker.CurrentPalette = xamColorPicker.ColorPalettes[0];
				}
				return;
			}
			int num = xamColorPicker.ColorPalettes.IndexOf(xamColorPicker.CurrentPalette);
			if (num < xamColorPicker.ColorPalettes.Count - 1)
			{
				xamColorPicker.CurrentPalette = xamColorPicker.ColorPalettes[++num];
			}
		}
	}

				xamColorPicker.ColorPickerDialog.IsOpen = true;
			}
		}
