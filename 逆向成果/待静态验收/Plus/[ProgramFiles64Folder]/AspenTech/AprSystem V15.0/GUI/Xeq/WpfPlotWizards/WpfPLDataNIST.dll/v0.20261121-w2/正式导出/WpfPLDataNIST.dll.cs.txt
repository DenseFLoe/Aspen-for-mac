		public string Join(List<string> xW, string sFill, bool bTrail)
		{
			string text = "";
			for (int i = 0; i < xW.Count; i++)
			{
				if (!string.IsNullOrEmpty(xW[i]))
				{
					text += xW[i];
					if (i != xW.Count - 1 || bTrail)
					{
						text += sFill;
					}
				}
				else if (i == xW.Count - 1 && bTrail && text.Length == 0)
				{
					text = sFill;
				}
			}
			return text;
		}

		private void Build_Node_Lookup()
		{
			int count = ComponentList.Count;
			Xc = new int[count];
			X2c = new int[count];
			Yc = new int[count];
			for (int i = 0; i < count; i++)
			{
				Xc[i] = -1;
				X2c[i] = -1;
				Yc[i] = -1;
			}
			IHNode iHNode = _context.FindNode(VarNames[2]);
			if (iHNode == null)
			{
				return;
			}
			int num = iHNode.Elements.get_RowCount(1);
			for (int j = 0; j < num; j++)
			{
				string text = iHNode.Elements.get_Label(1, j, Type.Missing);
				string text2 = iHNode.Elements.get_Label(2, j, Type.Missing);
				int num2 = -1;
				switch (text)
				{
				case "TEMPERATURE":
					if (TempElement == -1)
					{
						TempUnit = text2;
						TempElement = j;
					}
					break;
				case "PRESSURE":
					if (PresElement == -1)
					{
						PresUnit = text2;
						PresElement = j;
					}
					break;
				case "X":
				case "X1":
					num2 = ComponentList.IndexOf(text2);
					if (num2 >= 0)
					{
						Xc[num2] = j;
					}
					break;
				case "X2":
					num2 = ComponentList.IndexOf(text2);
					if (num2 >= 0)
					{
						X2c[num2] = j;
					}
					break;
				case "Y":
					num2 = ComponentList.IndexOf(text2);
					if (num2 >= 0)
					{
						Yc[num2] = j;
					}
					break;
				}
			}
		}

		private void DetermineMoleType(IHNode Context, out string moleMass, out string moleType)
		{
			moleMass = "";
			moleType = "";
			IHNode iHNode = Context.FindNode("Input/COMPOSITION");
			if (iHNode != null)
			{
				switch ((dynamic)iHNode.get_Value(Type.Missing))
				{
				case "MOLE-FRAC":
					moleType = "Mole fraction";
					moleMass = "fraction";
					break;
				case "MASS-FRAC":
					moleType = "Mass fraction";
					moleMass = "fraction";
					break;
				case "MOLE-PCT":
					moleType = "Mole fraction";
					moleMass = "";
					break;
				case "MASS-PCT":
					moleType = "Mass fraction";
					moleMass = "";
					break;
				}
			}
		}
