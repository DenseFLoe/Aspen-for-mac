		public string Join(string[] xW, string sFill, bool bTrail)
		{
			string text = "";
			if (xW != null)
			{
				for (int i = 0; i < xW.Length; i++)
				{
					if (!string.IsNullOrEmpty(xW[i]))
					{
						text += xW[i];
						if (i != xW.Length - 1 || bTrail)
						{
							text += sFill;
						}
					}
					else if (i == xW.Length - 1 && bTrail && text.Length == 0)
					{
						text = sFill;
					}
				}
			}
			return text;
		}

		public string[] ParseLine(string linein)
		{
			string[] array = linein.Split(' ');
			if (array == null)
			{
				return null;
			}
			List<string> list = new List<string>();
			for (int i = 0; i < array.Length; i++)
			{
				if (!string.IsNullOrEmpty(array[i]))
				{
					list.Add(array[i]);
				}
			}
			if (list.Count == 0)
			{
				return null;
			}
			return list.ToArray();
		}
