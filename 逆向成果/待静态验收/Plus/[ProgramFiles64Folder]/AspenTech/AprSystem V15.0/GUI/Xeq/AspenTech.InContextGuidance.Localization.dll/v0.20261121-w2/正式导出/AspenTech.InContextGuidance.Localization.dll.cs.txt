public static class AspenLanguageSelector
{
	private static readonly List<string> _supportedLanguages = new List<string> { "ENG", "FRA", "ITA", "DEU", "SPA", "JPN", "CHS", "KOR", "RUS", "PTB" };

	public static void ApplyAspenLanguageSelection()
	{
		string keyName = "HKEY_CURRENT_USER\\Software\\AspenTech\\Engineering\\Language";
		string text = Registry.GetValue(keyName, null, string.Empty) as string;
		if (string.IsNullOrEmpty(text))
		{
			string text2 = Thread.CurrentThread.CurrentUICulture.ThreeLetterWindowsLanguageName;
			if (!_supportedLanguages.Contains(text2))
			{
				text2 = "ENG";
			}
			Registry.SetValue(keyName, null, text2);
			text = text2;
		}
		string text3 = text.ToUpperInvariant() switch
		{
			"ENG" => "en-US", 
			"FRA" => "fr-FR", 
			"ITA" => "it-IT", 
			"DEU" => "de-DE", 
			"SPA" => "es-ES", 
			"JPN" => "ja-JP", 
			"CHS" => "zh-HANS", 
			"KOR" => "ko-KR", 
			"RUS" => "ru-RU", 
			"PTB" => "pt-BR", 
			_ => "en-US", 
		};
		if (!string.IsNullOrEmpty(text3))
		{
			CultureInfo cultureInfo = null;
			try
			{
				cultureInfo = new CultureInfo(CultureInfo.CreateSpecificCulture(text3).Name, useUserOverride: true);
			}
			catch (CultureNotFoundException)
			{
			}
			if (cultureInfo != null)
			{
				Thread.CurrentThread.CurrentUICulture = cultureInfo;
				return;
			}
			CultureInfo currentUICulture = new CultureInfo("en-US");
			Thread.CurrentThread.CurrentUICulture = currentUICulture;
		}
	}
}
