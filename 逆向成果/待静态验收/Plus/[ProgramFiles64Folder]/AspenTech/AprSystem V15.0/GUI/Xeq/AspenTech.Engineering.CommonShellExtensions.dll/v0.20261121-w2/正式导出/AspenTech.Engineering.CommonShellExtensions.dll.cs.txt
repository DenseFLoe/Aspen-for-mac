		{

			{
				return view as INavigatorService;
			}
			if (view is IServiceProvider serviceProvider)
			{
				object service = serviceProvider.GetService(typeof(INavigatorService));
				if (service != null)
				{
					return service as INavigatorService;
				}

	internal class InternalModuleInfoLoader : IDisposable
	{
		private Assembly[] _loadedAssemblies;

		private Type _iModuleType;

		private string ReflectionOnlyLoadDirectory { get; set; }

		private Assembly[] LoadedAssemblies
		{
			get
			{
				if (_loadedAssemblies == null)
				{
					_loadedAssemblies = AppDomain.CurrentDomain.ReflectionOnlyGetAssemblies();
				}
				return _loadedAssemblies;
			}
		}

		public Type IModuleType
		{
			get
			{
				if (_iModuleType == null)
				{
					Assembly assembly = LoadedAssemblies.First((Assembly asm) => asm.FullName == typeof(IModule).Assembly.FullName);
					_iModuleType = assembly.GetType(typeof(IModule).FullName);
				}
				return _iModuleType;
			}
		}

		public InternalModuleInfoLoader(string reflectionOnlyLoadDirectory = null)
		{
			ReflectionOnlyLoadDirectory = reflectionOnlyLoadDirectory;
			Prepare();
		}

		public void LoadModuleInfos(ModuleCatalog catalog, string filePath)
		{
			if (LoadedAssemblies.Any((Assembly x) => string.Equals(x.Location, filePath, StringComparison.OrdinalIgnoreCase)))
			{
				return;
			}
			Assembly assembly = null;
			try
			{
				assembly = Assembly.ReflectionOnlyLoadFrom(filePath);
			}
			catch (Exception ex)
			{
				Trace.TraceWarning(string.Format(CultureInfo.CurrentCulture, "Exception while loading modules from assembly {0}: {1}", filePath, ex.Message));
			}
			if (!(assembly != null))
			{
				return;
			}
			foreach (Type item in from t in assembly.GetExportedTypes().Where(IModuleType.IsAssignableFrom)
				where t != IModuleType
				where !t.IsAbstract
				select t)
			{
				ModuleInfo moduleInfo = ModuleLoader.CreateModuleInfo(item);
				if (moduleInfo != null)
				{
					catalog.AddModule(moduleInfo);
				}
			}
		}

		private void Prepare()
		{
			Assembly[] assemblies = AppDomain.CurrentDomain.GetAssemblies();
			foreach (Assembly assembly in assemblies)
			{
				if (assembly is AssemblyBuilder)
				{
					continue;
				}
				string text = string.Empty;
				try
				{
					text = assembly.Location;
				}
				catch (Exception)
				{
				}
				if (!string.IsNullOrEmpty(text))
				{
					try
					{
						Assembly.ReflectionOnlyLoadFrom(text);
					}
					catch (Exception)
					{
					}
				}
			}
			RegisterReflectionOnlyResolve();
		}

		private void RegisterReflectionOnlyResolve()
		{
			if (!string.IsNullOrWhiteSpace(ReflectionOnlyLoadDirectory))
			{
				AppDomain.CurrentDomain.ReflectionOnlyAssemblyResolve += CurrentDomain_ReflectionOnlyAssemblyResolve;
			}
		}

		private void UnregisterReflectionOnlyResolve()
		{
			if (!string.IsNullOrWhiteSpace(ReflectionOnlyLoadDirectory))
			{
				AppDomain.CurrentDomain.ReflectionOnlyAssemblyResolve -= CurrentDomain_ReflectionOnlyAssemblyResolve;
			}
		}

		private Assembly CurrentDomain_ReflectionOnlyAssemblyResolve(object sender, ResolveEventArgs args)
		{
			return OnReflectionOnlyResolve(args, ReflectionOnlyLoadDirectory);
		}

		private Assembly OnReflectionOnlyResolve(ResolveEventArgs args, string directory)
		{
			Assembly assembly = AppDomain.CurrentDomain.ReflectionOnlyGetAssemblies().FirstOrDefault((Assembly asm) => string.Equals(asm.FullName, args.Name, StringComparison.OrdinalIgnoreCase));
			if (assembly != null)
			{
				return assembly;
			}
			AssemblyName assemblyName = new AssemblyName(args.Name);
			string text = Path.Combine(directory, assemblyName.Name + ".dll");
			if (File.Exists(text))
			{
				return Assembly.ReflectionOnlyLoadFrom(text);
			}
			return Assembly.ReflectionOnlyLoad(args.Name);
		}

		public void Dispose()
		{
			UnregisterReflectionOnlyResolve();
		}
	}

		public static bool VerifyIsAspenSigned(string filePath)
		{
			if (string.IsNullOrEmpty(filePath))
			{
				throw new ArgumentException("Parameter filePath cannot be null or empty.");
			}
			if (!File.Exists(filePath))
			{
				throw new ArgumentException("Specified file doesn't exist.");
			}
			try
			{
				X509Certificate2 x509Certificate = new X509Certificate2(X509Certificate.CreateFromSignedFile(filePath));
				if (x509Certificate != null && !string.IsNullOrWhiteSpace(x509Certificate.Subject) && ValidateCertSubjectAsAspenTechSigned(x509Certificate.Subject))
				{
					return ValidateCert509Chain(x509Certificate);
				}
				return false;
			}
			catch (Exception)
			{
				return false;
			}
		}

		public static bool VerifyIsAspenSigned(Assembly assembly)
		{
			if (assembly == null)
			{
				throw new ArgumentNullException("assembly");
			}
			try
			{
				System.Reflection.Module[] modules = assembly.GetModules();
				if (modules != null && modules.Length != 0)
				{
					X509Certificate signerCertificate = modules[0].GetSignerCertificate();
					if (signerCertificate != null)
					{
						X509Certificate2 x509Certificate = new X509Certificate2(signerCertificate);
						if (x509Certificate != null && !string.IsNullOrWhiteSpace(x509Certificate.Subject) && ValidateCertSubjectAsAspenTechSigned(x509Certificate.Subject))
						{
							return ValidateCert509Chain(x509Certificate);
						}
					}
				}
				return false;
			}
			catch (Exception)
			{
				return false;
			}
		}

		private static bool ValidateCertSubjectAsAspenTechSigned(string certSubjectLine)
		{
			if (certSubjectLine.IndexOf("C=US") < 0)
			{
				return false;
			}
			string arg = "Aspen Technology, Inc.";
			if (certSubjectLine.IndexOf($"CN=\"{arg}\"") < 0)
			{
				return false;
			}
			if (certSubjectLine.IndexOf($"O=\"{arg}\"") < 0)
			{
				return false;
			}
			return true;
		}

		private static bool ValidateCert509Chain(X509Certificate2 c2)
		{
			try
			{
				return new X509Chain
				{
					ChainPolicy = 
					{
						RevocationMode = X509RevocationMode.Offline,
						VerificationFlags = X509VerificationFlags.IgnoreNotTimeValid
					}
				}.Build(c2);
			}
			catch (Exception)
			{
				return false;
			}
		}

		public static void AddModulesFromDirectory(this ModuleCatalog catalog, string directory, string searchPattern, SearchOption searchOption, bool verifySignedByAspenTech)
		{
			if (catalog == null)
			{
				throw new ArgumentNullException("catalog");

			if (string.IsNullOrEmpty(directory))
			{
				throw new ArgumentException("Parameter directory cannot be null or empty.");

			if (!Directory.Exists(directory))
			{
				return;
			}
			using InternalModuleInfoLoader internalModuleInfoLoader = new InternalModuleInfoLoader(directory);
			string[] files = Directory.GetFiles(directory, searchPattern, searchOption);
			foreach (string filePath in files)

				if (!verifySignedByAspenTech || ModuleChecker.VerifyIsAspenSigned(filePath))

					internalModuleInfoLoader.LoadModuleInfos(catalog, filePath);

		public static ModuleInfo CreateModuleInfo(Type type)
		{
			if (type == null)
			{
				return null;
			}
			string name = type.Name;
			bool flag = false;
			CustomAttributeData customAttributeData = CustomAttributeData.GetCustomAttributes(type).FirstOrDefault((CustomAttributeData cad) => cad.Constructor.DeclaringType.FullName == typeof(ModuleAttribute).FullName);
			if (customAttributeData != null)
			{
				foreach (CustomAttributeNamedArgument namedArgument in customAttributeData.NamedArguments)
				{
					switch (namedArgument.MemberInfo.Name)
					{
					case "ModuleName":
						name = (string)namedArgument.TypedValue.Value;
						break;
					case "OnDemand":
						flag = (bool)namedArgument.TypedValue.Value;
						break;
					case "StartupLoaded":
						flag = !(bool)namedArgument.TypedValue.Value;
						break;
					}
				}
			}
			ModuleInfo moduleInfo = new ModuleInfo(name, type.AssemblyQualifiedName)
			{
				InitializationMode = (flag ? InitializationMode.OnDemand : InitializationMode.WhenAvailable),
				Ref = type.Assembly.CodeBase
			};
			foreach (CustomAttributeData item in from cad in CustomAttributeData.GetCustomAttributes(type)
				where cad.Constructor.DeclaringType.FullName == typeof(ModuleDependencyAttribute).FullName
				select cad)
			{
				moduleInfo.DependsOn.Add((string)item.ConstructorArguments[0].Value);
			}
			return moduleInfo;
		}
	}

		private static IEnumerable<T> ResolveConvertersInScope<T>(IUnityContainer container, int sourceVersion, int targetVersion, Func<T> defaultCreator, Func<T> emptyCreator) where T : IVersionConverter
		{
			List<T> list = (from x in UnityContainerExtensions.ResolveAll<T>(container, Array.Empty<ResolverOverride>())
				orderby x.TargetVersion
				select x).ToList();
			if (list.Count == 0)
			{
				return new T[1] { defaultCreator() };
			}
			List<T> list2 = ResolveOrderedConverters<T>(container, sourceVersion, targetVersion).ToList();
			if (list2.Count == 0)
			{
				if (list.LastOrDefault((T x) => x.TargetVersion <= sourceVersion) != null)
				{
					return new T[1] { emptyCreator() };
				}
				return new T[1] { defaultCreator() };
			}
			if (list2[list2.Count - 1].TargetVersion < targetVersion)
			{
				list2.Add(emptyCreator());
			}
			return list2;
		}

		private static IEnumerable<T> ResolveOrderedConverters<T>(IUnityContainer container, int sourceVersion, int targetVersion) where T : IVersionConverter
		{
			return UnityContainerExtensions.ResolveAll<T>(container, Array.Empty<ResolverOverride>()).Where(delegate(T x)
			{
				T val = x;
				if (val.TargetVersion > sourceVersion)
				{
					val = x;
					return val.TargetVersion <= targetVersion;
				}
				return false;
			}).OrderBy(delegate(T x)
			{
				T val = x;
				return val.TargetVersion;
			});
		}

			foreach (object item in screenDatasInTargetVersion)
			{
				converters.BatchConvert(item, sourceVersion);
			}
		}

		public static void DoDefaultConvert(this IScreenDataConverter converter, object source, int sourceVersion, IUnityContainer container)
		{
			if (converter == null)
			{
				throw new ArgumentNullException("converter");
			}
			if (container == null)
			{
				throw new ArgumentNullException("container");
			}
			if (source == null)
			{
				return;
			}
			IScreenDataConverter contextConverter = converter.GetContextConverter(container, () => new DefaultScreenDataConverter(container, sourceVersion));
			string screenFactoryKeyInTargetVersion = contextConverter.GetScreenFactoryKeyInTargetVersion(source);
			if (!string.IsNullOrEmpty(screenFactoryKeyInTargetVersion))
			{
				IScreenConverterCatalog screenConverterCatalog = container.TryResolve<IScreenConverterCatalog>();
				if (screenConverterCatalog != null)
				{
					object screenSubjectInTargetVersion = contextConverter.GetScreenSubjectInTargetVersion(source);
					object screenStateInTargetVersion = contextConverter.GetScreenStateInTargetVersion(source);
					ConvertScreenInSequence(converter, screenConverterCatalog, source, screenFactoryKeyInTargetVersion, screenSubjectInTargetVersion, screenStateInTargetVersion, sourceVersion, converter.TargetVersion);
				}
			}
		}

		public static T GetContextConverter<T>(this T converter, IUnityContainer container, Func<T> defaultCreator) where T : IVersionConverter
		{
			if (converter == null)
			{
				throw new ArgumentNullException("converter");
			}
			if (container == null)
			{
				throw new ArgumentNullException("container");
			}
			if (defaultCreator == null)
			{
				throw new ArgumentNullException("defaultCreator");
			}
			T val = UnityContainerExtensions.ResolveAll<T>(container, Array.Empty<ResolverOverride>()).Where(delegate(T x)
			{
				T val2 = x;
				return val2.TargetVersion < converter.TargetVersion;
			}).OrderByDescending(delegate(T x)
			{
				T val2 = x;
				return val2.TargetVersion;
			})
				.FirstOrDefault();
			if (val != null)
			{
				return val;
			}
			return defaultCreator();
		}

		private static void ConvertScreenInSequence(IScreenDataConverter converter, IScreenConverterCatalog catalog, object source, string factoryKey, object subject, object state, int sourceVersion, int targetVersion)
		{
			foreach (IScreenConverter item in catalog.FindConvertersInOrder(factoryKey, sourceVersion, targetVersion))
			{
				item.Convert(subject, state);
				if (item.IsScreenObsolete)
				{
					converter.UpdateScreenFactoryKey(source, string.Empty);
					break;
				}
				if (!string.IsNullOrEmpty(item.NewScreenFactoryKey))
				{
					converter.UpdateScreenFactoryKey(source, item.NewScreenFactoryKey);
					ConvertScreenInSequence(converter, catalog, source, item.NewScreenFactoryKey, subject, state, item.TargetVersion, targetVersion);
					break;
				}
			}
		}

		public static XElement ElementWithLocalNamePath(this XElement node, string localNamePath)
		{
			if (node == null)
			{
				throw new ArgumentNullException("node");
			}
			if (string.IsNullOrWhiteSpace(localNamePath))
			{
				throw new ArgumentException("Parameter localNamePath cannot be null or white spaces.");
			}
			XElement xElement = node;
			string[] array = localNamePath.Split('/');
			foreach (string localName in array)
			{
				xElement = xElement.InternalElementWithLocalName(localName);
				if (xElement == null)
				{
					return null;
				}
			}
			return xElement;
		}

		public static IEnumerable<XElement> ElementsWithLocalNamePath(this XElement node, string localNamePath)
		{
			if (node == null)
			{
				throw new ArgumentNullException("node");
			}
			if (string.IsNullOrWhiteSpace(localNamePath))
			{
				throw new ArgumentException("Parameter localNamePath cannot be null or white spaces.");
			}
			XElement xElement = node;
			string[] array = localNamePath.Split('/');
			for (int i = 0; i < array.Length - 1; i++)
			{
				xElement = xElement.InternalElementWithLocalName(array[i]);
				if (xElement == null)
				{
					break;
				}
			}
			if (xElement == null)
			{
				yield break;
			}
			string lastSegment = array[array.Length - 1];
			foreach (XElement item in xElement.Elements())
			{
				if (string.Equals(item.Name.LocalName, lastSegment, StringComparison.CurrentCulture))
				{
					yield return item;
				}
			}
		}

		private static XElement InternalElementWithLocalName(this XElement node, string localName)
		{
			if (node == null)
			{
				throw new ArgumentNullException("node");
			}
			if (localName == null)
			{
				throw new ArgumentNullException("localName");
			}
			return node.Elements().FirstOrDefault((XElement x) => string.Equals(x.Name.LocalName, localName, StringComparison.CurrentCulture));
		}

	public interface IScreenConverterCatalog
	{
		IEnumerable<IScreenConverter> FindConvertersInOrder(string screenFactoryKey, int sourceVersion, int targetVersion);

		void Register<T>() where T : IScreenConverter;

		void Register(string screenFactoryKey, int targetVersion, Action<object, object> subjectAndStateConverter, string newScreenFactoryKey = null);
	}
	public class ScreenConverterCatalog : IScreenConverterCatalog
	{
		private readonly IUnityContainer _container;

		private readonly List<IScreenConverter> _convertersList = new List<IScreenConverter>();

		public ScreenConverterCatalog(IUnityContainer container)
		{
			if (container == null)
			{
				throw new ArgumentNullException("container");
			}
			_container = container;
		}

		public IEnumerable<IScreenConverter> FindConvertersInOrder(string screenFactoryKey, int sourceVersion, int targetVersion)
		{
			return from x in _convertersList
				where string.Equals(x.ScreenFactoryKey, screenFactoryKey, StringComparison.InvariantCulture) && x.TargetVersion > sourceVersion && x.TargetVersion <= targetVersion
				orderby x.TargetVersion
				select x;
		}

		public void Register<T>() where T : IScreenConverter
		{
			T val = UnityContainerExtensions.Resolve<T>(_container, Array.Empty<ResolverOverride>());
			AppendIfNotExist(val);
		}

		public void Register(string screenFactoryKey, int targetVersion, Action<object, object> subjectAndStateConverter, string newScreenFactoryKey = null)
		{
			DelegateScreenConverter converter = new DelegateScreenConverter(screenFactoryKey, targetVersion, subjectAndStateConverter, newScreenFactoryKey);
			AppendIfNotExist(converter);
		}

		private void AppendIfNotExist(IScreenConverter converter)
		{
			if (!_convertersList.Any((IScreenConverter x) => IsSameConverter(x, converter)))
			{
				_convertersList.Add(converter);
			}
		}

		private static bool IsSameConverter(IScreenConverter first, IScreenConverter second)
		{
			if (first == null && second == null)
			{
				return true;
			}
			if (first == null || second == null)
			{
				return false;
			}
			if (string.Equals(first.ScreenFactoryKey, second.ScreenFactoryKey, StringComparison.InvariantCulture))
			{
				return first.TargetVersion == second.TargetVersion;
			}
			return false;
		}
	}

	internal class SettingsTransformer : ISettingsTransformer
	{
		private readonly IUnityContainer _unityContainer;

		public SettingsTransformer(IUnityContainer unityContainer)
		{
			if (unityContainer == null)
			{
				throw new ArgumentNullException("unityContainer");
			}
			_unityContainer = unityContainer;
		}

		public string Transform(string settings, int sourceVersion, int targetVersion)
		{
			if (string.IsNullOrEmpty(settings))
			{
				return settings;
			}
			if (sourceVersion >= targetVersion)
			{
				return settings;
			}
			try
			{
				XElement root = XDocument.Parse(settings).Root;
				ConvertersManager.ResolveShellSettingConverters(_unityContainer, sourceVersion, targetVersion).BatchConvert(root, sourceVersion);
				return root.ToString();
			}
			catch (Exception ex)
			{
				Log(ex.Message);
				return settings;
			}
		}

		private void Log(string message)
		{
			_unityContainer.TryResolve<ILoggerFacade>()?.Log(message, Category.Exception, Priority.Medium);
		}
	}
