	public sealed class ExchangeAdapter : INotifyPropertyChanged
	{
		private static ExchangeAdapter _default;

		private ProductName _productName;

		private ProductVersion _productVersion;

		private Action<ExchangeErrorInfo> _errorHandler;

		private bool _successfullyInitialized;

		private bool? _useELearningContent;

		private string _selectedLanguage;

		public static ExchangeAdapter Default => _default ?? (_default = new ExchangeAdapter());

		public bool SuccessfullyInitialized
		{
			get
			{
				return _successfullyInitialized;
			}
			private set
			{
				if (_successfullyInitialized != value)
				{
					_successfullyInitialized = value;
					OnPropertyChanged("SuccessfullyInitialized");
				}
			}
		}

		private ProductName ExchangeProductName => ConvertEnum<ProductName, ProductName>(_productName);

		private ProductVersion ExchangeProductVersion => ConvertEnum<ProductVersion, ProductVersion>(_productVersion);

		public event PropertyChangedEventHandler PropertyChanged;

		private ExchangeAdapter()
		{
		}

		public void Initialize(ProductName productName, ProductVersion productVersion, Action<ExchangeErrorInfo> errorHandler, int tryingTimes = 10)
		{
			_productName = productName;
			_productVersion = productVersion;
			_errorHandler = errorHandler;
			if (!IsIcgConfiguredDisabled())
			{
				InitializeExchangeAsync(tryingTimes);
			}
		}

		private bool IsIcgConfiguredDisabled()
		{
			try
			{
				return File.Exists(Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData), "AspenTech", "Aspen Knowledge In-Context", ".aticg.disabled"));
			}
			catch
			{
				return false;
			}
		}

		private void OnPropertyChanged(string propertyName)
		{
			this.PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
		}

		public async void OpenBrowserAsync(string url)
		{
			await Task.CompletedTask.ContinueWith(delegate
			{
				if (SuccessfullyInitialized)
				{
					CheckELearningContent();
					if (!DoExchangeOpenUrl(url, out var errorInfo))
					{
						ProcessExchangeError(errorInfo);
					}
				}
			});
		}

		public void OpenBrowser(string url)
		{
			if (SuccessfullyInitialized)
			{
				CheckELearningContent();
				Cursor overrideCursor = Mouse.OverrideCursor;
				Mouse.OverrideCursor = Cursors.Wait;
				ExchangeErrorInfo errorInfo;
				bool num = DoExchangeOpenUrl(url, out errorInfo);
				Mouse.OverrideCursor = overrideCursor;
				if (!num)
				{
					ProcessExchangeError(errorInfo);
				}
			}
		}

		public async void SearchExchangeAsync(string searchText = "")
		{
			await Task.CompletedTask.ContinueWith(delegate
			{
				if (SuccessfullyInitialized)
				{
					CheckELearningContent();
					if (!DoExchangeSearch(searchText, out var errorInfo))
					{
						ProcessExchangeError(errorInfo);
					}
				}
			});
		}

		public void SearchExchange(string searchText = "")
		{
			if (SuccessfullyInitialized)
			{
				CheckELearningContent();
				Cursor overrideCursor = Mouse.OverrideCursor;
				Mouse.OverrideCursor = Cursors.Wait;
				ExchangeErrorInfo errorInfo;
				bool num = DoExchangeSearch(searchText, out errorInfo);
				Mouse.OverrideCursor = overrideCursor;
				if (!num)
				{
					ProcessExchangeError(errorInfo);
				}
			}
		}

		public void OpenElearning()
		{
			if (SuccessfullyInitialized)
			{
				CheckELearningContent();
				Cursor overrideCursor = Mouse.OverrideCursor;
				Mouse.OverrideCursor = Cursors.Wait;
				ExchangeErrorInfo errorInfo;
				bool num = DoElearningCatelog(out errorInfo);
				Mouse.OverrideCursor = overrideCursor;
				if (!num)
				{
					ProcessExchangeError(errorInfo);
				}
			}
		}

		public void Shutdown()
		{
			if (SuccessfullyInitialized)
			{
				Cursor overrideCursor = Mouse.OverrideCursor;
				Mouse.OverrideCursor = Cursors.Wait;
				ExchangeAgentImpl.OnShutdown();
				Mouse.OverrideCursor = overrideCursor;
			}
		}

		private void CheckELearningContent()
		{
			if (SuccessfullyInitialized && (_productName == ProductName.AspenPlus || _productName == ProductName.Hysys || _productName == ProductName.EDR || _productName == ProductName.ACCE || _productName == ProductName.Fidelis) && !_useELearningContent.HasValue)
			{
				MessageBoxResult messageBoxResult = MessageBox.Show(InContextGuidanceResources.ExchangeIntegration_ELearningChekingDialog_Question, InContextGuidanceResources.ExchangeIntegration_ELearningChekingDialog_Title, MessageBoxButton.YesNo, MessageBoxImage.Question);
				_useELearningContent = messageBoxResult == MessageBoxResult.Yes;
			}
		}

		private async void InitializeExchangeAsync(int tryingTimes)
		{
			await Task.Run(delegate
			{
				int num = Math.Max(1, tryingTimes);
				bool successfullyInitialized = false;
				ExchangeErrorInfo errorInfo = null;
				for (int i = 0; i < num; i++)
				{
					if (DoExchangeInitialize(out errorInfo))
					{
						successfullyInitialized = true;
						break;
					}
				}
				SuccessfullyInitialized = successfullyInitialized;
				if (!SuccessfullyInitialized)
				{
					ProcessExchangeError(errorInfo);
				}
			});
		}

		private bool DoExchangeInitialize(out ExchangeErrorInfo errorInfo)
		{
			//IL_0003: Unknown result type (might be due to invalid IL or missing references)
			try
			{
				ExchangeError error = default(ExchangeError);
				bool flag = ExchangeAgentImpl.Initialize(ref error, GetSupportedLanguage());
				errorInfo = (flag ? null : new ExchangeErrorInfo(error, isInitializationError: true, null, null));
				return flag;
			}
			catch (Exception ex)
			{
				errorInfo = new ExchangeErrorInfo(ex, isInitializationError: true, null, null);
				return false;
			}
		}

		private bool DoExchangeSearch(string searchText, out ExchangeErrorInfo errorInfo)
		{
			//IL_0001: Unknown result type (might be due to invalid IL or missing references)
			//IL_0007: Unknown result type (might be due to invalid IL or missing references)
			//IL_001f: Unknown result type (might be due to invalid IL or missing references)
			try
			{
				ExchangeError error = default(ExchangeError);
				bool flag = ExchangeAgentImpl.StartBrowserUrl(ExchangeProductName, ExchangeProductVersion, string.Empty, _useELearningContent == true, ref error, GetSupportedLanguage(), searchText);
				errorInfo = (flag ? null : new ExchangeErrorInfo(error, isInitializationError: false, searchText, null));
				return flag;
			}
			catch (Exception ex)
			{
				errorInfo = new ExchangeErrorInfo(ex, isInitializationError: false, searchText, null);
				return false;
			}
		}

		private bool DoElearningCatelog(out ExchangeErrorInfo errorInfo)
		{
			//IL_0006: Unknown result type (might be due to invalid IL or missing references)
			//IL_000c: Unknown result type (might be due to invalid IL or missing references)
			//IL_0020: Unknown result type (might be due to invalid IL or missing references)
			try
			{
				ExchangeError error = default(ExchangeError);
				bool flag = ExchangeAgentImpl.GetExchangeAgent().StartELearningCatalog(ExchangeProductName, ExchangeProductVersion, _useELearningContent == true, ref error, true, GetSupportedLanguage());
				errorInfo = (flag ? null : new ExchangeErrorInfo(error, isInitializationError: false, null, null));
				return flag;
			}
			catch (Exception ex)
			{
				errorInfo = new ExchangeErrorInfo(ex, isInitializationError: false, null, null);
				return false;
			}
		}

		private bool DoExchangeOpenUrl(string url, out ExchangeErrorInfo errorInfo)
		{
			//IL_0001: Unknown result type (might be due to invalid IL or missing references)
			//IL_0007: Unknown result type (might be due to invalid IL or missing references)
			//IL_001b: Unknown result type (might be due to invalid IL or missing references)
			try
			{
				ExchangeError error = default(ExchangeError);
				bool flag = ExchangeAgentImpl.StartBrowserUrl(ExchangeProductName, ExchangeProductVersion, url, _useELearningContent == true, ref error, GetSupportedLanguage(), "");
				errorInfo = (flag ? null : new ExchangeErrorInfo(error, isInitializationError: false, null, url));
				return flag;
			}
			catch (Exception ex)
			{
				errorInfo = new ExchangeErrorInfo(ex, isInitializationError: false, null, url);
				return false;
			}
		}

		private void ProcessExchangeError(ExchangeErrorInfo errorInfo)
		{
			if (errorInfo != null)
			{
				_errorHandler?.Invoke(errorInfo);
			}
		}

		private SupportedLanguage GetSupportedLanguage()
		{
			//IL_0049: Unknown result type (might be due to invalid IL or missing references)
			if (string.IsNullOrWhiteSpace(_selectedLanguage))
			{
				string text = Registry.GetValue("HKEY_CURRENT_USER\\Software\\AspenTech\\Engineering\\Language", null, string.Empty) as string;
				_selectedLanguage = (string.IsNullOrWhiteSpace(text) ? "eng" : text.Trim().ToLowerInvariant());
			}
			return GetSupportedLanguage(_selectedLanguage);
		}

		private static SupportedLanguage GetSupportedLanguage(string threeLettersLanguage)
		{
			return (SupportedLanguage)(threeLettersLanguage switch
			{
				"eng" => 0, 
				"chs" => 1, 
				"zho" => 1, 
				"jpn" => 2, 
				_ => 3, 
			});
		}

		private static TTo ConvertEnum<TFrom, TTo>(TFrom value) where TFrom : Enum where TTo : Enum
		{
			return (TTo)Enum.Parse(typeof(TTo), value.ToString());
		}
	}
	public class ExchangeErrorInfo
	{
		public bool IsInitializationError { get; }

		public string SearchText { get; }

		public string NavigationUrl { get; }

		public string ExceptionMessage { get; }

		public string StackTrace { get; }

		public string TranslatedDescription { get; }

		internal ExchangeErrorInfo(ExchangeError error, bool isInitializationError, string searchText, string navigationUrl)
			: this(isInitializationError, searchText, navigationUrl, (error != null) ? error.description : null, (error != null) ? error.stackTrace : null, (error != null) ? error.ToDescriptionString() : null)
		{
		}

		internal ExchangeErrorInfo(Exception ex, bool isInitializationError, string searchText, string navigationUrl)
			: this(isInitializationError, searchText, navigationUrl, ex?.ToString(), ex?.StackTrace, ex?.Message)
		{
		}

		internal ExchangeErrorInfo(bool isInitializationError, string searchText, string navigationUrl, string exceptionMessage, string stackTrace, string translatedDescription)
		{
			IsInitializationError = isInitializationError;
			SearchText = searchText;
			NavigationUrl = navigationUrl;
			ExceptionMessage = exceptionMessage;
			StackTrace = stackTrace;
			TranslatedDescription = translatedDescription;
		}
	}

		public string DisabledTooltip { get; set; }

		public object Convert(object value, Type targetType, object parameter, CultureInfo culture)
		{
			if (!(bool)value)
			{
				return DisabledTooltip;
			}
			return EnabledTooltip;
		}

	}

		private static ImageSource GetImageSource(string imageFileName)
		{
			if (!_imageSources.ContainsKey(imageFileName))
			{
				StringBuilder stringBuilder = new StringBuilder();
				stringBuilder.Append("pack://application:,,,/");
				stringBuilder.Append(typeof(ImageResources).Assembly.FullName);
				stringBuilder.Append(";component/Images/");
				stringBuilder.Append(imageFileName);
				_imageSources.Add(imageFileName, new BitmapImage(new Uri(stringBuilder.ToString(), UriKind.RelativeOrAbsolute)));
			}
			return _imageSources[imageFileName];
		}
	}
