	public class CHtmlHelpHelper : IHtmlHelpHelper
	{
		private readonly IUnityContainer _unityContainer;

		private readonly Dispatcher _dispatcher;

		private static IEventAggregator myEventAggregator;

		private string myHtmlHelpRoot;

		private string myHtmlHelpRootPrefix;

		private string myProductName;

		private string myProductExternalVersion;

		private string myProductInternalVersion;

		private string myProdDisplayName;

		private string myHelpSelfExtractApp;

		private bool mydisplayMoreThanOnce;

		private IHtmlHelpViewer htmlHelpHostWin;

		private CHtmlHelpLogging myLogger = new CHtmlHelpLogging();

		private bool myManageLibLifetime = true;

		public bool UseWinFormsBrowser { get; set; }

		public bool UseAspenTechStdIcon { get; set; }

		public IUnityContainer Container => _unityContainer;

		public int HtmlHelpWinHeight
		{
			get
			{
				return 0;
			}
			set
			{
			}
		}

		public int HtmlHelpWinWidth
		{
			get
			{
				return 0;
			}
			set
			{
			}
		}

		public CHtmlHelpHelper(IUnityContainer unityContainer)
		{
			if (unityContainer == null)
			{
				throw new ArgumentNullException("unityContainer");
			}
			_unityContainer = unityContainer;
			_dispatcher = Dispatcher.CurrentDispatcher;
			myHtmlHelpRoot = "about:blank";
			myProductName = "ProductName";
			myProductExternalVersion = "ExternalVersion";
			myProductInternalVersion = "100";
			myEventAggregator = UnityContainerExtensions.Resolve<IEventAggregator>(Container, Array.Empty<ResolverOverride>());
			if (myEventAggregator != null)
			{
				myEventAggregator.GetEvent<HtmlHelpShowHelpViewerEvent>().Subscribe(ExchangeHelpTopicHandle);
				myEventAggregator.GetEvent<HtmlHelpNavigateToEvent>().Subscribe(HtmlHelpNavigateToHandle);
			}
		}

		public void InitHtmlHelpSystem(string strProduct, string strExternalVersion, string strInternalVersion, string strHelpRoot)
		{
			InitHtmlHelpSystem(strProduct, strExternalVersion, strInternalVersion, strHelpRoot, string.Empty, string.Empty, bNoSelfExtract: false);
		}

		public void InitHtmlHelpSystem(string strProduct, string strExternalVersion, string strInternalVersion, string strHelpRoot, string strProdDspName)
		{
			InitHtmlHelpSystem(strProduct, strExternalVersion, strInternalVersion, strHelpRoot, strProdDspName, string.Empty, bNoSelfExtract: false);
		}

		public void InitHtmlHelpSystem(string strProduct, string strExternalVersion, string strInternalVersion, string strHelpRoot, string strProdDspName, string strHelpExtractApp, bool bNoSelfExtract, bool displayMoreThanOnce)
		{
			mydisplayMoreThanOnce = displayMoreThanOnce;
			InitHtmlHelpSystem(strProduct, strExternalVersion, strInternalVersion, strHelpRoot, strProdDspName, strHelpExtractApp, bNoSelfExtract);
		}

		public void InitHtmlHelpSystem(string strProduct, string strExternalVersion, string strInternalVersion, string strHelpRoot, string strProdDspName, string strHelpExtractApp, bool bNoSelfExtract)
		{
			myProductName = strProduct;
			myProductExternalVersion = strExternalVersion;
			myProductInternalVersion = strInternalVersion;
			myHtmlHelpRoot = strHelpRoot;
			myHtmlHelpRootPrefix = myHtmlHelpRoot + "#";
			myProdDisplayName = strProdDspName;
			myHelpSelfExtractApp = strHelpExtractApp;
		}

		private void CheckHelpExtractStatCode(string strHelpFolder)
		{
			HtmlHelpSystemUtil.CheckExtractStat(myProductName, myProdDisplayName, myProductExternalVersion, myProductInternalVersion, strHelpFolder, myHelpSelfExtractApp);
		}

		public void HtmlHelpDisplayF1Link(string HelpWebLink)
		{
			try
			{
				if (!string.IsNullOrWhiteSpace(HelpWebLink))
				{
					HtmlHelpDisplayLink(HelpWebLink, bLogExclude: false);
				}
			}
			catch (Exception)
			{
			}
		}

		public void HtmlHelpDispose()
		{
			if (htmlHelpHostWin != null)
			{
				htmlHelpHostWin.FreeHtmlHelpSystem();
				htmlHelpHostWin = null;
			}
			if (myManageLibLifetime)
			{
				HtmlHelpCefLibMgr.Shutdown();
			}
			if (myLogger != null)
			{
				myLogger.StopLogging();
			}
		}

		public void SetVisible(bool visible)
		{
			if (htmlHelpHostWin != null)
			{
				htmlHelpHostWin.SetVisible(visible);
			}
		}

		private void ExtHelpLinkForExchange(string strExtHelpLink)
		{
			if (string.IsNullOrEmpty(strExtHelpLink) || myEventAggregator == null || _dispatcher == null)
			{
				return;
			}
			_dispatcher.BeginInvoke(DispatcherPriority.Input, (Action)delegate
			{
				try
				{
					myEventAggregator.GetEvent<HtmlHelpViewerNavigatingEvent>().Publish(new HtmlHelpViewerNavigatingEventArgs
					{
						HelpFileRelativeURL = strExtHelpLink
					});
				}
				catch (Exception)
				{
				}
			});
		}

		private void HelpTopicVisitLogForExchange(string strHelpLink, bool bFromExt)
		{
			string.IsNullOrEmpty(strHelpLink);
		}

		private void ReleaseHtmlHelpHandle()
		{
			if (htmlHelpHostWin != null)
			{
				htmlHelpHostWin = null;
			}
			if (myLogger != null)
			{
				myLogger.PauseLogging();
			}
		}

		private void HelpDisplayLangHandle(string strHelpDisplayLang)
		{
			if (myLogger != null)
			{
				myLogger.HelpDisplayLangHandle(strHelpDisplayLang);
			}
		}

		private void ConfigHtmlHelpSystem(string strHelpWebLink, bool bLogExclude)
		{
			if (htmlHelpHostWin != null)
			{
				htmlHelpHostWin.FreeHtmlHelpSystem();
			}
			if (myLogger != null && myLogger.InitLogging())
			{
				htmlHelpHostWin = new HtmlHelpWindowHost(strHelpWebLink, myProductName, myProductExternalVersion, myProductInternalVersion, myHtmlHelpRoot, myProdDisplayName, UseWinFormsBrowser, UseAspenTechStdIcon);
				if (htmlHelpHostWin != null)
				{
					htmlHelpHostWin.ConfigHtmlHelpWin(ExtHelpLinkForExchange, HelpTopicVisitLogForExchange, ReleaseHtmlHelpHandle, HelpDisplayLangHandle);
					htmlHelpHostWin.ActiveHelpWindow();
				}
			}
		}

		private void HtmlHelpDisplayLink(string HelpWebLink, bool bLogExclude)
		{
			if (myHtmlHelpRoot == "about:blank" || myProductName == "ProductName" || myProductExternalVersion == "ExternalVersion")
			{
				StringBuilder stringBuilder = new StringBuilder();
				stringBuilder.AppendFormat("Please provide the proper landing page, product and version information to set up the help system.  \n\nCurrently, landing page={0}\nProduct={1}\nExternalVersion={2}", myHtmlHelpRoot, myProductName, myProductExternalVersion);
				MessageBox.Show(stringBuilder.ToString());
			}
			else if (htmlHelpHostWin == null || mydisplayMoreThanOnce)
			{
				ConfigHtmlHelpSystem(HelpWebLink, bLogExclude);
			}
			else
			{
				htmlHelpHostWin.HtmlHelpNavigateTo(HelpWebLink, bLogExclude);
			}
		}

		private void ExchangeHelpTopicHandle(HtmlHelpShowHelpViewerEventArgs exHelpTopicArg)
		{
			if (exHelpTopicArg == null)
			{
				return;
			}
			string helpFileRelativeURL = exHelpTopicArg.HelpFileRelativeURL;
			if (string.IsNullOrWhiteSpace(helpFileRelativeURL))
			{
				return;
			}
			helpFileRelativeURL = ((!helpFileRelativeURL.StartsWith("#")) ? (myHtmlHelpRootPrefix + helpFileRelativeURL) : (myHtmlHelpRoot + helpFileRelativeURL));
			try
			{
				HtmlHelpDisplayLink(helpFileRelativeURL, bLogExclude: true);
			}
			catch (Exception)
			{
			}
		}

		private void HtmlHelpNavigateToHandle(HtmlHelpNavigateToEventArgs helpNavArgs)
		{
			if (helpNavArgs != null)
			{
				HtmlHelpDisplayF1Link(helpNavArgs.HelpTopicUrl);
			}
		}

		public void IgnoreExchangeEvents()
		{
			myManageLibLifetime = false;
			if (myEventAggregator != null)
			{
				myEventAggregator.GetEvent<HtmlHelpShowHelpViewerEvent>().Unsubscribe(ExchangeHelpTopicHandle);
			}
		}

		public bool IsHelpExtractDone(ref bool bIsDone)
		{
			return HtmlHelpSystemUtil.IsHelpExtractDone(myProductName, myProductInternalVersion, ref bIsDone);
		}

		public void CloseOpenedHelpWindow()
		{
			if (htmlHelpHostWin != null)
			{
				htmlHelpHostWin.CloseOpenedHelpWindow();
			}
		}
	}
