	protected override void CreateSink()
	{
		try
		{
			eventMulticaster = new AxPfsplotEventMulticaster(this);
			cookie = new ConnectionPointCookie(ocx, eventMulticaster, typeof(_DPfsplotEvents));
		}
		catch (Exception)
		{
		}
	}

	protected override void DetachSink()
	{
		try
		{
			cookie.Disconnect();
		}
		catch (Exception)
		{
		}
	}

	protected override void AttachInterfaces()
	{
		try
		{
			ocx = (_DPfsplot)GetOcx();
		}
		catch (Exception)
		{
		}
	}
