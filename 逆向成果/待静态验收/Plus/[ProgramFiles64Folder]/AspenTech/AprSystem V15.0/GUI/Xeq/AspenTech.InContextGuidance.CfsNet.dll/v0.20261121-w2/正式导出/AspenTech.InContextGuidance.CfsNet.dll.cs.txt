		private static async Task<CfsResult<T>> DoHttpRequest<T>(string url) where T : class
		{
			for (int i = 0; i < CfsClientSettings.TryingTimesForHttpRequest - 1; i++)
			{
				CfsResult<T> cfsResult = await DoSingleHttpRequest<T>(url);
				if (cfsResult.Succeeded || cfsResult.FailedToConnectServer)
				{
					return cfsResult;
				}
			}
			return await DoSingleHttpRequest<T>(url);
		}

		private static async Task<CfsResult<T>> DoSingleHttpRequest<T>(string url) where T : class
		{
			string text;
			try
			{
				text = await _httpClientSingleton.GetStringAsync(url);
			}
			catch (Exception exception)
			{
				return new CfsResult<T>(exception);
			}
			if (!string.IsNullOrWhiteSpace(text))
			{
				try
				{
					return new CfsResult<T>(new JavaScriptSerializer().Deserialize<T>(text));
				}
				catch (Exception exception2)
				{
					return new CfsResult<T>(exception2);
				}
			}
			return new CfsResult<T>((T)null);
		}

		private static string ResolveLanguageString(CfsProductLanguage? language = null)
		{
			if (language.HasValue)
			{
				return language.Value.GetCfsText();
			}
			return Extensions.GetCfsLanguageByThreadUICulture().GetCfsText();
		}

		private static string ResolveCfsUrl(string cfsUrl = null)
		{
			if (!string.IsNullOrWhiteSpace(cfsUrl))
			{
				return CfsClientSettings.NormalizeUrl(cfsUrl);
			}
			return CfsClientSettings.GetConfiguredServiceUrl();
		}

		private static CfsResult<T> Encapsulate<F, T>(CfsResult<F> from, Func<F, T> accessor)
		{
			if (from.Exception != null)
			{
				return new CfsResult<T>(from.Exception);
			}
			return new CfsResult<T>(accessor(from.Data));
		}
	}
	public class CfsResult<T>
	{
		private const string UnableToConnect = "Unable to connect to the remote server";

		public bool Succeeded { get; }

		public bool FailedToConnectServer => IsUnableToConnectToRemoteServer(Exception);

		public T Data { get; }

		public Exception Exception { get; }

		public string ExceptionMessage
		{
			get
			{
				if (!FailedToConnectServer)
				{
					return Exception?.Message;
				}
				return "Unable to connect to the remote server";
			}
		}

		public CfsResult(T succeededResult)
		{
			Data = succeededResult;
			Succeeded = true;
		}

		public CfsResult(Exception exception)
		{
			Exception = exception ?? throw new ArgumentNullException("exception");
		}

		private static bool IsUnableToConnectToRemoteServer(Exception exception)
		{
			if (exception != null)
			{
				if (exception is WebException { Status: WebExceptionStatus.ConnectFailure })
				{
					return true;
				}
				return IsUnableToConnectToRemoteServer(exception.InnerException);
			}
			return false;
		}
	}

	public class IndexedContent
	{
		private readonly List<IndexedContentItem> _allItems = new List<IndexedContentItem>();

		private readonly List<IndexedContentItem> _contextItems = new List<IndexedContentItem>();

		private readonly List<IndexedContentItem> _secondaryContextItems = new List<IndexedContentItem>();

		public string ProductName { get; }

		public string ProductVersion { get; }

		public string ProductLanguage { get; }

		public string CfsAddress { get; }

		public string Id { get; }

		public string Title { get; }

		public string SearchString { get; }

		public string FeedbackPrefix { get; }

		public IndexedContentItem OnlineHelpItem { get; private set; }

		public IndexedContentItem PrimaryContextItem { get; private set; }

		public IEnumerable<IndexedContentItem> Items => _allItems;

		public IEnumerable<IndexedContentItem> ContextItems => _contextItems;

		public IEnumerable<IndexedContentItem> SecondaryContextItems => _secondaryContextItems;

		internal IndexedContent(string productName, string productVersion, string productLanguage, string cfsAddress, CfsContextPackage package)
		{
			ProductName = productName;
			ProductVersion = productVersion;
			ProductLanguage = productLanguage;
			CfsAddress = cfsAddress;
			if (package == null)
			{
				return;
			}
			FeedbackPrefix = package.FeedbackPrefix;
			if (package.Context != null)
			{
				Id = package.Context.Id;
				Title = package.Context.Title;
				Description = package.Context.Description;
				SearchString = package.Context.SearchString;
				IEnumerable<CfsContextDataItem> items = package.Context.Items;
				if (items != null)
				{
					AllocateItems(items);
				}
			}
		}

		private void AllocateItems(IEnumerable<CfsContextDataItem> cfsItems)
		{
			_allItems.AddRange(cfsItems.Select((CfsContextDataItem x) => new IndexedContentItem(x, this)));
			OnlineHelpItem = _allItems.FirstOrDefault((IndexedContentItem x) => !x.IsPrimary);
			PrimaryContextItem = _allItems.FirstOrDefault((IndexedContentItem x) => x.IsPrimary) ?? _allItems.FirstOrDefault((IndexedContentItem x) => x != OnlineHelpItem);
			foreach (IndexedContentItem allItem in _allItems)
			{
				if (allItem != OnlineHelpItem)
				{
					_contextItems.Add(allItem);
					if (allItem != PrimaryContextItem)
					{
						_secondaryContextItems.Add(allItem);
					}
				}
			}
			LoadItemsIconsAsync();
		}

		private async void LoadItemsIconsAsync()
		{
			List<string> list = new List<string>();
			List<IndexedContentItem> notLoadedItems = new List<IndexedContentItem>();
			List<Task> list2 = new List<Task>();
			foreach (IndexedContentItem allItem in _allItems)
			{
				if (!list.Contains(allItem.ContentType))
				{
					Task item = allItem.LoadIconsAsync();
					list2.Add(item);
					list.Add(allItem.ContentType);
				}
				else
				{
					notLoadedItems.Add(allItem);
				}
			}
			await Task.WhenAll(list2);
			foreach (IndexedContentItem item2 in notLoadedItems)
			{
				await item2.LoadIconsAsync();
			}
		}
	}

	internal static class ImageResources
	{
		private class ContentImageSources
		{
			public ImageSource Regular { get; set; }

			public ImageSource Large { get; set; }
		}

		private static readonly Dictionary<string, ContentImageSources> _contentImages = new Dictionary<string, ContentImageSources>();

		public static async Task<(ImageSource Regular, ImageSource Large)> GetContentImageSources(string productName, string productVersion, string language, string contentType, string cfsUrl = null)
		{
			if (!string.IsNullOrWhiteSpace(contentType))
			{
				string key = productName + "." + productVersion + "." + language + "." + contentType;
				if (_contentImages.ContainsKey(key))
				{
					ContentImageSources contentImageSources = _contentImages[key];
					return (Regular: contentImageSources.Regular, Large: contentImageSources.Large);
				}
				CfsResult<CfsIconData> cfsResult = await CfsClient.GetCfsIconDataAsync(productName, productVersion, language, contentType, cfsUrl);
				if (cfsResult.Succeeded && cfsResult.Data != null)
				{
					ImageSource imageSource = CreateFromBase64String(cfsResult.Data.MimeType, cfsResult.Data.Image);
					ImageSource imageSource2 = CreateFromBase64String(cfsResult.Data.LargeMimeType, cfsResult.Data.LargeImage);
					_contentImages[key] = new ContentImageSources
					{
						Regular = imageSource,
						Large = imageSource2
					};
					return (Regular: imageSource, Large: imageSource2);
				}
			}
			return default((ImageSource, ImageSource));
		}

		private static ImageSource CreateFromBase64String(string mimeType, string base64EncodedImage)
		{
			if (!string.IsNullOrWhiteSpace(base64EncodedImage))
			{
				byte[] array;
				try
				{
					array = Convert.FromBase64String(ExtractBase64Code(mimeType, base64EncodedImage));
				}
				catch (FormatException ex)
				{
					Console.WriteLine(ex.Message);
					array = null;
				}
				if (array != null)
				{
					try
					{
						BitmapImage bitmapImage = new BitmapImage();
						bitmapImage.BeginInit();
						bitmapImage.StreamSource = new MemoryStream(array);
						bitmapImage.EndInit();
						bitmapImage.Freeze();
						return bitmapImage;
					}
					catch (Exception ex2)
					{
						Console.WriteLine(ex2.Message);
					}
				}
			}
			return null;
		}

		private static string ExtractBase64Code(string mimeType, string base64EncodedImage)
		{
			int num = base64EncodedImage.IndexOf(";base64,");
			if (num > 0)
			{
				return base64EncodedImage.Substring(num + ";base64,".Length).Trim();
			}
			return base64EncodedImage;
		}
