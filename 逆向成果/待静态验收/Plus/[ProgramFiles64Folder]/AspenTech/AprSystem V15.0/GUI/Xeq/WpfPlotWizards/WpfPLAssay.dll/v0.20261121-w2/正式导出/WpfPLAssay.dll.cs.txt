		long IRibbonPlotWizardMgr.SupportedPlotTypes(ref List<string> lsTagIds)
		{
			lsTagIds.Clear();
			for (int i = 0; i < _lspwrTypes.Count; i++)
			{
				PWRPlotType val = _lspwrTypes[i];
				IRibbonPlotWizardType iplotType = null;
				_tempWizard.Clear();
				_pwrParser.FillDefaultPlotVal(val, ref _hnode, ref _tempWizard);
				if (DoesPlotTypeHaveData(val.TagId, ref iplotType, ref _tempWizard))
				{
					lsTagIds.Add(val.TagId);
				}
			}
			return _lspwrTypes.Count;
		}

		long IRibbonPlotWizardMgr.GetPlotType(string sTagId, ref IRibbonPlotWizardType plwType)
		{
			if (_dicTagIdToPlotType.ContainsKey(sTagId))
			{
				plwType = _dicTagIdToPlotType[sTagId];
				if (plwType == null)
				{
					return 0L;
				}
				PWRPlotType val = null;
				_pwrParser.FindPlotType(sTagId, ref _lspwrTypes, ref val);
				_pwrParser.FillDefaultPlotVal(val, ref _hnode, ref _wizardPlot);
			}
			return 0L;
		}

		private void Initialize()
		{
			_asm = Assembly.GetExecutingAssembly();
			StreamReader streamReader = new StreamReader(Assembly.GetExecutingAssembly().GetManifestResourceStream("WpfPLAssay.Text.Assaypwr.txt"));
			_lspwrTypes.Clear();
			_pwrParser.ParsePwrFile(streamReader, ref _lspwrTypes);
			_dicTagIdToPlotType = new Dictionary<string, IRibbonPlotWizardType>();
			string[] array = supportedTypes;
			foreach (string text in array)
			{
				if (text == supportedTypes[0])
				{
					IRibbonPlotWizardType value = (IRibbonPlotWizardType)(object)new Distillation();
					_dicTagIdToPlotType.Add(text, value);
				}
				else if (text == supportedTypes[1])
				{
					IRibbonPlotWizardType value2 = (IRibbonPlotWizardType)(object)new Gravity();
					_dicTagIdToPlotType.Add(text, value2);
				}
				else if (text == supportedTypes[2])
				{
					IRibbonPlotWizardType value3 = (IRibbonPlotWizardType)(object)new MolecularWt();
					_dicTagIdToPlotType.Add(text, value3);
				}
				else if (text == supportedTypes[3])
				{
					IRibbonPlotWizardType value4 = (IRibbonPlotWizardType)(object)new Petro();
					_dicTagIdToPlotType.Add(text, value4);
				}
				else if (text == supportedTypes[4])
				{
					IRibbonPlotWizardType value5 = (IRibbonPlotWizardType)(object)new Viscosity();
					_dicTagIdToPlotType.Add(text, value5);
				}
			}
		}

		private bool DoesPlotTypeHaveData(string sTagId, ref IRibbonPlotWizardType iplotType, ref IHWizardPlot iWizardPlot)
		{
			int num = 0;
			if (_dicTagIdToPlotType.ContainsKey(sTagId))
			{
				iplotType = _dicTagIdToPlotType[sTagId];
				if (iplotType == null)
				{
					return false;
				}
				object obj = iWizardPlot;
				iplotType.Context((object)_hnode, ref obj, ref num);
				if (num == 0)
				{
					return true;
				}
			}
			return false;
		}
	}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = _context.FindNode("Input/DISTTYPE");
			string text = ((dynamic)iHNode.get_Value((object)null)).ToString();
			if (string.IsNullOrEmpty(text))
			{
				ierror = 1;
				return;
			}
			SetVariables();
			radioButtonReport.IsEnabled = false;
			for (int i = 0; i < _outputList.GetLength(0); i++)
			{
				iHNode = _context.FindNode(_outputList[i]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					radioButtonReport.IsEnabled = true;
					break;
				}
			}
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				radioButtonInput.IsChecked = true;
				List<string> list = new List<string>();
				for (int j = 0; j < _inputList.GetLength(0); j++)
				{
					if (text.ToLower() == _inputList[j].ToString().ToLower())
					{
						string item = _curveNames[j].Content.ToString();
						list.Add(item);
						break;
					}
				}
				CurveList.SetCurves(list.ToArray());
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, "Input/TEMP");
		}

		public void UpdateWizardData()
		{
			_wizardPlot.RemoveAllDependentVariable();
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			int count = list.Count;
			int num = 0;
			for (int i = 0; i < count; i++)
			{
				for (int j = 0; j < _curveNames.Count; j++)
				{
					if (list[i].ToLower() == _curveNames[j].Content.ToString().ToLower())
					{
						if (radioButtonInput.IsChecked == true)
						{
							_wizardPlot.SetIndependentVariable(_context, "Input/TEMP!Label2");
							_wizardPlot.AddDependentVariable(_context, "Input/TEMP");
						}
						else
						{
							_wizardPlot.SetIndependentVariable(_context, "Output/DIST_RATIO");
							_wizardPlot.AddDependentVariable(_context, _outputList[j]);
						}
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(num++, _curveNames[j].Content.ToString());
						if (j == 3 || j > 4)
						{
							_wizardPlot.PlotVal.XAxisTitle = WpfPLAssayResources.Distillation_XAxisTitle_W;
						}
						else
						{
							_wizardPlot.PlotVal.XAxisTitle = WpfPLAssayResources.Distillation_XAxisTitle_VL;
						}
					}
				}
			}
			_help.SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentTempUnit);
			_currentConfigValue = SaveConfigValue();
		}

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			int num = CurveList.GetSelectedCurves().Count();
			if (num == CurveList.Curves.Count)
			{

			}
			else
			{
				string[] selectedCurves = CurveList.GetSelectedCurves();
				CurveList.IsCheckAll = false;
				CurveList.SetSelectedCurves(selectedCurves);
			}
			e.CanExecute = num != 0;
		}

		private void SetInputOption()
		{
			if (radioButtonInput.IsChecked != true)
			{
				return;
			}
			_bReportCheckAll = CurveList.IsCheckAll;
			selReport = new List<string>(CurveList.GetSelectedCurves());
			IHNode iHNode = null;
			List<string> list = new List<string>();
			for (int i = 0; i < _inputList.GetLength(0); i++)
			{
				iHNode = _context.FindNode("Input/DISTTYPE");
				if (((dynamic)iHNode.get_Value((object)null)).ToString().ToLower() == _inputList[i].ToString().ToLower())
				{
					string item = _curveNames[i].Content.ToString();
					list.Add(item);
				}
			}
			CurveList.IsCheckAll = _bInputCheckAll;
			CurveList.SetCurves(list.ToArray());
			if (selInput != null)
			{
				CurveList.SetSelectedCurves(selInput.ToArray());
			}
		}

		private void SetReportOption()
		{
			if (radioButtonReport.IsChecked != true)
			{
				return;
			}
			_bInputCheckAll = CurveList.IsCheckAll;
			selInput = new List<string>(CurveList.GetSelectedCurves());
			List<string> list = new List<string>();
			IHNode iHNode = null;
			for (int i = 0; i < _outputList.GetLength(0); i++)
			{
				iHNode = _context.FindNode(_outputList[i].ToString());
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					string item = _curveNames[i].Content.ToString();
					list.Add(item);
				}
			}
			CurveList.IsCheckAll = _bReportCheckAll;
			CurveList.SetCurves(list.ToArray());
			if (selReport != null)
			{
				CurveList.SetSelectedCurves(selReport.ToArray());
			}
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("Option=");
			stringBuilder.Append(radioButtonInput.IsChecked);
			stringBuilder.Append(";");
			stringBuilder.Append("ReportEnable=");
			stringBuilder.Append(radioButtonReport.IsEnabled);
			stringBuilder.Append(";");
			stringBuilder.Append("CurveSelection=");
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			int count = list.Count;
			for (int i = 0; i < count; i++)
			{
				if (i > 0)
				{
					stringBuilder.Append(",");
				}
				stringBuilder.Append(list[i]);
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CheckAll=");
			stringBuilder.Append(CurveList.IsCheckAll);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string value = array[num].Substring(array[num].IndexOf('=') + 1);
			radioButtonInput.IsChecked = Convert.ToBoolean(value);
			radioButtonReport.IsChecked = !radioButtonInput.IsChecked;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			radioButtonReport.IsEnabled = Convert.ToBoolean(value);
			if (radioButtonInput.IsChecked == true)
			{
				SetInputOption();
			}
			else
			{
				SetReportOption();
			}
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = value.Split(',');
			List<string> list = new List<string>();
			string[] array3 = array2;
			foreach (string item in array3)
			{
				list.Add(item);
			}
			CurveList.SetSelectedCurves(list.ToArray());
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			CurveList.IsCheckAll = Convert.ToBoolean(value);
		}

		public void UpdateWizardData()
		{
			string empty = string.Empty;
			string empty2 = string.Empty;
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			int count = list.Count;
			if (count > 0)
			{
				IHNode iHNode;
				string text;

				{
					iHNode = _context.FindNode("Input/VISCTEMP");

					empty = WpfPLAssayResources.Viscosity_AbsoluteVisc_Title;

				}
				else
				{
					iHNode = _context.FindNode("Input/KVISCTEMP");
					text = "Input/KVISCOSITY";
					empty = WpfPLAssayResources.Viscosity_KinematicVisc_Title;
					empty2 = WpfPLAssayResources.Viscosity_KinematicVisc_YAxisTitle + "(%Unit%)";
				}
				int count2 = iHNode.Elements.Count;
				_wizardPlot.RemoveAllIndependentVariable();
				_wizardPlot.RemoveAllDependentVariable();
				for (int i = 0; i < count; i++)
				{
					string text2 = list[i];
					for (int j = 0; j < count2; j++)
					{
						string text3 = ((dynamic)iHNode.Elements[j, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value((object)null)).ToString().ToLower();
						if (text2.ToLower() == text3)
						{
							_wizardPlot.SetMultIndependentVariables(_context, text + "/#" + j + "!Label2");
							_wizardPlot.AddDependentVariable(_context, text + "/#" + j);
							((IHPlotVal)_wizardPlot.PlotVal).set_Label(i, text3);
						}
					}
				}
				_wizardPlot.PlotVal.Title = empty;
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, empty2);
				_wizardPlot.PlotVal.XAxisTitle = WpfPLAssayResources.Percent_distilled;

			_currentConfigValue = SaveConfigValue();
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("CurveSelection=");
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			int count = list.Count;
			for (int i = 0; i < count; i++)
			{
				if (i > 0)
				{
					stringBuilder.Append(",");
				}
				stringBuilder.Append(list[i]);
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CheckAll=");
			stringBuilder.Append(CurveList.IsCheckAll);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string text = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = text.Split(',');
			List<string> list = new List<string>();
			string[] array3 = array2;
			foreach (string item in array3)
			{
				list.Add(item);
			}
			CurveList.SetSelectedCurves(list.ToArray());
			num++;
			text = array[num].Substring(array[num].IndexOf('=') + 1);
			CurveList.IsCheckAll = Convert.ToBoolean(text);
		}
