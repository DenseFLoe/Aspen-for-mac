	internal class DashboardItemFactory : IDashboardItemFactory
	{
		private class RegisteredItem
		{
			public Type Type { get; private set; }

			public int Sequence { get; private set; }

			public RegisteredItem(Type type, int sequence)
			{
				Type = type;
				Sequence = sequence;
			}
		}

		private readonly IUnityContainer _unityContainer;

		private readonly List<RegisteredItem> _registeredTypes = new List<RegisteredItem>();

		public DashboardItemFactory(IUnityContainer unityContainer)
		{
			if (unityContainer == null)
			{
				throw new ArgumentNullException("unityContainer");
			}
			_unityContainer = unityContainer;
		}

		public void Register<T>(int sequence = 0) where T : UIElement, IDashboardItem
		{
			Type dit = typeof(T);
			if (!_registeredTypes.Any((RegisteredItem x) => x.Type == dit))
			{
				_registeredTypes.Add(new RegisteredItem(dit, sequence));
			}
		}

		public IEnumerable<IDashboardItem> CreateItems()
		{
			if (_registeredTypes.Count <= 0)
			{
				yield break;
			}
			IOrderedEnumerable<RegisteredItem> orderedEnumerable = _registeredTypes.OrderBy((RegisteredItem x) => x.Sequence);
			foreach (RegisteredItem item in orderedEnumerable)
			{
				IDashboardItem dashboardItem = null;
				try
				{
					dashboardItem = UnityContainerExtensions.Resolve(_unityContainer, item.Type, Array.Empty<ResolverOverride>()) as IDashboardItem;
				}
				catch (Exception ex)
				{
					UnityContainerExtensions.Resolve<ILoggerFacade>(_unityContainer, Array.Empty<ResolverOverride>())?.Log(ex.ToString(), Category.Exception, Priority.High);
				}
				if (dashboardItem != null)
				{
					yield return dashboardItem;
				}
			}
		}
	}

		void IWorkspacePersist.Restore(object obj)
		{
			string text = obj as string;
			if (string.IsNullOrEmpty(text))
			{
				return;
			}
			try
			{
				XElement xElement = XElement.Parse(text);
				IsExpanded = XmlConvert.ToBoolean(xElement.Attribute("IsExpanded").Value);
				XAttribute xAttribute = xElement.Attribute("IsTurnedOff");
				if (xAttribute == null || string.Equals(xAttribute.Value, "False", StringComparison.CurrentCultureIgnoreCase))
				{
					IsTurnedOff = false;
				}
				else
				{
					IsTurnedOff = true;
				}
				int? num = null;
				XAttribute xAttribute2 = xElement.Attribute("Version");
				if (xAttribute2 != null)
				{
					num = XmlConvert.ToInt16(xAttribute2.Value);
				}
				foreach (XElement item in xElement.Elements("Item"))
				{
					if (string.IsNullOrEmpty(item.Value))
					{
						continue;
					}
					string type = item.Attribute("Type").Value;
					IDashboardItem dashboardItem = Items.FirstOrDefault((IDashboardItem x) => string.Equals(x.GetType().FullName, type, StringComparison.InvariantCulture));
					if (dashboardItem == null)
					{
						continue;
					}
					if (num.HasValue)
					{
						string text2 = DashboardItemStateManager.Convert(dashboardItem, item.Value, num.Value);
						if (!string.IsNullOrEmpty(text2))
						{
							dashboardItem.RestoreState(text2);
						}
					}
					else
					{
						dashboardItem.RestoreState(item.Value);
					}
				}
			}
			catch (Exception ex)
			{
				Log(ex);
			}
		}

		object IWorkspacePersist.Save()
		{
			IDashboardPanelAgent agent = Agent;
			if (agent == null)
			{
				return null;
			}
			if (!agent.ShouldPersistState)
			{
				return null;
			}
			XElement xElement = new XElement("DashboardPanel");
			xElement.SetAttributeValue("IsExpanded", IsExpanded);
			xElement.SetAttributeValue("IsTurnedOff", IsTurnedOff);
			int? currentAppVersion = GetCurrentAppVersion();
			if (currentAppVersion.HasValue)
			{
				xElement.SetAttributeValue("Version", currentAppVersion.Value);
			}
			foreach (IDashboardItem item in Items)
			{
				XElement xElement2 = new XElement("Item");
				xElement2.SetAttributeValue("Type", item.GetType().FullName);
				string value = item.SaveState();
				if (!string.IsNullOrEmpty(value))
				{
					xElement2.SetValue(value);
				}
				xElement.Add(xElement2);
			}
			return xElement.ToString();
		}

		private int? GetCurrentAppVersion()
		{
			return Agent?.CurrentApplicationVersion;
		}

		protected override void OnPreviewMouseLeftButtonDown(MouseButtonEventArgs e)
		{
			_downPos = e.GetPosition(this);
			base.OnPreviewMouseLeftButtonDown(e);
		}

		protected override void OnPreviewMouseLeftButtonUp(MouseButtonEventArgs e)
		{
			if (_downPos.HasValue && (e.GetPosition(this) - _downPos.Value).Length <= 2.0 && !SkipInvokingClick(e.OriginalSource))
			{
				OnClicked(e.OriginalSource);
			}
			_downPos = null;
			base.OnPreviewMouseLeftButtonUp(e);
		}

		private bool SkipInvokingClick(object originalSource)
		{
			if (originalSource == null)
			{
				return true;
			}
			if (originalSource is ButtonBase)
			{
				return true;
			}
			DependencyObject dependencyObject = originalSource as DependencyObject;
			while (dependencyObject != null && dependencyObject != this)
			{
				if (dependencyObject is OnOffControlBase)
				{
					return true;
				}
				if (dependencyObject is ButtonBase)
				{
					return true;
				}
				dependencyObject = VisualTreeHelper.GetParent(dependencyObject);
			}
			return false;
		}

		protected virtual void OnClicked(object originalSource)
		{
		}

		public static void ShowOrActivateScreen(IUnityContainer container, string screenFactoryKey)
		{
			if (container == null)
			{
				throw new ArgumentNullException("container");
			}
			if (string.IsNullOrEmpty(screenFactoryKey))
			{
				throw new ArgumentException("Parameter screenFactoryKey cannot be null or empty.");
			}
			IShellApplication shellApplication = UnityContainerExtensions.Resolve<IShellApplication>(container, Array.Empty<ResolverOverride>());
			if (shellApplication != null && shellApplication.ActiveWindow != null)
			{
				ScreenContentControl screenContentControl = shellApplication.ActiveWindow.FindScreen((ScreenContentControl x) => string.Equals(x.ScreenFactoryKey, screenFactoryKey, StringComparison.InvariantCulture));
				if (screenContentControl != null)
				{
					screenContentControl.Activate();
					return;
				}
				shellApplication.ActiveWindow.ShowScreen(new ShowScreenEventArgs
				{
					ScreenFactoryKey = screenFactoryKey,
					CreateNewTab = true
				});
			}
		}

	public class DoubleValueSumConverter : IMultiValueConverter
	{
		public object Convert(object[] values, Type targetType, object parameter, CultureInfo culture)
		{
			if (values != null)
			{
				double num = 0.0;
				foreach (object obj in values)
				{
					if (obj is double)
					{
						num += (double)obj;
					}
				}
				return num;
			}
			return 0.0;
		}

		public object[] ConvertBack(object value, Type[] targetTypes, object parameter, CultureInfo culture)
		{
			throw new NotImplementedException();
		}
	}

		private static double ConvertUOM(EDRActivateSimulatorType siumlatorType, EDRActivateQuantityType rType, double srcValue, string srcUnitID, string trgtUnitID, EDRActivateGlobalUnitSetting globalUnitSetting)
		{
			if (double.IsNaN(srcValue))
			{
				return double.NaN;
			}
			double num = double.NaN;
			Dictionary<string, EDRActivateUnitConversionFactor> unitFactorList = EDRActivateUnitData.GetUnitFactorList(siumlatorType, rType);
			if (!unitFactorList.ContainsKey(srcUnitID))
			{
				throw new KeyNotFoundException($"\"{srcUnitID}\" is not a valid unit of {rType.ToString()} for {siumlatorType.ToString()}.");
			}
			if (!unitFactorList.ContainsKey(trgtUnitID))
			{
				throw new KeyNotFoundException($"\"{trgtUnitID}\" is not a valid unit of {rType.ToString()} for {siumlatorType.ToString()}.");
			}
			_ = unitFactorList[srcUnitID];
			EDRActivateUnitConversionFactor eDRActivateUnitConversionFactor = unitFactorList[trgtUnitID];
			string calculationUnit = EDRActivateUnitData.GetCalculationUnit(siumlatorType, rType);
			_ = unitFactorList[calculationUnit];
			num = ConvertToBaseUnit(siumlatorType, rType, srcValue, srcUnitID, globalUnitSetting);
			switch (siumlatorType)
			{
			case EDRActivateSimulatorType.HYSYS:
				if (trgtUnitID != calculationUnit)
				{
					num = num * eDRActivateUnitConversionFactor.Factor + eDRActivateUnitConversionFactor.Offset;
				}
				break;
			case EDRActivateSimulatorType.AspenPlus:
				if (eDRActivateUnitConversionFactor.Offset != _plusUserOffset)
				{
					num = (num - eDRActivateUnitConversionFactor.Offset) / eDRActivateUnitConversionFactor.Factor;
					break;
				}
				switch (rType)
				{
				case EDRActivateQuantityType.Pressure:
					num -= globalUnitSetting.AmbientPressure;
					num /= eDRActivateUnitConversionFactor.Factor;
					break;
				case EDRActivateQuantityType.MoleFlow:
				case EDRActivateQuantityType.MassFlow:
				case EDRActivateQuantityType.VolumFlow:
					if (trgtUnitID.EndsWith(_operYear))
					{
						num /= eDRActivateUnitConversionFactor.Factor * 365.25 * 24.0 * 3600.0 / globalUnitSetting.OperationalYear;
					}
					break;
				}
				break;
			}
			return num;
		}

		private static double ConvertToBaseUnit(EDRActivateSimulatorType siumlatorType, EDRActivateQuantityType rType, double srcValue, string srcUnitID, EDRActivateGlobalUnitSetting globalUnitSetting)
		{
			double result = double.NaN;
			EDRActivateUnitConversionFactor eDRActivateUnitConversionFactor = EDRActivateUnitData.GetUnitFactorList(siumlatorType, rType)[srcUnitID];
			switch (siumlatorType)
			{
			case EDRActivateSimulatorType.HYSYS:
				result = (srcValue - eDRActivateUnitConversionFactor.Offset) / eDRActivateUnitConversionFactor.Factor;
				break;
			case EDRActivateSimulatorType.AspenPlus:
				if (eDRActivateUnitConversionFactor.Offset != _plusUserOffset)
				{
					result = srcValue * eDRActivateUnitConversionFactor.Factor + eDRActivateUnitConversionFactor.Offset;
					break;
				}
				switch (rType)
				{
				case EDRActivateQuantityType.Pressure:
					result = srcValue * eDRActivateUnitConversionFactor.Factor;
					result += globalUnitSetting.AmbientPressure;
					break;
				case EDRActivateQuantityType.MoleFlow:
				case EDRActivateQuantityType.MassFlow:
				case EDRActivateQuantityType.VolumFlow:
					if (srcUnitID.EndsWith(_operYear))
					{
						result = srcValue * eDRActivateUnitConversionFactor.Factor * 365.25 * 24.0 * 3600.0 / globalUnitSetting.OperationalYear;
					}
					break;
				}
				break;
			}
			return result;
		}

		{
			if (value != DependencyProperty.UnsetValue)
			{
				if (string.IsNullOrEmpty(SubProperty))
				{
					return GetReturnValue(IsEqual(value, parameter));
				}
				return GetReturnValue(IsEqual(value, SubProperty, parameter));
			}
			return FalseResult;
		}

		public object ConvertBack(object value, Type targetType, object parameter, CultureInfo culture)
		{
			throw new NotImplementedException();
		}

		private bool IsEqual(object value, object parameter)
		{
			if (value == null)
			{
				return parameter == null;
			}
			if (parameter == null)
			{
				return false;
			}
			if (value.GetType() == parameter.GetType())
			{
				return object.Equals(value, parameter);
			}
			return string.Equals(value.ToString(), parameter.ToString(), StringComparison.InvariantCultureIgnoreCase);
		}

		private bool IsEqual(object value, string subProperty, object parameter)
		{
			if (string.Equals(subProperty, "HasValue", StringComparison.InvariantCultureIgnoreCase))
			{
				if (value == null)
				{
					return IsEqual("False", parameter);
				}
				return IsEqual("True", parameter);
			}
			if (value != null)
			{
				PropertyInfo propertyInfo = value.GetType().GetProperties().FirstOrDefault((PropertyInfo x) => string.Equals(x.Name, subProperty, StringComparison.InvariantCultureIgnoreCase));
				if (propertyInfo != null)
				{
					object value2 = propertyInfo.GetValue(value, null);
					return IsEqual(value2, parameter);
				}
			}
			return false;
		}

		private object GetReturnValue(bool condition)
		{
			if (!condition)
			{
				return FalseResult;
			}
			return TrueResult;
		}

	{
		private const double OneBillion = 10000000000.0;

		public object Convert(object value, Type targetType, object parameter, CultureInfo culture)
		{
			if (value != null && value != DependencyProperty.UnsetValue)
			{
				if (value is double)
				{
					return FormatDouble((double)value, culture);
				}
				if (value is int num)
				{
					return FormatDouble(num, culture);
				}
				string text = value.ToString();
				double result = 0.0;
				if (double.TryParse(text, out result))
				{
					return FormatDouble(result, culture);
				}
				return text;
			}
			return null;
		}

		public object ConvertBack(object value, Type targetType, object parameter, CultureInfo culture)
		{
			throw new NotImplementedException();
		}

		private object FormatDouble(double value, CultureInfo culture)
		{
			if (value < 10000000000.0)
			{
				if (value % 1.0 == 0.0)
				{
					return value.ToString("N0", culture);
				}
				return value.ToString("N4", culture);
			}
			return value.ToString("E5", culture);
		}

		{
			_timer = new DispatcherTimer(new TimeSpan(0, 0, 0, 0, 300), DispatcherPriority.Normal, _timer_tick, base.Dispatcher)
			{
				IsEnabled = false
			};
			base.Loaded += RevolvingLamp_Loaded;

			base.IsVisibleChanged += RevolvingLamp_IsVisibleChanged;

		}

			UpdateStoryboardForMovingBand();
		}

		}

		{

			base.OnRenderSizeChanged(sizeInfo);

		private void _timer_tick(object sender, EventArgs e)
		{
			if (!string.IsNullOrEmpty(RevolvingText))
			{
				string[] array = RevolvingText.Split(',');
				if (array.Length != 0)
				{
					_currentIndex++;
					_currentIndex %= array.Length * 4;
					int num = _currentIndex / 4;
					int repeatCount = _currentIndex % 4;
					StringBuilder stringBuilder = new StringBuilder();
					stringBuilder.Append(array[num]);
					stringBuilder.Append('.', repeatCount);
					ComputedDisplayText = stringBuilder.ToString();
				}
			}
		}

		protected virtual void OnIsRunningChanged(bool oldValue, bool newValue)
		{
			UpdateTimerStatus();
		}

		private void OnRevolvingTextChanged(string oldValue, string newValue)
		{
			UpdateTimerStatus();
		}

		private void UpdateTimerStatus()
		{
			_timer.IsEnabled = base.IsLoaded && IsRunning;
			if (string.IsNullOrEmpty(RevolvingText))
			{
				ComputedDisplayText = null;
			}
			else
			{
				ComputedDisplayText = RevolvingText;
			}
			ThreeDotsVisibility = ((!string.IsNullOrEmpty(RevolvingText)) ? Visibility.Collapsed : Visibility.Visible);
		}

		private void UpdateStoryboardForMovingBand()
		{
			if (UseMovingColorBand && base.IsLoaded && base.IsVisible)
			{
				StartStoryboard();
			}
			else
			{
				EndStoryboard();
			}
		}

		private void StartStoryboard()
		{
			if (_movingBandStoryboard == null)
			{
				ThicknessAnimation thicknessAnimation = new ThicknessAnimation
				{
					From = new Thickness(-6.0, 0.0, 0.0, 0.0),
					To = new Thickness(base.ActualWidth, 0.0, 0.0, 0.0),
					RepeatBehavior = RepeatBehavior.Forever,
					Duration = new Duration(TimeSpan.FromSeconds(1.0))
				};
				Storyboard.SetTarget(thicknessAnimation, this);
				Storyboard.SetTargetProperty(thicknessAnimation, new PropertyPath(ComputedBandMarginProperty));
				_movingBandStoryboard = new Storyboard();
				_movingBandStoryboard.Children.Add(thicknessAnimation);
			}
			else if (_movingBandStoryboard.Children.FirstOrDefault() is ThicknessAnimation thicknessAnimation2)
			{
				thicknessAnimation2.To = new Thickness(base.ActualWidth, 0.0, 0.0, 0.0);
			}
			_movingBandStoryboard.Begin();
		}

		private void EndStoryboard()
		{
			if (_movingBandStoryboard != null)
			{
				_movingBandStoryboard.Stop();
			}
		}

		protected override void OnPreviewMouseLeftButtonDown(MouseButtonEventArgs e)
		{
			if (e.OriginalSource is Ellipse)
			{
				InitializeDragging(e);
			}
			else
			{
				CaptureMouse();
			}
			e.Handled = true;
			base.OnPreviewMouseLeftButtonDown(e);
		}

		protected override void OnPreviewMouseMove(MouseEventArgs e)
		{
			if (e.OriginalSource is Ellipse)
			{
				ProcessDragging(e);
			}
			base.OnPreviewMouseMove(e);
		}

		protected override void OnPreviewMouseLeftButtonUp(MouseButtonEventArgs e)
		{
			if (e.OriginalSource is Ellipse)
			{
				EndDragging(e);
			}
			else if (Mouse.Captured == this)
			{
				ReleaseMouseCapture();
				DispathcerInvokeClicked();
			}
			e.Handled = true;
			base.OnPreviewMouseLeftButtonUp(e);
		}

		private void InitializeDragging(MouseButtonEventArgs e)
		{
			_draggingEllipse = e.OriginalSource as Ellipse;
			_offDragging = _draggingEllipse == _ellipseOff;
			_maskEllipse = (_offDragging ? _ellipseOffDrag : _ellipseOnDrag);
			_mouseDownPosition = e.GetPosition(_draggingEllipse.Parent as FrameworkElement).X;
			_everDragged = false;
			_draggingEllipse.CaptureMouse();
		}

		private void ProcessDragging(MouseEventArgs e)
		{
			if (_draggingEllipse == null || _maskEllipse == null || Mouse.Captured != _draggingEllipse)
			{
				return;
			}
			double num = e.GetPosition(_draggingEllipse.Parent as FrameworkElement).X - _mouseDownPosition;
			if ((_offDragging && num >= 3.0) || (!_offDragging && num <= -3.0))
			{
				_draggingEllipse.Opacity = 0.5;
				_maskEllipse.Visibility = Visibility.Visible;
				_everDragged = true;
				double actualWidth = (_draggingEllipse.Parent as FrameworkElement).ActualWidth;
				if (_offDragging)
				{
					_maskEllipse.Margin = new Thickness(Math.Min(num + 2.0, actualWidth - 14.0), 2.0, 0.0, 2.0);
				}
				else
				{
					_maskEllipse.Margin = new Thickness(0.0, 2.0, Math.Min(actualWidth - 14.0, 2.0 - num), 2.0);
				}
			}
			else
			{
				_draggingEllipse.Opacity = 1.0;
				_maskEllipse.Visibility = Visibility.Collapsed;
				_maskEllipse.Margin = new Thickness(2.0);
			}
		}

		private void EndDragging(MouseButtonEventArgs e)
		{
			if (_draggingEllipse != null && _maskEllipse != null && Mouse.Captured == _draggingEllipse)
			{
				double num = e.GetPosition(_draggingEllipse.Parent as FrameworkElement).X - _mouseDownPosition;
				_maskEllipse.Margin = new Thickness(2.0);
				_maskEllipse.Visibility = Visibility.Collapsed;
				_draggingEllipse.ReleaseMouseCapture();
				_draggingEllipse.Opacity = 1.0;
				_draggingEllipse = null;
				if ((_offDragging && num >= 3.0) || (!_offDragging && num <= -3.0))
				{
					DispathcerInvokeClicked();
				}
				else if (!_everDragged)
				{
					DispathcerInvokeClicked();
				}
			}
		}

		private void DispathcerInvokeClicked()
		{
			base.Dispatcher.BeginInvoke(new Action(OnClicked), DispatcherPriority.Loaded);
		}

	public abstract class OnOffControlBase : Control
	{
		public static readonly DependencyProperty StateProperty = DependencyProperty.Register("State", typeof(OnOffState), typeof(OnOffControlBase), new FrameworkPropertyMetadata(OnOffState.Off, OnStatePropertyChanged));

		public static readonly RoutedEvent StateChangingEvent = EventManager.RegisterRoutedEvent("StateChanging", RoutingStrategy.Bubble, typeof(EventHandler<PropertyChangingRoutedEventArgs<OnOffState>>), typeof(OnOffControlBase));

		public static readonly RoutedEvent StateChangedEvent = EventManager.RegisterRoutedEvent("StateChanged", RoutingStrategy.Bubble, typeof(RoutedEventHandler), typeof(OnOffControlBase));

		public static readonly RoutedEvent ClickedEvent = EventManager.RegisterRoutedEvent("Clicked", RoutingStrategy.Bubble, typeof(RoutedEventHandler), typeof(OnOffControlBase));

		public OnOffState State
		{
			get
			{
				return (OnOffState)GetValue(StateProperty);
			}
			set
			{
				SetValue(StateProperty, value);
			}
		}

		public event EventHandler<PropertyChangingRoutedEventArgs<OnOffState>> StateChanging
		{
			add
			{
				AddHandler(StateChangingEvent, value);
			}
			remove
			{
				RemoveHandler(StateChangingEvent, value);
			}
		}

		public event RoutedEventHandler StateChanged
		{
			add
			{
				AddHandler(StateChangedEvent, value);
			}
			remove
			{
				RemoveHandler(StateChangedEvent, value);
			}
		}

		public event RoutedEventHandler Clicked
		{
			add
			{
				AddHandler(ClickedEvent, value);
			}
			remove
			{
				RemoveHandler(ClickedEvent, value);
			}
		}

		protected virtual void OnStateChanging(PropertyChangingRoutedEventArgs<OnOffState> args)
		{
			RaiseEvent(args);
		}

		protected virtual void OnStateChanged(OnOffState oldValue, OnOffState newValue)
		{
			RaiseEvent(new RoutedEventArgs(StateChangedEvent, this));
		}

		protected virtual void OnClicked()
		{
			OnOffState state = State;
			OnOffState onOffState = ((state == OnOffState.Off) ? OnOffState.On : OnOffState.Off);
			PropertyChangingRoutedEventArgs<OnOffState> e = new PropertyChangingRoutedEventArgs<OnOffState>(this, StateChangingEvent, state, onOffState);
			OnStateChanging(e);
			if (!e.Cancel)
			{
				State = onOffState;
			}
			RaiseEvent(new RoutedEventArgs(ClickedEvent, this));
		}

		private static void OnStatePropertyChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
		{
			(d as OnOffControlBase).OnStateChanged((OnOffState)e.OldValue, (OnOffState)e.NewValue);
		}
	}

			: base(routedEvent, sender)

		protected virtual void OnValueChanged(double? oldValue, double? newValue)
		{
			ComputedDisplayText = CalculateDisplayText();
		}

		protected virtual void OnDisplayFormatDividingPointyChanged(double oldValue, double newValue)
		{
			ComputedDisplayText = CalculateDisplayText();
		}

		protected virtual void OnDisplayFormatAboveDividingPointChanged(string oldValue, string newValue)
		{
			ComputedDisplayText = CalculateDisplayText();
		}

		protected virtual void OnDisplayFormatBelowDividingPointChanged(string oldValue, string newValue)
		{
			ComputedDisplayText = CalculateDisplayText();
		}

		private string CalculateDisplayText()
		{
			if (Value.HasValue)
			{
				double value = Value.Value;
				if (value >= DisplayFormatDividingPoint && !string.IsNullOrEmpty(DisplayFormatAboveDividingPoint))
				{
					return value.ToString(DisplayFormatAboveDividingPoint);
				}
				if (value < DisplayFormatDividingPoint && !string.IsNullOrEmpty(DisplayFormatBelowDividingPoint))
				{
					return value.ToString(DisplayFormatBelowDividingPoint);
				}
				return value.ToString();
			}
			return null;
		}

		protected virtual void OnTeaserClicked()
		{
			RaiseEvent(new RoutedEventArgs(TeaserClickedEvent, this));
		}

		private void tool_Click(object sender, RoutedEventArgs e)
		{
			OnTeaserClicked();
		}

		private static void OnValuePropertyChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
		{
			(d as KpiControl).OnValueChanged((double?)e.OldValue, (double?)e.NewValue);
		}

		private static void OnDisplayFormatDividingPointPropertyChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
		{
			(d as KpiControl).OnDisplayFormatDividingPointyChanged((double)e.OldValue, (double)e.NewValue);

		}

		{
			(d as KpiControl).OnDisplayFormatBelowDividingPointChanged((string)e.OldValue, (string)e.NewValue);
		}

	public class DashboardScreenDataConverterV34 : CommonScreenConverterBase
	{
		public override bool IsScreenObsolete => true;

		public DashboardScreenDataConverterV34()
			: base("DashboardScreenFactoryKey", 34)
		{
		}

		protected override void ConvertOverride(XElement subject, XElement state)
		{
			if (state == null || state.Parent == null || state.Parent.Parent == null || state.Parent.Parent.Parent == null || state.Parent.Parent.Parent.Parent == null || !(state.FirstNode is XElement xElement))
			{
				return;
			}
			XElement xElement2 = new XElement("DashboardPanel");
			xElement2.SetAttributeValue("IsExpanded", true);
			xElement2.SetAttributeValue("Version", 32);
			foreach (XElement item in xElement.Elements())
			{
				XAttribute xAttribute = item.Attribute("Key");
				if (xAttribute == null)
				{
					continue;
				}
				XAttribute xAttribute2 = item.Attribute("State");
				if (xAttribute2 != null)
				{
					string itemTypeText = GetItemTypeText(xAttribute.Value);
					if (!string.IsNullOrEmpty(itemTypeText))
					{
						XElement xElement3 = new XElement("Item");
						xElement3.SetAttributeValue("Type", itemTypeText);
						xElement3.SetValue(xAttribute2.Value);
						xElement2.Add(xElement3);
					}
				}
			}
			state.Parent.Parent.Parent.Parent.SetAttributeValue("ActivationBarState", xElement2.ToString());
		}

		private string GetItemTypeText(string key)
		{
			if (string.Equals(key, "EconomicsAnalysisDashboardItem", StringComparison.InvariantCultureIgnoreCase))
			{
				return "AspenTech.ActivationDashboard.EconomicAnalysis.DashboardItem";
			}
			if (string.Equals(key, "EnergyAnalysisDashboardItem", StringComparison.InvariantCultureIgnoreCase))
			{
				return "AspenTech.ActivationDashboard.EnergyAnalysis.DashboardItem";
			}
			if (string.Equals(key, "EdrActivationDashboardItem", StringComparison.InvariantCultureIgnoreCase))
			{
				return "AspenTech.ActivationDashboard.EdrActivation.DashboardItem";
			}
			return null;
		}
	}
	internal static class DashboardItemStateManager
	{
		public static string Convert(IDashboardItem item, string state, int version)
		{
			if (string.IsNullOrEmpty(state))
			{
				return null;
			}
			IOrderedEnumerable<IDashboardItemStateConverter> orderedEnumerable = from x in item.Converters
				where x != null && x.TargetVersion > version
				orderby x.TargetVersion
				select x;
			string text = state;
			foreach (IDashboardItemStateConverter item2 in orderedEnumerable)
			{
				text = item2.Convert(text);
				if (string.IsNullOrEmpty(text))
				{
					return null;
				}
			}
			return text;
		}
	}
