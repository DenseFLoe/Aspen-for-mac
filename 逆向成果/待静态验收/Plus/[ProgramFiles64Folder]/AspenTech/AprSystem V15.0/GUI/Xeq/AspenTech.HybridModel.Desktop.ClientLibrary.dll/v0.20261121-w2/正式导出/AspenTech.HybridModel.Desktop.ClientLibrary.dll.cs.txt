	public static class AimbButton
	{
		public static readonly DependencyProperty UseLaunchCommandProperty = DependencyProperty.RegisterAttached("UseLaunchCommand", typeof(bool), typeof(AimbButton), new PropertyMetadata(false, OnUseLaunchCommandPropertyChanged));

		public static string Title => Resources.AimbButton_Title;

		public static string EnabledTooltip => Resources.AimbButton_EnabledTooltip;

		public static string DisabledTooltip => Resources.AimbButton_DisabledTooltip;

		public static ImageSource LargeRibbonImage => ImageResources.AimbIcon;

		public static bool GetUseLaunchCommand(UIElement d)
		{
			return (bool)d.GetValue(UseLaunchCommandProperty);
		}

		public static void SetUseLaunchCommand(UIElement d, bool value)
		{
			d.SetValue(UseLaunchCommandProperty, value);
		}

		private static void OnUseLaunchCommandPropertyChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
		{
			if ((bool)e.NewValue)
			{
				HookCommand(d);
			}
			else
			{
				UnhookCommand(d);
			}
		}

		private static void HookCommand(DependencyObject d)
		{
			if (d is ButtonBase buttonBase)
			{
				buttonBase.Command = new LaunchCommand();
			}
			else if (d is MenuItem menuItem)
			{
				menuItem.Command = new LaunchCommand();
			}
		}

		private static void UnhookCommand(DependencyObject d)
		{
			if (d is ButtonBase buttonBase)
			{
				buttonBase.ClearValue(ButtonBase.CommandProperty);
			}
			else if (d is MenuItem menuItem)
			{
				menuItem.ClearValue(MenuItem.CommandProperty);
			}
		}
	}

	public static class AimbAppHelper
	{
		public static void Launch()
		{
			Launch(delegate(string file, string arguments)
			{
				Process.Start(file, arguments);
			});
		}

		public static void Launch(Action<string, string> processStarter)
		{
			if (string.Equals(Process.GetCurrentProcess().ProcessName, "AspenTech.HybridModel.Desktop.ClientApp", StringComparison.OrdinalIgnoreCase))
			{
				NamedPipeHelper.SendClientMessage("Open_Web_Page");
				return;
			}
			string aimbDesktopExecutable = GetAimbDesktopExecutable();
			if (!string.IsNullOrEmpty(aimbDesktopExecutable))
			{
				processStarter(aimbDesktopExecutable, null);
				return;
			}
			if (Process.GetProcesses().Any((Process x) => string.Equals(x.ProcessName, "AspenTech.HybridModel.Desktop.ClientApp", StringComparison.OrdinalIgnoreCase)))
			{
				NamedPipeHelper.SendClientMessage("Open_Web_Page");
				return;
			}
			string aimbLauncherExecutable = GetAimbLauncherExecutable();
			if (!string.IsNullOrEmpty(aimbLauncherExecutable))
			{
				processStarter(aimbLauncherExecutable, "Open_Web_Page");
				return;
			}
			throw new InvalidOperationException(Resources.AimbButton_DisabledTooltip);
		}

		private static string GetAimbDesktopExecutable()
		{
			if (!string.IsNullOrWhiteSpace(LocalMachineRegistryKeys.AspenRoot))
			{
				return TryFindExecutableFilePath(LocalMachineRegistryKeys.AspenRoot, new string[2] { "Aspen AI Model Builder for Desktop V15.0", "Aspen AI Model Builder for Desktop V14.6" }, "AspenTech.Aimb.WpfApp.exe");
			}
			return null;
		}

		private static string GetAimbLauncherExecutable()
		{
			if (!string.IsNullOrWhiteSpace(LocalMachineRegistryKeys.AspenCommonRoot))
			{
				return TryFindExecutableFilePath(LocalMachineRegistryKeys.AspenCommonRoot, new string[2] { "Aspen AI Model Builder Cloud Launcher", "Aspen AI Model Builder" }, "AspenTech.HybridModel.Desktop.ClientApp.exe");
			}
			return null;
		}

		private static string TryFindExecutableFilePath(string parentFolder, IEnumerable<string> testingChildFolders, string executableFileName)
		{
			foreach (string testingChildFolder in testingChildFolders)
			{
				string text = Path.Combine(parentFolder, testingChildFolder, executableFileName);
				if (File.Exists(text))
				{
					return text;
				}
			}
			return null;
		}
	}
	public static class AspenLanguageSelector
	{
		private static readonly List<string> _supportedLanguages = new List<string> { "ENG", "JPN", "CHS" };

		public static void ApplyAspenLanguageSelection()
		{
			string keyName = "HKEY_CURRENT_USER\\Software\\AspenTech\\Engineering\\Language";
			string text = Registry.GetValue(keyName, null, string.Empty) as string;
			if (string.IsNullOrEmpty(text))
			{
				string text2 = Thread.CurrentThread.CurrentUICulture.ThreeLetterWindowsLanguageName;
				if (!_supportedLanguages.Contains(text2))
				{
					text2 = "ENG";
				}
				Registry.SetValue(keyName, null, text2);
				text = text2;
			}
			string text3 = text.ToUpperInvariant();
			string text4 = ((text3 == "CHS") ? "zh-HANS" : ((!(text3 == "JPN")) ? "en-US" : "ja-JP"));
			if (!string.IsNullOrEmpty(text4))
			{
				CultureInfo cultureInfo = null;
				try
				{
					cultureInfo = new CultureInfo(CultureInfo.CreateSpecificCulture(text4).Name, useUserOverride: true);
				}
				catch (CultureNotFoundException)
				{
				}
				Thread.CurrentThread.CurrentUICulture = cultureInfo ?? new CultureInfo("en-US");
			}
		}
	}

	public class LaunchCommand : ICommand
	{
		private bool _isExecuting;

		public event EventHandler CanExecuteChanged
		{
			add
			{
				CommandManager.RequerySuggested += value;
			}
			remove
			{
				CommandManager.RequerySuggested -= value;
			}
		}

		public void RaiseCanExecuteChanged()
		{
			CommandManager.InvalidateRequerySuggested();
		}

		public bool CanExecute(object parameter)
		{
			return true;
		}

		public void Execute(object parameter)
		{
			if (!_isExecuting)
			{
				Cursor overrideCursor = Mouse.OverrideCursor;
				try
				{
					_isExecuting = true;
					Mouse.OverrideCursor = Cursors.Wait;
					AimbAppHelper.Launch();
					Mouse.OverrideCursor = overrideCursor;
					_isExecuting = false;
				}
				catch (Exception ex)
				{
					Mouse.OverrideCursor = overrideCursor;
					_isExecuting = false;
					MessageBox.Show(Resources.AimbButton_Failed + "\n" + ex.Message, Resources.AimbButton_DialogCaption, MessageBoxButton.OK, MessageBoxImage.Hand);
				}
			}
		}
	}
