		public static string StrConvMixCase(string InpVal)
		{
			string text = "";
			try
			{
				int result = 0;
				if (int.TryParse(InpVal, out result))
				{
					text = InpVal;
					return text;
				}
				string[] array = InpVal.Split('-');
				for (int i = 0; i < array.Length; i++)
				{
					if (array[i].Length > 1)
					{
						array[i] = array[i].Substring(0, 1).ToUpper() + array[i].Substring(1).ToLower();
					}
					else if (array[i].Length > 0)
					{
						array[i] = array[i].ToUpper();
					}
				}
				for (int j = 0; j < array.Length - 1; j++)
				{
					text = text + array[j] + "-";
				}
				text += array[array.Length - 1];
			}
			catch
			{
			}
			return text;
		}

		public static string StrConvMixCase(string InpVal)
		{
			string text = "";
			try
			{
				int result = 0;
				if (int.TryParse(InpVal, out result))
				{
					text = InpVal;
					return text;
				}
				string[] array = InpVal.Split('-');
				for (int i = 0; i < array.Length; i++)
				{
					if (array[i].Length > 1)
					{
						array[i] = array[i].Substring(0, 1).ToUpper() + array[i].Substring(1).ToLower();
					}
					else if (array[i].Length > 0)
					{
						array[i] = array[i].ToUpper();
					}
				}
				for (int j = 0; j < array.Length - 1; j++)
				{
					text = text + array[j] + "-";
				}
				text += array[array.Length - 1];
			}
			catch
			{
			}
			return text;
		}

							}
						}
						num2 -= 5.0;
					}
					if (ItemsSourceData[i].StartOption.Equals("TIME"))
					{
						if (ItemsSourceData[i].StartOptionTime > 0.0)
						{
							ItemsSourceData[i].Multiple = (ItemsSourceData[i].StartOptionTime / minTime).Value;

					}

				{
					haveoverwidth = true;
					if (maxwidth < num)
					{
						maxwidth = num;
						removeMarginWidth = num2;
					}

				}

		private void TreeNodeBindingGriphics(FrameworkElement element, TreeNodeModel node)
		{
			TreeNodeModel treeNodeModel = new TreeNodeModel();
			IValueConverter converter;
			if (node.IsLeafNode)
			{
				treeNodeModel = node.Parent;
				converter = new BoolToChildrenVisibilityConverter();
			}
			else
			{
				treeNodeModel = node;
				converter = new BoolToParentVisibilityConverter();
			}
			Binding binding = new Binding
			{
				Source = treeNodeModel,
				Path = new PropertyPath("IsExpanded"),
				Mode = BindingMode.TwoWay,
				UpdateSourceTrigger = UpdateSourceTrigger.PropertyChanged,
				Converter = converter
			};
			BindingOperations.SetBinding(element, UIElement.VisibilityProperty, binding);
		}
