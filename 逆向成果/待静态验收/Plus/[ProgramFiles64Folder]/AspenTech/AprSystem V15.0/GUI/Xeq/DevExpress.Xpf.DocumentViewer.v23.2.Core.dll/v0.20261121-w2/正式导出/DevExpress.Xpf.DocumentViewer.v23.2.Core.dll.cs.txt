		public double CalcZoomFactor(ZoomMode zoomMode, double currentZoomFactor)
		{
			return zoomMode switch
			{
				ZoomMode.ActualSize => 1.0, 
				ZoomMode.PageLevel => CalcViewWholePage(), 
				ZoomMode.FitToWidth => CalcViewPageWidth(), 
				ZoomMode.FitToVisible => CalcViewPageVisibleWidth(), 
				ZoomMode.Custom => currentZoomFactor, 
				_ => throw new ArgumentException("zoomMode"), 
			};
		}

		public double CalcZoomFactor(ZoomMode zoomMode)
		{
			return CalcZoomFactor(zoomMode, 1.0);
		}

		private double CalcViewPageVisibleWidth()
		{
			if (Viewport.IsEmpty || PageVisibleSize.IsEmpty)
			{
				return 1.0;
			}
			return Viewport.Width / PageVisibleSize.Width;
		}

		protected virtual double CalcViewPageWidth()
		{
			if (Viewport.IsEmpty || PageSize.IsEmpty)
			{
				return 1.0;
			}
			return Viewport.Width / PageSize.Width;
		}

		protected virtual double CalcViewWholePage()
		{
			if (Viewport.IsEmpty || PageSize.IsEmpty)
			{
				return 1.0;
			}
			double val = Viewport.Width / PageSize.Width;
			double val2 = Viewport.Height / PageSize.Height;
			return Math.Min(val, val2);
		}

		public bool CanZoomIn()
		{
			List<double> zoomFactors = GetZoomFactors();
			int num = FindCurrentZoomIndex(zoomFactors);
			return num < zoomFactors.Count - 1;
		}

		public bool CanZoomOut()
		{
			List<double> zoomFactors = GetZoomFactors();
			int num = FindCurrentZoomIndex(zoomFactors);
			if (num <= zoomFactors.Count - 1)
			{
				return num > 0;
			}
			return false;
		}

		public double GetNextZoomFactor(bool isZoomIn)
		{
			List<double> zoomFactors = GetZoomFactors();
			int val = Math.Max(FindCurrentZoomIndex(zoomFactors) + (isZoomIn ? 1 : (-1)), 0);
			val = Math.Min(val, zoomFactors.Count - 1);
			return zoomFactors[val];
		}

		private ZoomMode GetZoomModeByZoomFactor(double zoomFactor)
		{
			if (DoubleExtensions.AreClose(ZoomHelper.CalcZoomFactor(ZoomMode.FitToWidth), zoomFactor))
			{
				return ZoomMode.FitToWidth;
			}
			if (DoubleExtensions.AreClose(ZoomHelper.CalcZoomFactor(ZoomMode.FitToVisible), zoomFactor))
			{
				return ZoomMode.FitToVisible;
			}
			if (DoubleExtensions.AreClose(ZoomHelper.CalcZoomFactor(ZoomMode.PageLevel), zoomFactor))
			{
				return ZoomMode.PageLevel;
			}
			if (DoubleExtensions.AreClose(zoomFactor, 1.0))
			{
				return ZoomMode.ActualSize;
			}
			return ZoomMode.Custom;
		}

		private void Zoom(bool isZoomIn)
		{
			double nextZoomFactor = GetNextZoomFactor(isZoomIn);
			ZoomFactor = nextZoomFactor;
		}

		private int FindCurrentZoomIndex(List<double> zoomFactors)
		{
			int num = zoomFactors.BinarySearch(ZoomFactor);
			if (num < 0)
			{
				return -num - 1;
			}
			return num;
		}

		protected void UpdateZoomFactor()
		{
			updateZoomLocker.DoLockedActionIfNotLocked((Action)delegate
			{
				ZoomFactor = ZoomHelper.CalcZoomFactor(ZoomMode, ZoomFactor);
			});
		}

		protected void UpdateZoomMode()
		{
			updateZoomLocker.DoLockedActionIfNotLocked((Action)delegate
			{
				ZoomMode = GetZoomModeByZoomFactor(ZoomFactor);
			});
		}

		private void RaiseZoomChanged(double oldZoomFactor, double newZoomFactor, ZoomMode oldZoomMode, ZoomMode newZoomMode)
		{
			if (this.ZoomChanged != null)
			{
				this.ZoomChanged(this, new ZoomChangedEventArgs(oldZoomFactor, newZoomFactor, oldZoomMode, newZoomMode));
			}
		}

		private void RaiseRotateAngleChanged(double oldValue, double newValue)
		{
			if (this.RotateAngleChanged != null)
			{
				this.RotateAngleChanged(this, new RotateAngleChangedEventArgs(oldValue, newValue));
			}
		}

		private void RaisePageIndexChanged()
		{
			if (this.PageIndexChanged != null)
			{
				this.PageIndexChanged(this, new PageIndexChangedEventArgs(PageIndex));
			}
		}

		protected virtual List<double> GetZoomFactors()
		{
			if (ZoomHelper == null)
			{
				return new List<double>(DefaultZoomLevels);
			}
			List<double> source = new List<double>(DefaultZoomLevels)
			{
				ZoomHelper.CalcZoomFactor(ZoomMode.FitToWidth),
				ZoomHelper.CalcZoomFactor(ZoomMode.PageLevel)
			};
			source = source.Distinct().ToList();
			source.Sort();
			return source;
		}

		protected virtual void OnPageVisibleSizeChanged()

			ZoomHelper.PageVisibleSize = PageVisibleSize;
			UpdateZoomFactor();
		}

		protected virtual void OnPageSizeChanged()

			UpdateZoomFactor();
		}

		protected virtual void OnViewportChanged()
		{
			ZoomHelper.Viewport = Viewport;
			UpdateZoomFactor();
		}

		protected virtual void OnZoomFactorChanged(double oldValue, double newValue)
		{
			ZoomMode oldZoomMode = ZoomMode;
			UpdateZoomMode();
			RaiseZoomChanged(oldValue, newValue, oldZoomMode, ZoomMode);
		}

		protected virtual void OnZoomModeChanged()
		{
			UpdateZoomFactor();
		}

		protected virtual void OnRotateAngleChanged(double oldValue, double newValue)
		{
			ZoomHelper.PageSize = PageSize;
			UpdateZoomFactor();
			RaiseRotateAngleChanged(oldValue, newValue);
		}

		protected virtual void OnPageIndexChanged()
		{
			RaisePageIndexChanged();
		}

		internal void UpdateZoomCommand()
		{
			CommandCheckItems input = ZoomCommand as CommandCheckItems;
			input.Do(delegate(CommandCheckItems x)
			{
				x.UpdateCheckState(UpdateZoomFactorCheckState);
			});
		}

		protected void UpdatePagination()
		{
			if (DocumentViewer != null)
			{
				((CommandPagination)PaginationCommand).CurrentPageNumber = ActualDocumentViewer.Return((DocumentViewerControl x) => x.CurrentPageNumber, () => 0);
				((CommandPagination)PaginationCommand).PageCount = ActualDocumentViewer.Return((DocumentViewerControl x) => x.PageCount, () => 0);
			}
		}

		{

		}

		protected virtual void OnZoomChanged(object sender, RoutedEventArgs e)

			UpdateZoomCommand();
		}

		protected virtual bool UpdateZoomFactorCheckState(CommandToggleButton item)
		{
			if (DocumentViewer == null)
			{
				return false;
			}
			return (item as CommandSetZoomFactorAndModeItem).Return((CommandSetZoomFactorAndModeItem x) => (ActualDocumentViewer.Return((DocumentViewerControl z) => z.ZoomMode, () => ZoomMode.Custom) != ZoomMode.Custom && x.ZoomMode == ActualDocumentViewer.Return((DocumentViewerControl z) => z.ZoomMode, () => ZoomMode.Custom)) || DoubleExtensions.AreClose(x.ZoomFactor, DocumentViewer.ZoomFactor), () => false);
		}

		protected virtual void UnsubscribeFromEvents()
		{
			ActualDocumentViewer.Do(delegate(DocumentViewerControl x)
			{
				x.ZoomChanged -= OnZoomChanged;
			});
			ActualDocumentViewer.Do(delegate(DocumentViewerControl x)
			{
				x.CurrentPageNumberChanged -= OnCurrentPageNumberChanged;
			});
			ActualDocumentViewer.Do(delegate(DocumentViewerControl x)
			{
				x.DocumentChanged -= OnDocumentChanged;
			});
		}

		protected virtual void SubscribeToEvents()
		{
			ActualDocumentViewer.Do(delegate(DocumentViewerControl x)
			{
				x.ZoomChanged += OnZoomChanged;
			});
			ActualDocumentViewer.Do(delegate(DocumentViewerControl x)

			});
			ActualDocumentViewer.Do(delegate(DocumentViewerControl x)

		protected virtual System.Windows.Size GetViewportSize()
		{
			if (ItemsPanel == null || !ItemsControl.Return((DocumentViewerItemsControl x) => x.IsInitialized, () => false) || !HasPages)
			{
				return System.Windows.Size.Empty;
			}
			System.Windows.Size size = CalcPagesMarginSize();
			return new System.Windows.Size(Math.Max(0.0, ItemsPanel.ActualWidth - size.Width), Math.Max(0.0, ItemsPanel.ActualHeight - size.Height));
		}

		protected virtual System.Windows.Size CalcPagesMarginSize()
		{
			System.Windows.Size result = default(System.Windows.Size);
			foreach (PageWrapper page in Pages)
			{
				System.Windows.Size size = page.CalcMarginSize();
				if (DoubleExtensions.GreaterThanOrClose(size.Width * size.Height, result.Width * result.Height))
				{
					result = size;
				}
			}
			return result;
		}

		protected bool IsInBounds(System.Windows.Point point)
		{
			if (!HasPages)
			{
				return false;
			}
			HitTestResult hitTestResult = VisualTreeHelper.HitTest(this, point);
			if (hitTestResult.With((HitTestResult x) => x.VisualHit) == null)
			{
				return false;
			}
			ScrollBar scrollBar = LayoutHelper.FindParentObject<ScrollBar>(hitTestResult.VisualHit);
			if (scrollBar != null)
			{
				return false;
			}
			return true;
		}

		protected virtual ObservableCollection<PageWrapper> GeneratePageWrappers(IEnumerable<IPage> pages)
		{
			List<PageWrapper> list = new List<PageWrapper>();
			switch (PageDisplayMode)
			{
			case PageDisplayMode.Single:
				list = pages.Select(CreatePageWrapper).ToList();
				break;
			case PageDisplayMode.Columns:
				if (ShowCoverPage)
				{
					list.Add(CreatePageWrapper(new IPage[1] { pages.First() }));
					list[0].IsCoverPage = true;
					list.AddRange(pages.Skip(1).Partition(ColumnsCount).Select(CreatePageWrapper));
				}
				else
				{
					list = pages.Partition(ColumnsCount).Select(CreatePageWrapper).ToList();
				}
				list.ForEach(delegate(PageWrapper x)
				{
					x.IsColumnMode = true;
				});
				break;
			case PageDisplayMode.Wrap:
			{
				list = WrapPages(pages).Select(CreatePageWrapper).ToList();
				int maxPageCount = 0;
				list.ForEach(delegate(PageWrapper x)
				{
					maxPageCount = Math.Max(maxPageCount, x.Pages.Count());
				});
				list.ForEach(delegate(PageWrapper x)
				{
					x.MaxPageCount = maxPageCount;
					x.PageDisplayMode = PageDisplayMode;
					x.ZoomFactor = BehaviorProvider.ZoomFactor;
					x.Viewport = BehaviorProvider.Viewport;
				});
				break;
			}
			}
			list.ForEach(delegate(PageWrapper x)
			{
				x.HorizontalPageSpacing = HorizontalPageSpacing;
			});
			list.ForEach(delegate(PageWrapper x)
			{
				x.VerticalPageSpacing = VerticalPageSpacing;
			});
			return new ObservableCollection<PageWrapper>(list);
		}

		protected virtual IEnumerable<IPage[]> WrapPages(IEnumerable<IPage> pages)
		{
			if (BehaviorProvider.ZoomMode != ZoomMode.Custom && BehaviorProvider.ZoomMode != ZoomMode.ActualSize)
			{
				return pages.Select((IPage x) => new IPage[1] { x });
			}
			System.Windows.Size viewportSize = GetViewportSize();
			List<IPage[]> list = new List<IPage[]>();
			double num = 0.0;
			List<IPage> list2 = new List<IPage>();
			foreach (IPage page in pages)
			{
				double num2 = page.PageSize.Width * BehaviorProvider.ZoomFactor + page.Margin.Left + page.Margin.Right;
				num += num2;
				if (DoubleExtensions.GreaterThan(num, viewportSize.Width) && list2.Count > 0)
				{
					list.Add(list2.ToArray());
					list2.Clear();
					num = num2;
				}
				list2.Add(page);
			}
			if (list2.Count > 0)
			{
				list.Add(list2.ToArray());
			}
			return list;
		}

		internal virtual void OnDocumentChangedInternal(IDocument oldValue, IDocument newValue)
		{
			documentInternal = newValue;
			PageCount = newValue.With((IDocument x) => x.Pages).Return((IEnumerable<IPage> x) => x.Count(), () => 0);
			SetCurrentPageNumber(1);
			AssignDocumentPresenterProperties();
			OnDocumentChanged(oldValue, newValue);
			RaiseDocumentChanged();
		}

		private void OnActualCommandProviderChangedInternal(CommandProvider oldValue, CommandProvider newValue)
		{
			oldValue.Do(delegate(CommandProvider x)
			{
				x.DocumentViewer = null;
			});
			newValue.Do(delegate(CommandProvider x)
			{
				x.DocumentViewer = this;
			});
			OnActualCommandProviderChanged(oldValue, newValue);
		}

		private void OnActualBehaviorProviderChangedInternal(BehaviorProvider oldValue, BehaviorProvider newValue)
		{
			oldValue.Do(delegate(BehaviorProvider x)
			{
				x.ZoomChanged -= OnBehaviorProviderZoomChanged;
			});
			oldValue.Do(delegate(BehaviorProvider x)
			{
				x.RotateAngleChanged -= OnBehaviorProviderRotateAngleChanged;
			});
			oldValue.Do(delegate(BehaviorProvider x)
			{
				x.PageIndexChanged -= OnBehaviorProviderPageIndexChanged;
			});
			newValue.Do(delegate(BehaviorProvider x)
			{
				x.ZoomFactor = ZoomFactor;
			});
			newValue.Do(delegate(BehaviorProvider x)
			{
				x.ZoomChanged += OnBehaviorProviderZoomChanged;
			});
			newValue.Do(delegate(BehaviorProvider x)
			{
				x.RotateAngleChanged += OnBehaviorProviderRotateAngleChanged;
			});
			newValue.Do(delegate(BehaviorProvider x)
			{
				x.PageIndexChanged += OnBehaviorProviderPageIndexChanged;
			});
			AssignDocumentPresenterProperties();
			OnActualBehaviorProviderChanged(oldValue, newValue);
		}

		private bool ChooseFile(string documentName, out string fileName)
		{
			IOpenFileDialogService openFileDialogService = OpenFileDialogTemplate.Return((Func<DataTemplate, IOpenFileDialogService>)TemplateHelper.LoadFromTemplate<OpenFileDialogService>, (Func<IOpenFileDialogService>)CreateDefaultOpenFileDialogService);
			IFileInfo fileInfo = null;
			AssignableServiceHelper2<DocumentViewerControl, IOpenFileDialogService>.DoServiceAction((DependencyObject)this, openFileDialogService, (Action<IOpenFileDialogService>)delegate(IOpenFileDialogService service)
			{
				fileInfo = (service.ShowDialog() ? service.Files.FirstOrDefault() : null);
			});
			if (fileInfo == null)
			{
				fileName = null;
				return false;
			}
			fileName = Path.Combine(fileInfo.DirectoryName, fileInfo.Name);
			return true;
		}

		private Rotation GetClockwiseRotation(Rotation rotation)
		{
			int num = (int)(rotation + 1);
			if (num <= Enum.GetValues(typeof(Rotation)).Length - 1)
			{
				return (Rotation)num;
			}
			return Rotation.Rotate0;
		}

		private Rotation GetCounterClockwiseRotation(Rotation rotation)
		{
			int num = (int)(rotation - 1);
			if (num >= 0)
			{
				return (Rotation)num;
			}
			return Rotation.Rotate270;
		}

		}

	public class IndexCalculator
	{
		private readonly DocumentViewerPanel documentViewerPanel;

		protected IDictionary<int, System.Windows.Size> ItemSizes { get; set; }

		protected internal DocumentViewerPanel Panel => documentViewerPanel;

		public IndexCalculator(DocumentViewerPanel panel)
		{
			ItemSizes = new Dictionary<int, System.Windows.Size>();
			documentViewerPanel = panel;
		}

		public double IndexToVerticalOffset(int index, bool showSingleItem)
		{
			if (showSingleItem)
			{
				double num = Panel.ExtentHeight / (Panel.ExtentHeight - Panel.ViewportHeight);
				if (DoubleExtensions.LessThanOrClose(GetRealItemSize(index).Height, Panel.ViewportHeight))
				{
					return IndexToVerticalOffset(index, showSingleItem: false);
				}
				return IndexToVerticalOffset(index, showSingleItem: false) / num;
			}
			double num2 = 0.0;
			for (int i = 0; i < index; i++)
			{
				if (!HasItemSize(i))
				{
					return double.PositiveInfinity;
				}
				num2 += GetItemSize(i).Height;
			}
			return num2;
		}

		public int VerticalOffsetToIndex(double offset)
		{
			double num = offset;
			int num2 = 0;
			while (HasItemSize(num2) && (DoubleExtensions.GreaterThan(num, 0.0) || (int)num == 0))
			{
				num -= GetItemSize(num2++).Height;
			}
			if (DoubleExtensions.AreClose(num, 0.0))
			{
				return num2;
			}
			DoubleExtensions.GreaterThan(num, 0.0);
			return num2 - 1;
		}

		public void SetItemSize(int index, System.Windows.Size size)
		{
			if (ItemSizes.ContainsKey(index))
			{
				ItemSizes[index] = size;
			}
			else
			{
				ItemSizes.Add(index, size);
			}
		}

		public System.Windows.Size GetRealItemSize(int index)
		{
			if (!HasItemSize(index))
			{
				return System.Windows.Size.Empty;
			}
			return ItemSizes[index];
		}

		public System.Windows.Size GetItemSize(int index)
		{
			if (!HasItemSize(index))
			{
				return System.Windows.Size.Empty;
			}
			if (!Panel.ActualShowSingleItem || !DoubleExtensions.GreaterThanOrClose(Panel.ViewportHeight, ItemSizes[index].Height))
			{
				return ItemSizes[index];
			}
			return new System.Windows.Size(ItemSizes[index].Width, Panel.ViewportHeight);
		}

		public bool HasItemSize(int index)
		{
			return ItemSizes.ContainsKey(index);
		}
	}

			}
			System.Windows.Size viewportSize2 = new System.Windows.Size(DoubleExtensions.Round(viewportSize.Width, false), DoubleExtensions.Round(viewportSize.Height, false));
			System.Windows.Size extentSize = CalcExtentSize();
			InvalidateScrollInfo(viewportSize2, extentSize);

		private System.Windows.Size CalcExtentSize()
		{
			int count = ItemsControl.GetItemsOwner(this).Items.Count;
			ItemCollection items = ItemsControl.GetItemsOwner(this).Items;
			double num = 0.0;
			double num2 = 0.0;
			for (int i = 0; i < count; i++)
			{
				System.Windows.Size renderSize = ((PageWrapper)items[i]).RenderSize;
				IndexCalculator.SetItemSize(i, renderSize);
				num = Math.Max(renderSize.Width, num);
				num2 += renderSize.Height;
			}
			if (ShowSingleItem)
			{
				num2 = Math.Max(ViewportHeight * (double)count, num2);
			}
			return new System.Windows.Size(DoubleExtensions.Round(num, false), DoubleExtensions.Round(num2, false));
		}

		private void CleanUpItems(int startIndex, int endIndex)
		{
			UIElementCollection internalChildren = InternalChildren;
			IItemContainerGenerator itemContainerGenerator = base.ItemContainerGenerator;
			int count = internalChildren.Count;
			for (int num = count - 1; num >= 0; num--)
			{
				GeneratorPosition position = new GeneratorPosition(num, 0);
				int num2 = itemContainerGenerator.IndexFromGeneratorPosition(position);
				if (num2 < startIndex || num2 > endIndex)
				{
					itemContainerGenerator.Remove(position, 1);
					RemoveInternalChildRange(num, 1);
				}
			}
		}

		internal double CalcStartPosition(bool showSingleItem)
		{
			int num = IndexCalculator.VerticalOffsetToIndex(showSingleItem ? GetVirtualVerticalOffset() : VerticalOffset);
			if (showSingleItem)
			{
				double height = IndexCalculator.GetRealItemSize(num).Height;
				if (DoubleExtensions.LessThanOrClose(height, ViewportHeight))
				{
					return (ViewportHeight - height) / 2.0;
				}
				double num2 = IndexCalculator.IndexToVerticalOffset(num, showSingleItem: false);
				double value = (GetVirtualVerticalOffset() - num2) / height;
				double num3 = height - ViewportHeight;
				return (0.0 - num3) * Math.Abs(value);
			}
			double num4 = VerticalOffset;
			for (int i = 0; i < num; i++)
			{
				num4 -= IndexCalculator.GetRealItemSize(i).Height;
			}
			return 0.0 - num4;
		}

		private void CalcVisibleItemsRange(double viewportHeight, out int firstIndex, out int lastIndex)
		{
			if (ShowSingleItem)
			{
				firstIndex = IndexCalculator.VerticalOffsetToIndex(GetVirtualVerticalOffset());
				lastIndex = firstIndex;
				return;
			}
			int count = ItemsControl.GetItemsOwner(this).Items.Count;
			if (count == 0)
			{
				firstIndex = 0;
				lastIndex = 0;
				return;
			}
			int num = IndexCalculator.VerticalOffsetToIndex(VerticalOffset);
			viewportHeight += VerticalOffset - IndexCalculator.IndexToVerticalOffset(num, ShowSingleItem);
			firstIndex = ((num >= 0) ? num : 0);
			lastIndex = count - 1;
			for (int i = firstIndex; i < lastIndex; i++)
			{
				if (DoubleExtensions.LessThanOrClose(viewportHeight, 0.0))
				{
					lastIndex = i;
					break;
				}
				viewportHeight -= IndexCalculator.GetRealItemSize(i).Height;
			}
		}

		{
			int index = IndexCalculator.VerticalOffsetToIndex((!showSingleItem) ? GetVirtualVerticalOffset() : VerticalOffset);
			double pageRelativeVerticalOffset = GetPageRelativeVerticalOffset(!showSingleItem);
			ActualShowSingleItem = showSingleItem;
			double verticalOffset = IndexCalculator.IndexToVerticalOffset(index, showSingleItem) + RelativeOffsetToRealOffset(pageRelativeVerticalOffset, IndexCalculator.GetRealItemSize(index).Height, showSingleItem);
			SetVerticalOffset(verticalOffset);
		}

		private double RelativeOffsetToRealOffset(double relativeOffset, double pageHeight, bool showSingleItem)
		{
			if (!showSingleItem)
			{
				return relativeOffset * pageHeight;
			}
			if (DoubleExtensions.LessThanOrClose(pageHeight, ViewportHeight))
			{
				return 0.0;
			}
			double num = pageHeight - ViewportHeight;
			double num2 = ExtentHeight / (ExtentHeight - ViewportHeight);
			double num3 = pageHeight / num2;
			if (DoubleExtensions.GreaterThanOrClose(relativeOffset, num / pageHeight))
			{
				return num3 - 1.0;
			}
			double num4 = pageHeight / num;
			double num5 = relativeOffset * num4;
			return num3 * num5;
		}

		private void ScrollToRect(Rect rectangle, ScrollIntoViewMode mode)
		{
			switch (mode)
			{
			case ScrollIntoViewMode.Center:
				CenterRect(rectangle);
				break;
			case ScrollIntoViewMode.Edge:
				ScrollToEdge(rectangle);
				break;
			case ScrollIntoViewMode.TopLeft:
				ScrollToRect(rectangle);
				break;
			}
		}

		private void CenterRect(Rect rectangle)
		{
			if (ShowSingleItem)
			{
				CenterRectSingleItemMode(rectangle);
			}
			else if (!IsRectangleVisible(rectangle))
			{
				if (DoubleExtensions.GreaterThanOrClose(rectangle.Height, ViewportHeight))
				{
					SetVerticalOffset(rectangle.Top);
				}
				else
				{
					double num = (ViewportHeight - rectangle.Height) / 2.0;
					SetVerticalOffset(rectangle.Top - num);
				}
				if (DoubleExtensions.GreaterThanOrClose(rectangle.Width, ViewportWidth))
				{
					SetHorizontalOffset(rectangle.Left);
					return;
				}
				double num2 = (ViewportWidth - rectangle.Width) / 2.0;
				SetHorizontalOffset(rectangle.Left - num2);
			}
		}

		private void CenterRectSingleItemMode(Rect rectangle)
		{
			if (!IsRectangleVisible(rectangle))
			{
				int index = IndexCalculator.VerticalOffsetToIndex(rectangle.Top);
				if (DoubleExtensions.LessThanOrClose(IndexCalculator.GetRealItemSize(index).Height, ViewportHeight))
				{
					SetVerticalOffset(IndexCalculator.IndexToVerticalOffset(index, showSingleItem: true));
					return;
				}
				double num = IndexCalculator.IndexToVerticalOffset(index, showSingleItem: false);
				double relativeOffset = (rectangle.Top - num) / IndexCalculator.GetRealItemSize(index).Height;
				double verticalOffset = IndexCalculator.IndexToVerticalOffset(index, showSingleItem: true) + RelativeOffsetToRealOffset(relativeOffset, IndexCalculator.GetRealItemSize(index).Height, showSingleItem: true);
				SetVerticalOffset(verticalOffset);
				SetHorizontalOffset(rectangle.Left);
			}
		}

		private void ScrollToEdge(Rect rectangle)
		{
			if (ShowSingleItem)
			{
				ScrollToEdgeSingleItemMode(rectangle);
				return;
			}
			if (!IsRectangleVerticalVisible(rectangle))
			{
				if (DoubleExtensions.LessThanOrClose(rectangle.Top, VerticalOffset + 10.0))
				{
					SetVerticalOffset(rectangle.Top - 10.0);
				}
				if (DoubleExtensions.GreaterThanOrClose(rectangle.Bottom, VerticalOffset + ViewportHeight - 10.0))
				{
					SetVerticalOffset(rectangle.Bottom - ViewportHeight + 10.0);
				}
			}
			if (!IsRectangleHorizontalVisible(rectangle))
			{
				if (DoubleExtensions.GreaterThanOrClose(rectangle.Right, HorizontalOffset + ViewportWidth - 10.0))
				{
					SetHorizontalOffset(rectangle.Right - ViewportWidth + 10.0);
				}
				if (DoubleExtensions.LessThanOrClose(rectangle.Left, HorizontalOffset + 10.0))
				{
					SetHorizontalOffset(rectangle.Left - 10.0);
				}
			}
		}

		private void ScrollToEdgeSingleItemMode(Rect rectangle)
		{
			if (!IsRectangleVerticalVisible(rectangle))
			{
				int num = IndexCalculator.VerticalOffsetToIndex(GetVirtualVerticalOffset());
				double num2 = IndexCalculator.IndexToVerticalOffset(num, showSingleItem: false);
				double height = IndexCalculator.GetRealItemSize(num).Height;
				if (DoubleExtensions.GreaterThan(height, ViewportHeight))
				{
					double pageRelativeVerticalOffset = GetPageRelativeVerticalOffset(showSingleItem: true);
					num2 += height * pageRelativeVerticalOffset;
				}
				if (DoubleExtensions.LessThanOrClose(rectangle.Top, num2 + 10.0))
				{
					SetVerticalOffset(GetSinglePageModeVerticalOffset(rectangle.Top - 10.0));
				}
				if (DoubleExtensions.GreaterThanOrClose(rectangle.Bottom, num2 + ViewportHeight - 10.0))
				{
					int num3 = IndexCalculator.VerticalOffsetToIndex(rectangle.Bottom);
					if (num3 != num)
					{
						SetVerticalOffset(IndexCalculator.IndexToVerticalOffset(num3, showSingleItem: true));
						ScrollToEdgeSingleItemMode(rectangle);
					}
					else
					{
						SetVerticalOffset(GetSinglePageModeVerticalOffset(rectangle.Bottom - ViewportHeight + 10.0));
					}
				}
			}
			if (!IsRectangleHorizontalVisible(rectangle))
			{
				if (DoubleExtensions.GreaterThanOrClose(rectangle.Right, HorizontalOffset + ViewportWidth - 10.0))
				{
					SetHorizontalOffset(rectangle.Right - ViewportWidth + 10.0);
				}
				if (DoubleExtensions.LessThanOrClose(rectangle.Left, HorizontalOffset + 10.0))
				{
					SetHorizontalOffset(rectangle.Left - 10.0);
				}
			}
		}

		private void ScrollToRect(Rect rectangle)
		{
			SetVerticalOffset(ShowSingleItem ? GetSinglePageModeVerticalOffset(rectangle.Top) : rectangle.Top);
			SetHorizontalOffset(rectangle.Left);
		}

		private double GetSinglePageModeVerticalOffset(double offset)
		{
			int index = IndexCalculator.VerticalOffsetToIndex(offset);
			double num = IndexCalculator.IndexToVerticalOffset(index, showSingleItem: false);
			double relativeOffset = (offset - num) / IndexCalculator.GetRealItemSize(index).Height;
			return IndexCalculator.IndexToVerticalOffset(index, showSingleItem: true) + RelativeOffsetToRealOffset(relativeOffset, IndexCalculator.GetRealItemSize(index).Height, showSingleItem: true);
		}

		internal bool IsRectangleVisible(Rect rect)
		{
			if (IsRectangleVerticalVisible(rect))
			{
				return IsRectangleHorizontalVisible(rect);
			}
			return false;
		}

		private bool IsRectangleVerticalVisible(Rect rect)
		{
			Rect empty = Rect.Empty;
			if (ShowSingleItem)
			{
				int index = IndexCalculator.VerticalOffsetToIndex(GetVirtualVerticalOffset());
				double num = IndexCalculator.IndexToVerticalOffset(index, showSingleItem: false);
				double height = IndexCalculator.GetRealItemSize(index).Height;
				if (DoubleExtensions.GreaterThan(height, ViewportHeight))
				{
					double pageRelativeVerticalOffset = GetPageRelativeVerticalOffset(showSingleItem: true);
					num += height * pageRelativeVerticalOffset;
				}
				double height2 = (DoubleExtensions.LessThanOrClose(height, ViewportHeight) ? height : ViewportHeight);
				empty = new Rect(HorizontalOffset, num, ViewportWidth, height2);
			}
			else
			{
				empty = new Rect(HorizontalOffset, VerticalOffset, ViewportWidth, ViewportHeight);
			}
			Rect rect2 = Rect.Union(empty, rect);
			return DoubleExtensions.AreClose(DoubleExtensions.Round(empty.Height, false), DoubleExtensions.Round(rect2.Height, false));
		}

		private bool IsRectangleHorizontalVisible(Rect rect)
		{
			Rect rect2 = new Rect(HorizontalOffset, 0.0, ViewportWidth, ExtentHeight);
			Rect rect3 = Rect.Union(rect2, rect);
			return DoubleExtensions.AreClose(DoubleExtensions.Round(rect2.Width, false), DoubleExtensions.Round(rect3.Width, false));
		}

		private void InvalidateScrollInfo(System.Windows.Size viewportSize, System.Windows.Size extentSize)
		{
			if (scrollData.ViewportSize != viewportSize || scrollData.ExtentSize != extentSize)
			{
				scrollData.ViewportSize = viewportSize;
				scrollData.ExtentSize = extentSize;
				scrollData.VerticalLineSize = extentSize.Height / ((double)ItemsControl.GetItemsOwner(this).Items.Count * 10.0);
				scrollData.HorizontalLineSize = extentSize.Width / 50.0;
				scrollData.HorizontalOffset = Math.Max(0.0, Math.Min(ExtentWidth - ViewportWidth, HorizontalOffset));
				scrollData.VerticalOffset = Math.Max(0.0, Math.Min(ExtentHeight - ViewportHeight, VerticalOffset));
				ScrollOwner.Do(delegate(ScrollViewer x)
				{
					x.InvalidateScrollInfo();
				});
				InvalidateMeasure();
			}
		}

		{

		public virtual void ProcessKeyDown(KeyEventArgs e)
		{
			switch (e.Key)
			{
			case Key.Up:
				CommandProvider.With((CommandProvider x) => x.ScrollCommand).Do(delegate(ICommand x)
				{
					x.Execute(ScrollCommand.LineUp);
				});
				break;
			case Key.Down:
				CommandProvider.With((CommandProvider x) => x.ScrollCommand).Do(delegate(ICommand x)
				{
					x.Execute(ScrollCommand.LineDown);
				});
				break;
			case Key.Left:
				CommandProvider.With((CommandProvider x) => x.ScrollCommand).Do(delegate(ICommand x)
				{
					x.Execute(ScrollCommand.LineLeft);
				});
				break;
			case Key.Right:
				CommandProvider.With((CommandProvider x) => x.ScrollCommand).Do(delegate(ICommand x)
				{
					x.Execute(ScrollCommand.LineRight);
				});
				break;
			case Key.Prior:
				CommandProvider.With((CommandProvider x) => x.ScrollCommand).Do(delegate(ICommand x)
				{
					x.Execute(ScrollCommand.PageUp);
				});
				break;
			case Key.Next:

				});
				break;

				{

				});
				break;
			}

	public class NavigationStrategy
	{
		private readonly Locker pageIndexSyncLocker = new Locker();

		private readonly Locker backPageIndexSyncLocker = new Locker();

		private readonly Locker scrollLocker = new Locker();

		private readonly Locker changedPositionLocker = new Locker();

		private readonly Locker defferedScrollLocker = new Locker();

		private readonly DocumentPresenterControl presenter;

		protected DocumentViewerPanel ItemsPanel => presenter.ItemsPanel;

		protected ScrollViewer ScrollViewer => presenter.ScrollViewer;

		protected BehaviorProvider BehaviorProvider => presenter.BehaviorProvider;

		protected DocumentPresenterControl Presenter => presenter;

		protected internal PositionCalculator PositionCalculator { get; private set; }

		private int PageCount => presenter.Pages.Return((ObservableCollection<PageWrapper> x) => x.Count(), () => 0);

		private int PageIndex => BehaviorProvider.PageIndex;

		private IEnumerable<PageWrapper> Pages => presenter.Pages;

		public NavigationStrategy(DocumentPresenterControl documentPresenter)
		{
			//IL_0001: Unknown result type (might be due to invalid IL or missing references)
			//IL_000b: Expected O, but got Unknown
			//IL_000c: Unknown result type (might be due to invalid IL or missing references)
			//IL_0016: Expected O, but got Unknown
			//IL_0017: Unknown result type (might be due to invalid IL or missing references)
			//IL_0021: Expected O, but got Unknown
			//IL_0022: Unknown result type (might be due to invalid IL or missing references)
			//IL_002c: Expected O, but got Unknown
			//IL_002d: Unknown result type (might be due to invalid IL or missing references)
			//IL_0037: Expected O, but got Unknown
			presenter = documentPresenter;
			PositionCalculator = CreatePositionCalculator();
		}

		public virtual void ProcessScroll(ScrollCommand command)
		{
			switch (command)
			{
			case ScrollCommand.LineUp:
				ScrollViewer.Do(delegate(ScrollViewer x)
				{
					x.LineUp();
				});
				break;
			case ScrollCommand.LineDown:
				ScrollViewer.Do(delegate(ScrollViewer x)
				{
					x.LineDown();
				});
				break;
			case ScrollCommand.LineLeft:
				ScrollViewer.Do(delegate(ScrollViewer x)
				{
					x.LineLeft();
				});
				break;
			case ScrollCommand.LineRight:
				ScrollViewer.Do(delegate(ScrollViewer x)
				{
					x.LineRight();
				});
				break;
			case ScrollCommand.PageUp:
				ScrollViewer.Do(delegate(ScrollViewer x)
				{
					x.PageUp();
				});
				break;
			case ScrollCommand.PageDown:
				ScrollViewer.Do(delegate(ScrollViewer x)
				{
					x.PageDown();
				});
				break;
			case ScrollCommand.Home:
				ScrollViewer.Do(delegate(ScrollViewer x)
				{
					x.ScrollToHome();
				});
				break;
			case ScrollCommand.End:
				if (PageIndex == PageCount - 1)
				{
					ScrollViewer.ScrollToEnd();
				}
				else
				{
					ScrollIntoView(PageCount - 1, Rect.Empty, ScrollIntoViewMode.TopLeft);
				}
				break;
			}
		}

		public virtual void ProcessScrollViewerScrollChanged(ScrollChangedEventArgs e)
		{
			UpdatePageIndex();
			if (!scrollLocker.IsLocked)
			{
				RegisterUndoAction(GenerateCurrentState(BehaviorProvider.RotateAngle, BehaviorProvider.ZoomFactor, BehaviorProvider.ZoomMode), UndoActionType.DeferredScroll);
			}
		}

		public virtual void ProcessPageIndexChanged(int pageIndex)
		{
			pageIndexSyncLocker.DoIfNotLocked((Action)delegate
			{
				backPageIndexSyncLocker.LockOnce();
				int pageWrapperIndex = PositionCalculator.GetPageWrapperIndex(pageIndex);
				ScrollIntoView(pageWrapperIndex, Rect.Empty, ScrollIntoViewMode.TopLeft);
			});
		}

		public virtual void ProcessRotateAngleChanged(RotateAngleChangedEventArgs e)
		{
			if (!changedPositionLocker.IsLocked && ScrollViewer != null)
			{
				NavigationState navigationState = GenerateCurrentState(e.OldValue, BehaviorProvider.ZoomFactor, BehaviorProvider.ZoomMode);
				NavigationState navigationState2 = navigationState.Clone();
				navigationState2.Angle = e.NewValue;
				UpdatePagesRotateAngle(e.NewValue);
				ChangePositionAndRegisterUndoAction(navigationState, navigationState2, UndoActionType.Rotate);
			}
			else
			{
				UpdatePagesRotateAngle(e.NewValue);
			}
		}

		public virtual void ProcessZoomChanged(ZoomChangedEventArgs e)
		{
			if (!changedPositionLocker.IsLocked && ScrollViewer != null)
			{
				NavigationState navigationState = GenerateCurrentState(BehaviorProvider.RotateAngle, e.OldZoomFactor, e.OldZoomMode);
				NavigationState navigationState2 = navigationState.Clone();
				navigationState2.ZoomFactor = e.ZoomFactor;
				navigationState2.ZoomMode = e.ZoomMode;
				UpdatePagesZoomFactor(e.ZoomFactor);
				ChangePositionAndRegisterUndoAction(navigationState, navigationState2, UndoActionType.Zoom);
			}
			else
			{
				UpdatePagesZoomFactor(e.ZoomFactor);
			}
		}

		public virtual void ProcessScrollTo(double offset, bool isVertical)
		{
			if (isVertical)
			{
				ScrollViewer.ScrollToVerticalOffset(offset);
			}
			else
			{
				ScrollViewer.ScrollToHorizontalOffset(offset);
			}
		}

		public virtual void ProcessMarqueeZoom(Rect rect, double x, double y)
		{
			double zoomFactor = CalcFitRectangleZoomFactor(BehaviorProvider.Viewport, rect);
			ScrollToAnchorPoint(zoomFactor, CalcHorizontalAnchorPointForFitRectangle(rect, x), CalcVerticalAnchorPointForFitRectangle(rect, y), UndoActionType.Scroll);
		}

		public virtual void ZoomToAnchorPoint(bool isZoomIn, System.Windows.Point visibleAnchorPoint)
		{
			double num = CalcMouseWheelZoomFactor(isZoomIn);
			double num2 = BehaviorProvider.ZoomFactor / num;
			double num3 = visibleAnchorPoint.X + ScrollViewer.HorizontalOffset;
			double horizontalAnchor = Math.Max(num3 - num2 * visibleAnchorPoint.X, 0.0);
			double num4 = 0.0;
			if (Presenter.ShowSingleItem)
			{
				double pageRelativeVerticalOffset = ItemsPanel.GetPageRelativeVerticalOffset(showSingleItem: true);
				num4 = visibleAnchorPoint.Y + PositionCalculator.GetPageVerticalOffset(PageIndex, pageRelativeVerticalOffset);
				double height = ItemsPanel.IndexCalculator.GetRealItemSize(PageIndex).Height;
				if (DoubleExtensions.LessThanOrClose(height, ItemsPanel.ViewportHeight))
				{
					num4 -= (ItemsPanel.ViewportHeight - height) / 2.0;
				}
			}
			else
			{
				num4 = visibleAnchorPoint.Y + ScrollViewer.VerticalOffset;
			}
			double verticalAnchor = Math.Max(num4 - num2 * visibleAnchorPoint.Y, 0.0);
			ScrollToAnchorPoint(num, horizontalAnchor, verticalAnchor, UndoActionType.Zoom);
		}

		public void ChangePosition(NavigationState state)
		{
			ChangePositionInternal(state);
		}

		public void ScrollIntoView(int pageWrapperIndex, Rect rect, ScrollIntoViewMode mode)

		public void GenerateStartUpState()
		{

			{
				x.Flush();
			});

		public void ScrollToStartUp()
		{
			ScrollToStartUp(BehaviorProvider.Return((BehaviorProvider x) => x.PageIndex, () => 1));
		}

		public void ScrollToStartUp(int pageIndex)
		{
			ChangePositionWithDeferredScrollLocking(new NavigationState
			{
				Angle = BehaviorProvider.Return((BehaviorProvider x) => x.RotateAngle, () => 0.0),
				OffsetX = 0.0,
				OffsetY = 0.0,
				PageIndex = pageIndex,
				ZoomFactor = (BehaviorProvider.Return((BehaviorProvider x) => DoubleExtension.IsNumber(x.ZoomFactor), () => false) ? BehaviorProvider.ZoomFactor : 1.0),
				ZoomMode = BehaviorProvider.Return((BehaviorProvider x) => x.ZoomMode, () => ZoomMode.ActualSize)
			});
		}

		protected virtual PositionCalculator CreatePositionCalculator()
		{
			return new PositionCalculator(() => Pages);
		}

		private double CalcMouseWheelZoomFactor(bool isZoomIn)
		{
			return BehaviorProvider.GetNextZoomFactor(isZoomIn);
		}

		private void ChangePositionAndRegisterUndoAction(NavigationState oldState, NavigationState newState, UndoActionType state)
		{
			RegisterUndoAction(newState, state);
			defferedScrollLocker.Lock();
			ChangePosition(newState);
		}

		private void RegisterUndoAction(NavigationState newState, UndoActionType state)
		{
			//IL_0083: Unknown result type (might be due to invalid IL or missing references)
			//IL_008d: Expected O, but got Unknown
			if (presenter.UndoRedoManager == null)
			{
				return;
			}
			if (defferedScrollLocker.IsLocked && state == UndoActionType.DeferredScroll)
			{
				defferedScrollLocker.Unlock();
				return;
			}
			scrollLocker.LockOnce();
			presenter.UndoRedoManager.RegisterAction(new UndoState
			{
				State = newState,
				Perform = ChangePositionWithDeferredScrollLocking,
				ActionType = state
			});
			presenter.ImmediateActionsManager.EnqueueAction((IAction)new DelegateAction((Action)delegate
			{
				scrollLocker.Unlock();
			}));
		}

		private void ChangePositionInternal(NavigationState state)
		{
			changedPositionLocker.DoLockedActionIfNotLocked((Action)delegate
			{
				BehaviorProvider.Do(delegate(BehaviorProvider x)
				{
					x.RotateAngle = state.Angle;
				});
				if (state.ZoomMode != ZoomMode.Custom)
				{
					BehaviorProvider.Do(delegate(BehaviorProvider x)
					{
						x.ZoomMode = state.ZoomMode;
					});
				}
				else
				{
					BehaviorProvider.Do(delegate(BehaviorProvider x)
					{
						x.ZoomFactor = state.ZoomFactor;
					});
				}
				PositionCalculator calc = PositionCalculator;
				double horizontalOffset = calc.Return((PositionCalculator x) => x.GetPageHorizontalOffset(state.OffsetX), () => 0.0);
				double verticalOffset = calc.Return((PositionCalculator x) => x.GetPageVerticalOffset(state.PageIndex, state.OffsetY) - x.GetPageVerticalOffset(state.PageIndex, 0.0), () => 0.0);
				Presenter.With((DocumentPresenterControl x) => x.ImmediateActionsManager).Do(delegate(ImmediateActionsManager x)
				{
					//IL_0020: Unknown result type (might be due to invalid IL or missing references)
					//IL_002a: Expected O, but got Unknown
					x.EnqueueAction((IAction)new DelegateAction((Action)delegate
					{
						ScrollIntoView(calc.GetPageWrapperIndex(state.PageIndex), new Rect(horizontalOffset, verticalOffset, 1.0, 1.0), ScrollIntoViewMode.TopLeft);
					}));
				});
			});
		}

		private void ChangePositionWithDeferredScrollLocking(NavigationState state)
		{
			//IL_002a: Unknown result type (might be due to invalid IL or missing references)
			//IL_0034: Expected O, but got Unknown
			scrollLocker.LockOnce();
			ChangePosition(state);
			presenter.ImmediateActionsManager.EnqueueAction((IAction)new DelegateAction((Action)delegate
			{
				scrollLocker.Unlock();
			}));
		}

		protected NavigationState GenerateCurrentState(double rotateAngle, double zoomFactor, ZoomMode zoomMode)
		{

			double relativeOffsetY = PositionCalculator.GetRelativeOffsetY(verticalAnchor);
			navigationState2.OffsetY = relativeOffsetY;
			navigationState2.PageIndex = PositionCalculator.GetPageIndex(verticalAnchor);

		protected double CalcHorizontalAnchorPointForFitRectangle(Rect rect, double x)

			double num4 = rect.Height * num3;

			double val = num5 - num4 / 2.0;

			double num3 = rect.Width / num2;
			double num4 = y + rect.Height / 2.0 + num;
			double val = num4 - num3 / 2.0;
			return Math.Max(0.0, val);
		}

(Pages.First().Margin.Right + Pages.Last().Margin.Left) : ((!DoubleExtensions.AreClose(RotateAngle, 270.0)) ? (Pages.First().Margin.Left + Pages.Last().Margin.Right) : (Pages.First().Margin.Top + Pages.Last().Margin.Bottom))));
			double width = num + (double)(Pages.Count() - 1) * HorizontalPageSpacing;
			double num2 = 0.0;
			foreach (IPage page in Pages)
			{
				num2 = Math.Max(IsVertical ? (page.Margin.Top + page.Margin.Bottom) : (page.Margin.Left + page.Margin.Right), num2);

				num2 = Math.Max(num2, IsVertical ? size.Height : size.Width);

			return new System.Windows.Size(num, num2);

		public Rect GetPageRect(IPage page)

		protected Rect GetPageRectCore(IPage page)
		{
			double num;
			double num2;
			if (DoubleExtensions.AreClose(RotateAngle, 90.0))
			{
				num = Pages.First().Margin.Bottom;
				num2 = page.Margin.Left;
			}
			else if (DoubleExtensions.AreClose(RotateAngle, 180.0))
			{
				num = Pages.First().Margin.Right;
				num2 = page.Margin.Bottom;
			}
			else if (DoubleExtensions.AreClose(RotateAngle, 270.0))
			{
				num = Pages.First().Margin.Top;
				num2 = page.Margin.Right;
			}
			else
			{
				num = Pages.First().Margin.Left;
				num2 = page.Margin.Top;
			}
			double num3 = CalcFirstPageLeftOffset() + num;
			double y = CalcPageTopOffset(page) + num2;
			foreach (IPage page2 in Pages)
			{
				if (page2.PageIndex == page.PageIndex)
				{
					break;
				}
				System.Windows.Size originalPageSize = GetOriginalPageSize(page2);
				num3 += HorizontalPageSpacing + (IsVertical ? originalPageSize.Width : originalPageSize.Height) * ZoomFactor;
			}
			System.Windows.Size originalPageSize2 = GetOriginalPageSize(page);
			if (!IsVertical)
			{
				return new Rect(num3, y, originalPageSize2.Height * ZoomFactor, originalPageSize2.Width * ZoomFactor);
			}
			return new Rect(num3, y, originalPageSize2.Width * ZoomFactor, originalPageSize2.Height * ZoomFactor);

			if (DoubleExtensions.AreClose(RotateAngle, 90.0))

			else

		{
			if (PageWrapper == null)
			{
				return base.ArrangeOverride(finalSize);
			}

			{
				System.Windows.Size desiredSize = internalChild.DesiredSize;
				IPage page = (IPage)internalChild.DataContext;
				Rect pageRect = PageWrapper.GetPageRect(page);
				bool flag = PageWrapper.Pages.First() == page;
				Rect finalRect = new Rect(new System.Windows.Point(pageRect.Left - base.HorizontalOffset - (flag ? page.Margin.Left : (PageWrapper.HorizontalPageSpacing / 2.0)), 0.0), desiredSize);
				internalChild.Arrange(finalRect);
			}
			return finalSize;
		}

		protected virtual void OnPageWrapperChanged(PageWrapper pageWrapper)
		{
			IPage page = base.DataContext as IPage;
			IsCoverPage = pageWrapper.IsCoverPage;
			int num = pageWrapper.Pages.ToList().IndexOf(page);
			if (num == 0)
			{
				Position = PageControlItemPosition.Left;
			}
			else if (num == PageWrapper.Pages.Count() - 1)
			{
				Position = PageControlItemPosition.Right;
			}
			else
			{
				Position = PageControlItemPosition.Middle;
			}
			UpdateMargin(page.Margin);
		}

			PageNumber = input.Return((IPage x) => x.PageIndex, () => 0) + 1;
		}

		internal void UpdateMargin(Thickness margin)
		{
			if (IsCoverPage)
			{
				base.Margin = margin;
				return;
			}
			switch (Position)
			{
			case PageControlItemPosition.Left:
				margin.Right = DocumentViewer.Return((IDocumentViewerControl x) => x.HorizontalPageSpacing, () => 0.0) / 2.0;
				break;
			case PageControlItemPosition.Middle:
				margin.Left = DocumentViewer.Return((IDocumentViewerControl x) => x.HorizontalPageSpacing, () => 0.0) / 2.0;
				margin.Right = DocumentViewer.Return((IDocumentViewerControl x) => x.HorizontalPageSpacing, () => 0.0) / 2.0;
				break;
			case PageControlItemPosition.Right:
				margin.Left = DocumentViewer.Return((IDocumentViewerControl x) => x.HorizontalPageSpacing, () => 0.0) / 2.0;
				break;
			}
			base.Margin = margin;
		}

		protected override System.Windows.Size MeasureOverride(System.Windows.Size constraint)
		{
			IPage page = base.DataContext as IPage;
			if (DocumentViewer == null || page == null)
			{
				return base.MeasureOverride(constraint);
			}
			if (!IsVertical)
			{
				return new System.Windows.Size(page.PageSize.Height * DocumentViewer.ZoomFactor, page.PageSize.Width * DocumentViewer.ZoomFactor + DocumentViewer.VerticalPageSpacing);
			}
			return new System.Windows.Size(page.PageSize.Width * DocumentViewer.ZoomFactor, page.PageSize.Height * DocumentViewer.ZoomFactor + DocumentViewer.VerticalPageSpacing);
		}
	}

	{
		private double manipulationScale;

		protected DocumentPresenterControl Presenter { get; private set; }

		protected DocumentPresenterDecorator PresenterDecorator => Presenter.With((DocumentPresenterControl x) => x.PresenterDecorator);

		protected BehaviorProvider BehaviorProvider => Presenter.With((DocumentPresenterControl x) => x.BehaviorProvider);

		protected virtual double MinManipulationScale => 0.1;

		protected bool LockManipulationZooming { get; set; }

		public TouchController(DocumentPresenterControl presenter)
		{
			Presenter = presenter;
		}

		public virtual void ProcessManipulationCompleted(ManipulationCompletedEventArgs e)
		{
			e.Handled = true;
		}

		public virtual void ProcessManipulationDelta(ManipulationDeltaEventArgs e)
		{
			if (e.Handled)
			{
				return;
			}
			if (e.Manipulators.Count() != 2)
			{
				manipulationScale = 0.0;
				return;
			}
			manipulationScale += e.DeltaManipulation.Scale.X - 1.0;
			if (DoubleExtensions.GreaterThan(Math.Abs(manipulationScale), MinManipulationScale) && !LockManipulationZooming)
			{
				BehaviorProvider.Do(delegate(BehaviorProvider x)
				{
					x.ZoomFactor *= e.DeltaManipulation.Scale.X;
				});
			}
			e.Handled = true;
		}

		public virtual void ProcessManipulationStarted(ManipulationStartedEventArgs e)
		{
			manipulationScale = 0.0;
			e.Handled = true;
		}

		public virtual void ProcessManipulationStarting(ManipulationStartingEventArgs e)
		{
			if (e.Manipulators.Count() <= 2)
			{
				e.ManipulationContainer = PresenterDecorator;
				e.Handled = true;
			}
		}

		public virtual void ProcessManipulationInertiaStarting(ManipulationInertiaStartingEventArgs e)
		{
		}

		public virtual void ProcessStylusSystemGesture(StylusSystemGestureEventArgs e)

		}

	public class UndoRedoManager : BindableBase
	{
		private readonly TimeSpan timeSpan = TimeSpan.FromMilliseconds(300.0);

		private readonly DispatcherTimer timer;

		private readonly Locker undoRedoLocker;

		public bool CanUndo
		{
			get
			{
				if (UndoStack.Count <= 0)
				{
					return DeferredQueue.Count > 0;
				}
				return true;
			}
		}

		public bool CanRedo => RedoStack.Count > 0;

		private Stack<UndoState> UndoStack { get; set; }

		private UndoState CurrentState { get; set; }

		private Stack<UndoState> RedoStack { get; set; }

		private Queue<UndoState> DeferredQueue { get; set; }

		public UndoRedoManager(Dispatcher dispatcher)
		{
			//IL_003c: Unknown result type (might be due to invalid IL or missing references)
			//IL_0046: Expected O, but got Unknown
			UndoStack = new Stack<UndoState>();
			RedoStack = new Stack<UndoState>();
			DeferredQueue = new Queue<UndoState>();
			undoRedoLocker = new Locker();
			timer = new DispatcherTimer(timeSpan, DispatcherPriority.Normal, delegate
			{
				TimerTick();
			}, dispatcher);
		}

		private void TimerTick()
		{
			timer.Stop();
			FlushDeferredQueue();
		}

		public void Undo()
		{
			FlushDeferredQueue();
			if (CanUndo)
			{
				UndoState action = UndoStack.Pop();
				undoRedoLocker.DoLockedAction((Action)delegate
				{
					action.Perform(action.State);
				});
				RedoStack.Push(CurrentState);
				CurrentState = action;
				RaiseUndoRedoPropertiesChanged();
			}
		}

		public void Redo()
		{
			FlushDeferredQueue();
			if (CanRedo)
			{
				UndoState action = RedoStack.Pop();
				undoRedoLocker.DoLockedAction((Action)delegate
				{
					action.Perform(action.State);
				});
				UndoStack.Push(CurrentState);
				CurrentState = action;
				RaiseUndoRedoPropertiesChanged();
			}
		}

		public void Flush()
		{
			CurrentState = null;
			UndoStack.Clear();
			RedoStack.Clear();
			DeferredQueue.Clear();
			RaiseUndoRedoPropertiesChanged();
		}

		public void RegisterAction(UndoState action)
		{
			if (undoRedoLocker.IsLocked)
			{
				return;
			}
			if (CurrentState == null)
			{
				CurrentState = action;
			}
			else if (!IsSameAsCurrentState(action))
			{
				if (IsDeferredAction(action))
				{
					RegisterDeferredAction(action);
				}
				else
				{
					RegisterImmediateAction(action);
				}
			}
		}

		private bool IsSameAsCurrentState(UndoState action)
		{
			return action.Equals(CurrentState);
		}

		private void FlushDeferredQueue()
		{
			UndoState undoState = DeferredQueue.FirstOrDefault();
			UndoState undoState2 = DeferredQueue.LastOrDefault();
			DeferredQueue.Clear();
			if (undoState != null && undoState2 != null)
			{
				UndoState action = new UndoState
				{
					ActionType = undoState.ActionType,
					State = undoState2.State,
					Perform = undoState.Perform
				};
				RegisterImmediateAction(action);
			}
		}

		private void RegisterImmediateAction(UndoState action)
		{
			FlushDeferredQueue();
			CurrentState.Do(delegate(UndoState x)
			{
				UndoStack.Push(x);
			});
			CurrentState = action;
			RedoStack.Clear();
			RaiseUndoRedoPropertiesChanged();
		}

		private void RegisterDeferredAction(UndoState action)
		{
			DeferredQueue.Enqueue(action);
			timer.Stop();
			timer.Start();
		}

		private bool IsDeferredAction(UndoState action)
		{
			return action.ActionType == UndoActionType.DeferredScroll;
		}

		private void RaiseUndoRedoPropertiesChanged()
		{
			RaisePropertyChanged(() => CanUndo);
			RaisePropertyChanged(() => CanRedo);
		}
	}
