		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			SetVariables();
			IHNode iHNode = _context.FindNode(_varName);
			if (iHNode != null && iHNode.Elements.Count > 0)
			{
				_bDilutePhase = true;
				_bHasInputDialog = false;
			}
			else
			{
				iHNode = _context.FindNode(_varNames[0]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					_bDensePhase = true;
					_bHasInputDialog = true;
				}
			}
			if (!_bDilutePhase && !_bDensePhase)
			{
				ierror = 1;
				return;
			}
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				for (int i = 0; i < _cbArr.Count(); i++)
				{
					_cbArr[i].IsChecked = true;
				}
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
			_wizardPlot.RemoveAllDependentVariable();
			if (_bDilutePhase)
			{
				_wizardPlot.SetIndependentVariable(_context, _varName + "!Label");
				_wizardPlot.AddDependentVariable(_context, _varName);
				return;
			}
			for (int j = 0; j < _cbArr.Count(); j++)
			{
				if (_cbArr[j].IsChecked == true)
				{
					_wizardPlot.SetIndependentVariable(_context, _varNames[j] + "!Label");
					_wizardPlot.AddDependentVariable(_context, _varNames[j]);
				}
			}
		}

		public void UpdateWizardData()
		{
			SetVariables();
			_wizardPlot.RemoveAllDependentVariable();
			if (_bDilutePhase)
			{
				_wizardPlot.SetIndependentVariable(_context, _varName + "!Label");
				_wizardPlot.AddDependentVariable(_context, _varName);
				_bHasInputDialog = false;
			}
			else
			{
				int num = 0;
				for (int i = 0; i < _cbArr.Count(); i++)
				{
					if (_cbArr[i].IsChecked == true)
					{
						_wizardPlot.SetIndependentVariable(_context, _varNames[i] + "!Label");
						_wizardPlot.AddDependentVariable(_context, _varNames[i]);
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(num, _strLabels[i]);
						num++;
					}
				}
				_currentConfigValue = SaveConfigValue();
			}
			_help.SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentpDropUnit);
		}

		private void ExecuteOKCommand(object sender, ExecutedRoutedEventArgs e)

		}

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			int num = 0;
			for (int i = 0; i < _cbArr.Count(); i++)
			{
				if (_cbArr[i].IsChecked.Value)
				{
					num++;
					break;
				}
			}
			e.CanExecute = num != 0;
		}

		private void ExecuteCancelCommand(object sender, ExecutedRoutedEventArgs e)
		{
			_help.CloseForm((UserControl)this, (bool?)false);
		}

		private void SetVariables()
		{
			_cbArr = (AesCheckBox[])(object)new AesCheckBox[3] { checkBoxFricPresDrop, checkBoxElevPresDrop, checkBoxTotalPresDrop };
			_varNames = new string[3] { "Output/S_CUMDPH", "Output/S_CUMDPV", "Output/DENS_CUMDP" };
			_varName = "Output/S_CUMDPT";
			_strLabels = new string[3]
			{
				WpfPLPipelineResources.Cpdrop_Label_Fric,
				WpfPLPipelineResources.Cpdrop_Label_Elev,
				WpfPLPipelineResources.Cpdrop_Label_Total
			};
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			int num = 0;
			stringBuilder.Append("Cpdrop=");
			for (int i = 0; i < _cbArr.Count(); i++)
			{
				if (_cbArr[i].IsChecked == true)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(i.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			num = 0;
			stringBuilder.Append("CpdropEnable=");
			for (int j = 0; j < _cbArr.Count(); j++)
			{
				if (((UIElement)(object)_cbArr[j]).IsEnabled)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(j.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string[] array2 = array[num].Substring(array[num].IndexOf('=') + 1).Split(',');
			foreach (string value in array2)
			{
				_cbArr[Convert.ToInt16(value)].IsChecked = true;
			}
			num++;
			array2 = array[num].Substring(array[num].IndexOf('=') + 1).Split(',');
			foreach (string value2 in array2)
			{
				((UIElement)(object)_cbArr[Convert.ToInt16(value2)]).IsEnabled = true;
			}
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			SetVariables();
			IHNode iHNode = _context.FindNode(_varNames[0]);
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				ierror = 1;
				return;
			}
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				for (int i = 0; i < _cbArr.Count(); i++)
				{
					_cbArr[i].IsChecked = true;
				}
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
			_wizardPlot.RemoveAllDependentVariable();
			for (int j = 0; j < _cbArr.Count(); j++)
			{
				if (_cbArr[j].IsChecked == true)
				{
					_wizardPlot.SetIndependentVariable(_context, _varNames[j] + "!Label");
					_wizardPlot.AddDependentVariable(_context, _varNames[j]);
				}
			}
		}

		public void UpdateWizardData()
		{
			SetVariables();
			_wizardPlot.RemoveAllDependentVariable();
			int num = 0;
			for (int i = 0; i < _cbArr.Count(); i++)
			{
				if (_cbArr[i].IsChecked == true)
				{
					_wizardPlot.SetIndependentVariable(_context, _varNames[i] + "!Label");
					_wizardPlot.AddDependentVariable(_context, _varNames[i]);
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(num, _strLabels[i]);
					num++;
				}
			}
			_help.SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentpDropUnit);
			_currentConfigValue = SaveConfigValue();
		}

		}
