		protected internal virtual void ArrangeThumb(XamSliderThumb<T> thumb, bool isFirstTime)
		{
			if (base.Orientation == Orientation.Horizontal)
			{
				if (base.HorizontalTrack == null)
				{
					return;
				}
				Canvas.SetTop(thumb, 0.0);
				if (!base.IsDirectionReversed)
				{
					if (isFirstTime)
					{
						Canvas.SetLeft(thumb, ResolveThumbLocation(thumb.Value, base.HorizontalTrack.ActualWidth));
					}
					else
					{
						Canvas.SetTop(thumb, base.VerticalTrack.ActualHeight - ResolveThumbLocation(thumb.Value, base.VerticalTrack.ActualHeight) - thumb.ActualHeight / 2.0);
					}
				}
				else if (isFirstTime)
				{
					Canvas.SetLeft(thumb, ResolveThumbLocation(thumb.Value, base.HorizontalTrack.ActualWidth));
				}
				else
				{
					Canvas.SetLeft(thumb, ResolveThumbLocation(thumb.Value, base.HorizontalTrack.ActualWidth) - thumb.ActualWidth / 2.0);
				}
			}
			else if (base.VerticalTrack != null)
			{
				Canvas.SetLeft(thumb, 0.0);
				if (isFirstTime)
				{
					Canvas.SetTop(thumb, base.VerticalTrack.ActualHeight - ResolveThumbLocation(thumb.Value, base.VerticalTrack.ActualHeight));
				}
			}
		}

		protected override void EnsureOrientation()
		{
			if (base.Orientation == Orientation.Horizontal)
			{
				if (base.VerticalThumbsPanel != null)
				{
					base.VerticalThumbsPanel.Children.Clear();
				}
				if (base.VerticalTrackFillsPanel != null)
				{
					base.VerticalTrackFillsPanel.Children.Clear();
				}
				if (base.VerticalTickMarksContainer != null)
				{
					base.VerticalTickMarksContainer.Children.Clear();
				}
				if (base.HorizontalTickMarksContainer != null)
				{
					if (!base.HorizontalTickMarksContainer.Children.Contains(TickMarksPanel))
					{
						base.HorizontalTickMarksContainer.Children.Add(TickMarksPanel);
					}
					_isOrientationChanged = true;
				}
				if (base.HorizontalTrackFillsPanel != null)
				{
					foreach (XamSliderThumb<T> thumb in Thumbs)
					{
						if (!base.HorizontalTrackFillsPanel.Children.Contains(thumb.TrackFill))
						{
							base.HorizontalTrackFillsPanel.Children.Add(thumb.TrackFill);
						}
					}
				}
				if (base.HorizontalThumbsPanel != null)
				{
					foreach (XamSliderThumb<T> thumb2 in Thumbs)
					{
						if (!base.HorizontalThumbsPanel.Children.Contains(thumb2))
						{
							RemoveLogicalChild(thumb2);
							base.HorizontalThumbsPanel.Children.Add(thumb2);
						}
					}
				}
			}
			else
			{
				if (base.HorizontalThumbsPanel != null)
				{
					base.HorizontalThumbsPanel.Children.Clear();
				}
				if (base.HorizontalTrackFillsPanel != null)
				{
					base.HorizontalTrackFillsPanel.Children.Clear();
				}
				if (base.HorizontalTickMarksContainer != null)
				{
					base.HorizontalTickMarksContainer.Children.Clear();
				}
				if (base.VerticalTickMarksContainer != null)
				{
					if (!base.VerticalTickMarksContainer.Children.Contains(TickMarksPanel))
					{
						base.VerticalTickMarksContainer.Children.Add(TickMarksPanel);
					}
					_isOrientationChanged = true;
				}
				if (base.VerticalTrackFillsPanel != null)
				{
					foreach (XamSliderThumb<T> thumb3 in Thumbs)
					{
						if (!base.VerticalTrackFillsPanel.Children.Contains(thumb3.TrackFill))
						{
							base.VerticalTrackFillsPanel.Children.Add(thumb3.TrackFill);
						}
					}
				}
				if (base.VerticalThumbsPanel != null)
				{
					foreach (XamSliderThumb<T> thumb4 in Thumbs)
					{
						if (!base.VerticalThumbsPanel.Children.Contains(thumb4))
						{
							RemoveLogicalChild(thumb4);
							base.VerticalThumbsPanel.Children.Add(thumb4);
						}
					}
				}
			}
			ArrangeThumbs();
		}

		internal void EnsureTrackFill()
		{
			if (Thumbs.Count <= 0)
			{
				return;
			}
			for (int i = 0; i < SortedThumbs.Count; i++)
			{
				XamSliderThumb<T> xamSliderThumb = SortedThumbs[i];
				if (xamSliderThumb.IsTrackFillVisible)
				{
					xamSliderThumb.TrackFill.Visibility = Visibility.Visible;
					xamSliderThumb.ResolveTrackFill();
				}
				else
				{
					xamSliderThumb.TrackFill.Visibility = Visibility.Collapsed;
				}
			}
		}

		public override void OnApplyTemplate()
		{
			if (_tickMarksPanel != null)
			{
				if (base.HorizontalTickMarksContainer != null && base.HorizontalTickMarksContainer.Children.Contains(_tickMarksPanel))
				{
					base.HorizontalTickMarksContainer.Children.Remove(_tickMarksPanel);
				}
				if (base.VerticalTickMarksContainer != null && base.VerticalTickMarksContainer.Children.Contains(_tickMarksPanel))
				{
					base.VerticalTickMarksContainer.Children.Remove(_tickMarksPanel);
				}
				_tickMarksPanel = null;
			}
			RemoveThumbsFromHorizontalThumbsPanel();
			RemoveThumbsFromVerticalThumbsPanel();
			RemoveTrackFillFromHorizontalTrackFillsPanel();
			RemoveTrackFillFromVerticalTrackFillsPanel();
			base.OnApplyTemplate();
			if (ActiveThumb == null && Thumbs.Count > 0)
			{
				Thumbs[Thumbs.Count - 1].IsActive = true;
			}
		}

		protected override void OnKeyDown(KeyEventArgs e)
		{
			base.OnKeyDown(e);
			if (!base.EnableKeyboardNavigation || Thumbs.Count == 0 || ActiveThumb == null)
			{
				return;
			}
			if (e.Key == Key.Tab)
			{
				int num = SortedThumbs.IndexOf(ActiveThumb);
				e.Handled = true;

				double num4 = ((Owner.Orientation != Orientation.Horizontal) ? (-1.0 * (num2 - num3) * (verticalChange / (num - base.ActualHeight / 2.0))) : ((num2 - num3) * (horizontalChange / (num - base.ActualWidth / 2.0))));
