	public class RectangleIndents : ModelElement
	{
		public const int Undefined = -1;

		public int Left { get; set; }

		public int Top { get; set; }

		public int Right { get; set; }

		public int Bottom { get; set; }

		public int All
		{
			get
			{
				if (Left != Top || Left != Right || Left != Bottom)
				{
					return -1;
				}
				return Left;
			}
			set
			{
				Left = value;
				Top = value;
				Right = value;
				Bottom = value;
			}
		}

		public RectangleIndents(int left, int top, int right, int bottom)
		{
			Left = left;
			Top = top;
			Right = right;
			Bottom = bottom;
		}

		public RectangleIndents(int all)
			: this(all, all, all, all)
		{
		}

		public RectangleIndents()
			: this(0)
		{
		}
	}

	public static class LegendPositionConverter
	{
		private static readonly Dictionary<LegendPosition, LayoutAlignmentHorizontal> horizontalAlignmentDict;

		private static readonly Dictionary<LegendPosition, LayoutAlignmentVertical> verticalAlignmentDict;

		static LegendPositionConverter()
		{
			horizontalAlignmentDict = new Dictionary<LegendPosition, LayoutAlignmentHorizontal>();
			verticalAlignmentDict = new Dictionary<LegendPosition, LayoutAlignmentVertical>();
			horizontalAlignmentDict[LegendPosition.Left] = LayoutAlignmentHorizontal.Left;
			horizontalAlignmentDict[LegendPosition.Top] = LayoutAlignmentHorizontal.Center;
			horizontalAlignmentDict[LegendPosition.Right] = LayoutAlignmentHorizontal.Right;
			horizontalAlignmentDict[LegendPosition.Bottom] = LayoutAlignmentHorizontal.Center;
			horizontalAlignmentDict[LegendPosition.TopRight] = LayoutAlignmentHorizontal.Right;
			verticalAlignmentDict[LegendPosition.Left] = LayoutAlignmentVertical.Center;
			verticalAlignmentDict[LegendPosition.Top] = LayoutAlignmentVertical.Top;
			verticalAlignmentDict[LegendPosition.Right] = LayoutAlignmentVertical.Center;
			verticalAlignmentDict[LegendPosition.Bottom] = LayoutAlignmentVertical.Bottom;
			verticalAlignmentDict[LegendPosition.TopRight] = LayoutAlignmentVertical.Top;
		}

		public static LayoutAlignmentHorizontal ConverterToHorizontalAlignment(LegendPosition position, bool overlay)
		{
			LayoutAlignmentHorizontal layoutAlignmentHorizontal = horizontalAlignmentDict[position];
			if (!overlay)
			{
				if (layoutAlignmentHorizontal == LayoutAlignmentHorizontal.Left)
				{
					layoutAlignmentHorizontal = LayoutAlignmentHorizontal.LeftOutside;
				}
				if (layoutAlignmentHorizontal == LayoutAlignmentHorizontal.Right)
				{
					layoutAlignmentHorizontal = LayoutAlignmentHorizontal.RightOutside;
				}
			}
			return layoutAlignmentHorizontal;
		}

		public static LayoutAlignmentVertical ConverterToVerticalAlignment(LegendPosition position, bool overlay)
		{
			LayoutAlignmentVertical layoutAlignmentVertical = verticalAlignmentDict[position];
			if (!overlay)
			{
				if (layoutAlignmentVertical == LayoutAlignmentVertical.Top)
				{
					layoutAlignmentVertical = LayoutAlignmentVertical.TopOutside;
				}
				if (layoutAlignmentVertical == LayoutAlignmentVertical.Bottom)
				{
					layoutAlignmentVertical = LayoutAlignmentVertical.BottomOutside;
				}
			}
			return layoutAlignmentVertical;
		}
	}

	public class ModelElementContainer : IModelElementContainer
	{
		private readonly Dictionary<object, ModelElement> modelElements;

		private readonly Dictionary<ModelElement, object> viewObjects;

		public ModelElementContainer()
		{
			modelElements = new Dictionary<object, ModelElement>();
			viewObjects = new Dictionary<ModelElement, object>();
		}

		public void Clear()
		{
			modelElements.Clear();
			viewObjects.Clear();
		}

		public void Register(object viewObject, ModelElement element)
		{
			if (!modelElements.ContainsKey(viewObject))
			{
				modelElements[viewObject] = element;
			}
			if (!viewObjects.ContainsKey(element))
			{
				viewObjects[element] = viewObject;
			}
		}

		public ModelElement FindModelElement(object viewObject)
		{
			if (modelElements.TryGetValue(viewObject, out var value))
			{
				return value;
			}
			return null;
		}

		public object FindViewObject(ModelElement element)
		{
			if (viewObjects.TryGetValue(element, out var value))
			{
				return value;
			}
			return null;
		}
	}

	{
		private readonly ModelElement parent;

		protected ModelElement Parent => parent;

		protected ModelElementCollection(ModelElement parent)
		{
			this.parent = parent;
		}

		protected override void OnInsertComplete(int index, T value)
		{
			((DXCollectionBase<T>)(object)this).OnInsertComplete(index, value);
			value.Parent = parent;
		}

		protected override void OnRemoveComplete(int index, T value)
		{
			((DXCollectionBase<T>)(object)this).OnRemoveComplete(index, value);
			value.Parent = null;
		}

		protected override void OnSetComplete(int index, T oldValue, T newValue)
		{
			((DXCollectionBase<T>)(object)this).OnSetComplete(index, oldValue, newValue);
			oldValue.Parent = null;
			newValue.Parent = parent;
		}

		protected override bool OnClear()
		{
			for (int i = 0; i < ((DXCollectionBase<T>)(object)this).Count; i++)
			{
				((DXCollection<T>)this)[i].Parent = null;
			}
			return ((DXCollectionBase<T>)(object)this).OnClear();
		}
	}

	public struct ColorARGB : IComparable<ColorARGB>, IEquatable<ColorARGB>
	{
		public static ColorARGB Empty = default(ColorARGB);

		public static ColorARGB Transparent = new ColorARGB(0, byte.MaxValue, byte.MaxValue, byte.MaxValue);

		private readonly uint argb;

		public byte A => (byte)(argb >> 24);

		public byte R => (byte)(argb >> 16);

		public byte G => (byte)(argb >> 8);

		public byte B => (byte)argb;

		public bool IsEmpty => this == Empty;

		public bool IsTransparent => this == Transparent;

		public static ColorARGB Parse(string argbColor)
		{
			if (argbColor.Length != 8)
			{
				return Empty;
			}
			byte alpha = byte.Parse(argbColor.Substring(0, 2), NumberStyles.HexNumber);
			byte blue = byte.Parse(argbColor.Substring(2, 2), NumberStyles.HexNumber);
			byte green = byte.Parse(argbColor.Substring(4, 2), NumberStyles.HexNumber);
			byte red = byte.Parse(argbColor.Substring(6, 2), NumberStyles.HexNumber);
			return new ColorARGB(alpha, red, green, blue);
		}

		public static bool operator ==(ColorARGB color1, ColorARGB color2)
		{
			return color1.argb == color2.argb;
		}

		public static bool operator !=(ColorARGB color1, ColorARGB color2)
		{
			return !(color1 == color2);
		}

		public static bool Equals(ColorARGB color1, ColorARGB color2)
		{
			return color1 == color2;
		}

		public ColorARGB(byte alpha, byte red, byte green, byte blue)
		{
			argb = (uint)((alpha << 24) | (red << 16) | (green << 8) | blue);
		}

		public bool Equals(ColorARGB other)
		{
			return Equals(this, other);
		}

		public override bool Equals(object obj)
		{
			ColorARGB? colorARGB = obj as ColorARGB?;
			if (colorARGB.HasValue)
			{
				return Equals(this, colorARGB.Value);
			}
			return false;
		}

		public override int GetHashCode()
		{
			return argb.GetHashCode();
		}

		public int CompareTo(ColorARGB other)
		{
			return argb.CompareTo(other.argb);
		}
	}

	public class ColorHSL
	{
		private const float minLuminance = 0.5f;

		private const float maxLuminance = 0.8f;

		private float hue;

		private float saturation;

		private float luminance;

		public float Hue
		{
			get
			{
				return hue;
			}
			set
			{
				hue = value;
			}
		}

		public float Saturation
		{
			get
			{
				return saturation;
			}
			set
			{
				saturation = value;
			}
		}

		public float Luminance
		{
			get
			{
				return luminance;
			}
			set
			{
				luminance = value;
			}
		}

		public float MinLuminance => Math.Min(0.5f, luminance * 0.9f);

		public float MaxLuminance => Math.Max(0.8f, luminance + (1f - luminance) * 0.15f);

		private static float GetHue(ColorARGB color)
		{
			if (color.R == color.G && color.G == color.B)
			{
				return 0f;
			}
			float num = (float)(int)color.R / 255f;
			float num2 = (float)(int)color.G / 255f;
			float num3 = (float)(int)color.B / 255f;
			float num4 = 0f;
			float num5 = num;
			float num6 = num;
			if (num2 > num5)
			{
				num5 = num2;
			}
			if (num3 > num5)
			{
				num5 = num3;
			}
			if (num2 < num6)
			{
				num6 = num2;
			}
			if (num3 < num6)
			{
				num6 = num3;
			}
			float num7 = num5 - num6;
			if (num == num5)
			{
				num4 = (num2 - num3) / num7;
			}
			else if (num2 == num5)
			{
				num4 = 2f + (num3 - num) / num7;
			}
			else if (num3 == num5)
			{
				num4 = 4f + (num - num2) / num7;
			}
			num4 *= 60f;
			if (num4 < 0f)
			{
				num4 += 360f;
			}
			return num4;
		}

		private static float GetSaturation(ColorARGB color)
		{
			float num = (float)(int)color.R / 255f;
			float num2 = (float)(int)color.G / 255f;
			float num3 = (float)(int)color.B / 255f;
			float result = 0f;
			float num4 = num;
			float num5 = num;
			if (num2 > num4)
			{
				num4 = num2;
			}
			if (num3 > num4)
			{
				num4 = num3;
			}
			if (num2 < num5)
			{
				num5 = num2;
			}
			if (num3 < num5)
			{
				num5 = num3;
			}
			if (num4 != num5)
			{
				float num6 = (num4 + num5) / 2f;
				result = ((!((double)num6 <= 0.5)) ? ((num4 - num5) / (2f - num4 - num5)) : ((num4 - num5) / (num4 + num5)));
			}
			return result;
		}

		private static float GetBrightness(ColorARGB color)
		{
			float num = (float)(int)color.R / 255f;
			float num2 = (float)(int)color.G / 255f;
			float num3 = (float)(int)color.B / 255f;
			float num4 = num;
			float num5 = num;
			if (num2 > num4)
			{
				num4 = num2;
			}
			if (num3 > num4)
			{
				num4 = num3;
			}
			if (num2 < num5)
			{
				num5 = num2;
			}
			if (num3 < num5)
			{
				num5 = num3;
			}
			return (num4 + num5) / 2f;
		}

		private static byte GetComponent(float q, float p, float t)
		{
			while (t < 0f)
			{
				t += 1f;
			}
			while (t > 1f)
			{
				t -= 1f;
			}
			float num = ((t < 1f / 6f) ? (p + (q - p) * 6f * t) : ((t < 0.5f) ? q : ((!(t < 2f / 3f)) ? p : (p + (q - p) * (2f / 3f - t) * 6f))));
			return (byte)Math.Round(num * 255f);
		}

		public static explicit operator ColorHSL(ColorARGB color)
		{
			return new ColorHSL(GetHue(color), GetSaturation(color), GetBrightness(color));
		}

		public static explicit operator ColorARGB(ColorHSL color)
		{
			float num = ((color.luminance < 0.5f) ? (color.luminance * (1f + color.saturation)) : (color.luminance + color.saturation - color.luminance * color.saturation));
			float p = 2f * color.luminance - num;
			float num2 = color.hue / 360f;
			return new ColorARGB(byte.MaxValue, GetComponent(num, p, num2 + 1f / 3f), GetComponent(num, p, num2), GetComponent(num, p, num2 - 1f / 3f));
		}

		public ColorHSL(float hue, float saturation, float luminance)
		{
			this.hue = hue;
			this.saturation = saturation;
			this.luminance = luminance;
		}

	public abstract class ColorIndexCacheCore
	{
		private readonly Dictionary<int, int> indexCount = new Dictionary<int, int>();

		protected int GetActualIndexRepeatCount(int index, int actualIndex)
		{
			int num = 0;
			if (!indexCount.ContainsKey(actualIndex))
			{
				indexCount[actualIndex] = num;
			}
			else
			{
				num = ++indexCount[actualIndex];
			}
			return num;
		}

		protected ColorARGB ConvertColor(ColorARGB color, int colorRepeat)
		{
			ColorHSL colorHSL = (ColorHSL)color;
			float num = (((double)colorHSL.Luminance > 0.5) ? (1f - colorHSL.Luminance) : colorHSL.Luminance) / 4f;
			float num2 = colorHSL.Luminance + num;
			for (int i = 1; i < colorRepeat; i++)
			{
				num /= 1.5f;
				num2 = ((!(num2 + num > 1f)) ? (num2 + num) : (num2 + (1f - num2) / 2f));
			}
			colorHSL.Luminance = num2;
			return (ColorARGB)colorHSL;
		}
