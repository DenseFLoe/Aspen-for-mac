		{
			get
			{
				return _defaultOwner;
			}
			set
			{
				if (value == _defaultOwner)
				{
					return;
				}
				_defaultOwner = value;
				if (_defaultOwner == null)
				{
					return;
				}
				using IEnumerator<NavigatorTreeNode> enumerator = GetEnumerator();
				while (enumerator.MoveNext())
				{
					NavigatorTreeNode current = enumerator.Current;
					if (current != null && current.Owner == null)
					{
						current.Owner = _defaultOwner;
					}
				}
			}
		}

		}

		{
			_defaultOwner = defaultOwner;
		}

		protected override void InsertItem(int index, NavigatorTreeNode item)
		{
			if (_defaultOwner != null)
			{
				item.Owner = _defaultOwner;
			}
			base.InsertItem(index, item);
		}

	public enum CompositeEventRoutingStrategy
	{
		Children,
		Parents,
		ChildrenThenParents,
		ParentsThenChildren,
		FirstOfParentsThenChildren,
		FirstOfChildrenThenParents,
		FirstOfParents,

	}
	[CLSCompliant(false)]
	public class RoutingEventAggregatorUnityExtension : UnityContainerExtension
	{
		protected override void Initialize()
		{
		}

		private IEnumerable<IUnityContainer> EnumerateContainerChildren()
		{
			ILifetimeContainer lifetime = base.Context.Lifetime;
			if (lifetime == null)
			{
				yield break;
			}
			foreach (object item in lifetime)
			{
				if (item is IUnityContainer unityContainer)
				{
					yield return unityContainer;
				}
			}
		}

		public static IEnumerable<IUnityContainer> EnumerateContainerChildren(IUnityContainer container)
		{
			RoutingEventAggregatorUnityExtension routingEventAggregatorUnityExtension = container.Configure<RoutingEventAggregatorUnityExtension>();
			if (routingEventAggregatorUnityExtension == null)
			{
				yield break;
			}
			foreach (IUnityContainer item in routingEventAggregatorUnityExtension.EnumerateContainerChildren())
			{
				yield return item;
			}
		}
	}
	public class RoutingEventAggregator : IEventAggregator
	{
		private readonly List<EventBase> _events = new List<EventBase>();

		public IUnityContainer Container { get; private set; }

		public RoutingEventAggregator(IUnityContainer container)
		{
			Container = container;
		}

		public TEventType GetEvent<TEventType>() where TEventType : EventBase
		{
			TEventType val = _events.FirstOrDefault((EventBase evt) => evt.GetType() == typeof(TEventType)) as TEventType;
			if (val == null)
			{
				val = Container.Resolve<TEventType>(Array.Empty<ResolverOverride>());
				_events.Add(val);
			}
			return val;
		}

		internal void RoutedPublish<TPayload>(CompositeEventRoutingStrategy strategy, RoutedCompositePresentationEvent<TPayload> eventInstance, TPayload payload)
		{
			Type type = eventInstance.GetType();
			if (strategy == CompositeEventRoutingStrategy.AllTopDown)
			{
				IUnityContainer unityContainer = Container;
				while (unityContainer.Parent != null)
				{
					unityContainer = unityContainer.Parent;
				}
				PublishToContainer(unityContainer, type, payload);
				PublishToChildren(unityContainer, type, payload, stopAtFirstSubscriber: false);
				return;
			}
			bool flag = false;
			if (eventInstance.PublishToPeers(payload))
			{
				flag = true;
			}
			switch (strategy)
			{
			case CompositeEventRoutingStrategy.Children:
				PublishToChildren(Container, type, payload, stopAtFirstSubscriber: false);
				break;
			case CompositeEventRoutingStrategy.Parents:
				PublishToParents(Container, type, payload, stopAtFirstSubscriber: false);
				break;
			case CompositeEventRoutingStrategy.ChildrenThenParents:
				PublishToChildren(Container, type, payload, stopAtFirstSubscriber: false);
				PublishToParents(Container, type, payload, stopAtFirstSubscriber: false);
				break;
			case CompositeEventRoutingStrategy.ParentsThenChildren:
				PublishToParents(Container, type, payload, stopAtFirstSubscriber: false);
				PublishToChildren(Container, type, payload, stopAtFirstSubscriber: false);
				break;
			case CompositeEventRoutingStrategy.FirstOfChildrenThenParents:
				if (!flag && !PublishToChildren(Container, type, payload, stopAtFirstSubscriber: true))
				{
					PublishToParents(Container, type, payload, stopAtFirstSubscriber: true);
				}
				break;
			case CompositeEventRoutingStrategy.FirstOfParentsThenChildren:
				if (!flag && !PublishToParents(Container, type, payload, stopAtFirstSubscriber: true))
				{
					PublishToChildren(Container, type, payload, stopAtFirstSubscriber: true);
				}
				break;
			case CompositeEventRoutingStrategy.FirstOfParents:
				if (!flag)
				{
					PublishToParents(Container, type, payload, stopAtFirstSubscriber: true);
				}
				break;
			case CompositeEventRoutingStrategy.FirstOfParentsThenActiveWindow:
				if (!flag && !PublishToParents(Container, type, payload, stopAtFirstSubscriber: true))
				{
					PublishToActiveWindow(type, payload, stopAtFirstSubscriber: false);
				}
				break;
			}
		}

		private bool PublishToParents<TPayload>(IUnityContainer container, Type eventType, TPayload payload, bool stopAtFirstSubscriber)
		{
			bool result = false;
			IUnityContainer parent = container.Parent;
			if (parent != null)
			{
				RoutingEventAggregator routingEventAggregator = null;
				try
				{
					routingEventAggregator = parent.Resolve<IEventAggregator>(Array.Empty<ResolverOverride>()) as RoutingEventAggregator;
				}
				catch (Exception)
				{
				}
				if (routingEventAggregator != null && routingEventAggregator._events.FirstOrDefault((EventBase evt) => evt.GetType() == eventType) is RoutedCompositePresentationEvent<TPayload> routedCompositePresentationEvent && routedCompositePresentationEvent.PublishToPeers(payload))
				{
					result = true;
					if (stopAtFirstSubscriber)
					{
						return result;
					}
				}
				if (PublishToParents(parent, eventType, payload, stopAtFirstSubscriber))
				{
					result = true;
				}
			}
			return result;
		}

		private bool PublishToContainer<TPayload>(IUnityContainer container, Type eventType, TPayload payload)
		{
			bool result = false;
			RoutingEventAggregator routingEventAggregator = null;
			try
			{
				routingEventAggregator = container.Resolve<IEventAggregator>(Array.Empty<ResolverOverride>()) as RoutingEventAggregator;
			}
			catch (Exception)
			{
			}
			if (routingEventAggregator != null && routingEventAggregator._events.FirstOrDefault((EventBase evt) => evt.GetType() == eventType) is RoutedCompositePresentationEvent<TPayload> routedCompositePresentationEvent && routedCompositePresentationEvent.PublishToPeers(payload))
			{
				result = true;
			}
			return result;
		}

		private bool PublishToChildren<TPayload>(IUnityContainer container, Type eventType, TPayload payload, bool stopAtFirstSubscriber)
		{
			bool result = false;
			foreach (IUnityContainer item in RoutingEventAggregatorUnityExtension.EnumerateContainerChildren(container).ToList())
			{
				RoutingEventAggregator routingEventAggregator = null;
				try
				{
					routingEventAggregator = item.Resolve<IEventAggregator>(Array.Empty<ResolverOverride>()) as RoutingEventAggregator;
				}
				catch (Exception)
				{
				}
				if (routingEventAggregator != null && routingEventAggregator._events.FirstOrDefault((EventBase evt) => evt.GetType() == eventType) is RoutedCompositePresentationEvent<TPayload> routedCompositePresentationEvent && routedCompositePresentationEvent.PublishToPeers(payload))
				{
					result = true;
					if (stopAtFirstSubscriber)
					{
						break;
					}
				}
				if (PublishToChildren(item, eventType, payload, stopAtFirstSubscriber))
				{
					result = true;
					if (stopAtFirstSubscriber)
					{
						break;
					}
				}
			}
			return result;
		}

		private bool PublishToActiveWindow<TPayload>(Type eventType, TPayload payload, bool stopAtFirstSubscriber)
		{
			bool flag = false;
			IShellMainWindow shellMainWindow = null;
			try
			{
				IShellApplication shellApplication = Container.Resolve<IShellApplication>(Array.Empty<ResolverOverride>());
				if (shellApplication != null)
				{
					shellMainWindow = shellApplication.ActiveWindow;
				}
			}
			catch (Exception)
			{
			}
			if (shellMainWindow != null)
			{
				if (shellMainWindow.EventAggregator is RoutingEventAggregator routingEventAggregator && routingEventAggregator._events.FirstOrDefault((EventBase evt) => evt.GetType() == eventType) is RoutedCompositePresentationEvent<TPayload> routedCompositePresentationEvent)
				{
					flag = routedCompositePresentationEvent.PublishToPeers(payload);
				}
				if ((!flag || !stopAtFirstSubscriber) && PublishToChildren(shellMainWindow.Container, eventType, payload, stopAtFirstSubscriber))
				{
					flag = true;
				}
			}
			return flag;
		}
	}
	public class RoutedCompositePresentationEvent<TPayload> : CompositePresentationEvent<TPayload>
	{
		public CompositeEventRoutingStrategy RoutingStrategy { get; set; }

		[Microsoft.Practices.Unity.Dependency]
		public IEventAggregator EventAggregator { get; set; }

		public RoutedCompositePresentationEvent()
		{
			RoutingStrategy = CompositeEventRoutingStrategy.ChildrenThenParents;
		}

		public override void Publish(TPayload payload)
		{
			Publish(RoutingStrategy, payload);
		}

		public void Publish(CompositeEventRoutingStrategy strategy, TPayload payload)
		{
			if (EventAggregator is RoutingEventAggregator routingEventAggregator)
			{
				routingEventAggregator.RoutedPublish(strategy, this, payload);
			}
			else
			{
				PublishToPeers(payload);
			}
		}

		public bool PublishToPeers(TPayload payload)
		{
			if (base.Subscriptions.Count == 0)
			{
				return false;
			}
			base.InternalPublish(payload);
			return true;
		}
	}

		{
			UnityContainer = container;
			if (container != null)
			{
				EventAggregator = container.Resolve<IEventAggregator>(Array.Empty<ResolverOverride>());
			}
			ScreenSubject = screenSubject;
			if (screenSubject != null)
			{
				NavigationGroupKey = screenSubject.NavigationGroupKey;
			}
			if (screenState != null)
			{
				Zoom = screenState.Zoom;
			}
		}

		public virtual bool SatisfiesScreenRequirements(string screenFactoryKey, ScreenSubject screenSubject)
		{
			if (!string.Equals(screenFactoryKey, ScreenFactoryKey))
			{
				return false;
			}
			if (ScreenSubject == screenSubject)
			{
				return true;
			}
			if (ScreenSubject == null)
			{
				return false;
			}
			return ScreenSubject.SatisfiesScreenRequirements(screenSubject);
		}

		protected virtual void ExecuteZoomInCommand()
		{
			Zoom = Math.Min(ZoomMaximum, Zoom + ZoomSmallChange);
		}

		protected virtual bool CanExecuteZoomInCommand()
		{
			if (ZoomEnabled)
			{
				return Zoom < ZoomMaximum;
			}
			return false;
		}

		protected void RaiseCanExecuteZoomInCommandChanged()
		{
			if (_zoomInCommand != null)
			{
				_zoomInCommand.RaiseCanExecuteChanged();
			}
		}

		protected virtual void ExecuteZoomOutCommand()
		{
			Zoom = Math.Max(ZoomMinimum, Zoom - ZoomSmallChange);
		}

		protected virtual bool CanExecuteZoomOutCommand()
		{
			if (ZoomEnabled)
			{
				return Zoom > ZoomMinimum;
			}
			return false;
		}

		protected void RaiseCanExecuteZoomOutCommandChanged()
		{
			if (_zoomOutCommand != null)
			{
				_zoomOutCommand.RaiseCanExecuteChanged();
			}
		}

		protected virtual void ExecuteShowZoomDialogCommand()
		{
			ScreenZoomDialog screenZoomDialog = new ScreenZoomDialog(this);
			foreach (int item in ListZoomDialogPercentages())
			{
				screenZoomDialog.AddZoomLevel(item);
			}
			screenZoomDialog.ShowDialog();
		}

		public virtual IEnumerable<int> ListZoomDialogPercentages()
		{
			int[] array = new int[5] { 200, 100, 75, 50, 25 };
			int[] array2 = array;
			foreach (int num in array2)
			{
				if (!((double)num / 100.0 < ZoomMinimum) && !((double)num / 100.0 > ZoomMaximum))
				{
					yield return num;
				}
			}
		}

		protected virtual bool CanExecuteShowZoomDialogCommand()
		{
			return ZoomEnabled;
		}

		protected void RaiseCanExecuteShowZoomDialogCommandChanged()
		{
			if (_showZoomDialogCommand != null)
			{
				_showZoomDialogCommand.RaiseCanExecuteChanged();
			}
		}

		public override void InitializeScreen(IUnityContainer container, ScreenSubject screenSubject, ScreenState screenState)
		{
			base.InitializeScreen(container, screenSubject, screenState);
			IShellMainWindow shellMainWindow = container?.Resolve<IShellMainWindow>(Array.Empty<ResolverOverride>());
			if (shellMainWindow != null)
			{
				BindingOperations.SetBinding(this, ActiveDocumentScreenProperty, new Binding
				{
					Source = shellMainWindow,
					Path = new PropertyPath("ActiveDocumentScreen"),
					Mode = BindingMode.OneWay
				});
			}
			if (screenSubject is TrackingScreenPanelSubject trackingScreenPanelSubject)
			{
				base.Header = trackingScreenPanelSubject.Header;
				PanelContentId = trackingScreenPanelSubject.PanelId;
			}
		}

		{
			if (d is TrackingScreenPanelControl trackingScreenPanelControl)

				trackingScreenPanelControl.OnActiveDocumentScreenChanged(args.OldValue as ScreenContentControl, args.NewValue as ScreenContentControl);
			}

		}

		private static void OnPanelContentIdPropertyChanged(DependencyObject d, DependencyPropertyChangedEventArgs args)
		{
			if (d is TrackingScreenPanelControl trackingScreenPanelControl)
			{
				trackingScreenPanelControl.OnPanelContentIdChanged(args.OldValue as string, args.NewValue as string);
			}
		}

		protected virtual void OnPanelContentIdChanged(string oldValue, string newValue)
		{
			RefreshPanelContent();
		}

		protected virtual void RefreshPanelContent()
		{
			if (base.IsClosed)
			{
				return;
			}
			string panelContentId = PanelContentId;
			if (!string.IsNullOrEmpty(panelContentId))
			{
				FrameworkElement frameworkElement = ActiveDocumentScreen?.GetPanelContent(panelContentId);
				if (frameworkElement != base.Content)
				{
					base.Content = frameworkElement;
				}
			}
		}

		protected override void OnClosed(EventArgs args)
		{
			base.OnClosed(args);
			base.Content = null;
		}

		internal RadioButton _rbCustomZoom;

		internal XamMaskedEditor _zoomTextBox;

		public int ZoomMinimum { get; set; }

		public int ZoomMaximum { get; set; }

		private RadioButton SelectedRadioButton
		{
			get
			{
				foreach (object child in _gridRadioButtons.Children)
				{
					if (child is RadioButton { IsChecked: var isChecked } radioButton && isChecked == true)
					{
						return radioButton;
					}
				}
				return null;

		public ScreenZoomDialog(ScreenContentControl screenContentControl)
		{
			if (screenContentControl == null)
			{
				throw new ArgumentNullException("screenContentControl");
			}
			_screenContentControl = screenContentControl;
			base.Owner = Window.GetWindow(_screenContentControl);
			if (base.Owner != null)
			{
				base.Icon = base.Owner.Icon;
			}
			InitializeComponent();
			ZoomMinimum = (int)Math.Round(_screenContentControl.ZoomMinimum * 100.0);
			ZoomMaximum = (int)Math.Round(_screenContentControl.ZoomMaximum * 100.0);
			_zoomTextBox.PromptChar = ' ';
			_zoomTextBox.ValueType = typeof(int);
			_zoomTextBox.Mask = string.Format(CultureInfo.InvariantCulture, "{{number:{0}-{1}}}", ZoomMinimum, ZoomMaximum);
			_zoomTextBox.ValueChanged += _zoomTextBox_ValueChanged;
			_zoomTextBox.EditModeValidationError += _zoomTextBox_EditModeValidationError;
		}

		private void _zoomTextBox_EditModeValidationError(object sender, EditModeValidationErrorEventArgs e)
		{
			e.InvalidValueBehavior = InvalidValueBehavior.RetainValue;
		}

		public void AddZoomLevel(int zoomPercentage)
		{
			int num = _gridRadioButtons.Children.Count - 1;
			RadioButton element = new RadioButton
			{
				Style = (FindResource("zoomButtonStyle") as Style),
				Content = zoomPercentage.ToString(CultureInfo.CurrentCulture) + "%",
				Tag = zoomPercentage
			};
			_gridRadioButtons.RowDefinitions.Insert(num, new RowDefinition
			{
				Height = GridLength.Auto
			});
			Grid.SetRow(element, num);
			_gridRadioButtons.Children.Insert(num, element);
			Grid.SetRow(_rbCustomZoom, num + 1);
		}

		private void Window_Loaded(object sender, RoutedEventArgs e)
		{
			int num = (int)Math.Round(_screenContentControl.Zoom * 100.0);
			foreach (object child in _gridRadioButtons.Children)
			{
				if (child is RadioButton radioButton && radioButton != _rbCustomZoom)
				{
					int? zoomValue = GetZoomValue(radioButton);
					if (zoomValue.HasValue && zoomValue.Value == num)
					{
						radioButton.IsChecked = true;
						break;
					}
				}
			}
			_zoomTextBox.Value = num;
		}

		private void _zoomTextBox_ValueChanged(object sender, RoutedPropertyChangedEventArgs<object> e)
		{
			int? num = ((e.NewValue != DBNull.Value) ? ((int?)e.NewValue) : ((int?)null));
			if (GetZoomValue() != num)
			{
				_rbCustomZoom.IsChecked = true;
			}
			buttonOK.IsEnabled = num.HasValue && num.Value >= ZoomMinimum && num.Value <= ZoomMaximum;
			if (!buttonOK.IsEnabled && _rbCustomZoom.IsChecked == true)
			{
				if (_errorText.Visibility == Visibility.Collapsed)
				{
					_errorText.Text = string.Format(CultureInfo.CurrentCulture, CommonShellResources.ZoomValueOutofRange, ZoomMinimum, ZoomMaximum);
					_errorText.Visibility = Visibility.Visible;
					_errorText.Width = base.Width - 50.0;
					_zoomTextBox.BorderBrush = Brushes.Red;
				}
			}
			else if (_errorText.Visibility == Visibility.Visible)
			{
				_errorText.Visibility = Visibility.Collapsed;
				_errorText.Text = null;
				_zoomTextBox.ClearValue(Control.BorderBrushProperty);
			}
		}

		private void _gridRadioButtons_Checked(object sender, RoutedEventArgs e)
		{
			if (base.IsLoaded && e.Source is RadioButton radioButton && radioButton != _rbCustomZoom)
			{
				int? zoomValue = GetZoomValue(radioButton);
				if (zoomValue.HasValue)
				{
					_zoomTextBox.Value = zoomValue.Value;
				}
			}
		}

		private void buttonOK_Click(object sender, RoutedEventArgs e)
		{
			int? zoomValue = GetZoomValue();
			if (!zoomValue.HasValue)
			{
				MessageBox.Show(CommonShellResources.ScreenZoomControlConfigWindow_InputException, CommonShellResources.ScreenZoomControlConfigWindow_InputExceptionHeader);
				return;
			}
			_screenContentControl.Zoom = (double)zoomValue.Value / 100.0;
			base.DialogResult = true;
			Close();
		}

		private void buttonCancel_Click(object sender, RoutedEventArgs e)
		{
			base.DialogResult = false;
			Close();
		}

		private int? GetZoomValue()
		{
			int? result = null;
			RadioButton selectedRadioButton = SelectedRadioButton;
			if (selectedRadioButton != null)
			{
				return GetZoomValue(selectedRadioButton);
			}
			return result;
		}

		private int? GetZoomValue(RadioButton rb)
		{
			if (rb == _rbCustomZoom)
			{
				if (_zoomTextBox.Value == DBNull.Value)
				{
					return null;
				}
				return (int?)_zoomTextBox.Value;
			}
			return (int?)rb.Tag;
		}

			}
		}

		public override void RefreshScreen()
		{
			if (_showScreenPending || !HasScreenToShow())
			{
				return;
			}
			IShellMainWindow mainWindow = MainWindow;
			if (mainWindow != null)
			{
				ScreenContentControl activeDocumentScreen = mainWindow.ActiveDocumentScreen;
				if (activeDocumentScreen == null || !activeDocumentScreen.SatisfiesScreenRequirements(ScreenFactoryKey, ScreenSubject))
				{
					DispatcherShowScreen();
				}
			}
		}

		private void UpdateViewModelSelectedTreeNode()
		{
			if (base.RootOwner is SimpleTreeNavigatorViewModel simpleTreeNavigatorViewModel)
			{
				if (IsSelected)
				{
					simpleTreeNavigatorViewModel.SetSelectedTreeNode(this);
				}
				else if (this == simpleTreeNavigatorViewModel.SelectedTreeNode)
				{
					simpleTreeNavigatorViewModel.SetSelectedTreeNode(null);
				}
			}
		}

		public override bool HasScreenToShow()
		{
			return !string.IsNullOrEmpty(ScreenFactoryKey);
		}

		public override bool CanShowScreenInNewTab()
		{
			return true;
		}

		public override bool CanShowScreenInNewWindow()
		{
			return true;
		}

		public void DispatcherShowScreen()
		{
			if (HasScreenToShow() && !_showScreenPending)
			{
				_showScreenPending = true;
				Dispatcher.CurrentDispatcher.BeginInvoke((Action<object>)delegate
				{
					ExecuteShowScreen(null);
				}, this);
			}
		}

		public sealed override void ExecuteShowScreen(ShowScreenOptions options)
		{
			try
			{
				InternalExecuteShowScreen(options);
			}
			finally
			{
				_showScreenPending = false;
			}
		}

		protected virtual void InternalExecuteShowScreen(ShowScreenOptions options)
		{
			if (EventAggregator != null)
			{
				ShowScreenEventArgs e = new ShowScreenEventArgs
				{
					ScreenFactoryKey = ScreenFactoryKey,
					ScreenSubject = ScreenSubject
				};
				e.ApplyShowScreenOptions(options);
				EventAggregator.GetEvent<ShowScreenEvent>().Publish(e);
			}
		}

		protected virtual void OnIsExpandedChanged()
		{
			OnPropertyChanged("IsExpanded");
		}

		protected virtual void OnIsEnabledChanged()
		{
			OnPropertyChanged("IsEnabled");
		}

		protected virtual void OnIsVisibleChanged()
		{
			OnPropertyChanged("IsVisible");
		}

		public virtual bool IsMatchToScreenConfiguration(ScreenContentControl screen)
		{
			if (screen == null)
			{
				return false;
			}
			if (!string.Equals(screen.ScreenFactoryKey, ScreenFactoryKey))
			{
				return false;
			}
			if (ScreenSubject == screen.ScreenSubject)
			{
				return true;
			}
			if (ScreenSubject == null)
			{
				return false;
			}
			return ScreenSubject.SatisfiesScreenRequirements(screen.ScreenSubject);
		}

		public virtual bool ExpandToScreen(ScreenContentControl screen)
		{
			if (screen == null)
			{
				return false;
			}
			if (IsMatchToScreenConfiguration(screen))
			{
				SetSelectedWithoutSideEffects(value: true);
				return true;
			}
			foreach (SimpleNavigatorTreeNode child in Children)
			{
				if (child != null && child.ExpandToScreen(screen))
				{
					IsExpanded = true;
					return true;
				}
			}
			return false;
		}

		public virtual void ApplyViewFilter(string newFilter)
		{
			foreach (NavigatorTreeNode child in Children)
			{
				if (child is SimpleNavigatorTreeNode simpleNavigatorTreeNode)
				{
					simpleNavigatorTreeNode.ApplyViewFilter(newFilter);
				}
			}
		}
	}

			}
			if (treeViewItem == null)
			{
				treeView = _favoritesTree;

			}
			if (treeViewItem == null)
			{
				return;
			}
			ScrollViewer ancestor = GetAncestor<ScrollViewer>(treeViewItem);
			if (ancestor == null)
			{
				treeViewItem.BringIntoView();
				return;
			}
			double? num = EstimateItemHeight(treeView);
			if (!num.HasValue)
			{
				treeViewItem.BringIntoView();
				return;
			}
			int? itemDisplayIndex = GetItemDisplayIndex(treeView, treeViewItem);
			if (!itemDisplayIndex.HasValue)
			{
				treeViewItem.BringIntoView();
				return;
			}
			double num2 = num.Value * (double)itemDisplayIndex.Value;
			if (num2 > ancestor.VerticalOffset && num2 + num < ancestor.VerticalOffset + ancestor.ViewportHeight)
			{

			}

			double num3 = (ancestor.ViewportHeight - targetRectangle.Height) / 2.0;

		private static TreeViewItem VirtualizingFindItemContainer(ItemsControl container, NavigatorTreeNode item)
		{
			if (container == null || item == null)
			{
				return null;
			}
			Stack<NavigatorTreeNode> nodes = EnumerateNodesInPath(item);

		private static TreeViewItem VirtualizingFindItemContainerIterate(ItemsControl container, Stack<NavigatorTreeNode> nodes)
		{
			if (container == null || nodes == null || nodes.Count == 0)
			{
				return null;
			}
			if (container is TreeViewItem && !((TreeViewItem)container).IsExpanded)
			{
				container.SetValue(TreeViewItem.IsExpandedProperty, true);
			}
			NavigatorTreeNode navigatorTreeNode = nodes.Pop();
			if (navigatorTreeNode == null || navigatorTreeNode.Owner == null)
			{
				return null;
			}
			int num = navigatorTreeNode.Owner.Children.IndexOf(navigatorTreeNode);
			if (num < 0)
			{
				return null;
			}
			TreeViewItem treeViewItem = null;
			if (container.ItemContainerGenerator != null)
			{
				treeViewItem = (TreeViewItem)container.ItemContainerGenerator.ContainerFromIndex(num);
			}
			if (treeViewItem == null)
			{
				ItemsPresenter itemsPresenter = null;
				container.ApplyTemplate();
				if (container.Template != null)
				{
					itemsPresenter = (ItemsPresenter)container.Template.FindName("ItemsHost", container);
				}
				if (itemsPresenter != null)
				{
					itemsPresenter.ApplyTemplate();
				}
				else
				{
					itemsPresenter = FindVisualChild<ItemsPresenter>(container);
					if (itemsPresenter == null)
					{
						container.UpdateLayout();
						itemsPresenter = FindVisualChild<ItemsPresenter>(container);
					}
				}
				if (itemsPresenter == null)
				{
					return null;
				}
				Panel panel = (Panel)VisualTreeHelper.GetChild(itemsPresenter, 0);
				if (panel == null)
				{
					return null;
				}
				_ = panel.Children;
				if (panel is VirtualizingStackPanel virtualizingPanel)
				{
					CallBringIndexIntoView(virtualizingPanel, num);
					if (container.ItemContainerGenerator != null)
					{
						treeViewItem = (TreeViewItem)container.ItemContainerGenerator.ContainerFromIndex(num);
					}
				}
			}
			if (nodes.Count <= 0)
			{
				return treeViewItem;
			}
			if (treeViewItem == null)
			{
				return null;
			}
			return VirtualizingFindItemContainerIterate(treeViewItem, nodes);
		}

							Header = CommonShellResources.NavTreeNode_OpenInNewWindow,

						});

		{

			{
				foreach (object item in e.OldItems)
				{
					FrameworkElement frameworkElement = base.MinimizedItems.Cast<FrameworkElement>().FirstOrDefault((FrameworkElement x) => x.Tag == item);
					if (frameworkElement != null)
					{
						base.MinimizedItems.Remove(frameworkElement);
					}

			{
				foreach (object newItem in e.NewItems)
				{
					base.MinimizedItems.Add(CreateFavoriteElement(newItem as NavigatorTreeNode));
				}
			}
			RefreshFavoritesVisibility();
		}

		private T GetAncestor<T>(DependencyObject d) where T : class
		{
			DependencyObject dependencyObject = d;
			while (true)
			{
				if (dependencyObject == null)
				{
					return null;
				}
				if (dependencyObject == this)
				{
					return null;
				}
				if (typeof(T).IsInstanceOfType(dependencyObject))
				{
					break;
				}
				dependencyObject = VisualTreeHelper.GetParent(dependencyObject);
			}
			return dependencyObject as T;
		}

	public abstract class TreeNavigatorViewModel : INotifyPropertyChanged, INavigatorTreeNodeOwner
	{
		private char _pathSeparator = '/';

		public INavigatorTreeNodeOwner Owner => null;

		public abstract NavigatorTreeNodeCollection Children { get; }

		public abstract NavigatorTreeNode SelectedTreeNode { get; }

		public virtual bool IsFilterVisibleReadOnly => false;

		public virtual IEnumerable<string> FilterOptionsReadOnly => null;

		public virtual string SelectedFilter
		{
			get
			{
				return null;
			}
			set
			{
			}
		}

		public char PathSeparator
		{
			get
			{
				return _pathSeparator;
			}
			set
			{
				_pathSeparator = value;
			}
		}

		public event PropertyChangedEventHandler PropertyChanged;

		protected virtual void OnPropertyChanged(string propertyName)
		{
			if (this.PropertyChanged != null)
			{
				this.PropertyChanged(this, new PropertyChangedEventArgs(propertyName));
			}
		}

		public virtual void NavigateTo(string navigationPath, ShowScreenOptions options)
		{
			if (string.IsNullOrEmpty(navigationPath))
			{
				return;
			}
			NavigatorTreeNode navigatorTreeNode = null;
			NavigatorTreeNodeCollection children = Children;
			string[] array = navigationPath.Split(new char[1] { PathSeparator }, StringSplitOptions.None);
			foreach (string key in array)
			{
				if (navigatorTreeNode != null)
				{
					navigatorTreeNode.IsExpanded = true;
					children = navigatorTreeNode.Children;
				}
				navigatorTreeNode = FindChildByKey(children, key);
				if (navigatorTreeNode == null)
				{
					break;
				}
			}
			if (navigatorTreeNode != null)
			{
				navigatorTreeNode.SetSelectedWithoutSideEffects(value: true);
				navigatorTreeNode.ExecuteShowScreen(options);
			}
		}

		public abstract bool SynchronizeWithScreen(ScreenContentControl screen);

		public virtual void SelectDefaultNode()
		{
			InternalSelectDefaultNode(Children);
		}

		protected static bool InternalSelectDefaultNode(NavigatorTreeNodeCollection children)
		{
			if (children == null)
			{
				return false;
			}
			bool result = false;
			foreach (NavigatorTreeNode child in children)
			{
				if (child.HasScreenToShow())
				{
					child.IsSelected = true;
					result = true;
					break;
				}
				if (InternalSelectDefaultNode(child.Children))
				{
					result = true;
					break;
				}
			}
			return result;
		}

		protected static NavigatorTreeNode FindChildByKey(NavigatorTreeNodeCollection children, string key)
		{
			if (children == null)
			{
				return null;
			}
			foreach (NavigatorTreeNode child in children)
			{
				if (child != null && child.KeyReadOnly == key)
				{
					return child;
				}
			}
			return null;
		}
	}

	public class TreeNavigatorViewTreeViewItem : TreeViewItem
	{
		public static readonly DependencyProperty DataContextIsSelectedProperty = DependencyProperty.Register("DataContextIsSelected", typeof(bool), typeof(TreeNavigatorViewTreeViewItem), new PropertyMetadata(false, OnDataContextIsSelectedPropertyChanged));

		public bool DataContextIsSelected
		{
			get
			{
				return (bool)GetValue(DataContextIsSelectedProperty);
			}
			set
			{
				SetValue(DataContextIsSelectedProperty, value);
			}
		}

		public TreeNavigatorViewTreeViewItem()
		{
			BindingOperations.SetBinding(this, DataContextIsSelectedProperty, new Binding
			{
				Path = new PropertyPath("IsSelected"),
				Mode = BindingMode.TwoWay
			});
			BindingOperations.SetBinding(this, TreeViewItem.IsExpandedProperty, new Binding
			{
				Path = new PropertyPath("IsExpanded"),
				Mode = BindingMode.TwoWay
			});
			BindingOperations.SetBinding(this, UIElement.IsEnabledProperty, new Binding
			{
				Path = new PropertyPath("IsEnabled"),
				Mode = BindingMode.TwoWay
			});
			BindingOperations.SetBinding(this, FrameworkElement.ToolTipProperty, new Binding
			{
				Path = new PropertyPath("TooltipReadOnly"),
				Mode = BindingMode.OneWay
			});
			BindingOperations.SetBinding(this, UIElement.VisibilityProperty, new Binding
			{
				Path = new PropertyPath("IsVisible"),
				Mode = BindingMode.OneWay,
				Converter = new BooleanToVisibilityConverter()
			});
		}

		protected override void OnKeyDown(KeyEventArgs e)
		{
			if (e == null)
			{
				throw new ArgumentNullException("e");
			}
			base.OnKeyDown(e);
			if (e.Key != Key.Return)
			{
				return;
			}
			if (base.IsSelected && DataContextIsSelected)
			{
				if (base.DataContext is NavigatorTreeNode navigatorTreeNode)
				{
					navigatorTreeNode.RefreshScreen();
				}
			}
			else
			{
				DataContextIsSelected = base.IsSelected;
			}
		}

		protected override void OnPropertyChanged(DependencyPropertyChangedEventArgs e)
		{
			base.OnPropertyChanged(e);
			if (e.Property == TreeViewItem.IsSelectedProperty && !base.IsSelected)
			{
				DataContextIsSelected = false;
			}
		}

		private static void OnDataContextIsSelectedPropertyChanged(DependencyObject d, DependencyPropertyChangedEventArgs args)
		{
			if (d is TreeNavigatorViewTreeViewItem treeNavigatorViewTreeViewItem)
			{
				treeNavigatorViewTreeViewItem.OnDataContextIsSelectedChanged((bool)args.OldValue, (bool)args.NewValue);
			}
		}

		protected virtual void OnDataContextIsSelectedChanged(bool oldValue, bool newValue)
		{
			if (oldValue != newValue)
			{
				base.IsSelected = newValue;
			}
		}

		protected override void OnMouseLeftButtonDown(MouseButtonEventArgs e)
		{
			base.OnMouseLeftButtonDown(e);
			if (base.IsSelected && DataContextIsSelected)
			{
				if (base.DataContext is NavigatorTreeNode navigatorTreeNode)
				{
					navigatorTreeNode.RefreshScreen();
				}
			}
			else
			{
				DataContextIsSelected = base.IsSelected;
			}
		}

		protected override DependencyObject GetContainerForItemOverride()
		{
			return new TreeNavigatorViewTreeViewItem();
		}

		protected override bool IsItemItsOwnContainerOverride(object item)
		{
			return item is TreeNavigatorViewTreeViewItem;
		}
	}
