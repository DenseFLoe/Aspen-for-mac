		private void responseCallback(IAsyncResult result)
		{
			try
			{
				_requestState = (RequestState)result.AsyncState;
				HttpWebRequest httpWebRequest = _requestState.request;
				_requestState.response = (HttpWebResponse)httpWebRequest.EndGetResponse(result);
			}
			catch (Exception e)
			{
				_requestState.response = null;
				ShareLog.LogException("Error ending request to server.", e);
			}
			allDone.Set();
		}

		private static bool CheckAspenONEVersion(Version simClientVersion, VersionInfo ServerVersionInfo)
		{
			if (simClientVersion == null || ServerVersionInfo == null || ServerVersionInfo.SystemVersion == null)
			{
				ShareLog.LogWarning("Checking AspenONEVersion Failed: Null Checks Failed");
				return false;
			}
			string systemVersion = ServerVersionInfo.SystemVersion;
			if (Version.TryParse(systemVersion, out var result))
			{
				int num = result.CompareTo(simClientVersion);
				if (num < 0)
				{
					ShareLog.LogWarning("Checking AspenONEVersion Failed: A1 Version->" + result.ToString() + " SimVersion->" + simClientVersion.ToString());
					return false;
				}
				return true;
			}
			ShareLog.LogWarning("Checking AspenONEVersion Failed: A1 Version->" + result.ToString() + " SimVersion->" + simClientVersion.ToString());
			return false;
		}

		private static bool CheckSimulatorVersion(Version simClientVersion, VersionInfo ServerVersionInfo, string appName)
		{
			if (simClientVersion == null || string.IsNullOrWhiteSpace(appName) || ServerVersionInfo == null || ServerVersionInfo.Components == null || ServerVersionInfo.Components.Length == 0)
			{
				ShareLog.LogWarning("Checking SimVersion Failed: Null Checks Failed");
				return false;
			}
			bool flag = false;
			string text = appName.Replace(" ", "").ToUpper();
			ComponentVersionInfo[] components = ServerVersionInfo.Components;
			foreach (ComponentVersionInfo componentVersionInfo in components)
			{
				if (string.IsNullOrWhiteSpace(componentVersionInfo.ResourceType) || !text.Contains(componentVersionInfo.ResourceType.ToUpper()))
				{
					continue;
				}
				string resourceVersion = componentVersionInfo.ResourceVersion;
				if (Version.TryParse(Regex.Replace(resourceVersion, "[^0-9.]", ""), out var result))
				{
					result = new Version(result.Major, (result.Minor >= 0) ? result.Minor : 0, (result.Build >= 0) ? result.Build : 0, (result.Revision >= 0) ? result.Revision : 0);
					int num = result.CompareTo(simClientVersion);
					if (num >= 0)
					{
						flag = true;
						break;
					}
				}
			}
			if (flag)
			{
				return true;
			}
			return false;
		}

		private bool CheckVersion(Version simClientVersion, VersionInfo ServerVersionInfo, string appName)
		{
			ShareLog.LogInfo("Checking Versions...");
			_serverVersionIsValid = CheckAspenONEVersion(simClientVersion, ServerVersionInfo);
			if (_serverVersionIsValid)
			{
				_serverSimVersionIsValid = CheckSimulatorVersion(simClientVersion, ServerVersionInfo, appName);
				return _serverSimVersionIsValid;
			}
			return false;
		}

		protected void connectToServer(object sender, DoWorkEventArgs e)
		{
			if (sender is BackgroundWorker { CancellationPending: false } backgroundWorker)
			{
				string text = e.Argument as string;
				VersionInfo versionInfo = null;
				connectStatus = ConnectStatus.UNKNOWN;
				try
				{
					if (backgroundWorker.CancellationPending)
					{
						e.Cancel = true;
						return;
					}
					using ShareClient shareClient = ShareServiceHelper.CreateClient(Utilities.GetRelativeShareUrl(text));
					versionInfo = shareClient.GetVersion();
					if (backgroundWorker.CancellationPending)
					{
						e.Cancel = true;
						return;
					}
					if (versionInfo != null)
					{
						connectStatus = ConnectStatus.SERVERHASSHARE;
					}
					owner = shareClient.GetUserProfile(WindowsIdentity.GetCurrent().Name.ToLower());
					if (owner != null && (owner.IsAdmin || (owner.Roles != null && owner.Roles.Contains("aspenONESimulationUsers") && owner.Roles.Contains("aspenONE Members"))))
					{
						connectStatus = ConnectStatus.USERHASACCESS;
					}
				}
				catch (EndpointNotFoundException)
				{
					connectionStatusCode = HttpStatusCode.NotFound;
					if (backgroundWorker.CancellationPending)
					{
						e.Cancel = true;
						return;
					}
				}
				catch (FaultException<SecurityAccessDeniedException>)
				{
					connectStatus = ConnectStatus.SERVERHASSHARE;
					ConnectedSuccessfully = false;
					return;
				}
				catch (Exception ex3)
				{
					versionInfo = null;
					if (ex3 is ArgumentException || ex3 is UriFormatException)
					{
						connectStatus = ConnectStatus.URLISMALFORMATTED;
						ConnectedSuccessfully = false;
						return;
					}
					ShareLog.LogException("Error connecting to server.", ex3);
				}
				if (backgroundWorker.CancellationPending)
				{
					e.Cancel = true;
					return;
				}
				if (versionInfo != null)
				{
					ConnectedSuccessfully = CheckVersion(_AppVersion, versionInfo, _AppHostName) && connectStatus == ConnectStatus.USERHASACCESS;
					return;
				}
				ConnectedSuccessfully = false;
				bool flag = false;
				_request = null;
				try
				{
					_request = WebRequest.Create(GetANYHomeURL(text));
					_request.Timeout = 10000;
					_request.Method = "HEAD";
					_request.UseDefaultCredentials = true;
					_requestState.request = request as HttpWebRequest;
				}
				catch (Exception ex4)
				{
					if (ex4 is NotSupportedException || ex4 is ArgumentNullException || ex4 is UriFormatException)
					{
						connectStatus = ConnectStatus.URLISMALFORMATTED;
						return;
					}
					ShareLog.LogException("Unexpected error while testing URL.", ex4);
				}
				try
				{
					if (backgroundWorker.CancellationPending)
					{
						e.Cancel = true;
						return;
					}
					_request.BeginGetResponse(responseCallback, _requestState);
					allDone.WaitOne();
					flag = _requestState.response != null;
				}
				finally
				{
					if (_requestState.response != null)
					{
						_requestState.response.Close();
					}
				}
				if (!flag)
				{
					_request = WebRequest.Create(new Uri(text));
					_request.Method = "HEAD";
					_request.Timeout = 10000;
					_request.UseDefaultCredentials = true;
					_requestState.request = request as HttpWebRequest;
					try
					{
						if (backgroundWorker.CancellationPending)
						{
							e.Cancel = true;
							Monitor.Exit(_requestState);
							return;
						}
						_request.BeginGetResponse(responseCallback, _requestState);
						allDone.WaitOne();
						if (_requestState.response != null)
						{
							connectStatus = ConnectStatus.SERVEREXISTS;
						}
						return;
					}
					catch (Exception e2)
					{
						ShareLog.LogException("Error while testing server status", e2);
						return;
					}
					finally
					{
						if (_requestState.response != null)
						{
							_requestState.response.Close();
						}
					}
				}
				connectStatus = ConnectStatus.SERVERISA1;
			}
			else
			{
				e.Cancel = true;
			}
		}

		private Uri GetANYHomeURL(string urlString)
		{
			Uri result = null;
			if (!string.IsNullOrEmpty(urlString))
			{
				result = new Uri((urlString.EndsWith("/") ? urlString.Substring(0, urlString.Length - 1) : urlString) + "/ANYHome.html");
			}
			return result;
		}

	public class DriveControl : UserControl, IWeakEventListener, IComponentConnector
	{
		private bool _suppressUpdates;

		internal Label ShareStatusLabel;

		internal Label ModelStatusLabel;

		internal Label LastSharedLabel;

		internal Button UploadButton;

		internal Button aspenONEMyDriveButton;

		public DriveControl()
		{
			InitializeComponent();
			base.Loaded += Control_Loaded;
			SessionInfoEventManager.AddListener(SessionInfo.Instance, this);
			UpdateUI();
		}

		private void Control_Loaded(object sender, RoutedEventArgs e)
		{
			UpdateUI();
		}

		private void SessionInfo_MessageReceived(object sender, SessionInfo.SessionMessageArgs e)
		{
			switch (e.Message)
			{
			case "ActionComplete":
				OnActionComplete();
				break;
			case "ConnectedToServerSuccessfully":
			case "ServerURL":
			case "ModelDetails":
			case "ModelModified":
				UpdateUI();
				break;
			case "SimUpdate":
				UpdateUI();
				ShareOpStatusBlock.Visibility = Visibility.Hidden;
				ShareOpStatusImage.Visibility = Visibility.Hidden;
				break;
			}
		}

		private void OnActionComplete()
		{
			try
			{
				_suppressUpdates = true;
				if (SessionInfo.Instance.ConnectedToServerSuccessfully && SessionInfo.Instance.ModelShareStatus == ModelStatus.COMPLETED && SessionInfo.Instance.FileActionType == FileActionType.INGESTION)
				{
					ShareOpStatusImage.Source = StatusImages.greenCheckMark;
					ShareOpStatusImage.RenderTransform = null;
					ShareOpStatusBlock.Foreground = Palette.GreenStatus;
					ShareOpStatusBlock.Text = Strings.DRIVECONTROL_UPLOADED;
					FadeOut(ShareOpStatusBlock, 2);
					FadeOut(ShareOpStatusImage, 2);
				}
				else if (SessionInfo.Instance.ConnectedToServerSuccessfully && SessionInfo.Instance.ModelShareStatus == ModelStatus.PROCESSING && SessionInfo.Instance.FileActionType == FileActionType.INGESTION)
				{
					ShareOpStatusBlock.Text = Strings.SHAREWINDOW_STATUS_QUEUED_INFO;
					ShareOpStatusImage.RenderTransform = null;
					ShareOpStatusBlock.Foreground = Palette.GreenStatus;
					FadeOut(ShareOpStatusImage, 2);
				}
				else if (SessionInfo.Instance.FileActionType == FileActionType.INGESTION)
				{
					ShareOpStatusImage.Source = StatusImages.redX;
					ShareOpStatusImage.RenderTransform = null;
					ShareOpStatusBlock.Foreground = Palette.RedStatus;
					ShareOpStatusBlock.Text = Strings.DRIVECONTROL_UNABLE_TO_UPLOAD;
				}
				else
				{
					ShareOpStatusImage.Source = StatusImages.greenCheckMark;
					ShareOpStatusImage.RenderTransform = null;
					ShareOpStatusBlock.Foreground = Palette.GreenStatus;
					ShareOpStatusBlock.Text = Strings.DRIVECONTROL_REMOVED;
					FadeOut(ShareOpStatusBlock, 2);
					FadeOut(ShareOpStatusImage, 2);
				}
			}
			finally
			{
				_suppressUpdates = false;
			}
			UpdateUI();
		}

		private void UpdateUI()
		{
			if (_suppressUpdates)
			{
				return;
			}
			aspenONEMyDriveButton.IsEnabled = SessionInfo.Instance.ConnectedToServerSuccessfully;
			if (string.IsNullOrWhiteSpace(SessionInfo.Instance.ActiveDocumentFilePath))
			{
				ShareStatusLabel.Content = Strings.SHAREWINDOW_OPEN_A_MODEL;
				LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				ModelStatusLabel.Content = string.Empty;
				UploadButton.IsEnabled = false;
				RemoveButton.IsEnabled = false;
			}
			else if (!SessionInfo.Instance.FileExtensionIsValid)
			{
				if (SessionInfo.Instance.HostApplicationName == "Aspen Plus")
				{
					ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_UNSUPPORTED_FILE_APLUS;
				}
				else if (SessionInfo.Instance.HostApplicationName == "Aspen HYSYS")
				{
					ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_UNSUPPORTED_FILE_HYSYS;
				}
				LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				ModelStatusLabel.Content = string.Empty;
				UploadButton.IsEnabled = false;
				RemoveButton.IsEnabled = false;
			}
			else if (!SessionInfo.Instance.ConnectedToServerSuccessfully)
			{
				ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_NOT_CONNECTED;
				LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				ModelStatusLabel.Content = string.Empty;
				UploadButton.IsEnabled = false;
				aspenONEMyDriveButton.IsEnabled = false;
				RemoveButton.IsEnabled = false;
			}
			else if (SessionInfo.Instance.ModelShareStatus == ModelStatus.PROCESSING)
			{
				RemoveButton.IsEnabled = false;
				UploadButton.IsEnabled = true;
				ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_QUEUED;
			}
			else if (SessionInfo.Instance.ModelHasBeenShared)
			{
				ShareStatusLabel.Content = Strings.DRIVECONTROL_UPLOADED;
				RemoveButton.IsEnabled = true;
				if (!SessionInfo.Instance.ModelShareTime.HasValue)
				{
					LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				}
				else if (SessionInfo.Instance.ModelShareTime == DateTime.MinValue)
				{
					LastSharedLabel.Content = DateTime.Now.ToString("MMMM dd, yyyy") + " - " + DateTime.Now.ToShortTimeString();
				}
				else
				{
					LastSharedLabel.Content = SessionInfo.Instance.ModelShareTime.Value.ToString("MMMM dd, yyyy") + " - " + SessionInfo.Instance.ModelShareTime.Value.ToShortTimeString();
				}
				if (SessionInfo.Instance.ModelModified)
				{
					UploadButton.IsEnabled = true;
					ModelStatusLabel.Content = Strings.SHAREWINDOW_STATUS_MODIFIED;
				}
				else
				{
					UploadButton.IsEnabled = false;
					ModelStatusLabel.Content = string.Empty;
				}
			}
			else
			{
				RemoveButton.IsEnabled = false;
				UploadButton.IsEnabled = true;
				LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				ModelStatusLabel.Content = string.Empty;
				ShareStatusLabel.Content = Strings.DRIVECONTROL_NOT_UPLOADED;
			}
		}

			}
		}

		}

		private static void FadeOut(UIElement control, int fadeDuration)
		{
			DoubleAnimation doubleAnimation = new DoubleAnimation();

	internal class PendingProfileParser
	{
		public delegate void PendingTextChangedEvent();

		public delegate void PendingProfileListChangedEvent();

		public delegate void FullTextChangedEvent();

		public string DefaultText = Strings.SHAREWINDOW_DEFAULT_EMAIL_TEXT;

		private string _fullText = string.Empty;

		private BindingList<Profile> _pendingProfiles = new BindingList<Profile>();

		private List<Profile> _knownProfiles = new List<Profile>();

		private Dictionary<Profile, bool> _dicSelectedProfiles = new Dictionary<Profile, bool>();

		private List<Profile> _selectedProfiles = new List<Profile>();

		private string _pendingText = string.Empty;

		public bool IsClean
		{
			get
			{
				if (string.IsNullOrWhiteSpace(PendingText))
				{
					return true;
				}
				return false;
			}
		}

		public string FullText
		{
			get
			{
				return _fullText;
			}
			set
			{
				if (_fullText != value)
				{
					_fullText = value;
					ParseFullText();
					OnFullTextChanged();
				}
			}
		}

		public BindingList<Profile> PendingProfiles => _pendingProfiles;

		public List<Profile> KnownProfiles => _knownProfiles;

		public List<Profile> SelectedProfiles => _selectedProfiles;

		public string PendingText
		{
			get
			{
				return _pendingText;
			}
			private set
			{
				if (_pendingText != value)
				{
					_pendingText = value;
					OnPendingTextChanged();
				}
			}
		}

		public event PendingTextChangedEvent PendingTextChanged;

		public event PendingProfileListChangedEvent PendingProfileListChanged;

		public event FullTextChangedEvent FullTextChanged;

		public void WipeOut()
		{
			PendingProfiles.Clear();
			KnownProfiles.Clear();
			_dicSelectedProfiles.Clear();
			SelectedProfiles.Clear();
			FullText = DefaultText;
		}

		public void Clear()
		{
			PendingProfiles.Clear();
			_dicSelectedProfiles.Clear();
			SelectedProfiles.Clear();
			FullText = DefaultText;
		}

		private void ResetSelectedProfileDictionary()
		{
			_dicSelectedProfiles = new Dictionary<Profile, bool>();
			foreach (Profile selectedProfile in SelectedProfiles)
			{
				_dicSelectedProfiles.Add(selectedProfile, value: false);
			}
		}

		private void ParseFullText()
		{
			if (string.IsNullOrWhiteSpace(_fullText))
			{
				_pendingProfiles.Clear();
				PendingText = string.Empty;
				return;
			}
			List<string> list = new List<string>();
			string[] array = _fullText.Split(';');
			if (array != null && array.Length > 0)
			{
				for (int i = 0; i < array.Length; i++)
				{
					if (i < array.Length - 1)
					{
						list.Add(array[i].Trim());
					}
					else
					{
						list.Add(array[i].TrimStart());
					}
				}
			}
			_pendingProfiles.Clear();
			ResetSelectedProfileDictionary();
			List<string> list2 = new List<string>();
			int num = 0;
			foreach (string item in list)
			{
				if (num != list.Count - 1)
				{
					VerifyText(item);
				}
				list2.Add(item);
				num++;
			}
			_fullText = string.Empty;
			num = 0;
			foreach (string item2 in list2)
			{
				if (num < list2.Count - 1)
				{
					_fullText = _fullText + item2 + "; ";
				}
				else
				{
					_fullText += list2[num].TrimStart(' ');
				}
				num++;
			}
			if (list2.Count == 0)
			{
				PendingText = string.Empty;
			}
			else
			{
				PendingText = list2[list2.Count - 1];
			}
		}

		private bool VerifyText(string text)
		{
			if (!string.IsNullOrWhiteSpace(text))
			{
				text = text.TrimEnd();
			}
			if (string.IsNullOrWhiteSpace(text))
			{
				return false;
			}
			foreach (KeyValuePair<Profile, bool> dicSelectedProfile in _dicSelectedProfiles)
			{
				if (dicSelectedProfile.Value)
				{
					continue;
				}
				Profile key = dicSelectedProfile.Key;
				if (key != null && (string.Equals(key.FullName, text, StringComparison.CurrentCultureIgnoreCase) || string.Equals(key.EmailAddress, text, StringComparison.CurrentCultureIgnoreCase)))
				{
					if (!_pendingProfiles.Contains(key))
					{
						_pendingProfiles.Add(key);
					}
					_dicSelectedProfiles[dicSelectedProfile.Key] = true;
					return true;
				}
			}
			foreach (Profile knownProfile in _knownProfiles)

		private string _extractionErrorMessages = string.Empty;

		private bool Success = true;

		private string _tempDirectory = string.Empty;

		public string PathToPng { get; private set; }

		public event ExtractionFailedEventHandler ExtractionFailed;

		public event ExtractionCompleteEventHandler ExtractionComplete;

		private void BeginExtraction(object sender, DoWorkEventArgs e)
		{
			//IL_003b: Unknown result type (might be due to invalid IL or missing references)
			//IL_0041: Expected O, but got Unknown
			if (string.IsNullOrWhiteSpace(FullCaseFilePath))
			{
				return;
			}
			try
			{
				_extractionErrorMessages = string.Empty;
				string location = Program.GetLocation();
				string text = System.IO.Path.GetDirectoryName(location) + "\\";
				MetadataSDK val = new MetadataSDK(901000, 901000, text);
				val.Initialize(FullCaseFilePath);
				string fileNameWithoutExtension = System.IO.Path.GetFileNameWithoutExtension(FullCaseFilePath);
				string tempPath = System.IO.Path.GetTempPath();
				string text2 = System.IO.Path.Combine(tempPath, fileNameWithoutExtension + ".png");
				if (File.Exists(text2))
				{
					File.Delete(text2);
				}
				bool flag = val.ExtractPFD(1536, 1152, text2);
				if (flag)
				{
					PathToPng = text2;
				}
				Success = flag;
			}
			catch (Exception ex)
			{
				Success = false;
				ShareLog.LogException("Preview Extraction Failed", ex);
				_extractionErrorMessages = ex.Message;
			}
		}

		private void ExtractionCompleted(object sender, RunWorkerCompletedEventArgs e)
		{
			if (Success)
			{
				OnExtractionComplete();
			}
			else
			{
				OnExtractionFailed(_extractionErrorMessages);
			}
			if (string.IsNullOrWhiteSpace(_tempDirectory))
			{
				return;
			}
			try
			{
				bool flag = false;
				int num = 0;
				while (!flag)
				{
					try
					{
						Directory.Delete(_tempDirectory, recursive: true);
						flag = true;
					}
					catch (IOException)
					{
						if (num > 50)
						{
							break;
						}
						Thread.Sleep(50);
					}
					num++;
				}
			}
			catch (Exception)
			{
			}
			finally
			{
				_tempDirectory = string.Empty;
			}
		}

		private void OnExtractionFailed(string message)
		{
			if (this.ExtractionFailed != null)
			{
				this.ExtractionFailed(message);
			}
		}

		private void OnExtractionComplete()
		{
			if (this.ExtractionComplete != null)
			{
				this.ExtractionComplete();
			}
		}

		private int GetVersionNumber(string fileName)
		{
			if (File.Exists(fileName))
			{
				using (FileStream fileStream = new FileStream(fileName, FileMode.Open, FileAccess.Read, FileShare.Read))
				{
					BinaryReader binaryReader = new BinaryReader(fileStream);
					fileStream.Position = 22L;
					return binaryReader.ReadInt32();
				}
			}
			return 0;
		}

		public void Execute()
		{
			if (string.IsNullOrWhiteSpace(FullCaseFilePath))
			{
				return;
			}
			switch (System.IO.Path.GetExtension(FullCaseFilePath))
			{
			case ".hsc":
				if (GetVersionNumber(FullCaseFilePath) < 8615)
				{
					OnExtractionFailed(Strings.PREVIEWSCREENCONTROLLER_VERSION);
					return;
				}
				break;
			case ".apwz":
			case ".hscz":
				try
				{
					_tempDirectory = Utilities.GenerateZipTempDirectory();
					FullCaseFilePath = Utilities.GetSimulationFileFromZip(FullCaseFilePath, _tempDirectory);
				}
				catch (Exception ex)
				{
					OnExtractionFailed("Preview failed to create. " + ex.Message);
					return;
				}
				break;
			}
			Success = false;
			BackgroundWorker backgroundWorker = new BackgroundWorker();
			backgroundWorker.DoWork += BeginExtraction;
			backgroundWorker.RunWorkerCompleted += ExtractionCompleted;
			backgroundWorker.RunWorkerAsync();
		}
	}

					_popUp.DataGridPopUp.SelectedIndex++;
				}
				break;
			case Key.Up:
				if (_popUp.DataGridPopUp.Items != null && _popUp.DataGridPopUp.Items.Count > _popUp.DataGridPopUp.SelectedIndex - 1 && _popUp.DataGridPopUp.SelectedIndex - 1 > -1)
				{
					_popUp.DataGridPopUp.SelectedIndex--;
				}
				break;
			case Key.Tab:
			case Key.Return:
				if (_popUp.IsOpen && _popUp.DataGridPopUp.SelectedIndex != -1)
				{
					_popUp.AddCurrentSelection();
					e.Handled = true;
				}
				break;
			}
			if (_popUp.DataGridPopUp.SelectedIndex != -1)
			{
				_popUp.DataGridPopUp.ScrollIntoView(_popUp.DataGridPopUp.SelectedItem);
			}
		}

		{
			GetProfilesFromCore(Environment.UserName);
		}

		private static void requestUsernames(object sender, DoWorkEventArgs e)
		{
			List<Profile> list = new List<Profile>();
			try
			{
				object[] array = e.Argument as object[];
				string text = array[0] as string;
				if (string.IsNullOrEmpty(text))
				{
					return;
				}
				string term = array[1] as string;
				using ShareClient shareClient = ShareServiceHelper.CreateClient(text);
				ShareUser[] array2 = shareClient.QueryUserName(term);
				ShareUser[] array3 = array2;
				foreach (ShareUser shareUser in array3)
				{
					string[] array4 = shareUser.Account.Split('\\');
					list.Add(new Profile(array4[0], array4[1], shareUser.EMail, shareUser.DisplayName));
				}
			}
			catch (Exception e2)
			{
				ShareLog.LogException("Error requesting usernames.", e2);
			}
			e.Result = list;
		}

		private void ProcessProfilesFromCore(object sender, RunWorkerCompletedEventArgs e)
		{
			_availableProfiles = (e.Result as List<Profile>) ?? new List<Profile>();
			foreach (Profile availableProfile in _availableProfiles)
			{
				if (!PendingEmailTextParser.KnownProfiles.Contains(availableProfile))
				{
					PendingEmailTextParser.KnownProfiles.Add(availableProfile);
				}
				if (_activeUserProfile == null && availableProfile.UserName.ToUpper() == Environment.UserName.ToUpper() && availableProfile.Domain.ToUpper() == Environment.UserDomainName.ToUpper())
				{
					_activeUserProfile = availableProfile;
				}
				string[] filterableFields = availableProfile.GetFilterableFields();
				foreach (string text in filterableFields)
				{
					if (text != null)
					{
						if (!_profileMap.ContainsKey(text))
						{
							_profileMap.Add(text, new List<Profile>(new Profile[1] { availableProfile }));
						}
						else
						{
							_profileMap[text].Add(availableProfile);
						}
					}
				}
			}
			if (_activeUserProfile != null)
			{
				SessionInfo.Instance.ActiveUserProfile = _activeUserProfile;
			}
			PopulateTypeAhead(PendingEmailTextParser.PendingText);
		}

		private void GetProfilesFromCore(string queryTerm)
		{
			_availableProfiles.Clear();
			_profileMap.Clear();
			string relativeServerURL = SessionInfo.Instance.RelativeServerURL;
			BackgroundWorker backgroundWorker = new BackgroundWorker();
			backgroundWorker.DoWork += requestUsernames;
			backgroundWorker.RunWorkerCompleted += ProcessProfilesFromCore;
			backgroundWorker.RunWorkerAsync(new object[2] { relativeServerURL, queryTerm });
		}

		private void PopulateTypeAhead(string userInput)
		{
			TypeAheadContents.Clear();
			TypeAheadContents = _availableProfiles;
			if (userInput != PendingEmailTextParser.DefaultText)

				_popUp.SetItemsSource(TypeAheadContents);
				_popUp.Placement = PlacementMode.Bottom;
				_popUp.PlacementTarget = this;
				_popUp.IsOpen = true;
			}
			else
			{
				_popUp.IsOpen = false;
			}
		}

			{
				_popUp.IsOpen = false;
				_popUp.IsOpen = true;
			}
		}

			}
		}

		private void ProcessModelDetails()
		{
			_modelShareStatus = ((LatestDetail != null) ? LatestDetail.Status : ModelStatus.NOTSTARTED);
			if (ModelShareStatus == ModelStatus.NOTSTARTED)
			{
				_modelHasBeenShared = false;
				_modelShareTime = null;
				_deepLink = null;
				_modelModified = false;
				_sharedWithProfiles.Clear();
				return;
			}
			_modelShareTime = LatestDetail.SharedTime.ToLocalTime();
			_modelHasBeenShared = true;
			_deepLink = LatestDetail.DeepLink;
			_shareUsersCanDownload = LatestDetail.IsDownloadable;
			if (!_modelModified)
			{
				_modelModified = Math.Abs((LatestDetail.LastModified.ToLocalTime() - File.GetLastWriteTime(ActiveDocumentFilePath)).TotalMilliseconds) > 1.0;
			}
			if (LatestDetail.SharedUsers != null)
			{
				BindingList<Profile> bindingList = new BindingList<Profile>();
				ShareUser[] sharedUsers = LatestDetail.SharedUsers;
				foreach (ShareUser shareUser in sharedUsers)
				{
					string[] array = shareUser.Account.Split('\\');
					bindingList.Add(new Profile(array[0], array[1], shareUser.EMail, shareUser.DisplayName));
				}
				_sharedWithProfiles = bindingList;
			}
		}

		private void UpdateModelDetails(string message = "")
		{
			if (_connectedToServerSuccessfully)
			{
				BackgroundWorker backgroundWorker = new BackgroundWorker();
				backgroundWorker.DoWork += delegate(object sender, DoWorkEventArgs args)
				{
					args.Result = Utilities.RequestModelDetail(args.Argument as SessionInfo);
				};
				backgroundWorker.RunWorkerCompleted += delegate(object sender, RunWorkerCompletedEventArgs args)
				{
					_latestDetail = args.Result as ModelDetail;
					ProcessModelDetails();
					if (string.IsNullOrWhiteSpace(message))
					{
						SendMessage("ModelDetails");
					}
					else
					{
						SendMessage(message);
					}
				};
				backgroundWorker.RunWorkerAsync(this);
			}
			else
			{
				SendMessage("ModelDetails");
			}
		}

		private bool IsPreviewUpToDate()
		{
			bool flag = true;
			DateTime dateTime = (string.IsNullOrWhiteSpace(_fullCaseFilePath) ? DateTime.MinValue : File.GetLastWriteTimeUtc(_fullCaseFilePath));
			if (_CaseFileLastWriteTime.Key == _fullCaseFilePath)
			{
				if (_CaseFileLastWriteTime.Value != dateTime)
				{
					flag = false;
				}
			}
			else
			{
				flag = false;
			}
			if (!flag)
			{
				_CaseFileLastWriteTime = new KeyValuePair<string, DateTime>(_fullCaseFilePath, dateTime);
			}
			return flag;
		}

		private void triggerAsyncConnect()
		{
			if (connectWorker.IsBusy)
			{
				connectWorker.CancelAsync();
				if (base.requestState != null && base.requestState.request != null)

					base.requestState.request.Abort();

		private void afterConnectToServer(object sender, RunWorkerCompletedEventArgs e)
		{
			ServerNameTextBox.IsEnabled = true;
			StatusImage.RenderTransform = null;
			if (!e.Cancelled && (e.Error != null || !ConnectedSuccessfully))
			{
				switch (connectStatus)
				{
				case ConnectStatus.URLISMALFORMATTED:
					base.CurrentStatusMessage = Strings.URLVALIDATOR_MALFORMATTED_URL;
					break;
				case ConnectStatus.USERHASACCESS:
					if (!_serverVersionIsValid)
					{
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_BAD_VERSION;
					}
					else if (!_serverSimVersionIsValid)
					{
						base.CurrentStatusMessage = SessionInfo.Instance.HostApplicationName + " " + Strings.URLVALIDATOR_CANNOT_CONNECT_BAD_SIMVERSION;
					}
					else
					{
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_DEFAULT;
					}
					break;
				case ConnectStatus.SERVERHASSHARE:
					base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_UNAUTHORIZED;
					break;
				case ConnectStatus.SERVERISA1:
					base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_NO_SHARE;
					break;
				case ConnectStatus.SERVEREXISTS:
					base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_NOT_A1_SERVER;
					break;
				default:
					switch (connectionStatusCode)
					{
					case HttpStatusCode.NotFound:
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_SERVER_NOT_FOUND;
						break;
					case HttpStatusCode.Unauthorized:
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_UNAUTHORIZED;
						break;
					default:
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_DEFAULT;
						break;
					}
					break;
				}
				SessionInfo.Instance.Owner = null;
				SessionInfo.Instance.ServerURL = base.CurrentURL;
				SessionInfo.Instance.ActiveUserProfile = null;
				SessionInfo.Instance.ConnectedToServerSuccessfully = false;
				base.CurrentStatusColor = Palette.RedStatus;
				base.CurrentImage = StatusImages.redX;
				currentMode = mode.Editing;
			}
			else if (!e.Cancelled)
			{
				base.CurrentStatusMessage = Strings.URLVALIDATOR_CONNECTED;
				string[] array = base.Owner.Account.Split('\\');
				SessionInfo.Instance.ActiveUserProfile = new Profile((array.Length > 0) ? array[0] : string.Empty, (array.Length > 1) ? array[1] : string.Empty, base.Owner.EMail, base.Owner.DisplayName);
				SessionInfo.Instance.Owner = base.Owner;
				SessionInfo.Instance.ServerURL = base.CurrentURL;
				SessionInfo.Instance.ConnectedToServerSuccessfully = true;
				base.CurrentStatusColor = Palette.GreenStatus;
				base.CurrentImage = StatusImages.greenCheckMark;
				ShareConfiguration.Set(ConfigurationSetting.URLSetting, base.CurrentURL, trySetDefault: true);
				fadeOutLabel(3);
				currentMode = mode.Connected;
			}
			else if (e.Cancelled)
			{
				base.CurrentStatusMessage = "";
				stackPanelStatus.Visibility = Visibility.Hidden;
			}
			buttonConnect.IsEnabled = true;
			if (waitForWorkerToFinish)
			{
				waitForWorkerToFinish = false;
				triggerAsyncConnect();
			}
		}

		private static string Unzip(string zipToUnpack, string unzipDirectory)
		{
			string empty = string.Empty;
			int num = 0;
			if (zipToUnpack.IndexOf(".zip", StringComparison.OrdinalIgnoreCase) > -1)
			{
				num = zipToUnpack.IndexOf(".zip/", StringComparison.OrdinalIgnoreCase) + 4;
			}
			if (num == 0 && zipToUnpack.IndexOf(".apwz", StringComparison.OrdinalIgnoreCase) > -1)
			{
				num = zipToUnpack.IndexOf(".apwz/", StringComparison.OrdinalIgnoreCase) + 5;
			}
			if (num == 0 && zipToUnpack.IndexOf(".hscz", StringComparison.OrdinalIgnoreCase) > -1)
			{
				num = zipToUnpack.IndexOf(".hscz/", StringComparison.OrdinalIgnoreCase) + 5;
			}
			string result = string.Empty;
			ZipFile val = ZipFile.Read(zipToUnpack);
			try
			{
				int num2 = 0;
				string text = System.IO.Path.GetExtension(zipToUnpack).ToLower();
				for (int i = 0; i < val.Count; i++)
				{
					ZipEntry val2 = val[i];
					if (val2.FileName.EndsWith("/"))
					{
						continue;
					}
					string text2 = System.IO.Path.GetExtension(val2.FileName).ToLower();
					if (text.Equals(".apwz") || text.Equals(".hscz"))
					{
						if (!text2.Equals(".bkp") && !text2.Equals(".apw") && !text2.Equals(".hsc"))
						{
							continue;
						}
						_ = val2.FileName;
						int num3 = val2.FileName.LastIndexOf("/");
						string empty2 = string.Empty;
						if (num3 > 0)
						{
							empty2 = val2.FileName.Remove(0, num3);
							val2.FileName = num2 + empty2;
							num2++;
						}
					}
					else if (text.Equals(".zip") && empty != val2.FileName)
					{
						continue;
					}
					string text3 = val2.FileName.Replace("/", "\\");
					result = ((!text3.StartsWith("\\")) ? (unzipDirectory + "\\" + text3) : (unzipDirectory + text3));
					val2.Extract(unzipDirectory, (ExtractExistingFileAction)1);
				}
				return result;
			}
			finally
			{
				((IDisposable)val)?.Dispose();
			}
		}

		private static ModelStatus pollModelStatus(ModelStatus desiredStatus, SessionInfo caseInfo)
		{
			ModelStatus modelStatus = RequestModelStatus(caseInfo.RelativeServerURL, caseInfo.ActiveDocumentFilePath, caseInfo.Owner);
			for (int i = 0; i < 10; i++)
			{

				{
					break;
				}
				Thread.Sleep(5000);
				modelStatus = RequestModelStatus(caseInfo.RelativeServerURL, caseInfo.ActiveDocumentFilePath, caseInfo.Owner);
				if (modelStatus == ModelStatus.ERROR || modelStatus == ModelStatus.INTERRUPTED)
				{
					break;
				}

		}

		private static void sendFile(object sender, DoWorkEventArgs e)
		{
			if (!(e.Argument is SessionInfo { Owner: var owner, AddedProfiles: var addedProfiles, ActiveDocumentFilePath: var activeDocumentFilePath } sessionInfo))
			{
				return;
			}
			string text = sessionInfo.ActiveUserProfile.FullName + " has shared the model \"" + sessionInfo.ActiveDocumentName + "\" with you.";
			if (sessionInfo.OptionalEmailContent != Strings.SHAREWINDOW_EMAIL_BODY_INSTRUCTIONS)
			{
				text = text + Environment.NewLine + Environment.NewLine + sessionInfo.OptionalEmailContent;
			}
			text = text.Replace("<", "&lt;");
			text = text.Replace(">", "&gt;");
			text = text.Replace(Environment.NewLine, "<br/>");
			string relativeServerURL = sessionInfo.RelativeServerURL;
			List<ShareUser> list = new List<ShareUser>();
			foreach (Profile item in addedProfiles)
			{
				if (item.IsValid)
				{
					list.Add(new ShareUser
					{
						Account = item.Domain + "\\" + item.UserName,
						EMail = (string.IsNullOrWhiteSpace(item.EmailAddress) ? "NULL" : item.EmailAddress),
						DisplayName = item.FullName
					});
				}
			}
			FileInfo fileInfo = new FileInfo(activeDocumentFilePath);
			using ShareClient shareClient = ShareServiceHelper.CreateClient(relativeServerURL);
			try
			{
				ShareUser shareUser = new ShareUser();
				shareUser.Account = owner.Account;
				shareUser.DisplayName = owner.DisplayName;
				shareUser.EMail = owner.EMail;
				ShareUser owner2 = shareUser;
				bool shareUsersCanDownload = SessionInfo.Instance.ShareUsersCanDownload;
				if (shareClient.PublishModel(shareUsersCanDownload, System.IO.Path.GetFileName(activeDocumentFilePath), owner2, fileInfo.CreationTime, fileInfo.LastWriteTime, list.ToArray(), text, File.ReadAllBytes(activeDocumentFilePath)))
				{
					e.Result = new object[2]
					{
						ModelStatus.COMPLETED,
						sessionInfo
					};
				}
				else
				{
					e.Result = new object[2]
					{
						ModelStatus.ERROR,
						sessionInfo
					};
				}
			}
			catch (FaultException<SecurityAccessDeniedException>)
			{
				e.Result = new object[2]
				{
					ModelStatus.ERROR,
					sessionInfo
				};
			}
		}

		public static bool VerifyExtension(string ActiveDocumentName)
		{
			if (string.IsNullOrWhiteSpace(ActiveDocumentName))
			{
				return false;
			}
			string hostApplicationName = SessionInfo.Instance.HostApplicationName;
			if (string.IsNullOrWhiteSpace(hostApplicationName))
			{
				return false;
			}
			bool result = false;
			string empty = string.Empty;
			try
			{
				empty = System.IO.Path.GetExtension(ActiveDocumentName);
			}
			catch (ArgumentException)
			{
				return false;
			}
			if (string.Compare("Aspen Plus", hostApplicationName, StringComparison.OrdinalIgnoreCase) == 0)
			{
				result = _validAspenPlusExtensions.Contains(empty);
			}
			else if (string.Compare("Aspen HYSYS", hostApplicationName, StringComparison.OrdinalIgnoreCase) == 0)
			{
				result = _validHYSYSExtensions.Contains(empty);
			}
			return result;
		}

			{
			}
			return (urlString.EndsWith("/") ? urlString.Substring(0, urlString.Length - 1) : urlString) + "/" + RelativeShareUrl;
		}

			return Visibility.Hidden;

		}

		{
			switch (Action)
			{
			case ShareProperty.IsDirty:
				_IsDirty = (bool)value;
				break;
			case ShareProperty.ActiveDocumentName:
				_ActiveDocumentName = (string)value;
				break;
			case ShareProperty.ActiveDocumentFilePath:
				_ActiveDocumentFilePath = (string)value;
				break;
			case ShareProperty.HostApplicationName:
				_HostApplicationName = (string)value;
				break;
			case ShareProperty.ApplicationVersion:
				_applicationVersion = value as Version;
				if (_applicationVersion == null)
				{
					_applicationVersion = Assembly.GetExecutingAssembly().GetName().Version;
				}
				break;
			case ShareProperty.DoSave:
				ActionComplete = true;
				break;
			case ShareProperty.IsCompoundFileFormat:
			case ShareProperty.CompoundFilePath:
				break;
			}
		}

		public IntroURLValidator()
		{
			InitializeComponent();
			ServerNameTextBox.Text = "";
			connectWorker = new BackgroundWorker();
			connectWorker.DoWork += base.connectToServer;
			connectWorker.RunWorkerCompleted += afterConnectToServer;
			connectWorker.WorkerSupportsCancellation = true;
		}

		private void CancelWorker()
		{
			connectWorker.CancelAsync();
			connectWorker.DoWork -= base.connectToServer;
			connectWorker.RunWorkerCompleted -= afterConnectToServer;
			connectWorker = null;
			connectWorker = new BackgroundWorker();
			connectWorker.DoWork += base.connectToServer;
			connectWorker.RunWorkerCompleted += afterConnectToServer;
			connectWorker.WorkerSupportsCancellation = true;
		}

		private void buttonConnect_Click(object sender, RoutedEventArgs e)
		{
			switch (currentMode)
			{
			case mode.Editing:
				triggerAsyncConnect();
				currentMode = mode.Connecting;
				break;
			case mode.Connecting:
				CancelWorker();
				if (base.requestState != null && base.requestState.request != null)
				{
					base.requestState.request.Abort();
				}
				currentMode = mode.Editing;
				stackPanelStatus.Visibility = Visibility.Hidden;
				ServerNameTextBox.IsEnabled = true;
				break;
			}
		}

		private void triggerAsyncConnect()
		{
			if (connectWorker.IsBusy)
			{
				connectWorker.CancelAsync();
				if (base.requestState != null && base.requestState.request != null)
				{
					base.requestState.request.Abort();
				}
				waitForWorkerToFinish = true;
			}
			else
			{
				_AppVersion = (SessionInfo.Instance.IsLoaded ? SessionInfo.Instance.ApplicationVersion : null);
				_AppHostName = (SessionInfo.Instance.IsLoaded ? SessionInfo.Instance.HostApplicationName : null);
				ServerNameTextBox.IsEnabled = false;
				base.CurrentImage = StatusImages.spinner;
				base.CurrentStatusMessage = Strings.URLVALIDATOR_CONNECTING;
				base.CurrentStatusColor = Brushes.Gray;
				StatusImages.Spin(StatusImage);
				stackPanelStatus.Visibility = Visibility.Visible;
				connectWorker.RunWorkerAsync(base.CurrentURL);
			}
		}

		private void afterConnectToServer(object sender, RunWorkerCompletedEventArgs e)
		{
			ServerNameTextBox.IsEnabled = true;
			StatusImage.RenderTransform = null;
			if (!e.Cancelled && (e.Error != null || !ConnectedSuccessfully))
			{
				switch (connectStatus)
				{
				case ConnectStatus.URLISMALFORMATTED:
					base.CurrentStatusMessage = Strings.URLVALIDATOR_MALFORMATTED_URL;
					break;
				case ConnectStatus.USERHASACCESS:
					if (!_serverVersionIsValid)
					{
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_BAD_VERSION;
					}
					else if (!_serverSimVersionIsValid)
					{
						base.CurrentStatusMessage = SessionInfo.Instance.HostApplicationName + " " + Strings.URLVALIDATOR_CANNOT_CONNECT_BAD_SIMVERSION;
					}
					else
					{
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_DEFAULT;
					}
					break;
				case ConnectStatus.SERVERHASSHARE:
					base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_UNAUTHORIZED;
					break;
				case ConnectStatus.SERVERISA1:
					base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_NO_SHARE;
					break;
				case ConnectStatus.SERVEREXISTS:
					base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_NOT_A1_SERVER;
					break;
				default:
					switch (connectionStatusCode)
					{
					case HttpStatusCode.NotFound:
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_SERVER_NOT_FOUND;
						break;
					case HttpStatusCode.Unauthorized:
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_UNAUTHORIZED;
						break;
					default:
						base.CurrentStatusMessage = Strings.URLVALIDATOR_CANNOT_CONNECT_DEFAULT;
						break;
					}
					break;
				}
				SessionInfo.Instance.Owner = null;
				SessionInfo.Instance.ServerURL = base.CurrentURL;
				SessionInfo.Instance.ConnectedToServerSuccessfully = false;
				base.CurrentStatusColor = Palette.RedStatus;
				base.CurrentImage = StatusImages.redX;
				currentMode = mode.Editing;
			}
			else if (!e.Cancelled)
			{
				base.CurrentStatusMessage = Strings.URLVALIDATOR_CONNECTED;
				string[] array = base.Owner.Account.Split('\\');
				SessionInfo.Instance.ActiveUserProfile = new Profile((array.Length > 0) ? array[0] : string.Empty, (array.Length > 1) ? array[1] : string.Empty, base.Owner.EMail, base.Owner.DisplayName);
				SessionInfo.Instance.Owner = base.Owner;
				SessionInfo.Instance.ServerURL = base.CurrentURL;
				SessionInfo.Instance.ConnectedToServerSuccessfully = true;
				base.CurrentStatusColor = Palette.GreenStatus;
				base.CurrentImage = StatusImages.greenCheckMark;
				ShareConfiguration.Set(ConfigurationSetting.URLSetting, base.CurrentURL, trySetDefault: true);
			}
			else if (e.Cancelled)
			{
				base.CurrentStatusMessage = "";
				stackPanelStatus.Visibility = Visibility.Hidden;
			}
			buttonConnect.IsEnabled = true;
			if (waitForWorkerToFinish)
			{
				waitForWorkerToFinish = false;
				triggerAsyncConnect();
			}
		}

			MessageBoxResult messageBoxResult = MessageBox.Show(Strings.SHAREWINDOW_DIALOG_REMOVE_ALL, SessionInfo.Instance.HostApplicationName, MessageBoxButton.OKCancel, MessageBoxImage.None, MessageBoxResult.No);
			if (messageBoxResult == MessageBoxResult.OK)
			{
				ClearProfiles();
			}
		}

		private void ClearProfiles()
		{
			AddedProfiles.Clear();
			SharedUsersDataGrid.ItemsSource = null;
			if (SessionInfo.Instance.ActiveUserProfile != null)
			{
				AddedProfiles.Add(SessionInfo.Instance.ActiveUserProfile);
			}
			SharedUsersDataGrid.ItemsSource = AddedProfiles;
		}

		private void PendingProfiles_ListChanged(object sender, ListChangedEventArgs e)
		{
			if (EmailTextBox == null || EmailTextBox.Profiles == null)
			{
				AddButton.IsEnabled = false;
			}
			else if (EmailTextBox.Profiles.Count > 0)
			{
				AddButton.IsEnabled = true;
			}
			else
			{
				AddButton.IsEnabled = false;
			}
		}

		private void _addedProfiles_ListChanged(object sender, ListChangedEventArgs e)
		{
			SharedUsersDataGrid.ItemsSource = null;
			SharedUsersDataGrid.ItemsSource = AddedProfiles;
			SessionInfo instance = SessionInfo.Instance;
			instance.AddedProfiles = AddedProfiles;
			if (!instance.ModelHasBeenShared)
			{
				return;
			}
			bool flag = false;
			int num = 0;
			foreach (Profile addedProfile in AddedProfiles)
			{
				if (addedProfile.IsValid)
				{
					num++;
					if (!instance.SharedWithProfiles.Contains(addedProfile))
					{
						flag = true;
						break;
					}
				}
			}
			flag |= num != instance.SharedWithProfiles.Count;
			instance.ModelModified = flag || Math.Abs((instance.LatestDetail.LastModified.ToLocalTime() - File.GetLastWriteTime(instance.ActiveDocumentFilePath)).TotalMilliseconds) > 1.0;
		}

		private void SessionInfo_MessageReceived(object sender, SessionInfo.SessionMessageArgs e)
		{
			switch (e.Message)
			{
			case "ActionComplete":
				OnActionComplete();
				break;
			case "ActiveUserProfile":
				if (SessionInfo.Instance.ActiveUserProfile != null)
				{
					SessionInfo.Instance.ActiveUserProfile.IsActiveUser = true;
					if (!AddedProfiles.Contains(SessionInfo.Instance.ActiveUserProfile))
					{
						AddedProfiles.Add(SessionInfo.Instance.ActiveUserProfile);
					}
				}
				break;
			case "SimUpdate":
				ClearProfiles();
				ProcessSharedProfiles();
				UpdateUI();
				ShareOpStatusBlock.Visibility = Visibility.Hidden;
				ShareOpStatusImage.Visibility = Visibility.Hidden;
				break;
			case "ServerURL":
			case "ConnectedToServerSuccessfully":
			case "ModelDetails":
				ClearProfiles();
				ProcessSharedProfiles();
				UpdateUI();
				break;
			case "ModelModified":
				UpdateUI();
				break;
			}
		}

		private void UpdateUI()
		{
			if (_suppressUpdates)
			{
				return;
			}
			if (string.IsNullOrWhiteSpace(SessionInfo.Instance.ActiveDocumentFilePath))
			{
				ShareStatusLabel.Content = Strings.SHAREWINDOW_OPEN_A_MODEL;
				LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				ModelStatusLabel.Content = string.Empty;
				ShareButton.IsEnabled = false;
				UnshareButton.IsEnabled = false;
				EmailTextBox.IsEnabled = false;
				return;
			}
			if (!SessionInfo.Instance.FileExtensionIsValid)
			{
				if (SessionInfo.Instance.HostApplicationName == "Aspen Plus")
				{
					ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_UNSUPPORTED_FILE_APLUS;
				}
				else if (SessionInfo.Instance.HostApplicationName == "Aspen HYSYS")
				{
					ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_UNSUPPORTED_FILE_HYSYS;
				}
				LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				ModelStatusLabel.Content = string.Empty;
				ShareButton.IsEnabled = false;
				UnshareButton.IsEnabled = false;
				EmailTextBox.IsEnabled = false;
				return;
			}
			if (!SessionInfo.Instance.ConnectedToServerSuccessfully)
			{
				ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_NOT_CONNECTED;
				LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				ModelStatusLabel.Content = string.Empty;
				ShareButton.IsEnabled = false;
				UnshareButton.IsEnabled = false;
				EmailTextBox.IsEnabled = false;
				return;
			}
			EmailTextBox.IsEnabled = true;
			if (SessionInfo.Instance.ModelShareStatus == ModelStatus.PROCESSING)
			{
				UnshareButton.IsEnabled = false;
				ShareButton.IsEnabled = true;
				ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_QUEUED;
			}
			else if (SessionInfo.Instance.ModelHasBeenShared)
			{
				ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_SHARED;
				UnshareButton.IsEnabled = true;
				if (!SessionInfo.Instance.ModelShareTime.HasValue)
				{
					LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				}
				else if (SessionInfo.Instance.ModelShareTime == DateTime.MinValue)
				{
					LastSharedLabel.Content = DateTime.Now.ToString("MMMM dd, yyyy") + " - " + DateTime.Now.ToShortTimeString();
				}
				else
				{
					LastSharedLabel.Content = SessionInfo.Instance.ModelShareTime.Value.ToString("MMMM dd, yyyy") + " - " + SessionInfo.Instance.ModelShareTime.Value.ToShortTimeString();
				}
				CanDownloadCheckBox.IsChecked = SessionInfo.Instance.ShareUsersCanDownload;
				if (SessionInfo.Instance.ModelModified)
				{
					ShareButton.IsEnabled = true;
					ModelStatusLabel.Content = Strings.SHAREWINDOW_STATUS_MODIFIED;
				}
				else
				{
					ShareButton.IsEnabled = false;
					ModelStatusLabel.Content = string.Empty;
				}
			}
			else
			{
				UnshareButton.IsEnabled = false;
				ShareButton.IsEnabled = true;
				LastSharedLabel.Content = Strings.SHAREWINDOW_NA;
				ModelStatusLabel.Content = string.Empty;
				ShareStatusLabel.Content = Strings.SHAREWINDOW_STATUS_NOT_SHARED;
			}
		}

		private void ProcessSharedProfiles()
		{
			BindingList<Profile> sharedWithProfiles = SessionInfo.Instance.SharedWithProfiles;
			if (sharedWithProfiles == null)
			{
				return;
			}
			foreach (Profile item in sharedWithProfiles)
			{
				if (!AddedProfiles.Contains(item))
				{
					if (item != null && item.Equals(SessionInfo.Instance.ActiveUserProfile))
					{
						item.IsActiveUser = true;
					}
					AddedProfiles.Add(item);
				}
			}
		}

		private void PrepopulateUserDatagrid(ShareUser[] sharedUsers)
		{
			SharedUsersDataGrid.ItemsSource = null;
			AddedProfiles.Clear();
			if (SessionInfo.Instance.ActiveUserProfile != null)
			{
				AddedProfiles.Add(SessionInfo.Instance.ActiveUserProfile);
			}
			foreach (ShareUser shareUser in sharedUsers)
			{
				string[] array = shareUser.Account.Split('\\');
				Profile profile = new Profile(array[0], array[1], shareUser.EMail, shareUser.DisplayName);
				if (!profile.Equals(SessionInfo.Instance.ActiveUserProfile))
				{
					AddedProfiles.Add(profile);
				}
			}
			SharedUsersDataGrid.ItemsSource = AddedProfiles;
		}

		private void AddProfilesFromTypeAhead()
		{
			if (EmailTextBox == null || EmailTextBox.Profiles == null)
			{
				return;
			}
			foreach (Profile profile in EmailTextBox.Profiles)
			{
				if (profile.IsValid && !AddedProfiles.Contains(profile))
				{
					AddedProfiles.Add(profile);
				}
			}
			EmailTextBox.ClearAfterAdd();
			SharedUsersDataGrid.Focus();
			EmailTextBox.Focus();
		}

		private void OnActionComplete()
		{
			try
			{
				_suppressUpdates = true;
				if (SessionInfo.Instance.ConnectedToServerSuccessfully && SessionInfo.Instance.ModelShareStatus == ModelStatus.COMPLETED && SessionInfo.Instance.FileActionType == FileActionType.INGESTION)
				{
					ShareOpStatusImage.Source = StatusImages.greenCheckMark;
					ShareOpStatusImage.RenderTransform = null;
					ShareOpStatusBlock.Foreground = Palette.GreenStatus;
					ShareOpStatusBlock.Text = Strings.SHAREWINDOW_STATUS_SHARED;
					FadeOut(ShareOpStatusBlock, 2);
					FadeOut(ShareOpStatusImage, 2);
				}
				else if (SessionInfo.Instance.ConnectedToServerSuccessfully && SessionInfo.Instance.ModelShareStatus == ModelStatus.PROCESSING && SessionInfo.Instance.FileActionType == FileActionType.INGESTION)
				{
					ShareOpStatusBlock.Text = Strings.SHAREWINDOW_STATUS_QUEUED_INFO;
					ShareOpStatusImage.RenderTransform = null;
					ShareOpStatusBlock.Foreground = Palette.GreenStatus;
					FadeOut(ShareOpStatusImage, 2);
				}
				else if (SessionInfo.Instance.FileActionType == FileActionType.INGESTION)
				{
					ShareOpStatusImage.Source = StatusImages.redX;
					ShareOpStatusImage.RenderTransform = null;
					ShareOpStatusBlock.Foreground = Palette.RedStatus;
					ShareOpStatusBlock.Text = Strings.SHAREWINDOW_UNABLE_TO_SHARE;
				}
				else
				{
					ShareOpStatusImage.Source = StatusImages.greenCheckMark;
					ShareOpStatusImage.RenderTransform = null;
					ShareOpStatusBlock.Foreground = Palette.GreenStatus;
					ShareOpStatusBlock.Text = Strings.SHAREWINDOW_STATUS_NOT_SHARED;
					FadeOut(ShareOpStatusBlock, 2);
					FadeOut(ShareOpStatusImage, 2);
				}
			}
			finally
			{
				_suppressUpdates = false;
			}
			UpdateUI();
		}

		private void UnShareActiveCase()
		{
			MessageBoxResult messageBoxResult = MessageBox.Show(Strings.SHAREWINDOW_DIALOG_UNSHARE, SessionInfo.Instance.HostApplicationName, MessageBoxButton.OKCancel, MessageBoxImage.None, MessageBoxResult.No);
			if (messageBoxResult == MessageBoxResult.OK)
			{
				UnshareButton.IsEnabled = false;
				ShareOpStatusImage.Source = StatusImages.spinner;
				ShareOpStatusBlock.Foreground = Brushes.Black;
				ShareOpStatusBlock.Text = Strings.SHAREWINDOW_STATUS_UNSHARING;
				StatusImages.Spin(ShareOpStatusImage);
				ShareButton.IsEnabled = false;
				ShareOpStatusBlock.Visibility = Visibility.Visible;
				ShareOpStatusImage.Visibility = Visibility.Visible;
				Cursor overrideCursor = Mouse.OverrideCursor;
				try
				{
					Mouse.OverrideCursor = Cursors.Wait;
					SessionInfo.Instance.UnShareCase();
				}
				finally
				{
					Mouse.OverrideCursor = overrideCursor;
				}
			}
		}

		private void ShareActiveCase()
		{
			if (SessionInfo.Instance.IsDirty)
			{
				MessageBoxResult messageBoxResult = MessageBox.Show(Strings.SHAREWINDOW_DIALOG_SAVE_AND_SHARE, SessionInfo.Instance.HostApplicationName, MessageBoxButton.OKCancel, MessageBoxImage.None, MessageBoxResult.Yes);
				if (messageBoxResult != MessageBoxResult.OK)
				{
					return;
				}
				SessionInfo.Instance.SaveCase();
			}
			ShareOpStatusImage.Source = StatusImages.spinner;
			ShareOpStatusBlock.Foreground = Brushes.Black;
			ShareOpStatusBlock.Text = Strings.SHAREWINDOW_STATUS_SHARING;
			StatusImages.Spin(ShareOpStatusImage);
			ShareButton.IsEnabled = false;
			ShareOpStatusBlock.Visibility = Visibility.Visible;
			ShareOpStatusImage.Visibility = Visibility.Visible;
			Cursor overrideCursor = Mouse.OverrideCursor;
			try
			{
				Mouse.OverrideCursor = Cursors.Wait;
				SessionInfo.Instance.ShareCase();
			}
			finally
			{
				Mouse.OverrideCursor = overrideCursor;
			}
		}

		private static void FadeOut(UIElement control, int fadeDuration)
		{
			DoubleAnimation doubleAnimation = new DoubleAnimation();
			doubleAnimation.From = 1.0;
			doubleAnimation.To = 0.0;
			doubleAnimation.Duration = new Duration(TimeSpan.FromSeconds(fadeDuration));
			doubleAnimation.Completed += delegate
			{
				control.Visibility = Visibility.Hidden;
				control.Opacity = 1.0;
			};
			doubleAnimation.FillBehavior = FillBehavior.Stop;
			control.BeginAnimation(UIElement.OpacityProperty, doubleAnimation);
		}
