		public void ToggleAlign(bool align, double distance)
		{
			if (align == _align || _source == null || _sink == null)
			{
				return;
			}
			_align = align;
			if (!_align)
			{
				_distance = 0.0;
				System.Windows.Point point = _points[0];
				System.Windows.Point point2 = _points[1];
				if (point.Y == point2.Y)
				{
					System.Windows.Point item = new System.Windows.Point((point.X + point2.X) / 2.0, point.Y);
					_points.Insert(1, item);
					item = new System.Windows.Point((point.X + point2.X) / 2.0, point.Y);
					_points.Insert(1, item);
				}
				else
				{
					System.Windows.Point item2 = new System.Windows.Point(point.X, (point.Y + point2.Y) / 2.0);
					_points.Insert(1, item2);
					item2 = new System.Windows.Point(point.X, (point.Y + point2.Y) / 2.0);
					_points.Insert(1, item2);
				}
				RefreshPoints();
			}
			else
			{
				_distance = distance;
				base.Children.Remove(_pl);
				Connect();
			}
		}

		public void ToggleArrow(bool on)
		{
			if (on)
			{
				int num = _points.Count - 1;
				if (_sinkOffset > 0)
				{
					num = _points.Count - 2;
				}
				System.Windows.Point point = _points[num];
				double x = point.X;
				double y = point.Y;
				System.Windows.Point point2 = _points[num - 1];
				double x2 = point2.X;
				double y2 = point2.Y;
				_arrowhead = new ArrowHead();
				_arrowhead.Render();
				_arrowhead.SetPosition(x, y);
				double angle = 0.0;
				if (Math.Abs(x - x2) < 0.01)
				{
					angle = ((!(y < y2)) ? 90.0 : (-90.0));
				}
				else if (x < x2)
				{
					angle = 180.0;
				}
				_arrowhead.SetAngle(angle);
				base.Children.Add(_arrowhead);
			}
			else
			{
				if (_arrowhead != null)
				{
					base.Children.Remove(_arrowhead);
				}
				_arrowhead = null;
			}

		{

			{
				num = distance - _distance;

			}

			{
				System.Windows.Point item = _points[1];
				item.X += num;
				item.Y += num2;

			{
				System.Windows.Point item2 = _points[0];
				item2.X -= num;
				item2.Y -= num2;
				_points.RemoveAt(0);
				_points.Insert(0, item2);

				{
					num = (num2 = 0.0);
				}
			}
			RefreshPoints();
			if (_arrowhead != null)
			{
				_arrowhead.Move(num, num2);
			}

		public void CheckRange(ref double minx, ref double miny, ref double maxx, ref double maxy)
		{
			foreach (System.Windows.Point point in _points)
			{
				if (point.X < minx)
				{
					minx = point.X;
				}
				if (point.X > maxx)
				{
					maxx = point.X;
				}
				if (point.Y < miny)
				{
					miny = point.Y;
				}
				if (point.Y > maxy)
				{
					maxy = point.Y;
				}
			}
		}

		{
			double num = 0.0;
			double num2 = 0.0;

			System.Windows.Point item = _points[0];

			{
				_source.Ctrl.GetPortPosition(_source, out var x, out var y, out var _);
				num = x - item.X;
				num2 = y - item.Y;
				item.X = x;
				item.Y = y;
				_points.Clear();
				_points.Add(item);
				if (_align || _sink == null)
				{
					p.X += num;
					p.Y += num2;
				}
				else if (!_align)
				{
					Reroute(ref p);
				}
				_points.Add(p);
				OptimizeRoute();

				{
					_sink.Ctrl.GetPosition(out var x2, out var y2, out var scale);
					_sink.Ctrl.SetPosition(x2 + num, y2 + num2, scale);
				}
				RefreshPoints();

				{
					_arrowhead.Move(num, num2);
				}
			}
			else if (port == _sink)
			{
				_sink.Ctrl.GetPortPosition(_sink, out var x3, out var y3, out var _);
				num = p.X - x3;
				num2 = p.Y - y3;
				if (_align)
				{
					_sink.Ctrl.GetPosition(out var x4, out var y4, out var scale2);
					_sink.Ctrl.SetPosition(x4 + num, y4 + num2, scale2);
				}
				else
				{
					p.X = x3;
					p.Y = y3;
					if (_source == null)
					{
						item.X -= num;
						item.Y -= num2;
					}
					_points.Clear();
					_points.Add(item);
					if (_source != null)
					{
						Reroute(ref p);
					}
					_points.Add(p);
					OptimizeRoute();
					RefreshPoints();
					if (_arrowhead != null)
					{
						_arrowhead.Move(0.0 - num, 0.0 - num2);
					}
				}
			}
			if (_align)
			{
				_container.PropagateMove(this, connConnected: false);
			}
		}

		public void Move(double dx, double dy)
		{
			if (_pl != null)
			{
				_pl.Points.Clear();
				for (int i = 0; i < _points.Count; i++)
				{
					System.Windows.Point item = _points[i];
					item.X += dx;
					item.Y += dy;
					_points.RemoveAt(i);
					_points.Insert(i, item);
					_pl.Points.Add(_points[i]);
				}
				if (_arrowhead != null)
				{
					_arrowhead.Move(dx, dy);
				}
				_back.Points.Clear();
				for (int j = 0; j < _points.Count; j++)
				{
					_back.Points.Add(_points[j]);
				}
			}
		}

		public void Initialize(double xl, double xr, double yt, double yb)
		{
			SnapToGrid(ref xl);
			SnapToGrid(ref xr);
			SnapToGrid(ref yt);
			SnapToGrid(ref yb);
			if (xl > xr)
			{
				double num = xr;
				xr = xl;
				xl = num;
			}
			if (yb > yt)
			{
				double num = yt;
				yt = yb;
				yb = num;
			}
			bx = xl;
			by = yb;
			bx2 = xr;
			by2 = yt;
			while (true)
			{
				double num2 = (bx2 - bx) / MESH_DELTA / 4.0;
				double num3 = (by2 - by) / MESH_DELTA;
				double num4 = num2 * num3;
				if (num4 < 2000000000.0)
				{
					break;
				}
				MESH_DELTA *= 2.0;
			}
			int num5 = 0;
			int num6;
			int num8;
			while (true)
			{
				lx = (int)((bx2 - bx) / MESH_DELTA) + 1;
				ly = (int)((by2 - by) / MESH_DELTA) + 1;
				num6 = (lx - 1 >> 2) + 1;
				long num7 = (long)num6 * (long)ly;
				num8 = (int)((num7 - 1) / 1200 + 1);
				if (num8 > 300)
				{
					MESH_DELTA *= 2.0;
					continue;
				}
				if (!(MESH_DELTA > 20.0) || num8 >= 10 || num5 != 0)
				{
					break;
				}
				num5 = 1;
				MESH_DELTA = 4.0;
			}
			DELTA = MESH_DELTA;
			DEL = DELTA * 0.5;
			DDEL = DELTA * 0.2;
			while (true)
			{
				ndivx = (ndivy = 0);
				int num9 = 0;
				while (true)
				{
					int num10 = (1 << ndivx) * (1 << ndivy);
					if (num8 <= num10)
					{
						break;
					}
					if (num9 != 0)
					{
						ndivx++;
					}
					else
					{
						ndivy++;
					}
					num9 = 1 - num9;
				}
				dx = (num6 - 1 >> ndivx) + 1;
				lx2 = dx << 2;
				dy = (ly - 1 >> ndivy) + 1;
				int num11 = dx * dy;
				if (num11 <= 1200)
				{
					break;
				}
				num8++;
			}
			ndx = 1 << ndivx;
			ndy = 1 << ndivy;
			num8 = ndx * ndy;
			for (int i = 0; i < nB; i++)
			{
				for (int j = 0; j < 1200; j++)
				{
					B[i][j] = 0;
				}
			}
			if (num8 == nB)
			{
				return;
			}
			byte[][] b = B;
			B = new byte[num8][];
			int num12 = ((nB < num8) ? nB : num8);
			for (int i = 0; i < num12; i++)
			{
				B[i] = b[i];
			}
			for (int i = nB; i < num8; i++)
			{
				B[i] = new byte[1200];
				for (int j = 0; j < 1200; j++)
				{
					B[i][j] = 0;
				}
			}
			nB = num8;

		}

		}

			double num4 = GetDouble(range, "MaxY");
			icon.SetSize(num, num2, num3, num4);
			if (x + num < _minX)
			{
				_minX = x + num;
			}
			if (x + num3 > _maxX)
			{
				_maxX = x + num3;
			}
			if (y + num2 < _minY)
			{
				_minY = y + num2;
			}
			if (y + num4 > _maxY)
			{
				_maxY = y + num4;
			}
		}

		private bool GetPosition(string hint, double minx, double miny, double maxx, double maxy, out double bx, out double by)
		{
			bx = (by = 0.0);
			string[] array = hint.Split(';');
			global::IconCtrl.IconCtrl iconCtrl = FindIcon(array[0]);
			if (iconCtrl == null)
			{
				return false;
			}
			iconCtrl.GetPosition(out bx, out by, out var scale);
			iconCtrl.GetSize(out var minx2, out var maxx2, out var miny2, out var maxy2);
			double num = 0.0;
			double num2 = 0.0;
			double num3 = 0.0;
			double num4 = 0.0;
			double num5 = _defDistance;
			if (array.Length >= 4)
			{
				num5 = double.Parse(array[3]);
			}
			switch (array[1])
			{
			case "E":
				num = scale * maxx2;
				num3 = 0.0 - minx + num5;
				break;
			case "W":
				num = scale * minx2;
				num3 = 0.0 - maxx - num5;
				break;
			case "N":
				num2 = scale * miny2;
				num4 = 0.0 - maxy - num5;
				break;
			case "S":
				num2 = scale * maxy2;
				num4 = 0.0 - miny + num5;
				break;
			case "NE":
				num = scale * maxx2;
				num3 = 0.0 - minx + num5;
				num2 = scale * miny2;
				num4 = 0.0 - maxy - num5;
				break;
			case "NW":
				num = scale * minx2;
				num3 = 0.0 - maxx - num5;
				num2 = scale * miny2;
				num4 = 0.0 - maxy - num5;
				break;
			case "SE":
				num = scale * maxx2;
				num3 = 0.0 - minx + num5;
				num2 = scale * maxy2;
				num4 = 0.0 - miny + num5;
				break;
			case "SW":
				num = scale * minx2;
				num3 = 0.0 - maxx - num5;
				num2 = scale * maxy2;
				num4 = 0.0 - miny + num5;
				break;
			}
			if (array.Length > 2 && !string.IsNullOrEmpty(array[2]))
			{
				Port port = iconCtrl.FindPort(array[2]);
				if (port != null)
				{
					iconCtrl.GetPortPosition(port, out var x, out var y, out var _);
					num = x - bx;
					num2 = y - by;
				}
			}
			bx += num + num3;
			by += num2 + num4;
			return true;
		}

		{
			_scannedIcons = new List<global::IconCtrl.IconCtrl>();
			if (icon != null)
			{
				_scannedIcons.Add(icon);

		public void PropagateMove(Connection connection, bool connConnected, bool needRescan = false)
		{
			if (_connections.Count == 1)
			{
				return;
			}
			if (!needRescan && !_scannedConnections.Contains(connection))
			{
				_scannedConnections.Add(connection);
			}
			global::IconCtrl.IconCtrl iconCtrl = null;
			if (!connConnected)
			{
				iconCtrl = connection.SinkPort.Ctrl;
			}
			foreach (Connection connection2 in _connections)
			{
				if (_scannedConnections.Contains(connection2))
				{
					continue;
				}
				if (connConnected)
				{
					if (connection2.SourceConn == connection)
					{
						connection2.MoveEnd(start: true, 0.0, 0.0);
					}
					if (connection2.SinkConn == connection)
					{
						connection2.MoveEnd(start: false, 0.0, 0.0);
					}
				}
				else if (connection2.SourcePort.Ctrl == iconCtrl)
				{
					connection2.MovePort(connection2.SourcePort);
				}
				else if (connection2.SinkPort.Ctrl == iconCtrl)
				{
					connection2.MovePort(connection2.SinkPort);
				}
			}
		}

		public void PropagateMove(global::IconCtrl.IconCtrl icon, double dx, double dy)
		{
			if (_scannedIcons.Contains(icon))
			{
				return;
			}
			_scannedIcons.Add(icon);
			if (icon != _curIcon)
			{
				icon.GetPosition(out var x, out var y, out var scale);
				icon.SetPosition(x + dx, y + dy, scale);
			}
			foreach (Connection connection in _connections)
			{
				if (_scannedConnections.Contains(connection))
				{
					continue;
				}
				if (connection.SourcePort != null && connection.SourcePort.Ctrl == icon)
				{
					_scannedConnections.Add(connection);
					if (connection.Aligned)
					{
						PropagateMove(connection.SinkPort.Ctrl, dx, dy);
						connection.Move(dx, dy);
					}
					else
					{
						connection.MoveEnd(start: true, dx, dy);
					}
				}
				else if (connection.SinkPort != null && connection.SinkPort.Ctrl == icon)
				{
					_scannedConnections.Add(connection);
					if (connection.Aligned)
					{
						PropagateMove(connection.SourcePort.Ctrl, dx, dy);
						connection.Move(dx, dy);
					}
					else
					{
						connection.MoveEnd(start: false, dx, dy);
					}
				}
			}
		}

						}
						else if (list.Contains(childNode.Name))
						{
							continue;
						}
					}
					iconParameterEditor.AddItem(i, num, childNode);
					num++;
				}
			}
			iconParameterEditor.ShowDialog();
		}

		private void VerifyIconPosition(global::IconCtrl.IconCtrl icon, ref double delx, ref double dely)
		{
			List<global::IconCtrl.IconCtrl> attached = new List<global::IconCtrl.IconCtrl>();
			FindAttached(icon, ref attached);
			MeshInitialize(null, attached);
			icon.GetPosition(out var x, out var y, out var scale);
			icon.GetSize(out var minx, out var maxx, out var miny, out var maxy);
			double num = 0.0;
			double num2;
			for (num2 = 0.0; _mesh.CheckRegion(x + num + scale * minx, y + num2 + scale * miny, x + num + scale * maxx, y + num2 + scale * maxy) == 0; num += 10.0)
			{
			}
			if (num != 0.0 || num2 != 0.0)
			{
				icon.SetPosition(x + num, y + num2, scale);

			{

		public void MeshInitialize(Connection connection, List<global::IconCtrl.IconCtrl> iconToExclude)
		{
			double num = 100.0;
			_mesh.Initialize(_minX - num, _maxX + num, _minY - num, _maxY + num);
			foreach (global::IconCtrl.IconCtrl icon in _icons)
			{
				if (iconToExclude == null || !iconToExclude.Contains(icon))
				{
					icon.GetPosition(out var x, out var y, out var scale);
					icon.GetSize(out var minx, out var maxx, out var miny, out var maxy);
					_mesh.MapBlock(x + scale * minx, y + scale * miny, x + scale * maxx, y + scale * maxy);
				}
			}
			if (iconToExclude != null)
			{
				return;
			}
			foreach (Connection connection2 in _connections)
			{
				if (connection2 == connection)
				{
					continue;
				}
				int count = connection2.Points.Count;
				for (int i = 1; i < count; i++)
				{
					System.Windows.Point point = connection2.Points[i - 1];
					System.Windows.Point point2 = connection2.Points[i];
					if (Math.Abs(point.Y - point2.Y) < 1.0)
					{
						_mesh.MapHLine(point.Y, point.X, point2.X, bInclusive: true);
					}
					else
					{
						_mesh.MapVLine(point.X, point.Y, point2.Y, bInclusive: true);
					}
				}
			}
		}

		public int CountConnection(Port port, Connection exclude)
		{
			int num = 0;
			foreach (Connection connection in _connections)
			{
				if (exclude != null && connection == exclude)
				{
					break;
				}
				if (port.IOtype == "0")
				{
					if (connection.SinkPort == port)
					{
						num++;
					}
				}
				else if (connection.SourcePort == port)
				{
					num++;
				}
			}
			return num;
		}
