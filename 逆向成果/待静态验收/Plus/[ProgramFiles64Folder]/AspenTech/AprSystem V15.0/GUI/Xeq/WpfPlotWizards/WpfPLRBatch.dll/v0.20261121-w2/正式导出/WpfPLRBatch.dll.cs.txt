		List<string> IRibbonPlotWizardType.FieldDataOptionList(string sFieldName)
		{
			if (sFieldName.CompareTo(_timeUnitFieldId) == 0)
			{
				List<string> list = new List<string>();
				_help.LoadXUnitsList(_context.Application.Tree, _wizardPlot, list);
				return list;
			}
			if (sFieldName.CompareTo(_timeOrderFieldId) == 0)
			{
				return new List<string> { _ascending, _descending };
			}
			if (sFieldName.CompareTo(_timeOnFieldId) == 0)
			{
				return new List<string> { _xAxis, _yAxis };
			}
			return null;
		}

		void IRibbonPlotWizardType.GetFieldStyle(string sFieldName, ref string sCurrentVal)
		{
			if (sFieldName.CompareTo(_timeOrderFieldId) == 0)
			{
				if (_wizardPlot.PlotVal.ReverseXAxis)
				{
					sCurrentVal = _descending;
				}
				else
				{
					sCurrentVal = _ascending;
				}
			}
			else if (sFieldName.CompareTo(_timeOnFieldId) == 0)
			{
				if (_wizardPlot.PlotVal.SwapAxises)
				{
					sCurrentVal = _yAxis;
				}
				else
				{
					sCurrentVal = _xAxis;
				}
			}
		}

		void IRibbonPlotWizardType.SetFieldStyle(string sFieldName, string sValue)
		{
			if (sFieldName.CompareTo(_timeOrderFieldId) == 0)
			{
				if (sValue.CompareTo(_ascending) == 0)
				{
					_wizardPlot.PlotVal.ReverseXAxis = false;
				}
				else
				{
					_wizardPlot.PlotVal.ReverseXAxis = true;
				}
			}
			else if (sFieldName.CompareTo(_timeOnFieldId) == 0)
			{
				if (sValue.CompareTo(_yAxis) == 0)
				{
					_wizardPlot.PlotVal.SwapAxises = true;
				}
				else
				{
					_wizardPlot.PlotVal.SwapAxises = false;
				}
			}
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			IHNodeCol curves = null;
			bool flag = false;
			for (int i = 0; i < _VarNames.Count(); i++)
			{
				iHNode = _context.FindNode(_VarNames[i]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					_radBtnArr[i].IsEnabled = true;
					flag = true;
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			labelSubstream.IsEnabled = false;
			comboBoxSubstream.Items.Clear();
			comboBoxSubstream.IsEnabled = false;
			labelUnit.IsEnabled = false;
			comboBoxUnit.Items.Clear();
			comboBoxUnit.IsEnabled = false;
			for (int j = 0; j < _VarNames.GetLength(0); j++)
			{
				if (!_radBtnArr[j].IsEnabled)
				{
					continue;
				}
				_radBtnArr[j].IsChecked = true;
				iHNode = _context.FindNode(_VarNames[j]);
				if (iHNode == null || iHNode.Elements.Count <= 0)
				{
					break;
				}
				if (j == 5)
				{
					int count = iHNode.Elements.Count;
					if (count > 0)
					{
						comboBoxSubstream.IsEnabled = true;
						for (int k = 0; k < count; k++)
						{
							string newItem = iHNode.Elements.get_Label(0, j, (object)null);
							comboBoxSubstream.Items.Add(newItem);
						}
						comboBoxSubstream.SelectedIndex = 0;
						string text = comboBoxSubstream.Text;
						if (iHNode.Elements[text, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null && iHNode.Elements[text, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null)
						{
							curves = iHNode.Elements[text, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
						}
					}
				}
				else
				{
					curves = iHNode.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
				}
				break;
			}
			SetCurves(curves);
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				RestoreConfigValue(_currentConfigValue);
			}
			_wizardPlot.SetIndependentVariable(_context, "Output/TIME");
		}

		public void UpdateWizardData()
		{
			string text = "";
			int num = -1;
			for (int i = 0; i < _radBtnArr.Count(); i++)
			{
				if (_radBtnArr[i].IsChecked == true)
				{
					num = i;
					text = _radBtnArr[i].Content.ToString();
					break;
				}
			}
			string text2 = comboBoxSubstream.Text;
			string path = _VarNames[num];
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.SetIndependentVariable(_context, "Output/TIME");
			IHNode iHNode = _context.FindNode(path);
			IHNode iHNode2 = iHNode;
			if (iHNode != null)
			{
				if (num == 5 && iHNode.Elements[text2, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null && iHNode.Elements[text2, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null)
				{
					iHNode2 = iHNode.Elements[text2, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing];
				}
				if (iHNode2 != null)
				{
					int count = iHNode2.Elements.Count;
					_wizardPlot.PlotVal.NPoints = count;
					List<string> list = new List<string>(CurveList.GetSelectedCurves());
					int count2 = list.Count;
					_wizardPlot.PlotVal.NVariables = count2;
					for (int j = 0; j < count2; j++)
					{
						string text3 = list[j];
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(j, text3);
						for (int k = 0; k < count; k++)
						{
							if (num == 5)
							{
								if (iHNode.Elements[text2, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null && iHNode.Elements[text2, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text3, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null && iHNode.Elements[text2, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text3, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null)
								{
									((IHPlotVal)_wizardPlot.PlotVal).set_YData(j, k, (short)1, (object)iHNode.Elements[text2, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text3, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing]);
								}
							}
							else if (iHNode.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null && iHNode.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text3, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null)
							{
								((IHPlotVal)_wizardPlot.PlotVal).set_YData(j, k, (short)1, (object)iHNode.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text3, Type.Missing, Type.Missing, Type.Missing, Type.Missing]);
							}
						}
					}
				}
			}
			_wizardPlot.PlotVal.XAxisTitle = WpfPLRBatchResources.XAxisTitle_Time;
			string ytitle;
			if (num == 5)
			{
				_wizardPlot.PlotVal.Title = WpfPLRBatchResources.Block + " " + WpfPLRBatchResources.Substream + text2 + " " + WpfPLRBatchResources.Title;
				ytitle = text + " " + comboBoxUnit.Text;
			}
			else
			{
				_wizardPlot.PlotVal.Title = WpfPLRBatchResources.Block + " " + WpfPLRBatchResources.Composition;
				ytitle = text + " " + WpfPLRBatchResources.Composition_Mole;
			}
			((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle((object)null, ytitle);
			_help.SaveXUnitsComboText(_context.Application.Tree, _wizardPlot, _currentUnit);
			_currentConfigValue = SaveConfigValue();
		}

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			int num = CurveList.GetSelectedCurves().Count();
			if (num == CurveList.Curves.Count)
			{
				CurveList.IsCheckAll = true;
			}
			else
			{
				string[] selectedCurves = CurveList.GetSelectedCurves();
				CurveList.IsCheckAll = false;
				CurveList.SetSelectedCurves(selectedCurves);
			}
			e.CanExecute = num != 0;
		}

		private void SetUnit(int index, string strSubstream)
		{
			string path = _VarNames[index];
			labelSubstream.IsEnabled = false;
			comboBoxSubstream.IsEnabled = false;
			labelUnit.IsEnabled = false;
			comboBoxUnit.Items.Clear();
			comboBoxUnit.IsEnabled = false;
			IHNode iHNode = _context.FindNode(path);
			if (iHNode == null || iHNode.Elements == null)
			{
				return;
			}
			IHNodeCol curves = null;
			if (index == 5)
			{
				labelSubstream.IsEnabled = true;
				comboBoxSubstream.IsEnabled = true;
				labelUnit.IsEnabled = true;
				comboBoxUnit.IsEnabled = true;
				comboBoxSubstream.Items.Clear();
				int count = iHNode.Elements.Count;
				for (int i = 0; i < count; i++)
				{
					comboBoxSubstream.Items.Add(iHNode.Elements.get_Label(0, i, (object)null));
				}
				if (string.IsNullOrEmpty(strSubstream))
				{
					comboBoxSubstream.SelectedIndex = 0;
					strSubstream = comboBoxSubstream.Text;
				}
				if (iHNode.Elements[strSubstream, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null)
				{
					curves = iHNode.Elements[strSubstream, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
					try
					{
						if (iHNode.Elements[strSubstream, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null && iHNode.Elements[strSubstream, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing] != null)
						{
							_wizardPlot.PlotVal.NVariables = 1;
							((IHPlotVal)_wizardPlot.PlotVal).set_YData(0, 0, (short)1, (object)iHNode.Elements[strSubstream, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing]);
							_help.LoadUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxUnit);
						}
					}
					catch
					{
					}
				}
			}
			else
			{
				curves = iHNode.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
			}
			SetCurves(curves);
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("CompSelection=");
			for (int i = 0; i < _radBtnArr.Count(); i++)
			{
				if (_radBtnArr[i].IsChecked == true)
				{
					stringBuilder.Append(i.ToString());
					break;
				}
			}
			stringBuilder.Append(";");
			int num = 0;
			stringBuilder.Append("CompButtonEnable=");
			for (int j = 0; j < _radBtnArr.Count(); j++)
			{
				if (_radBtnArr[j].IsEnabled)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(j.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("SubstreamEnable=");
			stringBuilder.Append(comboBoxSubstream.IsEnabled);
			stringBuilder.Append(";");
			stringBuilder.Append("Substream=");
			stringBuilder.Append(comboBoxSubstream.Text);
			stringBuilder.Append(";");
			stringBuilder.Append("UnitSelection=");
			stringBuilder.Append(comboBoxUnit.Text);
			stringBuilder.Append(";");
			stringBuilder.Append("UnitEnable=");
			stringBuilder.Append(comboBoxUnit.IsEnabled);
			stringBuilder.Append(";");
			stringBuilder.Append("CurveSelection=");
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			int count = list.Count;
			for (int k = 0; k < count; k++)
			{
				if (k > 0)
				{
					stringBuilder.Append(",");
				}
				stringBuilder.Append(list[k]);
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CheckAll=");
			stringBuilder.Append(CurveList.IsCheckAll);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string value = array[num].Substring(array[num].IndexOf('=') + 1);
			int index = Convert.ToInt16(value);
			_radBtnArr[index].IsChecked = true;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = value.Split(',');
			foreach (string value2 in array2)
			{
				_radBtnArr[Convert.ToInt16(value2)].IsEnabled = true;
			}
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxSubstream.IsEnabled = Convert.ToBoolean(value);
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			if (comboBoxSubstream.IsEnabled)
			{
				SetUnit(index, value);
			}
			comboBoxSubstream.Text = value;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxUnit.Text = value;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxUnit.IsEnabled = Convert.ToBoolean(value);
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array3 = value.Split(',');
			List<string> list = new List<string>();
			array2 = array3;
			foreach (string item in array2)
			{
				list.Add(item);
			}
			CurveList.SetSelectedCurves(list.ToArray());
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);

		public void UpdateWizardData()
		{
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.SetIndependentVariable(_context, "Output/TIME");
			_wizardPlot.PlotVal.XAxisTitle = WpfPLRBatchResources.XAxisTitle_Time;
			_wizardPlot.PlotVal.NVariables = 1;
			int selectedIndex = comboBoxVariable.SelectedIndex;
			string text = comboBoxVariable.SelectedItem.ToString();
			IHNode iHNode = _context.FindNode("Output/USER");
			if (iHNode != null && iHNode.Elements.Count > 0)
			{
				int count = iHNode.Elements[selectedIndex, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements.Count;
				for (int i = 0; i < count; i++)
				{
					try
					{
						double num = ((double?)(dynamic)iHNode.Elements[selectedIndex, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[i, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing)) ?? 0.0;
						((IHPlotVal)_wizardPlot.PlotVal).set_YData(0, i, (short)0, (object)num);
					}
					catch
					{
					}
				}
			}
			if (!string.IsNullOrEmpty(text) && !string.IsNullOrEmpty(txtBoxUnit.Text))
			{
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle((object)null, text + "  " + txtBoxUnit.Text);
			}
			if (!string.IsNullOrEmpty(text))
			{
				((IHPlotVal)_wizardPlot.PlotVal).set_Label(0, text);
			}
			else
			{
				((IHPlotVal)_wizardPlot.PlotVal).set_Label(0, ((IHPlotVal)_wizardPlot.PlotVal).get_YAxisTitle(Type.Missing));
			}
			_help.SaveXUnitsComboText(_context.Application.Tree, _wizardPlot, _currentTimeUnit);
			_currentConfigValue = SaveConfigValue();
		}
