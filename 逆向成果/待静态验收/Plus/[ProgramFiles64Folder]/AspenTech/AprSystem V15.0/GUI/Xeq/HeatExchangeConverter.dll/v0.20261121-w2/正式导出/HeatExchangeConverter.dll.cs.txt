		{

			string text = "";
			for (int i = 0; i < streamList.Count; i++)
			{

				if (streamConnection.blockPort.ToString() == "F(IN)")
				{
					text = ((!(side == "HOT")) ? "C(IN)" : "H(IN)");
				}
				else if (streamConnection.blockPort.ToString() == "P(OUT)")
				{
					text = ((!(side == "HOT")) ? "C(OUT)" : "H(OUT)");
				}
				if (!string.IsNullOrEmpty(text))
				{
					string streamName = streamConnection.streamName;

				}

				iHNode2 = pNode.FindNode("Input/" + convertData.varName);
				try
				{
					if (iHNode2 != null && convertData.Value != null)
					{
						iHNode2.set_Value(Type.Missing, convertData.Value);
					}
					else if (iHNode2 != null && convertData.Value == null && j > 4 && j < 11)
					{

			}

			{
				ConvertData convertData2 = varWithUnitList[k] as ConvertData;
				iHNode2 = pNode.FindNode("Input/" + convertData2.varName);

					{
