		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = _context.FindNode(_colVarNames[0]);
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				ierror = 1;
				return;
			}
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, _colVarNames[0]);
			if (_bEnthalpyBox)
			{
				_wizardPlot.AddDependentVariable(_context, _colVarNames[1]);
			}
			_wizardPlot.PlotVal.SwapAxises = true;
			iHNode = _context.FindNode(_stripperVarNames[0]);
			_petrofracHelp.LoadColumn(iHNode, comboBoxStripper, radBtnStripper);
			if (comboBoxStripper.Items.Count <= 0)
			{
				_bHasInputDialog = false;
			}
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				radBtnMainColumn.IsChecked = true;
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		public void UpdateWizardData()
		{
			string text = "";
			string depvar = "";
			string text2 = "";
			_wizardPlot.RemoveAllDependentVariable();
			if (radBtnMainColumn.IsChecked == true)
			{
				text = _colVarNames[0];
				text2 = WpfPLPetroFracResources.Title_MainCol;
				if (_bEnthalpyBox)
				{
					depvar = _colVarNames[1];
				}
			}
			else
			{
				string text3 = comboBoxStripper.Text.Trim();
				text = _stripperVarNames[0] + "/" + text3;
				text2 = WpfPLPetroFracResources.Title_Stripper + " " + text3;
				if (_bEnthalpyBox)
				{
					depvar = _stripperVarNames[1] + "/" + text3;
				}
			}
			_wizardPlot.SetIndependentVariable(_context, text + "!Label");
			_wizardPlot.AddDependentVariable(_context, text);
			((IHPlotVal)_wizardPlot.PlotVal).set_Label(0, WpfPLPetroFracResources.CGCCstgH_Legends0);
			_wizardPlot.PlotVal.Title = WpfPLPetroFracResources.CGCCStgH_Title + " " + text2;
			if (_bEnthalpyBox)
			{
				_wizardPlot.AddDependentVariable(_context, depvar);
				((IHPlotVal)_wizardPlot.PlotVal).set_Label(1, WpfPLPetroFracResources.CGCCstgH_Legends1);
			}
			_help.SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentEnthalpyFlowUnit);
			_currentConfigValue = SaveConfigValue();
		}

		private void SetVariables()
		{
			_colVarNames = new string[2] { "Output/ENTHCOL", "Output/THBOXCOL" };
			_stripperVarNames = new string[2] { "Output/ENTHSTR", "Output/THBOXSTR" };
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("ColumnOption=");
			stringBuilder.Append(radBtnMainColumn.IsChecked);
			stringBuilder.Append(";");
			stringBuilder.Append("StripperEnabled=");
			stringBuilder.Append(comboBoxStripper.IsEnabled);
			stringBuilder.Append(";");
			stringBuilder.Append("Stripper=");
			stringBuilder.Append(comboBoxStripper.Text);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string value = array[num].Substring(array[num].IndexOf('=') + 1);
			radBtnMainColumn.IsChecked = Convert.ToBoolean(value);
			radBtnStripper.IsChecked = !radBtnMainColumn.IsChecked;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxStripper.IsEnabled = Convert.ToBoolean(value);
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxStripper.Text = value;
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			bool flag = false;
			for (int i = 0; i < _colVarNames.Count(); i++)
			{
				iHNode = _context.FindNode(_colVarNames[i]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					flag = true;
					break;
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			List<string> list = new List<string>();
			CurveList.IsCheckAll = false;
			if (iHNode != null && iHNode.Elements.Count > 0)
			{
				IHNodeCol elements = iHNode.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
				if (elements != null && elements.Count > 0)
				{
					for (int j = 0; j < elements.Count; j++)
					{
						string item = elements.get_Label(0, j, (object)null);
						list.Add(item);
					}
					CurveList.SetCurves(list.ToArray());
					CurveList.Header = WpfPLPetroFracResources.Comp_Header;
				}
			}
			iHNode = _context.FindNode(_stripperVarNames[0]);
			_petrofracHelp.LoadColumn(iHNode, comboBoxStripper, radBtnStripper);
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				radBtnMainColumn.IsChecked = true;
				for (int k = 0; k < _colVarNames.Count(); k++)
				{
					iHNode = _context.FindNode(_colVarNames[k]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_phaseRadBtnList[k].IsEnabled = true;
					}
				}
				for (int l = 0; l < _phaseRadBtnList.Count(); l++)
				{
					if (_phaseRadBtnList[l].IsEnabled)
					{
						_phaseRadBtnList[l].IsChecked = true;
						iHNode = _context.FindNode(_colVarNames[l]);
						break;
					}
				}
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		{
			int num = -1;
			string text = "";
			string ytitle = "";
			for (int i = 0; i < _phaseRadBtnList.Count; i++)
			{
				if (_phaseRadBtnList[i].IsChecked == true)
				{
					num = i;
					text = _strTitle[i];
					ytitle = _strYTitle[i];
					break;
				}
			}
			_wizardPlot.RemoveAllDependentVariable();
			IHNode iHNode = null;
			string text2 = "";
			if (radBtnMainColumn.IsChecked == true)
			{
				text2 = _colVarNames[num];
				iHNode = _context.FindNode(text2);
				text += WpfPLPetroFracResources.Title_MainCol;
			}
			else
			{
				string text3 = comboBoxStripper.Text.Trim();
				text2 = _stripperVarNames[num] + "/" + text3;
				iHNode = _context.FindNode(text2);
				text = text + WpfPLPetroFracResources.Title_Stripper + " " + text3;
			}
			_wizardPlot.SetIndependentVariable(_context, text2 + "!Label");
			if (iHNode != null)
			{
				List<string> list = new List<string>(CurveList.GetSelectedCurves());
				int count = list.Count;
				int count2 = iHNode.Elements.Count;
				_wizardPlot.PlotVal.NVariables = count;
				for (int j = 0; j < count; j++)
				{
					string text4 = list[j];
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(j, text4);
					for (int k = 0; k < count2; k++)
					{
						((IHPlotVal)_wizardPlot.PlotVal).set_YData(j, k, (short)1, (object)iHNode.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text4, Type.Missing, Type.Missing, Type.Missing, Type.Missing]);
					}
				}
			}
			_wizardPlot.PlotVal.Title = text;
			((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle((object)null, ytitle);
			_currentConfigValue = SaveConfigValue();
			_currentConfigValue = SaveConfigValue();
		}

		{
			lsTagIds.Clear();
			for (int i = 0; i < _lspwrTypes.Count; i++)
			{
				PWRPlotType val = _lspwrTypes[i];
				IRibbonPlotWizardType iplotType = null;
				_tempWizard.Clear();
				_pwrParser.FillDefaultPlotVal(val, ref _hnode, ref _tempWizard);
				if (DoesPlotTypeHaveData(val.TagId, ref iplotType, ref _tempWizard))

		private bool DoesPlotTypeHaveData(string sTagId, ref IRibbonPlotWizardType iplotType, ref IHWizardPlot iWizardPlot)
		{
			int num = 0;
			if (_dicTagIdToPlotType.ContainsKey(sTagId))
			{
				iplotType = _dicTagIdToPlotType[sTagId];
				if (iplotType == null)
				{
					return false;
				}
				object obj = iWizardPlot;
				iplotType.Context((object)_hnode, ref obj, ref num);
				if (num == 0)
				{
					return true;
				}
			}
			return false;
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = _context.FindNode(_colDepVarNames[0]);
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				ierror = 1;
				return;
			}
			iHNode = _context.FindNode(_stripDepVarNames[0]);
			_petrofracHelp.LoadColumn(iHNode, comboBoxStripper, radBtnStripper);
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.SetIndependentVariable(_context, _colIndepVarName);
			_wizardPlot.AddDependentVariable(_context, _colDepVarNames[0] + "!Label");
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				radBtnMainColumn.IsChecked = true;
				_profileRadBtnList[0].IsChecked = true;
				labTemperature.IsEnabled = false;
				comboBoxTemperature.IsEnabled = false;
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		public void UpdateWizardData()
		{
			string text = comboBoxStripper.Text.Trim();
			_wizardPlot.RemoveAllDependentVariable();
			string text2 = "";
			string depvar = "";
			string ytitle = "";
			string title = "";
			if (radBtnMainColumn.IsChecked == true)
			{
				text2 = _colIndepVarName;
				for (int i = 0; i < _profileRadBtnList.Count(); i++)
				{
					if (_profileRadBtnList[i].IsChecked == true)
					{
						switch (i)
						{
						case 0:
							depvar = _colDepVarNames[0] + "!Label";
							break;
						case 1:
							depvar = _colDepVarNames[0];
							_help.SaveUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxTemperature);
							break;
						default:
							depvar = _colDepVarNames[1];
							break;
						}
						ytitle = _strYTitle[i];
						title = _strTitle[i] + WpfPLPetroFracResources.Title_MainCol;
					}
				}
			}
			else
			{
				text2 = _stripIndepVarName + "/" + text;
				for (int j = 0; j < _profileRadBtnList.Count(); j++)
				{
					if (_profileRadBtnList[j].IsChecked == true)
					{
						switch (j)
						{
						case 0:
							depvar = _stripDepVarNames[0] + "/" + text + "!Label";
							break;
						case 1:
							depvar = _stripDepVarNames[0] + "/" + text;
							_help.SaveUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxTemperature);
							break;
						default:
							depvar = _stripDepVarNames[1] + "/" + text;
							break;
						}
						ytitle = _strYTitle[j];
						title = _strTitle[j] + WpfPLPetroFracResources.Title_Stripper + " " + text;
					}
				}
			}
			_wizardPlot.SetIndependentVariable(_context, text2);
			_wizardPlot.AddDependentVariable(_context, depvar);
			((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, ytitle);
			_wizardPlot.PlotVal.Title = title;
			_help.SaveUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxTemperature);
			_help.SaveXUnitsComboText(_context.Application.Tree, _wizardPlot, _currentExergyUnit);
			_currentConfigValue = SaveConfigValue();
		}

		private void SetUnit()
		{
			for (int i = 0; i < _profileRadBtnList.Count(); i++)
			{
				if (_profileRadBtnList[i].IsChecked != true)
				{
					continue;
				}
				if (i == 1)
				{
					labTemperature.IsEnabled = true;
					comboBoxTemperature.IsEnabled = true;
					_wizardPlot.RemoveAllDependentVariable();
					if (radBtnMainColumn.IsChecked == true)
					{
						_wizardPlot.AddDependentVariable(_context, _colDepVarNames[0]);
					}
					else
					{
						string text = comboBoxStripper.Text.Trim();
						_wizardPlot.AddDependentVariable(_context, _stripDepVarNames[0] + "/" + text);
					}
					_help.LoadUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxTemperature);
				}
				else
				{
					labTemperature.IsEnabled = false;
					comboBoxTemperature.IsEnabled = false;
				}
			}
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("ColumnOption=");
			stringBuilder.Append(radBtnMainColumn.IsChecked);
			stringBuilder.Append(";");
			stringBuilder.Append("StripperEnabled=");
			stringBuilder.Append(comboBoxStripper.IsEnabled);
			stringBuilder.Append(";");
			stringBuilder.Append("Stripper=");
			stringBuilder.Append(comboBoxStripper.Text);
			stringBuilder.Append(";");
			stringBuilder.Append("ProfileOption=");
			for (int i = 0; i < _profileRadBtnList.Count; i++)
			{
				if (_profileRadBtnList[i].IsChecked == true)
				{
					stringBuilder.Append(i.ToString());
					break;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("ProfileEnable=");
			int num = 0;
			for (int j = 0; j < _profileRadBtnList.Count; j++)
			{
				if (_profileRadBtnList[j].IsEnabled)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(j.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("Unit=");
			stringBuilder.Append(comboBoxTemperature.Text);
			stringBuilder.Append(";");
			stringBuilder.Append("UnitEnabel=");
			stringBuilder.Append(comboBoxTemperature.IsEnabled);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string value = array[num].Substring(array[num].IndexOf('=') + 1);
			radBtnMainColumn.IsChecked = Convert.ToBoolean(value);
			radBtnStripper.IsChecked = !radBtnMainColumn.IsChecked;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxStripper.IsEnabled = Convert.ToBoolean(value);
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxStripper.Text = value;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			_profileRadBtnList[Convert.ToInt16(value)].IsChecked = true;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = value.Split(',');
			foreach (string value2 in array2)
			{
				_profileRadBtnList[Convert.ToInt16(value2)].IsEnabled = true;
			}
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			SetUnit();
			comboBoxTemperature.Text = value;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxTemperature.IsEnabled = Convert.ToBoolean(value);
		}

		{

			{
				if (_wizardPlot.PlotVal.ReverseXAxis)

					sCurrentVal = _descending;
				}

				{
					sCurrentVal = _ascending;
				}
			}
			else if (sFieldName.CompareTo(_stageOnFieldId) == 0)
			{
				if (_wizardPlot.PlotVal.SwapAxises)
				{
					sCurrentVal = _yAxis;
				}
				else
				{
					sCurrentVal = _xAxis;
				}
			}
		}

				{

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			bool flag = false;
			for (int i = 0; i < _colVarNames.GetLength(0); i++)
			{
				for (int j = 0; j < _colVarNames.GetLength(1); j++)
				{
					iHNode = _context.FindNode(_colVarNames[i, j]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_flowCheckList[j].IsEnabled = true;
						_flowCheckList[j].IsChecked = true;
						_basisradBtnList[i].IsEnabled = true;
						flag = true;
					}
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			iHNode = _context.FindNode(_varName);
			_petrofracHelp.LoadColumn(iHNode, comboBoxStripper, _columnradBtnList[1]);
			if (comboBoxStripper.Items.Count > 0)
			{
				_columnradBtnList[1].IsEnabled = true;
			}
			_wizardPlot.SetIndependentVariable(_context, _colVarNames[0, 0] + "!Label");
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, _colVarNames[0, 0]);
			_wizardPlot.basis = 1;
			_help.LoadUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxUnits);
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				_columnradBtnList[0].IsEnabled = true;
				_columnradBtnList[0].IsChecked = true;
				_basisradBtnList[0].IsEnabled = true;
				_basisradBtnList[0].IsChecked = true;
				return;
			}
			_columnradBtnList[0].IsChecked = false;
			for (int k = 0; k < _flowCheckList.Count(); k++)
			{
				_flowCheckList[k].IsChecked = false;
			}
			RestoreConfigValue(_currentConfigValue);
		}

		public void UpdateWizardData()
		{
			string text = "";
			int basis = _wizardPlot.basis;
			text = basis switch
			{
				1 => WpfPLPetroFracResources.Molar, 
				2 => WpfPLPetroFracResources.Mass, 
				_ => WpfPLPetroFracResources.StdVol, 
			};
			string text2 = comboBoxStripper.Text.Trim();
			int num = 0;
			string text3 = (text3 = WpfPLPetroFracResources.Flowrate_Title);
			_wizardPlot.RemoveAllDependentVariable();
			if (_columnradBtnList[0].IsChecked == true)
			{
				_wizardPlot.PlotVal.Title = text3 + " " + WpfPLPetroFracResources.Title_MainCol;
				_wizardPlot.SetIndependentVariable(_context, _colVarNames[0, 1] + "!Label");
				for (int i = 0; i < _flowCheckList.Count; i++)
				{
					if (_flowCheckList[i].IsEnabled && _flowCheckList[i].IsChecked == true)
					{
						string depvar = _colVarNames[basis - 1, i];
						_wizardPlot.AddDependentVariable(_context, depvar);
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(num, _flowCheckList[i].Content.ToString() + " " + text);
						num++;
					}
				}
			}
			else
			{
				_wizardPlot.PlotVal.Title = text3 + " " + WpfPLPetroFracResources.Title_Stripper + " " + text2;
				_wizardPlot.SetIndependentVariable(_context, _stripperVarNames[0, 1] + "/" + text2 + "!Label");
				for (int j = 0; j < _flowCheckList.Count; j++)
				{
					if (_flowCheckList[j].IsEnabled && _flowCheckList[j].IsChecked == true)
					{
						string depvar2 = _stripperVarNames[basis - 1, j] + "/" + text2;
						_wizardPlot.AddDependentVariable(_context, depvar2);
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(num, _flowCheckList[j].Content.ToString() + " " + text);
						num++;
					}
				}
			}
			_help.SaveUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxUnits);
			_currentConfigValue = SaveConfigValue();
		}

		{
			RadioButton radioButton = sender as RadioButton;

				if (radioButton.Name.ToLower() == _basisradBtnList[i].Name.ToLower())
				{
					SetUnit(i + 1);
					break;
				}
			}
		}

		private void SetUnit(int index)
		{
			int num = 0;
			string text = comboBoxStripper.Text.Trim();
			_wizardPlot.basis = index;
			_wizardPlot.RemoveAllDependentVariable();
			if (_columnradBtnList[0].IsChecked == true)
			{
				for (int i = 0; i < _flowCheckList.Count; i++)
				{
					if (_flowCheckList[i].IsEnabled && _flowCheckList[i].IsChecked == true)
					{
						string depvar = _colVarNames[index - 1, i];
						_wizardPlot.AddDependentVariable(_context, depvar);
						num++;
					}
				}
			}
			else
			{
				for (int j = 0; j < _flowCheckList.Count; j++)
				{
					string depvar2 = _stripperVarNames[index - 1, j] + "/" + text;
					_wizardPlot.AddDependentVariable(_context, depvar2);
					num++;
				}
			}
			_help.LoadUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxUnits);
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("ColumnOption=");
			stringBuilder.Append(radBtnMainColumn.IsChecked);
			stringBuilder.Append(";");
			stringBuilder.Append("StripperEnabled=");
			stringBuilder.Append(comboBoxStripper.IsEnabled);
			stringBuilder.Append(";");
			stringBuilder.Append("Stripper=");
			stringBuilder.Append(comboBoxStripper.Text);
			stringBuilder.Append(";");
			stringBuilder.Append("BasisOption=");
			for (int i = 0; i < _basisradBtnList.Count; i++)
			{
				if (_basisradBtnList[i].IsChecked == true)
				{
					stringBuilder.Append(i.ToString());
					break;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("BasisEnable=");
			int num = 0;
			for (int j = 0; j < _basisradBtnList.Count; j++)
			{
				if (_basisradBtnList[j].IsEnabled)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(j.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("FlowEnable=");
			num = 0;
			for (int k = 0; k < _flowCheckList.Count; k++)
			{
				if (_flowCheckList[k].IsEnabled)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(k.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("FlowOption=");
			num = 0;
			for (int l = 0; l < _flowCheckList.Count; l++)
			{
				if (_flowCheckList[l].IsChecked == true)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(l.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("Unit=");
			stringBuilder.Append(comboBoxUnits.Text);
			stringBuilder.Append(";");
			stringBuilder.Append("UnitEnabel=");
			stringBuilder.Append(comboBoxUnits.IsEnabled);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string value = array[num].Substring(array[num].IndexOf('=') + 1);
			radBtnMainColumn.IsChecked = Convert.ToBoolean(value);
			radBtnStripper.IsChecked = !radBtnMainColumn.IsChecked;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxStripper.IsEnabled = Convert.ToBoolean(value);
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxStripper.Text = value;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			int num2 = Convert.ToInt16(value);
			_basisradBtnList[num2].IsChecked = true;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = value.Split(',');
			foreach (string value2 in array2)
			{
				_basisradBtnList[Convert.ToInt16(value2)].IsEnabled = true;
			}
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			array2 = value.Split(',');
			foreach (string value3 in array2)
			{
				_flowCheckList[Convert.ToInt16(value3)].IsEnabled = true;
			}
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			array2 = value.Split(',');
			foreach (string value4 in array2)
			{
				_flowCheckList[Convert.ToInt16(value4)].IsChecked = true;
			}
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			SetUnit(num2 + 1);
			comboBoxUnits.Text = value;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxUnits.IsEnabled = Convert.ToBoolean(value);
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			bool flag = false;
			for (int i = 0; i < _colVarNames.GetLength(0); i++)
			{
				flag = false;
				for (int j = 0; j < _colVarNames.GetLength(1); j++)
				{
					iHNode = _context.FindNode(_colVarNames[i, j]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_basisRadBtnList[i].IsEnabled = true;
						flag = true;
					}
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			iHNode = _context.FindNode(_varName);
			_petrofracHelp.LoadColumn(iHNode, comboBoxStripper, _columnradBtnList[1]);
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				_columnradBtnList[0].IsEnabled = true;
				_columnradBtnList[0].IsChecked = true;
				_basisRadBtnList[0].IsEnabled = true;
				_basisRadBtnList[0].IsChecked = true;
				_flowCheckList[0].IsEnabled = true;
				_flowCheckList[0].IsChecked = true;
				for (int k = 0; k < _basisRadBtnList.Count; k++)
				{
					if (_basisRadBtnList[k].IsEnabled)
					{
						_basisRadBtnList[k].IsChecked = true;
						break;
					}
				}
				if (comboBoxStripper.Items.Count > 0)
				{
					_columnradBtnList[1].IsEnabled = true;
				}
				iHNode = _context.FindNode("Input/NO_PHASE");
				if ((dynamic)iHNode.get_Value(Type.Missing) == 3)
				{
					_flowCheckList[1].IsEnabled = true;
				}
				else
				{
					_flowCheckList[1].IsEnabled = false;
				}
			}
			else
			{
				for (int l = 0; l < _flowCheckList.Count; l++)
				{
					_flowCheckList[0].IsChecked = false;
				}
				RestoreConfigValue(_currentConfigValue);
			}
			_wizardPlot.basis = 1;
		}

		public void UpdateWizardData()
		{
			string text = "";
			string text2 = "";
			int num = 0;
			for (int i = 0; i < _basisRadBtnList.Count; i++)
			{
				if (_basisRadBtnList[i].IsChecked == true)
				{
					num = i;
					text2 = _strTitle[i];
					text = _strLabel[i];
					break;
				}
			}
			_wizardPlot.basis = num + 1;
			int index = 0;
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.PlotVal.NVariables = 0;
			((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLPetroFracResources.FlowRatio + text);
			IHNode iHNode = null;
			IHNode iHNode2 = null;
			int num2 = -1;
			if (_columnradBtnList[0].IsChecked == true)
			{
				for (int j = 0; j < _colVarNames.GetLength(1); j += 2)
				{
					num2++;
					if (!_flowCheckList[num2].IsEnabled || _flowCheckList[num2].IsChecked != true)
					{
						continue;
					}
					iHNode = _context.FindNode(_colVarNames[num, j]);
					iHNode2 = _context.FindNode(_colVarNames[num, j + 1]);
					_wizardPlot.PlotVal.Title = text2 + " " + WpfPLPetroFracResources.Title_MainCol;
					_wizardPlot.SetIndependentVariable(_context, _colVarNames[num, 0] + "!Label");
					if (iHNode == null || iHNode2 == null)
					{
						continue;
					}
					_wizardPlot.PlotVal.NVariables = _wizardPlot.PlotVal.NVariables + 1;
					for (int k = 0; k < iHNode.Elements.Count; k++)
					{
						double num3 = ((double?)(dynamic)iHNode2.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing)) ?? 0.0;
						if (num3 != 0.0)
						{
							double num4 = ((double?)(dynamic)iHNode.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing)) ?? 0.0;
							((IHPlotVal)_wizardPlot.PlotVal).set_YData(index, k, (short)0, (object)(num4 / num3));
						}
					}
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(index++, _flowCheckList[num2].Content.ToString() + " " + text);
				}
			}
			else
			{
				num2 = -1;
				for (int l = 0; l < _stripperVarNames.GetLength(1); l += 2)
				{
					num2++;
					if (!_flowCheckList[num2].IsEnabled || _flowCheckList[num2].IsChecked != true)
					{
						continue;
					}
					string text3 = comboBoxStripper.Text.Trim();
					iHNode = _context.FindNode(_stripperVarNames[num, l] + "/" + text3);
					iHNode2 = _context.FindNode(_stripperVarNames[num, l + 1] + "/" + text3);
					_wizardPlot.PlotVal.Title = text2 + " " + WpfPLPetroFracResources.Title_Stripper + " " + text3;
					_wizardPlot.SetIndependentVariable(_context, _stripperVarNames[num, 0] + "/" + text3 + "!Label");
					if (iHNode == null || iHNode2 == null)
					{
						continue;
					}
					_wizardPlot.PlotVal.NVariables = _wizardPlot.PlotVal.NVariables + 1;
					for (int m = 0; m < iHNode.Elements.Count; m++)
					{
						double num5 = ((double?)(dynamic)iHNode2.Elements[m, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing)) ?? 0.0;
						if (num5 != 0.0)
						{
							double num6 = ((double?)(dynamic)iHNode.Elements[m, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing)) ?? 0.0;
							((IHPlotVal)_wizardPlot.PlotVal).set_YData(index, m, (short)0, (object)(num6 / num5));
						}
					}
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(index++, _flowCheckList[num2].Content.ToString() + " " + text);
				}
			}
			_currentConfigValue = SaveConfigValue();
		}

				if (_flowCheckList[k].IsChecked == true)

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			_context.FindNode(_YColVarNames[0, 0]);
			_context.FindNode(_XColVarNames[0, 0]);
			IHNode iHNode = null;
			bool flag = false;
			bool flag2 = false;
			for (int i = 0; i < _YColVarNames.GetLength(1); i++)
			{
				for (int j = 0; j < _YColVarNames.GetLength(0); j++)
				{
					iHNode = _context.FindNode(_YColVarNames[i, j]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_flowCheckList[j].IsEnabled = true;
						_flowCheckList[j].IsChecked = true;
						_basisradBtnList[i].IsEnabled = true;
						_phaseList[0].IsEnabled = true;
						_phaseList[0].IsChecked = true;
						flag = true;
					}
				}
			}
			for (int k = 0; k < _XColVarNames.GetLength(1); k++)
			{
				for (int l = 0; l < _XColVarNames.GetLength(0); l++)
				{
					iHNode = _context.FindNode(_XColVarNames[k, l]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_flowCheckList[l].IsEnabled = true;
						_flowCheckList[l].IsChecked = true;
						_basisradBtnList[k].IsEnabled = true;
						_phaseList[1].IsEnabled = true;
						if (!flag)
						{
							_phaseList[1].IsChecked = true;
						}
						flag2 = true;
					}
				}
			}
			if (!flag && !flag2)
			{
				ierror = 1;
				return;
			}
			iHNode = _context.FindNode(_varName);
			_petrofracHelp.LoadColumn(iHNode, comboBoxStripper, radBtnStripper);
			_wizardPlot.SetIndependentVariable(_context, _YColVarNames[0, 1] + "!Label");
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, _YColVarNames[0, 1]);
			_wizardPlot.basis = 1;
			_help.LoadUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxUnits);
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				_columnradBtnList[0].IsEnabled = true;
				_columnradBtnList[0].IsChecked = true;
				for (int m = 0; m < _basisradBtnList.Count(); m++)
				{
					if (_basisradBtnList[m].IsEnabled)
					{
						_basisradBtnList[m].IsChecked = true;
						break;
					}
				}
			}
			else
			{
				for (int n = 0; n < _flowCheckList.Count(); n++)
				{
					_flowCheckList[n].IsChecked = false;
				}
				RestoreConfigValue(_currentConfigValue);
			}
		}

		public void UpdateWizardData()
		{
			string text = "";
			int basis = _wizardPlot.basis;
			text = basis switch
			{
				1 => WpfPLPetroFracResources.Hydraulic_Title_Mole, 
				2 => WpfPLPetroFracResources.Hydraulic_Title_Mass, 
				_ => WpfPLPetroFracResources.Hydraulic_Title_StdVol, 
			};
			string text2 = comboBoxStripper.Text.Trim();
			if (radBtnVapor.IsChecked == true)
			{
				_colVarNames = _YColVarNames;
				_stripperVarNames = _YStripperVarNames;
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLPetroFracResources.Hydraulic_YAxisTitle_Vapor);
			}
			else
			{
				_colVarNames = _XColVarNames;
				_stripperVarNames = _XStripperVarNames;
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, WpfPLPetroFracResources.Hydraulic_YAxisTitle_Liquid);
			}
			int num = 0;
			_wizardPlot.RemoveAllDependentVariable();
			string text3 = "";
			string text4 = "";
			if (radBtnMainColumn.IsChecked == true)
			{
				text3 = WpfPLPetroFracResources.Title_MainCol;
				text4 = _colVarNames[basis - 1, 1] + "!Label";
				_wizardPlot.SetIndependentVariable(_context, text4);
				for (int i = 0; i < _flowCheckList.Count; i++)
				{
					if (_flowCheckList[i].IsEnabled && _flowCheckList[i].IsChecked == true)
					{
						string depvar = _colVarNames[basis - 1, i];
						_wizardPlot.AddDependentVariable(_context, depvar);
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(num, _flowCheckList[i].Content.ToString());
						num++;
					}
				}
			}
			else
			{
				text3 = WpfPLPetroFracResources.Title_Stripper + " " + text2;
				text4 = _stripperVarNames[basis - 1, 1] + "/" + text2 + "!Label";
				_wizardPlot.SetIndependentVariable(_context, text4);
				for (int j = 0; j < _flowCheckList.Count; j++)
				{
					if (_flowCheckList[j].IsEnabled && _flowCheckList[j].IsChecked == true)
					{
						string depvar2 = _stripperVarNames[basis - 1, j] + "/" + text2;
						_wizardPlot.AddDependentVariable(_context, depvar2);
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(num, _flowCheckList[j].Content.ToString());
						num++;
					}
				}
			}
			_wizardPlot.PlotVal.Title = text + " " + text3;
			_help.SaveUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxUnits);
			_currentConfigValue = SaveConfigValue();
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			bool flag = false;
			for (int i = 0; i < _colVarNames.Count(); i++)
			{
				iHNode = _context.FindNode(_colVarNames[i]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					flag = true;
					break;
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			List<string> list = new List<string>();
			CurveList.IsCheckAll = false;
			if (iHNode != null && iHNode.Elements.Count > 0)
			{
				IHNodeCol elements = iHNode.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
				if (elements != null && elements.Count > 0)
				{
					for (int j = 0; j < elements.Count; j++)
					{
						string item = elements.get_Label(0, j, (object)null);
						list.Add(item);
					}
					CurveList.SetCurves(list.ToArray());
					CurveList.Header = WpfPLPetroFracResources.Comp_Header;
				}
			}
			iHNode = _context.FindNode(_stripperVarNames[0]);
			_petrofracHelp.LoadColumn(iHNode, comboBoxStripper, radBtnStripper);
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				radBtnMainColumn.IsChecked = true;
				for (int k = 0; k < _colVarNames.Count(); k++)
				{
					iHNode = _context.FindNode(_colVarNames[k]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_phaseRadBtnList[k].IsEnabled = true;
					}
				}
				for (int l = 0; l < _phaseRadBtnList.Count(); l++)
				{
					if (_phaseRadBtnList[l].IsEnabled)
					{
						_phaseRadBtnList[l].IsChecked = true;
						iHNode = _context.FindNode(_colVarNames[l]);
						break;
					}
				}
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		public void UpdateWizardData()
		{
			string text = "";
			int num = -1;
			for (int i = 0; i < _phaseRadBtnList.Count(); i++)
			{
				if (_phaseRadBtnList[i].IsEnabled && _phaseRadBtnList[i].IsChecked == true)
				{
					num = i;
					text = _strTitle[i];
					break;
				}
			}
			_wizardPlot.RemoveAllDependentVariable();
			IHNode iHNode = null;
			if (radBtnMainColumn.IsChecked == true)
			{
				iHNode = _context.FindNode(_colVarNames[num]);
				text += WpfPLPetroFracResources.Title_MainCol;
				_wizardPlot.SetIndependentVariable(_context, _colVarNames[num] + "!Label");
			}
			else
			{
				string text2 = comboBoxStripper.Text.Trim();
				iHNode = _context.FindNode(_stripperVarNames[num] + "/" + text2);
				text = text + WpfPLPetroFracResources.Title_Stripper + " " + text2;
				_wizardPlot.SetIndependentVariable(_context, _stripperVarNames[num] + "/" + text2 + "!Label");
			}
			if (iHNode != null)
			{
				List<string> list = new List<string>(CurveList.GetSelectedCurves());
				int count = list.Count;
				int count2 = iHNode.Elements.Count;
				_wizardPlot.PlotVal.NVariables = count;
				for (int j = 0; j < count; j++)
				{
					string text3 = list[j];
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(j, text3);
					for (int k = 0; k < count2; k++)
					{
						((IHPlotVal)_wizardPlot.PlotVal).set_YData(j, k, (short)1, (object)iHNode.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text3, Type.Missing, Type.Missing, Type.Missing, Type.Missing]);
					}
				}
			}
			_wizardPlot.PlotVal.Title = text;
			((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle((object)null, WpfPLPetroFracResources.KValues_YAxisTitle);
			_currentConfigValue = SaveConfigValue();
		}

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{

			if (num == CurveList.Curves.Count)

				CurveList.IsCheckAll = true;
			}
			else
			{
				string[] selectedCurves = CurveList.GetSelectedCurves();
				CurveList.IsCheckAll = false;
				CurveList.SetSelectedCurves(selectedCurves);
			}
			e.CanExecute = num != 0;
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("ColumnOption=");
			stringBuilder.Append(radBtnMainColumn.IsChecked);
			stringBuilder.Append(";");
			stringBuilder.Append("StripperEnabled=");
			stringBuilder.Append(comboBoxStripper.IsEnabled);
			stringBuilder.Append(";");
			stringBuilder.Append("Stripper=");
			stringBuilder.Append(comboBoxStripper.Text);
			stringBuilder.Append(";");
			stringBuilder.Append("BasisOption=");
			for (int i = 0; i < _phaseRadBtnList.Count; i++)
			{
				if (_phaseRadBtnList[i].IsChecked == true)
				{
					stringBuilder.Append(i.ToString());
					break;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("BasisEnable=");
			int num = 0;
			for (int j = 0; j < _phaseRadBtnList.Count; j++)
			{
				if (_phaseRadBtnList[j].IsEnabled)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(j.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CurveSelection=");
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			int count = list.Count;
			for (int k = 0; k < count; k++)
			{
				if (k > 0)
				{
					stringBuilder.Append(",");
				}
				stringBuilder.Append(list[k]);
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CheckAll=");
			stringBuilder.Append(CurveList.IsCheckAll);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string value = array[num].Substring(array[num].IndexOf('=') + 1);
			radBtnMainColumn.IsChecked = Convert.ToBoolean(value);
			radBtnStripper.IsChecked = !radBtnMainColumn.IsChecked;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxStripper.IsEnabled = Convert.ToBoolean(value);
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxStripper.Text = value;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			_phaseRadBtnList[Convert.ToInt16(value)].IsChecked = true;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = value.Split(',');
			foreach (string value2 in array2)
			{
				_phaseRadBtnList[Convert.ToInt16(value2)].IsEnabled = true;
			}
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array3 = value.Split(',');
			List<string> list = new List<string>();
			array2 = array3;
			foreach (string item in array2)
			{
				list.Add(item);
			}
			CurveList.SetSelectedCurves(list.ToArray());
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			CurveList.IsCheckAll = Convert.ToBoolean(value);
		}
