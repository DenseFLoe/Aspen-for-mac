		public IBrowserSettings BrowserSettings
		{
			get
			{
				return browserSettings;
			}
			set
			{
				if (browserCreated)
				{
					throw new Exception("Browser has already been created. BrowserSettings must be set before the underlying CEF browser is created.");
				}
				if (browserSettings != null && browserSettings.AutoDispose)
				{
					browserSettings.Dispose();
				}
				browserSettings = value;
			}
		}

			}
		}

		static ChromiumWebBrowser()
		{
			CanGoBackProperty = DependencyProperty.Register("CanGoBack", typeof(bool), typeof(ChromiumWebBrowser));
			CanGoForwardProperty = DependencyProperty.Register("CanGoForward", typeof(bool), typeof(ChromiumWebBrowser));
			AddressProperty = DependencyProperty.Register("Address", typeof(string), typeof(ChromiumWebBrowser), new UIPropertyMetadata(null, OnAddressChanged));
			IsLoadingProperty = DependencyProperty.Register("IsLoading", typeof(bool), typeof(ChromiumWebBrowser), new PropertyMetadata(false));
			IsBrowserInitializedProperty = DependencyProperty.Register("IsBrowserInitialized", typeof(bool), typeof(ChromiumWebBrowser), new PropertyMetadata(false, OnIsBrowserInitializedChanged));
			TitleProperty = DependencyProperty.Register("Title", typeof(string), typeof(ChromiumWebBrowser), new PropertyMetadata(null, OnTitleChanged));
			ZoomLevelProperty = DependencyProperty.Register("ZoomLevel", typeof(double), typeof(ChromiumWebBrowser), new UIPropertyMetadata(0.0, OnZoomLevelChanged));
			ZoomLevelIncrementProperty = DependencyProperty.Register("ZoomLevelIncrement", typeof(double), typeof(ChromiumWebBrowser), new PropertyMetadata(0.1));
			CleanupElementProperty = DependencyProperty.Register("CleanupElement", typeof(FrameworkElement), typeof(ChromiumWebBrowser), new PropertyMetadata(null, OnCleanupElementChanged));
			TooltipTextProperty = DependencyProperty.Register("TooltipText", typeof(string), typeof(ChromiumWebBrowser), new PropertyMetadata(null, OnTooltipTextChanged));
			WebBrowserProperty = DependencyProperty.Register("WebBrowser", typeof(IWebBrowser), typeof(ChromiumWebBrowser), new UIPropertyMetadata((object)null));
			FrameworkElement.DefaultStyleKeyProperty.OverrideMetadata(typeof(ChromiumWebBrowser), new FrameworkPropertyMetadata(typeof(ChromiumWebBrowser)));
			if (!CefSharpSettings.ShutdownOnExit)
			{
				return;
			}
			Dispatcher dispatcher = Dispatcher.FromThread(Thread.CurrentThread);
			if (dispatcher == null)
			{
				Application current = Application.Current;
				if (current != null)
				{
					current.Exit += OnApplicationExit;
				}
			}
			else
			{
				dispatcher.ShutdownStarted += DispatcherShutdownStarted;
				dispatcher.ShutdownFinished += DispatcherShutdownFinished;
			}
		}

		private static void DispatcherShutdownStarted(object sender, EventArgs e)
		{
			Dispatcher dispatcher = (Dispatcher)sender;
			dispatcher.ShutdownStarted -= DispatcherShutdownStarted;
			if (!DesignMode)
			{
				CefPreShutdown();
			}
		}

		private static void DispatcherShutdownFinished(object sender, EventArgs e)
		{
			Dispatcher dispatcher = (Dispatcher)sender;
			dispatcher.ShutdownFinished -= DispatcherShutdownFinished;
			if (!DesignMode)
			{
				CefShutdown();
			}
		}

		private static void OnApplicationExit(object sender, ExitEventArgs e)
		{
			if (!DesignMode)
			{
				CefShutdown();
			}
		}

		public static void UnregisterShutdownHandler()
		{
			Dispatcher dispatcher = Dispatcher.FromThread(Thread.CurrentThread);
			if (dispatcher != null)
			{
				dispatcher.ShutdownStarted -= DispatcherShutdownStarted;
				dispatcher.ShutdownFinished -= DispatcherShutdownFinished;
			}
			Application current = Application.Current;
			if (current != null)
			{
				current.Exit -= OnApplicationExit;
			}
		}

		{
			if (!Cef.IsInitialized)
			{
				CefSettings settings = new CefSettings();

					throw new InvalidOperationException("Cef.Initialize() failed.Check the log file see https://github.com/cefsharp/CefSharp/wiki/Trouble-Shooting#log-file for details.");
				}
			}
			Cef.AddDisposable(this);
			base.Focusable = true;
			base.FocusVisualStyle = null;
			base.IsTabStop = true;
			base.Dispatcher.BeginInvoke((Action)delegate
			{
				WebBrowser = this;
			});
			base.Loaded += OnLoaded;
			base.SizeChanged += OnActualSizeChanged;
			base.GotKeyboardFocus += OnGotKeyboardFocus;
			base.LostKeyboardFocus += OnLostKeyboardFocus;
			base.DragEnter += OnDragEnter;
			base.DragOver += OnDragOver;
			base.DragLeave += OnDragLeave;
			base.Drop += OnDrop;
			base.IsVisibleChanged += OnIsVisibleChanged;
			base.ToolTip = (toolTip = new ToolTip());

			toolTip.Visibility = Visibility.Collapsed;
			toolTip.Closed += OnTooltipClosed;

			ForwardCommand = new DelegateCommand(this.Forward, () => CanGoForward);
			ReloadCommand = new DelegateCommand(((IChromiumWebBrowserBase)this).Reload, () => !IsLoading);

		protected virtual void Dispose(bool disposing)
		{
			if (Interlocked.CompareExchange(ref disposeSignaled, 1, 0) == 0 && !DesignMode)
			{
				InternalDispose(disposing);
			}
		}

		[MethodImpl(MethodImplOptions.NoInlining)]
		private void InternalDispose(bool disposing)
		{
			if (disposing)
			{
				CanExecuteJavascriptInMainFrame = false;
				Interlocked.Exchange(ref browserInitialized, 0);
				browser?.GetHost().WasHidden(hidden: true);
				UiThreadRunAsync(delegate
				{
					SetCurrentValue(IsBrowserInitializedProperty, false);
					WebBrowser = null;
				});
				this.ConsoleMessage = null;
				this.FrameLoadEnd = null;
				this.FrameLoadStart = null;
				this.AddressChanged = null;
				this.IsBrowserInitializedChanged = null;
				this.LoadError = null;
				this.LoadingStateChanged = null;
				this.Paint = null;
				this.StatusMessage = null;
				this.TitleChanged = null;
				this.VirtualKeyboardRequested = null;
				this.JavascriptMessageReceived = null;
				FreeHandlersExceptLifeSpanAndFocus();
				FocusHandler = new NoFocusHandler();
				browser = null;
				BrowserCore = null;
				currentDragData?.Dispose();
				currentDragData = null;
				PresentationSource.RemoveSourceChangedHandler(this, PresentationSourceChangedHandler);
				if (sourceWindow != null)
				{
					sourceWindow.StateChanged -= OnWindowStateChanged;
					sourceWindow.LocationChanged -= OnWindowLocationChanged;
					sourceWindow = null;
				}
				base.Loaded -= OnLoaded;
				base.SizeChanged -= OnActualSizeChanged;
				base.GotKeyboardFocus -= OnGotKeyboardFocus;
				base.LostKeyboardFocus -= OnLostKeyboardFocus;
				base.DragEnter -= OnDragEnter;
				base.DragOver -= OnDragOver;
				base.DragLeave -= OnDragLeave;
				base.Drop -= OnDrop;
				base.IsVisibleChanged -= OnIsVisibleChanged;
				if (tooltipTimer != null)
				{
					tooltipTimer.Tick -= OnTooltipTimerTick;
					tooltipTimer.Stop();
					tooltipTimer = null;
				}
				if (CleanupElement != null)
				{
					CleanupElement.Unloaded -= OnCleanupElementUnloaded;
				}
				managedCefBrowserAdapter?.Dispose();
				managedCefBrowserAdapter = null;
				LifeSpanHandler = null;
				WpfKeyboardHandler?.Dispose();
				WpfKeyboardHandler = null;
				IRenderHandler renderHandler = RenderHandler;
				RenderHandler = null;
				renderHandler?.Dispose();
				source = null;
			}
			Cef.RemoveDisposable(this);
		}

		ScreenInfo? IRenderWebBrowser.GetScreenInfo()
		{
			return GetScreenInfo();
		}

		protected virtual ScreenInfo? GetScreenInfo()
		{
			CefSharp.Structs.Rect value = monitorInfo.Monitor;
			CefSharp.Structs.Rect value2 = monitorInfo.WorkArea;
			if ((double)DpiScaleFactor > 1.0)
			{
				value = value.ScaleByDpi(DpiScaleFactor);
				value2 = value2.ScaleByDpi(DpiScaleFactor);
			}
			return new ScreenInfo
			{
				DeviceScaleFactor = DpiScaleFactor,
				Rect = value,
				AvailableRect = value2
			};
		}

		CefSharp.Structs.Rect IRenderWebBrowser.GetViewRect()
		{
			return GetViewRect();
		}

		protected virtual CefSharp.Structs.Rect GetViewRect()
		{
			CefSharp.Structs.Size? size = resizeHackSize;
			if (!size.HasValue)
			{
				return viewRect;
			}
			CefSharp.Structs.Size value = size.Value;
			return new CefSharp.Structs.Rect(0, 0, value.Width, value.Height);
		}

		bool IRenderWebBrowser.GetScreenPoint(int viewX, int viewY, out int screenX, out int screenY)
		{
			return GetScreenPoint(viewX, viewY, out screenX, out screenY);
		}

		protected virtual bool GetScreenPoint(int viewX, int viewY, out int screenX, out int screenY)
		{
			screenX = 0;
			screenY = 0;
			if (DpiScaleFactor > 1f)
			{
				screenX = (int)(browserScreenLocation.X + (double)((float)viewX * DpiScaleFactor));
				screenY = (int)(browserScreenLocation.Y + (double)((float)viewY * DpiScaleFactor));
			}
			else
			{
				screenX = (int)(browserScreenLocation.X + (double)viewX);
				screenY = (int)(browserScreenLocation.Y + (double)viewY);
			}
			return true;
		}

		bool IRenderWebBrowser.StartDragging(IDragData dragData, DragOperationsMask allowedOps, int x, int y)
		{
			return StartDragging(dragData, allowedOps, x, y);
		}

		protected virtual bool StartDragging(IDragData dragData, DragOperationsMask allowedOps, int x, int y)
		{
			DataObject dataObject = new DataObject();
			dataObject.SetText(dragData.FragmentText, TextDataFormat.Text);
			dataObject.SetText(dragData.FragmentText, TextDataFormat.UnicodeText);
			dataObject.SetText(dragData.FragmentHtml, TextDataFormat.Html);
			currentDragData = dragData.Clone();
			currentDragData.ResetFileContents();
			UiThreadRunAsync(delegate
			{
				if (browser != null)
				{
					DragDropEffects dragDropEffects = DragDrop.DoDragDrop(this, dataObject, allowedOps.GetDragEffects());
					currentDragData = null;
					browser.GetHost().DragSourceEndedAt(x, y, dragDropEffects.GetDragOperationsMask());
					browser.GetHost().DragSourceSystemDragEnded();
				}
			});
			return true;
		}

		{

			{
				return;
			}

			zoomLevelAsync.ContinueWith(delegate(Task<double> previous)
			{

				}
				throw new InvalidOperationException("Unexpected failure of calling CEF->GetZoomLevelAsync", previous.Exception);
			}, TaskContinuationOptions.ExecuteSynchronously);
		}

		private void PresentationSourceChangedHandler(object sender, SourceChangedEventArgs args)
		{
			if (args.NewSource != null)
			{
				source = (HwndSource)args.NewSource;
				WpfKeyboardHandler.Setup(source);
				NotifyDpiChange((float)source.CompositionTarget.TransformToDevice.M11);
				if (source.RootVisual is Window window)
				{
					window.StateChanged += OnWindowStateChanged;
					window.LocationChanged += OnWindowLocationChanged;
					sourceWindow = window;
					if (CleanupElement == null)
					{
						CleanupElement = window;
					}
					else if (CleanupElement is Window window2 && window2 != window)
					{
						CleanupElement = window;
					}
				}
				browserScreenLocation = GetBrowserScreenLocation();
				monitorInfo.Init();
				MonitorInfo.GetMonitorInfoForWindowHandle(source.Handle, ref monitorInfo);
			}
			else if (args.OldSource != null)
			{
				WpfKeyboardHandler.Dispose();
				if (args.OldSource.RootVisual is Window window3)
				{
					window3.StateChanged -= OnWindowStateChanged;
					window3.LocationChanged -= OnWindowLocationChanged;
					sourceWindow = null;
				}
			}
		}

		private void OnWindowStateChanged(object sender, EventArgs e)
		{
			Window window = (Window)sender;
			switch (window.WindowState)
			{
			case WindowState.Normal:
			case WindowState.Maximized:
				if (previousWindowState == WindowState.Minimized && base.IsVisible)
				{
					OnBrowserWasHidden(hidden: false);
				}
				break;
			case WindowState.Minimized:
				OnBrowserWasHidden(hidden: true);
				break;
			}
			previousWindowState = window.WindowState;
		}

		protected virtual void OnBrowserWasHidden(bool hidden)

			if (browser == null)

		}

		public bool CreateBrowser(HwndSource parentWindowHwndSource, System.Windows.Size initialSize)
		{
			if (initialSize.IsEmpty || initialSize.Equals(new System.Windows.Size(0.0, 0.0)))
			{
				throw new Exception("Invalid size, must be greater than 0,0");
			}
			source = parentWindowHwndSource;
			if (!CreateOffscreenBrowser(initialSize))
			{
				return false;
			}
			RoutedEventHandler handler = null;
			handler = delegate
			{
				base.Loaded -= handler;
				this.GetBrowserHost()?.Invalidate(PaintElementType.View);
			};
			base.Loaded += handler;
			return true;
		}

		protected virtual bool CreateOffscreenBrowser(System.Windows.Size size)
		{
			if (browserCreated || size.IsEmpty || size.Equals(new System.Windows.Size(0.0, 0.0)))
			{
				return false;
			}
			if (!((IWebBrowserInternal)this).HasParent)
			{
				IWindowInfo windowInfo = CreateOffscreenBrowserWindowInfo((source == null) ? IntPtr.Zero : source.Handle);
				managedCefBrowserAdapter.CreateBrowser(windowInfo, browserSettings, requestContext, initialAddress);
				if (browserSettings.AutoDispose)
				{
					browserSettings.Dispose();
				}
				browserSettings = null;
			}
			browserCreated = true;
			return true;
		}

		protected virtual IWindowInfo CreateOffscreenBrowserWindowInfo(IntPtr handle)
		{
			IWindowInfo windowInfo = ObjectFactory.CreateWindowInfo();
			windowInfo.SetAsWindowless(handle);
			return windowInfo;
		}

		protected virtual void OpenOrCloseToolTip(string text)
		{
			if (string.IsNullOrEmpty(text))
			{
				if (toolTip.IsOpen)
				{
					toolTip.IsOpen = false;
				}
				return;
			}
			if (toolTip.IsOpen)
			{
				toolTip.IsOpen = false;
			}
			if (toolTip.Style == null)
			{
				toolTip.Content = new TextBlock
				{
					Text = text,
					TextWrapping = TextWrapping.Wrap
				};
			}
			else
			{
				toolTip.SetCurrentValue(ContentControl.ContentProperty, text);
			}
			toolTip.Placement = PlacementMode.Mouse;
			toolTip.Visibility = Visibility.Visible;
			toolTip.IsOpen = true;
		}

		private void OnMouseButton(MouseButtonEventArgs e)
		{
			if (e.Handled || browser == null)
			{
				return;
			}
			CefEventFlags modifiers = e.GetModifiers();
			bool flag = e.ButtonState == MouseButtonState.Released;
			System.Windows.Point position = e.GetPosition(this);
			if (e.ChangedButton == MouseButton.XButton1)
			{
				if (CanGoBack && flag)
				{
					this.Back();
				}
			}
			else if (e.ChangedButton == MouseButton.XButton2)
			{
				if (CanGoForward && flag)
				{
					this.Forward();
				}
			}
			else
			{
				int num = e.ClickCount;
				if (num > 3)
				{
					num = 1;
				}
				browser.GetHost().SendMouseClickEvent((int)position.X, (int)position.Y, (MouseButtonType)e.ChangedButton, flag, num, modifiers);
			}
			e.Handled = true;
		}

		private void OnTouch(TouchEventArgs e)
		{
			if (!e.Handled && browser != null)
			{
				CefEventFlags modifierKeys = WpfExtensions.GetModifierKeys();
				TouchPoint touchPoint = e.GetTouchPoint(this);
				TouchEventType touchEventType = TouchEventType.Cancelled;
				touchEventType = touchPoint.Action switch
				{
					TouchAction.Down => TouchEventType.Pressed, 
					TouchAction.Move => TouchEventType.Moved, 
					TouchAction.Up => TouchEventType.Released, 
					_ => TouchEventType.Cancelled, 
				};
				TouchEvent evt = new TouchEvent
				{
					Id = e.TouchDevice.Id,
					X = (float)touchPoint.Position.X,
					Y = (float)touchPoint.Position.Y,
					PointerType = PointerType.Touch,
					Type = touchEventType,
					Modifiers = modifierKeys
				};
				browser.GetHost().SendTouchEvent(evt);
				e.Handled = true;

		public virtual void NotifyDpiChange(float newDpi)
		{
			if (DpiScaleFactor.Equals(newDpi))
			{
				return;
			}
			bool flag = DpiScaleFactor > 0f && !DpiScaleFactor.Equals(newDpi);
			DpiScaleFactor = newDpi;
			if (flag && browser != null)
			{
				browser.GetHost().NotifyScreenInfoChanged();
			}
			if (RenderHandler == null || RenderHandler is WritableBitmapRenderHandler || RenderHandler is DirectWritableBitmapRenderHandler)
			{
				float num = 96f * DpiScaleFactor;
				IRenderHandler renderHandler = RenderHandler;
				if (Cef.CurrentlyOnThread(CefThreadIds.TID_UI))
				{
					RenderHandler = new DirectWritableBitmapRenderHandler(num, num);
				}
				else
				{
					RenderHandler = new WritableBitmapRenderHandler(num, num);
				}
				renderHandler?.Dispose();
			}
		}

		private void InitialLoad(bool? isLoading, CefErrorCode? errorCode)
		{
			if (IsDisposed)
			{
				initialLoadAction = null;
				initialLoadTaskCompletionSource.TrySetCanceled();
			}
			else if (isLoading.HasValue)
			{
				if (!isLoading.Value)
				{
					initialLoadAction = null;
					int num = ((browser?.GetHost())?.GetVisibleNavigationEntry())?.HttpStatusCode ?? (-1);
					if (num == 0)
					{
						num = -1;
					}
					initialLoadTaskCompletionSource.TrySetResultAsync(new LoadUrlAsyncResponse(CefErrorCode.None, num));
				}
			}
			else if (errorCode.HasValue && errorCode != CefErrorCode.Aborted)
			{
				initialLoadAction = null;
				initialLoadTaskCompletionSource.TrySetResultAsync(new LoadUrlAsyncResponse(errorCode.Value, -1));
			}
		}

	public abstract class AbstractRenderHandler : IDisposable, IRenderHandler
	{
		internal static readonly PixelFormat PixelFormat = PixelFormats.Pbgra32;

		internal static int BytesPerPixel = PixelFormat.BitsPerPixel / 8;

		protected object lockObject = new object();

		protected System.Windows.Size viewSize;

		protected System.Windows.Size popupSize;

		protected DispatcherPriority dispatcherPriority;

		protected MemoryMappedFile viewMemoryMappedFile;

		protected MemoryMappedFile popupMemoryMappedFile;

		protected MemoryMappedViewAccessor viewMemoryMappedViewAccessor;

		protected MemoryMappedViewAccessor popupMemoryMappedViewAccessor;

		private int disposeSignaled;

		public bool IsDisposed => Interlocked.CompareExchange(ref disposeSignaled, 1, 1) == 1;

		public void Dispose()
		{
			Dispose(disposing: true);
			GC.SuppressFinalize(this);
		}

		protected virtual void Dispose(bool disposing)
		{
			if (Interlocked.CompareExchange(ref disposeSignaled, 1, 0) != 0 || !disposing)
			{
				return;
			}
			lock (lockObject)
			{
				ReleaseMemoryMappedView(ref popupMemoryMappedFile, ref popupMemoryMappedViewAccessor);
				ReleaseMemoryMappedView(ref viewMemoryMappedFile, ref viewMemoryMappedViewAccessor);
			}
		}

		protected void ReleaseMemoryMappedView(ref MemoryMappedFile mappedFile, ref MemoryMappedViewAccessor stream)
		{
			if (stream != null)
			{
				stream.Dispose();
				stream = null;
			}
			if (mappedFile != null)
			{
				mappedFile.Dispose();
				mappedFile = null;
			}
		}

		public virtual void OnAcceleratedPaint(bool isPopup, CefSharp.Structs.Rect dirtyRect, IntPtr sharedHandle)
		{
		}

		public virtual void OnPaint(bool isPopup, CefSharp.Structs.Rect dirtyRect, IntPtr buffer, int width, int height, Image image)
		{
			if (!IsDisposed && !image.Dispatcher.HasShutdownStarted)
			{
				if (isPopup)
				{
					CreateOrUpdateBitmap(isPopup, dirtyRect, buffer, width, height, image, ref popupSize, ref popupMemoryMappedFile, ref popupMemoryMappedViewAccessor);
				}
				else
				{
					CreateOrUpdateBitmap(isPopup, dirtyRect, buffer, width, height, image, ref viewSize, ref viewMemoryMappedFile, ref viewMemoryMappedViewAccessor);
				}
			}
		}

		protected abstract void CreateOrUpdateBitmap(bool isPopup, CefSharp.Structs.Rect dirtyRect, IntPtr buffer, int width, int height, Image image, ref System.Windows.Size currentSize, ref MemoryMappedFile mappedFile, ref MemoryMappedViewAccessor viewAccessor);
	}
	public class AllocHGlobalWritableBitmapRenderHandler : IRenderHandler, IDisposable
	{
		private class PaintElement
		{
			private readonly double dpiX;

			private readonly double dpiY;

			private Image image;

			private int width;

			private int height;

			private CefSharp.Structs.Rect dirtyRect;

			private IntPtr buffer;

			private int bufferSize;

			private int imageSize;

			private readonly bool invalidateDirtyRect;

			internal bool IsDirty { get; set; }

			internal PaintElement(double dpiX, double dpiY, bool invalidateDirtyRect)
			{
				this.dpiX = dpiX;
				this.dpiY = dpiY;
				this.invalidateDirtyRect = invalidateDirtyRect;
			}

			internal void Dispose()
			{
				image = null;
				if (buffer != IntPtr.Zero)
				{
					Marshal.FreeHGlobal(buffer);
					buffer = IntPtr.Zero;
				}
			}

			internal void UpdateBuffer(CefSharp.Structs.Rect dirtyRect, IntPtr sourceBuffer, int width, int height, Image image)
			{
				imageSize = width * height * AbstractRenderHandler.BytesPerPixel;
				if (bufferSize < imageSize)
				{
					Marshal.FreeHGlobal(buffer);
					buffer = Marshal.AllocHGlobal(imageSize);
					bufferSize = imageSize;
				}
				this.width = width;
				this.height = height;
				this.dirtyRect = dirtyRect;
				NativeMethodWrapper.MemoryCopy(buffer, sourceBuffer, imageSize);
				this.image = image;
				IsDirty = true;
			}

			internal void UpdateImage(object lockObject)
			{
				image.Dispatcher.BeginInvoke((Action)delegate
				{
					lock (lockObject)
					{
						if (IsDirty && image != null)
						{
							WriteableBitmap writeableBitmap = image.Source as WriteableBitmap;
							if (writeableBitmap == null || writeableBitmap.PixelWidth != width || writeableBitmap.PixelHeight != height)
							{
								if (image.Source != null)
								{
									image.Source = null;
									GC.Collect(1);
								}
								writeableBitmap = (WriteableBitmap)(image.Source = new WriteableBitmap(width, height, dpiX, dpiY, AbstractRenderHandler.PixelFormat, null));
							}
							if (writeableBitmap != null)
							{
								if (invalidateDirtyRect)
								{
									Int32Rect sourceRect = new Int32Rect(dirtyRect.X, dirtyRect.Y, dirtyRect.Width, dirtyRect.Height);
									writeableBitmap.Lock();
									writeableBitmap.WritePixels(sourceRect, buffer, imageSize, width * AbstractRenderHandler.BytesPerPixel, sourceRect.X, sourceRect.Y);
									writeableBitmap.Unlock();
								}
								else
								{
									Int32Rect sourceRect2 = new Int32Rect(0, 0, width, height);
									writeableBitmap.Lock();
									writeableBitmap.WritePixels(sourceRect2, buffer, imageSize, width * AbstractRenderHandler.BytesPerPixel, sourceRect2.X, sourceRect2.Y);
									writeableBitmap.Unlock();
								}
							}
							IsDirty = false;
						}
					}
				});
			}
		}

		private readonly double dpiX;

		private readonly double dpiY;

		private readonly PaintElement view;

		private readonly PaintElement popup;

		private readonly DispatcherPriority dispatcherPriority;

		private readonly object lockObject = new object();

		private int disposeSignaled;

		public bool IsDisposed => Interlocked.CompareExchange(ref disposeSignaled, 1, 1) == 1;

		public AllocHGlobalWritableBitmapRenderHandler(double dpiX, double dpiY, bool invalidateDirtyRect = true, DispatcherPriority dispatcherPriority = DispatcherPriority.Render)
		{
			this.dpiX = dpiX;
			this.dpiY = dpiY;
			this.dispatcherPriority = dispatcherPriority;
			view = new PaintElement(dpiX, dpiY, invalidateDirtyRect);
			popup = new PaintElement(dpiX, dpiY, invalidateDirtyRect);
		}

		public void Dispose()
		{
			Dispose(disposing: true);
			GC.SuppressFinalize(this);
		}

		protected virtual void Dispose(bool disposing)
		{
			if (Interlocked.CompareExchange(ref disposeSignaled, 1, 0) != 0 || !disposing)
			{
				return;
			}
			lock (lockObject)
			{
				view?.Dispose();
				popup?.Dispose();
			}
		}

		void IRenderHandler.OnAcceleratedPaint(bool isPopup, CefSharp.Structs.Rect dirtyRect, IntPtr sharedHandle)
		{
		}

		void IRenderHandler.OnPaint(bool isPopup, CefSharp.Structs.Rect dirtyRect, IntPtr buffer, int width, int height, Image image)
		{
			lock (lockObject)
			{
				if (!IsDisposed && !image.Dispatcher.HasShutdownStarted)
				{
					PaintElement paintElement = (isPopup ? popup : view);
					paintElement?.UpdateBuffer(dirtyRect, buffer, width, height, image);
					paintElement?.UpdateImage(lockObject);
				}
			}
		}
	}

		public DirectWritableBitmapRenderHandler(double dpiX, double dpiY, bool invalidateDirtyRect = true, DispatcherPriority dispatcherPriority = DispatcherPriority.Render)
		{

			}

		void IRenderHandler.OnPaint(bool isPopup, CefSharp.Structs.Rect dirtyRect, IntPtr buffer, int width, int height, Image image)
		{
			WriteableBitmap writeableBitmap = image.Source as WriteableBitmap;
			if (writeableBitmap == null || writeableBitmap.PixelWidth != width || writeableBitmap.PixelHeight != height)
			{
				writeableBitmap = (WriteableBitmap)(image.Source = new WriteableBitmap(width, height, dpiX, dpiY, AbstractRenderHandler.PixelFormat, null));
			}
			if (writeableBitmap != null)
			{
				writeableBitmap.Lock();
				NativeMethodWrapper.MemoryCopy(writeableBitmap.BackBuffer, buffer, writeableBitmap.BackBufferStride * writeableBitmap.PixelHeight);
				if (invalidateDirtyRect)
				{
					writeableBitmap.AddDirtyRect(new Int32Rect(dirtyRect.X, dirtyRect.Y, dirtyRect.Width, dirtyRect.Height));
				}
				else
				{
					writeableBitmap.AddDirtyRect(new Int32Rect(0, 0, writeableBitmap.PixelWidth, writeableBitmap.PixelHeight));
				}
				writeableBitmap.Unlock();
			}
		}

	public class InteropBitmapRenderHandler : AbstractRenderHandler
	{
		public InteropBitmapRenderHandler(DispatcherPriority dispatcherPriority = DispatcherPriority.Render)
		{

		}

		protected override void CreateOrUpdateBitmap(bool isPopup, CefSharp.Structs.Rect dirtyRect, IntPtr buffer, int width, int height, Image image, ref System.Windows.Size currentSize, ref MemoryMappedFile mappedFile, ref MemoryMappedViewAccessor viewAccessor)
		{
			bool createNewBitmap = false;
			lock (lockObject)
			{
				int num = width * height;
				int num2 = num * AbstractRenderHandler.BytesPerPixel;
				createNewBitmap = mappedFile == null || currentSize.Height != (double)height || currentSize.Width != (double)width;
				if (createNewBitmap)
				{
					ReleaseMemoryMappedView(ref mappedFile, ref viewAccessor);
					mappedFile = MemoryMappedFile.CreateNew(null, num2, MemoryMappedFileAccess.ReadWrite);
					viewAccessor = mappedFile.CreateViewAccessor();
					currentSize.Height = height;
					currentSize.Width = width;
				}
				NativeMethodWrapper.MemoryCopy(viewAccessor.SafeMemoryMappedViewHandle.DangerousGetHandle(), buffer, num2);
				SafeMemoryMappedFileHandle backBufferHandle = mappedFile.SafeMemoryMappedFileHandle;
				image.Dispatcher.BeginInvoke((Action)delegate
				{
					lock (lockObject)
					{
						if (!backBufferHandle.IsClosed && !backBufferHandle.IsInvalid)

			internal int Width { get; private set; }

			internal int BufferSize { get; private set; }

			internal PaintElement(double dpiX, double dpiY)
			{
				DpiX = dpiX;
				DpiY = dpiY;
			}

			internal void Dispose()
			{
				if (Buffer != IntPtr.Zero)
				{
					Marshal.FreeHGlobal(Buffer);
					Buffer = IntPtr.Zero;
				}
			}

			internal void OnPaint(CefSharp.Structs.Rect dirtyRect, IntPtr sourceBuffer, int width, int height, Image image)
			{
				ImageSize = width * height * AbstractRenderHandler.BytesPerPixel;
				if (BufferSize < ImageSize)
				{
					Marshal.FreeHGlobal(Buffer);
					Buffer = Marshal.AllocHGlobal(ImageSize);
					BufferSize = ImageSize;
				}
				Width = width;
				Height = height;
				DirtyRect = dirtyRect;
				NativeMethodWrapper.MemoryCopy(Buffer, sourceBuffer, ImageSize);
				Image = image;
				IsDirty = true;
			}
		}

		private PaintElement view;

		private PaintElement popup;

		private readonly object lockObj = new object();

		private ChromiumWebBrowser browser;

		public CompositionTargetRenderHandler(ChromiumWebBrowser browser, double dpiX, double dpiY)
		{
			this.browser = browser;
			this.browser.IsVisibleChanged += BrowserIsVisibleChanged;
			view = new PaintElement(dpiX, dpiY);
			popup = new PaintElement(dpiX, dpiY);
			if (browser.IsVisible)
			{
				CompositionTarget.Rendering += OnRendering;
			}
		}

		private void BrowserIsVisibleChanged(object sender, DependencyPropertyChangedEventArgs e)
		{
			if ((bool)e.NewValue)
			{
				CompositionTarget.Rendering += OnRendering;
			}
			else
			{
				CompositionTarget.Rendering -= OnRendering;
			}
		}

		void IDisposable.Dispose()
		{
			CompositionTarget.Rendering -= OnRendering;
			if (browser != null)
			{
				browser.IsVisibleChanged -= BrowserIsVisibleChanged;
				browser = null;
			}
			lock (lockObj)
			{
				view?.Dispose();
				view = null;
				popup?.Dispose();
				popup = null;
			}
		}

		void IRenderHandler.OnAcceleratedPaint(bool isPopup, CefSharp.Structs.Rect dirtyRect, IntPtr sharedHandle)
		{
			throw new NotImplementedException();
		}

		void IRenderHandler.OnPaint(bool isPopup, CefSharp.Structs.Rect dirtyRect, IntPtr buffer, int width, int height, Image image)
		{
			if (image.Dispatcher.HasShutdownStarted)
			{
				return;
			}
			lock (lockObj)
			{
				(isPopup ? popup : view)?.OnPaint(dirtyRect, buffer, width, height, image);
			}
		}

		private void OnRendering(object sender, EventArgs args)
		{
			lock (lockObj)
			{
				UpdateImage(view);
				UpdateImage(popup);
			}
		}

		private void UpdateImage(PaintElement element)
		{
			if (element != null && element.IsDirty && element.Image != null)
			{
				WriteableBitmap writeableBitmap = element.Image.Source as WriteableBitmap;
				if (writeableBitmap == null || writeableBitmap.PixelWidth != element.Width || writeableBitmap.PixelHeight != element.Height)
				{
					writeableBitmap = (WriteableBitmap)(element.Image.Source = new WriteableBitmap(element.Width, element.Height, element.DpiX, element.DpiY, AbstractRenderHandler.PixelFormat, null));
				}
				if (writeableBitmap != null)
				{
					CefSharp.Structs.Rect dirtyRect = element.DirtyRect;
					Int32Rect sourceRect = new Int32Rect(dirtyRect.X, dirtyRect.Y, dirtyRect.Width, dirtyRect.Height);
					writeableBitmap.Lock();
					writeableBitmap.WritePixels(sourceRect, element.Buffer, element.ImageSize, element.Width * AbstractRenderHandler.BytesPerPixel, sourceRect.X, sourceRect.Y);
					writeableBitmap.Unlock();
				}
				element.IsDirty = false;
			}
		}
	}

		{

			if ((mask & DragOperationsMask.Move) == DragOperationsMask.Move)

				return DragDropEffects.Move;

				}

	public static class WpfExtensions
	{
		public static CefEventFlags GetModifiers(this MouseEventArgs e)
		{
			CefEventFlags cefEventFlags = CefEventFlags.None;
			if (e.LeftButton == MouseButtonState.Pressed)
			{
				cefEventFlags |= CefEventFlags.LeftMouseButton;
			}
			if (e.MiddleButton == MouseButtonState.Pressed)
			{
				cefEventFlags |= CefEventFlags.MiddleMouseButton;
			}
			if (e.RightButton == MouseButtonState.Pressed)
			{
				cefEventFlags |= CefEventFlags.RightMouseButton;
			}
			return cefEventFlags | GetModifierKeys(cefEventFlags);
		}

		public static CefEventFlags GetModifiers(this DragEventArgs e)
		{
			return GetModifierKeys();
		}

		public static CefEventFlags GetModifierKeys(CefEventFlags modifiers = CefEventFlags.None)
		{
			if (Keyboard.IsKeyDown(Key.LeftCtrl))
			{
				modifiers |= CefEventFlags.ControlDown | CefEventFlags.IsLeft;
			}
			if (Keyboard.IsKeyDown(Key.RightCtrl))
			{
				modifiers |= CefEventFlags.ControlDown | CefEventFlags.IsRight;
			}
			if (Keyboard.IsKeyDown(Key.LeftShift))
			{
				modifiers |= CefEventFlags.ShiftDown | CefEventFlags.IsLeft;
			}
			if (Keyboard.IsKeyDown(Key.RightShift))
			{
				modifiers |= CefEventFlags.ShiftDown | CefEventFlags.IsRight;
			}
			if (Keyboard.IsKeyDown(Key.LeftAlt))
			{
				modifiers |= CefEventFlags.AltDown | CefEventFlags.IsLeft;
			}
			if (Keyboard.IsKeyDown(Key.RightAlt))
			{
				modifiers |= CefEventFlags.AltDown | CefEventFlags.IsRight;
			}
			return modifiers;
		}

		public static CefEventFlags GetModifiers(this KeyEventArgs e)
		{
			CefEventFlags cefEventFlags = CefEventFlags.None;
			ModifierKeys modifiers = e.KeyboardDevice.Modifiers;
			if (modifiers.HasFlag(ModifierKeys.Shift))
			{
				cefEventFlags |= CefEventFlags.ShiftDown;
			}
			if (modifiers.HasFlag(ModifierKeys.Alt))
			{
				cefEventFlags |= CefEventFlags.AltDown;
			}
			if (modifiers.HasFlag(ModifierKeys.Control))
			{
				cefEventFlags |= CefEventFlags.ControlDown;
			}
			return cefEventFlags;
		}

		public static IDragData GetDragData(this DragEventArgs e)
		{
			IDragData dragData = DragData.Create();
			dragData.IsFile = e.Data.GetDataPresent(DataFormats.FileDrop);
			if (dragData.IsFile)
			{
				string[] array = (string[])e.Data.GetData(DataFormats.FileDrop);
				foreach (string text in array)
				{
					string fileName = Path.GetFileName(text);
					dragData.AddFile(text.Replace("\\", "/"), fileName);
				}
			}
			string link = GetLink(e.Data);
			dragData.IsLink = !string.IsNullOrEmpty(link);
			if (dragData.IsLink)
			{
				dragData.LinkUrl = link;
			}
			dragData.IsFragment = e.Data.GetDataPresent(DataFormats.Text);
			if (dragData.IsFragment)
			{
				dragData.FragmentText = (string)e.Data.GetData(DataFormats.Text);
				string text2 = (string)e.Data.GetData(DataFormats.Html);
				if (text2 != string.Empty)
				{
					dragData.FragmentHtml = text2;
				}
			}
			return dragData;
		}

		private static string GetLink(IDataObject data)
		{
			if (data.GetDataPresent("UniformResourceLocatorW"))
			{
				string text = ReadUrlFromDragDropData(data, "UniformResourceLocatorW", Encoding.Unicode);
				if (text != null)
				{
					return text;
				}
			}
			if (data.GetDataPresent("UniformResourceLocator"))
			{
				return ReadUrlFromDragDropData(data, "UniformResourceLocator", Encoding.ASCII);
			}
			return null;
		}

		private static string ReadUrlFromDragDropData(IDataObject data, string urlDataFormatName, Encoding urlEncoding)
		{
			string text;
			using (Stream stream = (Stream)data.GetData(urlDataFormatName))
			{
				using TextReader textReader = new StreamReader(stream, urlEncoding);
				text = textReader.ReadToEnd();
			}
			return text.TrimEnd(default(char));
		}
	}
	public class WpfKeyboardHandler : IWpfKeyboardHandler, IDisposable
	{
		protected readonly ChromiumWebBrowser owner;

		public WpfKeyboardHandler(ChromiumWebBrowser owner)
		{
			this.owner = owner;
		}

		public virtual void Setup(HwndSource source)
		{
		}

		public virtual void Dispose()
		{
		}

		public virtual void HandleKeyPress(KeyEventArgs e)
		{
			IBrowser browserCore = owner.BrowserCore;
			Key key = ((e.SystemKey == Key.None) ? e.Key : e.SystemKey);
			if (browserCore != null)
			{
				int num = 0;
				switch (key)
				{
				case Key.LeftAlt:
				case Key.RightAlt:
					num = 18;
					break;
				case Key.LeftCtrl:
				case Key.RightCtrl:
					num = 17;
					break;
				case Key.LeftShift:
				case Key.RightShift:
					num = 16;
					break;
				default:
					num = KeyInterop.VirtualKeyFromKey(key);
					break;
				}
				int message = ((!e.IsDown) ? ((e.SystemKey != Key.None) ? 261 : 257) : ((e.SystemKey != Key.None) ? 260 : 256));
				browserCore.GetHost().SendKeyEvent(message, num, 0);
			}
			switch (key)
			{
			case Key.A:
				if (Keyboard.Modifiers != ModifierKeys.Control)
				{
					break;
				}
				goto case Key.Tab;
			case Key.Tab:
			case Key.End:
			case Key.Home:
			case Key.Left:
			case Key.Up:
			case Key.Right:
			case Key.Down:
				e.Handled = true;
				break;
			}
		}

		public virtual void HandleTextInput(TextCompositionEventArgs e)
		{
			IBrowser browserCore = owner.BrowserCore;
			if (browserCore != null)
			{
				IBrowserHost host = browserCore.GetHost();
				for (int i = 0; i < e.Text.Length; i++)
				{
					host.SendKeyEvent(258, e.Text[i], 0);
				}
				e.Handled = true;
			}
		}
	}
	public class WpfLegacyKeyboardHandler : IWpfKeyboardHandler, IDisposable
	{
		private HwndSourceHook sourceHook;

		private HwndSource source;

		private readonly ChromiumWebBrowser owner;

		public WpfLegacyKeyboardHandler(ChromiumWebBrowser owner)
		{
			this.owner = owner;
		}

		public virtual void Setup(HwndSource source)
		{
			this.source = source;
			if (source != null)
			{
				sourceHook = SourceHook;
				source.AddHook(SourceHook);
			}
		}

		public virtual void Dispose()
		{
			if (source != null && sourceHook != null)
			{
				source.RemoveHook(sourceHook);
			}
			source = null;
		}

		protected virtual IntPtr SourceHook(IntPtr hWnd, int message, IntPtr wParam, IntPtr lParam, ref bool handled)
		{
			if (handled)
			{
				return IntPtr.Zero;
			}
			if (((uint)(message - 256) <= 2u || (uint)(message - 260) <= 2u || message == 646) && owner.IsKeyboardFocused)
			{
				if (message == 260 && wParam.ToInt32() == KeyInterop.VirtualKeyFromKey(Key.F4))
				{
					return IntPtr.Zero;
				}
				IBrowser browserCore = owner.BrowserCore;
				if (browserCore != null)
				{
					browserCore.GetHost().SendKeyEvent(message, wParam.CastToInt32(), lParam.CastToInt32());
					handled = true;
				}
			}
			return IntPtr.Zero;
		}

		public virtual void HandleKeyPress(KeyEventArgs e)
		{
			if (e.Key == Key.Tab || e.Key == Key.Home || e.Key == Key.End || e.Key == Key.Up || e.Key == Key.Down || e.Key == Key.Left || e.Key == Key.Right || (e.Key == Key.A && Keyboard.Modifiers == ModifierKeys.Control))
			{
				CefEventFlags modifiers = e.GetModifiers();
				int message = (e.IsDown ? 256 : 257);
				int wParam = KeyInterop.VirtualKeyFromKey(e.Key);
				IBrowser browserCore = owner.BrowserCore;
				if (browserCore != null)
				{
					browserCore.GetHost().SendKeyEvent(message, wParam, (int)modifiers);
					e.Handled = true;
				}
			}
		}

		public virtual void HandleTextInput(TextCompositionEventArgs e)
		{
		}
	}

	public class ContextMenuHandler : CefSharp.Handler.ContextMenuHandler
	{
		internal class MenuModel
		{
			internal string Label { get; set; }

			internal CefMenuCommand CommandId { get; set; }

			internal bool IsEnabled { get; set; }

			internal bool IsSeperator { get; set; }

			internal bool? IsChecked { get; set; }

			internal MenuItemType Type { get; set; }

			internal IList<MenuModel> SubMenus { get; set; }
		}

		public const int CefMenuCommandShowDevToolsId = 28440;

		public const int CefMenuCommandCloseDevToolsId = 28441;

		private readonly bool addDevtoolsMenuItems;

		public ContextMenuHandler(bool addDevtoolsMenuItems = false)
		{
			this.addDevtoolsMenuItems = addDevtoolsMenuItems;
		}

		protected override void OnBeforeContextMenu(IWebBrowser chromiumWebBrowser, IBrowser browser, IFrame frame, IContextMenuParams parameters, IMenuModel model)
		{
			if (addDevtoolsMenuItems)
			{
				if (model.Count > 0)
				{
					model.AddSeparator();
				}
				model.AddItem((CefMenuCommand)28440, "Show DevTools (Inspect)");
				model.AddItem((CefMenuCommand)28441, "Close DevTools");
			}
		}

		protected override void OnContextMenuDismissed(IWebBrowser chromiumWebBrowser, IBrowser browser, IFrame frame)
		{
			ChromiumWebBrowser webBrowser = (ChromiumWebBrowser)chromiumWebBrowser;
			webBrowser.UiThreadRunAsync(delegate
			{
				webBrowser.ContextMenu = null;
			});
		}

		protected override bool RunContextMenu(IWebBrowser chromiumWebBrowser, IBrowser browser, IFrame frame, IContextMenuParams parameters, IMenuModel model, IRunContextMenuCallback callback)
		{
			ChromiumWebBrowser webBrowser = (ChromiumWebBrowser)chromiumWebBrowser;
			IList<MenuModel> menuItems = GetMenuItems(model);
			List<string> dictionarySuggestions = parameters.DictionarySuggestions;
			int xCoord = parameters.XCoord;
			int yCoord = parameters.YCoord;
			string misspelledWord = parameters.MisspelledWord;
			string selectionText = parameters.SelectionText;
			webBrowser.UiThreadRunAsync(delegate
			{
				ContextMenu menu = new ContextMenu
				{
					IsOpen = true,
					Placement = PlacementMode.Mouse
				};
				RoutedEventHandler handler = null;
				handler = delegate
				{
					menu.Closed -= handler;
					if (!callback.IsDisposed)
					{
						callback.Cancel();
					}
				};
				menu.Closed += handler;
				foreach (MenuModel item in menuItems)
				{
					if (item.IsSeperator)
					{
						menu.Items.Add(new Separator());
					}
					else if (item.CommandId != CefMenuCommand.NotFound)
					{
						MenuItem menuItem = new MenuItem
						{
							Header = item.Label.Replace("&", "_"),
							IsEnabled = item.IsEnabled,
							IsChecked = (item.IsChecked == true),
							IsCheckable = item.IsChecked.HasValue,
							Command = new DelegateCommand(delegate
							{
								ExecuteCommand(browser, new ContextMenuExecuteModel(item.CommandId, dictionarySuggestions, xCoord, yCoord, selectionText, misspelledWord));
							})
						};
						if (item.SubMenus != null && item.SubMenus.Count > 0)
						{
							foreach (MenuModel subItem in item.SubMenus)
							{
								if (subItem.CommandId != CefMenuCommand.NotFound)
								{
									if (subItem.IsSeperator)
									{
										menu.Items.Add(new Separator());
									}
									else
									{
										MenuItem newItem = new MenuItem
										{
											Header = subItem.Label.Replace("&", "_"),
											IsEnabled = subItem.IsEnabled,
											IsChecked = (subItem.IsChecked == true),
											IsCheckable = subItem.IsChecked.HasValue,
											Command = new DelegateCommand(delegate
											{
												ExecuteCommand(browser, new ContextMenuExecuteModel(subItem.CommandId, dictionarySuggestions, xCoord, yCoord, selectionText, misspelledWord));
											})
										};
										menuItem.Items.Add(newItem);
									}
								}
							}
						}
						menu.Items.Add(menuItem);
					}
				}
				webBrowser.ContextMenu = menu;
			});
			return true;
		}

		protected virtual void ExecuteCommand(IBrowser browser, ContextMenuExecuteModel model)
		{
			if (model.MenuCommand >= CefMenuCommand.SpellCheckSuggestion0 && model.MenuCommand <= CefMenuCommand.SpellCheckSuggestion4)
			{
				int num = (int)(model.MenuCommand - 200);
				if (num < model.DictionarySuggestions.Count)
				{
					string word = model.DictionarySuggestions[num];
					browser.ReplaceMisspelling(word);
				}
				return;
			}
			switch (model.MenuCommand)
			{
			case CefMenuCommand.Back:
				browser.GoBack();
				break;
			case CefMenuCommand.Forward:
				browser.GoForward();
				break;
			case CefMenuCommand.Reload:
				browser.Reload();
				break;
			case CefMenuCommand.ReloadNoCache:
				browser.Reload(ignoreCache: true);
				break;
			case CefMenuCommand.StopLoad:
				browser.StopLoad();
				break;
			case CefMenuCommand.Undo:
				browser.FocusedFrame.Undo();
				break;
			case CefMenuCommand.Redo:
				browser.FocusedFrame.Redo();
				break;
			case CefMenuCommand.Cut:
				browser.FocusedFrame.Cut();
				break;
			case CefMenuCommand.Copy:
				browser.FocusedFrame.Copy();
				break;
			case CefMenuCommand.Paste:
				browser.FocusedFrame.Paste();
				break;
			case CefMenuCommand.Delete:
				browser.FocusedFrame.Delete();
				break;
			case CefMenuCommand.SelectAll:
				browser.FocusedFrame.SelectAll();
				break;
			case CefMenuCommand.Print:
				browser.GetHost().Print();
				break;
			case CefMenuCommand.ViewSource:
				browser.FocusedFrame.ViewSource();
				break;
			case CefMenuCommand.Find:
				browser.GetHost().Find(model.SelectionText, forward: true, matchCase: false, findNext: false);
				break;
			case CefMenuCommand.AddToDictionary:
				browser.GetHost().AddWordToDictionary(model.MisspelledWord);
				break;
			case (CefMenuCommand)28440:
				browser.GetHost().ShowDevTools(null, model.XCoord, model.YCoord);
				break;
			case (CefMenuCommand)28441:
				browser.GetHost().CloseDevTools();
				break;
			}
		}

		private static IList<MenuModel> GetMenuItems(IMenuModel model)
		{
			List<MenuModel> list = new List<MenuModel>();
			for (int i = 0; i < model.Count; i++)
			{
				MenuItemType typeAt = model.GetTypeAt(i);
				bool? isChecked = null;
				if (typeAt == MenuItemType.Check)
				{
					isChecked = model.IsCheckedAt(i);
				}
				IMenuModel subMenuAt = model.GetSubMenuAt(i);
				IList<MenuModel> subMenus = ((subMenuAt == null) ? null : GetMenuItems(subMenuAt));
				MenuModel item = new MenuModel
				{
					Label = model.GetLabelAt(i),
					CommandId = model.GetCommandIdAt(i),
					IsEnabled = model.IsEnabledAt(i),
					Type = typeAt,
					IsSeperator = (typeAt == MenuItemType.Separator),
					IsChecked = isChecked,
					SubMenus = subMenus
				};
				list.Add(item);
			}
			return list;
		}
	}

	public class ChromiumWebBrowserWithTouchSupport : ChromiumWebBrowser
	{
		protected override void OnStylusDown(StylusDownEventArgs e)
		{
			if (e.StylusDevice.TabletDevice.Type == TabletDeviceType.Stylus)
			{
				Focus();
				CaptureStylus();
				OnStylus(e, TouchEventType.Pressed);
			}
			base.OnStylusDown(e);
		}

		protected override void OnStylusMove(StylusEventArgs e)
		{
			if (e.StylusDevice.TabletDevice.Type == TabletDeviceType.Stylus)
			{
				OnStylus(e, TouchEventType.Moved);
			}
			base.OnStylusMove(e);
		}

		protected override void OnStylusUp(StylusEventArgs e)
		{
			if (e.StylusDevice.TabletDevice.Type == TabletDeviceType.Stylus)
			{
				ReleaseStylusCapture();
				OnStylus(e, TouchEventType.Released);
			}
			base.OnStylusUp(e);
		}

		private void OnStylus(StylusEventArgs e, TouchEventType touchEventType)
		{
			IBrowser browser = GetBrowser();
			if (e.Handled || browser == null)
			{
				return;
			}
			CefEventFlags modifierKeys = WpfExtensions.GetModifierKeys();
			PointerType pointerType = (e.StylusDevice.Inverted ? PointerType.Eraser : PointerType.Pen);
			foreach (StylusPoint stylusPoint in e.GetStylusPoints(this))
			{
				TouchEvent evt = new TouchEvent
				{
					Id = e.StylusDevice.Id,
					X = (float)stylusPoint.X,
					Y = (float)stylusPoint.Y,
					PointerType = pointerType,
					Pressure = stylusPoint.PressureFactor,
					Type = touchEventType,
					Modifiers = modifierKeys
				};
				browser.GetHost().SendTouchEvent(evt);
			}
			e.Handled = true;
		}

	public class LifeSpanHandler : CefSharp.Handler.LifeSpanHandler
	{
		private LifeSpanHandlerOnBeforePopupCreatedDelegate onBeforePopupCreated;

		private LifeSpanHandlerOnPopupDestroyedDelegate onPopupDestroyed;

		private LifeSpanHandlerOnPopupBrowserCreatedDelegate onPopupBrowserCreated;

		private LifeSpanHandlerOnPopupCreatedDelegate onPopupCreated;

		private LifeSpanHandlerCreatePopupChromiumWebBrowser chromiumWebBrowserCreatedDelegate;

		private Dictionary<IntPtr, ChromiumWebBrowser> chromiumWebBrowserMap = new Dictionary<IntPtr, ChromiumWebBrowser>();

		private ChromiumWebBrowser pendingChromiumWebBrowser;

		public LifeSpanHandler(LifeSpanHandlerCreatePopupChromiumWebBrowser chromiumWebBrowserCreatedDelegate = null)
		{
			this.chromiumWebBrowserCreatedDelegate = chromiumWebBrowserCreatedDelegate;
		}

		public LifeSpanHandler OnBeforePopupCreated(LifeSpanHandlerOnBeforePopupCreatedDelegate onBeforePopupCreated)
		{
			this.onBeforePopupCreated = onBeforePopupCreated;
			return this;
		}

		public LifeSpanHandler OnPopupCreated(LifeSpanHandlerOnPopupCreatedDelegate onPopupCreated)
		{
			this.onPopupCreated = onPopupCreated;
			return this;
		}

		public LifeSpanHandler OnPopupBrowserCreated(LifeSpanHandlerOnPopupBrowserCreatedDelegate onPopupBrowserCreated)
		{
			this.onPopupBrowserCreated = onPopupBrowserCreated;
			return this;
		}

		public LifeSpanHandler OnPopupDestroyed(LifeSpanHandlerOnPopupDestroyedDelegate onPopupDestroyed)
		{
			this.onPopupDestroyed = onPopupDestroyed;
			return this;
		}

		public static LifeSpanHandlerBuilder Create(LifeSpanHandlerCreatePopupChromiumWebBrowser chromiumWebBrowserCreatedDelegate = null)
		{
			return new LifeSpanHandlerBuilder(chromiumWebBrowserCreatedDelegate);
		}

		protected override bool OnBeforePopup(IWebBrowser browserControl, IBrowser browser, IFrame frame, string targetUrl, string targetFrameName, WindowOpenDisposition targetDisposition, bool userGesture, IPopupFeatures popupFeatures, IWindowInfo windowInfo, IBrowserSettings browserSettings, ref bool noJavascriptAccess, out IWebBrowser newBrowser)
		{
			newBrowser = null;
			switch (onBeforePopupCreated?.Invoke(browserControl, browser, frame, targetUrl, targetFrameName, targetDisposition, userGesture, browserSettings) ?? PopupCreation.Continue)
			{
			case PopupCreation.Cancel:
				return true;
			case PopupCreation.ContinueWithJavascriptDisabled:
				noJavascriptAccess = true;
				break;
			}
			if (onPopupCreated == null)
			{
				return false;
			}
			ChromiumWebBrowser chromiumWebBrowser = (ChromiumWebBrowser)browserControl;
			ChromiumWebBrowser control = null;
			chromiumWebBrowser.Dispatcher.Invoke(delegate
			{
				control = chromiumWebBrowserCreatedDelegate?.Invoke();
				if (control == null)
				{
					control = new ChromiumWebBrowser();
				}
				control.SetAsPopup();
				control.LifeSpanHandler = this;
				pendingChromiumWebBrowser = control;
				onPopupCreated?.Invoke(control, targetUrl, targetFrameName, windowInfo);
				Window window = Window.GetWindow(control);
				if (window == null)
				{
					windowInfo.SetAsWindowless(IntPtr.Zero);
				}
				else
				{
					WindowInteropHelper windowInteropHelper = new WindowInteropHelper(window);
					IntPtr asWindowless = windowInteropHelper.EnsureHandle();
					windowInfo.SetAsWindowless(asWindowless);
				}
			});
			newBrowser = control;
			return false;
		}

		protected override void OnAfterCreated(IWebBrowser browserControl, IBrowser browser)
		{
			if (browser.IsPopup && pendingChromiumWebBrowser != null)
			{
				chromiumWebBrowserMap.Add(browser.GetHost().GetWindowHandle(), pendingChromiumWebBrowser);
				onPopupBrowserCreated?.Invoke(pendingChromiumWebBrowser, browser);
				pendingChromiumWebBrowser = null;
			}
		}

		protected override bool DoClose(IWebBrowser chromiumWebBrowser, IBrowser browser)
		{
			if (browser.IsPopup && !browser.IsDisposed)
			{
				IntPtr windowHandle = browser.GetHost().GetWindowHandle();
				if (chromiumWebBrowserMap.TryGetValue(windowHandle, out var control))
				{
					if (control == null)
					{
						return false;
					}
					if (!control.Dispatcher.HasShutdownStarted)
					{
						control.Dispatcher.Invoke(delegate
						{
							onPopupDestroyed?.Invoke(control, browser);
							if (!control.IsDisposed)
							{
								control.Dispose();
							}
						});
					}
					chromiumWebBrowserMap.Remove(windowHandle);
					return true;
				}
			}
			return false;
		}
	}

	public class WpfImeKeyboardHandler : WpfKeyboardHandler
	{
		private int languageCodeId;

		private bool systemCaret;

		private bool isSetup;

		private List<CefSharp.Structs.Rect> compositionBounds = new List<CefSharp.Structs.Rect>();

		private HwndSource source;

		private HwndSourceHook sourceHook;

		private bool hasImeComposition;

		private MouseButtonEventHandler mouseDownEventHandler;

		private bool isActive;

		public WpfImeKeyboardHandler(ChromiumWebBrowser owner)
			: base(owner)
		{
		}

		public void ChangeCompositionRange(Range selectionRange, CefSharp.Structs.Rect[] characterBounds)
		{
			if (!isActive)
			{
				return;
			}
			ScreenInfo? screenInfo = ((IRenderWebBrowser)owner).GetScreenInfo();
			float scaleFactor = (screenInfo.HasValue ? screenInfo.Value.DeviceScaleFactor : 1f);
			owner.UiThreadRunAsync(delegate
			{
				Window window = Window.GetWindow(owner);
				if (window != null)
				{
					System.Windows.Point point = owner.TransformToAncestor(window).Transform(new System.Windows.Point(0.0, 0.0));
					List<CefSharp.Structs.Rect> list = new List<CefSharp.Structs.Rect>();
					CefSharp.Structs.Rect[] array = characterBounds;
					for (int i = 0; i < array.Length; i++)
					{
						CefSharp.Structs.Rect rect = array[i];
						list.Add(new CefSharp.Structs.Rect((int)((point.X + (double)rect.X) * (double)scaleFactor), (int)((point.Y + (double)rect.Y) * (double)scaleFactor), (int)((float)rect.Width * scaleFactor), (int)((float)rect.Height * scaleFactor)));
					}
					compositionBounds = list;
					MoveImeWindow(source.Handle);
				}
			});
		}

		public override void Setup(HwndSource source)
		{
			if (!isSetup)
			{
				isSetup = true;
				this.source = source;
				sourceHook = SourceHook;
				source.AddHook(SourceHook);
				owner.GotFocus += OwnerGotFocus;
				owner.LostFocus += OwnerLostFocus;
				mouseDownEventHandler = OwnerMouseDown;
				owner.AddHandler(UIElement.MouseDownEvent, mouseDownEventHandler, handledEventsToo: true);
				if (owner.IsFocused)
				{
					SetActive();
				}
			}
		}

		public override void Dispose()
		{
			if (isSetup)
			{
				isSetup = false;
				owner.GotFocus -= OwnerGotFocus;
				owner.LostFocus -= OwnerLostFocus;
				owner.RemoveHandler(UIElement.MouseDownEvent, mouseDownEventHandler);
				if (source != null && sourceHook != null)
				{
					source.RemoveHook(sourceHook);
					source = null;
				}
			}
		}

		private void OwnerMouseDown(object sender, MouseButtonEventArgs e)
		{
			CloseImeComposition();
		}

		private void OwnerGotFocus(object sender, RoutedEventArgs e)
		{
			SetActive();
		}

		private void OwnerLostFocus(object sender, RoutedEventArgs e)
		{
			SetInactive();
		}

		private void SetActive()
		{
			if (InputMethod.GetIsInputMethodEnabled(owner))
			{
				InputMethod.SetIsInputMethodEnabled(owner, value: false);
			}
			if (InputMethod.GetIsInputMethodSuspended(owner))
			{
				InputMethod.SetIsInputMethodSuspended(owner, value: false);
			}
			InputMethod.SetIsInputMethodEnabled(owner, value: true);
			InputMethod.SetIsInputMethodSuspended(owner, value: true);
			isActive = true;
		}

		private void SetInactive()
		{
			isActive = false;
			InputMethod.SetIsInputMethodEnabled(owner, value: false);
			InputMethod.SetIsInputMethodSuspended(owner, value: false);
		}

		private IntPtr SourceHook(IntPtr hwnd, int msg, IntPtr wParam, IntPtr lParam, ref bool handled)
		{
			handled = false;
			if (!isActive || !isSetup || owner == null || owner.IsDisposed || !owner.IsBrowserInitialized)
			{
				return IntPtr.Zero;
			}
			IBrowserHost browserHost = owner.GetBrowserHost();
			if (browserHost == null)
			{
				return IntPtr.Zero;
			}
			switch ((WM)msg)
			{
			case WM.IME_SETCONTEXT:
				OnImeSetContext(hwnd, (uint)msg, wParam, lParam);
				handled = true;
				break;
			case WM.IME_STARTCOMPOSITION:
				OnIMEStartComposition(hwnd);
				hasImeComposition = true;
				handled = true;
				break;
			case WM.IME_COMPOSITION:
				OnImeComposition(browserHost, hwnd, lParam.ToInt32());
				handled = true;
				break;
			case WM.IME_ENDCOMPOSITION:
				OnImeEndComposition(browserHost, hwnd);
				hasImeComposition = false;
				handled = true;
				break;
			}
			if (!handled)
			{
				return new IntPtr(1);
			}
			return IntPtr.Zero;
		}

		private void CloseImeComposition()
		{
			if (hasImeComposition)
			{
				ImeNative.SetFocus(IntPtr.Zero);
				ImeNative.SetFocus(source.Handle);
			}
		}

		private void OnImeComposition(IBrowserHost browserHost, IntPtr hwnd, int lParam)
		{
			string text = string.Empty;
			if (ImeHandler.GetResult(hwnd, (uint)lParam, out text))
			{
				browserHost.ImeCommitText(text, new Range(int.MaxValue, int.MaxValue), 0);
				browserHost.ImeSetComposition(text, new CompositionUnderline[0], new Range(int.MaxValue, int.MaxValue), new Range(0, 0));
				browserHost.ImeFinishComposingText(keepSelection: false);
				return;
			}
			List<CompositionUnderline> list = new List<CompositionUnderline>();
			int compositionStart = 0;
			if (ImeHandler.GetComposition(hwnd, (uint)lParam, list, ref compositionStart, out text))
			{
				if ((long)languageCodeId == 18)
				{
					browserHost.ImeSetComposition(text, list.ToArray(), new Range(int.MaxValue, int.MaxValue), new Range(compositionStart + list.Count, compositionStart + list.Count));
				}
				else
				{
					browserHost.ImeSetComposition(text, list.ToArray(), new Range(int.MaxValue, int.MaxValue), new Range(compositionStart, compositionStart));
				}
				UpdateCaretPosition(compositionStart - 1);
			}
			else
			{
				CancelComposition(browserHost, hwnd);
			}
		}

		private void CancelComposition(IBrowserHost browserHost, IntPtr hwnd)
		{
			browserHost.ImeCancelComposition();
			DestroyImeWindow(hwnd);
		}

		private void OnImeEndComposition(IBrowserHost browserHost, IntPtr hwnd)
		{
			if ((long)languageCodeId != 18)
			{
				browserHost.ImeFinishComposingText(keepSelection: false);
			}
			DestroyImeWindow(hwnd);
		}

		private void OnImeSetContext(IntPtr hwnd, uint msg, IntPtr wParam, IntPtr lParam)
		{
			ImeNative.DefWindowProc(hwnd, msg, wParam, (IntPtr)(lParam.ToInt64() & 0x7FFFFFFF));
			CreateImeWindow(hwnd);
			MoveImeWindow(hwnd);
		}

		private void OnIMEStartComposition(IntPtr hwnd)
		{
			CreateImeWindow(hwnd);
			MoveImeWindow(hwnd);
		}

		private void CreateImeWindow(IntPtr hwnd)
		{
			languageCodeId = PrimaryLangId(InputLanguageManager.Current.CurrentInputLanguage.KeyboardLayoutId);
			if (((long)languageCodeId == 17 || (long)languageCodeId == 4) && !systemCaret && ImeNative.CreateCaret(hwnd, IntPtr.Zero, 1, 1))
			{
				systemCaret = true;
			}
		}

		private int PrimaryLangId(int lgid)
		{
			return lgid & 0x3FF;
		}

		private void MoveImeWindow(IntPtr hwnd)
		{
			if (compositionBounds.Count != 0)
			{
				IntPtr hIMC = ImeNative.ImmGetContext(hwnd);
				CefSharp.Structs.Rect rect = compositionBounds[0];
				int num = rect.X + rect.Width;
				int num2 = rect.Y + rect.Height;
				ImeNative.CANDIDATEFORM lpCandidateForm = new ImeNative.CANDIDATEFORM
				{
					dwIndex = 0,
					dwStyle = 64,
					ptCurrentPos = new ImeNative.POINT(num, num2),
					rcArea = new ImeNative.RECT(0, 0, 0, 0)
				};
				ImeNative.ImmSetCandidateWindow(hIMC, ref lpCandidateForm);
				if (systemCaret)
				{
					ImeNative.SetCaretPos(num, num2);
				}
				if ((long)languageCodeId == 4)
				{
					ImeNative.COMPOSITIONFORM lpCompForm = new ImeNative.COMPOSITIONFORM
					{
						dwStyle = 2,
						ptCurrentPos = new ImeNative.POINT(num, num2),
						rcArea = new ImeNative.RECT(0, 0, 0, 0)
					};
					ImeNative.ImmSetCompositionWindow(hIMC, ref lpCompForm);
				}
				if ((long)languageCodeId == 18)
				{
					num2++;
				}
				ImeNative.CANDIDATEFORM lpCandidateForm2 = new ImeNative.CANDIDATEFORM
				{
					dwIndex = 0,
					dwStyle = 128,
					ptCurrentPos = new ImeNative.POINT(num, num2),
					rcArea = new ImeNative.RECT(rect.X, rect.Y, num, num2 + 1)
				};
				ImeNative.ImmSetCandidateWindow(hIMC, ref lpCandidateForm2);
				ImeNative.ImmReleaseContext(hwnd, hIMC);
			}
		}

		private void DestroyImeWindow(IntPtr hwnd)
		{
			if (systemCaret)
			{
				ImeNative.DestroyCaret();
				systemCaret = false;
			}
		}

		private void UpdateCaretPosition(int index)
		{
			MoveImeWindow(source.Handle);
		}
	}

					int y = (int)dictionary["y"].GetDouble();
					int width = (int)dictionary["width"].GetDouble();
					int height = (int)dictionary["height"].GetDouble();
					location = new CefSharp.Structs.Rect(x, y, width, height);
				}
			}
			if (node.ContainsKey("offset_container_id"))
			{
				offsetContainerId = node["offset_container_id"].GetInt();
			}
			if (!node.ContainsKey("attributes"))
			{
				return;
			}
			IDictionary<string, IValue> dictionary2 = node["attributes"].GetDictionary();
			if (dictionary2 != null)
			{
				if (dictionary2.ContainsKey("name"))

		protected virtual CefSharp.Structs.Rect GetLocation()
		{
			AccessibilityNode node = accessibilityTree.GetNode(offsetContainerId);

			int y = location.Y;
			CefSharp.Structs.Rect rect = node.GetLocation();
			x += rect.X - node.scroll.X;
			y += rect.Y - node.scroll.Y;

	public class AccessibilityTree : FrameworkElementAutomationPeer
	{
		private readonly Dictionary<int, AccessibilityNode> accessibilityNodeMap;

		public string Id { get; }

		public int RootNodeId { get; private set; } = -1;

		public int FocusedNodeId { get; private set; } = -1;

		public AccessibilityTree(FrameworkElement owner, string Id)
			: base(owner)
		{
			this.Id = Id;
			accessibilityNodeMap = new Dictionary<int, AccessibilityNode>();
		}

		protected override List<AutomationPeer> GetChildrenCore()
		{
			AccessibilityNode rootNode = GetRootNode();
			if (rootNode == null)
			{
				return new List<AutomationPeer>(0);
			}
			return new List<AutomationPeer>(1) { rootNode };
		}

		protected virtual AccessibilityNode GetRootNode()
		{
			return GetNode(RootNodeId);
		}

		protected virtual AccessibilityNode GetFocusedNode()
		{
			return GetNode(FocusedNodeId);
		}

		public virtual void Update(IDictionary<string, IValue> accessibilityUpdateDictionary)
		{
			if (accessibilityUpdateDictionary == null || !accessibilityUpdateDictionary.ContainsKey("events") || !accessibilityUpdateDictionary.ContainsKey("updates"))
			{
				return;
			}
			IList<IValue> list = accessibilityUpdateDictionary["events"].GetList();
			IList<IValue> list2 = accessibilityUpdateDictionary["updates"].GetList();
			if (list == null || list2 == null)
			{
				return;
			}
			List<IDictionary<string, IValue>> list3 = (from x in list
				select x.GetDictionary() into y
				where y.ContainsKey("event_type")
				select y).ToList();
			List<IDictionary<string, IValue>> updateDictionaries = list2.Select((IValue x) => x.GetDictionary()).ToList();
			foreach (IDictionary<string, IValue> item in list3)
			{
				string text = item["event_type"].GetString();
				if (text == "layoutComplete")
				{
					UpdateLayout(item, updateDictionaries);
				}
				else if (text == "focus")
				{
					UpdateFocus(item, updateDictionaries);
				}
			}
		}

		protected virtual void UpdateLayout(IDictionary<string, IValue> eventDictionary, List<IDictionary<string, IValue>> updateDictionaries)
		{
			foreach (IDictionary<string, IValue> updateDictionary in updateDictionaries)
			{
				bool flag = false;
				if (updateDictionary.ContainsKey("node_id_to_clear"))
				{
					int num = updateDictionary["node_id_to_clear"].GetInt();
					if (num == RootNodeId)
					{
						RootNodeId = -1;
					}
					AccessibilityNode node = GetNode(num);
					RemoveNode(node);
				}
				if (updateDictionary.ContainsKey("root_id"))
				{
					int num2 = updateDictionary["root_id"].GetInt();
					if (num2 != RootNodeId)
					{
						RootNodeId = num2;
						flag = true;
					}
				}
				UpdateNodes(updateDictionary);
				if (flag)
				{
					RaiseAutomationEvent(AutomationEvents.StructureChanged);
				}
			}
		}

		protected virtual void UpdateFocus(IDictionary<string, IValue> eventDictionary, List<IDictionary<string, IValue>> updateDictionaries)
		{
			bool flag = false;
			if (eventDictionary.ContainsKey("id"))
			{
				int num = eventDictionary["id"].GetInt();
				if (num != FocusedNodeId)
				{
					FocusedNodeId = num;
					flag = true;
				}
			}
			if (!flag)
			{
				return;
			}
			AccessibilityNode focusedNode = GetFocusedNode();
			if (focusedNode == null)
			{
				foreach (IDictionary<string, IValue> updateDictionary in updateDictionaries)
				{
					UpdateNodes(updateDictionary);
				}
			}
			GetFocusedNode()?.OnFocusChanged();
		}

		protected virtual void RemoveNode(AccessibilityNode node)
		{
			if (node == null)
			{
				return;
			}
			foreach (AccessibilityNode childAccessibilityNode in node.GetChildAccessibilityNodes())
			{
				RemoveNode(childAccessibilityNode);
			}
			accessibilityNodeMap.Remove(node.Id);
		}

		protected virtual void UpdateNodes(IDictionary<string, IValue> update)
		{
			if (update == null || !update.ContainsKey("nodes"))
			{
				return;
			}
			IList<IValue> list = update["nodes"].GetList();
			foreach (IValue item in list)
			{
				IDictionary<string, IValue> dictionary = item.GetDictionary();
				if (dictionary != null)
				{
					int num = dictionary["id"].GetInt();
					AccessibilityNode node = GetNode(num);
					if (node != null)
					{
						node.Update(dictionary);
						continue;
					}
					node = new AccessibilityNode((FrameworkElement)base.Owner, dictionary, this);
					accessibilityNodeMap[num] = node;
				}
			}
		}

		internal AccessibilityNode GetNode(int nodeId)
		{
			if (accessibilityNodeMap.ContainsKey(nodeId))
			{
				return accessibilityNodeMap[nodeId];
			}
			return null;
		}
	}

			}
			accessibilityTree.Update(accessibilityUpdateDictionary);
		}

		}
