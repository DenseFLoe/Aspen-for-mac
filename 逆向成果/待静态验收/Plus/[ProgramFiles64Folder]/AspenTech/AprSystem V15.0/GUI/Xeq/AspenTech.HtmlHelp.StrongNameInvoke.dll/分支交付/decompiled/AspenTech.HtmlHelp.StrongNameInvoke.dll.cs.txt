namespace AspenTech.HtmlHelp.StrongNameInvoke.Util
{
	internal static class Global
	{
		public static string nameApp = "AspenTech HtmlHelp";

		internal static string GetLogFilename()
		{
			return Path.Combine(Path.GetTempPath(), nameApp + DateTime.Now.ToString("MM-dd-yyyy HH-mm-ss") + ".log");
		}
	}
	internal static class UtilError
	{
		private static LogWriter logWritter;

		public static void WriteToEventLog(string message, string className, string method)
		{
			try
			{
				if (logWritter == null)
				{
					logWritter = new LogWriter();
				}
				logWritter.WriteToLog(message, className, method);
			}
			catch (Exception)
			{
			}
		}
	}
}
namespace AspenTech.HtmlHelp.StrongNameInvoke.Classes
{
	internal interface ILogWriter
	{
		void WriteToLog(string message, string className, string method);
	}
	internal class LogWriter : ILogWriter
	{
		private string _fileName = Global.GetLogFilename();

		public string FileName
		{
			get
			{
				return _fileName;
			}
			set
			{
				_fileName = value;
			}
		}

		public void WriteToLog(string message, string className, string method)
		{
			DateTime.Now.ToString("HH-mm-ss");
			try
			{
				using StreamWriter streamWriter = File.AppendText(_fileName);
				streamWriter.Write("\r\nLog Entry : ");
				streamWriter.WriteLine(DateTime.Now.ToLongTimeString() + " " + DateTime.Now.ToLongDateString());
				streamWriter.WriteLine(className + " " + method + " :");
				streamWriter.WriteLine("  :" + message);
				streamWriter.WriteLine("-------------------------------");
			}
			catch (Exception ex)
			{
				ex.ToString();
			}
		}
	}
