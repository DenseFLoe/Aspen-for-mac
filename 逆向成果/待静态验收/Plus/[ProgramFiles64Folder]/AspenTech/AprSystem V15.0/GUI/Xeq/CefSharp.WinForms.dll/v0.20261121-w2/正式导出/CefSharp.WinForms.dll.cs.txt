		[MethodImpl(MethodImplOptions.NoInlining)]
		private void InitializeFieldsAndCefIfRequired()
		{
			if (!initialized)
			{
				if (!Cef.IsInitialized && !Cef.Initialize(new CefSettings()))
				{
					throw new InvalidOperationException("Cef.Initialize() failed.Check the log file see https://github.com/cefsharp/CefSharp/wiki/Trouble-Shooting#log-file for details.");
				}
				Cef.AddDisposable(this);
				if (FocusHandler == null && !Cef.CurrentlyOnThread(CefThreadIds.TID_UI))
				{
					FocusHandler = new DefaultFocusHandler();
				}
				if (browserSettings == null)
				{
					browserSettings = ObjectFactory.CreateBrowserSettings(autoDispose: true);
				}
				managedCefBrowserAdapter = ManagedCefBrowserAdapter.Create(this, offScreenRendering: false);
				initialized = true;
			}
		}

		protected override void Dispose(bool disposing)
		{
			if (Interlocked.CompareExchange(ref disposeSignaled, 1, 0) == 0)
			{
				if (!designMode)
				{
					InternalDispose(disposing);
				}
				base.Dispose(disposing);
			}
		}

		[MethodImpl(MethodImplOptions.NoInlining)]
		private void InternalDispose(bool disposing)
		{
			if (disposing)
			{
				Interlocked.Exchange(ref browserInitialized, 0);
				CanExecuteJavascriptInMainFrame = false;
				this.AddressChanged = null;
				this.ConsoleMessage = null;
				this.FrameLoadEnd = null;
				this.FrameLoadStart = null;
				this.LoadError = null;
				this.LoadingStateChanged = null;
				this.StatusMessage = null;
				this.TitleChanged = null;
				this.JavascriptMessageReceived = null;
				FreeHandlersExceptLifeSpanAndFocus();
				FocusHandler = new NoFocusHandler();
				browser = null;
				BrowserCore = null;
				if (parentFormMessageInterceptor != null)
				{
					parentFormMessageInterceptor.Dispose();
					parentFormMessageInterceptor = null;
				}
				if (managedCefBrowserAdapter != null)
				{
					managedCefBrowserAdapter.Dispose();
					managedCefBrowserAdapter = null;
				}
				if (browserSettings != null && browserSettings.AutoDispose)
				{
					browserSettings.Dispose();
				}
				browserSettings = null;
				parkingControl?.Dispose();
				parkingControl = null;
				LifeSpanHandler = null;
			}
			Cef.RemoveDisposable(this);
		}

		public void Load(string url)
		{
			if (IsDisposed)
			{
				return;
			}
			IBrowser browserCore = BrowserCore;
			if (browserCore == null)
			{
				Address = url;
			}
			else
			{
				if (browserCore.IsDisposed)
				{
					return;
				}
				if (!browserCore.IsValid)
				{
					throw new InvalidOperationException("IBrowser instance is no longer valid. Control.Handle was likely destroyed.");
				}
				using IFrame frame = browserCore.MainFrame;
				if (frame.IsValid)
				{
					frame.LoadUrl(url);
				}
			}
		}

		public async Task<byte[]> CaptureScreenshotAsync(CaptureScreenshotFormat format = CaptureScreenshotFormat.Png, int? quality = null, Viewport viewPort = null, bool fromSurface = true, bool captureBeyondViewport = false)
		{
			ThrowExceptionIfDisposed();
			ThrowExceptionIfBrowserNotInitialized();
			if (viewPort != null && viewPort.Scale <= 0.0)
			{
				throw new ArgumentException("viewPort.Scale must be greater than 0.");
			}
			using DevToolsClient devToolsClient = browser.GetDevToolsClient();
			return (await devToolsClient.Page.CaptureScreenshotAsync(format, quality, viewPort, fromSurface, captureBeyondViewport).ConfigureAwait(continueOnCapturedContext: false)).Data;
		}

		private bool IsParentInDesignMode(Control control)
		{
			if (control == null)
			{
				throw new ArgumentNullException("control");
			}
			ISite obj = control.Site;
			if (obj != null && obj.DesignMode)
			{
				return true;
			}
			if (control.Parent == null)
			{
				return false;
			}
			return IsParentInDesignMode(control.Parent);
		}

		protected override void OnHandleCreated(EventArgs e)
		{
			designMode = base.DesignMode;
			if (!designMode)
			{
				try
				{
					designMode = IsParentInDesignMode(this);
				}
				catch (Exception)
				{
				}
			}
			if (designMode)
			{
				Application.ApplicationExit -= OnApplicationExit;
			}
			else
			{
				InitializeFieldsAndCefIfRequired();
				CreateBrowser();
				ResizeBrowser(base.Width, base.Height);
			}
			base.OnHandleCreated(e);
		}

		protected override void OnHandleDestroyed(EventArgs e)
		{
			if (!designMode)
			{
				OnHandleDestroyedInternal();
			}
			base.OnHandleDestroyed(e);
		}

		[MethodImpl(MethodImplOptions.NoInlining)]
		private void OnHandleDestroyedInternal()
		{
			if (base.RecreatingHandle || ParkControlOnHandleDestroyed)
			{
				parkingControl = new Control();
				parkingControl.CreateControl();
				IBrowserHost browserHost = this.GetBrowserHost();
				if (browserHost != null)
				{
					IntPtr windowHandle = browserHost.GetWindowHandle();
					NativeMethodWrapper.SetWindowParent(windowHandle, parkingControl.Handle);
				}
			}
		}

		protected virtual IWindowInfo CreateBrowserWindowInfo(IntPtr handle)
		{
			IWindowInfo windowInfo = ObjectFactory.CreateWindowInfo();
			windowInfo.SetAsChild(handle);
			if (!ActivateBrowserOnCreation)
			{
				windowInfo.ExStyle |= 134217728u;
			}
			return windowInfo;
		}

		[MethodImpl(MethodImplOptions.NoInlining)]
		private void CreateBrowser()
		{
			browserCreated = true;
			if (!((IWebBrowserInternal)this).HasParent)
			{
				if ((base.RecreatingHandle || ParkControlOnHandleDestroyed) && IsBrowserInitialized && browser != null)
				{
					IBrowserHost browserHost = this.GetBrowserHost();
					IntPtr windowHandle = browserHost.GetWindowHandle();
					NativeMethodWrapper.SetWindowParent(windowHandle, base.Handle);
					parkingControl.Dispose();
					parkingControl = null;
				}
				else
				{
					IWindowInfo windowInfo = CreateBrowserWindowInfo(base.Handle);
					removeExNoActivateStyle = (windowInfo.ExStyle & 0x8000000) == 134217728;
					initialAddressLoaded = !string.IsNullOrEmpty(Address);
					managedCefBrowserAdapter.CreateBrowser(windowInfo, browserSettings, requestContext, Address);
				}
			}
		}

		protected virtual void OnSetBrowserInitialFocus()
		{
			if (!base.InvokeRequired)
			{
				return;
			}
			BeginInvoke((Action)delegate
			{
				if (!base.Disposing && !IsDisposed)
				{
					IBrowser obj = browser;
					if ((obj == null || !obj.IsDisposed) && Focused)
					{
						browser?.GetHost()?.SetFocus(focus: true);
					}
				}
			});
		}

		public async Task<DomRect> GetContentSizeAsync()
		{
			ThrowExceptionIfDisposed();

		private void InitialLoad(bool? isLoading, CefErrorCode? errorCode)
		{
			if (IsDisposed)
			{
				initialLoadAction = null;
				initialLoadTaskCompletionSource.TrySetCanceled();
			}
			else if (isLoading.HasValue)
			{
				if (!isLoading.Value)
				{
					initialLoadAction = null;
					int num = ((browser?.GetHost())?.GetVisibleNavigationEntry())?.HttpStatusCode ?? (-1);
					if (num == 0)
					{
						num = -1;
					}
					initialLoadTaskCompletionSource.TrySetResultAsync(new LoadUrlAsyncResponse(CefErrorCode.None, num));
				}
			}
			else if (errorCode.HasValue && errorCode != CefErrorCode.Aborted)
			{
				initialLoadAction = null;
				initialLoadTaskCompletionSource.TrySetResultAsync(new LoadUrlAsyncResponse(errorCode.Value, -1));
			}
		}

		private void OnAfterBrowserCreated(IBrowser browser)
		{
			base.BrowserHwnd = browser.GetHost().GetWindowHandle();
			if (UseParentFormMessageInterceptor)
			{
				this.InvokeOnUiThreadIfRequired(delegate
				{
					parentFormMessageInterceptor = new ParentFormMessageInterceptor(this);
					parentFormMessageInterceptor.Moving += delegate
					{
						if (IsBrowserInitialized && !IsDisposed)
						{
							browser?.GetHost()?.NotifyMoveOrResizeStarted();
						}
					};
				});
			}
			ResizeBrowser(base.Width, base.Height);
			if (!initialAddressLoaded && !string.IsNullOrEmpty(Address))
			{
				browser.MainFrame.LoadUrl(Address);
			}
			if (initialFocus)
			{
				OnSetBrowserInitialFocus();
			}
			RaiseIsBrowserInitializedChangedEvent();
		}

		internal static void InvokeOnUiThreadIfRequired(this Control control, Action action)
		{
			if (control.Disposing || control.IsDisposed || !control.IsHandleCreated)
			{
				return;
			}
			if (control.InvokeRequired)
			{
				control.BeginInvoke((Action)delegate
				{
					if (!control.Disposing && !control.IsDisposed && control.IsHandleCreated)
					{
						action();
					}
				});
			}
			else
			{
				action();
			}
		}

		internal static void InvokeSyncOnUiThreadIfRequired(this Control control, Action action)
		{
			if (!control.Disposing && !control.IsDisposed && control.IsHandleCreated)
			{
				if (control.InvokeRequired)
				{
					control.Invoke(action);
				}
				else
				{
					action();
				}
			}
		}

		public static bool Activate(this Control control)
		{
			return control.GetContainerControl()?.ActivateControl(control) ?? false;
		}

		public static bool IsActiveControl(this Control control)
		{
			Form form = control.FindForm();
			if (form == null)
			{
				return false;
			}
			Control activeControl = form.ActiveControl;
			while (activeControl != null && activeControl is ContainerControl && control != activeControl)
			{
				ContainerControl containerControl = activeControl as ContainerControl;
				activeControl = containerControl.ActiveControl;
			}
			return control == activeControl;
		}

		public static void SelectNextControl(this Control control, bool next)
		{
			IContainerControl containerControl = control.GetContainerControl();
			while (containerControl != null && containerControl is Control control2)
			{
				Control activeControl = containerControl.ActiveControl;
				if (!control2.SelectNextControl(activeControl, next, tabStopOnly: true, nested: true, wrap: false) && control2.Parent != null)
				{
					containerControl = control2.Parent.GetContainerControl();
					continue;
				}
				break;
			}
		}
	}

	public class ParentFormMessageInterceptor : NativeWindow, IDisposable
	{
		private bool isMoving;

		private Rectangle movingRectangle;

		private FormWindowState previousWindowState;

		private ChromiumHostControlBase Browser { get; set; }

		private Form ParentForm { get; set; }

		public event EventHandler<EventArgs> Moving;

		public ParentFormMessageInterceptor(ChromiumHostControlBase browser)
		{
			Browser = browser;
			Browser.ParentChanged += ParentParentChanged;
			RefindParentForm();
		}

		public void RefindParentForm()
		{
			ParentParentChanged(Browser, null);
		}

		private void ParentParentChanged(object sender, EventArgs e)
		{
			Control control = (Control)sender;
			Form parentForm = ParentForm;
			Form form = control.FindForm();
			if (parentForm != null && form != null && !(parentForm.Handle != form.Handle))
			{
				return;
			}
			if (base.Handle != IntPtr.Zero)
			{
				ReleaseHandle();
			}
			if (parentForm != null)
			{
				parentForm.HandleCreated -= OnHandleCreated;
				parentForm.HandleDestroyed -= OnHandleDestroyed;
				parentForm.Resize -= OnResize;
			}
			ParentForm = form;
			if (form != null)
			{
				form.HandleCreated += OnHandleCreated;
				form.HandleDestroyed += OnHandleDestroyed;
				form.Resize += OnResize;
				previousWindowState = form.WindowState;
				if (form.IsHandleCreated)
				{
					OnHandleCreated(form, null);
				}
			}
		}

		private void OnResize(object sender, EventArgs e)
		{
			Form form = (Form)sender;
			if (previousWindowState == form.WindowState)
			{
				return;
			}
			switch (form.WindowState)
			{
			case FormWindowState.Normal:
			case FormWindowState.Maximized:
				if (previousWindowState == FormWindowState.Minimized)
				{
					Browser?.ShowInternal();
				}
				break;
			case FormWindowState.Minimized:
				Browser?.HideInternal();
				break;
			}
			previousWindowState = form.WindowState;
		}

		private void OnHandleCreated(object sender, EventArgs e)
		{
			AssignHandle(((Form)sender).Handle);
		}

		private void OnHandleDestroyed(object sender, EventArgs e)
		{
			ReleaseHandle();
		}

		protected override void WndProc(ref Message m)
		{
			bool flag = false;
			int num = -1;
			int num2 = -1;
			switch (m.Msg)
			{
			case 6:
			{
				ChromiumHostControlBase browser = Browser;
				if ((int)m.WParam == 0)
				{
					browser.IsActivating = false;
					DefWndProc(ref m);
				}
				else
				{
					browser.IsActivating = browser.IsActiveControl();
					DefWndProc(ref m);
				}
				return;
			}
			case 534:
				movingRectangle = (Rectangle)Marshal.PtrToStructure(m.LParam, typeof(Rectangle));
				num = movingRectangle.Left;
				num2 = movingRectangle.Top;
				flag = true;
				break;
			case 3:
			{
				int num3 = m.LParam.CastToInt32();
				num = num3 & 0xFFFF;
				num2 = (num3 >> 16) & 0xFFFF;
				flag = true;
				break;
			}
			}
			if (flag && Browser.IsHandleCreated && ParentForm.WindowState == FormWindowState.Normal && (ParentForm.Left != num || ParentForm.Top != num2) && !isMoving && ParentForm.Left >= 0 && ParentForm.Right >= 0)
			{
				OnMoving();
			}
			DefWndProc(ref m);
		}

		protected virtual void OnMoving()
		{
			isMoving = true;
			this.Moving?.Invoke(Browser, EventArgs.Empty);
			isMoving = false;
		}

		public void Dispose()
		{
			Dispose(disposing: true);
		}

		protected virtual void Dispose(bool disposing)
		{
			if (disposing)
			{
				this.Moving = null;
				if (ParentForm != null)
				{
					ParentForm.HandleCreated -= OnHandleCreated;
					ParentForm.HandleDestroyed -= OnHandleDestroyed;
					ParentForm.Resize -= OnResize;
					ParentForm = null;
				}
				if (base.Handle != IntPtr.Zero)
				{
					ReleaseHandle();
				}
				if (Browser != null)
				{
					Browser.ParentChanged -= ParentParentChanged;
					Browser = null;
				}
			}
		}

		protected override void OnThreadException(Exception e)
		{
			base.OnThreadException(e);
		}
	}

	public abstract class ChromiumHostControlBase : Control
	{
		[Browsable(false)]
		[DesignerSerializationVisibility(DesignerSerializationVisibility.Hidden)]
		public IntPtr BrowserHwnd { get; set; }

		[Browsable(false)]
		[DesignerSerializationVisibility(DesignerSerializationVisibility.Hidden)]
		[DefaultValue(false)]
		public bool IsActivating { get; set; }

		protected override System.Drawing.Size DefaultSize => new System.Drawing.Size(200, 100);

		public event EventHandler IsBrowserInitializedChanged;

		[DllImport("user32.dll", SetLastError = true)]
		[return: MarshalAs(UnmanagedType.Bool)]
		private static extern bool SetWindowPos(IntPtr hWnd, IntPtr hWndInsertAfter, int X, int Y, int cx, int cy, uint uFlags);

		protected override bool IsInputKey(Keys keyData)
		{
			switch (keyData)
			{
			case Keys.Tab:
			case Keys.Left:
			case Keys.Up:
			case Keys.Right:
			case Keys.Down:
				return true;
			case Keys.Tab | Keys.Shift:
			case Keys.Left | Keys.Shift:
			case Keys.Up | Keys.Shift:
			case Keys.Right | Keys.Shift:
			case Keys.Down | Keys.Shift:
				return true;
			default:
				return base.IsInputKey(keyData);
			}
		}

		protected override void OnSizeChanged(EventArgs e)
		{
			ResizeBrowser(base.Width, base.Height);
			base.OnSizeChanged(e);
		}

		protected override void OnVisibleChanged(EventArgs e)
		{
			if (base.Visible)
			{
				ShowInternal();
			}
			else
			{
				HideInternal();
			}
			base.OnVisibleChanged(e);
		}

		protected virtual void ResizeBrowser(int width, int height)
		{
			if (BrowserHwnd != IntPtr.Zero)
			{
				SetWindowPosition(BrowserHwnd, 0, 0, width, height);
			}
		}

		internal virtual void HideInternal()
		{
			if (BrowserHwnd != IntPtr.Zero)
			{
				SetWindowPosition(BrowserHwnd, 0, 0, 0, 0);
			}
		}

		internal virtual void ShowInternal()
		{
			if (BrowserHwnd != IntPtr.Zero)
			{
				SetWindowPosition(BrowserHwnd, 0, 0, base.Width, base.Height);
			}
		}

		protected override void Dispose(bool disposing)
		{
			if (disposing)
			{
				BrowserHwnd = IntPtr.Zero;
				this.IsBrowserInitializedChanged = null;
			}
			base.Dispose(disposing);
		}

		internal void RaiseIsBrowserInitializedChangedEvent()
		{
			this.IsBrowserInitializedChanged?.Invoke(this, EventArgs.Empty);
		}

		private void SetWindowPosition(IntPtr handle, int x, int y, int width, int height)
		{
			if (handle != IntPtr.Zero)
			{
				if (width == 0 && height == 0)
				{
					SetWindowPos(handle, IntPtr.Zero, 0, 0, 0, 0, 22u);
				}
				else
				{
					SetWindowPos(handle, IntPtr.Zero, x, y, width, height, 4u);
				}
			}
		}

		public static T FromBrowser<T>(IBrowser browser) where T : ChromiumHostControlBase
		{
			if (browser.IsDisposed)
			{
				return null;
			}
			IntPtr windowHandle = browser.GetHost().GetWindowHandle();
			if (windowHandle == IntPtr.Zero)
			{
				return null;
			}
			return Control.FromChildHandle(windowHandle) as T;
		}
	}

	public class LifeSpanHandler : CefSharp.Handler.LifeSpanHandler
	{
		private readonly Dictionary<int, ParentFormMessageInterceptor> popupParentFormMessageInterceptors = new Dictionary<int, ParentFormMessageInterceptor>();

		private OnBeforePopupCreatedDelegate onBeforePopupCreated;

		private OnPopupDestroyedDelegate onPopupDestroyed;

		private OnPopupBrowserCreatedDelegate onPopupBrowserCreated;

		private OnPopupCreatedDelegate onPopupCreated;

		private CreatePopupChromiumHostControl chromiumHostControlCreatedDelegate;

		public LifeSpanHandler(CreatePopupChromiumHostControl chromiumHostControlCreatedDelegate = null)
		{
			this.chromiumHostControlCreatedDelegate = chromiumHostControlCreatedDelegate;
		}

		protected override bool DoClose(IWebBrowser chromiumWebBrowser, IBrowser browser)
		{
			if (browser.IsPopup)
			{
				ChromiumHostControl control = ChromiumHostControl.FromBrowser(browser);
				if (control == null)
				{
					return false;
				}
				if (!control.IsDisposed && control.IsHandleCreated)
				{
					try
					{
						control.InvokeSyncOnUiThreadIfRequired(delegate
						{
							onPopupDestroyed?.Invoke(control, browser);
							control.Dispose();
						});
					}
					catch (ObjectDisposedException)
					{
						return false;
					}
				}
			}
			return true;
		}

		protected override void OnAfterCreated(IWebBrowser chromiumWebBrowser, IBrowser browser)
		{
			if (!browser.IsPopup)
			{
				return;
			}
			ChromiumHostControl control = ChromiumHostControl.FromBrowser(browser);
			if (control != null && !control.IsDisposed)
			{
				control.BrowserHwnd = browser.GetHost().GetWindowHandle();
				control.InvokeOnUiThreadIfRequired(delegate
				{
					ParentFormMessageInterceptor parentFormMessageInterceptor = new ParentFormMessageInterceptor(control);
					parentFormMessageInterceptor.Moving += delegate
					{
						if (!browser.IsDisposed)
						{
							browser?.GetHost()?.NotifyMoveOrResizeStarted();
						}
					};
					popupParentFormMessageInterceptors.Add(browser.Identifier, parentFormMessageInterceptor);
				});
				control.BrowserCore = browser;
				control.RaiseIsBrowserInitializedChangedEvent();
			}
			onPopupBrowserCreated?.Invoke(control, browser);
		}

		protected override void OnBeforeClose(IWebBrowser chromiumWebBrowser, IBrowser browser)
		{
			if (!browser.IsDisposed && browser.IsPopup && popupParentFormMessageInterceptors.TryGetValue(browser.Identifier, out var value))
			{
				popupParentFormMessageInterceptors.Remove(browser.Identifier);
				value?.Dispose();
			}
		}

		protected override bool OnBeforePopup(IWebBrowser chromiumWebBrowser, IBrowser browser, IFrame frame, string targetUrl, string targetFrameName, WindowOpenDisposition targetDisposition, bool userGesture, IPopupFeatures popupFeatures, IWindowInfo windowInfo, IBrowserSettings browserSettings, ref bool noJavascriptAccess, out IWebBrowser newBrowser)
		{
			newBrowser = null;
			switch (onBeforePopupCreated?.Invoke(chromiumWebBrowser, browser, frame, targetUrl, targetFrameName, targetDisposition, userGesture, browserSettings) ?? PopupCreation.Continue)
			{
			case PopupCreation.Cancel:
				return true;
			case PopupCreation.ContinueWithJavascriptDisabled:
				noJavascriptAccess = true;
				break;
			}
			if (onPopupCreated == null)
			{
				return false;
			}
			ChromiumWebBrowser chromiumWebBrowser2 = (ChromiumWebBrowser)chromiumWebBrowser;
			if (chromiumWebBrowser2.LoadHandler == null)
			{
				chromiumWebBrowser2.LoadHandler = new LoadHandler();
			}
			if (chromiumWebBrowser2.DisplayHandler == null)
			{
				chromiumWebBrowser2.DisplayHandler = new DisplayHandler();
			}
			chromiumWebBrowser2.InvokeSyncOnUiThreadIfRequired(delegate
			{
				ChromiumHostControl chromiumHostControl = chromiumHostControlCreatedDelegate?.Invoke();
				if (chromiumHostControl == null)
				{
					chromiumHostControl = new ChromiumHostControl
					{
						Dock = DockStyle.Fill
					};
				}
				chromiumHostControl.CreateControl();
				onPopupCreated?.Invoke(chromiumHostControl, targetUrl);
				Rectangle clientRectangle = chromiumHostControl.ClientRectangle;
				CefSharp.Structs.Rect windowBounds = new CefSharp.Structs.Rect(clientRectangle.X, clientRectangle.Y, clientRectangle.Width, clientRectangle.Height);
				windowInfo.SetAsChild(chromiumHostControl.Handle, windowBounds);
			});
			return false;
		}

		public LifeSpanHandler OnBeforePopupCreated(OnBeforePopupCreatedDelegate onBeforePopupCreated)
		{
			this.onBeforePopupCreated = onBeforePopupCreated;
			return this;
		}

		public LifeSpanHandler OnPopupCreated(OnPopupCreatedDelegate onPopupCreated)
		{
			this.onPopupCreated = onPopupCreated;
			return this;
		}

		public LifeSpanHandler OnPopupBrowserCreated(OnPopupBrowserCreatedDelegate onPopupBrowserCreated)
		{
			this.onPopupBrowserCreated = onPopupBrowserCreated;
			return this;
		}

		public LifeSpanHandler OnPopupDestroyed(OnPopupDestroyedDelegate onPopupDestroyed)
		{
			this.onPopupDestroyed = onPopupDestroyed;
			return this;
		}

		public static LifeSpanHandlerBuilder Create(CreatePopupChromiumHostControl chromiumHostControlCreatedDelegate = null)
		{
			return new LifeSpanHandlerBuilder(chromiumHostControlCreatedDelegate);
		}
	}

	public static class ChromiumRenderWidgetHandleFinder
	{
		private delegate bool EnumWindowProc(IntPtr hwnd, IntPtr lParam);

		public const string ChromeRenderWidgetHostClassName = "Chrome_RenderWidgetHostHWND";

		[DllImport("user32")]
		[return: MarshalAs(UnmanagedType.Bool)]
		private static extern bool EnumChildWindows(IntPtr window, EnumWindowProc callback, IntPtr lParam);

		[DllImport("user32.dll", CharSet = CharSet.Auto)]
		private static extern int GetClassName(IntPtr hWnd, StringBuilder lpClassName, int nMaxCount);

		public static bool TryFindHandle(IWebBrowser chromiumWebBrowser, out IntPtr chromerRenderWidgetHostHandle)
		{
			IBrowserHost browserHost = chromiumWebBrowser.GetBrowserHost();
			if (browserHost == null)
			{
				throw new Exception("IBrowserHost is null, you've likely call this method before the underlying browser has been created.");
			}
			IntPtr windowHandle = browserHost.GetWindowHandle();
			return TryFindHandle(windowHandle, "Chrome_RenderWidgetHostHWND", out chromerRenderWidgetHostHandle);
		}

		public static bool TryFindHandle(IBrowser browser, out IntPtr chromerRenderWidgetHostHandle)
		{
			IBrowserHost host = browser.GetHost();
			if (host == null)
			{
				throw new Exception("IBrowserHost is null, you've likely call this method before the underlying browser has been created.");
			}
			IntPtr windowHandle = host.GetWindowHandle();
			return TryFindHandle(windowHandle, "Chrome_RenderWidgetHostHWND", out chromerRenderWidgetHostHandle);
		}

		public static bool TryFindHandle(IntPtr chromiumWebBrowserHandle, string chromeRenderWidgetHostClassName, out IntPtr chromerRenderWidgetHostHandle)
		{
			IntPtr chromeRenderWidgetHostHwnd = IntPtr.Zero;
			EnumWindowProc callback = delegate(IntPtr hWnd, IntPtr lParam)
			{
				StringBuilder stringBuilder = new StringBuilder(128);
				GetClassName(hWnd, stringBuilder, stringBuilder.Capacity);
				if (stringBuilder.ToString() == chromeRenderWidgetHostClassName)
				{
					chromeRenderWidgetHostHwnd = hWnd;
					return false;
				}
				return true;
			};
			EnumChildWindows(chromiumWebBrowserHandle, callback, IntPtr.Zero);
			chromerRenderWidgetHostHandle = chromeRenderWidgetHostHwnd;
			return chromerRenderWidgetHostHandle != IntPtr.Zero;
		}
	}
	public class ChromiumWidgetNativeWindow : NativeWindow
	{
		private Func<Message, bool> wndProcHandler;

		public ChromiumWidgetNativeWindow(Control control, IntPtr chromeWidgetHostHandle)
		{
			AssignHandle(chromeWidgetHostHandle);
			if (control != null)
			{
				control.HandleDestroyed += BrowserHandleDestroyed;
			}
		}

		private void BrowserHandleDestroyed(object sender, EventArgs e)
		{
			ReleaseHandle();
			Control control = (Control)sender;
			control.HandleDestroyed -= BrowserHandleDestroyed;
			wndProcHandler = null;
		}

		public void OnWndProc(Func<Message, bool> wndProcHandler)
		{
			this.wndProcHandler = wndProcHandler;
		}

		protected override void WndProc(ref Message m)
		{
			if (wndProcHandler?.Invoke(m) == false)
			{
				base.WndProc(ref m);
			}
		}
	}
