	public class ConditionalFormatting : Control
	{
		private IFlyover _flyover;

		private IUnityContainer _container;

		private IPFDHighlight _pfdHighlight;

		private IEconomicsOverlay _economicsOverlay;

		private GenericUnitOpCollection _uniopCollection;

		~ConditionalFormatting()
		{
			_container = null;
			_economicsOverlay = null;
			_flyover = null;
			_pfdHighlight = null;
			_uniopCollection = null;
		}

		public void DisplayFormatting(IFlyover flyover, bool bDisplay = true)
		{
			if (flyover == null)
			{
				throw new ArgumentNullException("flyover");
			}
			_flyover = flyover;
			if (_container == null)
			{
				return;
			}
			_pfdHighlight = UnityContainerExtensions.Resolve<IPFDHighlight>(_container, Array.Empty<ResolverOverride>());
			if (_pfdHighlight == null)
			{
				return;
			}
			_pfdHighlight.ClearHighlights();
			if (bDisplay && _economicsOverlay != null)
			{
				try
				{
					_uniopCollection = new GenericUnitOpCollection(_container);
					_uniopCollection.EquipmentCostItems = (List<EquipmentCostItem>)_economicsOverlay.OnEEOverlayGetEquipmentCostItems();
					_uniopCollection.SetBlockNameHandleMaps();
					_pfdHighlight.AddHighlights(false, _uniopCollection.Colors, (object)_uniopCollection, _flyover, false);
					return;
				}
				catch (Exception)
				{
					return;
				}
			}
			_uniopCollection = null;
		}

		private void OnOverlayRefresh(OverlayRefreshEventArgs e)
		{
			if (_uniopCollection == null || _container == null || _flyover == null)
			{
				return;
			}
			base.Dispatcher.BeginInvoke(DispatcherPriority.Normal, (Action)delegate
			{
				_pfdHighlight = UnityContainerExtensions.Resolve<IPFDHighlight>(_container, Array.Empty<ResolverOverride>());
				if (_pfdHighlight != null)
				{
					DisplayFormatting(_flyover);
				}
			});
		}

		private void RefreshHighlights()
		{
			if (_container != null)
			{
				base.Dispatcher.BeginInvoke(DispatcherPriority.Background, (Action)delegate
				{
					_pfdHighlight = UnityContainerExtensions.Resolve<IPFDHighlight>(_container, Array.Empty<ResolverOverride>());
					DisplayFormatting(_flyover);
				});
			}
		}

		public void Load(IUnityContainer container, IEconomicsOverlay economicsOverlay)
		{
			if (container == null)
			{
				throw new ArgumentNullException("container");
			}
			_container = container;
			if (economicsOverlay == null)
			{
				throw new ArgumentNullException("flyover");
			}
			_economicsOverlay = economicsOverlay;
			((CompositePresentationEvent<OverlayRefreshEventArgs>)(object)UnityContainerExtensions.Resolve<IEventAggregator>(_container, Array.Empty<ResolverOverride>()).GetEvent<OverlayRefreshEvent>()).Subscribe((Action<OverlayRefreshEventArgs>)OnOverlayRefresh);
			if (_economicsOverlay != null)
			{
				_economicsOverlay.Load();
			}
		}

		public void Unload()
		{
			if (_economicsOverlay != null)
			{
				_economicsOverlay.Unload();
			}
			if (_container != null)
			{
				((CompositePresentationEvent<OverlayRefreshEventArgs>)(object)UnityContainerExtensions.Resolve<IEventAggregator>(_container, Array.Empty<ResolverOverride>()).GetEvent<OverlayRefreshEvent>()).Unsubscribe((Action<OverlayRefreshEventArgs>)OnOverlayRefresh);
			}
		}
	}

	internal class GenericUnitOpCollection : IUnitOpCollection
	{
		private readonly IUnityContainer _container;

		private List<EquipmentCostItem> _equipmentCostItems;

		private IDictionary<string, Color> _colors;

		public List<EquipmentCostItem> EquipmentCostItems
		{
			set
			{
				_equipmentCostItems = value;
			}
		}

		public IDictionary<string, Color> Colors
		{
			get
			{
				if (_colors == null)
				{
					_colors = new SortedList<string, Color>();
				}
				if (_equipmentCostItems != null)
				{
					foreach (EquipmentCostItem equipmentCostItem in _equipmentCostItems)
					{
						if (!_colors.ContainsKey(equipmentCostItem.Handle))
						{
							_colors.Add(equipmentCostItem.Handle, GetUnitOpColor(equipmentCostItem.ColorFormat));
						}
					}
				}
				return _colors;
			}
			internal set
			{
				_colors = value;
			}
		}

		internal GenericUnitOpCollection(IUnityContainer container)
		{
			if (container != null)
			{
				_container = container;
			}
		}

		public void SetBlockNameHandleMaps()
		{
			if (_container == null || _equipmentCostItems == null || _equipmentCostItems.Count() <= 0)
			{
				return;
			}
			IDictionary<string, string> dictionary = new Dictionary<string, string>();
			IFlowsheetOverlayManager val = UnityContainerExtensions.Resolve<IFlowsheetOverlayManager>(_container, Array.Empty<ResolverOverride>());
			if (val == null)
			{
				return;
			}
			IFlowsheetOverlay val2 = null;
			List<string> list = new List<string>();
			foreach (EquipmentCostItem equipmentCostItem in _equipmentCostItems)
			{
				if (list.IndexOf(equipmentCostItem.Unit) == -1)
				{
					list.Add(equipmentCostItem.Unit);
				}
			}
			foreach (string name in val.Names)
			{
				val2 = val.GetOverlay(name);
				if (val2 == null)
				{
					continue;
				}
				dictionary = val2.MapBlockNamesToHandles(list);
				foreach (EquipmentCostItem equipmentCostItem2 in _equipmentCostItems)
				{
					if (dictionary.ContainsKey(equipmentCostItem2.Unit))
					{
						string value = string.Empty;
						dictionary.TryGetValue(equipmentCostItem2.Unit, out value);
						equipmentCostItem2.Handle = value;
					}
				}
			}
		}

		private Color GetUnitOpColor(int nColor)
		{
			return (Color)ColorConverter.ConvertFromString($"#{nColor:X6}");
		}

		public UserControl GetPopup(string key)
		{
			foreach (EquipmentCostItem equipmentCostItem in _equipmentCostItems)
			{
				if (equipmentCostItem.Handle == key)
				{
					return GetPopup(equipmentCostItem.Unit, equipmentCostItem);
				}
			}
			return null;
		}

		private UserControl GetPopup(string key, EquipmentCostItem costitem)
		{
			return new SimpleStatusPopup(key, costitem);
		}
	}
	public interface IUnitOpCollection
	{
		UserControl GetPopup(string key);
	}

		{
			if (base.Parent is Popup popup)
			{
				popup.IsOpen = false;
			}
		}

		{
			if (!_contentLoaded)
			{

				Uri resourceLocator = new Uri("/AspenTech.EconomicsAnalysis.Overlays;component/views/simplelstatuspopup.xaml", UriKind.Relative);
				Application.LoadComponent(this, resourceLocator);
			}
		}

		[DebuggerNonUserCode]

		[EditorBrowsable(EditorBrowsableState.Never)]

	public class ViewModelBase : INotifyPropertyChanged
	{
		public event PropertyChangedEventHandler PropertyChanged;

		protected void OnPropertyChanged(object sender, string propertyName)
		{
			if (this.PropertyChanged != null)
			{
				this.PropertyChanged(sender, new PropertyChangedEventArgs(propertyName));
			}
		}

		protected bool WindowOpened(string title)
		{
			foreach (Window window in Application.Current.Windows)
			{
				if (window.Title == title)
				{
					window.Activate();
					return true;
				}
			}
			return false;
		}

		internal Window ShowWindow(UserControl child, string title)
		{
			Window window = new Window();
			Grid grid = new Grid();
			Thumb thumb = new Thumb();
			grid.Children.Add(thumb);
			grid.Children.Add(child);
			window.Content = grid;
			window.WindowStyle = WindowStyle.None;
			window.ResizeMode = ResizeMode.NoResize;
			window.SizeToContent = SizeToContent.WidthAndHeight;
			window.Title = title;
			window.Tag = "EconomicsAnalysis";
			window.Owner = Application.Current.MainWindow;
			window.WindowStartupLocation = WindowStartupLocation.CenterOwner;
			window.ShowInTaskbar = false;
			window.BorderThickness = new Thickness(1.0);
			window.BorderBrush = Brushes.LightSteelBlue;
			window.MouseDown += delegate(object sender, MouseButtonEventArgs e)
			{
				window.Dispatcher.BeginInvoke(DispatcherPriority.Input, (Action)delegate
				{
					thumb.RaiseEvent(e);
				});
			};
			thumb.DragDelta += delegate(object sender, DragDeltaEventArgs e)
			{
				window.Left += e.HorizontalChange;
				window.Top += e.VerticalChange;
			};
			window.Show();
			return window;
		}
	}
