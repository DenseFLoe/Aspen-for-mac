		private void _toolbarSelectedNodes_ToolClick(object sender, ToolClickEventArgs e)
		{
			switch (e.Tool.Key)
			{
			case "Add To Selection":

				break;
			case "Remove From Selection":
				RemoveFromSelection((IEnumerable<CxsNode>)_gridSelectedNodes.SelectedNodes);
				break;
			case "Add From Clipboard":
				AddToSelectionFromClipboard();
				break;
			}
		}

		public bool IsInSelection(CxsNode node)
		{
			foreach (CxsNode item in (Collection<CxsNode>)(object)_Selection)
			{
				if (node == item)
				{
					return true;
				}
			}
			return false;
		}

		public bool AddToSelectionFromClipboard()
		{
			CxsCaseInfo caseInfoFromClipboard = CxsServices.GetCaseInfoFromClipboard();
			if (caseInfoFromClipboard == null || string.IsNullOrEmpty(caseInfoFromClipboard.FileMoniker))
			{
				return false;
			}
			bool result = false;
			foreach (object rootObject in _explorerView.RootObjects)
			{
				CxsCase val = (CxsCase)((rootObject is CxsCase) ? rootObject : null);
				if (val != null && val.CaseFilePath == caseInfoFromClipboard.FileMoniker)
				{
					result = true;
					string[] clipboardPaths = val.GetClipboardPaths();
					CxsNodeCollection nodes = val.ResolveNodes(clipboardPaths);
					AddToSelection((IEnumerable<CxsNode>)nodes);
					break;
				}
			}
			return result;
		}

		public void AddToSelection(CxsNode node)
		{
			if (node != null)
			{
				List<CxsNode> list = new List<CxsNode>();
				list.Add(node);
				AddToSelection((IEnumerable<CxsNode>)list);
			}
		}

		public void AddToSelection(IEnumerable<CxsNode> nodes)
		{
			if (nodes == null)
			{
				return;
			}
			List<CxsNode> list = new List<CxsNode>();
			foreach (CxsNode node in nodes)
			{
				if (node != null && !IsInSelection(node))
				{
					list.Add(node);
				}
			}
			if (list.Count == 0)
			{
				return;
			}
			if (this.AddingToSelection != null)
			{
				this.AddingToSelection(this, new AddingToSelectionEventArgs(list));
			}
			if (list.Count == 0)
			{
				return;
			}
			if (!MultiSelectRows && ((Collection<CxsNode>)(object)_Selection).Count > 0)
			{
				RemoveFromSelection(((Collection<CxsNode>)(object)_Selection)[0]);
			}
			List<CxsNode> list2 = new List<CxsNode>();
			foreach (CxsNode item in list)
			{
				if (item != null)
				{
					((Collection<CxsNode>)(object)_Selection).Add(item);
					list2.Add(item);
				}
			}
			if (this.AddedToSelection != null)
			{
				this.AddedToSelection(this, new AddedToSelectionEventArgs(list2));
			}
		}

		public void RemoveFromSelection(CxsNode node)
		{
			if (node != null)
			{
				List<CxsNode> list = new List<CxsNode>();
				list.Add(node);
				RemoveFromSelection((IEnumerable<CxsNode>)list);
			}
		}

		public void RemoveFromSelection(IEnumerable<CxsNode> nodes)
		{
			List<CxsNode> list = new List<CxsNode>();
			foreach (CxsNode node in nodes)
			{
				if (node == null)
				{
					continue;
				}
				foreach (CxsNode item in (Collection<CxsNode>)(object)_Selection)
				{
					if (item == node)
					{
						list.Add(item);
					}
				}
			}
			if (this.RemovingFromSelection != null)
			{
				this.RemovingFromSelection(this, new RemovingFromSelectionEventArgs(list));
			}
			if (list.Count == 0)
			{
				return;
			}
			List<CxsNode> list2 = new List<CxsNode>();
			foreach (CxsNode item2 in list)
			{
				if (item2 != null && ((Collection<CxsNode>)(object)_Selection).Remove(item2))
				{
					list2.Add(item2);
				}
			}
			if (this.RemovedFromSelection != null)
			{
				this.RemovedFromSelection(this, new RemovedFromSelectionEventArgs(list2));
			}
		}

		public void NavigateToNode(CxsNode node)
		{
			if (node == null)
			{
				return;
			}
			ExplorerTreeNode explorerTreeNode = null;
			if (!node.IsLeaf)
			{
				explorerTreeNode = _TreeRoot.ResolveToNode(node);
				if (explorerTreeNode != null)
				{
					_navigationBar.NavigateTo(explorerTreeNode.FullPath, fireEvent: true);
				}
				return;
			}
			string parentPath = node.ParentPath;
			if (string.IsNullOrEmpty(parentPath))
			{
				explorerTreeNode = _TreeRoot.ResolveToCase(node.Case);
			}
			else
			{
				explorerTreeNode = _TreeRoot.ResolveToNode(node.Case.ResolveNode(parentPath));
			}
			if (ListViewVisible)
			{
				_gridListView.FocusedNode = node;
			}
			else
			{
				_gridSearchResults.FocusedNode = node;
			}
		}

			//IL_0075: Unknown result type (might be due to invalid IL or missing references)

			switch (e.Tool.Key)
			{
			case "Add Condition":
			{
				UltraTreeNode activeNode2 = _treeSearchExpression.ActiveNode;
				if (activeNode2 != null)
				{
					CxsSearchExpression val2 = new CxsSearchExpression();
					val2.Operator = (CxsSearchExpressionOperator)2;
					val2.TestProperty = "Id";
					AddExpressionToNode(activeNode2, val2);
				}
				break;
			}
			case "Add Group":
			{
				UltraTreeNode activeNode3 = _treeSearchExpression.ActiveNode;
				if (activeNode3 != null)
				{
					CxsSearchExpression val3 = new CxsSearchExpression();
					val3.Operator = (CxsSearchExpressionOperator)0;
					AddExpressionToNode(activeNode3, val3);
				}
				break;
			}
			case "Remove":
				if (_treeSearchExpression.ActiveNode.DataKey != _SearchExpression)
				{
					UltraTreeNode activeNode = _treeSearchExpression.ActiveNode;
					object dataKey = activeNode.DataKey;
					CxsSearchExpression item = (CxsSearchExpression)((dataKey is CxsSearchExpression) ? dataKey : null);
					UltraTreeNode ultraTreeNode = activeNode.Parent;
					object dataKey2 = ultraTreeNode.DataKey;
					CxsSearchExpression val = (CxsSearchExpression)((dataKey2 is CxsSearchExpression) ? dataKey2 : null);
					val.Subexpressions.Remove(item);
					RefreshQueryTree(ultraTreeNode, val);
					ultraTreeNode.BringIntoView();
					_treeSearchExpression.ActiveNode = ultraTreeNode;
					ultraTreeNode.Selected = true;
				}
				break;
			}
		}

		private void _gridListView_OnAccessUnboundOverride(object sender, CxsNodeOverrideEventArgs e)
		{
			if (!(e.Override.Name == "Label") || !e.IsGet)
			{
				return;
			}
			if (ListViewExpansionDepth <= 1)
			{
				e.Value = e.Node.Label;
				return;
			}
			ExplorerTreeNode activeTreeNode = ActiveTreeNode;
			if (activeTreeNode == null || activeTreeNode.IsCase || activeTreeNode.IsRoot)
			{
				e.Value = e.Node.Path;
			}
			else
			{
				e.Value = CxsPath.RelativePathFromParent(activeTreeNode.Node.Path, e.Node.Path);
			}
		}

					if (indexChild < 0)
					{
						indexChild = _SearchEngine.IndexChildNum;
					}
					stringBuilder.Append($"{text} ({text2}={indexLevel}, {text3}: {indexChild})");
				}
			}
			else
			{
				if (isIndexing)
				{
					stringBuilder.Append($"Indexing (level: {indexLevel}, child: {indexChild})");
				}
				else
				{
					stringBuilder.Append(status);
				}
				stringBuilder.Append("... ");
				stringBuilder.Append(GetSearchResultString(nodesSearched, nodesMatched));
			}
			return stringBuilder.ToString();
		}

		private void UpdatePauseResumeResponseButton()
		{
			if (_SearchEngine != null)
			{
				string text = "";
				string text2 = "";
				bool flag = true;
				bool flag2 = true;
				if (_SearchEngine.SearchComplete)
				{
					text = Global.GetString("CxsNodeExplorer_PauseSearch");
					text2 = Global.GetString("CxsNodeExplorer_BeginSearch");
					flag = false;
					flag2 = true;
					SetAnimateState(AnimateState.Completed);
				}
				else if (_SearchEngine.IsPaused)
				{
					text = Global.GetString("CxsNodeExplorer_ResumeSearch");
					text2 = Global.GetString("CxsNodeExplorer_RestartSearch");
					flag = !_SearchEngine.PausePending;
					flag2 = true;
					SetAnimateState(AnimateState.Paused);
				}
				else
				{
					text = Global.GetString("CxsNodeExplorer_PauseSearch");
					text2 = Global.GetString("CxsNodeExplorer_RestartSearch");
					flag = !_SearchEngine.PausePending;
					flag2 = false;
					SetAnimateState(AnimateState.Running);
				}
				((Control)(object)_buttonSearchPauseResume).Enabled = !_pendingResume && flag;
				((Control)(object)_buttonSearchPauseResume).Text = text;
				((Control)(object)_buttonSearchBegin).Text = text2;
				((Control)(object)_buttonSearchBegin).Enabled = flag2;
			}
		}

			}

		public void SynchronizeGridWithColumnSpecs()
		{
			if (!_LayoutInitialized || _WaitingForEndInit || _gridControl.DisplayLayout.Bands.Count == 0)
			{
				return;
			}
			UltraGridBand ultraGridBand = _gridControl.DisplayLayout.Bands[0];
			List<UltraGridColumn> list = new List<UltraGridColumn>();
			foreach (UltraGridColumn column in ultraGridBand.Columns)
			{
				list.Add(column);
			}
			CxsNodeGridColumnCollection cxsNodeGridColumnCollection = ((_Columns.Count <= 0) ? _DefaultColumns : _Columns);
			int num = 0;
			foreach (CxsNodeGridColumn item in cxsNodeGridColumnCollection)
			{
				if (item.VisiblePosition >= 0)
				{
					num = item.VisiblePosition + 1;
				}
			}
			foreach (CxsNodeGridColumn item2 in cxsNodeGridColumnCollection)
			{
				if (string.IsNullOrEmpty(item2.Key))
				{
					continue;
				}
				UltraGridColumn ultraGridColumn = null;
				foreach (UltraGridColumn item3 in list)
				{
					if (item3.Key == item2.Key)
					{
						ultraGridColumn = item3;
						break;
					}
				}
				if (ultraGridColumn != null)
				{
					list.Remove(ultraGridColumn);
				}
				else
				{
					ultraGridColumn = ultraGridBand.Columns.Add(item2.Key, item2.Caption);
				}
				if (!string.IsNullOrEmpty(item2.Caption))
				{
					ultraGridColumn.Header.Caption = item2.Caption;
				}
				if (item2.VisiblePosition >= 0)
				{
					ultraGridColumn.Header.VisiblePosition = item2.VisiblePosition;
				}
				else

		public void SynchronizeColumnSpecsWithGrid()
		{
			if (_gridControl.DisplayLayout.Bands.Count == 0)
			{
				return;
			}
			bool raiseListChangedEvents = _Columns.RaiseListChangedEvents;
			try
			{
				UltraGridBand ultraGridBand = _gridControl.DisplayLayout.Bands[0];
				_Columns.RaiseListChangedEvents = false;
				_Columns.Clear();
				foreach (UltraGridColumn column in ultraGridBand.Columns)
				{
					CxsNodeGridColumn cxsNodeGridColumn = new CxsNodeGridColumn();
					cxsNodeGridColumn.Key = column.Key;
					cxsNodeGridColumn.Caption = column.Header.Caption;
					cxsNodeGridColumn.Hidden = column.Hidden;
					cxsNodeGridColumn.VisiblePosition = column.Header.VisiblePosition;
					cxsNodeGridColumn.Width = column.Width;
					switch (column.ExcludeFromColumnChooser)
					{
					case ExcludeFromColumnChooser.Default:
						cxsNodeGridColumn.ExcludeFromColumnChooser = null;
						break;
					case ExcludeFromColumnChooser.True:
						cxsNodeGridColumn.ExcludeFromColumnChooser = true;
						break;
					case ExcludeFromColumnChooser.False:
						cxsNodeGridColumn.ExcludeFromColumnChooser = false;
						break;
					}
					_Columns.Add(cxsNodeGridColumn);
				}
			}
			finally
			{
				_Columns.RaiseListChangedEvents = raiseListChangedEvents;
			}
		}

		public bool ShouldSerializeColumns()
		{
			if (_Columns.Count != _DefaultColumns.Count)
			{
				return true;
			}
			for (int i = 0; i < _Columns.Count; i++)
			{
				if (!_Columns[i].Equals(_DefaultColumns[i]))

					return true;
				}
			}
			return false;
		}

		{

				{
					list.Add(val);
				}
			}
			bool flag = false;
			List<CxsNode> list2 = new List<CxsNode>();
			foreach (CxsNode item in (Collection<CxsNode>)(object)_SelectedNodes)
			{
				int num = list.IndexOf(item);
				if (num >= 0)
				{
					list.RemoveAt(num);
					flag = true;
				}
				else
				{
					list2.Add(item);
				}
			}
			if (list.Count > 0 || list2.Count > 0)
			{
				flag = true;
			}
			foreach (CxsNode item2 in list2)
			{
				((Collection<CxsNode>)(object)_SelectedNodes).Remove(item2);
			}
			foreach (CxsNode item3 in list)
			{
				((Collection<CxsNode>)(object)_SelectedNodes).Add(item3);
			}
			if (flag && this.SelectedNodesChanged != null)
			{
				this.SelectedNodesChanged(this, new EventArgs());
			}
		}
