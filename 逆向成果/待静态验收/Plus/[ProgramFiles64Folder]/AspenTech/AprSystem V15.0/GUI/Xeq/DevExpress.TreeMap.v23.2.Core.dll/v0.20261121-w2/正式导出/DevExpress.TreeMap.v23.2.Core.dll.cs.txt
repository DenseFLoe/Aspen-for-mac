		private static Tuple<double, double> CheckIntersectWithArcAngGetCoordinates(double starAngle, double endAngle, double y, double intersectX, ref double firstIntersectX)
		{
			double num = Math.Atan2(y, intersectX);
			if (starAngle >= 0.0 && endAngle >= 0.0 && num < 0.0)
			{
				num += Math.PI * 2.0;
			}
			else if (starAngle < 0.0 && endAngle < 0.0 && num > 0.0)
			{
				num -= Math.PI * 2.0;
			}
			if (num <= starAngle && num >= endAngle)
			{
				return TryGetCoordinates(intersectX, ref firstIntersectX);
			}
			return null;
		}

		private static Tuple<double, double> TryGetCoordinates(double intersectX, ref double firstIntersectX)
		{
			if (!double.IsNaN(firstIntersectX))
			{
				return new Tuple<double, double>(Math.Min(firstIntersectX, intersectX), Math.Max(firstIntersectX, intersectX));
			}
			firstIntersectX = intersectX;
			return null;

			if (!(num2 > 0.0))
			{
				return;

			double num4 = 0.0;
			for (int num5 = 0; num5 < array.Length; num5++)
			{
				double weight = array[num5].Weight;
				if (array[num5].Weight > 0.0 && num2 > 0.0)
				{
					int count = list.Count;
					if (count == 0)
					{
						treeMapRect = RectCalculatorDirection.CalculateFirstRect(num5, treeMapRect2, array, flag, num2);
						num4 = Math.Max(treeMapRect.Height / treeMapRect.Width, treeMapRect.Width / treeMapRect.Height);
						list.Add(treeMapRect);
					}
					else
					{
						double num6 = 0.0;
						double[] sides = GetSides(array, num, num5 - num + 1, flag ? treeMapRect2.Height : treeMapRect2.Width);
						double anchorSide = array[num].Weight / sides[0] * num3;
						List<TreeMapRect> list2 = new List<TreeMapRect>();
						list2.AddRange(list);

						double num7 = 0.0;
						double num8 = 0.0;
						for (int num9 = 0; num9 <= count; num9++)
						{
							TreeMapRect item = RectCalculatorDirection.CalculateRect(sides[num9], anchorSide, num3, treeMapRect2, num6, flag, width, height);
							num6 += (flag ? item.Height : item.Width);
							num7 = Math.Max(item.Width / item.Height, item.Height / item.Width);
							if (num7 < num4)
							{
								list.Add(item);
								if (num8 < num7)
								{
									num8 = num7;
								}
								continue;
							}
							if (NeedContinueRow(width, height, treeMapRect2, list2[0]))
							{
								list.Add(item);
								continue;
							}
							FillItemsLayout(array, list2, num);
							list.Clear();
							treeMapRect2 = RectCalculatorDirection.CalculateNewInitialRect(treeMapRect2, list2[0], flag);
							num = num5;

							num4 = Math.Max(treeMapRect.Height / treeMapRect.Width, treeMapRect.Width / treeMapRect.Height);

	{
		private LayoutAlgorithmDirection direction;

		private SliceAndDiceAlgorithmLayoutMode mode;

		public SliceAndDiceAlgorithm()
			: this(LayoutAlgorithmDirection.TopLeftToBottomRight, SliceAndDiceAlgorithmLayoutMode.Auto)

		}

		public SliceAndDiceAlgorithm(LayoutAlgorithmDirection direction, SliceAndDiceAlgorithmLayoutMode mode)
		{
			this.direction = direction;
			this.mode = mode;
		}

		private bool IsVerticalDirection(double width, double height)
		{
			return mode switch
			{
				SliceAndDiceAlgorithmLayoutMode.Vertical => false, 
				SliceAndDiceAlgorithmLayoutMode.Horizontal => true, 
				_ => width < height, 
			};
		}

		private bool IsFromLeft()
		{
			if (direction != LayoutAlgorithmDirection.TopLeftToBottomRight)
			{
				return direction == LayoutAlgorithmDirection.BottomLeftToTopRight;
			}
			return true;
		}

		private bool IsFromTop()
		{
			if (direction != LayoutAlgorithmDirection.TopLeftToBottomRight)
			{
				return direction == LayoutAlgorithmDirection.TopRightToBottomLeft;
			}
			return true;
		}

		private void GetCoefficients(double width, double height, bool isVertical, out double begin, out int sign, out int coef)
		{
			if (isVertical)
			{
				if (IsFromTop())
				{
					begin = 0.0;
					sign = 1;
				}
				else
				{
					begin = height;
					sign = -1;
				}
			}
			else if (IsFromLeft())
			{
				begin = 0.0;
				sign = 1;
			}
			else
			{
				begin = width;
				sign = -1;
			}
			coef = ((sign == -1) ? (-1) : 0);
		}

		public override void Calculate(IEnumerable<ITreeMapItemLayout> sortedData, double width, double height)
		{
			double num = 0.0;
			double num2 = sortedData.Sum((ITreeMapItemLayout d) => (!(d.Weight > 0.0)) ? 0.0 : d.Weight);
			bool flag = IsVerticalDirection(width, height);
			GetCoefficients(width, height, flag, out var begin, out var sign, out var coef);
			foreach (ITreeMapItemLayout sortedDatum in sortedData)
			{
				if (sortedDatum.Weight > 0.0)
				{
					TreeMapRect layout = default(TreeMapRect);
					if (flag)
					{
						layout.Width = width;
						layout.Height = sortedDatum.Weight / num2 * height;
						layout.X = 0.0;
						layout.Y = begin + num * (double)sign + layout.Height * (double)coef;
						num += layout.Height;
					}
					else
					{
						layout.Width = sortedDatum.Weight / num2 * width;
						layout.Height = height;
						layout.X = begin + num * (double)sign + layout.Width * (double)coef;
						layout.Y = 0.0;
						num += layout.Width;
					}
					sortedDatum.Layout = layout;
				}
				else
				{
					sortedDatum.Layout = new TreeMapRect(0.0, 0.0, 0.0, 0.0);
				}
			}
		}

	public class PatternParser
	{
		private class PatternFragment
		{
			private readonly string fragment;

			private readonly string format;

			private readonly PatternDataProvider dataProvider;

			public string Fragment => fragment;

			public PatternFragment(string fragment, string format, PatternDataProvider dataProvider)
			{
				this.fragment = fragment;
				this.format = format;
				this.dataProvider = dataProvider;
			}

			public string GetText()
			{
				return dataProvider.GetText(format);
			}

			public void SetContext(object context)
			{
				dataProvider.TryAddContext(context);
			}
		}

		private static Dictionary<string, PatternConstants> constants;

		private readonly string pattern;

		private readonly IPatternHolder patternHolder;

		private readonly List<PatternFragment> fragments = new List<PatternFragment>();

		static PatternParser()
		{
			constants = new Dictionary<string, PatternConstants>();
			constants.Add("L", PatternConstants.Label);
			constants.Add("V", PatternConstants.Value);
			constants.Add("TV", PatternConstants.TotalValue);
			constants.Add("V1", PatternConstants.Value1);
			constants.Add("V2", PatternConstants.Value2);
		}

		public PatternParser(string pattern, IPatternHolder patternHolder)
		{
			this.pattern = pattern;
			this.patternHolder = patternHolder;
			Parse();
		}

		private void Parse()
		{
			List<string> list = SplitString(pattern, '{', '}');
			if (list == null)
			{
				return;
			}
			foreach (string item in list)
			{
				if (PrepareFragment(item, out var patternConstant, out var format))
				{
					PatternDataProvider dataProvider = patternHolder.GetDataProvider(patternConstant.Value);
					if (dataProvider != null)
					{
						fragments.Add(new PatternFragment(item, format, dataProvider));
					}
				}
			}
		}

		private bool PrepareFragment(string fragment, out PatternConstants? patternConstant, out string format)
		{
			patternConstant = null;
			format = string.Empty;
			if (!fragment.StartsWith("{") || !fragment.EndsWith("}"))
			{
				return false;
			}
			string text = fragment.Substring(1, fragment.Length - 2);
			int num = text.IndexOf(":");
			if (num >= 0)
			{
				format = text.Substring(num + 1).Trim();
				text = text.Substring(0, num);
			}
			text = text.Trim().ToUpper();
			if (!constants.ContainsKey(text))
			{
				return false;
			}
			patternConstant = constants[text];
			return true;
		}

		private List<string> SplitString(string splittingString, char leftSeparator, char rightSeparator)
		{
			List<string> list = new List<string>();
			int num = 0;
			int num2 = 0;
			int num3 = 0;
			if (!string.IsNullOrEmpty(splittingString))
			{
				foreach (char c in splittingString)
				{
					if (c == leftSeparator)
					{
						num = num3;
					}
					else if (c == rightSeparator)
					{
						num2 = num3;
						list.Add(splittingString.Substring(num, num2 - num + 1));
					}
					num3++;
				}
				return list;
			}
			return null;
		}

		public void SetContext(params object[] contexts)
		{
			foreach (PatternFragment fragment in fragments)
			{
				foreach (object context in contexts)
				{
					fragment.SetContext(context);
				}
			}
		}

		public string GetText()
		{
			string text = pattern;
			foreach (PatternFragment fragment in fragments)
			{
				text = text.Replace(fragment.Fragment, fragment.GetText());
			}
			return text;
		}
	}

	public class TreeMapItemHelper
	{
		private static bool FillItemPath<T>(T requestItem, List<T> itemPath, IEnumerable<T> items) where T : class, IHierarchicalItem
		{
			foreach (T item in items)
			{
				if (requestItem == item)
				{
					itemPath.Add(item);
					return true;
				}
				if (item.Children.Count > 0 && FillItemPath(requestItem, itemPath, (IEnumerable<T>)item.Children))
				{
					itemPath.Insert(0, item);
					return true;
				}
			}
			return false;
		}

		public static List<T> GetItemPath<T>(T item, IEnumerable<T> items) where T : class, IHierarchicalItem
		{
			List<T> list = new List<T>();
			FillItemPath(item, list, items);
			return list;
		}

		public static bool IsGroupItem(IHierarchicalItem item)
		{
			if (item.Children != null)
			{
				return item.Children.Count > 0;
			}
			return false;
		}
