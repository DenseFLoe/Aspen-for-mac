		public void PostConvert(IHNode pNode)
		{
			IHNode iHNode = pNode.FindNode("Ports");
			IHNodeCol elements = iHNode.Elements;
			pNode?.HoldNotifies(hold: true);
			if (elements != null)
			{
				for (int i = 0; i < streamList.Count; i++)
				{
					StreamConnection streamConnection = streamList[i] as StreamConnection;
					for (int j = 0; j < elements.Count; j++)
					{
						string text = elements.get_Label(0, j, force);
						if (text.CompareTo(streamConnection.blockPort) == 0)
						{
							string streamName = streamConnection.streamName;
							iHNode.FindNode(text).Elements?.Add(streamName);
							break;
						}
					}
				}
			}
			pVar = pNode.FindNode("Input/CONDENSER");
			if (pVar != null)
			{
				pVar.set_Value(force, (object)"TOTAL");
			}
			pVar = pNode.FindNode("Input/NSTAGE");
			if (pVar != null)
			{
				ConvertData convertData = variableList[0] as ConvertData;
				pVar.set_Value(convertData.force, convertData.Value);
			}
			pVar = pNode.FindNode("Input/FEED_STAGE");
			if (pVar != null)
			{
				IHNode iHNode2 = pVar.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing];
				ConvertData convertData2 = variableList[1] as ConvertData;
				iHNode2.set_Value(convertData2.force, convertData2.Value);
			}
			pVar = pNode.FindNode("Input/FEED_CONVEN");
			if (pVar != null)
			{
				pVar.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].set_Value(force, (object)"ON-STAGE");
			}
			pVar = pNode.FindNode("Input/FEED_CONVE2");
			if (pVar != null)
			{
				pVar.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].set_Value(force, (object)"ON-STAGE");
			}
			if (sRatioType.CompareTo("RR") == 0)
			{
				pVar = pNode.FindNode("Input/BASIS_RR");
			}
			else
			{
				pVar = pNode.FindNode("Input/BASIS_BR");
			}
			if (pVar != null)
			{
				ConvertData convertData3 = variableList[2] as ConvertData;
				pVar.set_Value(convertData3.force, convertData3.Value);
			}
			if (sRatioType.CompareTo("RR") == 0)

				pVar = pNode.FindNode("Input/RR_BASIS");

				pVar = pNode.FindNode("Input/BR_BASIS");
			}

			{
				pVar.set_Value(force, (object)sBasis);
			}
			pVar = pNode.FindNode("Input/PRES1");
			if (pVar != null)
			{
				ConvertData convertData4 = varWithUnitList[0] as ConvertData;
