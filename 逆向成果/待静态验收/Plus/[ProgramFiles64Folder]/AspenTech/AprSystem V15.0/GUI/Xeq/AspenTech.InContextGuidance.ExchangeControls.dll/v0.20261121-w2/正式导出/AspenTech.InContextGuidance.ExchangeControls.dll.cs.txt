namespace AspenTech.InContextGuidance.ExchangeControls
{
	public class ELearningRibbonButtonTool : ButtonTool
	{
		private class ToolTipContentConverter : IValueConverter
		{
			public object Convert(object value, Type targetType, object parameter, CultureInfo culture)
			{
				if (!(bool)value)
				{
					return InContextGuidanceResources.ExchangeTools_ELearningRibbon_Tooltip_Disabled;
				}
				return InContextGuidanceResources.ExchangeTools_ELearningRibbon_Tooltip_Enabled;
			}

			public object ConvertBack(object value, Type targetType, object parameter, CultureInfo culture)
			{
				throw new NotImplementedException();
			}
		}

		public ELearningRibbonButtonTool()
		{
			base.Caption = InContextGuidanceResources.ExchangeTools_ELearningRibbon_Caption;
			base.LargeImage = ImageResources.ELearningButtonImg;
			SetBinding(UIElement.IsEnabledProperty, new Binding("SuccessfullyInitialized")
			{
				Source = ExchangeAdapter.Default,
				Mode = BindingMode.OneWay
			});
			RibbonGroup.SetMaximumSize(this, RibbonToolSizingMode.ImageAndTextLarge);
			XamRibbonScreenTip xamRibbonScreenTip = new XamRibbonScreenTip
			{
				Header = InContextGuidanceResources.ExchangeTools_ELearningRibbon_Caption,
				FooterSeparatorVisibility = Visibility.Hidden
			};
			xamRibbonScreenTip.SetBinding(ContentControl.ContentProperty, new Binding
			{
				Source = this,
				Path = new PropertyPath(UIElement.IsEnabledProperty),
				Converter = new ToolTipContentConverter(),
				Mode = BindingMode.OneWay
			});
			base.ToolTip = xamRibbonScreenTip;
		}

		protected override void OnClick()
		{
			ExchangeAdapter.Default.OpenElearning();
			base.OnClick();
		}
	}

namespace AspenTech.InContextGuidance.ExchangeControls.Images
{
	internal static class ImageResources
	{
		private static readonly Dictionary<string, ImageSource> _imageSources = new Dictionary<string, ImageSource>();

		public static ImageSource AspenONEExchangeButtonImg => GetImageSource("InContextGuidance.png");

		public static ImageSource ELearningButtonImg => GetImageSource("eLearning.png");

		private static ImageSource GetImageSource(string imageFileName)
		{
			if (!_imageSources.ContainsKey(imageFileName))
			{
				StringBuilder stringBuilder = new StringBuilder();
				stringBuilder.Append("pack://application:,,,/");
				stringBuilder.Append(typeof(ImageResources).Assembly.FullName);
				stringBuilder.Append(";component/Images/");
				stringBuilder.Append(imageFileName);
				_imageSources.Add(imageFileName, new BitmapImage(new Uri(stringBuilder.ToString(), UriKind.RelativeOrAbsolute)));
			}
			return _imageSources[imageFileName];
		}
	}
