		}

		private void App_RunFinished()
		{
			lock (this)
			{
				if (!(_isRunning ?? true))
				{
					return;
				}
				if (worker != null)
				{
					worker.DoWork -= worker_DoWork;
					worker.Dispose();
					worker = null;
				}
				_isRunning = false;
			}

			{
				App.UpdateControlsInScalars(App.Scalars);
				App.UpdateControlsInArrays(App.Arrays);
				if (App.Name.Equals("Tasc"))

					IATVScalar val = App.ResultScalars["OptPathItemNum"];
					short result = 0;
					if (val != null && short.TryParse(val.ValuePureSI.ToString(), out result))
					{
						string text = App.ResultArrays["arDesignStatus"][result];
						if (!string.IsNullOrWhiteSpace(text))
						{
							if (text == "0")
							{
								MessageBox.Show(AspenTech.ExchangerDesign.UserInterface.Properties.Resources.SizingNearFailMessage, AspenTech.ExchangerDesign.UserInterface.Properties.Resources.btn_RunStatus, MessageBoxButton.OK, MessageBoxImage.Asterisk);
							}
							else if (text == "1")

								MessageBox.Show(AspenTech.ExchangerDesign.UserInterface.Properties.Resources.SizingNearMessage, AspenTech.ExchangerDesign.UserInterface.Properties.Resources.btn_RunStatus, MessageBoxButton.OK, MessageBoxImage.Asterisk);
							}
						}
					}
				}
				else if (App.Name.Equals("Acol"))
				{
					IATVScalar val2 = App.ResultScalars["OptPathItemNum"];
					short result2 = 0;
					if (val2 != null && short.TryParse(val2.ValuePureSI.ToString(), out result2))
					{
						string text2 = App.ResultArrays["arAreaRatio"][result2];
						string text3 = App.ResultArrays["arDpRatioTS"][result2];
						string text4 = App.ResultArrays["arDpRatioOT"][result2];
						if ((!string.IsNullOrEmpty(text2) && text2.Equals(" *")) || (!string.IsNullOrEmpty(text3) && text3.Equals(" *")) || (!string.IsNullOrEmpty(text3) && text4.Equals(" *")))
						{
							MessageBox.Show(AspenTech.ExchangerDesign.UserInterface.Properties.Resources.SizingNearMessage, AspenTech.ExchangerDesign.UserInterface.Properties.Resources.btn_RunStatus, MessageBoxButton.OK, MessageBoxImage.Asterisk);
						}
					}
				}
				if (!_isInteractiveSizing && App.ResultStatus == 1)
				{
					object app = App;
					object aTVService = App.ATVService;
					((IATVService)((aTVService is IATVService) ? aTVService : null)).UpdateApp(ref app);
				}
				if (EDRDocument != null)

					try
					{
						EDRDocument.GetType().InvokeMember("OnSizingCompleted", BindingFlags.Instance | BindingFlags.Public | BindingFlags.InvokeMethod, null, EDRDocument, new object[1] { App.Name });
					}
					catch (Exception)
					{
					}
				}
				progress.IsIndeterminate = false;
				if (Utilities.GetAncestorFromType(this, typeof(Window), allowSubclassOfType: true) is Window window)
				{
					UserCancelledProgress = false;
					window.Close();
				}
			});
		}
