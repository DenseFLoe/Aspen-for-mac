			public override DocumentPage GetPage(int pageNumber)
			{
				DocumentPage documentPage = null;
				_report._runningLogicalPageCount = 0;
				if (_report.Sections.Count == 0)
				{
					return null;
				}
				_report.PhysicalPageNumber = pageNumber + PageRange.PageFrom;
				bool flag = pageNumber > 0;
				int count = _report.Sections.Count;
				while (_currentSectionIndex < count)
				{
					ReportSection reportSection = _report.Sections[_currentSectionIndex];
					if (reportSection != null)
					{
						int pageNumber2 = pageNumber + _pageRange.PageFrom - _totalPagesProcessed - 1;
						documentPage = reportSection.GetPage(pageNumber2);
						if (documentPage != null)
						{
							flag = true;
							if (_currentSectionIndex == _report.Sections.Count - 1 && reportSection.IsEndReached)
							{
								_isPageCountValid = true;
							}
							if (pageNumber + _pageRange.PageFrom >= _pageRange.PageTo && _pageRange.PageTo != 0)
							{
								_isPageCountValid = true;
							}
							_maxLogicalNumbers[_currentSectionIndex] = Math.Max(_maxLogicalNumbers[_currentSectionIndex], reportSection.LogicalPageNumber);
							_report.LogicalPageNumber = reportSection.LogicalPageNumber + _report.RunningLogicalPageCount;
							_report.LogicalPagePartNumber = reportSection.LogicalPagePartNumber;
							_report.SectionPhysicalPageNumber = reportSection.PhysicalPageNumber;
							_report.SectionLogicalPageNumber = reportSection.LogicalPageNumber;
							_report.SectionLogicalPagePartNumber = reportSection.LogicalPagePartNumber;
							if (documentPage.Visual is UIElement uIElement)
							{
								uIElement.InvalidateMeasure();
								uIElement.InvalidateArrange();
								uIElement.Measure(documentPage.Size);
								uIElement.Arrange(new Rect(default(Point), documentPage.Size));
								uIElement.UpdateLayout();
							}
							if (_isPreviewMode)
							{
								_pageVisual = CreatePreviewCopy(documentPage.Visual);
							}
							RaisePrintProgress(pageNumber);
							return documentPage;
						}
						if (_currentSectionIndex < count - 1)
						{
							if (flag)
							{
								_totalPagesProcessed += reportSection.GeneratedPageCount;
							}
							else
							{
								int num = ((!(reportSection is EmbeddedVisualReportSection embeddedVisualReportSection)) ? 1 : Math.Max(embeddedVisualReportSection.EstmatedPageCount, 1));
								_totalPagesProcessed += num;
							}
						}
						else
						{
							_totalPagesProcessed += reportSection.GeneratedPageCount;
							_report._runningLogicalPageCount += _maxLogicalNumbers[_currentSectionIndex];
						}
					}
					_currentSectionIndex++;
				}
				if (pageNumber > _totalPagesProcessed)
				{
					throw new ArgumentOutOfRangeException("pageNumber", XamReportPreview.GetString("LE_ReportPaginator_OutOfRangeException"));
				}
				_isPageCountValid = true;
				Size size = new Size(100.0, 100.0);
				return new DocumentPage(new Canvas(), size, new Rect(size), new Rect(size));
			}

			internal void BeginPagination()
			{
				_currentSectionIndex = 0;
				_totalPagesProcessed = 0;
				_maxLogicalNumbers = new List<int>();
				_printPageCount = 0;
				foreach (ReportSection section in _report.Sections)

				}
			}

		internal void ExportToXPS()
		{
			_exportDoc = null;
			_exportPackage = null;
			try
			{
				string text = "";
				if (string.IsNullOrEmpty(ReportSettings.FileName))
				{
					text = Environment.CurrentDirectory + System.IO.Path.DirectorySeparatorChar + _defaultExportFileName;
				}
				else
				{
					string text2 = System.IO.Path.GetDirectoryName(ReportSettings.FileName);
					string text3 = System.IO.Path.GetFileName(ReportSettings.FileName);
					if (string.IsNullOrEmpty(text3))
					{
						text3 = _defaultExportFileName;
					}
					if (!Directory.Exists(text2))
					{
						text2 = Environment.CurrentDirectory;
					}
					text = text2 + System.IO.Path.DirectorySeparatorChar + text3;
				}
				CheckUserSettings();
				_exportPackage = Package.Open(text, FileMode.Create);
				_exportDoc = new XpsDocument(_exportPackage);
				GenerateReport(_exportDoc);
			}
			catch (IOException exception)
			{
				OnWritingCanceled(this, new WritingCancelledEventArgs(exception));
			}
		}

		internal void ExportToXPS(string fileName)
		{
			_exportDoc = null;
			_exportPackage = null;
			try
			{
				if (string.IsNullOrEmpty(fileName))
				{
					ExportToXPS();
					return;

		private void GenerateReport(object reportObject)
		{
			if (reportObject is XpsDocument)
			{
				XpsDocument xpsDocument = reportObject as XpsDocument;
				XpsDocumentWriter writer = XpsDocument.CreateXpsDocumentWriter(xpsDocument);
				PrintTicket printTicket = null;
				PrintQueue printQueue = null;
				try
				{
					printQueue = ReportSettings.PrintQueue;
					if (printQueue != null)
					{
						printTicket = ReportSettings.PrintQueue.DefaultPrintTicket;
					}
				}
				catch (Exception)
				{
				}
				try
				{
					ProcessXpsDocumentWriter(writer, printTicket, printQueue);
					return;
				}
				catch (PrintQueueException)
				{
					InvokeEndPaginationIfNeeded();
					writer = XpsDocument.CreateXpsDocumentWriter(xpsDocument);
					ProcessXpsDocumentWriter(writer, new PrintTicket(), printQueue);
					return;
				}
			}
			if (reportObject is PrintQueue)
			{
				PrintQueue printQueue2 = reportObject as PrintQueue;
				XpsDocumentWriter writer2 = PrintQueue.CreateXpsDocumentWriter(printQueue2);
				try
				{
					ProcessXpsDocumentWriter(writer2, printQueue2);
				}
				catch (PrintQueueException)
				{
					InvokeEndPaginationIfNeeded();
					writer2 = PrintQueue.CreateXpsDocumentWriter(printQueue2);
					ProcessXpsDocumentWriter(writer2, new PrintTicket(), printQueue2);
				}
			}
		}

		private void ProcessXpsDocumentWriter(XpsDocumentWriter writer, PrintTicket printTicket, PrintQueue printQueue)
		{
			ClearCachedOriginAndSize();
			if (printTicket == null)
			{
				if (printQueue != null)
				{
					printTicket = printQueue.UserPrintTicket;
					if (printTicket == null)
					{
						printTicket = printQueue.DefaultPrintTicket;
					}
				}
				if (printTicket == null)
				{
					printTicket = new PrintTicket();
					printTicket.PageMediaSize = ReportSettings.PageMediaSize;
					printTicket.PageOrientation = ((ReportSettings.PageOrientation == PageOrientation.Landscape) ? System.Printing.PageOrientation.Landscape : System.Printing.PageOrientation.Portrait);
				}
			}
			if (!string.IsNullOrEmpty(ReportSettings.PrintJobDescription) && printQueue != null && printQueue.CurrentJobSettings != null)
			{
				printQueue.CurrentJobSettings.Description = ReportSettings.PrintJobDescription;
			}
			base.ActivePrintTicket = printTicket;
			_paginator.Initialize(ReportSettings);
			writer.Write(_paginator, printTicket);
			base.ActivePrintTicket = null;
			if (_paginator.IsCancel)
			{
				CleanUpDocAndPackage();
			}
			else
			{
				OnWritingCompleted(this, new WritingCompletedEventArgs(cancelled: false, "State", null));
			}
		}

			PrintStatus status = ((e.Cancelled || e.Error != null) ? PrintStatus.Canceled : PrintStatus.Successful);
			PrintEndedEventArgs e2 = new PrintEndedEventArgs(status, _countPrintedPages);
			RaisePrintEndedEvent(e2);
			CloseReportProgressControl();
		}

		{

			RaisePrintEndedEvent(e2);
			InvokeEndPaginationIfNeeded();
			CloseReportProgressControl();
		}

		public void Print(bool showPrintDialog, bool showReportProgressControl, FrameworkElement reportProgressControlOwnedElement)
		{
			VerifyAtLeastOneSection();
			_explicitPageSize = Size.Empty;
			ClearCachedOriginAndSize();
			_reportProgressControlOwnedElement = reportProgressControlOwnedElement;
			_countPrintedPages = 0;
			RaisePrintStartEvent(new EventArgs());
			try
			{
				if (PrinterSettings.InstalledPrinters.Count == 0)
				{
					_reportProgressControlOwnedElement = null;
					RaisePrintEndedEvent(new PrintEndedEventArgs(PrintStatus.Canceled, 0, PrintCancelationReason.NoPrinterAvailable));
					return;
				}
			}
			catch (SecurityException)
			{
			}
			try
			{
				PermissionSet permissionSet = new PermissionSet(null);
				permissionSet.AddPermission(new PrintingPermission(PrintingPermissionLevel.DefaultPrinting));
				permissionSet.AddPermission(new RegistryPermission(PermissionState.Unrestricted));
				permissionSet.Demand();
				if (!InitializeReport(showPrintDialog))
				{
					return;
				}
				if (showReportProgressControl)
				{
					ShowReportProgressControl();
				}
				try
				{
					GenerateReport(ReportSettings.PrintQueue);
				}
				catch
				{
					if (showReportProgressControl)
					{
						CloseReportProgressControl();
					}
					throw;
				}
			}
			catch (SecurityException)
			{
				PrintViaPrintDialog(showReportProgressControl);
			}
			_reportProgressControlOwnedElement = null;
		}

		{
			get
			{
				if (_partContent != null)
				{
					Rect layoutSlot = LayoutInformation.GetLayoutSlot(_partContent);
					Thickness margin = _partContent.Margin;
					double width = CalculateConstrainedExtent(layoutSlot.Width, _partContent.ActualWidth, _partContent.Width, _partContent.MinWidth, _partContent.MaxWidth, margin.Left + margin.Right, _partContent.HorizontalAlignment == System.Windows.HorizontalAlignment.Stretch);
					double height = CalculateConstrainedExtent(layoutSlot.Height, _partContent.ActualHeight, _partContent.Height, _partContent.MinHeight, _partContent.MaxHeight, margin.Top + margin.Bottom, _partContent.VerticalAlignment == VerticalAlignment.Stretch);
					return new Size(width, height);
				}
				return new Size(0.0, 0.0);
			}
		}

		private static double CalculateConstrainedExtent(double layoutSlotExtent, double actualExtent, double explicitExtent, double minExtent, double maxExtent, double marginExtent, bool isStretched)
		{
			if (isStretched && actualExtent > 0.0)
			{
				return actualExtent;
			}
			double num = (double.IsNaN(explicitExtent) ? layoutSlotExtent : explicitExtent);
			if (minExtent > 0.0)
			{
				num = Math.Max(num, minExtent);
			}
			if (!double.IsPositiveInfinity(maxExtent))
			{
				num = Math.Min(maxExtent, num);
			}
			return Math.Max(num - marginExtent, 0.0);
		}

		private static object CoerceMargin(DependencyObject target, object value)

			thickness.Top += reportPagePresenter._pageOriginInternal.Y;
			return thickness;
		}

		public void GeneratePreview(Report report, bool showPrintDialog, bool showReportProgressControl)
		{
			if (_documentViewer != null)
			{
				_documentViewer.ClearValue(DocumentViewerBase.DocumentProperty);
			}
			SetValue(ReportPropertyKey, report);
			try
			{
				PermissionSet permissionSet = new PermissionSet(null);
				permissionSet.AddPermission(new PrintingPermission(PrintingPermissionLevel.DefaultPrinting));
				permissionSet.AddPermission(new RegistryPermission(PermissionState.Unrestricted));
				permissionSet.Demand();
				if (!report.InitializeReport(showPrintDialog))
				{
					return;
				}
			}
			catch (SecurityException)
			{
				System.Windows.Controls.PrintDialog printDialog = new System.Windows.Controls.PrintDialog();
				printDialog.UserPageRangeEnabled = report.ReportSettings.UserPageRangeEnabled;
				if (showPrintDialog)
				{
					try
					{
						if (printDialog.PrintTicket != null)
						{
							if (report.ReportSettings.PageMediaSize != null)
							{
								printDialog.PrintTicket.PageMediaSize = report.ReportSettings.PageMediaSize;
							}
							printDialog.PrintTicket.PageOrientation = ((report.ReportSettings.PageOrientation == PageOrientation.Landscape) ? System.Printing.PageOrientation.Landscape : System.Printing.PageOrientation.Portrait);
						}
					}
					catch (PrintQueueException)
					{
					}
					if (printDialog.ShowDialog() != true)
					{
						return;
					}
					report.ReportSettings.PageRange = printDialog.PageRange;
					try
					{
						if (printDialog.PrintTicket != null && printDialog.PrintTicket.PageMediaSize != null)
						{
							report.ReportSettings.PageMediaSize = printDialog.PrintTicket.PageMediaSize;
						}
						else
						{
							report.ReportSettings.PageSize = new Size(printDialog.PrintableAreaWidth, printDialog.PrintableAreaHeight);
						}
						if (printDialog.PrintTicket != null && printDialog.PrintTicket.PageOrientation.HasValue)
						{
							System.Printing.PageOrientation value = printDialog.PrintTicket.PageOrientation.Value;
							if (value == System.Printing.PageOrientation.Landscape || value == System.Printing.PageOrientation.ReverseLandscape)
							{
								report.ReportSettings.PageOrientation = PageOrientation.Landscape;
							}
							else
							{
								report.ReportSettings.PageOrientation = PageOrientation.Portrait;
							}
						}
					}
					catch (PrintQueueException)
					{
					}
				}
				else
				{
					_ = report.ReportSettings.PageRange;
					if (report.ReportSettings.PageRange.PageFrom == 0)
					{
						report.ReportSettings.PageRange = new PageRange(1, 0);
					}
				}
			}
			ReleaseDocumentResources();
			_stream = new MemoryStream();
			report.IsGeneratingPreview = true;
			if (showReportProgressControl)
			{
				report.ShowReportProgressControl(this);
			}
			report.Export(OutputFormat.XPS, _stream);
			_package = Package.Open(_stream, FileMode.Open, FileAccess.Read);
			string text = "pack://xamreportpreview.document";
			Uri uri = new Uri(text + ".xps");
			int num = 1;
			while (PackageStore.GetPackage(uri) != null)
			{
				num++;
				uri = new Uri(text + num + ".xps");
			}
			_packageRegisteredUri = uri;
			PackageStore.AddPackage(_packageRegisteredUri, _package);
			_xpsDocument = new XpsDocument(_package, CompressionOption.Maximum, _packageRegisteredUri.AbsoluteUri);
			_documentForPreview = _xpsDocument.GetFixedDocumentSequence();
			Refresh();
			report.IsGeneratingPreview = false;
			report.ReportSettings.PrintQueue = null;
		}

		private static void OnPreviewDocumentCommand(object sender, ExecutedRoutedEventArgs e)
		{
			if (e.Command == ApplicationCommands.Print && sender is XamReportPreview { Report: var report } xamReportPreview)
			{
				if (report != null)
				{
					report.Print(showPrintDialog: true, showReportProgressControl: true, xamReportPreview);
					report.ReportSettings.PrintQueue = null;
				}
				e.Handled = true;
			}
		}

		private void OnUnloaded(object sender, RoutedEventArgs e)
		{
			base.Dispatcher.BeginInvoke(DispatcherPriority.Background, new MethodDelegate(VerifyUnloaded));
		}

		private void ReleaseDocumentResources()
		{
			if (_packageRegisteredUri != null)
			{
				if (PackageStore.GetPackage(_packageRegisteredUri) == _package)
				{
					PackageStore.RemovePackage(_packageRegisteredUri);
				}
				_packageRegisteredUri = null;
			}
			if (_xpsDocument != null)
			{
				_xpsDocument.Close();
				_xpsDocument = null;
			}
			if (_package != null)
			{
				_package.Close();
				_package = null;
			}
			if (_stream != null)
			{
				_stream.Close();
				_stream = null;
			}
			if (_documentForPreview != null)
			{
				_documentForPreview = null;
			}
		}

		private void VerifyUnloaded()
		{
			if (!base.IsLoaded && !Utilities.HasLoadedAncestor(this))
			{
				if (_documentViewer != null)
				{
					_documentViewer.ClearValue(DocumentViewerBase.DocumentProperty);
				}
				ReleaseDocumentResources();
			}
		}
