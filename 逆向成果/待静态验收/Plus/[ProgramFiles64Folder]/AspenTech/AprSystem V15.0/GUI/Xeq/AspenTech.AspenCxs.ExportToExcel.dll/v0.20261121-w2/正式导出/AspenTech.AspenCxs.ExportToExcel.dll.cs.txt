		private void SetMsgBoxTitle()
		{
			if (Case == null)
			{
				return;
			}
			string applicationName = Case.ApplicationName;
			if (!(applicationName == "Aspen Plus"))
			{
				if (applicationName == "Hysys")
				{
					_msgBoxTitle = "Aspen Hysys";
				}
			}
			else
			{
				_msgBoxTitle = "Aspen Plus";
			}
		}

		private void OnPropertyChanged(string property)
		{
			switch (property)
			{
			default:
				_ = property == "Case";
				break;
			case "RunningCase":
				if (_Case != null)
				{
					DisconnectFromRunningCase();
				}
				if (_runningCase != null && (!string.IsNullOrEmpty(_caseFilePath) || !string.IsNullOrEmpty(_applicationName)))
				{
					ConnectToRunningCase();
				}
				break;
			case "CasePath":
				if (_Case == null)
				{
					if (_runningCase != null && !string.IsNullOrEmpty(_caseFilePath))
					{
						ConnectToRunningCase();
					}
				}
				else
				{
					_Case.CaseFilePath = _caseFilePath;
				}
				break;
			case "ApplicationName":
				if (_Case == null)
				{
					if (_runningCase != null && !string.IsNullOrEmpty(_applicationName))
					{
						ConnectToRunningCase();
					}
				}
				else
				{
					_Case.ApplicationRequirement = _applicationName;
				}
				break;
			}
		}

		private void DisconnectFromRunningCase()
		{
			CxsCase cxsCase = _Case;
			_Case = null;
			cxsCase?.Dispose();
		}

		private void ConnectToRunningCase()
		{
			if (_runningCase == null || (string.IsNullOrEmpty(_caseFilePath) && string.IsNullOrEmpty(_applicationName)))
			{
				return;
			}
			if (_Case != null)
			{
				if (_DisposeCase)
				{
					_Case.Dispose();
				}
				_Case = null;
			}
			try
			{
				_Case = new CxsCase();
				_DisposeCase = true;
				_Case.CaseFilePath = _caseFilePath;
				_Case.ApplicationRequirement = _applicationName;
				_Case.AttachToRunningCase(_runningCase);
			}
			catch (Exception)
			{
			}
			if (_Case == null)
			{
				return;
			}
			if (_Case.IsConnected)
			{
				try
				{
					simprocessID = _Case.SimulatorProcessId;
					Process processById = Process.GetProcessById(simprocessID);
					if (processById != null)
					{
						simprocessName = processById.ProcessName;
					}
					return;
				}
				catch
				{
					return;
				}
			}
			_Case.Dispose();
			_Case = null;
		}

		private List<ExportTable> CreateTableList()
		{
			if (_Case != null)
			{
				GetCase();
			}
			List<ExportTable> list = new List<ExportTable>();
			List<string> list2 = new List<string>();
			int num = 1;
			foreach (Table table2 in Tables)
			{
				if (table2 != null)
				{
					string empty = string.Empty;
					if (string.IsNullOrWhiteSpace(table2.Name))
					{
						empty = Strings.XlExport_DefaultTableName + " " + num;
						num++;
					}
					else
					{
						empty = Global.GenerateExcelWorksheetName(table2.Name, list2, _msgBoxTitle);
					}
					list2.Add(empty);
					object[,] Values = new object[table2.Columns.Count, table2.Variables.Count + 1];
					string[,] Paths = new string[table2.Columns.Count, table2.Variables.Count];
					FillValues(table2, ref Values, ref Paths, _Case);

		private void PreProcessTemplate(ref Table table)
		{
			foreach (Column column in table.Columns)
			{
				if (string.IsNullOrWhiteSpace(column.Attribute))
				{
					continue;
				}
				foreach (KeyValuePair<string, string> item in AttributeConverterDictionary)
				{
					if (column.Attribute.ToString().ToUpper() == item.Key.ToString().ToUpper())
					{
						if (item.Value != string.Empty)
						{
							column.Attribute = item.Value;
						}
						break;
					}
				}
				if (string.IsNullOrWhiteSpace(column.Header))
				{
					column.Header = string.Empty;
				}
				column.Header = column.Header.Replace("_", " ");
				column.Attribute = column.Attribute.Replace(" ", "");
			}
		}

		private void PostProcessTemplate(ref Table table)
		{
			foreach (Column column in table.Columns)
			{
				if (string.IsNullOrWhiteSpace(column.Attribute))
				{
					column.Attribute = string.Empty;
				}
				foreach (KeyValuePair<string, string> item in AttributeConverterDictionary)
				{
					if (column.Attribute == string.Empty)
					{
						break;
					}
					if (column.Attribute.ToString().ToUpper() == item.Value.ToString().ToUpper())
					{
						if (item.Key != string.Empty)
						{
							column.Attribute = item.Key;
						}
						break;
					}
				}
				column.Header = column.Header.Replace("_", " ");
				column.Attribute = column.Attribute.Replace(" ", "");
			}
		}

		private void OnPropertyChanged(string property)
		{
			switch (property)
			{
			default:
				_ = property == "Case";
				break;
			case "RunningCase":
				if (_Case != null)
				{
					DisconnectFromRunningCase();
				}
				if (_runningCase != null && (!string.IsNullOrEmpty(_caseFilePath) || !string.IsNullOrEmpty(_applicationName)))
				{
					ConnectToRunningCase();
				}
				break;
			case "CasePath":
				if (_Case == null)
				{
					if (_runningCase != null && !string.IsNullOrEmpty(_caseFilePath))
					{
						ConnectToRunningCase();
					}
				}
				else
				{
					_Case.CaseFilePath = _caseFilePath;
				}
				break;
			case "ApplicationName":
				if (_Case == null)
				{
					if (_runningCase != null && !string.IsNullOrEmpty(_applicationName))
					{
						ConnectToRunningCase();
					}
				}
				else
				{
					_Case.ApplicationRequirement = _applicationName;
				}
				break;
			}
		}

			{
				return;
			}
			if (_Case != null)

				if (_DisposeCase)
				{
					_Case.Dispose();
				}
				_Case = null;
			}
			try
			{
				_Case = new CxsCase();
				_DisposeCase = true;
				_Case.CaseFilePath = _caseFilePath;
				_Case.ApplicationRequirement = _applicationName;
				_Case.AttachToRunningCase(_runningCase);

			catch (Exception)
			{
			}
			if (_Case == null)
			{
				return;

			if (_Case.IsConnected)

				else
				{
					array = new object[table.FieldLabels.Length, table.Records.Count + 1];
					array2 = new string[table.FieldLabels.Length, table.Records.Count];
				}
				FillValues(table, ref array, ref array2, _Case);
				ExportTable exportTable = null;
				exportTable = ((table.Orientation != Orientation.Horizontal) ? new ExportTable(empty, empty, 0, 1, table.FieldLabels.Length, table.Records.Count, array, array2) : new ExportTable(empty, empty, 1, 0, table.Records.Count, table.FieldLabels.Length, array, array2));
				if (exportTable != null)
				{
					exportTable.Case = _Case;
					exportTable.Is1D = false;
					exportTable.Records2D = table.Records;
					exportTable.Orientation = table.Orientation;
				}
				list.Add(exportTable);
			}
			return list;
		}

		private string[] MonikerToPath(CxsCase cxsCase, string[] paths)
		{
			string[] result = null;
			if (cxsCase != null && paths != null && paths.Length != 0)
			{
				try
				{
					result = cxsCase.ConvertMonikersToPaths(paths);
				}
				catch
				{
				}
			}
			return result;
		}

		private string MonikerToPath(CxsCase cxsCase, string Path)
		{
			if (cxsCase != null && !string.IsNullOrWhiteSpace(Path))
			{
				string[] itemMonikers = new string[1] { Path };
				try
				{
					string[] array = cxsCase.ConvertMonikersToPaths(itemMonikers);
					if (array[0] != null)
					{
						return array[0];
					}
				}
				catch
				{
				}
			}
			return Path;
		}

		private void GetCase()
		{
			if (_Case == null)
			{
				return;
			}
			if (string.IsNullOrWhiteSpace(CasePath))
			{
				CasePath = _Case.CaseFilePath;
			}
			if (string.IsNullOrWhiteSpace(ApplicationName))
			{
				ApplicationName = _Case.ApplicationName;
			}
			if (_Case.IsConnected)
			{
				try
				{
					simprocessID = _Case.SimulatorProcessId;
					Process processById = Process.GetProcessById(simprocessID);
					if (processById != null)
					{
						simprocessName = processById.ProcessName;
					}
					return;
				}
				catch
				{
					return;
				}
			}
			try
			{
				_Case.AttachToRunningCase();
				simprocessID = _Case.SimulatorProcessId;
				Process processById2 = Process.GetProcessById(simprocessID);
				if (processById2 != null)
				{
					simprocessName = processById2.ProcessName;
				}
			}
			catch
			{
				_Case.Dispose();
				_Case = null;
			}
		}

		private void FillValues(Table2D table, ref object[,] Values, ref string[,] Paths, CxsCase cxsCase = null)
		{
			int num = 1;
			int num2 = 0;
			string[] fieldLabels = table.FieldLabels;
			foreach (string text in fieldLabels)
			{
				if (table.Orientation == Orientation.Horizontal)
				{
					Values[num - 1, num2] = text;
				}
				else
				{
					Values[num2, num - 1] = text;
				}
				num2++;
			}
			num = 1;
			foreach (Record record in table.Records)
			{
				num2 = 0;
				object[] values = record.Values;
				foreach (object obj in values)
				{
					if (table.Orientation == Orientation.Horizontal)
					{
						Values[num, num2] = obj;
					}
					else
					{
						Values[num2, num] = obj;
					}
					num2++;
				}
				num2 = 0;
				fieldLabels = MonikerToPath(cxsCase, record.Paths);
				foreach (string text2 in fieldLabels)
				{
					if (table.Orientation == Orientation.Horizontal)
					{
						Paths[num - 1, num2] = text2;
					}
					else
					{
						Paths[num2, num - 1] = text2;
					}
					num2++;
				}
				num++;
			}
		}

		private string AddSpacesToCamelCaseName(string origName)
		{
			StringBuilder stringBuilder = new StringBuilder();
			bool flag = false;
			char[] array = origName.ToCharArray();
			foreach (char c in array)
			{
				if (stringBuilder.Length > 0 && (char.IsUpper(c) || char.IsDigit(c)))
				{
					if (char.IsDigit(c))
					{
						if (!flag)
						{
							stringBuilder.Append(" ");
						}
						flag = true;
					}
					else
					{
						flag = false;
						stringBuilder.Append(" ");
					}
				}
				stringBuilder.Append(c);
			}
			return stringBuilder.ToString();
		}

			}

				return false;
			}
			num = styleName.Substring(styleName.Length - i);
			root = styleName.Substring(0, styleName.Length - i).Trim();
			return true;
		}

		private void AddToListOfCheckItems(ExportTable table, CheckableAllNoneController ctlr = null)
		{
			CheckableAllNoneItem checkableAllNoneItem = new CheckableAllNoneItem();
			checkableAllNoneItem.Caption = table.DisplayName;
			checkableAllNoneItem.UnderlyingObject = table;
			checkableAllNoneItem.UnderlyingObjectPropName = "ExportThisTable";
			if (!table.Is1D && table.Records2D != null)
			{
				foreach (Export2DTemplate.Record item in table.Records2D)
				{
					CheckableAllNoneItem checkableAllNoneItem2 = new CheckableAllNoneItem();
					checkableAllNoneItem2.Caption = DeGuidise(item.Name);
					checkableAllNoneItem2.UnderlyingObject = item;
					checkableAllNoneItem2.UnderlyingObjectPropName = "ExportThisRecord";
					checkableAllNoneItem.AddChild(checkableAllNoneItem2);
				}
			}
			if (ctlr == null)
			{
				foreach (ICheckableAllNone colChkTable in ColChkTables)
				{
					if (colChkTable is CheckableAllNoneController)
					{
						ctlr = colChkTable as CheckableAllNoneController;

					}
				}
			}

			ColChkTables.Add(checkableAllNoneItem);
		}

		private string DeGuidise(string name)
		{
			string text = name;
			int length = name.Length;
			if (length >= 36)
			{
				for (int i = 0; i <= length - 36; i++)
				{
					if (Guid.TryParse(name.Substring(i, 36), out var _))
					{
						text = name.Substring(0, i).TrimEnd(' ');
						if (length > i + 36)
						{
							text += name.Substring(i + 36, length - (i + 36));
						}
						break;
					}
				}
			}
			return text;
		}

		private void ExportTablesToExcelWbNow()
		{
			if (_isExporting)
			{
				EXMessageBox.Show(Global.GetString("XlExport_ExportExecuting"), Global.GetString("XlExport_ExportExecutingTitle"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Warning);
			}
			else
			{
				if (!MeetsRequirements)
				{
					return;
				}
				BackgroundWorker backgroundWorker = new BackgroundWorker();
				backgroundWorker.DoWork += delegate(object obj, DoWorkEventArgs args)
				{
					ExportResultReport result = ExportAsycReportProgress(obj as BackgroundWorker);
					args.Result = result;
				};
				backgroundWorker.WorkerReportsProgress = true;
				backgroundWorker.WorkerSupportsCancellation = true;
				backgroundWorker.RunWorkerCompleted += delegate(object obj, RunWorkerCompletedEventArgs args)
				{
					_totalTimeWatch.Stop();
					_currentTimeWatch.Stop();
					_timer.Stop();
					UpdateTimes();
					_isExporting = false;
					if (args.Error != null)
					{
						EXMessageBox.Show(Global.GetString("ExcelExportVm_ErrorOccurred") + " " + args.Error.Message, Global.GetString("ExcelExportVm_ExportFailed"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Error);
					}
					else if (!(args.Result is ExportResultReport exportResultReport))
					{
						EXMessageBox.Show(Global.GetString("ExcelExportVm_UnspecifiedError"), Global.GetString("ExcelExportVm_ExportFailed"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Error);
					}
					else
					{
						switch (exportResultReport.Result)
						{
						case ExportResult.XLSuccess:
							if (!string.IsNullOrWhiteSpace(exportResultReport.FilePath))
							{
								ShowOpenDialog(exportResultReport.FilePath);
							}
							break;
						case ExportResult.XLAppFailure:
							EXMessageBox.Show(Global.GetString("ExcelExportVm_CouldNotCreateExcelInstance"), Global.GetString("ExcelExportVm_ExportFailed"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Error);
							break;
						case ExportResult.NoFileSpecified:
							EXMessageBox.Show(Global.GetString("ExcelExportVm_NoFileSpecified"), Global.GetString("ExcelExportVm_ExportFailed"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Error);
							break;
						case ExportResult.ActivateWorkbookFailed:
							EXMessageBox.Show(Global.GetString("ExcelExportVm_ErrorOpeningWkBk"), Global.GetString("ExcelExportVm_ExportFailed"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Error);
							break;
						case ExportResult.CreateNewSheetFailed:
							EXMessageBox.Show(Global.GetString("ExcelExportVm_ErrorCreatingNewSheetInWb"), Global.GetString("ExcelExportVm_ExportFailed"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Error);
							break;
						case ExportResult.XLSaveFailed:
							EXMessageBox.Show(Global.GetString("ExcelExportVm_ErrorSavingWb"), Global.GetString("ExcelExportVm_ExportFailed"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Error);
							break;
						case ExportResult.FileAlreadyExists:
							EXMessageBox.Show(Global.GetString("ExcelExportVm_FilenameSelectedExists"), Global.GetString("ExcelExportVm_ExportFailed"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Error);
							break;
						case ExportResult.UnSpecifiedError:
							EXMessageBox.Show(Global.GetString("ExcelExportVm_UnspecifiedError"), Global.GetString("ExcelExportVm_ExportFailed"), null, EXMessageBoxButton.OK, EXMessageBoxImage.Error);
							break;
						}
					}
				};
				backgroundWorker.ProgressChanged += delegate(object obj, ProgressChangedEventArgs args)
				{
					ProgressReport progressReport = (ProgressReport)args.UserState;
					if (progressReport.Type == ProgressType.Current)
					{
						CurrentTabProgress = args.ProgressPercentage;
					}
					else if (progressReport.Type == ProgressType.Total)
					{
						TotalProgress = args.ProgressPercentage;
						if (progressReport.CurrentTab != CurrentTab)
						{
							_currentTimeWatch.Reset();
							if (!string.IsNullOrWhiteSpace(progressReport.CurrentTab))
							{
								_currentTimeWatch.Start();
							}
							else
							{
								_currentTimeWatch.Stop();
							}
						}
						CurrentTab = progressReport.CurrentTab;
					}
					else if (progressReport.Type == ProgressType.Start)
					{
						_totalTimeWatch.Reset();
						_currentTimeWatch.Reset();
						_totalTimeWatch.Start();
						_currentTimeWatch.Start();
						_timer.Interval = new TimeSpan(0, 0, 0, 0, 25);
						_timer.Tick += delegate
						{
							UpdateTimes();
						};
						_timer.Start();
						CurrentTab = progressReport.CurrentTab;
						TotalProgress = 0.0;
						CurrentTabProgress = 0.0;
					}
				};
				backgroundWorker.RunWorkerAsync();
				_isExporting = true;
			}
		}

			CurrentTime = $"{_currentTimeWatch.Elapsed.Minutes:00}:{_currentTimeWatch.Elapsed.Seconds:00}.{_currentTimeWatch.Elapsed.Milliseconds / 10:00}";

			else
			{
				string[,] cxsPaths = CxsPaths;
				foreach (string text in cxsPaths)
				{

						continue;
					}
					bool flag = false;
					if (!text.StartsWith("#"))
					{

						if (num != -1)
						{
							path = text.Remove(num);
						}
						CxsNode cxsNode = Case.ResolveNode(path);
						if (cxsNode != null && !cxsNode.ReadOnly)
						{
							flag = true;
						}
					}
					if (flag && !IsResultVariable(text))
					{
						if (!dictionary.ContainsKey(text))
						{
							dictionary.Add(text, value: true);
						}
					}
					else if (!dictionary.ContainsKey(text))
					{
						dictionary.Add(text, value: false);
					}
				}
