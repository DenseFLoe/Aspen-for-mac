	internal class MenuNavigation : INavigation
	{
		public Uri NavigationUri { get; set; }

		public string NavigationParameter { get; set; }

		public FrameworkElement NavigationElement { get; set; }

		public void NavigateTo()
		{
			if (NavigationUri == null)
			{
				throw new InvalidOperationException(Resources.GetString("Navigation_UriIsNotSet"));
			}
			Uri uri = null;
			uri = ((!string.IsNullOrEmpty(NavigationParameter)) ? BuildNavigationString(ParseParameters(NavigationParameter), NavigationUri.OriginalString) : NavigationUri);
			if (NavigationElement is Frame)
			{
				(NavigationElement as Frame).Navigate(uri);
				return;
			}
			if (NavigationElement is Page)
			{
				((NavigationElement as Page).NavigationService ?? throw new InvalidOperationException(Resources.GetString("Navigation_PageIsNotSet"))).Navigate(uri);
				return;
			}
			throw new InvalidOperationException(Resources.GetString("Navigation_ElementIsNotSet"));
		}

		private static Uri BuildNavigationString(List<string> parameters, string sourceUri)
		{
			int num = 0;
			int num2 = 0;
			int num3 = 0;
			int num4 = 0;
			string text = "";
			while (true)
			{
				num2 = sourceUri.IndexOf("{", num3, StringComparison.Ordinal);
				num = sourceUri.IndexOf("}", num3, StringComparison.Ordinal);
				if (num == -1 && num2 == -1)
				{
					break;
				}
				if (num == -1 || num2 == -1)
				{
					throw new InvalidOperationException(Resources.GetString("Navigation_WrongFormat"));
				}
				text += sourceUri.Substring(num3, num2 - num3);
				text += parameters[num4];
				num4++;
				num3 = num + 1;
			}
			if (num3 >= 0 && num3 < sourceUri.Length)
			{
				text += sourceUri.Substring(num3);
			}
			return new Uri(text, UriKind.RelativeOrAbsolute);
		}

		private static List<string> ParseParameters(string parameterString)
		{
			int num = 0;
			int num2 = 0;
			List<string> list = new List<string>();
			if (parameterString[0] != '/')
			{
				list.Add(parameterString);
				return list;
			}
			num = parameterString.IndexOf('/');
			while (num != -1)
			{
				num2 = num + 1;
				if (num2 >= parameterString.Length)
				{
					list.Add("");
					break;
				}
				num = parameterString.IndexOf('/', num2);
				if (num == -1)
				{
					list.Add(parameterString.Substring(num2, parameterString.Length - num2));
					break;
				}
				list.Add(parameterString.Substring(num2, num - num2));
			}
			return list;
		}

		private void PrepareOpenMode(UIElement element, OpenMode oldOpenMode, OpenMode newOpenMode)
		{
			if (element != null)
			{
				switch (oldOpenMode)
				{
				case OpenMode.RightClick:
					RemoveRightMouseButtonUpHandler(element);
					break;
				case OpenMode.LeftClick:
					RemoveLeftMouseButtonUpHandler(element);
					break;
				}
				switch (newOpenMode)
				{
				case OpenMode.RightClick:
					AddRightMouseButtonUpHandler(element);
					break;
				case OpenMode.LeftClick:
					AddLeftMouseButtonUpHandler(element);
					break;
				}
			}
		}

		private void PrepareContextMenu(UIElement element, XamContextMenu oldContextMenu, XamContextMenu newContextMenu)
		{
			if (element != null)
			{

				{
					PrepareOpenMode(oldContextMenu.ParentElement, OpenMode, OpenMode.None);
				}
				if (newContextMenu != null)
				{
					PrepareOpenMode(element, OpenMode.None, OpenMode);
				}
				if (newContextMenu != null)
				{
					newContextMenu.ParentElement = element;
				}
			}
			CommandSourceManager.NotifyCanExecuteChanged(typeof(OpenCommand));
			CommandSourceManager.NotifyCanExecuteChanged(typeof(CloseCommand));
		}

		private void RemoveRightMouseButtonUpHandler(UIElement element)

			element.MouseRightButtonUp -= Element_MouseRightButtonUp;
		}

		private void RemoveLeftMouseButtonUpHandler(UIElement element)
		{
			element.RemoveHandler(UIElement.MouseLeftButtonUpEvent, new MouseButtonEventHandler(Element_MouseLeftButtonUp));
		}

		private bool CheckModifierKeys()
		{

			{
				return (currentModifierKeys & ModifierKeys) == ModifierKeys;
			}
			return true;
		}

		private void Element_MouseLeftButtonUp(object sender, MouseButtonEventArgs e)

				{
					e.Handled = OpenContextMenu(sender, e.GetPosition(PlatformProxy.GetRootVisual(dependencyObject)));
				}
			}
		}

		{

			T val = ParentElement as T;
			IEnumerable<UIElement> elementsFromPoint = PlatformProxy.GetElementsFromPoint(PlatformProxy.GetRootVisual(ParentElement).TranslatePoint(MouseClickLocation, ParentElement), ParentElement);
			if (elementsFromPoint == null)
			{
				return null;
			}
			if (elementsFromPoint.GetEnumerator().MoveNext())
			{
				foreach (UIElement item2 in elementsFromPoint)
				{
					if (item2 is T item)
					{
						list.Insert(0, item);
					}
				}

		internal void OpenInternal()
		{
			if (IsClickPlacement(Placement))
			{
				bool num = IsClickPlacement(Placement);
				_preventPlacementChangeEvent = true;
				PlacementMode placement = Placement;
				if (num)
				{
					Placement = PlacementMode.Center;
				}
				IsOpen = true;
				if (num)
				{
					Placement = placement;
				}
				_preventPlacementChangeEvent = false;
			}
			else
			{
				IsOpen = true;
			}
		}

		private bool ResolveIsOpenChange(DependencyPropertyChangedEventArgs e)
		{
			if (_skipIsOpenChange)
			{
				_skipIsOpenChange = false;
				return false;
			}
			OpeningEventArgs e2 = new OpeningEventArgs(this, MouseClickLocation);
			if (ParentElement != null)
			{
				if ((bool)e.NewValue)
				{
					OnOpening(e2);
				}
				else
				{
					OnClosing(e2);
					if (_popupInitiatedClose)
					{
						e2.Cancel = false;
					}
				}
			}
			else
			{
				e2.Cancel = IsOpen;
			}
			IsCancelled = e2.Cancel;
			if (e2.Cancel)
			{
				_skipIsOpenChange = true;
				SetValue(IsOpenProperty, e.OldValue);
				return false;
			}
			if (!object.Equals(GetValue(IsOpenProperty), e.NewValue))
			{
				_skipIsOpenChange = true;
				SetValue(IsOpenProperty, e.NewValue);
			}
			return true;
		}

		private void OnContextMenuOpened()
		{
			_previouslyFocusedControl = PlatformProxy.GetFocusedElement(this) as Control;
			base.IsEnabled = true;
			if (!_contextMenuPopup.IsOpen)
			{
				AddContextMenuToPopup();
				Measure(new Size(double.MaxValue, double.MaxValue));
				SetPopupPosition();
				_contextMenuPopup.IsOpen = true;
				ContextMenuService.OnContextMenuOpened(this);
			}
			foreach (object item in (IEnumerable)base.Items)
			{
				if (base.ItemContainerGenerator.ContainerFromItem(item) is XamMenuItem xamMenuItem)
				{
					xamMenuItem.IsSelected = false;
					xamMenuItem.IsMouseOver = false;
					xamMenuItem.IsHighlighted = false;
					xamMenuItem.ChangeVisualState(useTransitions: false);
				}
			}
		}

		private void OnContextMenuClosed()
		{
			bool flag = false;
			flag = _popupInitiatedClose;
			if (_contextMenuPopup.IsOpen || flag)
			{
				if (IsChild(PlatformProxy.GetFocusedElement(this) as FrameworkElement))
				{
					if (_previouslyFocusedControl == null)
					{
						_previouslyFocusedControl = _focusedControl;
						_focusedControl = null;
					}
					ReturnFocus();
				}
				_previouslyFocusedControl = null;
				foreach (object item in (IEnumerable)base.Items)
				{
					if (base.ItemContainerGenerator.ContainerFromItem(item) is XamMenuItem { IsSubmenuOpen: not false } xamMenuItem)
					{
						xamMenuItem.CloseSubmenu(forceCloseImmediately: true);
					}
				}
				_contextMenuPopup.IsOpen = false;
				RemoveContextMenuFromPopup();
				_contextMenuPopup.Child = null;
				ContextMenuService.OnContextMenuClosed(this);
			}
			OnClosed(EventArgs.Empty);
		}

		private void SetPopupPosition()
		{
			double zoomFactor = PlatformProxy.GetZoomFactor();
			Size size = PlatformProxy.ResolveContainerSize();
			double width = size.Width;
			double height = size.Height;
			int num;
			if (zoomFactor > 0.0)
			{
				num = ((!PlatformProxy.IsVersionSupported("4.0")) ? 1 : 0);
				if (num != 0)
				{
					width /= zoomFactor;
					height /= zoomFactor;
					goto IL_0059;
				}
			}
			else
			{
				num = 0;
			}
			if (zoomFactor != 1.0)
			{
				width /= zoomFactor;
				height /= zoomFactor;
			}
			goto IL_0059;
			IL_0059:
			FrameworkElement frameworkElement = PlatformProxy.GetRootVisual(this) as FrameworkElement;
			if (frameworkElement == null)
			{
				frameworkElement = PlatformProxy.GetRootVisual(ParentElement) as FrameworkElement;
			}
			Rect targetRect = GetTargetRect();
			Point referencePoint = AlignToRect(targetRect, frameworkElement);
			FrameworkElement frameworkElement2 = ParentElement as FrameworkElement;
			if (frameworkElement2 != null && Placement == PlacementMode.MouseClick && FrameworkElement.GetFlowDirection(frameworkElement2) == FlowDirection.RightToLeft)
			{
				referencePoint.X = frameworkElement2.ActualWidth - referencePoint.X;
			}
			AdjustForScaling(frameworkElement2, Placement, ref referencePoint);
			ContextMenuLocation = referencePoint;
			_skipPopupEvents = _contextMenuPopup.IsOpen && !base.IsMouseCaptured;
			if (_skipPopupEvents)
			{
				_contextMenuPopup.IsOpen = false;
				_contextMenuPopup.IsOpen = true;
				_skipPopupEvents = false;
			}
			_mustSetPosition = false;
			if (num != 0)
			{
				ScaleTransform zoomTransform = _zoomTransform;
				double scaleX = (_zoomTransform.ScaleY = zoomFactor);
				zoomTransform.ScaleX = scaleX;
			}
		}

		private void SetPosition()

			}
		}

		private Point AlignToRect(Rect rect, FrameworkElement root)
		{
			double num = HorizontalOffset;
			double num2 = VerticalOffset;
			double actualWidth = GetActualWidth();
			double actualHeight = GetActualHeight();
			if (!rect.IsEmpty)
			{
				bool flag = root != null && root.FlowDirection == FlowDirection.RightToLeft;
				_ = base.FlowDirection;
				switch (Placement)
				{
				case PlacementMode.AlignedBelow:
					num = ((!flag) ? (num + rect.Left) : (num + (rect.Right - actualWidth)));
					num2 += rect.Bottom;
					break;
				case PlacementMode.Center:
					num += (rect.Left + rect.Right - actualWidth) / 2.0;
					num2 += (rect.Top + rect.Bottom - actualHeight) / 2.0;
					break;
				case PlacementMode.AlignedToTheRight:
					num = ((!flag) ? (num + rect.Right) : (num + (rect.Left - actualWidth)));
					num2 += rect.Top;
					break;
				case PlacementMode.AlignedToTheLeft:
					num = ((!flag) ? (num + (rect.Left - actualWidth)) : (num + rect.Right));
					num2 += rect.Top;
					break;
				case PlacementMode.AlignedAbove:
					num = ((!flag) ? (num + rect.Left) : (rect.Right - actualWidth));
					num2 += rect.Top - actualHeight;
					break;
				case PlacementMode.AlignedTop:
					num = ((!flag) ? (num + rect.Left) : (rect.Right - actualWidth));
					num2 += rect.Top;
					break;
				case PlacementMode.AlignedBottom:
					num = ((!flag) ? (num + rect.Left) : (rect.Right - actualWidth));
					num2 += rect.Bottom - actualHeight;
					break;
				case PlacementMode.AlignedLeft:
					num = ((!flag) ? (num + rect.Left) : (num + (rect.Right - actualWidth)));
					num2 += rect.Top;
					break;
				case PlacementMode.AlignedRight:
					num = ((!flag) ? (num + (rect.Right - actualWidth)) : (num + rect.Left));
					num2 += rect.Top;
					break;
				default:
					num += rect.Left;
					num2 += rect.Top;
					break;
				}
				if (SystemParameters.IsMenuDropRightAligned && Placement != PlacementMode.MouseClick)
				{
					num += actualWidth;
				}
			}
			return new Point(num, num2);
		}

		private Point AlignToScreen(Rect menu, Rect screen, Rect target, FrameworkElement root)
		{
			if (base.FlowDirection == FlowDirection.LeftToRight)
			{
				if (menu.Width >= screen.Width)
				{
					menu.X = screen.X;
				}
				else
				{
					if (menu.Left < screen.Left)
					{
						menu.X = screen.X;
					}
					if (menu.Right > screen.Right)
					{
						if (IsClickPlacement(Placement))
						{
							menu.X -= menu.Width - target.Width;
						}
						else
						{
							menu.X -= menu.Right - screen.Right;
						}
					}
					if (menu.X < screen.X)
					{
						menu.X = screen.X;
					}
					if (menu.Right > screen.Right)
					{
						menu.X -= menu.Right - screen.Right;
					}
				}
			}
			else if (menu.Width >= screen.Width)
			{
				menu.X = screen.Right;
			}
			else if (menu.Left < screen.Left)
			{
				menu.X = screen.Right;
			}
			else if (menu.Left < screen.Left)
			{
				if (IsClickPlacement(Placement))
				{
					menu.X -= menu.Width - target.Width;
				}
				else
				{
					menu.X -= screen.Left - menu.Left;
				}
			}
			else if (menu.X - menu.Width < screen.Left)
			{
				menu.X += menu.Width - menu.X;
			}
			if (menu.Height >= screen.Height)
			{
				menu.Y = screen.Y;
			}
			else
			{
				if (menu.Top < screen.Top)
				{
					menu.Y = screen.Y;
				}
				if (menu.Bottom > screen.Bottom)
				{
					if (IsClickPlacement(Placement))
					{
						menu.Y -= menu.Height - target.Height;
					}
					else
					{
						menu.Y -= menu.Bottom - screen.Bottom;
					}
				}
				if (menu.Top < screen.Top)
				{
					menu.Y = screen.Y;
				}
				if (menu.Bottom > screen.Bottom)
				{
					menu.Y -= menu.Bottom - screen.Bottom;
				}
			}
			return new Point(menu.Left, menu.Top);
		}

		private bool IsClickPlacement(PlacementMode placementMode)
		{
			return Placement == PlacementMode.MouseClick;
		}

		private Rect GetTargetRect()
		{
			Rect result = Rect.Empty;
			UIElement placementTargetResolvedInternal = PlacementTargetResolvedInternal;
			if (placementTargetResolvedInternal != null)
			{
				result = ContextMenuService.GetBounds(placementTargetResolvedInternal);
				if (!PlacementRectangle.IsEmpty)
				{
					result.X += PlacementRectangle.Left;
					result.Y += PlacementRectangle.Top;
					result.Width = PlacementRectangle.Width;
					result.Height = PlacementRectangle.Height;
				}
			}
			if (Placement == PlacementMode.MouseClick)
			{
				result.X = MouseClickLocation.X - result.X;
				result.Y = MouseClickLocation.Y - result.Y;
			}
			else
			{
				result.X = 0.0;
				result.Y = 0.0;
			}
			_contextMenuPopup.PlacementTarget = placementTargetResolvedInternal;
			_contextMenuPopup.Placement = System.Windows.Controls.Primitives.PlacementMode.Relative;
			_contextMenuPopup.PlacementRectangle = PlacementRectangle;
			RefreshPlacementTargetResolved();
			return result;
		}

			return this == ancestorFromType;
		}

		private static ScaleTransform GetScaleTransform(FrameworkElement pt)
		{
			ScaleTransform scaleTransform = new ScaleTransform();
			Transform renderTransform = pt.RenderTransform;
			if (renderTransform == null)
			{
				return scaleTransform;
			}
			if (renderTransform is TransformGroup transformGroup)
			{
				{
					foreach (ScaleTransform item in transformGroup.Children.OfType<ScaleTransform>())
					{
						scaleTransform.ScaleY *= item.ScaleY;
						scaleTransform.ScaleX *= item.ScaleX;
					}
					return scaleTransform;
				}
			}
			if (renderTransform is ScaleTransform scaleTransform2)
			{
				scaleTransform.ScaleX = scaleTransform2.ScaleX;
				scaleTransform.ScaleY = scaleTransform2.ScaleY;
				return scaleTransform;
			}
			return scaleTransform;
		}

		private static void AdjustForScaling(FrameworkElement parent, PlacementMode placement, ref Point referencePoint)
		{
			if (placement == PlacementMode.AlignedAbove || placement == PlacementMode.AlignedToTheRight)
			{
				ScaleTransform scaleTransform = GetScaleTransform(parent);
				if (placement == PlacementMode.AlignedAbove)
				{
					referencePoint.Y /= scaleTransform.ScaleY;
				}
				else
				{
					referencePoint.X /= scaleTransform.ScaleX;
				}
			}
		}

		private void Popup_Opened(object sender, EventArgs e)
		{
			if (_skipPopupEvents)
			{
				return;
			}
			if (GetActualHeight() < 1.0)
			{
				Measure(new Size(double.PositiveInfinity, double.PositiveInfinity));
				_mustSetPosition = true;
			}
			SetPopupPosition();
			OnOpened(new OpenedEventArgs(this, MouseClickLocation, ContextMenuLocation));
			if (base.Items.Count > 0 && base.ItemContainerGenerator.ContainerFromIndex(0) is XamMenuItem xamMenuItem)
			{
				if (xamMenuItem.IsEnabled)
				{
					xamMenuItem.Focus();
				}
				else
				{
					xamMenuItem.FindSiblingItem(isForward: true, recurse: false)?.Focus();
				}
			}
		}

			for (DependencyObject dependencyObject = e.OriginalSource as DependencyObject; dependencyObject != null; dependencyObject = GetParent(dependencyObject))
			{
				if (dependencyObject is XamMenuItem || dependencyObject == this)
				{
					return;
				}
			}
			CloseCurrentOpen();
		}

		private static void CurrentOpenChanged(DependencyObject obj, DependencyPropertyChangedEventArgs e)
		{
			XamMenuBase xamMenuBase = obj as XamMenuBase;
			XamMenuItem xamMenuItem = e.OldValue as XamMenuItem;
			XamMenuItem xamMenuItem2 = e.NewValue as XamMenuItem;
			if (xamMenuItem != null && xamMenuItem.IsSubmenuOpen)
			{
				xamMenuItem.ManipulateSubmenu(isOpen: false, isImmediately: false);
				xamMenuItem.IsHighlighted = false;
				xamMenuItem2?.ManipulateSubmenu(isOpen: true, isImmediately: false);
			}
			if (xamMenuItem == null && xamMenuItem2 != null)
			{
				Mouse.Capture(xamMenuBase, CaptureMode.SubTree);
			}
			else if (xamMenuItem != null && xamMenuItem2 == null)
			{
				Mouse.Capture(null);
			}
			xamMenuBase.OnCurrentOpenItemChanged(xamMenuItem, xamMenuItem2);
		}

		}

		{
			while (element != null)
			{
				DependencyObject parent = GetParent(element);
				if (parent == null)
				{
					break;
				}
				element = parent;
			}
			return element;
		}

		internal virtual void CloseCurrentOpen()
		{
			if (CurrentOpen != null)
			{
				CurrentOpen.ManipulateSubmenu(isOpen: false, isImmediately: true);
				CurrentOpen = null;
				if (CurrentSelected != null)
				{
					CurrentSelected.IsSelected = false;
					CurrentSelected.ChangeVisualState(useTransitions: false);
					CurrentSelected = null;
				}
			}
		}

		{
			_previouslyFocusedControl = null;
		}

		internal void ReturnFocus()
		{
			if (_previouslyFocusedControl != null && _previouslyFocusedControl.IsEnabled && _previouslyFocusedControl.Visibility == Visibility.Visible)
			{
				_previouslyFocusedControl.Focus();
				ClearPreviouslyFocusedControl();
			}
		}

		internal void RootVisual_MouseLeftButtonDown(object sender, MouseButtonEventArgs e)
		{
			object weakReferenceTargetSafe = CoreUtilities.GetWeakReferenceTargetSafe(_lastMouseArgs);
			if (e == weakReferenceTargetSafe)
			{
				return;
			}
			if (e.OriginalSource is DependencyObject dependencyObject)
			{
				if (IsAncestorOf(dependencyObject))
				{
					return;
				}
				UIElement rootVisual = PlatformProxy.GetRootVisual(dependencyObject);
				if (rootVisual != null && PlatformProxy.GetParent(rootVisual) is Popup descendant && IsAncestorOf(descendant))
				{
					return;
				}
			}
			CloseCurrentOpen();
			ClearPreviouslyFocusedControl();
		}

		internal static DependencyObject GetParent(DependencyObject element)
		{
			DependencyObject dependencyObject = null;
			if (element is Visual || element is Visual3D)
			{
				dependencyObject = VisualTreeHelper.GetParent(element);
			}
			if (dependencyObject == null)
			{
				if (element is FrameworkElement frameworkElement)
				{
					dependencyObject = frameworkElement.Parent;
				}
				if (element is FrameworkContentElement frameworkContentElement)
				{
					dependencyObject = frameworkContentElement.Parent;
				}
			}
			return dependencyObject;
		}

		public override void OnApplyTemplate()
		{
			AccessKeyManager.RemoveAccessKeyPressedHandler(this, HandleAccessKeyPressed);
			_scrollViewer = GetTemplateChild("ScrollViewer") as ScrollViewer;
			if (_dropDownPopup != null && _dropDownPopup.Child != null)
			{
				_dropDownPopup.Child.MouseEnter -= Items_MouseEnter;
				_dropDownPopup.Child.MouseLeave -= Items_MouseLeave;
				(_dropDownPopup.Child as FrameworkElement).SizeChanged -= Popup_SizeChanged;
			}
			base.OnApplyTemplate();
			if (_headerIconPresenter != null)
			{
				_headerIconPresenter.Content = null;
			}
			_headerIconPresenter = GetTemplateChild("HeaderIconPresenter") as ContentControl;
			if (_headerIconPresenter != null)
			{
				_headerIconPresenter.Content = Icon;
			}
			_subMenuClosedVisualState = GetTemplateChild("SubmenuClose") as VisualState;
			_subMenuClosedImmediatelyVisualState = GetTemplateChild("SubmenuCloseImmediately") as VisualState;
			_dropDownPopup = GetTemplateChild("Popup") as Popup;
			if (_dropDownPopup != null && _dropDownPopup.Child != null)
			{
				_dropDownPopup.PlacementTarget = this;
				_dropDownPopup.Placement = System.Windows.Controls.Primitives.PlacementMode.Bottom;
				SetDropdownPopupPlacement();
				_dropDownPopup.AllowsTransparency = true;
			}
			EnsureMenuItemRole();
			ChangeVisualState(useTransitions: false);
			VisualStateManager.GoToState(this, "SubmenuCloseImmediately", useTransitions: false);
			AccessKeyManager.AddAccessKeyPressedHandler(this, HandleAccessKeyPressed);
		}

		private void OnMouseLeftButtonDownImpl(MouseButtonEventArgs e)
		{
			XamMenuBase parentXamMenu = ParentXamMenu;
			if (parentXamMenu == null)
			{
				return;
			}
			parentXamMenu._lastMouseArgs = null;
			if (IsMouseOverSubling)
			{
				return;
			}
			if (_scrollViewer != null && (_scrollViewer.ComputedVerticalScrollBarVisibility == Visibility.Visible || _scrollViewer.ComputedHorizontalScrollBarVisibility == Visibility.Visible))
			{
				Rect rect = new Rect(e.GetPosition(this), new Size(0.0, 0.0));
				if (!new Rect(base.RenderSize).IntersectsWith(rect))
				{
					return;
				}
			}
			if (!e.Handled && base.IsEnabled)
			{
				if (Role == MenuItemRole.TopLevelItem)
				{
					if (!IsSelected)
					{
						IsSelected = true;
					}
				}
				else
				{
					FocusOrSelect();
				}
				parentXamMenu._lastMouseArgs = new WeakReference(e);
			}
			if (Role == MenuItemRole.TopLevelHeader)
			{
				if (IsSubmenuOpen && !((XamMenu)parentXamMenu).ExpandOnHover)
				{
					ManipulateSubmenu(isOpen: false, isImmediately: false);
					parentXamMenu.CloseCurrentOpen();
					parentXamMenu.ReturnFocus();
				}
				else
				{
					ManipulateSubmenu(isOpen: true, isImmediately: false);
				}
			}
			else if (Role == MenuItemRole.SubmenuHeader)
			{
				XamMenuItem sublingOpenItem = SublingOpenItem;
				if (sublingOpenItem != null)
				{
					sublingOpenItem.ManipulateSubmenu(isOpen: false, isImmediately: true);
					sublingOpenItem = null;
				}
				ManipulateSubmenu(isOpen: true, isImmediately: true);
			}
		}

		protected override void OnKeyDown(KeyEventArgs e)
		{
			if (e.Key == Key.System && Keyboard.Modifiers == ModifierKeys.Alt && _dropDownPopup != null)
			{
				_dropDownPopup.IsOpen = false;
			}
			base.OnKeyDown(e);
			if (ParentXamMenu == null)
			{
				return;
			}
			Orientation orientation = Orientation.Vertical;
			if (IsInContextMenu == false)
			{
				orientation = (ParentXamMenu as XamMenu).MenuOrientation;
			}
			if (PlatformProxy.GetFocusedElement(this) as DependencyObject != this || !base.IsEnabled || e.Handled)
			{
				return;
			}
			e.Handled = true;
			Key key = e.Key;
			if (ParentXamMenu.FlowDirection == FlowDirection.RightToLeft)
			{
				if (e.Key == Key.Left)
				{
					key = Key.Right;
				}
				else if (e.Key == Key.Right)
				{
					key = Key.Left;
				}
			}
			switch (key)
			{
			case Key.Left:
				if ((Role == MenuItemRole.TopLevelHeader && orientation == Orientation.Vertical) || (ParentXamMenuItem != null && ParentXamMenuItem.MenuOrientation == Orientation.Horizontal))
				{
					HandleUpDownKey(isDown: false);
				}
				else
				{
					HandleLeftKey();
				}
				break;
			case Key.Right:
				if ((Role == MenuItemRole.TopLevelHeader && orientation == Orientation.Vertical) || (ParentXamMenuItem != null && ParentXamMenuItem.MenuOrientation == Orientation.Horizontal))
				{
					HandleUpDownKey(isDown: true);
				}
				else
				{
					HandleRightKey();
				}
				break;
			case Key.Up:
				if ((Role == MenuItemRole.TopLevelHeader && orientation == Orientation.Vertical) || (ParentXamMenuItem != null && ParentXamMenuItem.MenuOrientation == Orientation.Horizontal))
				{
					HandleLeftKey();
				}
				else
				{
					HandleUpDownKey(isDown: false);
				}
				break;
			case Key.Down:
				if ((Role == MenuItemRole.TopLevelHeader && orientation == Orientation.Vertical) || (ParentXamMenuItem != null && ParentXamMenuItem.MenuOrientation == Orientation.Horizontal))
				{
					HandleRightKey();
				}
				else
				{
					HandleUpDownKey(isDown: true);
				}
				break;
			case Key.Escape:
			{
				XamMenuBase parentXamMenu = ParentXamMenu;
				if (ParentXamMenuItem == parentXamMenu.CurrentOpen)
				{
					parentXamMenu.CloseCurrentOpen();
					parentXamMenu.ReturnFocus();
				}
				else
				{
					HandleLeftKey();
				}
				break;
			}
			case Key.Space:
				if (!IsCheckable)
				{
					break;
				}
				IsChecked = !IsChecked;
				if (!StaysOpenOnClick && Role != MenuItemRole.SubmenuHeader)
				{
					XamMenuBase parentXamMenu2 = ParentXamMenu;
					RaiseClickEvent();
					if (parentXamMenu2 != null)
					{
						parentXamMenu2.CloseCurrentOpen();
						parentXamMenu2.ReturnFocus();
					}
				}
				break;
			case Key.Return:
				RaiseClickEvent();
				if (Role == MenuItemRole.TopLevelHeader || Role == MenuItemRole.SubmenuHeader)
				{
					if (IsSubmenuOpen)
					{
						ParentXamMenu.CloseCurrentOpen();
						ParentXamMenu.ReturnFocus();
						break;
					}
					ManipulateSubmenu(isOpen: true, isImmediately: true);
					XamMenuItem xamMenuItem = FindSiblingItem(isForward: true, recurse: true);
					if (xamMenuItem != null)
					{
						IsHighlighted = true;
						MoveToItem(xamMenuItem);
					}
				}
				else if (!StaysOpenOnClick && Role != MenuItemRole.SubmenuHeader && ParentXamMenu != null)
				{
					ParentXamMenu.CloseCurrentOpen();
					ParentXamMenu.ReturnFocus();
				}
				break;
			default:
				e.Handled = false;
				break;
			}
		}

		protected override void OnLostFocus(RoutedEventArgs e)
		{
			if (ParentXamMenu != null)

				object focusedElement = PlatformProxy.GetFocusedElement(ParentXamMenu);

					IsHighlighted = false;
				}
				if (!XamMenuBase.IsObjectFromMenu(ParentXamMenu, focusedElement))
				{
					ParentXamMenu._previouslyFocusedControl = null;
					IsSelected = false;
				}
			}
		}

		internal void OpenSubmenu()
		{
			if (_dropDownPopup == null)
			{
				return;
			}
			if (_dropDownPopup.Child != null)
			{
				_dropDownPopup.Child.MouseEnter -= Items_MouseEnter;
				_dropDownPopup.Child.MouseLeave -= Items_MouseLeave;
				(_dropDownPopup.Child as FrameworkElement).SizeChanged -= Popup_SizeChanged;
				_dropDownPopup.Child.MouseEnter += Items_MouseEnter;
				_dropDownPopup.Child.MouseLeave += Items_MouseLeave;
				(_dropDownPopup.Child as FrameworkElement).SizeChanged += Popup_SizeChanged;
			}
			if (Role == MenuItemRole.SubmenuHeader && ParentXamMenuItem != null && !ParentXamMenuItem.IsSubmenuOpen)
			{
				return;
			}
			if (_dropDownPopup.Child is FrameworkElement frameworkElement)
			{
				frameworkElement.MaxHeight = double.PositiveInfinity;
				frameworkElement.MinHeight = 0.0;
			}
			_dropDownPopup.IsOpen = true;
			if (Role == MenuItemRole.TopLevelHeader || _isImmediately)
			{
				VisualStateManager.GoToState(this, "SubmenuOpenImmediately", useTransitions: false);
			}
			else if (Role == MenuItemRole.SubmenuHeader)
			{
				VisualStateManager.GoToState(this, "SubmenuOpen", useTransitions: false);
			}
			if (Role == MenuItemRole.TopLevelHeader && ParentXamMenu != null)
			{
				ParentXamMenu.CurrentOpen = this;
			}
			if (!IsSubmenuOpen)
			{
				if (_isMenuOrientationNotSet && base.HeaderedItemContainerGenerator.ItemsHost is StackPanel stackPanel)
				{
					stackPanel.Orientation = MenuOrientation;
					_isMenuOrientationNotSet = false;
				}
				IgnorePropertyChange = true;
				IsSubmenuOpen = true;
				RaiseSubmenuOpenedEvent();
			}
			_dropDownPopup.UpdateLayout();
		}

		internal XamMenuItem FindSiblingItem(bool isForward, bool recurse)
		{
			if (recurse)
			{
				int index = 0;
				if (!isForward)
				{
					index = base.Items.Count - 1;
				}
				if (base.ItemContainerGenerator.ContainerFromIndex(index) is XamMenuItem xamMenuItem)
				{
					if (!xamMenuItem.IsEnabled)
					{
						return xamMenuItem.FindSiblingItem(isForward, recurse: false);
					}
					return xamMenuItem;
				}
			}
			ItemsControl parentItemsControl = ParentItemsControl;
			ItemContainerGenerator generator = XamMenuBase.GetGenerator(parentItemsControl);
			if (parentItemsControl == null && generator == null)
			{
				return null;
			}
			XamMenuItem xamMenuItem2 = null;
			int num = generator.IndexFromContainer(this);
			int count = parentItemsControl.Items.Count;
			bool flag = false;
			if (isForward)
			{
				if (num == count - 1)
				{
					num = -1;
					flag = true;
				}
				while (num++ < count)
				{

		private void ExecuteCommand()
		{
			ICommand command = Command;
			if (command == null)
			{
				return;
			}
			object commandParameter = CommandParameter;
			if (command is RoutedCommand routedCommand)
			{
				IInputElement target = CommandTarget ?? this;
				if (routedCommand.CanExecute(commandParameter, target))
				{
					routedCommand.Execute(commandParameter, target);
				}
			}
			else if (command.CanExecute(commandParameter))
			{
				command.Execute(commandParameter);
			}
		}

		{
			if (_isDesignMode)
			{
				return;
			}
			switch (Role)
			{
			case MenuItemRole.TopLevelHeader:
				if (CheckBoxVisibilityResolved == Visibility.Visible || IconVisibilityResolved == Visibility.Visible)
				{
					VisualStateManager.GoToState(this, "TopLevelHeaderWithIcon", useTransitions: false);
				}
				else
				{
					VisualStateManager.GoToState(this, "TopLevelHeader", useTransitions: false);
				}
				break;
			case MenuItemRole.TopLevelItem:
				if (CheckBoxVisibilityResolved == Visibility.Visible || IconVisibilityResolved == Visibility.Visible)
				{
					VisualStateManager.GoToState(this, "TopLevelItemWithIcon", useTransitions: false);
				}
				else

					VisualStateManager.GoToState(this, "TopLevelItem", useTransitions: false);
				}
				break;

				VisualStateManager.GoToState(this, "SubmenuHeader", useTransitions: false);

			case MenuItemRole.SubmenuItem:
				VisualStateManager.GoToState(this, "SubmenuItem", useTransitions: false);

		}

		private static void PlaceSubmenuOnPrefferedPosition(out double popupX, out double popupY, out bool below, double rootOffsetX, double rootOffsetY, double rootWidth, double rootHeight, double popupContentWidth, double popupContentHeight, double myControlWidth, double myControlHeight, MenuItemPosition prefPosition)
		{
			below = true;
			popupX = 0.0;
			popupY = 0.0;
			switch (prefPosition)
			{
			case MenuItemPosition.Bottom:
				popupX = DefineHorizontalOffset(rootOffsetX, rootWidth, popupContentWidth);
				popupY = rootOffsetY + myControlHeight;
				popupY = DefineTopOrBottom(popupY, ref below, rootOffsetY, rootHeight, popupContentHeight, myControlHeight);
				break;
			case MenuItemPosition.Top:
				popupX = DefineHorizontalOffset(rootOffsetX, rootWidth, popupContentWidth);
				popupY = rootHeight + 1.0;
				popupY = DefineTopOrBottom(popupY, ref below, rootOffsetY, rootHeight, popupContentHeight, myControlHeight);
				break;
			case MenuItemPosition.Left:
				popupX = rootOffsetX - popupContentWidth;
				if (popupX < 0.0)
				{
					popupX = 0.0;
					popupY = rootOffsetY + myControlHeight;
					popupY = DefineTopOrBottom(popupY, ref below, rootOffsetY, rootHeight, popupContentHeight, myControlHeight);
				}
				else
				{
					popupY = DefineVerticalOffset(rootOffsetY, rootHeight, popupContentHeight);
				}
				break;
			case MenuItemPosition.Right:
			{
				popupX = rootOffsetX + myControlWidth;
				double num = DefineHorizontalOffset(popupX, rootWidth, popupContentWidth);
				if (popupX != num)
				{
					popupX = num;
					popupY = rootOffsetY + myControlHeight;
					popupY = DefineTopOrBottom(popupY, ref below, rootOffsetY, rootHeight, popupContentHeight, myControlHeight);
				}
				else
				{
					popupY = DefineVerticalOffset(rootOffsetY, rootHeight, popupContentHeight);
				}
				break;
			}
			}
		}

		private static double DefineTopOrBottom(double popupY, ref bool below, double rootOffsetY, double rootHeight, double popupContentHeight, double myControlHeight)
		{
			double num = popupY;
			if (rootHeight < num + popupContentHeight)
			{
				below = false;
				num = rootOffsetY - popupContentHeight;
				if (num <= 0.0)
				{
					if (rootOffsetY < (rootHeight - myControlHeight) / 2.0)
					{
						below = true;
						num = rootOffsetY + myControlHeight;
					}
					else
					{
						below = false;
						num = rootOffsetY - popupContentHeight;
					}
				}
			}
			return num;
		}

		private static double DefineVerticalOffset(double rootOffsetY, double rootHeight, double popupContentHeight)
		{
			double num = rootOffsetY;
			if (rootHeight < num + popupContentHeight)
			{
				num = rootHeight - popupContentHeight;
				num = Math.Max(0.0, num);
			}
			return num;
		}

		private static double DefineVerticalAlignment(ref bool below, double rootOffsetY, double rootHeight, double popupContentHeight, double myControlHeight)
		{
			double num = rootOffsetY;
			if (rootHeight < num + popupContentHeight)
			{
				below = false;
				num = rootOffsetY + myControlHeight - popupContentHeight;
				if (num < 0.0)
				{
					if (rootOffsetY < (rootHeight - myControlHeight) / 2.0)
					{
						below = true;
						num = rootOffsetY + myControlHeight;
					}
					else
					{
						below = false;
						num = rootOffsetY - popupContentHeight;
					}
				}
			}
			return num;
		}

		private static double DefineHorizontalOffset(double refferPoint, double rootWidth, double popupContentWidth)
		{
			double zoomFactor = PlatformProxy.GetZoomFactor();
			double num = rootWidth / zoomFactor;
			double num2 = refferPoint;
			if (num < num2 + popupContentWidth)
			{
				num2 = num - popupContentWidth;
				num2 = Math.Max(0.0, num2);
			}
			return num2;
		}

		private static double DefineLeftOrRight(double popupX, double rootOffsetX, double rootWidth, double popupContentWidth)
		{
			bool isOverlay;
			return DefineLeftOrRight(popupX, rootOffsetX, rootWidth, popupContentWidth, out isOverlay);
		}

		private static double DefineLeftOrRight(double popupX, double rootOffsetX, double rootWidth, double popupContentWidth, out bool isOverlay)
		{
			isOverlay = false;
			double num = popupX;
			if (rootWidth < num + popupContentWidth)
			{
				num = rootOffsetX - popupContentWidth;
				if (num < 0.0)
				{
					isOverlay = true;
					num = 0.0;
				}
			}
			return num;
		}

		private void HandleUpDownKey(bool isDown)
		{
			if (!IsSelected && isDown)
			{
				IsSelected = true;
				ChangeVisualState(useTransitions: false);
				if (!IsSubmenuOpen)
				{
					return;
				}
			}
			XamMenuItem menuItem;
			if (Role == MenuItemRole.TopLevelHeader)
			{
				if (!IsSubmenuOpen)
				{
					ManipulateSubmenu(isOpen: true, isImmediately: true);
				}
				menuItem = FindSiblingItem(isDown, recurse: true);
				IsHighlighted = true;
			}
			else
			{
				menuItem = FindSiblingItem(isDown, recurse: false);
			}
			MoveToItem(menuItem);
		}

		private void HandleLeftKey()
		{
			XamMenuItem xamMenuItem = null;
			switch (Role)
			{
			case MenuItemRole.SubmenuItem:
			case MenuItemRole.SubmenuHeader:
				xamMenuItem = ParentXamMenuItem;
				if (xamMenuItem == null)
				{
					return;
				}
				xamMenuItem.ManipulateSubmenu(isOpen: false, isImmediately: true);
				if (xamMenuItem.Role != MenuItemRole.TopLevelHeader)
				{
					break;
				}
				xamMenuItem = ParentXamMenuItem.FindSiblingItem(isForward: false, recurse: false);
				if (xamMenuItem != null)
				{
					if (xamMenuItem.Role == MenuItemRole.TopLevelHeader)
					{
						xamMenuItem.ManipulateSubmenu(isOpen: true, isImmediately: true);
					}
					ParentXamMenuItem.IsHighlighted = false;
					ParentXamMenuItem.ChangeVisualState(useTransitions: false);
				}
				break;
			case MenuItemRole.TopLevelItem:
			case MenuItemRole.TopLevelHeader:
				xamMenuItem = FindSiblingItem(isForward: false, recurse: false);
				if (xamMenuItem != null)
				{
					if (xamMenuItem.Role == MenuItemRole.TopLevelHeader)
					{
						xamMenuItem.ManipulateSubmenu(isOpen: true, isImmediately: true);
					}
					else
					{
						ManipulateSubmenu(isOpen: false, isImmediately: true);
					}
				}
				break;
			}
			MoveToItem(xamMenuItem);
		}

		private void HandleRightKey()
		{
			XamMenuItem xamMenuItem = null;
			switch (Role)
			{
			case MenuItemRole.SubmenuHeader:
				ManipulateSubmenu(isOpen: true, isImmediately: true);
				xamMenuItem = FindSiblingItem(isForward: true, recurse: true);
				if (xamMenuItem != null)
				{
					IsHighlighted = true;
				}
				break;
			case MenuItemRole.TopLevelItem:
			case MenuItemRole.TopLevelHeader:
				xamMenuItem = FindSiblingItem(isForward: true, recurse: false);
				if (xamMenuItem != null)
				{
					if (xamMenuItem.Role == MenuItemRole.TopLevelHeader)
					{
						xamMenuItem.ManipulateSubmenu(isOpen: true, isImmediately: true);
					}
					else
					{
						ManipulateSubmenu(isOpen: false, isImmediately: true);
					}
				}
				break;
			case MenuItemRole.SubmenuItem:
				if (ParentXamMenu.CurrentOpen == null)
				{
					return;
				}
				xamMenuItem = ParentXamMenu.CurrentOpen.FindSiblingItem(isForward: true, recurse: false);
				if (xamMenuItem != null)
				{
					if (xamMenuItem.Role == MenuItemRole.TopLevelHeader)
					{
						xamMenuItem.ManipulateSubmenu(isOpen: true, isImmediately: true);
					}
					else if (xamMenuItem.Role == MenuItemRole.TopLevelItem)
					{
						ParentXamMenu.CloseCurrentOpen();
					}
				}
				break;
			}
			MoveToItem(xamMenuItem);
		}

		private void MoveToItem(XamMenuItem menuItem)
		{
			if (menuItem != null)
			{
				menuItem.FocusOrSelect();
				menuItem.ChangeVisualState(useTransitions: false);
				ChangeVisualState(useTransitions: false);
				if (ParentXamMenuItem != null)
				{
					ParentXamMenuItem.HeaderedItemContainerGenerator.ScrollIntoView(menuItem);
				}
			}
		}

		private void HandleItemSelection()
		{
			if (ParentXamMenuItem != null)
			{
				ParentXamMenuItem.IgnoreMouseUp = true;
			}
			if (IgnoreMouseUp)
			{
				IgnoreMouseUp = false;
			}
			else
			{
				if (ParentXamMenu == null)
				{
					return;
				}
				if (IsCheckable)
				{
					IsChecked = !IsChecked;
				}
				RaiseClickEvent();
				if (Role == MenuItemRole.SubmenuItem)
				{
					if (ParentXamMenu != null && !StaysOpenOnClick)
					{
						ParentXamMenu.ReturnFocus();
						IsMouseOver = false;
					}
					else
					{
						Focus();
						if (!StaysOpenOnClick)
						{
							IsMouseOver = false;
						}
					}
				}
				if (Role == MenuItemRole.TopLevelItem && ParentXamMenu != null)
				{
					ParentXamMenu.ReturnFocus();
				}
				if (_hasMouseCapture && !StaysOpenOnClick)
				{
					ReleaseMouseCapture();
