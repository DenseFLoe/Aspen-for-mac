	internal class CustomChromiumWebBrowser : ChromiumWebBrowser
	{
		public CustomChromiumWebBrowser(string defaultURL)
			: base(defaultURL)
		{
		}

		protected override void WndProc(ref Message m)
		{
			if (m.Msg == 528 && ((int)m.WParam == 66059 || (int)m.WParam == 131595))
			{
				if ((int)m.WParam == 66059)
				{
					if (base.CanGoBack)
					{
						GetBrowser().GoBack();
					}
				}
				else if ((int)m.WParam == 131595 && base.CanGoForward)
				{
					GetBrowser().GoForward();
				}
			}
			else
			{
				base.WndProc(ref m);
			}
		}
	}

		public HelpExtractMonitor(string strTitle, string strHtmlHelpFolder, string strHelpExtractAppFullName)
		{

		private void HelpExtractMonitor_Loaded(object sender, RoutedEventArgs e)
		{
			IntPtr handle = new WindowInteropHelper(this).Handle;
			SetWindowLong(handle, -16, GetWindowLong(handle, -16) & -524289);
			ProcessStartInfo processStartInfo = new ProcessStartInfo();
			processStartInfo.FileName = myHelpExtractApp;
			processStartInfo.Arguments = "\"" + myHelpFolder + "\"";
			processStartInfo.WorkingDirectory = myHelpFolder;
			processStartInfo.WindowStyle = ProcessWindowStyle.Hidden;
			processStartInfo.UseShellExecute = false;
			procHandle = new Process();
			procHandle.EnableRaisingEvents = true;
			procHandle.Exited += procHandle_Exited;
			procHandle.StartInfo = processStartInfo;
			bool flag = false;
			try
			{
				flag = procHandle.Start();
			}
			catch (Exception)
			{
				flag = false;
			}
			if (!flag)
			{
				DismissTask();
			}
		}

		private void DismissTask()
		{
			bFinished = true;
			base.Dispatcher.BeginInvoke(new Action(base.Close), DispatcherPriority.Input);
		}

		private void procHandle_Exited(object sender, EventArgs e)
		{
			DismissTask();
		}

		protected override void OnClosing(CancelEventArgs e)
		{
			base.OnClosing(e);
			if (!bFinished)
			{
				e.Cancel = true;
			}
		}

	public class HtmlHelpSystemUtil
	{
		private static int? GetStatCodeInFile(string strStatFile)
		{
			try
			{
				if (File.Exists(strStatFile))
				{
					using StreamReader streamReader = new StreamReader(strStatFile);
					int result = -5;
					string text = streamReader.ReadToEnd();
					if (!string.IsNullOrWhiteSpace(text) && int.TryParse(text, out result))
					{
						return result;
					}
				}
			}
			catch (Exception)
			{
			}
			return null;
		}

		public static void CheckExtractStat(string strProduct, string strProdDisplayName, string strExternalVer, string strInternalVer, string strHtmlHelpFolder, string strHelpExtractApp)
		{
			string strStatFile = Path.Combine(strHtmlHelpFolder, "status.txt");
			int? statCodeInFile = GetStatCodeInFile(strStatFile);
			if (statCodeInFile.HasValue && statCodeInFile.HasValue && statCodeInFile.Value == 0)
			{
				RegistryHelperSelfExtract.SetExtarctStatus(strProduct, strInternalVer, bSuccess: true);
				return;
			}
			RegistryHelperSelfExtract.SetExtarctStatus(strProduct, strInternalVer, bSuccess: false);
			HelpExtractMonitor helpExtractMonitor = new HelpExtractMonitor((string.IsNullOrWhiteSpace(strProdDisplayName) ? strProduct : strProdDisplayName) + " " + strExternalVer, strHtmlHelpFolder, strHelpExtractApp);
			if (helpExtractMonitor != null)
			{
				try
				{
					helpExtractMonitor.ShowDialog();
				}
				catch (Exception)
				{
				}
			}
			statCodeInFile = GetStatCodeInFile(strStatFile);
			if (statCodeInFile.HasValue && statCodeInFile.HasValue && statCodeInFile.Value == 0)
			{
				RegistryHelperSelfExtract.SetExtarctStatus(strProduct, strInternalVer, bSuccess: true);
			}

	internal class RegistryHelperSelfExtract
	{
		private static string strHtmlHelpBase = "Software\\AspenTech\\";

		private static string strExtractStat = "StatusCode";

		public static bool HasExtarctStatus(string strProduct, string strInternalVer, out bool bSuccess)
		{
			bSuccess = false;
			bool result = false;
			string name = strHtmlHelpBase + strProduct + "\\" + strInternalVer;
			RegistryKey registryKey = Registry.CurrentUser.OpenSubKey(name);
			if (registryKey != null)
			{
				if (registryKey.GetValue(strExtractStat) is int num && new int?(num).HasValue)
				{
					result = true;
					if (num == 0)
					{
						bSuccess = true;
					}
				}
				registryKey.Close();
			}
			return result;
		}

		public static void SetExtarctStatus(string strProduct, string strInternalVer, bool bSuccess)
		{
			string subkey = strHtmlHelpBase + strProduct + "\\" + strInternalVer;
			RegistryKey registryKey = Registry.CurrentUser.CreateSubKey(subkey);
			if (registryKey != null)
			{
				registryKey.SetValue(strExtractStat, (!bSuccess) ? 1 : 0);
				registryKey.Close();
			}
		}
	}
	internal class RegistryHelperWinInfo
	{
		private static string strHtmlHelpBase = "Software\\AspenTech\\";

		private static string strDefWinWidth = "DefWinWidth";

		private static string strDefWinHeight = "DefWinHeight";

		public static bool HasDefWinSize(string strProduct, string strInternalVer)
		{
			string name = strHtmlHelpBase + strProduct + "\\" + strInternalVer;
			RegistryKey registryKey = Registry.CurrentUser.OpenSubKey(name);
			if (registryKey == null)
			{
				return false;
			}
			registryKey.Close();
			return true;
		}

		public static void SetDefWinSize(string strProduct, string strInternalVer, int Width, int iHeight)
		{
			string subkey = strHtmlHelpBase + strProduct + "\\" + strInternalVer;
			RegistryKey registryKey = Registry.CurrentUser.CreateSubKey(subkey);
			if (registryKey != null)
			{
				registryKey.SetValue(strDefWinWidth, Width);
				registryKey.SetValue(strDefWinHeight, iHeight);
				registryKey.Close();
			}
		}

		public static bool GetDefWinSize(string strProduct, string strInternalVer, out int iWidth, out int iHeight)
		{
			iWidth = (iHeight = 0);
			string name = strHtmlHelpBase + strProduct + "\\" + strInternalVer;
			RegistryKey registryKey = Registry.CurrentUser.OpenSubKey(name);
			if (registryKey != null)
			{
				int? num = registryKey.GetValue(strDefWinWidth) as int?;
				int? num2 = registryKey.GetValue(strDefWinHeight) as int?;
				registryKey.Close();
				if (num.HasValue && num.Value > 0 && num2.HasValue && num2.Value > 0)
				{

			else if (strUrlNavigatingTo.StartsWith("http://www.aspentech.com/newdefault.html?", StringComparison.InvariantCultureIgnoreCase) || strUrlNavigatingTo.StartsWith("https://www.aspentech.com/newdefault.html?", StringComparison.InvariantCultureIgnoreCase))

		}
