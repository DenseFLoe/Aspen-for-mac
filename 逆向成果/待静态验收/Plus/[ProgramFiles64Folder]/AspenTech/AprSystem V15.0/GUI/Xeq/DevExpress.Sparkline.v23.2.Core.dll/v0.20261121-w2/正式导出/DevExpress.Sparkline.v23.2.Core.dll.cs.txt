		{

			{
				DataSet dataSet = (DataSet)itemsSource;
				dataSet.Tables.CollectionChanged += ItemsSourceDataSetCollectionChanged;
				for (int i = 0; i < dataSet.Tables.Count; i++)
				{

				}

		}

			{

				dataSet.Tables.CollectionChanged -= ItemsSourceDataSetCollectionChanged;
				for (int i = 0; i < dataSet.Tables.Count; i++)

	public sealed class SeriesInteraction
	{
		private readonly List<SeriesProviderPair> dataProviders;

		private readonly SparklineInteractionRanges interactionRanges;

		public int Count => dataProviders.Count;

		public bool HasDataToPresent => Count > 0;

		public SeriesProviderPair this[int index] => dataProviders[index];

		public SeriesInteraction()
		{
			dataProviders = new List<SeriesProviderPair>();
			interactionRanges = new SparklineInteractionRanges();
		}

		private void UpdateRanges(SparklineDataProvider provider)
		{
			List<SparklinePoint> sortedPoints = provider.SortedPoints;
			int count = sortedPoints.Count;
			if (count == 1)
			{
				interactionRanges.DataArgumentRange.Extend(sortedPoints[0].Argument - 0.5);
				interactionRanges.DataArgumentRange.Extend(sortedPoints[0].Argument + 0.5);
				interactionRanges.DataValueRange.Extend(sortedPoints[0].Value - 0.5);
				interactionRanges.DataValueRange.Extend(sortedPoints[0].Value + 0.5);
				interactionRanges.UpdateMinPointDistance(1.0);
				return;
			}
			for (int i = 0; i < count; i++)
			{
				SparklinePoint sparklinePoint = sortedPoints[i];
				interactionRanges.DataArgumentRange.Extend(sparklinePoint.Argument);
				interactionRanges.DataValueRange.Extend(sparklinePoint.Value);
				if (count > 1 && i > 0)
				{
					interactionRanges.UpdateMinPointDistance(sparklinePoint.Argument - sortedPoints[i - 1].Argument);
				}
			}
		}

		public void Clear()
		{
			dataProviders.Clear();
			interactionRanges.Invalidate();
		}

		public void Add(IClientSeries series)
		{
			SparklineDataProvider sparklineDataProvider = new SparklineDataProvider(series, interactionRanges.VisualArgumentRange);
			sparklineDataProvider.AllowPaddingCorrection = false;
			UpdateRanges(sparklineDataProvider);
			dataProviders.Add(new SeriesProviderPair(series, sparklineDataProvider));
		}

		public double NormalizeArgument(double argument)
		{
			SparklineRangeData actualDataArgumentRange = interactionRanges.ActualDataArgumentRange;
			double num = argument - actualDataArgumentRange.Min;
			double delta = actualDataArgumentRange.Delta;
			if (SparklineMathUtils.IsValidDouble(delta) && SparklineMathUtils.IsValidDouble(num))
			{
				if (delta == 0.0)
				{
					return 0.0;
				}
				return num / delta;
			}
			return 0.0;
		}

		public double GetArgument(double normalizedArgument)
		{
			if (SparklineMathUtils.IsValidDouble(normalizedArgument))
			{
				SparklineRangeData actualDataArgumentRange = interactionRanges.ActualDataArgumentRange;
				double delta = actualDataArgumentRange.Delta;
				if (SparklineMathUtils.IsValidDouble(delta) && SparklineMathUtils.IsValidDouble(actualDataArgumentRange.Min))
				{
					return normalizedArgument * delta + actualDataArgumentRange.Min;
				}
			}
			return 0.0;
		}

		{
			double value = GetValue(normalizedValue);
			double num = gridMapping.FloorValue(unit, value) + offset;
			if (num > 0.0)
			{
				return Normalize(gridMapping.GetGridValue(unit, num));
			}
			return 0.0;
		}

		public void DrawContent(IGraphicsCache cache, Rectangle bounds, Matrix normalTransform)
		{
			if (!interaction.HasDataToPresent)
			{
				return;
			}
			for (int i = 0; i < interaction.Count; i++)
			{
				SeriesProviderPair seriesProviderPair = interaction[i];
				if (seriesProviderPair.Series.View != null)
				{
					BaseSparklinePainter painter = sparklinePaintersCache.GetPainter(seriesProviderPair.Series.View);
					painter.Initialize(seriesProviderPair.DataProvider, seriesProviderPair.Series, bounds, interaction.Ranges, normalTransform);
					painter.Draw(cache);
				}
			}
		}

		public void DrawContent(Graphics graphics, Rectangle bounds, Matrix normalTransform)
		{
			if (!interaction.HasDataToPresent)

			}

					painter.Initialize(seriesProviderPair.DataProvider, seriesProviderPair.Series, bounds, interaction.Ranges, normalTransform);
					painter.Draw(graphics);
				}
			}
		}

		public double Normalize(double value)
		{
			double num = interaction.NormalizeArgument(value);
			if (num < 0.0)
			{
				num = 0.0;
			}
			else if (num > 1.0)
			{
				num = 1.0;
			}
			return num;
		}

		public double GetValue(double normalizedValue)
		{
			return interaction.GetArgument(normalizedValue);
		}

		public List<object> GenerateGrid(double min, double max, double width, IChartCoreClientGridOptions gridOptions)
		{
			List<object> list = new List<object>();
			GridUnit gridUnit = null;
			gridUnit = ((!gridOptions.Auto) ? gridOptions.GridUnit : GetAutoGridUnit(min, max, width, gridOptions));
			if (gridUnit != null)
			{
				IChartCoreClientGridMapping gridMapping = gridOptions.GridMapping;
				if ((max - min) / gridUnit.Step > width)
				{
					return list;
				}
				if (gridUnit != null)
				{
					double num = gridMapping.CeilValue(gridUnit, min);
					double num2 = gridMapping.FloorValue(gridUnit, max);
					double num3 = num;
					do
					{
						list.Add(gridOptions.GridMapping.GetGridValue(gridUnit, num3++));
					}
					while (num3 <= num2);
				}
			}
			return list;
		}

		public void SnapNormalRange(SparklineRangeData range, SnapBounds changedBounds, IChartCoreClientGridOptions gridOptions)
		{
			GridUnit snapUnit = gridOptions.SnapUnit;
			IChartCoreClientGridMapping gridMapping = gridOptions.GridMapping;
			double num = RoundNormalValue(range.Min, snapUnit, gridMapping);
			double num2 = RoundNormalValue(range.Max, snapUnit, gridMapping);
			if (SparklineMathUtils.AreDoublesEqual(num, num2))
			{
				double num3 = num;
				if (SparklineMathUtils.AreDoublesEqual(range.Min, range.Max))
				{
					switch (changedBounds)
					{
					case SnapBounds.None:
					case SnapBounds.Both:
						if (num3 - range.Min > 0.0)
						{
							num = ShiftNormalValue(num3, snapUnit, gridMapping, -1.0);
						}
						else
						{
							num2 = ShiftNormalValue(num3, snapUnit, gridMapping, 1.0);
						}
						break;
					case SnapBounds.Minimum:
						num = ShiftNormalValue(num3, snapUnit, gridMapping, -1.0);
						break;
					case SnapBounds.Maximum:
						num2 = ShiftNormalValue(num3, snapUnit, gridMapping, 1.0);
						break;
					}
				}
				else
				{
					double num4 = Math.Abs(num3 - range.Min);
					double num5 = Math.Abs(num3 - range.Max);
					if (num4 > num5)
					{
						num = FloorNormalValue(range.Min, snapUnit, gridMapping);
					}
					else
					{
						num2 = CeilNormalValue(range.Max, snapUnit, gridMapping);
					}
				}
			}
			range.Set(num, num2);
		}

		public void SetCustomArgumentRange(SparklineRangeData range)
		{
			Interaction.Ranges.CustomDataArgumentRange.Set(range.Min, range.Max);
		}

	public sealed class SparklineRangeData
	{
		private double min;

		private double max;

		public double Min => min;

		public double Max => max;

		public double Delta => Max - Min;

		public bool IsValid
		{
			get
			{
				if (!double.IsNaN(Min))
				{
					return !double.IsNaN(Max);
				}
				return false;
			}
		}

		public SparklineRangeData()
		{
			Invalidate();
		}

		public SparklineRangeData(double min, double max)
		{
			this.min = min;
			this.max = max;
		}

		internal void Extend(double value)
		{
			if (SparklineMathUtils.IsValidDouble(value))
			{
				if (double.IsNaN(min))
				{
					min = value;
				}
				else if (min > value)
				{
					min = value;
				}
				if (double.IsNaN(max))
				{
					max = value;
				}
				else if (max < value)
				{
					max = value;
				}
			}
		}

		internal void Invalidate()
		{
			min = double.NaN;
			max = double.NaN;
		}

		public void Set(double min, double max)
		{
			this.min = min;
			this.max = max;
		}

		public bool InRange(double value)
		{
			if (value >= min)
			{
				return value <= max;
			}
			return false;
		}

		public override string ToString()
		{
			if (IsValid)
			{
				return $"SparklineRange [{min}:{max}]";
			}
			return "Invalid SparklineRange";
		}
	}

			if (num >= 0.0 && num2 >= 0.0)
			{

					num = 0.0;
				}
				if (num == num2)

					num2 = 0.0;

			return base.CorrectValueRange(new SparklineRangeData(num, num2));
		}

		protected override SparklineRangeData CorrectArgumentRange(double minPointDistance, SparklineRangeData range)

	public abstract class SparklineMappingBase
	{
		private readonly Rectangle bounds;

		private double scaleX;

		private double yZeroValue;

		private double minPointsDistancePx;

		private int screenYZeroValue;

		private SparklineRangeData argumentRange;

		private SparklineRangeData valueRange;

		private SparklineRangeData originalArgumentRange;

		private SparklineRangeData originalValueRange;

		private Matrix normalTransform;

		public int ScreenYZeroValue => screenYZeroValue;

		public double MinPointsDistancePx => minPointsDistancePx;

		public double ScaleX => scaleX;

		public Rectangle Bounds => bounds;

		public SparklineRangeData ValueRange => valueRange;

		private static SparklineMappingBase Create(SparklineViewType viewType, Rectangle bounds)
		{
			if (bounds.Width <= 0 || bounds.Height <= 0)
			{
				return null;
			}
			return viewType switch
			{
				SparklineViewType.Bar => new BarSparklineMapping(bounds), 
				SparklineViewType.WinLoss => new WinLossSparklineMapping(bounds), 
				_ => new LineSparklineMapping(bounds), 
			};
		}

		public static SparklineMappingBase Create(SparklineViewType viewType, Rectangle bounds, SparklineDataProvider dataProvider)
		{
			SparklineMappingBase sparklineMappingBase = Create(viewType, bounds);
			sparklineMappingBase?.Initialize(dataProvider);
			return sparklineMappingBase;
		}

		public static SparklineMappingBase Create(SparklineViewType viewType, Rectangle bounds, SparklineDataProvider dataProvider, SparklineInteractionRanges ranges, Matrix normalTransform)
		{
			SparklineMappingBase sparklineMappingBase = Create(viewType, bounds);
			sparklineMappingBase?.Initialize(dataProvider, ranges, normalTransform);
			return sparklineMappingBase;
		}

		protected SparklineMappingBase(Rectangle bounds)
		{
			this.bounds = bounds;
		}

		private void Initialize(SparklineDataProvider dataProvider)
		{
			originalValueRange = dataProvider.OriginalValueRange;
			originalArgumentRange = dataProvider.DataArgumentRange;
			double minInternalArgumentDistance = GetMinInternalArgumentDistance(dataProvider.SortedPoints, originalArgumentRange);
			argumentRange = CorrectArgumentRange(minInternalArgumentDistance, originalArgumentRange);
			valueRange = CorrectValueRange(dataProvider.DataValueRange);
			scaleX = CalculateScaleX();
			minPointsDistancePx = scaleX * minInternalArgumentDistance;
			yZeroValue = CalculateZeroValue();
			screenYZeroValue = MapYValueToScreen(yZeroValue);
		}

		private void Initialize(SparklineDataProvider dataProvider, SparklineInteractionRanges ranges, Matrix normalTransform)
		{
			this.normalTransform = normalTransform;
			originalValueRange = dataProvider.OriginalValueRange;
			originalArgumentRange = ranges.ActualDataArgumentRange;
			argumentRange = CorrectArgumentRange(ranges.MinPointDistance, originalArgumentRange);
			valueRange = CorrectValueRange(ranges.DataValueRange);
			scaleX = CalculateScaleX();
			minPointsDistancePx = scaleX * ranges.MinPointDistance;
			yZeroValue = CalculateZeroValue();
			screenYZeroValue = MapYValueToScreen(yZeroValue);
		}

		private double GetMinInternalArgumentDistance(IList<SparklinePoint> points, SparklineRangeData argumentRange)
		{
			if (points.Count == 0)
			{
				return 0.0;
			}
			double num = double.NaN;
			for (int i = 1; i < points.Count; i++)
			{
				double argument = points[i].Argument;
				double argument2 = points[i - 1].Argument;
				if (argumentRange.InRange(argument) || argumentRange.InRange(argument2))
				{
					double num2 = argument - argument2;
					if (num2 != 0.0 && (double.IsNaN(num) || num > num2))
					{
						num = num2;
					}
				}
			}
			if (double.IsNaN(num))
			{
				return 0.5;
			}
			return num;
		}

		private double CalculateScaleX()
		{
			if (argumentRange.Max == argumentRange.Min)
			{
				return 0.0;
			}
			return (double)Bounds.Width / (argumentRange.Max - argumentRange.Min);
		}

		protected virtual SparklineRangeData CorrectArgumentRange(double minPointDistance, SparklineRangeData range)
		{
			return range;
		}

		protected virtual SparklineRangeData CorrectValueRange(SparklineRangeData range)
		{
			if (range.Min == range.Max)
			{
				return new SparklineRangeData(range.Min - 0.5, range.Max + 0.5);
			}
			return range;
		}

		protected virtual double CorrectValue(double value)
		{
			return value;
		}

		protected virtual double CalculateZeroValue()
		{
			if (!(valueRange.Min > 0.0))
			{
				if (!(valueRange.Max < 0.0))
				{
					return 0.0;
				}
				return valueRange.Max;
			}
			return valueRange.Min;
		}

		public bool IsMinValuePoint(double value)
		{
			return originalValueRange.Min == value;
		}

		public bool IsMaxValuePoint(double value)
		{
			return originalValueRange.Max == value;
		}

		public bool IsStartPoint(double argument)
		{
			return argument == originalArgumentRange.Min;
		}

		public bool IsEndPoint(double argument)
		{
			return argument == originalArgumentRange.Max;
		}

		public int MapYValueToScreen(double value)
		{
			return Bounds.Y + SparklineMathUtils.Round((valueRange.Max - CorrectValue(value)) / (valueRange.Max - valueRange.Min) * (double)Bounds.Height);
		}

		public int MapXValueToScreen(double value)
		{
			return Bounds.X + SparklineMathUtils.Round((value - argumentRange.Min) * ScaleX);
		}

		public Point MapPoint(double argument, double value)
		{
			double num = 0.0;
			if (valueRange.Delta != 0.0)
			{
				num = (valueRange.Max - CorrectValue(value)) / valueRange.Delta;
			}
			double num2 = 0.0;
			if (argumentRange.Delta != 0.0)
			{
				num2 = (argument - argumentRange.Min) / argumentRange.Delta;
			}
			PointF pointF = new PointF((float)num2, (float)num);
			if (normalTransform != null)
			{
				PointF[] array = new PointF[1] { pointF };
				normalTransform.TransformPoints(array);
				pointF = array[0];
			}
			return new Point(SparklineMathUtils.Round((float)Bounds.X + pointF.X * (float)bounds.Width), SparklineMathUtils.Round((float)Bounds.Y + pointF.Y * (float)bounds.Height));
		}

		public PointF MapPointF(double argument, double value)
		{
			double num = 0.0;
			if (valueRange.Delta != 0.0)
			{
				num = (valueRange.Max - CorrectValue(value)) / valueRange.Delta;
			}
			double num2 = 0.0;
			if (argumentRange.Delta != 0.0)
			{
				num2 = (argument - argumentRange.Min) / argumentRange.Delta;
			}
			PointF pointF = new PointF((float)num2, (float)num);
			if (normalTransform != null)
			{
				PointF[] array = new PointF[1] { pointF };
				normalTransform.TransformPoints(array);
				pointF = array[0];
			}
			return new PointF(SparklineMathUtils.Round((float)Bounds.X + pointF.X * (float)bounds.Width), SparklineMathUtils.Round((float)Bounds.Y + pointF.Y * (float)bounds.Height));
		}
	}

	public class BarSparklinePainter : BaseSparklinePainter
	{

		public override SparklineViewType SparklineType => SparklineViewType.Bar;

		private void DrawBar(ISparklineRenderer renderer, int barWidth, int halfBarWidth, int i, PointF screenPoint)
		{
			screenPoint.X -= halfBarWidth;
			float height = Math.Abs(screenPoint.Y - (float)base.Mapping.ScreenYZeroValue);
			PointPresentationType pointPresentationType = GetPointPresentationType(i);
			Color pointColor = GetPointColor(pointPresentationType);
			renderer.FillRectangle(screenPoint.X, Math.Min(screenPoint.Y, base.Mapping.ScreenYZeroValue), barWidth, height, pointColor);
		}

		protected override PaddingInfo GetMarkersPadding()
		{
			//IL_0002: Unknown result type (might be due to invalid IL or missing references)
			//IL_0008: Unknown result type (might be due to invalid IL or missing references)
			return default(PaddingInfo);
		}

		protected override void DrawInternal(ISparklineRenderer renderer)
		{
			renderer.SetClip(base.Mapping.Bounds);
			int num = Math.Max(1, SparklineMathUtils.Round(base.Mapping.MinPointsDistancePx) - BarView.BarDistance);
			int halfBarWidth = (int)Math.Round((double)num / 2.0);
			if (base.MappedPoints != null)
			{
				foreach (SparklineMappedPointInfo mappedPoint in base.MappedPoints)
				{
					DrawBar(renderer, num, halfBarWidth, mappedPoint.Index, mappedPoint.Point);
				}
			}
			else
			{
				int indexOfFirstPointForDrawing = GetIndexOfFirstPointForDrawing();
				int indexOfLastPointForDrawing = GetIndexOfLastPointForDrawing();
				for (int i = indexOfFirstPointForDrawing; i <= indexOfLastPointForDrawing; i++)
				{
					SparklinePoint sparklinePoint = base.DataProvider.SortedPoints[i];
					if (SparklineMathUtils.IsValidDouble(sparklinePoint.Value) && SparklineMathUtils.IsValidDouble(sparklinePoint.Argument))
					{
						PointF screenPoint = base.Mapping.MapPointF(sparklinePoint.Argument, sparklinePoint.Value);
						DrawBar(renderer, num, halfBarWidth, i, screenPoint);
					}
				}
			}
			renderer.RestoreClip();
		}

		public override List<SparklineMappedPointInfo> GetMappedPoints()
		{
			int indexOfFirstPointForDrawing = GetIndexOfFirstPointForDrawing();
			int indexOfLastPointForDrawing = GetIndexOfLastPointForDrawing();
			List<SparklineMappedPointInfo> list = new List<SparklineMappedPointInfo>();
			if (base.Mapping != null)
			{
				for (int i = indexOfFirstPointForDrawing; i <= indexOfLastPointForDrawing; i++)
				{
					SparklinePoint sparklinePoint = base.DataProvider.SortedPoints[i];
					if (SparklineMathUtils.IsValidDouble(sparklinePoint.Value) && SparklineMathUtils.IsValidDouble(sparklinePoint.Argument))
					{
						list.Add(new SparklineMappedPointInfo
						{
							Point = base.Mapping.MapPointF(sparklinePoint.Argument, sparklinePoint.Value),
							Index = i
						});
					}
				}
			}
			return list;
		}

	public class CohenSutherlandLineClipAlgorithm
	{
		private const int LEFT = 1;

		private const int RIGHT = 2;

		private const int TOP = 4;

		private const int BOTTOM = 8;

		private readonly Rectangle bounds;

		public CohenSutherlandLineClipAlgorithm(Rectangle bounds)
		{
			this.bounds = bounds;
		}

		private int CalculateOutCode(PointF point)
		{
			int num = 0;
			if (point.X < (float)bounds.Left)
			{
				num |= 1;
			}
			else if (point.X > (float)bounds.Right)
			{
				num |= 2;
			}
			if (point.Y < (float)bounds.Top)
			{
				num |= 4;
			}
			else if (point.Y > (float)bounds.Bottom)
			{
				num |= 8;
			}
			return num;
		}

		public Tuple<PointF, PointF> GetCutLine(PointF p1, PointF p2)
		{
			int num = CalculateOutCode(p1);
			int num2 = CalculateOutCode(p2);
			bool flag = false;
			while (true)
			{
				if ((num | num2) == 0)
				{
					flag = true;
					break;
				}
				if ((num & num2) != 0)
				{
					break;
				}
				float x = float.NaN;
				float y = float.NaN;
				int num3 = ((num != 0) ? num : num2);
				if ((num3 & 8) != 0)
				{
					x = p1.X + (p2.X - p1.X) * ((float)bounds.Bottom - p1.Y) / (p2.Y - p1.Y);
					y = bounds.Bottom;
				}
				else if ((num3 & 4) != 0)
				{
					x = p1.X + (p2.X - p1.X) * ((float)bounds.Top - p1.Y) / (p2.Y - p1.Y);
					y = bounds.Top;
				}
				else if ((num3 & 2) != 0)
				{
					y = p1.Y + (p2.Y - p1.Y) * ((float)bounds.Right - p1.X) / (p2.X - p1.X);
					x = bounds.Right;
				}
				else if ((num3 & 1) != 0)
				{
					y = p1.Y + (p2.Y - p1.Y) * ((float)bounds.Left - p1.X) / (p2.X - p1.X);
					x = bounds.Left;
				}
				if (num3 == num)
				{
					p1 = new PointF(x, y);
					num = CalculateOutCode(p1);
				}
				else

					p2 = new PointF(x, y);
					num2 = CalculateOutCode(p2);
				}
			}

		private static int GetActualDayOfWeek(DateTime dateTime)
		{
			int num = dateTime.DayOfWeek - FirstDayOfWeek;

				num += 7;

		private static double TotalFloorMonths(DateTime dateTime)
		{
			return dateTime.Year * 12 + dateTime.Month - 13;
		}

		private static DateTime AddMonths(DateTimeMeasureUnit measureUnit, DateTime date, double range, double factor)
		{
			range *= factor;
			DateTime fromDate = Floor(date, DateTimeMeasureUnit.Month);
			range += Difference(fromDate, date, measureUnit);
			int num = (int)Math.Floor(range);
			double num2 = range - (double)num;
			DateTime dateTime = fromDate.AddMonths(num);
			int days = (dateTime.AddMonths(1) - dateTime).Days;
			return dateTime.AddDays((double)days * num2);
		}

		public static DateTime Floor(DateTime dateTime, DateTimeMeasureUnit measureUnit)
		{
			DayOfWeek firstDayOfWeek = FirstDayOfWeek;
			switch (measureUnit)
			{
			case DateTimeMeasureUnit.Year:
				return new DateTime(dateTime.Year, 1, 1);
			case DateTimeMeasureUnit.Month:
				return new DateTime(dateTime.Year, dateTime.Month, 1);
			case DateTimeMeasureUnit.Day:
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day);
			case DateTimeMeasureUnit.Hour:
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day, dateTime.Hour, 0, 0);
			case DateTimeMeasureUnit.Minute:
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day, dateTime.Hour, dateTime.Minute, 0);
			case DateTimeMeasureUnit.Second:
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day, dateTime.Hour, dateTime.Minute, dateTime.Second);
			case DateTimeMeasureUnit.Millisecond:
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day, dateTime.Hour, dateTime.Minute, dateTime.Second, dateTime.Millisecond);
			case DateTimeMeasureUnit.Week:
			{
				int actualDayOfWeek = GetActualDayOfWeek(dateTime);
				DateTime dateTime3 = ((!((dateTime - DateTime.MinValue).TotalDays < (double)actualDayOfWeek)) ? dateTime.AddDays(-actualDayOfWeek) : DateTime.MinValue);
				return new DateTime(dateTime3.Year, dateTime3.Month, dateTime3.Day);
			}
			case DateTimeMeasureUnit.Quarter:
			{
				DateTime dateTime2 = dateTime.AddMonths(-((dateTime.Month - 1) % 3));
				return new DateTime(dateTime2.Year, dateTime2.Month, 1);
			}
			default:
				return dateTime;
			}
		}

		public static TimeSpan Floor(TimeSpan timeSpan, TimeSpanMeasureUnit measureUnit)
		{
			return measureUnit switch

				TimeSpanMeasureUnit.Day => new TimeSpan(timeSpan.Days, 0, 0, 0), 
				TimeSpanMeasureUnit.Hour => new TimeSpan(timeSpan.Days, timeSpan.Hours, 0, 0), 
				TimeSpanMeasureUnit.Minute => new TimeSpan(timeSpan.Days, timeSpan.Hours, timeSpan.Minutes, 0), 
				TimeSpanMeasureUnit.Second => new TimeSpan(timeSpan.Days, timeSpan.Hours, timeSpan.Minutes, timeSpan.Seconds), 
				TimeSpanMeasureUnit.Millisecond => new TimeSpan(timeSpan.Days, timeSpan.Hours, timeSpan.Minutes, timeSpan.Seconds, timeSpan.Milliseconds), 
				_ => timeSpan, 
			};
		}

		public static DateTime Add(DateTime dateTime, DateTimeMeasureUnit measureUnit, double range)
		{
			DayOfWeek firstDayOfWeek = FirstDayOfWeek;
			DateTime dateTime2 = Floor(dateTime, measureUnit);
			DateTime dateTime3;
			switch (measureUnit)
			{
			case DateTimeMeasureUnit.Year:
				dateTime3 = AddMonths(measureUnit, dateTime2, range, 12.0);
				break;
			case DateTimeMeasureUnit.Quarter:
				dateTime3 = AddMonths(measureUnit, dateTime2, range, 3.0);
				break;
			case DateTimeMeasureUnit.Month:
				dateTime3 = AddMonths(measureUnit, dateTime2, range, 1.0);
				break;
			case DateTimeMeasureUnit.Week:
			{
				int num = 0;
				if (dateTime2 == DateTime.MinValue)
				{
					num = GetActualDayOfWeek(dateTime);
				}
				double value = Math.Max(range * 7.0 - (double)num, 0.0);
				dateTime3 = dateTime2.AddDays(value);
				break;
			}
			case DateTimeMeasureUnit.Day:
				dateTime3 = dateTime2.AddDays(range);
				break;
			case DateTimeMeasureUnit.Hour:
				dateTime3 = dateTime2.AddHours(range);
				break;
			case DateTimeMeasureUnit.Minute:
				dateTime3 = dateTime2.AddMinutes(range);
				break;
			case DateTimeMeasureUnit.Second:
				dateTime3 = dateTime2.AddSeconds(range);
				break;
			case DateTimeMeasureUnit.Millisecond:
				dateTime3 = dateTime2.AddMilliseconds(range);
				break;
			default:
				dateTime3 = dateTime2;
				break;
			}
			return Round(dateTime3, measureUnit);
		}

		public static TimeSpan Add(TimeSpan value, TimeSpanMeasureUnit measureUnit, int range)

				_ => TimeSpan.Zero,

			return Round(timeSpan, measureUnit);
		}

		public static double Difference(DateTime fromDate, DateTime toDate, DateTimeMeasureUnit measureUnit)
		{
			DateTime dateTime = Floor(fromDate, measureUnit);
			DateTime dateTime2 = Floor(toDate, measureUnit);
			switch (measureUnit)
			{
			case DateTimeMeasureUnit.Year:
				return dateTime2.Year - dateTime.Year;
			case DateTimeMeasureUnit.Quarter:
				return (TotalFloorMonths(dateTime2) - TotalFloorMonths(dateTime)) / 3.0;
			case DateTimeMeasureUnit.Month:
				return TotalFloorMonths(dateTime2) - TotalFloorMonths(dateTime);
			case DateTimeMeasureUnit.Week:
			{
				int num = 0;
				if (dateTime == DateTime.MinValue)
				{
					num = GetActualDayOfWeek(fromDate);
				}
				else if (dateTime2 == DateTime.MinValue)
				{
					num = GetActualDayOfWeek(toDate);
				}
				return ((dateTime2 - dateTime).TotalDays + (double)num) / 7.0;
			}
			case DateTimeMeasureUnit.Day:
				return (dateTime2 - dateTime).TotalDays;
			case DateTimeMeasureUnit.Hour:
				return (dateTime2 - dateTime).TotalHours;
			case DateTimeMeasureUnit.Minute:
				return (dateTime2 - dateTime).TotalMinutes;
			case DateTimeMeasureUnit.Second:
				return (dateTime2 - dateTime).TotalSeconds;
			case DateTimeMeasureUnit.Millisecond:
				return (dateTime2 - dateTime).TotalMilliseconds;
			default:
				return 0.0;
			}
		}

			TimeSpan timeSpan2 = Floor(to, measureUnit);

				TimeSpanMeasureUnit.Hour => (timeSpan2 - timeSpan).TotalHours, 
				TimeSpanMeasureUnit.Minute => (timeSpan2 - timeSpan).TotalMinutes, 
				TimeSpanMeasureUnit.Second => (timeSpan2 - timeSpan).TotalSeconds, 
				TimeSpanMeasureUnit.Millisecond => (timeSpan2 - timeSpan).TotalMilliseconds, 
				_ => 0.0, 
			};
		}

		public static DateTime Round(DateTime dateTime, DateTimeMeasureUnit measureUnit)
		{
			DayOfWeek firstDayOfWeek = FirstDayOfWeek;
			int num = ((dateTime.Month == 2) ? 14 : 15);
			switch (measureUnit)
			{
			case DateTimeMeasureUnit.Year:
				if (dateTime.Month > 6)
				{
					dateTime = dateTime.AddYears(1);
				}
				return new DateTime(dateTime.Year, 1, 1);
			case DateTimeMeasureUnit.Month:
				if (dateTime.Day > num)
				{
					dateTime = dateTime.AddMonths(1);
				}
				return new DateTime(dateTime.Year, dateTime.Month, 1);
			case DateTimeMeasureUnit.Day:
				if (dateTime.Hour > 12 || (dateTime.Hour == 12 && (dateTime.Minute > 0 || dateTime.Second > 0 || dateTime.Millisecond > 0)))
				{
					dateTime = dateTime.AddDays(1.0);
				}
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day);
			case DateTimeMeasureUnit.Hour:
				if (dateTime.Minute >= 30)
				{
					dateTime = dateTime.AddHours(1.0);
				}
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day, dateTime.Hour, 0, 0);
			case DateTimeMeasureUnit.Minute:
				if (dateTime.Second >= 30)
				{
					dateTime = dateTime.AddMinutes(1.0);
				}
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day, dateTime.Hour, dateTime.Minute, 0);
			case DateTimeMeasureUnit.Second:
				if (dateTime.Millisecond >= 500)
				{
					dateTime = dateTime.AddSeconds(1.0);
				}
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day, dateTime.Hour, dateTime.Minute, dateTime.Second);
			case DateTimeMeasureUnit.Millisecond:
				return new DateTime(dateTime.Year, dateTime.Month, dateTime.Day, dateTime.Hour, dateTime.Minute, dateTime.Second, dateTime.Millisecond);
			case DateTimeMeasureUnit.Week:
			{
				int num2 = dateTime.DayOfWeek - firstDayOfWeek;
				if (num2 > 0)
				{
					if (num2 > 3)
					{
						num2 += 7 - num2;
					}
				}
				else if (num2 <= -4)
				{
					num2 -= 7 + num2;
				}
				DateTime dateTime3 = dateTime.AddDays(num2);
				return new DateTime(dateTime3.Year, dateTime3.Month, dateTime3.Day);
			}
			case DateTimeMeasureUnit.Quarter:
			{
				if (dateTime.Day > num)
				{
					dateTime = dateTime.AddMonths(1);
				}
				DateTime dateTime2 = dateTime.AddMonths(-1 * ((dateTime.Month - 1) % 3));
				return new DateTime(dateTime2.Year, dateTime2.Month, 1);
			}
			default:
				return dateTime;
			}
		}

		public static TimeSpan Round(TimeSpan timeSpan, TimeSpanMeasureUnit measureUnit)
		{
			TimeSpan ts = TimeSpan.Zero;
			switch (measureUnit)
			{
			case TimeSpanMeasureUnit.Day:
				if (timeSpan.Hours > 12 || (timeSpan.Hours == 12 && (timeSpan.Minutes > 0 || timeSpan.Seconds > 0 || timeSpan.Milliseconds > 0)))
				{
					ts = new TimeSpan(1, 0, 0, 0);
				}
				break;
			case TimeSpanMeasureUnit.Hour:
				if (timeSpan.Minutes >= 30)
				{
					ts = new TimeSpan(0, 1, 0, 0);
				}
				break;
			case TimeSpanMeasureUnit.Minute:
				if (timeSpan.Seconds >= 30)
				{
					ts = new TimeSpan(0, 0, 1, 0);
				}
				break;
			case TimeSpanMeasureUnit.Second:
				if (timeSpan.Milliseconds >= 500)
				{
					ts = new TimeSpan(0, 0, 0, 1);
				}
				break;
			}
			timeSpan = timeSpan.Add(ts);
			return Floor(timeSpan, measureUnit);
		}
