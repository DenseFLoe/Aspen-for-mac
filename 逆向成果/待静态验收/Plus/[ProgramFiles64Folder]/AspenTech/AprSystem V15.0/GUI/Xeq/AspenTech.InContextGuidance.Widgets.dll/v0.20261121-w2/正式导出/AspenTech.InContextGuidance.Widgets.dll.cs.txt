	public class InContextGuidanceRelayActionButton : InContextGuidanceButton
	{
		public static readonly DependencyProperty InContextGuidancePropertiesGeneratorProperty = DependencyProperty.Register("InContextGuidancePropertiesGenerator", typeof(Func<InContextGuidanceRelayActionButton, InContextGuidancePropertiesRequestedEventArgs>), typeof(InContextGuidanceRelayActionButton));

		public static readonly RoutedEvent InContextGuidancePropertiesRequestedEvent = EventManager.RegisterRoutedEvent("InContextGuidancePropertiesRequested", RoutingStrategy.Bubble, typeof(EventHandler<InContextGuidancePropertiesRequestedEventArgs>), typeof(InContextGuidanceRelayActionButton));

		public Func<InContextGuidanceRelayActionButton, InContextGuidancePropertiesRequestedEventArgs> InContextGuidancePropertiesGenerator
		{
			get
			{
				return (Func<InContextGuidanceRelayActionButton, InContextGuidancePropertiesRequestedEventArgs>)GetValue(InContextGuidancePropertiesGeneratorProperty);
			}
			set
			{
				SetValue(InContextGuidancePropertiesGeneratorProperty, value);
			}
		}

		public event EventHandler<InContextGuidancePropertiesRequestedEventArgs> InContextGuidancePropertiesRequested
		{
			add
			{
				AddHandler(InContextGuidancePropertiesRequestedEvent, value);
			}
			remove
			{
				RemoveHandler(InContextGuidancePropertiesRequestedEvent, value);
			}
		}

		public InContextGuidanceRelayActionButton()
		{
			base.Loaded += InContextGuidanceRelayActionButton_Loaded;
			CommandManager.RequerySuggested += delegate
			{
				OnCommandManagerRequerySuggested();
			};
		}

		protected virtual void OnInContextGuidancePropertiesRequested(InContextGuidancePropertiesRequestedEventArgs args)
		{
			args.RoutedEvent = InContextGuidancePropertiesRequestedEvent;
			args.Source = this;
			RaiseEvent(args);
		}

		protected virtual void OnCommandManagerRequerySuggested()
		{
			TryRequeryContextProperties();
		}

		private void InContextGuidanceRelayActionButton_Loaded(object sender, RoutedEventArgs e)
		{
			TryRequeryContextProperties();
		}

		public void TryRequeryContextProperties()
		{
			if (ExchangeAdapter.Default.SuccessfullyInitialized && base.IsLoaded)
			{
				RequestAndUpdateInContextGuidanceProperties();
			}
		}

		private void RequestAndUpdateInContextGuidanceProperties()
		{
			InContextGuidancePropertiesRequestedEventArgs e = InContextGuidancePropertiesGenerator?.Invoke(this);
			if (e == null)
			{
				e = new InContextGuidancePropertiesRequestedEventArgs
				{
					ProductName = base.ProductName,
					ProductVersion = base.ProductVersion,
					ContextId = base.ContextId,
					Visibility = base.Visibility
				};
			}
			OnInContextGuidancePropertiesRequested(e);
			if (base.ProductName != e.ProductName)
			{
				base.ProductName = e.ProductName;
			}
			if (base.ProductVersion != e.ProductVersion)
			{
				base.ProductVersion = e.ProductVersion;
			}
			if (base.ContextId != e.ContextId)
			{
				base.ContextId = e.ContextId;
			}
			if (base.Visibility != e.Visibility)
			{
				base.Visibility = e.Visibility;
			}
		}
	}

		{

		protected SearchBoxPopup InternalPopup => _internalPopup ?? (_internalPopup = new SearchBoxPopup(this));

		static SearchBox()
		{
			ImageProperty = DependencyProperty.Register("Image", typeof(ImageSource), typeof(SearchBox), new FrameworkPropertyMetadata(ImageResources.Search));
			PlaceholderProperty = DependencyProperty.Register("Placeholder", typeof(string), typeof(SearchBox), new FrameworkPropertyMetadata(InContextGuidanceResources.Widgets_SearchBox_Placeholder));
			TextProperty = DependencyProperty.Register("Text", typeof(string), typeof(SearchBox), new FrameworkPropertyMetadata(OnTextPropertyChanged));
			IsPopupSuppressedProperty = DependencyProperty.Register("IsPopupSuppressed", typeof(bool), typeof(SearchBox), new FrameworkPropertyMetadata(false, OnIsPopupSuppressedPropertyChanged));
			FrameworkElement.DefaultStyleKeyProperty.OverrideMetadata(typeof(SearchBox), new FrameworkPropertyMetadata(typeof(SearchBox)));
		}

		public override void OnApplyTemplate()
		{
			_textBox = GetTemplateChild("PART_TextBox") as TextBox;
			base.OnApplyTemplate();
		}

		protected virtual void OnTextChanged(string oldValue, string newValue)
		{
		}

		protected override void OnContextChanged(IndexedContent oldValue, IndexedContent newValue)
		{
			base.OnContextChanged(oldValue, newValue);
			UpdatePopup();
		}

		protected override async void OnPreviewMouseUp(MouseButtonEventArgs e)
		{
			base.OnPreviewMouseUp(e);
			if (base.Context == null)
			{
				await UpdateContextAsync();
			}
			InvalidatePopupStatus();
		}

		protected override void OnKeyDown(KeyEventArgs e)
		{
			if (e.Key == Key.Return)
			{
				NotifyEntered();
			}
			base.OnKeyDown(e);
		}

		private void NotifyEntered()
		{
			if (_processEnterOperation == null)
			{
				_processEnterOperation = base.Dispatcher.BeginInvoke((Action)delegate
				{
					base.Commands.ResolvedSearch.Execute(Text);
					_processEnterOperation = null;
				}, DispatcherPriority.Input);
			}
		}

		private void InvalidatePopupStatus()
		{
			if (_updatePopupOperation == null)
			{
				_updatePopupOperation = base.Dispatcher.BeginInvoke((Action)delegate
				{
					UpdatePopup();
					_updatePopupOperation = null;
				}, DispatcherPriority.Input);
			}
		}

		private void UpdatePopup()
		{
			if (CanShowPopup())
			{
				if (!InternalPopup.IsOpen)
				{
					InternalPopup.IsOpen = true;
				}
			}
			else if (InternalPopup.IsOpen)
			{
				InternalPopup.IsOpen = false;
			}
		}

		private bool CanShowPopup()
		{
			if (IsPopupSuppressed)
			{
				return false;
			}
			if (!string.IsNullOrWhiteSpace(Text))
			{
				return false;
			}
			if (base.Context == null)
			{
				return false;
			}
			if (_textBox == null)
			{
				return false;
			}
			if (!_textBox.IsKeyboardFocusWithin && !InternalPopup.IsKeyboardFocusWithin)
			{
				return false;
			}
			return true;
		}

		private static void OnTextPropertyChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
		{
			SearchBox obj = d as SearchBox;
			obj.OnTextChanged((string)e.OldValue, (string)e.NewValue);
			obj.UpdatePopup();
		}

		private static void OnIsPopupSuppressedPropertyChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
		{
			(d as SearchBox).UpdatePopup();
		}
	}

		private string CreateFeedbackUrl(string feedbackPrefix)
		{
			if (!string.IsNullOrWhiteSpace(feedbackPrefix))
			{
				string text = Uri.EscapeUriString(Owner.Context?.Id ?? Owner.ContextId);
				if (feedbackPrefix.Contains('?'))
				{
					return feedbackPrefix + "&ctxid=" + text;
				}
				return feedbackPrefix + "?ctxid=" + text;
			}
			return feedbackPrefix;
		}

		{

		public void UpdateUsability()
		{
			ResolvedIsEnabled = CalculateEnablement();
			ResolvedVisiblity = CalculateVisibility();
		}

		private Visibility CalculateVisibility()
		{
			if (!_owner.IsExchangeInitializedSuccessfully && WhenExchangeInitializationFailed == NegativeBehavior.Invisible)
			{
				return Visibility.Collapsed;
			}
			if (string.IsNullOrWhiteSpace(_owner.ContextId) && WhenContextIdIsNullOrEmpty == NegativeBehavior.Invisible)
			{
				return Visibility.Collapsed;
			}
			if (_owner.IsInContextGuidanceFeatureClosed && WhenInContxtGuidanceFeatureIsClosed == NegativeBehavior.Invisible)
			{
				return Visibility.Collapsed;
			}
			if (_owner.Visibility == Visibility.Visible && !string.IsNullOrWhiteSpace(_owner.ContextId) && !_owner.IsInContextGuidanceFeatureClosed && _owner.IsExchangeInitializedSuccessfully)
			{
				if (_owner.IsCfsAvailable != true && WhenCfsIsUnavailable == NegativeBehavior.Invisible)
				{
					return Visibility.Collapsed;
				}
				if (_owner.IsCfsAvailable == true && _owner.Context == null && WhenContextIsNotFound == NegativeBehavior.Invisible)
				{
					return Visibility.Collapsed;
				}
			}
			return Visibility.Visible;
		}

		private bool CalculateEnablement()
		{
			if (!_owner.IsInContextGuidanceFeatureClosed && _owner.IsExchangeInitializedSuccessfully)
			{
				return _owner.Context != null;
			}
			return false;
		}

		private async void InternalUpdateContextAsync()
		{
			await UpdateContextAsync();
			_updateOperation = null;
		}

		protected async Task UpdateContextAsync()
		{
			if (IsReadyToRequestContext())
			{
				CfsResult<IndexedContent> cfsResult = await CfsClient.GetIndexedContentAsync(ProductName, CfsProductVersion.V14, ContextId, ProductLanguage);
				IsCfsAvailable = !cfsResult.FailedToConnectServer;
				Context = cfsResult.Data;
			}
			else
			{
				Context = null;
			}
		}

		{

			{
				return false;
			}
			if (string.IsNullOrWhiteSpace(ContextId))
			{
				return false;
			}
			if (IsInContextGuidanceFeatureClosed)
			{
				return false;
			}
			if (!IsExchangeInitializedSuccessfully)
			{
				return false;
			}
			return true;
		}

		public static T GetAncestor<T>(DependencyObject descendant, DependencyObject upperLimit) where T : DependencyObject
		{
			if (descendant != null)
			{
				DependencyObject dependencyObject = descendant;
				while (dependencyObject != null && dependencyObject != upperLimit)
				{
					if (dependencyObject is T result)
					{
						return result;
					}
					DependencyObject dependencyObject2 = null;
					if (dependencyObject is Visual)
					{
						dependencyObject2 = VisualTreeHelper.GetParent(dependencyObject);
					}
					if (dependencyObject2 == null)
					{
						dependencyObject2 = LogicalTreeHelper.GetParent(dependencyObject);
					}
					dependencyObject = dependencyObject2;
				}
			}
			return null;
		}

		private readonly Action<object> _executeAction;

		private readonly Func<object, bool> _canExecuteFunc;

		public event EventHandler CanExecuteChanged;

		public DelegateCommand(Action<object> executeAction, Func<object, bool> canExecuteFunc = null)
		{
			_executeAction = executeAction ?? throw new ArgumentNullException("executeAction");
			if (canExecuteFunc != null)
			{
				_canExecuteFunc = canExecuteFunc;
				CommandManager.RequerySuggested += CommandManager_RequerySuggested;
			}
		}

		private void CommandManager_RequerySuggested(object sender, EventArgs e)
		{
			RaiseCanExecuteChanged();
		}

		public bool CanExecute(object parameter)
		{
			if (_canExecuteFunc != null)
			{
				return _canExecuteFunc(parameter);
			}
			return true;
		}

		public void Execute(object parameter)
		{
			if (CanExecute(parameter))
			{
				_executeAction(parameter);
			}
		}

		public void RaiseCanExecuteChanged()
		{
			this.CanExecuteChanged?.Invoke(this, EventArgs.Empty);
		}
	}
