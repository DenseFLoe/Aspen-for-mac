		List<string> IRibbonPlotWizardType.FieldDataOptionList(string sFieldName)
		{
			if (sFieldName.CompareTo(_lengthUnitFieldId) == 0)
			{
				List<string> list = new List<string>();
				_help.LoadXUnitsList(_context.Application.Tree, _wizardPlot, list);
				return list;
			}
			if (sFieldName.CompareTo(_lengthOrderFieldId) == 0)
			{
				return new List<string> { _ascending, _descending };
			}
			if (sFieldName.CompareTo(_lengthOnFieldId) == 0)
			{
				return new List<string> { _xAxis, _yAxis };
			}
			return null;
		}

		void IRibbonPlotWizardType.GetFieldStyle(string sFieldName, ref string sCurrentVal)
		{
			if (sFieldName.CompareTo(_lengthOrderFieldId) == 0)
			{
				if (_wizardPlot.PlotVal.ReverseXAxis)
				{
					sCurrentVal = _descending;
				}
				else
				{
					sCurrentVal = _ascending;
				}
			}
			else if (sFieldName.CompareTo(_lengthOnFieldId) == 0)
			{
				if (_wizardPlot.PlotVal.SwapAxises)
				{
					sCurrentVal = _yAxis;
				}
				else
				{
					sCurrentVal = _xAxis;
				}
			}
		}

		void IRibbonPlotWizardType.SetFieldStyle(string sFieldName, string sValue)
		{
			if (sFieldName.CompareTo(_lengthOrderFieldId) == 0)
			{
				if (sValue.CompareTo(_ascending) == 0)
				{
					_wizardPlot.PlotVal.ReverseXAxis = false;
				}
				else
				{
					_wizardPlot.PlotVal.ReverseXAxis = true;
				}
			}
			else if (sFieldName.CompareTo(_lengthOnFieldId) == 0)
			{
				if (sValue.CompareTo(_yAxis) == 0)
				{
					_wizardPlot.PlotVal.SwapAxises = true;
				}
				else
				{
					_wizardPlot.PlotVal.SwapAxises = false;
				}
			}
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = _context.FindNode("Output/COMP-ATTR");
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				ierror = 1;
			}
			else if (comboBoxSubstream.Items.Count <= 0)
			{
				comboBoxSubstream.Items.Clear();
				for (int i = 0; i < iHNode.Elements.Count; i++)
				{
					string newItem = iHNode.Elements.get_Label(0, i, (object)null);
					comboBoxSubstream.Items.Add(newItem);
				}
				if (string.IsNullOrEmpty(_currentConfigValue))
				{
					comboBoxSubstream.SelectedIndex = 0;
					_strSubstream = comboBoxSubstream.SelectedItem.ToString();
					SetComponent(_strSubstream);
					_strComponentID = comboBoxComponent.SelectedItem.ToString();
					SetAttribute(_strSubstream, _strComponentID);
					_strAttribute = comboBoxAttribute.SelectedItem.ToString();
					SetElement(_strSubstream, _strComponentID, _strAttribute);
				}
				else
				{
					RestoreConfigValue(_currentConfigValue);
				}
				_wizardPlot.SetIndependentVariable(_context, "Output/LENGTH_CAT");
			}
		}

		public void UpdateWizardData()
		{
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.SetIndependentVariable(_context, "Output/LENGTH_CAT");
			_wizardPlot.PlotVal.XAxisTitle = WpfPLRPlugResources.XAxisTitle_Length;
			((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle(Type.Missing, comboBoxAttribute.Text + " " + PlotPWRResource.RPlug_Comp_Attr_YAxisTitle);
			_wizardPlot.PlotVal.NVariables = 1;
			string text = comboBoxSubstream.Text;
			string text2 = comboBoxComponent.Text;
			string text3 = comboBoxAttribute.Text;
			string text4 = comboBoxElement.Text;
			int num = SetCompNum();
			IHNode iHNode = _context.FindNode("Output/LENGTH_CAT");
			IHNode iHNode2 = _context.FindNode("Output/COMP-ATTR");
			if (iHNode == null || iHNode2 == null)
			{
				return;
			}
			if (iHNode2.Elements.Count > 0)
			{
				IHNode iHNode3 = iHNode2.Elements[text, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[num, Type.Missing, Type.Missing, Type.Missing, Type.Missing];
				if (iHNode3 == null)
				{
					return;
				}
				int count = iHNode3.Elements.Count;
				_wizardPlot.PlotVal.NPoints = count;
				for (int i = 0; i < count; i++)
				{
					try
					{
						((IHPlotVal)_wizardPlot.PlotVal).set_XData(0, i, (short)1, (object)iHNode.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[i, Type.Missing, Type.Missing, Type.Missing, Type.Missing]);
						double num2 = ((double?)(dynamic)iHNode3.Elements[i, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing)) ?? 0.0;
						((IHPlotVal)_wizardPlot.PlotVal).set_YData(0, i, (short)0, (object)num2);
					}
					catch
					{
					}
				}
			}
			string label = WpfPLRPlugResources.Substream + text + "  " + WpfPLRPlugResources.CompID + " " + text2 + "  " + WpfPLRPlugResources.Attr + " " + text3 + "  " + WpfPLRPlugResources.Element + " " + text4;
			((IHPlotVal)_wizardPlot.PlotVal).set_Label(0, label);
			_help.SaveXUnitsComboText(_context.Application.Tree, _wizardPlot, _currentLengthUnit);
			_currentConfigValue = SaveConfigValue();
		}

			}
			int count = iHNode.Elements.Count;
			for (int i = 0; i < count; i++)

				string text = iHNode.Elements[i, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements.get_Label(0, 0, Type.Missing);

			stringBuilder.Append(";");
			stringBuilder.Append("Component=");

			stringBuilder.Append(";");
			stringBuilder.Append("Attribute=");

			stringBuilder.Append(";");

		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string text = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxSubstream.Text = text;
			_strSubstream = comboBoxSubstream.Text;
			SetComponent(_strSubstream);
			num++;
			text = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxComponent.Text = text;
			_strComponentID = comboBoxComponent.Text;
			SetAttribute(_strSubstream, _strComponentID);
			num++;
			text = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxAttribute.Text = text;
			_strAttribute = comboBoxAttribute.Text;
			SetElement(_strSubstream, _strComponentID, _strAttribute);
			num++;
			text = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxElement.Text = text;
		}

			{
				if (_wizardPlot.PlotVal.SwapAxises)

					sCurrentVal = _yAxis;

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			IHNodeCol curves = null;
			bool flag = false;
			for (int i = 0; i < _YVarNames.GetLength(0); i++)
			{
				iHNode = _context.FindNode(_YVarNames[i]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					flag = true;
					break;
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			for (int j = 0; j < _radBtnArr.Count(); j++)
			{
				iHNode = _context.FindNode(_YVarNames[j]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					_radBtnArr[j].IsEnabled = true;
				}
			}
			comboBoxSubstream.IsEnabled = false;
			string text = comboBoxSubstream.Text;
			for (int k = 0; k < _YVarNames.GetLength(0); k++)
			{
				if (!_radBtnArr[k].IsEnabled)
				{
					continue;
				}
				_radBtnArr[k].IsChecked = true;
				iHNode = _context.FindNode(_YVarNames[k]);
				if (iHNode == null || iHNode.Elements.Count <= 0)
				{
					break;
				}
				int count = iHNode.Elements.Count;
				if (count > 0)
				{
					comboBoxSubstream.Items.Clear();
					comboBoxSubstream.IsEnabled = true;
					for (int l = 0; l < count; l++)
					{
						string newItem = iHNode.Elements.get_Label(0, l, (object)null);
						comboBoxSubstream.Items.Add(newItem);
					}
					if (!string.IsNullOrEmpty(text))
					{
						comboBoxSubstream.SelectedItem = text;
					}
					else
					{
						comboBoxSubstream.SelectedIndex = 0;
					}
					string text2 = comboBoxSubstream.Text;
					curves = iHNode.Elements[text2, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
				}
				else
				{
					curves = iHNode.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
				}
				break;
			}
			SetCurves(curves);
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

			comboBoxSubstream.IsEnabled = false;
			IHNode iHNode = _context.FindNode(path);

				string text = comboBoxSubstream.Text;

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			int num = CurveList.GetSelectedCurves().Count();
			if (num == CurveList.Curves.Count)
			{
				CurveList.IsCheckAll = true;
			}
			else
			{
				string[] selectedCurves = CurveList.GetSelectedCurves();
				CurveList.IsCheckAll = false;
				CurveList.SetSelectedCurves(selectedCurves);
			}
			e.CanExecute = num != 0;
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("Substream=");
			stringBuilder.Append(comboBoxSubstream.Text);
			stringBuilder.Append(";");
			stringBuilder.Append("SubstreamEnable=");
			stringBuilder.Append(comboBoxSubstream.IsEnabled);
			stringBuilder.Append(";");
			stringBuilder.Append("CompSelection=");
			for (int i = 0; i < _radBtnArr.Count(); i++)
			{
				if (_radBtnArr[i].IsChecked == true)
				{
					stringBuilder.Append(i.ToString());
					break;
				}
			}
			stringBuilder.Append(";");
			int num = 0;
			stringBuilder.Append("CompButtonEnable=");
			for (int j = 0; j < _radBtnArr.Count(); j++)
			{
				if (_radBtnArr[j].IsEnabled)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(j.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CurveSelection=");
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			int count = list.Count;
			for (int k = 0; k < count; k++)
			{
				if (k > 0)
				{
					stringBuilder.Append(",");
				}
				stringBuilder.Append(list[k]);
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CheckAll=");
			stringBuilder.Append(CurveList.IsCheckAll);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		public void UpdateWizardData()
		{

			string text = comboBoxSubstream.SelectedItem.ToString();
			IHNode iHNode = null;
			string xAxisTitle = "";
			for (int i = 0; i < _radBtnArr.Count(); i++)
			{
				if (_radBtnArr[i].IsChecked == true)
				{
					iHNode = _context.FindNode(_XVarNames[i]);

					_help.SaveXUnitsCombo(_context.Application.Tree, _wizardPlot, _unitComboBox[i]);
					xAxisTitle = _strXTitle[i];
				}
			}
			_timeOrderFieldId = ((_radBtnArr[0].IsChecked == true) ? WpfPLRPlugResources.Time_order : WpfPLRPlugResources.Length_order);
			_timeOnFieldId = ((_radBtnArr[0].IsChecked == true) ? WpfPLRPlugResources.Time_on : WpfPLRPlugResources.Length_on);
			iHNode = iHNode.FindNode(text);
			int num = 0;
			if (iHNode != null)
			{
				num = iHNode.Elements.Count;
			}
			_wizardPlot.PlotVal.NPoints = num;
			_wizardPlot.PlotVal.XAxisTitle = xAxisTitle;
			int num2 = 0;
			for (int j = 0; j < _chkArr.Count(); j++)
			{
				if (_chkArr[j].IsChecked != true)
				{
					continue;
				}
				IHNode iHNode2 = _context.FindNode(_YVarNames[j]);
				_wizardPlot.PlotVal.NVariables = num2 + 1;
				string text2 = _chkArr[j].Content.ToString();

					text2 = text2 + " " + WpfPLRPlugResources.Substream + " " + text;
				}
				((IHPlotVal)_wizardPlot.PlotVal).set_Label(num2, text2);
				for (int k = 0; k < num; k++)
				{
					try
					{
						if (j == 0)
						{
							((IHPlotVal)_wizardPlot.PlotVal).set_YData(num2, k, (short)1, (object)iHNode2.Elements[text, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing]);

							((IHPlotVal)_wizardPlot.PlotVal).set_YData(num2, k, (short)1, (object)iHNode2.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing]);
						}
					}
					catch
					{
					}
				}
				num2++;
			}
			_help.SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentTempUnit);
			_currentConfigValue = SaveConfigValue();
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			bool flag = false;
			for (int i = 0; i < _chkArr.Count(); i++)
			{
				iHNode = _context.FindNode(_YVarNames[i]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					flag = true;
					_chkArr[i].IsEnabled = true;
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			for (int j = 0; j < _radBtnArr.Count(); j++)
			{
				iHNode = _context.FindNode(_XVarNames[j]);
				if (iHNode != null && iHNode.Elements.Count > 0)

					_radBtnArr[j].IsEnabled = true;
				}
			}
			if (_radBtnArr[0].IsEnabled)
			{
				_help.LoadXUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxResidenceTime);
			}
			else
			{
				_help.LoadXUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxLength);
			}
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				comboBoxSubstream.IsEnabled = false;
				comboBoxSubstream.Items.Clear();
				if (_chkArr[0].IsEnabled)
				{
					_chkArr[0].IsChecked = true;
					iHNode = _context.FindNode(_YVarNames[0]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						int count = iHNode.Elements.Count;
						comboBoxSubstream.IsEnabled = true;
						for (int k = 0; k < count; k++)
						{
							comboBoxSubstream.Items.Add(iHNode.Elements.get_Label(0, k, (object)null));
						}
						comboBoxSubstream.SelectedIndex = 0;
					}
				}
				if (_radBtnArr[0].IsEnabled)
				{
					_radBtnArr[0].IsChecked = true;
					comboBoxResidenceTime.IsEnabled = true;
					comboBoxLength.IsEnabled = false;
					_help.LoadXUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxResidenceTime);
				}
				else
				{
					comboBoxResidenceTime.IsEnabled = false;
					comboBoxLength.IsEnabled = true;
					_help.LoadXUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxLength);
				}
			}
			else
			{
				iHNode = _context.FindNode(_YVarNames[0]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					int count2 = iHNode.Elements.Count;

					for (int l = 0; l < count2; l++)

			}
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, _YVarNames[0]);
		}

			_timeOnFieldId = ((_radBtnArr[0].IsChecked == true) ? WpfPLRPlugResources.Time_on : WpfPLRPlugResources.Length_on);

			{
				num = iHNode.Elements.Count;
			}
			_wizardPlot.PlotVal.NPoints = num;
			_wizardPlot.PlotVal.XAxisTitle = xAxisTitle;

				if (_chkArr[j].IsChecked != true)

					continue;

				}
				((IHPlotVal)_wizardPlot.PlotVal).set_Label(num2, text2);
				for (int k = 0; k < num; k++)
