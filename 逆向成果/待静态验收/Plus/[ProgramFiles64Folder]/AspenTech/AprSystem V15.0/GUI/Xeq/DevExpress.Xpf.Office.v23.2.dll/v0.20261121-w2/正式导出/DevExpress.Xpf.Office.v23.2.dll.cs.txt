		void IDrawingEffectVisitor.Visit(BlurEffect drawingEffect)
		{
			if (parentFillBrush == null && parentOutlinePen == null)
			{
				return;
			}
			float blurRadius = GetBlurRadius(drawingEffect.Radius);
			if (blurRadius == 0f)
			{
				return;
			}
			bool grow = drawingEffect.Grow;
			BlurEffectSize = 1.7f * blurRadius;
			GlowCroppedByBlur = !grow;
			Rect boundsWithPenWidth = GetBoundsWithPenWidth();
			if (grow)
			{
				boundsWithPenWidth.Inflate(blurRadius, blurRadius);
			}
			DrawingVisual drawingVisual = new DrawingVisual();
			drawingVisual.Clip = new RectangleGeometry(boundsWithPenWidth);
			drawingVisual.Effect = new System.Windows.Media.Effects.BlurEffect
			{
				Radius = blurRadius
			};
			int count = Effects.Count;
			if (ParentIsRendered && count > 0)
			{
				DrawingVisual drawingVisual2 = Effects[count - 1];
				using (DrawingContext drawingContext = drawingVisual.RenderOpen())
				{
					drawingContext.DrawRectangle(System.Windows.Media.Brushes.Transparent, null, boundsWithPenWidth);
				}
				Effects.Remove(drawingVisual2);
				drawingVisual.Children.Add(drawingVisual2);
			}
			else
			{
				DrawParent(drawingVisual, boundsWithPenWidth);
			}
			Effects.Add(drawingVisual);
		}

		void IDrawingEffectVisitor.Visit(GlowEffect glowInfo)
		{
			if (parentFillBrush == null && parentOutlinePen == null)
			{
				return;
			}
			float blurRadius = GetBlurRadius(glowInfo.Radius);
			System.Windows.Media.Color glowColor = glowInfo.Color.FinalColor.ToWpfColor();
			if (blurRadius != 0f && glowColor.A != 0)
			{
				if (BlackAndWhitePrintMode)
				{
					glowColor = System.Windows.Media.Color.FromArgb(glowColor.A, byte.MaxValue, byte.MaxValue, byte.MaxValue);
				}
				Rect boundsWithPenWidth = GetBoundsWithPenWidth();
				double num = blurRadius + BlurEffectSize;
				boundsWithPenWidth.Inflate(num + GlowEffectSize, num + GlowEffectSize);
				num += GlowEffectSize * 1.8;
				num -= Math.Min(num, SoftEdgesSize / 4f);
				GlowEffectSize += blurRadius;
				DrawingVisual drawingVisual = new DrawingVisual();
				drawingVisual.Clip = GetGlowClipGeometry(blurRadius);
				using (DrawingContext drawingContext = drawingVisual.RenderOpen())
				{
					drawingContext.DrawRectangle(System.Windows.Media.Brushes.Transparent, null, boundsWithPenWidth);
					geometryInfo.DrawGlow(drawingContext, parentOutlinePen, parentFillBrush, glowColor, num);
				}
				DrawingVisual drawingVisual2 = new DrawingVisual();
				using (DrawingContext drawingContext2 = drawingVisual2.RenderOpen())
				{
					drawingContext2.DrawRectangle(CreateVisualContainerBrush(drawingVisual), null, boundsWithPenWidth);
				}
				drawingVisual2.Effect = new System.Windows.Media.Effects.BlurEffect
				{
					Radius = blurRadius / 2f
				};
				Effects.Insert(0, drawingVisual2);
			}
		}

		private Geometry GetGlowClipGeometry(float blurRadius)
		{
			if (GlowCroppedByBlur)
			{
				Rect bounds = Bounds;
				double num = ((double)blurRadius + penThickness) / 2.0;
				bounds.Inflate(num, num);
				return new RectangleGeometry(bounds);
			}
			return null;
		}

		void IDrawingEffectVisitor.Visit(InnerShadowEffect shadowInfo)
		{
			if (parentFillBrush == null || IsConnectorShape)
			{
				return;
			}
			System.Windows.Media.Color shadowColor = shadowInfo.Color.FinalColor.ToWpfColor();
			if (shadowColor.A == 0)
			{
				return;
			}
			if (BlackAndWhitePrintMode)
			{
				shadowColor = System.Windows.Media.Color.FromArgb(shadowColor.A, 0, 0, 0);
			}
			float blurRadius = GetBlurRadius(shadowInfo.BlurRadius);
			Geometry geometry = Geometry;
			Geometry geometry2 = CreateInnerShadowGeometry(shadowInfo.OffsetShadow, blurRadius);
			DrawingVisual drawingVisual = new DrawingVisual();
			DrawingVisual drawingVisual2 = new DrawingVisual();
			DrawParent(drawingVisual2);
			drawingVisual.Children.Add(drawingVisual2);
			InnerShadowBrushesInfo innerShadowBrushesInfo = CreateInnerShadowBrushes(shadowColor);
			if (innerShadowBrushesInfo.FillBrush != null)
			{
				if (innerShadowBrushesInfo.IsGradientFill)
				{
					drawingVisual2.OpacityMask = innerShadowBrushesInfo.FillBrush;
				}
				else
				{
					DrawingVisual drawingVisual3 = new DrawingVisual();
					using (DrawingContext dc = drawingVisual3.RenderOpen())
					{
						geometryInfo.Draw(dc, null, innerShadowBrushesInfo.FillBrush);
					}
					drawingVisual2.Children.Add(drawingVisual3);
				}
			}
			DrawingVisual drawingVisual4 = new DrawingVisual();
			drawingVisual4.Clip = geometry;
			using (DrawingContext drawingContext = drawingVisual4.RenderOpen())
			{
				drawingContext.DrawGeometry(innerShadowBrushesInfo.ShadowBrush, null, geometry2);
			}
			if (blurRadius > 0f)
			{
				drawingVisual4.Effect = new System.Windows.Media.Effects.BlurEffect
				{
					Radius = blurRadius
				};
			}
			drawingVisual2.Children.Add(drawingVisual4);
			if (parentOutlinePen != null)
			{
				DrawingVisual drawingVisual5 = new DrawingVisual();
				using (DrawingContext dc2 = drawingVisual5.RenderOpen())
				{
					geometryInfo.Draw(dc2, parentOutlinePen, null);
				}
				drawingVisual2.Children.Add(drawingVisual5);
			}
			Effects.Add(drawingVisual);
		}

		private Geometry CreateInnerShadowGeometry(OffsetShadowInfo shadowOffset, float blurRadius)
		{
			Rect bounds = Bounds;
			bounds.Inflate(blurRadius, blurRadius);
			Geometry geometry = new RectangleGeometry(bounds);
			float num = ToLayoutUnitConverter.ToLayoutUnits(UnitConverter.EmuToModelUnitsL(shadowOffset.Distance));
			double num2 = DrawingValueConverter.DegreeToRadian(DrawingValueConverter.FromPositiveFixedAngle(shadowOffset.Direction));
			float num3 = (float)((double)num * Math.Cos(num2));
			float num4 = (float)((double)num * Math.Sin(num2));
			Geometry geometry2 = Geometry.Clone();
			geometry2.Transform = new TranslateTransform(0f - num3, 0f - num4);
			return Geometry.Combine(geometry, geometry2, GeometryCombineMode.Exclude, Transform.Identity);
		}

		private InnerShadowBrushesInfo CreateInnerShadowBrushes(System.Windows.Media.Color shadowColor)
		{
			InnerShadowBrushesInfo innerShadowBrushesInfo = new InnerShadowBrushesInfo();
			if (parentFillBrush is SolidColorBrush)
			{
				byte a = ((SolidColorBrush)parentFillBrush).Color.A;
				PrepareSolidColorBrushes(innerShadowBrushesInfo, shadowColor, a);
			}
			else if (parentFillBrush is ImageBrush)
			{
				byte fillAlpha = Convert.ToByte(((ImageBrush)parentFillBrush).Opacity * 255.0);
				PrepareSolidColorBrushes(innerShadowBrushesInfo, shadowColor, fillAlpha);
			}
			else if (parentFillBrush is GradientBrush)
			{
				PrepareGradientBrushes(innerShadowBrushesInfo, (GradientBrush)parentFillBrush, shadowColor);
			}
			else
			{
				innerShadowBrushesInfo.ShadowBrush = CreateSolidColorBrush(shadowColor);
			}
			return innerShadowBrushesInfo;
		}

		private void PrepareSolidColorBrushes(InnerShadowBrushesInfo brushesInfo, System.Windows.Media.Color shadowColor, byte fillAlpha)
		{
			System.Windows.Media.Color color2;
			if (fillAlpha < byte.MaxValue)
			{
				System.Windows.Media.Color color = CreateFillLayerColor(shadowColor, fillAlpha);
				brushesInfo.FillBrush = CreateSolidColorBrush(color);
				color2 = CreateShadowLayerColor(shadowColor, fillAlpha, color.A);
			}
			else
			{
				color2 = shadowColor;
			}
			brushesInfo.ShadowBrush = CreateSolidColorBrush(color2);
		}

		private SolidColorBrush CreateSolidColorBrush(System.Windows.Media.Color color)
		{
			SolidColorBrush solidColorBrush = new SolidColorBrush(color);
			solidColorBrush.Freeze();
			return solidColorBrush;
		}

		private void PrepareGradientBrushes(InnerShadowBrushesInfo brushesInfo, GradientBrush parentFillBrush, System.Windows.Media.Color shadowColor)
		{
			System.Windows.Media.GradientStopCollection gradientStopCollection = new System.Windows.Media.GradientStopCollection();
			bool flag = false;
			int count = parentFillBrush.GradientStops.Count;
			for (int i = 0; i < count; i++)
			{
				GradientStop gradientStop = parentFillBrush.GradientStops[i];
				byte a = gradientStop.Color.A;
				if (a < byte.MaxValue)
				{
					flag = true;
				}
				gradientStopCollection.Add(new GradientStop(System.Windows.Media.Color.FromArgb(a, byte.MaxValue, byte.MaxValue, byte.MaxValue), gradientStop.Offset));
			}
			if (flag)
			{
				GradientBrush gradientBrush = parentFillBrush.Clone();
				gradientBrush.GradientStops = gradientStopCollection;
				gradientBrush.Freeze();
				brushesInfo.FillBrush = gradientBrush;
				brushesInfo.IsGradientFill = true;
			}
			brushesInfo.ShadowBrush = CreateSolidColorBrush(shadowColor);
		}

		private System.Windows.Media.Color CreateFillLayerColor(System.Windows.Media.Color shadowColor, byte fillAlpha)
		{
			byte a = Convert.ToByte(((float)(int)fillAlpha - (float)(fillAlpha * fillAlpha) / 255f) * ((float)(int)shadowColor.A / 255f));
			return System.Windows.Media.Color.FromArgb(a, shadowColor.R, shadowColor.G, shadowColor.B);
		}

		private System.Windows.Media.Color CreateShadowLayerColor(System.Windows.Media.Color shadowColor, byte fillAlpha, byte fillLayerAlpha)
		{
			byte b = Convert.ToByte(255f - (float)((255 - fillLayerAlpha) * (255 - fillAlpha)) / 255f);
			byte a = Convert.ToByte(255f - (float)(255 - fillAlpha) * (255f - (float)(int)fillAlpha * ((float)(int)shadowColor.A / 255f)) / (float)(255 - b));
			return System.Windows.Media.Color.FromArgb(a, shadowColor.R, shadowColor.G, shadowColor.B);
		}

		public DrawingVisual CalculateShadow(ContainerVisual parent)
		{
			if (shadow == null)
			{
				return null;
			}
			Rect transformedWidenedBounds = geometryInfo.GetTransformedWidenedBounds(parentOutlinePen, rotateTransform, 0.0);
			return CalculateShadowCore(parent, transformedWidenedBounds, rotateTransform);
		}

		public DrawingVisual CalculateGroupShadow(ContainerVisual parent, Rect parentBounds, Matrix rotateTransform)
		{
			if (shadow == null)
			{
				return null;
			}
			return CalculateShadowCore(parent, parentBounds, rotateTransform);
		}

		private DrawingVisual CalculateShadowCore(ContainerVisual parent, Rect transformedBounds, Matrix rotateTransform)
		{
			System.Windows.Media.Color color = shadow.Color.FinalColor.ToWpfColor();
			if (BlackAndWhitePrintMode)
			{
				color = System.Windows.Media.Color.FromArgb(color.A, 0, 0, 0);
			}
			SolidColorBrush solidColorBrush = new SolidColorBrush(color);
			solidColorBrush.Freeze();
			Rect descendantBounds = parent.DescendantBounds;
			Rect rectangle = descendantBounds;
			OuterShadowTransformationInfo info = CreateShadowTransformationInfo(shadow.Info, transformedBounds);
			DrawingVisual drawingVisual = new DrawingVisual();
			float blurRadius = GetBlurRadius(shadow.Info.BlurRadius);
			float num = Math.Min(Math.Abs(info.ScaleX), Math.Abs(info.ScaleY));
			if (blurRadius > 0f && num > 0f)
			{
				blurRadius /= num;
				drawingVisual.Effect = new System.Windows.Media.Effects.BlurEffect
				{
					Radius = blurRadius
				};
				rectangle.Inflate(blurRadius, blurRadius);
			}
			using (DrawingContext drawingContext = drawingVisual.RenderOpen())
			{
				drawingContext.DrawRectangle(System.Windows.Media.Brushes.Transparent, null, rectangle);
				drawingContext.DrawRectangle(solidColorBrush, null, descendantBounds);
			}
			drawingVisual.OpacityMask = CreateVisualContainerBrush(parent);
			drawingVisual.Transform = GetOuterShadowTransformation(info, rotateTransform);
			return drawingVisual;
		}

		private VisualBrush CreateVisualContainerBrush(ContainerVisual container)
		{
			VisualBrush visualBrush = new VisualBrush(container);
			visualBrush.Stretch = Stretch.None;
			visualBrush.Viewbox = container.DescendantBounds;
			visualBrush.ViewboxUnits = BrushMappingMode.Absolute;
			return visualBrush;
		}

		private OuterShadowTransformationInfo CreateShadowTransformationInfo(OuterShadowEffectInfo shadowInfo, Rect bounds)
		{
			OuterShadowTransformationCalculator outerShadowTransformationCalculator = new OuterShadowTransformationCalculator(UnitConverter, ToLayoutUnitConverter);
			return outerShadowTransformationCalculator.Calculate(bounds.ToRectangle(), shadowInfo);
		}

		private Transform GetOuterShadowTransformation(OuterShadowTransformationInfo info, Matrix rotateTransform)
		{
			Matrix matrix = new Matrix(info.ScaleX, info.SkewY, info.SkewX, info.ScaleY, info.OffsetX, info.OffsetY);
			if (!rotateTransform.IsIdentity)
			{
				matrix = Matrix.Multiply(rotateTransform, matrix);
				rotateTransform.Invert();
				matrix = Matrix.Multiply(matrix, rotateTransform);
			}
			return new MatrixTransform(matrix);
		}

		public DrawingVisual CalculateReflection(ContainerVisual parent)
		{
			if (reflectionInfo == null)
			{
				return null;
			}
			Rect transformedWidenedBounds = geometryInfo.GetTransformedWidenedBounds(parentOutlinePen, rotateTransform, 0.0);
			return CalculateReflectionCore(parent, Bounds, transformedWidenedBounds, rotateTransform);
		}

			if (reflectionInfo == null)

			}
			return CalculateReflectionCore(parent, groupBounds, widenedGroupGeometryBounds, rotateTransform);
		}

		private DrawingVisual CalculateReflectionCore(ContainerVisual parent, Rect shapeBounds, Rect widenedGeometryBounds, Matrix rotateTransform)
		{
			Rect descendantBounds = parent.DescendantBounds;
			Rect rectangle = descendantBounds;
			DrawingVisual drawingVisual = new DrawingVisual();
			float blurRadius = GetBlurRadius(reflectionInfo.OuterShadowEffectInfo.BlurRadius);
			if (blurRadius > 0f)
			{
				drawingVisual.Effect = new System.Windows.Media.Effects.BlurEffect
				{
					Radius = blurRadius
				};
				rectangle.Inflate(blurRadius, blurRadius);
			}
			VisualBrush brush = CreateVisualContainerBrush(parent);
			using (DrawingContext drawingContext = drawingVisual.RenderOpen())
			{
				drawingContext.DrawRectangle(System.Windows.Media.Brushes.Transparent, null, rectangle);
				drawingContext.DrawRectangle(brush, null, descendantBounds);
			}
			drawingVisual.OpacityMask = GetReflectionOpacityBrush(reflectionInfo.ReflectionOpacity, shapeBounds, rotateTransform);
			OuterShadowTransformationInfo info = CreateShadowTransformationInfo(reflectionInfo.OuterShadowEffectInfo, widenedGeometryBounds);
			drawingVisual.Transform = GetOuterShadowTransformation(info, rotateTransform);
			return drawingVisual;
		}

		private System.Windows.Media.Brush GetReflectionOpacityBrush(ReflectionOpacityInfo opacityInfo, Rect bounds, Matrix rotateTransform)
		{
			double offset = DrawingValueConverter.FromPercentage(opacityInfo.StartPosition);
			double offset2 = DrawingValueConverter.FromPercentage(opacityInfo.EndPosition);
			byte a = Convert.ToByte(DrawingValueConverter.FromPercentage(opacityInfo.StartOpacity) * 255.0);
			byte a2 = Convert.ToByte(DrawingValueConverter.FromPercentage(opacityInfo.EndOpacity) * 255.0);
			double num = DrawingValueConverter.FromPositiveFixedAngle(opacityInfo.FadeDirection);
			LinearGradientBrush linearGradientBrush = new LinearGradientBrush();
			linearGradientBrush.GradientStops.Add(new GradientStop(new System.Windows.Media.Color
			{
				A = a
			}, offset));
			linearGradientBrush.GradientStops.Add(new GradientStop(new System.Windows.Media.Color
			{
				A = a2
			}, offset2));
			rotateTransform.Invert();
			MatrixTransform matrixTransform = new MatrixTransform(rotateTransform);
			Rect rect = matrixTransform.TransformBounds(bounds);
			double height = bounds.Height;
			double width = bounds.Width;
			double num2 = (rect.Height - height) / (2.0 * height);
			double num3 = (rect.Width - width) / (2.0 * width);
			linearGradientBrush.StartPoint = new System.Windows.Point(0.5 + num3, 1.0 + num2);
			linearGradientBrush.EndPoint = new System.Windows.Point(0.5 - num3, 0.0 - num2);
			linearGradientBrush.Transform = matrixTransform;
			return linearGradientBrush;
		}

		void IDrawingEffectVisitor.Visit(SoftEdgeEffect softEdgesInfo)
		{
			float blurRadius = GetBlurRadius(softEdgesInfo.Radius);
			if (blurRadius != 0f)
			{
				SoftEdgesSize = blurRadius;
				Geometry widenedByPenGeometry = geometryInfo.GetWidenedByPenGeometry(parentOutlinePen);
				Geometry widenedPathGeometry = widenedByPenGeometry.GetWidenedPathGeometry(new System.Windows.Media.Pen(System.Windows.Media.Brushes.Black, 1.8 * (double)blurRadius));
				Geometry geometry = Geometry.Combine(widenedByPenGeometry, widenedPathGeometry, GeometryCombineMode.Exclude, Transform.Identity);
				DrawingVisual drawingVisual = new DrawingVisual();
				using (DrawingContext drawingContext = drawingVisual.RenderOpen())
				{
					drawingContext.DrawGeometry(System.Windows.Media.Brushes.Black, null, geometry);
				}
				drawingVisual.Effect = new System.Windows.Media.Effects.BlurEffect
				{
					Radius = blurRadius
				};
				VisualBrush visualBrush = new VisualBrush(drawingVisual);
				visualBrush.Stretch = Stretch.None;
				int count = Effects.Count;
				if (ParentIsRendered && count > 0)
				{
					DrawingVisual drawingVisual2 = Effects[count - 1];
					drawingVisual2.OpacityMask = visualBrush;
					return;
				}
				DrawingVisual drawingVisual3 = new DrawingVisual();
				drawingVisual3.OpacityMask = visualBrush;
				DrawParent(drawingVisual3);
				Effects.Add(drawingVisual3);
			}
		}

		public WPFShapeEffectRenderVisitor(IDocumentModel documentModel, IShapeGeometryInfo geometryInfo, System.Windows.Media.Pen outlinePen, Matrix rotateTransform)
		{
			Guard.ArgumentNotNull((object)documentModel, "documentModel");
			Guard.ArgumentNotNull((object)geometryInfo, "geometry");
			this.documentModel = documentModel;
			this.geometryInfo = geometryInfo;
			parentOutlinePen = outlinePen;
			penThickness = outlinePen?.Thickness ?? 0.0;
			this.rotateTransform = rotateTransform;
			Effects = new List<DrawingVisual>();
		}

		public void Clear()
		{
			Effects.Clear();
			reflectionInfo = null;
			shadow = null;
			IsConnectorShape = false;
			ParentIsRendered = false;
		}

		public Rect GetTransformedGeometryBounds(Matrix transform)
		{
			return geometryInfo.GetTransformedBounds(transform);
		}

		public Rect GetTransformedGeometryBoundsWithEffects(Matrix transform)
		{
			return geometryInfo.GetTransformedWidenedBounds(parentOutlinePen, transform, 2.0 * GlowEffectSize);
		}

		private float GetBlurRadius(long radius)
		{
			float val = ToLayoutUnitConverter.ToLayoutUnits(UnitConverter.EmuToModelUnitsL(radius));
			return Math.Max(0f, val);
		}

		private void DrawParent(DrawingVisual visual)
		{
			DrawParent(visual, Rect.Empty);
		}

		private void DrawParent(DrawingVisual visual, Rect bounds)
		{
			using (DrawingContext drawingContext = visual.RenderOpen())
			{
				drawingContext.DrawRectangle(System.Windows.Media.Brushes.Transparent, null, bounds);
				geometryInfo.Draw(drawingContext, parentOutlinePen, parentFillBrush);
			}
			ParentIsRendered = true;
		}

		private Rect GetBoundsWithPenWidth()
		{
			Rect bounds = Bounds;
			double num = penThickness / 2.0;
			bounds.Inflate(num, num);
			return bounds;
		}

		void IDrawingFillVisitor.Visit(DrawingGradientFill fill)
		{

		private System.Windows.Media.GradientStopCollection PrepareGradientStops(GradientStopCollection modelStops)
		{

			}
			System.Windows.Media.GradientStopCollection gradientStopCollection = new System.Windows.Media.GradientStopCollection();
			bool flag = false;
			foreach (DrawingGradientStop modelStop in modelStops)
			{

				gradientStopCollection.Add(new GradientStop(colorFromStyle, offset));

		}

		private System.Windows.Media.Brush CreateGradientBrush(DrawingGradientFill fill, System.Windows.Media.GradientStopCollection stops)
		{
			GradientType gradientType = fill.GradientType;
			Rect rect = bounds;
			if (geometry != null && (shapeType == ShapePreset.BlockArc || shapeType == ShapePreset.CircularArrow))
			{
				rect = geometry.Bounds;
			}
			switch (ShapeFillRenderVisitor.CorrectShapeGradientType(fill.GradientType, shapeType, OutlineFill))
			{
			case GradientType.Linear:
				return CreateLinearGradientBrush(rect, fill, stops);
			case GradientType.Circle:
				return CreateRadialGradientBrush(rect, fill, stops);
			case GradientType.Rectangle:
			{
				RectangleGradientBrushCalculator rectangleGradientBrushCalculator = new RectangleGradientBrushCalculator(rect, fill, stops, shapeType);
				return rectangleGradientBrushCalculator.CreateBrush();
			}
			case GradientType.Shape:
				if (geometry != null)
				{
					rect = ChooseBoundsForShapeGradients(shapeType);
					PathGradientBrushCalculator pathGradientBrushCalculator = new PathGradientBrushCalculator(geometry, rect, fill, stops, shapeType);
					return pathGradientBrushCalculator.CreateBrush();
				}
				break;
			}
			return null;
		}

		}

		private System.Windows.Media.Brush CreateLinearGradientBrush(Rect bounds, DrawingGradientFill fill, System.Windows.Media.GradientStopCollection stops)
		{
			bounds.Offset(0.0 - bounds.X, 0.0 - bounds.Y);
			GradientTileInfo tileInfo = new GradientTileInfo(bounds, fill.TileRect);
			if (tileInfo.Width == 0.0 || tileInfo.Height == 0.0)
			{
				return new SolidColorBrush(stops[0].Color);
			}
			LinearGradientBrush linearGradientBrush = new LinearGradientBrush();
			linearGradientBrush.SpreadMethod = GradientSpreadMethod.Reflect;
			float num = fill.DocumentModel.UnitConverter.ModelUnitsToDegreeF(fill.Angle);
			num %= 360f;
			System.Windows.Point[] array = CalculateGradientPoints(bounds, tileInfo, num, fill.Scaled);
			linearGradientBrush.StartPoint = array[0];
			linearGradientBrush.EndPoint = array[1];
			linearGradientBrush.GradientStops = stops;
			return linearGradientBrush;
		}

		private System.Windows.Media.Brush CreateRadialGradientBrush(Rect bounds, DrawingGradientFill fill, System.Windows.Media.GradientStopCollection stops)
		{
			GradientTileInfo gradientTileInfo = new GradientTileInfo(bounds, fill.TileRect);
			double width = gradientTileInfo.Width;
			double height = gradientTileInfo.Height;
			if (width == 0.0 && height == 0.0)
			{
				return new SolidColorBrush(stops[0].Color);
			}
			RadialGradientBrush radialGradientBrush = new RadialGradientBrush();
			radialGradientBrush.MappingMode = BrushMappingMode.Absolute;
			radialGradientBrush.RadiusX = gradientTileInfo.Radius;
			radialGradientBrush.RadiusY = gradientTileInfo.Radius;
			radialGradientBrush.Center = gradientTileInfo.Center;
			radialGradientBrush.GradientOrigin = new GradientTileInfo(bounds, fill.FillRect).Center;
			radialGradientBrush.GradientStops = stops;
			return radialGradientBrush;
		}

		private System.Windows.Point[] CalculateGradientPoints(Rect bounds, GradientTileInfo tileInfo, double gradientAngle, bool scaled)
		{
			double width = tileInfo.Width;
			double height = tileInfo.Height;
			if (scaled && width != height)
			{
				gradientAngle = ScaleAngle(gradientAngle, width, height);
			}
			bool flag = gradientAngle % 180.0 <= 90.0;
			double num = DrawingValueConverter.RadianToDegree(Math.Atan2(height, width));
			double degree = (flag ? (gradientAngle + 90.0 - num) : (gradientAngle - 90.0 + num));
			double num2 = tileInfo.Radius * Math.Sin(DrawingValueConverter.DegreeToRadian(degree));
			degree = (flag ? gradientAngle : (180.0 - gradientAngle));
			double num3 = num2 / width * Math.Cos(DrawingValueConverter.DegreeToRadian(degree));
			double num4 = num2 / height * Math.Sin(DrawingValueConverter.DegreeToRadian(degree));
			System.Windows.Point center = CalculateCenterPoint(tileInfo, bounds);
			double scaleX = width / bounds.Width;
			double scaleY = height / bounds.Height;
			System.Windows.Point point = CreateGradientPoint(center, flag ? (0.0 - num3) : num3, 0.0 - num4, scaleX, scaleY);
			System.Windows.Point point2 = CreateGradientPoint(center, flag ? num3 : (0.0 - num3), num4, scaleX, scaleY);
			if (gradientAngle < 180.0)
			{
				return new System.Windows.Point[2] { point, point2 };
			}
			return new System.Windows.Point[2] { point2, point };
		}

		private double ScaleAngle(double angle, double width, double height)
		{
			int num = 0;
			while (angle < 0.0)
			{
				angle += 90.0;
				num--;
			}
			while (angle > 90.0)
			{
				angle -= 90.0;
				num++;
			}
			double num2 = Math.Min(width, height);
			double num3 = num2 / width;
			double num4 = num2 / height;
			double num5 = Math.Tan(DrawingValueConverter.DegreeToRadian(angle));
			num5 *= ((num % 2 == 0) ? (num4 / num3) : (num3 / num4));
			double num6 = DrawingValueConverter.RadianToDegree(Math.Atan(num5));
			num6 += (double)(num * 90);
			return num6 % 360.0;
		}

		private System.Windows.Point CalculateCenterPoint(GradientTileInfo tileInfo, Rect bounds)
		{
			System.Windows.Point point = new System.Windows.Point(bounds.Width / 2.0, bounds.Height / 2.0);
			System.Windows.Point center = tileInfo.Center;
			double x = 0.5 + center.X / tileInfo.Width - point.X / bounds.Width;
			double y = 0.5 + center.Y / tileInfo.Height - point.Y / bounds.Height;
			return new System.Windows.Point(x, y);
		}

		{
			return new System.Windows.Point((center.X + dx) * scaleX, (center.Y + dy) * scaleY);
		}

		private bool IsIdentityMatrix(float[][] colorMatrix)

				for (int j = 0; j < 5; j++)

					if ((i == j && num != 1f) || (i != j && num != 0f))
					{
						return false;
					}
				}
			}
			return true;
		}

		private TransformGroup CreateStretchedBlipFillTransform(DrawingBlipFill fill)
		{
			TransformGroup transformGroup = new TransformGroup();
			RectangleOffset fillRectangle = fill.FillRectangle;
			double width = bounds.Width;
			double height = bounds.Height;
			double num = width * DrawingValueConverter.FromPercentage(fillRectangle.LeftOffset);
			double num2 = height * DrawingValueConverter.FromPercentage(fillRectangle.TopOffset);
			double num3 = width * DrawingValueConverter.FromPercentage(fillRectangle.RightOffset);
			double num4 = height * DrawingValueConverter.FromPercentage(fillRectangle.BottomOffset);
			double num5 = width - num - num3;
			double num6 = height - num2 - num4;
			if (num5 != width || num6 != height)
			{
				transformGroup.Children.Add(new ScaleTransform(num5 / width, num6 / height));
			}
			if (num != 0.0 || num2 != 0.0)
			{
				transformGroup.Children.Add(new TranslateTransform(num, num2));
			}
			if (!fill.RotateWithShape)
			{
				transformGroup.Children.Add(CreateNotRotatedScaleTransform());
			}
			return transformGroup;
		}

		private Transform CreateNotRotatedScaleTransform()
		{
			System.Windows.Point point = new System.Windows.Point(bounds.Width / 2.0, bounds.Height / 2.0);

			transformGroup.Children.Add(new ScaleTransform(rect.Width / bounds.Width, rect.Height / bounds.Height, point.X, point.Y));
			transformGroup.Children.Add(new RotateTransform(0f - shapeRotationAngle, point.X, point.Y));
			return transformGroup;
		}

				break;
			case RectangleAlignType.BottomRight:

				y = 0.0 - boundedRect.Y;

		}

		private System.Windows.Point CalculateAlignmentOffset(RectangleAlignType alignment, double width, double height, double imageWidth, double imageHeight)
		{
			double x = 0.0;

			switch (alignment)

			case RectangleAlignType.Top:

			case RectangleAlignType.TopRight:
				x = CalculateFarOffset(width, imageWidth);
				break;
			case RectangleAlignType.Left:
				y = CalculateCenterOffset(height, imageHeight);
				break;
			case RectangleAlignType.Center:
				x = CalculateCenterOffset(width, imageWidth);
				y = CalculateCenterOffset(height, imageHeight);
				break;
			case RectangleAlignType.Right:
				x = CalculateFarOffset(width, imageWidth);
				y = CalculateCenterOffset(height, imageHeight);
				break;
			case RectangleAlignType.BottomLeft:
				y = CalculateFarOffset(height, imageHeight);
				break;
			case RectangleAlignType.Bottom:
				x = CalculateCenterOffset(width, imageWidth);
				y = CalculateFarOffset(height, imageHeight);
				break;
			case RectangleAlignType.BottomRight:
				x = CalculateFarOffset(width, imageWidth);
				y = CalculateFarOffset(height, imageHeight);
				break;
			}
			return new System.Windows.Point(x, y);
		}

				System.Windows.Point point = CalculateNormalPoint(InnerPoint1, OuterPoint1.X, OuterPoint1.Y, OuterPoint2.X, OuterPoint2.Y);
				return new System.Windows.Point[2] { InnerPoint1, point };
			}

			private System.Windows.Point CalculateNormalPoint(System.Windows.Point center, double x1, double y1, double x2, double y2)
			{
				double num;
				double y3;
				if (Math.Abs(x2 - x1) < 9.999999960041972E-13)
				{
					num = x1;
					y3 = center.Y;
				}
				else if (Math.Abs(y2 - y1) < 9.999999960041972E-13)
				{
					num = center.X;
					y3 = y1;
				}
				else
				{
					double num2 = (y2 - y1) / (x2 - x1);
					double num3 = y2 - num2 * x2;
					double num4 = -1.0 / num2;
					double num5 = center.Y - num4 * center.X;
					num = (num3 - num5) / (num4 - num2);
					y3 = num4 * num + num5;
				}
				return new System.Windows.Point(num, y3);
			}

	public class PathGradientBrushCalculator : PathGradientBrushCalculatorBase
	{
		private readonly PathGeometry geometry;

		public PathGradientBrushCalculator(PathGeometry geometry, Rect bounds, DrawingGradientFill fill, System.Windows.Media.GradientStopCollection stops, ShapePreset shapeType)
			: base(bounds, fill, stops, shapeType)
		{
			Guard.ArgumentNotNull((object)geometry, "geometry");
			this.geometry = geometry;
		}

		protected override PathGeometry CreateInnerGeometry(RectangleOffset tileRect, RectangleOffset fillRect)
		{
			PathGeometry pathGeometry = geometry.Clone();
			System.Windows.Point[] innerPoints = new System.Windows.Point[2] { base.Center, base.Center };
			PathFigureCollection figures = pathGeometry.Figures;
			for (int num = figures.Count - 1; num >= 0; num--)
			{
				PathFigure pathFigure = figures[num];
				System.Windows.Point start = pathFigure.StartPoint;
				PathSegmentCollection segments = pathFigure.Segments;
				for (int i = 0; i < segments.Count; i++)
				{
					if (segments[i] is LineSegment { Point: var point } lineSegment)
					{
						ProcessLineSegment(innerPoints, start, point, pathFigure, lineSegment, i);
						start = point;
					}
					else if (segments[i] is BezierSegment { Point3: var point2 } bezierSegment)
					{
						ProcessBezierSegment(innerPoints, start, point2, pathFigure, bezierSegment, i);
						start = point2;
					}
				}
			}
			return pathGeometry;
		}

		private void ProcessLineSegment(System.Windows.Point[] innerPoints, System.Windows.Point start, System.Windows.Point end, PathFigure figure, LineSegment lineSegment, int index)
		{
			if (index == 0)
			{
				figure.StartPoint = innerPoints[0];
			}
			lineSegment.Point = innerPoints[1];
			lineSegment.IsStroked = true;
			if (!start.Equals(end))
			{
				AddTrapeze(start, end, innerPoints[0], innerPoints[1]);
			}
		}

		private void ProcessBezierSegment(System.Windows.Point[] innerPoints, System.Windows.Point start, System.Windows.Point end, PathFigure figure, BezierSegment bezierSegment, int index)
		{
			if (index == 0)
			{
				figure.StartPoint = innerPoints[0];
			}
			if (!start.Equals(end))
			{
				Vector[] bezierPoints = WpfDrawingShapeHelper.GetBezierPoints(start, bezierSegment.Point1, bezierSegment.Point2, bezierSegment.Point3);
				ConvertBezierCurveToTrapezes(bezierPoints, innerPoints);
			}
			bezierSegment.Point1 = innerPoints[0];
			bezierSegment.Point2 = innerPoints[0];
			bezierSegment.Point3 = innerPoints[1];
		}

		private void ConvertBezierCurveToTrapezes(Vector[] p, System.Windows.Point[] innerPoints)
		{
			double num = WpfDrawingShapeHelper.CalcBezierLength(p);
			int num2 = (int)Math.Max(10.0, num / 20.0);
			System.Windows.Point outerPoint = new System.Windows.Point(p[0].X, p[0].Y);
			for (int i = 0; i < num2; i++)
			{
				System.Windows.Point point = WpfDrawingShapeHelper.CalcBezier(p, ((double)i + 1.0) / (double)num2);
				AddTrapeze(outerPoint, point, innerPoints[0], innerPoints[1]);
				outerPoint = point;
			}
		}

			if (widenPen != null)
			{
				Geometry widenedPathGeometry = geometry.GetWidenedPathGeometry(widenPen);
				return Geometry.Combine(geometry, widenedPathGeometry, GeometryCombineMode.Union, Transform.Identity);
			}
			return geometry;
		}

		{
			System.Windows.Media.Pen pen = null;
			if (parentOutlinePen != null)
			{
				pen = parentOutlinePen.Clone();
				pen.Thickness += additionalSize;
				PrepareLineCapsAndFreeze(pen);
			}
			else if (additionalSize > 0.0)
			{
				pen = new System.Windows.Media.Pen(System.Windows.Media.Brushes.Black, additionalSize);
				PrepareLineCapsAndFreeze(pen);
			}
			return pen;
		}

		}

		{

		private System.Windows.Media.Pen CreateGlowPenCore(System.Windows.Media.Color glowColor, System.Windows.Media.Brush parentBrush, double thickness)
		{
			System.Windows.Media.Brush brush = ChooseGlowBrush(parentBrush, glowColor);
			System.Windows.Media.Pen pen = new System.Windows.Media.Pen(brush, thickness);
			pen.LineJoin = PenLineJoin.Round;
			return pen;
		}

		protected System.Windows.Media.Pen CreateGlowPen(System.Windows.Media.Color glowColor, System.Windows.Media.Brush parentBrush, double blurRadius)
		{
			System.Windows.Media.Pen pen = CreateGlowPenCore(glowColor, parentBrush, blurRadius);
			PrepareLineCapsAndFreeze(pen);
			return pen;
		}

		private System.Windows.Media.Pen CreateGlowPen(System.Windows.Media.Color glowColor, System.Windows.Media.Pen originalPen, double blurRadius)
		{
			double thickness = originalPen.Thickness;
			System.Windows.Media.Pen pen = CreateGlowPenCore(glowColor, originalPen.Brush, thickness);
			pen.DashCap = originalPen.DashCap;
			pen.DashStyle = GetDashStyle(originalPen.DashStyle.Clone(), thickness, blurRadius);
			pen.StartLineCap = GetCorrectedPenLineCap(originalPen.StartLineCap);
			pen.EndLineCap = GetCorrectedPenLineCap(originalPen.EndLineCap);
			pen.Thickness += blurRadius;
			if (pen.CanFreeze)
			{
				pen.Freeze();
			}
			return pen;
		}

		private DashStyle GetDashStyle(DashStyle dashStyle, double oldThickness, double blurRadius)
		{
			DoubleCollection dashes = dashStyle.Dashes;
			int count = dashes.Count;
			if (count > 0)
			{
				double num = oldThickness / (oldThickness + blurRadius);
				for (int i = 0; i < count; i++)
				{
					dashes[i] *= num;
				}
				dashStyle.Offset *= num;
			}
			return dashStyle;
		}

		private PenLineCap GetCorrectedPenLineCap(PenLineCap originalLineCap)
		{
			if (originalLineCap == PenLineCap.Square)
			{
				return originalLineCap;
			}
			return PenLineCap.Round;
		}

		private void PrepareLineCapsAndFreeze(System.Windows.Media.Pen pen)
		{
			pen.StartLineCap = PenLineCap.Round;
			pen.EndLineCap = PenLineCap.Round;
			if (pen.CanFreeze)
			{
				pen.Freeze();
			}
		}

		private System.Windows.Media.Brush ChooseGlowBrush(System.Windows.Media.Brush parentBrush, System.Windows.Media.Color glowColor)
		{
			if (parentBrush is GradientBrush)
			{
				System.Windows.Media.Brush brush = ConvertGradientBrushWithTransparentColors((GradientBrush)parentBrush, glowColor);
				if (brush != null)
				{
					return brush;
				}
			}
			SolidColorBrush solidColorBrush = new SolidColorBrush(glowColor);
			solidColorBrush.Freeze();
			return solidColorBrush;
		}

		private GradientBrush ConvertGradientBrushWithTransparentColors(GradientBrush brush, System.Windows.Media.Color glowColor)
		{
			System.Windows.Media.GradientStopCollection gradientStops = brush.GradientStops;
			System.Windows.Media.GradientStopCollection gradientStopCollection = new System.Windows.Media.GradientStopCollection();
			bool flag = false;
			bool flag2 = false;
			bool flag3 = false;
			for (int i = 0; i < gradientStops.Count; i++)
			{
				GradientStop gradientStop = gradientStops[i];
				System.Windows.Media.Color color = gradientStop.Color;
				double offset = gradientStop.Offset;
				flag2 = color.A == 0;
				if (flag2)
				{
					flag3 = true;
				}
				if (!flag2 && flag)
				{
					gradientStopCollection.Add(new GradientStop(glowColor, gradientStops[i - 1].Offset));
				}
				else if (!flag && flag2 && i != 0)
				{
					gradientStopCollection.Add(new GradientStop(glowColor, offset));
				}
				gradientStopCollection.Add(new GradientStop(flag2 ? color : glowColor, offset));
				flag = flag2;
			}
			GradientBrush gradientBrush = null;
			if (flag3)
			{
				gradientBrush = brush.Clone();
				gradientBrush.GradientStops = gradientStopCollection;
				gradientBrush.Freeze();
			}
			return gradientBrush;
		}

		public static System.Windows.Media.Brush CalculateFillBrush(ShapePropertiesBase shapeProperties, ShapePreset shapeType, ShapeStyle shapeStyle, Rect bounds, float shapeRotationAngle, PathGeometry geometry)

			{
				fill = shapeProperties.DocumentModel.OfficeTheme.FormatScheme.GetFill((StyleMatrixElementType)Math.Min(3, shapeStyle.FillReferenceIdx));
				styleColor = shapeStyle.FillColor.FinalColor.ToWpfColor();
			}
			WPFShapeFillRenderVisitor wPFShapeFillRenderVisitor = new WPFShapeFillRenderVisitor(styleColor, defaultColor, bounds, shapeType, shapeRotationAngle, geometry);
			fill.Visit(wPFShapeFillRenderVisitor);
			return wPFShapeFillRenderVisitor.Result;
		}

		public static System.Windows.Media.Pen CalculateLinePen(Outline outline, ShapeStyle shapeStyle, Rect bounds, ShapePreset shapeType)
		{
			return CalculateLinePen(outline, shapeStyle, bounds, shapeType, hasHeadArrow: false, hasTailArrow: false);
		}

		public static System.Windows.Media.Pen CalculateLinePen(Outline outline, ShapeStyle shapeStyle, Rect bounds, ShapePreset shapeType, bool hasHeadArrow, bool hasTailArrow)
		{
			System.Windows.Media.Brush brush = CalculateLineBrush(outline, shapeStyle, bounds, shapeType);
			if (brush == null)
			{
				return null;
			}
			double thickness = CalculatePenWidth(outline, shapeStyle);
			System.Windows.Media.Pen pen = new System.Windows.Media.Pen(brush, thickness);
			pen.DashStyle = GetPenDashStyle(outline.Dashing);
			pen.StartLineCap = GetPenLineCap(outline.EndCapStyle, hasHeadArrow);
			pen.EndLineCap = GetPenLineCap(outline.EndCapStyle, hasTailArrow);
			pen.DashCap = ((pen.EndLineCap != PenLineCap.Round) ? PenLineCap.Square : PenLineCap.Round);
			SetupLineJoinStyle(pen, outline.JoinStyle, outline.MiterLimit, shapeType);
			if (pen.CanFreeze)
			{
				pen.Freeze();
			}
			return pen;
		}

		public static System.Windows.Media.Brush CalculateLineBrush(Outline outline, ShapeStyle shapeStyle, Rect bounds, ShapePreset shapeType)
		{
			IDrawingFill fill = outline.Fill;
			System.Windows.Media.Color? defaultColor = null;
			System.Windows.Media.Color color;
			if (fill.FillType == DrawingFillType.Automatic)
			{
				color = shapeStyle.LineColor.FinalColor.ToWpfColor();
				defaultColor = color;
			}
			else
			{
				color = System.Drawing.Color.Empty.ToWpfColor();
			}
			if (outline.IsDefault || outline.Width == 0)
			{
				if (shapeStyle.LineReferenceIdx == 0 && outline.IsDefault)
				{
					return null;
				}
				if (fill.FillType == DrawingFillType.Automatic)
				{
					Outline styleOutline = GetStyleOutline(shapeStyle);
					fill = styleOutline.Fill;
				}
			}
			WPFShapeFillRenderVisitor wPFShapeFillRenderVisitor = new WPFShapeFillRenderVisitor(color, defaultColor, bounds, shapeType, 0f);
			wPFShapeFillRenderVisitor.OutlineFill = true;
			fill.Visit(wPFShapeFillRenderVisitor);
			return wPFShapeFillRenderVisitor.Result;
		}

		public static double CalculatePenWidth(Outline outline, ShapeStyle shapeStyle)
		{
			float value = outline.Width;
			if (outline.IsDefault || outline.Width == 0)
			{
				if (shapeStyle.LineReferenceIdx == 0 && outline.IsDefault)
				{
					return 0.0;
				}
				Outline styleOutline = GetStyleOutline(shapeStyle);
				value = styleOutline.Width;
			}
			double a = Math.Max(1f, shapeStyle.DocumentModel.ToDocumentLayoutUnitConverter.ToLayoutUnits(value));
			return Math.Round(a);
		}

		private static Outline GetStyleOutline(ShapeStyle shapeStyle)
		{
			IDocumentModel documentModel = shapeStyle.DocumentModel;
			int type = Math.Min(3, Math.Max(1, shapeStyle.LineReferenceIdx));
			return documentModel.OfficeTheme.FormatScheme.GetOutline((StyleMatrixElementType)type);
		}

		private static DashStyle GetPenDashStyle(OutlineDashing dashing)
		{
			switch (dashing)
			{
			case OutlineDashing.Solid:
				return DashStyles.Solid;
			case OutlineDashing.Dash:
				return new DashStyle(new double[2] { 3.0, 4.0 }, 0.0);
			case OutlineDashing.LongDash:
				return new DashStyle(new double[2] { 7.0, 4.0 }, 0.0);
			case OutlineDashing.SystemDash:
				return DashStyles.Dash;
			case OutlineDashing.DashDot:
				return new DashStyle(new double[4] { 3.0, 4.0, 0.20000000298023224, 4.0 }, 0.0);
			case OutlineDashing.LongDashDot:
				return new DashStyle(new double[4] { 7.0, 4.0, 0.20000000298023224, 4.0 }, 0.0);
			case OutlineDashing.SystemDashDot:
				return DashStyles.DashDot;
			case OutlineDashing.Dot:
				return new DashStyle(new double[2] { 1.0, 3.0 }, 0.0);
			case OutlineDashing.SystemDot:
				return DashStyles.Dot;
			case OutlineDashing.SystemDashDotDot:
				return DashStyles.DashDotDot;
			case OutlineDashing.LongDashDotDot:
				return new DashStyle(new double[6] { 7.0, 4.0, 0.20000000298023224, 4.0, 0.20000000298023224, 4.0 }, 0.0);
			default:
				Exceptions.ThrowInternalException();
				return DashStyles.Solid;
			}
		}

		private static PenLineCap GetPenLineCap(OutlineEndCapStyle endCapStyle, bool hasArrow)
		{
			if (hasArrow)
			{
				return PenLineCap.Round;
			}
			switch (endCapStyle)
			{
			case OutlineEndCapStyle.Round:
				return PenLineCap.Round;
			case OutlineEndCapStyle.Square:
				return PenLineCap.Square;
			case OutlineEndCapStyle.Flat:
				return PenLineCap.Flat;
			default:
				Exceptions.ThrowInternalException();
				return PenLineCap.Triangle;
			}
		}

		private static void SetupLineJoinStyle(System.Windows.Media.Pen pen, LineJoinStyle joinStyle, int miterLimit, ShapePreset shapeType)
		{
			if (joinStyle == LineJoinStyle.Round && ShapeRenderHelper.IsMiteredLineJoinShape(shapeType))
			{
				joinStyle = LineJoinStyle.Miter;
			}
			switch (joinStyle)
			{
			case LineJoinStyle.Bevel:
				pen.LineJoin = PenLineJoin.Bevel;
				break;
			case LineJoinStyle.Miter:
				pen.LineJoin = PenLineJoin.Miter;
				pen.MiterLimit = DrawingValueConverter.FromPercentage(miterLimit);
				break;
			case LineJoinStyle.Round:
				pen.LineJoin = PenLineJoin.Round;
				break;
			}
		}

		public static bool HasArrows(ShapeProperties shapeProperties)
		{
			if (!shapeProperties.IsConnectionShape() && !NonConnectorWithArrows(shapeProperties.ShapeType))
			{
				return false;
			}
			if (shapeProperties.Outline.HeadType == OutlineHeadTailType.None)
			{
				return shapeProperties.Outline.TailType != OutlineHeadTailType.None;
			}
			return true;
		}

		private static bool NonConnectorWithArrows(ShapePreset shapeType)
		{
			switch (shapeType)
			{
			case ShapePreset.None:
			case ShapePreset.Arc:
			case ShapePreset.LeftBracket:
			case ShapePreset.RightBracket:
			case ShapePreset.LeftBrace:
			case ShapePreset.RightBrace:
			case ShapePreset.BracketPair:
			case ShapePreset.BracePair:
				return true;
			default:
				return false;
			}
		}

		internal static Vector[] GetBezierPoints(System.Windows.Point p0, System.Windows.Point p1, System.Windows.Point p2, System.Windows.Point p3)
		{
			return new Vector[4]
			{
				new Vector(p0.X, p0.Y),
				new Vector(p1.X, p1.Y),
				new Vector(p2.X, p2.Y),
				new Vector(p3.X, p3.Y)
			};
		}

		internal static System.Windows.Point CalcBezier(Vector[] p, double t)
		{
			double num = t * t;
			double num2 = 1.0 - t;
			double num3 = num2 * num2;
			Vector vector = num2 * num3 * p[0] + 3.0 * t * num3 * p[1] + 3.0 * num * num2 * p[2] + t * num * p[3];
			return new System.Windows.Point(vector.X, vector.Y);
		}

		internal static double CalcBezierLength(Vector[] p)
		{
			double num = 0.0;
			System.Windows.Point point = new System.Windows.Point(p[0].X, p[0].Y);
			System.Windows.Point point2;
			for (int i = 0; i < 9; i++)
			{
				point2 = CalcBezier(p, (double)(i + 1) * 0.1);
				num += (point2 - point).Length;
				point = point2;
			}
			point2 = new System.Windows.Point(p[3].X, p[3].Y);
			return num + (point2 - point).Length;
		}
	}
	public class WpfCustomGeometryToPathGeometryConverter : IPathInstructionWalker
	{
		private PathGeometry pathGeometry;

		private PathFigure currentFigure;

		private Rect rect;

		private ModelShapePath path;

		private bool isFilled;

		private float defaultScale;

		private float scaleX;

		private float scaleY;

		private double left;

		private double top;

		private double lastX;

		private double lastY;

		public PathGeometry PathGeometry => pathGeometry;

		protected PathFigure CurrentFigure
		{
			get
			{
				return currentFigure;
			}
			set
			{
				currentFigure = value;
			}
		}

		public WpfCustomGeometryToPathGeometryConverter(Rect bounding, float defaultScale)
		{
			this.defaultScale = defaultScale;
			pathGeometry = new PathGeometry();
			pathGeometry.FillRule = FillRule.Nonzero;
			rect = bounding;
			left = bounding.Left;
			top = bounding.Top;
		}

		public void Walk(ModelShapePath shapePath)
		{
			path = shapePath;
			isFilled = path.FillMode != PathFillMode.None;
			scaleX = ((shapePath.Width == 0L) ? defaultScale : ((float)rect.Width / (float)shapePath.Width));
			scaleY = ((shapePath.Height == 0L) ? defaultScale : ((float)rect.Height / (float)shapePath.Height));
			foreach (IPathInstruction instruction in path.Instructions)
			{
				instruction.Visit(this);
			}
			if (path.Instructions.Last.GetType() != typeof(PathClose))
			{
				EndFigure();
			}
		}

		private void EndFigure()
		{
			currentFigure.IsClosed = false;
			AddCurrentFigure();
		}

		protected virtual void AddCurrentFigure()
		{
			PathGeometry.Figures.Add(currentFigure);
		}

		public void Visit(PathArc pathArc)
		{
		}

		public virtual void Visit(PathClose value)
		{
			currentFigure.Segments.Add(new LineSegment(currentFigure.StartPoint, path.Stroke));
			currentFigure.IsClosed = true;
			AddCurrentFigure();
			currentFigure = new PathFigure();
		}

		public void Visit(PathCubicBezier value)
		{
			double[] array = new double[3];
			double[] array2 = new double[3];
			for (int i = 0; i < 3; i++)
			{
				array[i] = value.Points[i].X.ValueEMU;
				array2[i] = value.Points[i].Y.ValueEMU;
			}
			currentFigure.Segments.Add(new BezierSegment(new System.Windows.Point(left + EMUToLayoutUnitsX(array[0]), top + EMUToLayoutUnitsY(array2[0])), new System.Windows.Point(left + EMUToLayoutUnitsX(array[1]), top + EMUToLayoutUnitsY(array2[1])), new System.Windows.Point(left + EMUToLayoutUnitsX(array[2]), top + EMUToLayoutUnitsY(array2[2])), path.Stroke));
		}

		public void Visit(PathLine pathLine)
		{
			double valueEMU = pathLine.Point.X.ValueEMU;
			double valueEMU2 = pathLine.Point.Y.ValueEMU;
			currentFigure.Segments.Add(new LineSegment(new System.Windows.Point(left + EMUToLayoutUnitsX(valueEMU), top + EMUToLayoutUnitsY(valueEMU2)), path.Stroke));
		}

		public void Visit(PathMove pathMove)
		{
			double valueEMU = pathMove.Point.X.ValueEMU;
			double valueEMU2 = pathMove.Point.Y.ValueEMU;
			if (currentFigure != null)
			{
				AddCurrentFigure();
			}
			currentFigure = new PathFigure();
			lastX = EMUToLayoutUnitsX(valueEMU) + left;
			lastY = EMUToLayoutUnitsY(valueEMU2) + top;
			currentFigure.StartPoint = new System.Windows.Point(lastX, lastY);
			currentFigure.IsClosed = false;
			currentFigure.IsFilled = isFilled;
		}

		public void Visit(PathQuadraticBezier value)
		{
		}

		private double EMUToLayoutUnitsX(double emu)
		{
			return emu * (double)scaleX;
		}

		private double EMUToLayoutUnitsY(double emu)
		{
			return emu * (double)scaleY;
		}
	}

	public class WpfHeadTailArrowGeometryCreator
	{
		private struct ArrowSegmentInfo
		{
			public static ArrowSegmentInfo Invalid = new ArrowSegmentInfo(-1, default(System.Windows.Point), default(System.Windows.Point));

			public int SegmentIndex { get; private set; }

			public System.Windows.Point InsidePoint { get; private set; }

			public System.Windows.Point OutsidePoint { get; private set; }

			public bool IsBezierCurve { get; private set; }

			public System.Windows.Point[] BezierPoints { get; private set; }

			public bool IsValid => SegmentIndex >= 0;

			public ArrowSegmentInfo(int segmentIndex, System.Windows.Point insidePoint, System.Windows.Point outsidePoint)
			{
				this = default(ArrowSegmentInfo);
				SegmentIndex = segmentIndex;
				InsidePoint = insidePoint;
				OutsidePoint = outsidePoint;
			}

			public ArrowSegmentInfo(int segmentIndex, System.Windows.Point insidePoint, System.Windows.Point outsidePoint, System.Windows.Point[] bezierPoints)
				: this(segmentIndex, insidePoint, outsidePoint)
			{
				BezierPoints = bezierPoints;
				IsBezierCurve = true;
			}
		}

		private abstract class WpfArrowGeometryCreatorBase
		{
			private ArrowSegmentInfo arrowSegmentInfo;

			private Geometry boundingGeometry;

			private OutlineHeadTailSize lengthSize;

			private OutlineHeadTailSize widthSize;

			private double penWidth;

			public Geometry BoundingGeometry => boundingGeometry;

			public ArrowSegmentInfo ArrowSegmentInfo
			{
				get
				{
					return arrowSegmentInfo;
				}
				protected set
				{
					arrowSegmentInfo = value;
				}
			}

			public Geometry CreateHeadArrow(PathGeometry geometry, System.Windows.Media.Pen pen, double penWidthLimit, OutlineHeadTailSize lengthSize, OutlineHeadTailSize widthSize)
			{
				return CreateArrowCore(geometry, pen, penWidthLimit, lengthSize, widthSize, CreateStartArrowTransform);
			}

			public Geometry CreateTailArrow(PathGeometry geometry, System.Windows.Media.Pen pen, double penWidthLimit, OutlineHeadTailSize lengthSize, OutlineHeadTailSize widthSize)
			{
				return CreateArrowCore(geometry, pen, penWidthLimit, lengthSize, widthSize, CreateEndArrowTransform);
			}

			private Geometry CreateArrowCore(PathGeometry geometry, System.Windows.Media.Pen pen, double penWidthLimit, OutlineHeadTailSize lengthSize, OutlineHeadTailSize widthSize, Func<PathGeometry, double, TransformGroup> createTransform)
			{
				penWidth = pen.Thickness;
				this.lengthSize = lengthSize;
				this.widthSize = widthSize;
				double num = Math.Max(penWidthLimit, penWidth);
				double arrowLength = num * GetArrowLengthCoefficient(lengthSize);
				double arrowWidth = num * GetArrowWidthCoefficient(widthSize);
				TransformGroup transform = createTransform(geometry, CorrectArrowLenght(arrowLength, arrowWidth, penWidth));
				boundingGeometry = CreateBoundingGeometry(arrowLength, arrowWidth, transform);
				return CreateArrowGeometry(pen, arrowLength, arrowWidth, transform);
			}

			protected abstract Geometry CreateArrowGeometry(System.Windows.Media.Pen pen, double arrowLength, double arrowWidth, TransformGroup transform);

			protected abstract Geometry CreateBoundingGeometry(double arrowLength, double arrowWidth, TransformGroup transform);

			protected virtual double CorrectArrowLenght(double arrowLength, double arrowWidth, double penWidth)
			{
				return arrowLength;
			}

			protected virtual System.Windows.Point GetPointInsideArrow(System.Windows.Point start, System.Windows.Point end, double arrowLength)
			{
				return end;
			}

			protected virtual double GetArrowLengthCoefficient(OutlineHeadTailSize length)
			{
				return GetStandartArrowWidthOrLength(length);
			}

			protected virtual double GetArrowWidthCoefficient(OutlineHeadTailSize width)
			{
				return GetStandartArrowWidthOrLength(width);
			}

			private double GetStandartArrowWidthOrLength(OutlineHeadTailSize size)
			{
				return size switch
				{
					OutlineHeadTailSize.Large => 5.0, 
					OutlineHeadTailSize.Medium => 3.0, 
					_ => 2.0, 
				};
			}

			private TransformGroup CreateStartArrowTransform(PathGeometry geometry, double arrowLength)
			{
				if (geometry.IsEmpty())
				{
					return null;
				}
				PathFigure pathFigure = geometry.Figures[0];
				System.Windows.Point startPoint = pathFigure.StartPoint;
				System.Windows.Point nextPoint = GetNextPoint(pathFigure, arrowLength, startPoint);
				return CreateTransformCore(startPoint, nextPoint);
			}

			private TransformGroup CreateEndArrowTransform(PathGeometry geometry, double arrowLength)
			{
				if (geometry.IsEmpty())
				{
					return null;
				}
				PathFigure figure = geometry.Figures[0];
				System.Windows.Point lastPoint = GetLastPoint(figure);
				System.Windows.Point prevPoint = GetPrevPoint(figure, arrowLength, lastPoint);
				return CreateTransformCore(lastPoint, prevPoint);
			}

			private TransformGroup CreateTransformCore(System.Windows.Point firstPoint, System.Windows.Point secondPoint)
			{
				double angle = Math.Atan2(secondPoint.Y - firstPoint.Y, secondPoint.X - firstPoint.X) * 180.0 / Math.PI;
				TransformGroup transformGroup = new TransformGroup();
				transformGroup.Children.Add(new RotateTransform(angle));
				transformGroup.Children.Add(new TranslateTransform(firstPoint.X, firstPoint.Y));
				return transformGroup;
			}

			private System.Windows.Point GetNextPoint(PathFigure figure, double arrowLength, System.Windows.Point firstPoint)
			{
				PathSegmentCollection segments = figure.Segments;
				System.Windows.Point point = firstPoint;
				System.Windows.Point point2 = default(System.Windows.Point);
				for (int i = 0; i < segments.Count; i++)
				{
					PathSegment pathSegment = segments[i];
					if (pathSegment is LineSegment lineSegment)
					{
						point2 = lineSegment.Point;
						double segmentLength = GetSegmentLength(firstPoint, point2);
						if (segmentLength >= arrowLength)
						{
							System.Windows.Point point3 = ((segmentLength == arrowLength) ? point2 : CalculatePointOnLine(firstPoint, point, point2, arrowLength, segmentLength));
							System.Windows.Point pointInsideArrow = GetPointInsideArrow(firstPoint, point3, arrowLength);
							arrowSegmentInfo = new ArrowSegmentInfo(i, pointInsideArrow, point2);
							return point3;
						}
					}
					else if (pathSegment is BezierSegment bezierSegment)
					{
						point2 = bezierSegment.Point3;
						Vector[] bezierPoints = WpfDrawingShapeHelper.GetBezierPoints(point, bezierSegment.Point1, bezierSegment.Point2, point2);
						Tuple<double, System.Windows.Point> tuple = CalculatePointOnBezier(bezierPoints, firstPoint, arrowLength, 0.0, 1.0);
						if (tuple != null)
						{
							System.Windows.Point[] splitBezierSegmentPoints = GetSplitBezierSegmentPoints(bezierPoints, 1.0 - tuple.Item1);
							arrowSegmentInfo = new ArrowSegmentInfo(i, splitBezierSegmentPoints[3], point2, new System.Windows.Point[3]
							{
								splitBezierSegmentPoints[4],
								splitBezierSegmentPoints[5],
								point2
							});
							return tuple.Item2;
						}
					}
					point = point2;
				}
				arrowSegmentInfo = ArrowSegmentInfo.Invalid;
				return point2;
			}

			private System.Windows.Point GetPrevPoint(PathFigure figure, double arrowLength, System.Windows.Point firstPoint)
			{
				PathSegmentCollection segments = figure.Segments;
				System.Windows.Point prev = firstPoint;
				System.Windows.Point point = default(System.Windows.Point);
				for (int num = segments.Count - 1; num >= 0; num--)
				{
					PathSegment pathSegment = segments[num];
					if (pathSegment is LineSegment)
					{
						point = GetPrevSegmentEndPoint(figure, num);
						double segmentLength = GetSegmentLength(firstPoint, point);
						if (segmentLength >= arrowLength)
						{
							System.Windows.Point point2 = ((segmentLength == arrowLength) ? point : CalculatePointOnLine(firstPoint, prev, point, arrowLength, segmentLength));
							System.Windows.Point pointInsideArrow = GetPointInsideArrow(firstPoint, point2, arrowLength);
							arrowSegmentInfo = new ArrowSegmentInfo(num, pointInsideArrow, point);
							return point2;
						}
					}
					else if (pathSegment is BezierSegment bezierSegment)
					{
						point = GetPrevSegmentEndPoint(figure, num);
						Vector[] bezierPoints = WpfDrawingShapeHelper.GetBezierPoints(point, bezierSegment.Point1, bezierSegment.Point2, bezierSegment.Point3);
						Tuple<double, System.Windows.Point> tuple = CalculatePointOnBezier(bezierPoints, firstPoint, arrowLength, 1.0, 0.0);
						if (tuple != null)
						{
							System.Windows.Point[] splitBezierSegmentPoints = GetSplitBezierSegmentPoints(bezierPoints, 1.0 - tuple.Item1);
							arrowSegmentInfo = new ArrowSegmentInfo(num, splitBezierSegmentPoints[3], point, new System.Windows.Point[3]
							{
								splitBezierSegmentPoints[1],
								splitBezierSegmentPoints[2],
								splitBezierSegmentPoints[3]
							});
							return tuple.Item2;
						}
					}
					prev = point;
				}
				arrowSegmentInfo = ArrowSegmentInfo.Invalid;
				return point;
			}

			private System.Windows.Point GetLastPoint(PathFigure figure)
			{
				PathSegment pathSegment = figure.Segments[figure.Segments.Count - 1];
				if (pathSegment is LineSegment lineSegment)
				{
					return lineSegment.Point;
				}
				if (pathSegment is BezierSegment bezierSegment)
				{
					return bezierSegment.Point3;
				}
				return default(System.Windows.Point);
			}

			private System.Windows.Point GetPrevSegmentEndPoint(PathFigure figure, int index)
			{
				if (index <= 0)
				{
					return figure.StartPoint;
				}
				PathSegment pathSegment = figure.Segments[index - 1];
				if (pathSegment is LineSegment lineSegment)
				{
					return lineSegment.Point;
				}
				if (pathSegment is BezierSegment bezierSegment)
				{
					return bezierSegment.Point3;
				}
				return default(System.Windows.Point);
			}

			private double GetSegmentLength(System.Windows.Point start, System.Windows.Point end)
			{
				return Math.Sqrt(Math.Pow(end.X - start.X, 2.0) + Math.Pow(end.Y - start.Y, 2.0));
			}

			private System.Windows.Point CalculatePointOnLine(System.Windows.Point start, System.Windows.Point prev, System.Windows.Point next, double arrowLength, double lengthFromStart)
			{
				bool flag = prev.Equals(start);
				double num = (flag ? lengthFromStart : GetSegmentLength(prev, next));
				double num2 = (next.X - prev.X) / num;
				double num3 = (next.Y - prev.Y) / num;
				if (!flag)
				{
					double num4 = num3 * (start.Y - prev.Y) + num2 * (start.X - prev.X);
					double num5 = num3 * (start.X - prev.X) - num2 * (start.Y - prev.Y);
					num4 += Math.Sqrt(arrowLength * arrowLength - num5 * num5);
					arrowLength = num4;
				}
				double x = prev.X + num2 * arrowLength;
				double y = prev.Y + num3 * arrowLength;
				return new System.Windows.Point(x, y);
			}

			private System.Windows.Point[] GetSplitBezierSegmentPoints(Vector[] bezierPoints, double t)
			{
				System.Windows.Point point = (System.Windows.Point)bezierPoints[0];
				System.Windows.Point bezierPoint = GetBezierPoint((System.Windows.Point)bezierPoints[1], (System.Windows.Point)bezierPoints[0], t);
				System.Windows.Point bezierPoint2 = GetBezierPoint((System.Windows.Point)bezierPoints[2], (System.Windows.Point)bezierPoints[1], t);
				System.Windows.Point bezierPoint3 = GetBezierPoint((System.Windows.Point)bezierPoints[3], (System.Windows.Point)bezierPoints[2], t);
				System.Windows.Point bezierPoint4 = GetBezierPoint(bezierPoint2, bezierPoint, t);
				System.Windows.Point bezierPoint5 = GetBezierPoint(bezierPoint3, bezierPoint2, t);
				System.Windows.Point bezierPoint6 = GetBezierPoint(bezierPoint5, bezierPoint4, t);
				System.Windows.Point point2 = (System.Windows.Point)bezierPoints[3];
				return new System.Windows.Point[7] { point, bezierPoint, bezierPoint4, bezierPoint6, bezierPoint5, bezierPoint3, point2 };
			}

			private System.Windows.Point GetBezierPoint(System.Windows.Point start, System.Windows.Point end, double t)
			{
				return new System.Windows.Point((end.X - start.X) * t + start.X, (end.Y - start.Y) * t + start.Y);
			}

			private Tuple<double, System.Windows.Point> CalculatePointOnBezier(Vector[] bezierPoints, System.Windows.Point startPoint, double arrowLength, double minT, double maxT)
			{
				double num = WpfDrawingShapeHelper.CalcBezierLength(bezierPoints);
				int num2 = (int)Math.Max(10.0, num / 20.0);
				double num3 = (maxT - minT) / (double)num2;
				double minT2 = minT;
				for (int i = 0; i < num2; i++)
				{
					double num4 = minT + ((double)i + 1.0) * num3;
					System.Windows.Point point = WpfDrawingShapeHelper.CalcBezier(bezierPoints, num4);
					double segmentLength = GetSegmentLength(startPoint, point);
					if (segmentLength == arrowLength)
					{
						return new Tuple<double, System.Windows.Point>(num4, point);
					}
					if (segmentLength > arrowLength)
					{
						return CalculatePointOnBezierMorePrecisely(bezierPoints, startPoint, arrowLength, minT2, num4, point, num2);
					}
					minT2 = num4;
				}
				return null;
			}

			private Tuple<double, System.Windows.Point> CalculatePointOnBezierMorePrecisely(Vector[] bezierPoints, System.Windows.Point startPoint, double arrowLength, double minT, double maxT, System.Windows.Point prevPoint, int steps)
			{
				if (steps <= 0)
				{
					return new Tuple<double, System.Windows.Point>(maxT, prevPoint);
				}
				double num = (minT + maxT) / 2.0;
				System.Windows.Point point = WpfDrawingShapeHelper.CalcBezier(bezierPoints, num);
				double segmentLength = GetSegmentLength(startPoint, point);
				steps--;
				if (segmentLength > arrowLength)
				{
					return CalculatePointOnBezierMorePrecisely(bezierPoints, startPoint, arrowLength, minT, num, point, steps);
				}
				return CalculatePointOnBezierMorePrecisely(bezierPoints, startPoint, arrowLength, num, maxT, point, steps);
			}

		{
			return new LineGeometry(new System.Windows.Point(x, y), new System.Windows.Point(x + 1, y + 1));
		}

		private static LineGeometry CreateLine(int x1, int y1, int x2, int y2)
		{
			if (x1 != x2)
			{
				x2++;
			}
			if (y1 != y2)
			{
				y2++;
			}

	public static class ImageExtensions
	{
		private static float defaultTargetDpi = 220f;

		private static BitmapImage CreateBitmapImageFrom(MemoryStream ms)
		{
			ms.Position = 0L;
			BitmapImage bitmapImage = new BitmapImage();
			bitmapImage.BeginInit();
			bitmapImage.StreamSource = ms;
			bitmapImage.EndInit();
			return bitmapImage;
		}

		public static ImageSource ToImageSource(this DXImage image)
		{
			return image.ToImageSource(defaultTargetDpi);
		}

		public static ImageSource ToImageSource(this DXImage image, float targetDpi)
		{
			System.Windows.Size targetSize = new System.Windows.Size((float)image.Width * targetDpi / image.HorizontalResolution, (float)image.Height * targetDpi / image.VerticalResolution);
			return image.ToImageSource(targetSize);
		}

		public static ImageSource ToImageSource(this DXImage image, System.Windows.Size targetSize)
		{
			MemoryStream memoryStream = new MemoryStream();
			if (!(image is DXBitmap))
			{
				int num = (int)(targetSize.Width * (double)DocumentModelDpi.Dpi / 96.0);
				int num2 = (int)(targetSize.Height * (double)DocumentModelDpi.Dpi / 96.0);
				if (num <= 0 || num2 <= 0)
				{
					return null;
				}
				image = ((!(image is DXMetafile)) ? DXBitmapCreator.CreateFromSvgImage(image as DXSvgImage, num, num2) : DXBitmapCreator.CreateFromMetafile(image as DXMetafile, num, num2));
			}
			try
			{
				image.Save(memoryStream, DXImageFormat.Png);
			}
			catch
			{
				memoryStream.Dispose();
				memoryStream = new MemoryStream();
				image.Save(memoryStream, image.ImageFormat);
			}
			return CreateBitmapImageFrom(memoryStream);
		}

		public static ImageSource ToImageSource(this DXImage dxImage, bool scaleToScreenDpi)
		{
			MemoryStream memoryStream = new MemoryStream();
			if (scaleToScreenDpi)
			{
				int num = (int)((double)((float)dxImage.Width * DocumentModelDpi.Dpi) / 96.0);
				int num2 = (int)((double)((float)dxImage.Height * DocumentModelDpi.Dpi) / 96.0);
				using DXBitmap dXBitmap = new DXBitmap(num, num2, DXPixelFormat.ARGB32bpp);
				dXBitmap.SetResolution(96f, 96f);
				using IDXGraphics iDXGraphics = dXBitmap.CreateGraphics();
				iDXGraphics.DrawImage(dxImage, new RectangleF(0f, 0f, num, num2));
				dXBitmap.Save(memoryStream, DXImageFormat.Png);
			}
			else
			{
				try
				{
					dxImage.Save(memoryStream, DXImageFormat.Png);
				}
				catch
				{
					memoryStream.Dispose();
					memoryStream = new MemoryStream();
					dxImage.Save(memoryStream, dxImage.ImageFormat);
				}
			}
			return CreateBitmapImageFrom(memoryStream);
		}
	}
	public static class RectangleExtensions
	{
		public static Rect GetSaveRect(double x, double y, double width, double height)
		{
			return new Rect(x, y, Math.Max(0.0, width), Math.Max(0.0, height));
		}

		public static Rect ToRect(this Rectangle rectangle)
		{
			return GetSaveRect(rectangle.X, rectangle.Y, rectangle.Width, rectangle.Height);
		}

		public static Rect ToRect(this RectangleF rectangle)
		{
			return GetSaveRect(rectangle.X, rectangle.Y, rectangle.Width, rectangle.Height);
		}

		public static Rectangle ToRectangle(this Rect rect)
		{
			return new Rectangle((int)Math.Round(rect.X), (int)Math.Round(rect.Y), (int)Math.Round(rect.Width), (int)Math.Round(rect.Height));
		}

		public static Rect BoundingRectangle(this Rect bounds, Matrix transform)
		{
			System.Windows.Point[] array = new System.Windows.Point[4]
			{
				new System.Windows.Point(bounds.Right, bounds.Bottom),
				new System.Windows.Point(bounds.Right, bounds.Top),
				new System.Windows.Point(bounds.Left, bounds.Bottom),
				bounds.Location
			};
			transform.Transform(array);
			return BoundingRectangle(array[0], array[1], array[2], array[3]);
		}

		public static Rect BoundingRectangle(params System.Windows.Point[] points)
		{
			int num = points.Length;
			if (num == 0)
			{
				return Rect.Empty;
			}
			double num2 = points[0].X;
			double num3 = num2;
			double num4 = points[0].Y;
			double num5 = num4;
			for (int i = 1; i < num; i++)
			{
				System.Windows.Point point = points[i];
				num2 = Math.Min(point.X, num2);
				num4 = Math.Min(point.Y, num4);
				num3 = Math.Max(point.X, num3);
				num5 = Math.Max(point.Y, num5);
			}
			return new Rect(num2, num4, num3 - num2, num5 - num4);
		}

		public static System.Windows.Size ToSize(this Rectangle rectangle)
		{
			return new System.Windows.Size(Math.Max(0, rectangle.Width), Math.Max(0, rectangle.Height));
		}
	}
