		public UtilityInformation Clone()
		{
			UtilityInformation result = null;
			try
			{
				BinaryFormatter binaryFormatter = new BinaryFormatter();
				using MemoryStream memoryStream = new MemoryStream();
				binaryFormatter.Serialize(memoryStream, this);
				memoryStream.Flush();
				memoryStream.Seek(0L, SeekOrigin.Begin);
				result = binaryFormatter.Deserialize(memoryStream) as UtilityInformation;
				memoryStream.Close();
			}
			catch (Exception)
			{
				result = new UtilityInformation
				{
					Name = Name,
					Current = _current,
					Target = _target,
					CostIndices = CostIndices,
					UtilityType = UtilityType,
					ValueUnitConvert = ValueUnitConvert,
					GHGEmissionCoefficient = _emission,
					DeltaMinTemperature = _deltaMinTemp,
					OriginalOrder = OriginalOrder,
					HeatExchangers = _heatExchangers
				};
			}
			return result;
		}

						IsTMinChanged = true;
					}
				}
			}

		private static XmlDocument CreateDocument()
		{
			XmlDocument xmlDocument = new XmlDocument();
			XmlNode newChild = xmlDocument.CreateXmlDeclaration("1.0", "utf-8", "yes");
			xmlDocument.AppendChild(newChild);
			XmlNode newChild2 = xmlDocument.CreateNode(XmlNodeType.Element, "Root", null);
			xmlDocument.AppendChild(newChild2);

		public CaseDocument(string filePath = null)
		{
			if (string.IsNullOrEmpty(filePath))
			{
				return;
			}
			_filePath = filePath;
			if (File.Exists(_filePath))
			{
				_doc.Load(_filePath);
				return;
			}
			string outerXml = _doc.OuterXml;
			try
			{
				_doc.LoadXml(filePath);
			}
			catch (XmlException)
			{
				_doc.LoadXml(outerXml);
			}
		}

		private void ConvertToNewVersionDocument(XmlDocument doc)
		{
			if (doc == null || doc.DocumentElement == null)
			{
				return;
			}
			Stack<XmlNode> stack = new Stack<XmlNode>();
			Stack<XmlNode> stack2 = new Stack<XmlNode>();
			stack.Push(doc.DocumentElement);
			while (stack.Count > 0)
			{
				XmlNode xmlNode = stack.Pop();
				if (xmlNode.HasChildNodes)
				{
					bool flag = true;
					foreach (XmlNode childNode in xmlNode.ChildNodes)
					{
						if (childNode.NodeType == XmlNodeType.Element)
						{
							flag = false;
							stack.Push(childNode);
						}
					}
					if (flag)
					{
						stack2.Push(xmlNode);
					}
				}
				else if (xmlNode != null && xmlNode.NodeType == XmlNodeType.Element)
				{
					stack2.Push(xmlNode);
				}
			}
			while (stack2.Count > 0)
			{
				XmlNode xmlNode = stack2.Pop();
				XmlAttribute xmlAttribute = doc.CreateAttribute(null, "ProjectID", null);
				xmlAttribute.Value = "Project 1";
				xmlNode.Attributes.Append(xmlAttribute);
			}
		}

		public dynamic Read(string path)
		{
			string xpath = FormatXPath(path);
			XmlNodeList xmlNodeList = _doc.DocumentElement.SelectNodes(xpath);
			if (xmlNodeList == null || xmlNodeList.Count == 0)
			{
				return null;
			}
			bool flag = xmlNodeList.Count > 1;
			ArrayList arrayList = new ArrayList();
			foreach (XmlNode item in xmlNodeList)
			{
				if (item.Attributes != null && item.Attributes.Count > 1)
				{
					flag = true;
					Dictionary<string, string> dictionary = new Dictionary<string, string>();
					foreach (XmlAttribute attribute in item.Attributes)
					{
						dictionary.Add(UnescapeXmlAttribute(attribute.Name).Replace('-', ' '), UnescapeXmlCharactors(attribute.Value));
					}
					arrayList.Add(new object[2] { item.InnerText, dictionary });
				}
				else
				{
					arrayList.Add(item.InnerText);
				}
			}
			if (arrayList.Count == 1 && !flag)
			{
				return arrayList[0].ToString();
			}
			return arrayList;
		}

		private string FormatXPath(string path)
		{

		private void ProceedToNode(string path, object value, Dictionary<string, string> attributes = null)
		{
			string text = string.Empty;
			if (value is IEnumerable<string>)
			{
				foreach (string item in value as IEnumerable<string>)
				{
					text = text + item + ";";
				}
				text = text.TrimEnd(';');
			}
			else
			{
				text = (value ?? string.Empty).ToString();
			}
			XmlNode node = FindNode(path);
			if (attributes != null)
			{
				try
				{
					Parallel.ForEach(attributes.Keys, delegate(string key)
					{
						XmlAttribute xmlAttribute = _doc.CreateAttribute(EscapeXmlAttribute(key).Replace(' ', '-'), null);
						xmlAttribute.Value = EscapeXmlCharactors(attributes[key]);
						lock (node)
						{
							node.Attributes.Append(xmlAttribute);
						}
					});
				}
				catch
				{
				}
			}
			node.InnerText = EscapeXmlCharactors(text);
		}

		private XmlNode FindNode(string path)
		{
			XmlNodeList xmlNodeList = _doc.DocumentElement.SelectNodes(path.Replace('.', '/'));
			XmlNode xmlNode = null;
			if (xmlNodeList == null || xmlNodeList.Count == 0)
			{
				XmlNode xmlNode2 = ((path.IndexOf('.') < 0) ? _doc.DocumentElement : FindNode(path.Substring(0, path.LastIndexOf('.'))));
				Regex regex = new Regex("\\[.+\\]");
				string input = path.Substring(path.LastIndexOf('.') + 1);
				xmlNode = _doc.CreateNode(XmlNodeType.Element, regex.Replace(input, string.Empty), null);
				if (regex.IsMatch(input))
				{
					regex = new Regex("@(.+?)='(.+?)'");
					foreach (Match item in regex.Matches(input))
					{
						XmlAttribute xmlAttribute = _doc.CreateAttribute(null, item.Groups[1].Value, null);
						xmlAttribute.Value = item.Groups[2].Value;
						xmlNode.Attributes.Append(xmlAttribute);
					}
				}
				xmlNode2.AppendChild(xmlNode);
			}
			else
			{
				xmlNode = xmlNodeList[xmlNodeList.Count - 1];
			}
			return xmlNode;
		}

		private static string EscapeXmlCharactors(string input)
		{
			return input.Replace("<", "&lt;").Replace(">", "&gt;").Replace("&", "&amp;")
				.Replace("'", "&apos;")
				.Replace("\"", "&quot;");
		}

		private static string UnescapeXmlCharactors(string input)
		{
			return input.Replace("&lt;", "<").Replace("&gt;", ">").Replace("&amp;", "&")
				.Replace("&apos;", "'")
				.Replace("&quot;", "\"");
		}

		private static string EscapeXmlAttribute(string attribute)
		{
			return EscapeXmlCharactors(_attributeKeyStartRulerRegex.IsMatch(attribute) ? ("A" + attribute) : attribute);
		}

		private static string UnescapeXmlAttribute(string attribute)
		{
			return UnescapeXmlCharactors(_attributeKeyStartRuleReserveRegex.IsMatch(attribute) ? attribute.Remove(0, "A".Length) : attribute);
		}

				memoryStream.Close();
			}
			return obj ?? source;
		}

	[Serializable]
	internal class SimulatorValueUnit : IValueUnit
	{
		public ValueType Convert(ValueType source, bool isGHG = false)
		{
			if (!double.TryParse(source.ToString(), out var result))
			{
				result = -1.0;
			}
			result = ((!isGHG) ? (result * SystemUnit_conversionFactors.KWToUserUnit) : (result * SystemUnit_conversionFactors.KgpersToUserUnit));
			return result;
		}

		public ValueType Convert(ValueType source, PropertyEnum propType)
		{
			if (!double.TryParse(source.ToString(), out var result))
			{
				result = -1.0;
			}
			switch (propType)
			{
			case PropertyEnum.Enthalpy:
				result *= SystemUnit_conversionFactors.KWToUserUnit;
				break;
			case PropertyEnum.GHGFlow:
				result *= SystemUnit_conversionFactors.KgpersToUserUnit;
				break;
			case PropertyEnum.Area:
				result *= SystemUnit_conversionFactors.M2ToUserUnit;
				break;
			case PropertyEnum.ToArea:
				result /= SystemUnit_conversionFactors.M2ToUserUnit;
				break;
			case PropertyEnum.Temperature:
			{
				_ = SystemUnit_conversionFactors.TempCToUserUnitA;
				double tempCToUserUnitB2 = SystemUnit_conversionFactors.TempCToUserUnitB;
				result = SystemUnit_conversionFactors.TempCToUserUnitA * result + tempCToUserUnitB2;
				break;
			}
			case PropertyEnum.ToTemperature:
			{
				double tempCToUserUnitA = SystemUnit_conversionFactors.TempCToUserUnitA;
				double tempCToUserUnitB = SystemUnit_conversionFactors.TempCToUserUnitB;
				result = (result - tempCToUserUnitB) / tempCToUserUnitA;
				break;
			}
			case PropertyEnum.DeltaTemperature:
				result *= SystemUnit_conversionFactors.DeltaTempCToUserUnit;
				break;
			case PropertyEnum.ToDeltaTemperature:
				result /= SystemUnit_conversionFactors.DeltaTempCToUserUnit;
				break;
			case PropertyEnum.Fiscal:
				result *= SystemUnit_conversionFactors.DollarToUserUnit;
				break;
			case PropertyEnum.UArea:
				result *= SystemUnit_conversionFactors.OverallUAToUserUnit;
				break;
			case PropertyEnum.OverallU:
				result *= SystemUnit_conversionFactors.OverallHTCToUserUnit;
				break;
			case PropertyEnum.ToOverallU:
				result /= SystemUnit_conversionFactors.OverallHTCToUserUnit;
				break;
			case PropertyEnum.ToCostPerMass:
				result *= SystemUnit_conversionFactors.CostMassRateToUserUnit;
				break;
			case PropertyEnum.FromCostPerMass:
				result /= SystemUnit_conversionFactors.CostMassRateToUserUnit;
				break;
			}
			return result;
		}
	}

	{
		private static IValueUnit _defaultSimulator = new HYSYSSimulatorValueUnit();

		public static IValueUnit DefaultValueUnit
		{
			get
			{
				return _defaultSimulator;
			}
			set
			{
				if (value != null)
				{
					_defaultSimulator = value;
				}
			}
		}

		public static IValueUnit Create(ValueUnitEnum unit)
		{
			IValueUnit valueUnit = null;
			try
			{
				return Activator.CreateInstance("Aspentech.EnergyAnalysis.ClassLibrary", "Aspentech.EnergyAnalysis.ClassLibrary.Converts." + unit.ToString() + "ValueUnit").Unwrap() as IValueUnit;
			}
			catch (Exception)
			{
				return new MWValueUnit();
			}
		}

		public static IValueUnit Create(string simulatorName)
		{
			if (string.IsNullOrEmpty(simulatorName))
			{
				return null;
			}
			string text = simulatorName.ToLower(CultureInfo.CurrentCulture);
			if (!(text == "hysys"))
			{
				if (text == "aspenplus")
				{
					_defaultSimulator = new SimulatorValueUnit();
					return _defaultSimulator;
				}
				return null;
			}
			_defaultSimulator = new HYSYSSimulatorValueUnit();
			return _defaultSimulator;
		}

		public static string ParseValueUnitName(IValueUnit valueUnit)
		{
			string[] array = valueUnit.ToString().Split('.');
			return ((array.Length != 0) ? array[array.Length - 1] : valueUnit.ToString()).Replace("PerHour", "/hr").Replace("PerSecond", "/sec").Replace("ValueUnit", "");
		}

		internal static string ParseValueUnitName(ValueUnitEnum valueUnit)
		{
			return valueUnit.ToString().Replace("PerHour", "/hr").Replace("PerSecond", "/sec");
		}
