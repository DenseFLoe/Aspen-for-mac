	public ImageSource GetImageSource(string imageName)
	{
		ImageSource value = null;
		string text = Path.GetExtension(imageName);
		if (string.IsNullOrEmpty(text))
		{
			text = (imageName.EndsWith("_Icon", StringComparison.OrdinalIgnoreCase) ? ".ico" : ".png");
			imageName += text;
		}
		if (!_imageCache.TryGetValue(imageName, out value))
		{
			try
			{
				string fullName = Assembly.GetExecutingAssembly().FullName;
				string uriString = "pack://application:,,,/" + fullName + ";component/images/" + imageName;
				if (text.Equals(".ico", StringComparison.OrdinalIgnoreCase))
				{
					value = BitmapFrame.Create(new Uri(uriString, UriKind.Absolute));
				}
				else
				{
					BitmapImage bitmapImage = new BitmapImage();
					bitmapImage.BeginInit();
					bitmapImage.UriSource = new Uri(uriString, UriKind.Absolute);
					bitmapImage.EndInit();
					if (bitmapImage.CanFreeze)
					{
						bitmapImage.Freeze();
					}
					value = bitmapImage;
				}
			}
			catch (Exception)
			{
			}
			if (value != null)
			{
				_imageCache[imageName] = value;
			}
		}
		return value;
	}

	public ImageSource GetCustomImageSource(string imageName, int? pixelHeight = null, int? bitsPerPixel = null)
	{
		string text = (imageName.EndsWith("_Icon", StringComparison.OrdinalIgnoreCase) ? (imageName + ".ico") : (imageName + ".png"));
		string fullName = Assembly.GetExecutingAssembly().FullName;
		string uriString = "pack://application:,,,/" + fullName + ";component/images/" + text;
		if (text.EndsWith(".ico", StringComparison.OrdinalIgnoreCase))
		{
			BitmapFrame bitmapFrame = BitmapFrame.Create(new Uri(uriString, UriKind.Absolute));
			if (bitmapFrame == null)
			{
				return null;
			}
			BitmapFrame bitmapFrame2 = null;
			{
				foreach (BitmapFrame frame in bitmapFrame.Decoder.Frames)
				{
					if (bitmapFrame2 == null)
					{
						bitmapFrame2 = frame;
						continue;
					}
					if (pixelHeight.HasValue)
					{
						if (frame.PixelHeight != pixelHeight)
						{
							continue;
						}
					}
					else if (frame.PixelHeight < bitmapFrame2.PixelHeight)
					{
						continue;
					}
					if (bitsPerPixel.HasValue)
					{
						int num = Math.Abs(bitmapFrame2.Format.BitsPerPixel - bitsPerPixel.Value);
						if (Math.Abs(frame.Format.BitsPerPixel - bitsPerPixel.Value) < num)
						{
							bitmapFrame2 = frame;
						}
					}
					else if (bitmapFrame2.Format.BitsPerPixel < frame.Format.BitsPerPixel)
					{
						bitmapFrame2 = frame;
					}
				}
				return bitmapFrame2;
			}
		}
		BitmapImage bitmapImage = new BitmapImage();
		bitmapImage.BeginInit();
		bitmapImage.UriSource = new Uri(uriString, UriKind.Absolute);
		if (pixelHeight.HasValue)
		{
			bitmapImage.DecodePixelHeight = pixelHeight.Value;
		}
		bitmapImage.EndInit();
		if (bitmapImage.CanFreeze)
		{
			bitmapImage.Freeze();
		}
		return bitmapImage;
	}
