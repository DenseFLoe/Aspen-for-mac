		private void AnalyzeEquipmentXml(string xml)
		{
			if (string.IsNullOrEmpty(xml))
			{
				return;
			}
			_allEquipments.Clear();
			_tabEquipmentsPair.Clear();
			foreach (XElement item in XDocument.Parse(xml).XPathSelectElements(".//Tab"))
			{
				XElement xElement = item.Element("Caption");

				{
					continue;
				}
				XElement xElement2 = item.Element("Rows");
				if (xElement2 == null)
				{
					continue;
				}
				IEnumerable<XElement> enumerable = xElement2.XPathSelectElements("./Row");
				if (enumerable == null)
				{
					continue;
				}
				List<string> list = new List<string>();
				foreach (XElement item2 in enumerable)
				{
					XElement xElement3 = item2.Element("Name");
					if (xElement3 != null && !string.IsNullOrEmpty(xElement3.Value))
					{
						list.Add(xElement3.Value);
					}
				}
				if (list.Count > 0)
				{
					if (string.Equals("Equipment", xElement.Value))
					{
						_allEquipments.AddRange(list);
					}
					_tabEquipmentsPair[xElement.Value] = list;
				}
			}
		}
