		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)

		public void UpdateWizardData()
		{
			_wizardPlot.SetIndependentVariable(_context, _indepVarName);
			_wizardPlot.RemoveAllDependentVariable();
			for (int i = 0; i < _profileRadBtnList.Count(); i++)
			{
				if (_profileRadBtnList[i].IsChecked == true)
				{
					switch (i)
					{
					case 0:
						_wizardPlot.AddDependentVariable(_context, _depVarNames[0] + "!Label");
						break;
					case 1:
						_wizardPlot.AddDependentVariable(_context, _depVarNames[0]);
						_help.SaveUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxTemperature);
						break;
					default:
						_wizardPlot.AddDependentVariable(_context, _depVarNames[1]);
						break;
					}
					_wizardPlot.PlotVal.Title = _strTitle[i];
					((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle((object)null, _strYTitle[i]);
					break;
				}
			}
			_help.SaveXUnitsComboText(_context.Application.Tree, _wizardPlot, _currentExergyUnit);
			_currentConfigValue = SaveConfigValue();
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("ProfileOption=");
			for (int i = 0; i < _profileRadBtnList.Count; i++)
			{
				if (_profileRadBtnList[i].IsChecked == true)
				{
					stringBuilder.Append(i.ToString());
					break;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("ProfileEnable=");
			int num = 0;
			for (int j = 0; j < _profileRadBtnList.Count; j++)
			{
				if (_profileRadBtnList[j].IsEnabled)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");
					}
					stringBuilder.Append(j.ToString());
					num++;
				}
			}
			stringBuilder.Append(";");
			stringBuilder.Append("Unit=");
			stringBuilder.Append(comboBoxTemperature.Text);
			stringBuilder.Append(";");
			stringBuilder.Append("UnitEnabel=");
			stringBuilder.Append(comboBoxTemperature.IsEnabled);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

		private void RestoreConfigValue(string ConfigString)
		{
			string[] array = ConfigString.Split(';');
			int num = 0;
			string value = array[num].Substring(array[num].IndexOf('=') + 1);
			_profileRadBtnList[Convert.ToInt16(value)].IsChecked = true;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = value.Split(',');
			foreach (string value2 in array2)
			{
				_profileRadBtnList[Convert.ToInt16(value2)].IsEnabled = true;
			}
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			SetUnit();
			comboBoxTemperature.Text = value;
			num++;
			value = array[num].Substring(array[num].IndexOf('=') + 1);
			comboBoxTemperature.IsEnabled = Convert.ToBoolean(value);
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			bool flag = false;
			iHNode = _context.FindNode(_3phaseVarNames[1, 1]);
			if (iHNode != null && iHNode.Elements.Count > 0)
			{
				_varNames = _3phaseVarNames;
			}
			else
			{
				_varNames = _2phaseVarNames;
			}
			for (int i = 0; i < _varNames.GetLength(0); i++)
			{
				for (int j = 0; j < _varNames.GetLength(1); j++)
				{
					iHNode = _context.FindNode(_varNames[i, j]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_flowCheckList[j].IsEnabled = true;
						if (j < 2)
						{
							_flowCheckList[j].IsChecked = true;
						}
						_basisRadBtnList[i].IsEnabled = true;
						flag = true;
					}
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			_wizardPlot.SetIndependentVariable(_context, _varNames[0, 0] + "!Label");
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, _varNames[0, 0]);
			_wizardPlot.basis = 1;
			_help.LoadUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxUnits);
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				for (int k = 0; k < _basisRadBtnList.Count(); k++)
				{
					if (_basisRadBtnList[k].IsEnabled)
					{
						_basisRadBtnList[k].IsChecked = true;
						break;
					}
				}
			}
			else
			{
				for (int l = 0; l < _flowCheckList.Count(); l++)
				{
					_flowCheckList[l].IsChecked = false;
				}
				RestoreConfigValue(_currentConfigValue);
			}
		}

			{
				num2++;
				if (!_flowCheckList[num2].IsEnabled || _flowCheckList[num2].IsChecked != true)

					continue;
				}
				IHNode iHNode = _context.FindNode(_varNames[num, j]);
				IHNode iHNode2 = _context.FindNode(_varNames[num, j + 1]);

				_wizardPlot.PlotVal.NVariables = _wizardPlot.PlotVal.NVariables + 1;

				{
					double num3 = ((double?)(dynamic)iHNode2.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing)) ?? 0.0;
					if (num3 != 0.0)
					{
						double num4 = ((double?)(dynamic)iHNode.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing)) ?? 0.0;
						((IHPlotVal)_wizardPlot.PlotVal).set_YData(index, k, (short)0, (object)(num4 / num3));
					}
				}
				((IHPlotVal)_wizardPlot.PlotVal).set_Label(index++, _flowCheckList[num2].Content.ToString());
			}
			_wizardPlot.PlotVal.Title = title;
			((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle((object)null, ytitle);
			_currentConfigValue = SaveConfigValue();
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			_context.FindNode(_YVarNames[0, 0]);
			_context.FindNode(_XVarNames[0, 0]);
			IHNode iHNode = null;
			bool flag = false;
			bool flag2 = false;
			for (int i = 0; i < _YVarNames.GetLength(1); i++)
			{
				for (int j = 0; j < _YVarNames.GetLength(0); j++)
				{
					iHNode = _context.FindNode(_YVarNames[i, j]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_flowCheckList[j].IsEnabled = true;
						_flowCheckList[j].IsChecked = true;
						_basisRadBtnList[i].IsEnabled = true;
						_phaseRadBtnList[0].IsEnabled = true;
						_phaseRadBtnList[0].IsChecked = true;
						flag = true;
					}
				}
			}
			for (int k = 0; k < _XVarNames.GetLength(1); k++)
			{
				for (int l = 0; l < _XVarNames.GetLength(0); l++)
				{
					iHNode = _context.FindNode(_XVarNames[k, l]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_flowCheckList[l].IsEnabled = true;
						_flowCheckList[l].IsChecked = true;
						_basisRadBtnList[k].IsEnabled = true;
						_phaseRadBtnList[1].IsEnabled = true;
						if (!flag)
						{
							_phaseRadBtnList[1].IsChecked = true;
						}
						flag2 = true;
					}
				}
			}
			if (!flag && !flag2)
			{
				ierror = 1;
				return;
			}
			_wizardPlot.SetIndependentVariable(_context, _YVarNames[0, 1] + "!Label");
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, _YVarNames[0, 1]);
			_wizardPlot.basis = 1;
			_help.LoadUnitsCombo(_context.Application.Tree, _wizardPlot, comboBoxUnits);
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				for (int m = 0; m < _basisRadBtnList.Count(); m++)
				{
					if (_basisRadBtnList[m].IsEnabled)
					{
						_basisRadBtnList[m].IsChecked = true;
						break;
					}
				}
			}
			else
			{
				for (int n = 0; n < _flowCheckList.Count(); n++)
				{
					_flowCheckList[n].IsChecked = false;
				}
				RestoreConfigValue(_currentConfigValue);
			}
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			bool flag = false;
			for (int i = 0; i < _varNames.Count(); i++)
			{
				iHNode = _context.FindNode(_varNames[i]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					_phaseRadBtnList[i].IsEnabled = true;
					flag = true;
				}
			}
			if (!flag)
			{
				ierror = 1;
			}
			else if (string.IsNullOrEmpty(_currentConfigValue))
			{
				for (int j = 0; j < _phaseRadBtnList.Count(); j++)
				{
					if (_phaseRadBtnList[j].IsEnabled)
					{
						_phaseRadBtnList[j].IsChecked = true;
						iHNode = _context.FindNode(_varNames[j]);
						break;
					}
				}
				List<string> list = new List<string>();
				if (iHNode == null || iHNode.Elements.Count <= 0)
				{
					return;
				}
				IHNodeCol elements = iHNode.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
				if (elements != null && elements.Count > 0)
				{
					for (int k = 0; k < elements.Count; k++)
					{
						string item = elements.get_Label(0, k, (object)null);
						list.Add(item);
					}
					CurveList.IsCheckAll = false;
					CurveList.SetCurves(list.ToArray());
					CurveList.Header = WpfPLRadFracResources.Select_Comp;
				}
			}
			else
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		public void UpdateWizardData()
		{
			string title = "";
			int num = -1;
			for (int i = 0; i < _phaseRadBtnList.Count(); i++)
			{
				if (_phaseRadBtnList[i].IsEnabled && _phaseRadBtnList[i].IsChecked == true)
				{
					num = i;
					title = _strTitle[i];
					break;
				}
			}
			_wizardPlot.SetIndependentVariable(_context, "Input/NSTAGE");
			_wizardPlot.RemoveAllDependentVariable();
			IHNode iHNode = _context.FindNode(_varNames[num]);
			if (iHNode != null)
			{
				List<string> list = new List<string>(CurveList.GetSelectedCurves());
				int count = list.Count;
				int count2 = iHNode.Elements.Count;
				_wizardPlot.PlotVal.NVariables = count;
				for (int j = 0; j < count; j++)
				{
					string text = list[j];
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(j, text);
					try
					{
						for (int k = 0; k < count2; k++)
						{
							object obj = iHNode.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements[text, Type.Missing, Type.Missing, Type.Missing, Type.Missing];
							((IHPlotVal)_wizardPlot.PlotVal).set_YData(j, k, (short)1, obj);
						}
					}
					catch (Exception)
					{
					}
				}
			}
			_wizardPlot.PlotVal.Title = title;
			((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle((object)null, WpfPLRadFracResources.KValues_YAxisTitle);
			_currentConfigValue = SaveConfigValue();
		}

		List<string> IRibbonPlotWizardType.FieldDataOptionList(string sFieldName)
		{
			if (sFieldName.CompareTo(_stageOrderFieldId) == 0)
			{
				return new List<string> { _ascending, _descending };
			}
			if (sFieldName.CompareTo(_stageOnFieldId) == 0)
			{
				return new List<string> { _xAxis, _yAxis };

		RibbonControlType IRibbonPlotWizardType.GetFieldCtrlType(string sField)

		void IRibbonPlotWizardType.GetFieldStyle(string sFieldName, ref string sCurrentVal)
		{
			if (sFieldName.CompareTo(_stageOrderFieldId) == 0)
			{
				if (_wizardPlot.PlotVal.ReverseXAxis)
				{
					sCurrentVal = _descending;
				}
				else
				{
					sCurrentVal = _ascending;
				}
			}
			else if (sFieldName.CompareTo(_stageOnFieldId) == 0)
			{
				if (_wizardPlot.PlotVal.SwapAxises)
				{
					sCurrentVal = _yAxis;
				}
				else
				{
					sCurrentVal = _xAxis;
				}
			}
		}

		void IRibbonPlotWizardType.SetFieldStyle(string sFieldName, string sValue)
		{
			if (sFieldName.CompareTo(_stageOrderFieldId) == 0)
			{
				if (sValue.CompareTo(_ascending) == 0)
				{
					_wizardPlot.PlotVal.ReverseXAxis = false;
				}
				else
				{
					_wizardPlot.PlotVal.ReverseXAxis = true;
				}
			}
			else if (sFieldName.CompareTo(_stageOnFieldId) == 0)
			{
				if (sValue.CompareTo(_yAxis) == 0)
				{
					_wizardPlot.PlotVal.SwapAxises = true;
				}
				else
				{
					_wizardPlot.PlotVal.SwapAxises = false;
				}
			}
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			bool flag = false;
			radBtnComponent.IsChecked = true;
			radBtnCompGroup.IsEnabled = false;
			CompGList.IsEnabled = false;
			radBtnLinear.IsChecked = true;
			radBtnLog.IsChecked = false;
			IHNode iHNode = _context.FindNode("Output/Y");
			if (iHNode != null && iHNode.Elements.Count > 0)
			{
				flag = true;
				List<string> list = new List<string>();
				IHNodeCol elements = iHNode.Elements[0, Type.Missing, Type.Missing, Type.Missing, Type.Missing].Elements;
				for (int i = 0; i < elements.Count; i++)
				{
					string item = elements.get_Label(0, i, (object)null);
					list.Add(item);
				}
				CompList.IsCheckAll = false;
				CompList.SetCurves(list.ToArray());
				if (radBtnComponent.IsChecked == true)
				{
					HCompList.IsCheckAll = false;
					HCompList.SetCurves(list.ToArray());
					HCompList.Header = WpfPLRadFracResources.HComp_Header;
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			CompGList.IsCheckAll = false;
			iHNode = _context.Application.Tree.FindNode("Data/Components/Comp-Groups");
			if (iHNode != null && iHNode.Elements.Count > 0)
			{
				radBtnCompGroup.IsEnabled = true;
				CompGList.IsEnabled = true;
				List<string> list2 = new List<string>();
				for (int j = 0; j < iHNode.Elements.Count; j++)
				{
					string item2 = iHNode.Elements.get_Label(0, j, (object)null);
					list2.Add(item2);
				}
				CompGList.SetCurves(list2.ToArray());
				if (radBtnCompGroup.IsChecked == true)
				{
					HCompList.IsCheckAll = false;
					HCompList.SetCurves(list2.ToArray());
					HCompList.Header = WpfPLRadFracResources.HCompG_Header;
				}
			}
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			int num = CompList.GetSelectedCurves().Count();
			if (num == CompList.Curves.Count)
			{
				CompList.IsCheckAll = true;
			}
			else
			{
				string[] selectedCurves = CompList.GetSelectedCurves();
				CompList.IsCheckAll = false;
				CompList.SetSelectedCurves(selectedCurves);
			}
			int num2 = CompGList.GetSelectedCurves().Count();
			if (num2 > 0 && num2 == CompGList.Curves.Count)
			{
				CompGList.IsCheckAll = true;
			}
			else
			{
				string[] selectedCurves2 = CompGList.GetSelectedCurves();
				CompGList.IsCheckAll = false;
				CompGList.SetSelectedCurves(selectedCurves2);
			}
			int num3 = HCompList.GetSelectedCurves().Count();
			if (num3 == HCompList.Curves.Count)
			{
				HCompList.IsCheckAll = true;
			}
			else
			{
				string[] selectedCurves3 = HCompList.GetSelectedCurves();
				HCompList.IsCheckAll = false;
				HCompList.SetSelectedCurves(selectedCurves3);
			}
			if (num3 > 0 && (num > 0 || num2 > 0))
			{
				e.CanExecute = true;
			}
			else
			{
				e.CanExecute = false;
			}
		}

		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = null;
			bool flag = false;
			_bRateSep = false;
			iHNode = _context.FindNode("Output/RATESEP");
			if (iHNode != null && (((int?)(dynamic)iHNode.get_Value(Type.Missing)) ?? 0) == 2)
			{
				_bRateSep = true;
			}
			if (_bRateSep)
			{
				_bHasInputDialog = true;
				for (int i = 0; i < _varNames.Count(); i++)
				{
					iHNode = _context.FindNode(_varNames[i]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_checkBoxList[i].IsEnabled = true;
						_checkBoxList[i].IsChecked = true;
						flag = true;
					}
				}
			}
			else
			{
				_bHasInputDialog = false;
				iHNode = _context.FindNode(_varNames[0]);
				if (iHNode != null && iHNode.Elements.Count > 0)
				{
					flag = true;
				}
			}
			if (!flag)
			{
				ierror = 1;
				return;
			}
			if (!string.IsNullOrEmpty(_currentConfigValue) && _bRateSep)
			{
				RestoreConfigValue(_currentConfigValue);
			}
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.AddDependentVariable(_context, _varNames[0]);
		}

		public void UpdateWizardData()
		{
			if (_bRateSep)
			{
				_wizardPlot.RemoveAllDependentVariable();
				int num = 0;
				for (int i = 0; i < _checkBoxList.Count; i++)
				{
					if (_checkBoxList[i].IsChecked != true)
					{
						continue;
					}
					IHNode iHNode = _context.FindNode(_varNames[i]);
					if (iHNode != null && iHNode.Elements.Count > 0)
					{
						_wizardPlot.PlotVal.NVariables = num + 1;
						_wizardPlot.SetIndependentVariable(_context, _varNames[i] + "/!Label");
						((IHPlotVal)_wizardPlot.PlotVal).set_Label(num, _checkBoxList[i].Content.ToString());
						for (int j = 0; j < iHNode.Elements.Count; j++)
						{
							((IHPlotVal)_wizardPlot.PlotVal).set_YData(num, j, (short)1, (object)iHNode.Elements[j, Type.Missing, Type.Missing, Type.Missing, Type.Missing]);
						}
						num++;
					}
				}
			}
			else
			{
				_wizardPlot.SetIndependentVariable(_context, _varNames[0] + "!Label");
				_wizardPlot.RemoveAllDependentVariable();
				_wizardPlot.AddDependentVariable(_context, _varNames[0]);
			}
			_help.SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentTempUnit);
			if (_bRateSep)
			{
				_currentConfigValue = SaveConfigValue();
			}
		}

			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();

			stringBuilder.Append("Temperature=");

			{
				if (_checkBoxList[i].IsChecked == true)
				{
					if (num > 0)
					{
						stringBuilder.Append(",");

					stringBuilder.Append(i.ToString());
