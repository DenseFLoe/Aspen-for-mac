		void IRibbonPlotWizardType.Context(object HNode, ref object HWizardPlot, ref int ierror)
		{
			if (HWizardPlot == null || HNode == null)
			{
				ierror = 1;
				return;
			}
			_context = HNode as IHNode;
			_wizardPlot = HWizardPlot as IHWizardPlot;
			IHNode iHNode = _context.FindNode(_varName[0]);
			if (iHNode == null || iHNode.Elements.Count <= 0)
			{
				ierror = 1;
				return;
			}
			ComboBoxXAxis.Items.Clear();
			ComboBoxParaVar.Items.Clear();
			List<string> list = new List<string>();
			CurveList.IsCheckAll = false;
			int count = iHNode.Elements.Count;
			for (int i = 0; i < count; i++)
			{
				string text = "";
				for (int j = 0; j < _varName.GetLength(0); j++)
				{
					string path = _varName[j] + "/" + (i + 1);
					string text2 = (dynamic)_context.FindNode(path).get_Value(Type.Missing);
					if (!string.IsNullOrEmpty(text2))
					{
						text2 = text2.Trim();
						text += text2;
						text += " ";
					}
				}
				text = text.Trim();
				ComboBoxXAxis.Items.Add(text);
				ComboBoxParaVar.Items.Add(text);
				list.Add(text);
				CurveList.SetCurves(list.ToArray());
			}
			int num = _context.FindNode("Input/VARY_VARTYPE").Elements.Count;
			IHNode iHNode2 = _context.FindNode("Input/DISABLE");
			for (int k = 0; k < num; k++)
			{
				if ((dynamic)iHNode2.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Value(Type.Missing) == "YES")
				{
					num--;
				}
			}
			chkParaVar.IsChecked = false;
			if (string.IsNullOrEmpty(_currentConfigValue))
			{
				ComboBoxXAxis.Text = ComboBoxXAxis.Items[0].ToString();
				if (ComboBoxParaVar.Items.Count > 1 && num > 1)
				{
					ComboBoxParaVar.Text = ComboBoxParaVar.Items[1].ToString();
					chkParaVar.IsChecked = true;
				}
				int num2 = 0;
				List<string> list2 = new List<string>();
				for (int l = num; l < count; l++)
				{
					if (num2 >= 5)
					{
						break;
					}
					list2.Add(list[l].ToString());
					num2++;
				}
				CurveList.SetSelectedCurves(list2.ToArray());
			}
			if (!string.IsNullOrEmpty(_currentConfigValue))
			{
				RestoreConfigValue(_currentConfigValue);
			}
		}

		public void UpdateWizardData()
		{
			int selectedIndex = ComboBoxXAxis.SelectedIndex;
			string indepvar = "Output/SENSVAR2/" + (selectedIndex + 1);
			_wizardPlot.SetIndependentVariable(_context, indepvar);
			_wizardPlot.RemoveAllDependentVariable();
			_wizardPlot.PlotVal.Parametric = false;
			if (chkParaVar.IsChecked == true)
			{
				_wizardPlot.PlotVal.Parametric = true;
				int selectedIndex2 = ComboBoxParaVar.SelectedIndex;
				string depvar = "Output/SENSVAR2/" + (selectedIndex2 + 1);
				_wizardPlot.AddDependentVariable(_context, depvar);
			}
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			List<string> list2 = new List<string>();
			int count = list.Count;
			for (int i = 0; i < count; i++)
			{
				string text = list[i];
				int selectedCurveIndex = CurveList.GetSelectedCurveIndex(text);
				string depvar2 = "Output/SENSVAR2/" + (selectedCurveIndex + 1);
				_wizardPlot.AddDependentVariable(_context, depvar2);
				((IHPlotVal)_wizardPlot.PlotVal).set_AxisMap(i, i);
				((IHPlotVal)_wizardPlot.PlotVal).set_YAxisTitle((object)i, text);
				list2.Add(text);
				if (chkParaVar.IsChecked == true)
				{
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(i + 1, ComboBoxParaVar.Text + " = %param%");
				}
				else
				{
					((IHPlotVal)_wizardPlot.PlotVal).set_Label(i, text);
				}
			}
			if (chkParaVar.IsChecked != true)
			{
				for (int j = 0; j < count; j++)
				{
					((IHPlotVal)_wizardPlot.PlotVal).set_AxisMap(j, count - j - 1);
				}
			}
			else if (list2.Contains(ComboBoxParaVar.Text))
			{
				for (int k = 0; k < count; k++)
				{
					((IHPlotVal)_wizardPlot.PlotVal).set_AxisMap(k, count - k);
				}
			}
			_wizardPlot.PlotVal.Title = WpfPLSensitivityResources.Title;
			_wizardPlot.PlotVal.XAxisTitle = ComboBoxXAxis.Text;
			_currentConfigValue = SaveConfigValue();
		}

		private void CanExecuteOKCommand(object sender, CanExecuteRoutedEventArgs e)
		{
			int selectedIndex = ComboBoxXAxis.SelectedIndex;
			int num = CurveList.GetSelectedCurves().Count();
			if (num == CurveList.Curves.Count)
			{
				CurveList.IsCheckAll = true;
			}
			else
			{
				string[] selectedCurves = CurveList.GetSelectedCurves();
				CurveList.IsCheckAll = false;
				CurveList.SetSelectedCurves(selectedCurves);
			}
			if (selectedIndex >= 0 && num > 0)
			{
				e.CanExecute = true;
			}
			else
			{
				e.CanExecute = false;
			}
		}

		private string SaveConfigValue()
		{
			_ = string.Empty;
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("X-Axis=");
			stringBuilder.Append(ComboBoxXAxis.Text.ToString());
			stringBuilder.Append(";");
			stringBuilder.Append("ParamatricChecked=");
			if (chkParaVar.IsChecked == true)
			{
				stringBuilder.Append(1);
			}
			else
			{
				stringBuilder.Append(0);
			}
			stringBuilder.Append(";");
			stringBuilder.Append("Parametric=");
			stringBuilder.Append(ComboBoxParaVar.Text.ToString());
			stringBuilder.Append(";");
			stringBuilder.Append("CurveSelection=");
			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			int count = list.Count;
			for (int i = 0; i < count; i++)
			{
				if (i > 0)
				{
					stringBuilder.Append(",");
				}
				string value = list[i];
				stringBuilder.Append(value);
			}
			stringBuilder.Append(";");
			stringBuilder.Append("CheckAll=");
			stringBuilder.Append(CurveList.IsCheckAll);
			stringBuilder.Append(";");
			return stringBuilder.ToString();
		}

			string text = array[num].Substring(array[num].IndexOf('=') + 1);
			ComboBoxXAxis.Text = text;

			text = array[num].Substring(array[num].IndexOf('=') + 1);

			{
				chkParaVar.IsChecked = true;
			}
			else
			{
				chkParaVar.IsChecked = false;
			}
			num++;
			text = array[num].Substring(array[num].IndexOf('=') + 1);
			ComboBoxParaVar.Text = text;
			num++;
			text = array[num].Substring(array[num].IndexOf('=') + 1);
			string[] array2 = text.Split(',');
			List<string> list = new List<string>();
			string[] array3 = array2;
			foreach (string item in array3)
			{

			}
			CurveList.SetSelectedCurves(list.ToArray());
