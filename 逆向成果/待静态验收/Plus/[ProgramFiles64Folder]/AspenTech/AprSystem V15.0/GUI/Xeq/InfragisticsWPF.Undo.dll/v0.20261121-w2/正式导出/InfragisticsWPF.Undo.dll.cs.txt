public class UndoExecuteContext
{
	private UndoManager _undoManager;

	private UndoExecuteReason _reason;

	private StackList<UndoUnit> _executeStack = new StackList<UndoUnit>();

	public UndoHistoryItemType ExecuteItemType
	{
		get
		{
			switch (_reason)
			{
			case UndoExecuteReason.Undo:
				return UndoHistoryItemType.Undo;
			case UndoExecuteReason.Redo:
				return UndoHistoryItemType.Redo;
			case UndoExecuteReason.Rollback:
				if (_undoManager.IsPerformingUndo)
				{
					return UndoHistoryItemType.Redo;
				}
				return UndoHistoryItemType.Undo;
			default:
				return UndoHistoryItemType.Undo;
			}
		}
	}

	public UndoExecuteReason Reason => _reason;

	internal int UnitCount => _executeStack.Count;

	public UndoManager UndoManager => _undoManager;

	internal UndoExecuteContext(UndoManager undoManager, UndoExecuteReason reason)
	{
		CoreUtilities.ValidateNotNull(undoManager, "undoManager");
		_undoManager = undoManager;
		_reason = reason;
	}

	internal UndoUnit GetUnit(int offset)
	{
		if (offset < 0 || offset > UnitCount)
		{
			return null;
		}
		return _executeStack.GetItem(offset);
	}

	internal bool Execute(UndoUnit unit)
	{
		CoreUtilities.ValidateNotNull(unit, "unit");
		_executeStack.Push(unit);
		try
		{
			return unit.Execute(this);
		}
		finally
		{
			_executeStack.Pop();
		}
	}
}

	{
		bool flag = _itemType == UndoHistoryItemType.Redo;
		int num = (flag ? _undoManager.RedoHistory : _undoManager.UndoHistory).IndexOf(this);
		if (num >= 0)
		{
			_undoManager.PerformUndoRedo(num + 1, !flag);
		}
		else if (validate)
		{
			throw new InvalidOperationException(Utils.GetString("LE_HistoryItemNotInCurrentHistory"));
		}
	}

		{
			_shortDescription = null;
			OnPropertyChanged("ShortDescription");
		}
		if (_longDescription != null)

			OnPropertyChanged("LongDescription");
		}
	}

	internal void PerformUndoRedo(int count, bool isUndo)
	{
		if (count <= 0)
		{
			throw new ArgumentOutOfRangeException("count", Utils.GetString("LE_ArgumentIsNegative", "count", count));
		}
		if (IsPerformingRedo || IsPerformingUndo)
		{
			throw new InvalidOperationException(Utils.GetString("LE_UndoRedoInUndoRedo"));
		}
		if (IsPerformingRollback)
		{
			throw new InvalidOperationException(Utils.GetString("LE_UndoRedoInRollback"));
		}
		if (IsSuspended)
		{
			throw new InvalidOperationException(Utils.GetString("LE_UndoRedoWhileSuspended"));
		}
		if (_rootTransaction != null)
		{
			throw new InvalidOperationException(Utils.GetString("LE_UndoRedoInTransaction"));
		}
		VerifyCanAffectHistory();
		if (!GetBool(isUndo ? BoolProperties.CanUndo : BoolProperties.CanRedo))
		{
			return;
		}
		BoolProperties property = (isUndo ? BoolProperties.IsPerformingUndo : BoolProperties.IsPerformingRedo);
		UndoExecuteReason reason = ((!isUndo) ? UndoExecuteReason.Redo : UndoExecuteReason.Undo);
		try
		{
			SetBool(property, value: true);
			ClearCachedRedoStack();
			StackList<UndoHistoryItem> stackList = (isUndo ? _undoStack : _redoStack);
			UndoExecuteContext undoExecuteContext = new UndoExecuteContext(this, reason);
			if (stackList.Count > 0)
			{
				ShouldPreventMerge = true;
			}
			int i = 0;
			for (int num = Math.Min(count, stackList.Count); i < num; i++)
			{
				UndoHistoryItem undoHistoryItem = (CurrentHistoryItem = stackList.Pop());
				undoExecuteContext.Execute(undoHistoryItem.Unit);
			}
		}
		finally
		{
			CurrentHistoryItem = null;
			SetBool(property, value: false);
		}
	}

	private UndoUnit AddChangeHelper(UndoUnit unit, bool allowTransaction = false)
	{
		if (unit == null)
		{
			return null;
		}
		if (IsPerformingRollback)
		{
			return null;
		}
		if (IsSuspended)
		{
			return null;
		}
		VerifyCanAffectHistory();
		if (unit is UndoTransaction { OpenState: var openState })
		{
			if (openState != false)
			{
				throw new ArgumentException(Utils.GetString("LE_AddOpenTransaction"));
			}
			if (!allowTransaction)
			{
				throw new ArgumentException(Utils.GetString("LE_AddTransactionDirect"));
			}
		}
		if (_rootTransaction != null)
		{
			UndoTransaction currentTransaction = CurrentTransaction;
			bool flag = IsPerformingRedo || IsPerformingUndo;
			if (!flag)
			{
				UndoMergeAction undoMergeAction = Merge(currentTransaction, unit);
				if (undoMergeAction != UndoMergeAction.NotMerged)
				{
					if (undoMergeAction == UndoMergeAction.MergedRemoveUnit)
					{
						currentTransaction.Cancel();
					}
					return null;
				}
				ShouldPreventMerge = false;
			}
			currentTransaction.Add(unit);
			if (!flag)
			{
				ResetLastNewOperationTime();
			}
		}
		else if (IsPerformingUndo)
		{
			ClearCachedRedoStack();
			_redoStack.Push(new UndoHistoryItem(unit, UndoHistoryItemType.Redo, this));
		}
		else
		{
			bool flag2 = true;
			bool isPerformingRedo = IsPerformingRedo;
			if (!isPerformingRedo && _undoStack.Count > 0)
			{
				UndoHistoryItem undoHistoryItem = _undoStack.Peek();
				UndoMergeAction undoMergeAction2 = Merge(undoHistoryItem.Unit, unit);
				if (undoMergeAction2 != UndoMergeAction.NotMerged)
				{
					ResetLastNewOperationTime();
					flag2 = false;
					if (undoMergeAction2 == UndoMergeAction.Merged)
					{
						undoHistoryItem.ResetDescription();
					}
					else
					{
						_undoStack.Pop();
						RestoreCachedRedoStack();
						PreventMerge();
					}
				}
			}
			ClearCachedRedoStack();
			if (flag2)
			{
				if (!isPerformingRedo)
				{
					CacheRedoStack();
					_redoStack.Clear();
					ShouldPreventMerge = false;
				}
				UndoHistoryItem item = new UndoHistoryItem(unit, UndoHistoryItemType.Undo, this);
				_undoStack.Push(item);
				if (!isPerformingRedo)
				{
					ResetLastNewOperationTime();
				}
			}
		}
		return unit;
	}

	private UndoMergeAction Merge(UndoUnit previousUnit, UndoUnit unitToMerge)
	{
		UndoMergeAction undoMergeAction = UndoMergeAction.NotMerged;
		if (AllowMergingResolved)
		{
			bool flag = _mergeTimeout == TimeSpan.Zero;
			if (!flag)
			{
				flag = DateTime.UtcNow - _lastNewOperation <= _mergeTimeout;
			}
			if (flag)
			{
				try
				{
					IsMerging = true;
					UndoMergeContext mergeInfo = new UndoMergeContext(this, unitToMerge);
					undoMergeAction = previousUnit.Merge(mergeInfo);
				}
				finally
				{
					IsMerging = false;
				}
				if (undoMergeAction != UndoMergeAction.NotMerged)
				{
					ResetLastNewOperationTime();
				}
			}
		}
		return undoMergeAction;
	}

		if (!RegisteredUndoManagers.TryGetValue(reference, out var value) && validate)
		{
			throw new ArgumentException(Utils.GetString("LE_ReferenceNotRegistered", reference));
		}
		return value;
	}

	public void RegisterReference(object reference)
	{
		if (reference is UndoManager)
		{
			throw new ArgumentException(Utils.GetString("LE_UndoManagerAsReference"), "reference");
		}
		WeakDictionary<object, UndoManager> registeredUndoManagers = RegisteredUndoManagers;
		if (registeredUndoManagers.TryGetValue(reference, out var value))
		{
			if (value != this)
			{
				throw new ArgumentException(Utils.GetString("LE_ReferenceRegisteredToOther", reference), "reference");
			}
		}
		else
		{
			registeredUndoManagers[reference] = this;
		}
	}

	public void UnregisterReference(object reference)
	{
		WeakDictionary<object, UndoManager> registeredUndoManagers = RegisteredUndoManagers;
		if (!registeredUndoManagers.TryGetValue(reference, out var value))
		{
			throw new ArgumentException(Utils.GetString("LE_ReferenceNotRegistered"), "reference");
		}
		if (value != this)
		{
			throw new ArgumentException(Utils.GetString("LE_ReferenceRegisteredToOther"), "reference");
		}
		registeredUndoManagers.Remove(reference);
	}

	private UndoUnit AddPropertyChangeHelper<TProperty>(UndoUnit unit, bool? preventMerge)
	{
		if (unit == null)
		{
			return null;
		}
		if (preventMerge != false)
		{
			PreventPropertyMerge(preventMerge, typeof(TProperty));
		}
		return AddChangeHelper(unit);
	}

	private void PreventPropertyMerge(bool? preventMerge, Type propertyType)
	{
		if (preventMerge ?? ShouldPreventPropertyMerge(propertyType))
		{
			PreventMerge();
		}
	}

	private bool ShouldAddPropertyChange<TProperty>(TProperty oldValue, TProperty newValue)
	{
		if (EqualityComparer<TProperty>.Default.Equals(oldValue, newValue))
		{
			return false;
		}
		if (IsSuspended)
		{
			return false;
		}
		return true;
	}

	private static bool ShouldPreventPropertyMerge(Type propertyType)
	{
		propertyType = CoreUtilities.GetUnderlyingType(propertyType);
		return propertyType == typeof(bool);
	}

	public UndoUnit AddPropertyChange(DependencyObject owner, DependencyPropertyChangedEventArgs propertyChangedArgs, bool? preventMerge = null, string propertyDisplayName = null, string typeDisplayName = null)
	{
		if (IsSuspended)
		{
			return null;
		}
		CoreUtilities.ValidateNotNull(propertyChangedArgs, "propertyChangedArgs");
		PropertyChangeUndoUnitBase propertyChangeUndoUnitBase = UndoUnitFactoryResolved.CreatePropertyChange(owner, propertyChangedArgs, propertyDisplayName, typeDisplayName);

			return null;
		}
		if (preventMerge != false)
		{
			Type propertyType = propertyChangedArgs.Property.PropertyType;
			PreventPropertyMerge(preventMerge, propertyType);
		}
		return AddChangeHelper(propertyChangeUndoUnitBase);
	}

[DebuggerDisplay("{DebuggerDisplay,nq}")]
public class UndoTransaction : UndoUnit, IUndoTransactionOwner
{
	private StackList<UndoUnit> _unitsSource;

	private ReadOnlyCollection<UndoUnit> _units;

	private IUndoTransactionOwner _parent;

	private bool? _isOpened;

	private UndoTransaction _openTransaction;

	private string _description;

	private string _descriptionDetailed;

	public override object Target => null;

	public bool IsClosed => _isOpened == false;

	public UndoTransaction OpenTransaction => _openTransaction;

	public UndoTransaction Parent => _parent as UndoTransaction;

	public IList<UndoUnit> Units => _units;

	internal bool? OpenState => _isOpened;

	internal IUndoTransactionOwner Owner => _parent;

	private string DebuggerDisplay => string.Format("{1}: {0} items", _units.Count, GetType().Name);

	protected string Description => _description;

	protected string DetailedDescription => _descriptionDetailed;

	UndoManager IUndoTransactionOwner.UndoManager => _parent.UndoManager;

	protected internal UndoTransaction(string description, string detailedDescription)
	{
		_description = description;
		_descriptionDetailed = detailedDescription;
		_unitsSource = new StackList<UndoUnit>();
		_units = new ReadOnlyCollection<UndoUnit>(_unitsSource);
	}

	protected internal override bool Execute(UndoExecuteContext executeInfo)
	{
		if (_openTransaction != null)
		{
			throw new InvalidOperationException(Utils.GetString("LE_CannotExecuteOpenTransaction", _openTransaction));
		}
		bool result = false;
		UndoTransaction undoTransaction = StartTransaction(executeInfo);
		try
		{
			foreach (UndoUnit item in _unitsSource)
			{
				if (executeInfo.Execute(item))
				{
					result = true;
				}
			}
			return result;
		}
		finally
		{
			if (undoTransaction != null && !undoTransaction.IsClosed)
			{
				undoTransaction.Commit();
			}
		}
	}

	internal override void ForEach(Action<UndoUnit> action)
	{
		base.ForEach(action);
		foreach (UndoUnit unit in _units)
		{
			action(unit);
			unit.ForEach(action);
		}
	}

	public override string GetDescription(UndoHistoryItemType itemType, bool detailed)
	{
		string text = (detailed ? _descriptionDetailed : _description);
		if (text == null)
		{
			if (_unitsSource.Count > 0)
			{
				foreach (UndoUnit item in _unitsSource)
				{
					_ = item;
					text = _unitsSource.Peek().GetDescription(itemType, detailed);
					if (text != null)
					{
						break;
					}
				}
			}
			if (text == null)
			{
				text = Utils.GetString("FallbackTransactionDescription");
			}
		}
		return text;
	}

	protected internal override UndoMergeAction Merge(UndoMergeContext mergeInfo)
	{
		UndoMergeAction undoMergeAction = UndoMergeAction.NotMerged;
		bool? isOpened = _isOpened;
		bool flag = true;
		if (isOpened == true == flag && isOpened.HasValue && _unitsSource.Count > 0)
		{
			undoMergeAction = _unitsSource.Peek().Merge(mergeInfo);
			if (undoMergeAction == UndoMergeAction.MergedRemoveUnit)
			{
				_unitsSource.Pop();
				undoMergeAction = UndoMergeAction.Merged;
			}
		}
		return undoMergeAction;
	}

	internal override bool RemoveAll(Func<UndoUnit, bool> match)
	{
		_unitsSource.RemoveAll(match);
		return _unitsSource.Count == 0;
	}

	private void Close(UndoTransactionCloseAction closeAction)
	{
		VerifyCanModify();
		UndoManager undoManager = _parent.UndoManager;
		undoManager.VerifyCanClose(this);
		if (_openTransaction != null)
		{
			_openTransaction.Close(closeAction);
		}
		_isOpened = false;
		if (closeAction == UndoTransactionCloseAction.Rollback)
		{
			undoManager.Rollback(this);
		}
		_parent.OnChildClosed(this, closeAction);
		undoManager.OnTransactionOpenedOrClosed(this);
	}

	private UndoTransaction StartTransaction(UndoExecuteContext executeInfo)
	{
		switch (executeInfo.Reason)
		{
		case UndoExecuteReason.Rollback:
			return null;
		case UndoExecuteReason.Undo:
		case UndoExecuteReason.Redo:
		{
			UndoTransaction undoTransaction = CreateExecuteTransaction(executeInfo);
			if (undoTransaction != null)
			{
				executeInfo.UndoManager.StartTransaction(undoTransaction);
			}
			return undoTransaction;
		}
		default:
			return null;
		}
	}

	private void VerifyCanModify()
	{
		if (!_isOpened.HasValue)
		{
			throw new InvalidOperationException(Utils.GetString("LE_TransactionNotStarted"));
		}
		if (_isOpened == false)
		{
			throw new InvalidOperationException(Utils.GetString("LE_TransactionClosed"));
		}
	}

	internal void Add(UndoUnit unit)
	{
		CoreUtilities.ValidateNotNull(unit, "unit");
		VerifyCanModify();
		if (_openTransaction != null)
		{
			throw new InvalidOperationException(Utils.GetString("LE_AddUnitWhileOpenTransaction", _openTransaction));
		}
		_unitsSource.Push(unit);
	}

	internal void Cancel()
	{
		Close(UndoTransactionCloseAction.Cancel);
	}

	internal bool ContainsNonTransactionUnits()
	{
		foreach (UndoUnit item in _unitsSource)
		{
			if (item is UndoTransaction undoTransaction)
			{
				if (undoTransaction.ContainsNonTransactionUnits())
				{
					return true;
				}
				continue;
			}
			return true;
		}
		return false;
	}

	internal static UndoTransaction GetLeafOpenGroup(UndoTransaction group)
	{
		CoreUtilities.ValidateNotNull(group, "group");
		if (group.OpenState != true)
		{
			throw new ArgumentException(Utils.GetString("LE_TransactionNotOpened", group));
		}
		while (group.OpenTransaction != null)
		{
			group = group.OpenTransaction;
		}
		return group;
	}

	internal void Open(IUndoTransactionOwner parent)
	{
		CoreUtilities.ValidateNotNull(parent, "parent");
		if (_isOpened.HasValue)
		{
			throw new InvalidOperationException(Utils.GetString("LE_TransactionAlreadyOpened"));
		}
		_parent = parent;
		_isOpened = true;
		if (!parent.OnChildOpened(this))
		{
			_isOpened = null;
			_parent = null;
		}
		else
		{
			parent.UndoManager.OnTransactionOpenedOrClosed(this);
		}
	}

	public void Commit()
	{
		Close(UndoTransactionCloseAction.Commit);
	}

	public void Rollback()
	{
		Close(UndoTransactionCloseAction.Rollback);
	}

	protected virtual UndoTransaction CreateExecuteTransaction(UndoExecuteContext executeInfo)
	{
		return executeInfo.UndoManager.CreateTransaction(_description, _descriptionDetailed);
	}

	bool IUndoTransactionOwner.OnChildOpened(UndoTransaction child)
	{
		CoreUtilities.ValidateNotNull(child, "child");
		VerifyCanModify();
		if (_openTransaction != null)
		{
			throw new InvalidOperationException(Utils.GetString("LE_HasOpenTransaction"));
		}
		if (!_unitsSource.Contains(child))
		{
			throw new ArgumentException(Utils.GetString("LE_ChildTransactionNotInUnits", child));
		}
		if (child.OpenState != true)
		{
			throw new ArgumentException(Utils.GetString("LE_TransactionNotOpened", child));
		}
		if (child.Owner != this)
		{
			throw new ArgumentException(Utils.GetString("LE_InvalidTransactionOwner"));
		}
		_openTransaction = child;
		return true;
	}

	void IUndoTransactionOwner.OnChildClosed(UndoTransaction child, UndoTransactionCloseAction closeAction)
	{
		CoreUtilities.ValidateNotNull(child, "child");
		VerifyCanModify();
		if (child != _openTransaction)
		{
			throw new ArgumentException(Utils.GetString("LE_ClosingOtherTransaction", child, _openTransaction));
		}
		_openTransaction = null;
		if (closeAction != UndoTransactionCloseAction.Commit || !child.ContainsNonTransactionUnits())
		{
			_unitsSource.Pop();
		}
	}
}

			{

					bool flag = index + _items.Length <= list.Count;

					{
						EqualityComparer<T> equalityComparer = EqualityComparer<T>.Default;

					}
					if (flag)
					{
						if (list is ObservableCollectionExtended<T> observableCollectionExtended)
						{
							observableCollectionExtended.RemoveRange(index, _items.Length);
						}
						else
						{
							ExecuteInTransaction(collection, executeInfo, delegate

								for (int j = 0; j < _items.Length; j++)

							});

					else

							if (num2 < 0)

								return false;
							}
							indexes[num] = num2;
						}
						Array.Sort(indexes);
						ExecuteInTransaction(collection, executeInfo, delegate
						{
							for (int num3 = indexes.Length - 1; num3 >= 0; num3--)
							{
								list.RemoveAt(indexes[num3]);

					});
				}

			}
			ExecuteInTransaction(collection, executeInfo, delegate
			{

	private class ResetChange : CollectionChange
	{

		internal ResetChange(T[] items)
		{
			_items = items;
		}

		internal override bool Execute(ICollection<T> collection, UndoExecuteContext executeInfo)
		{
			if (collection is ObservableCollectionExtended<T> observableCollectionExtended)
			{
				observableCollectionExtended.ReInitialize(_items);
			}
			else if (_items == null || _items.Length == 0)
			{
				collection.Clear();
			}
			else
			{
				ExecuteInTransaction(collection, executeInfo, delegate
				{
					collection.Clear();
					if (collection is List<T> list)
					{
						list.AddRange(_items);
					}
					else
					{
						T[] items = _items;
						foreach (T item in items)
						{
							collection.Add(item);
						}
					}
				});
			}
			return true;
		}

		internal override string GetDescription(ICollection<T> collection, UndoHistoryItemType itemType, bool detailed, string itemTypeDisplayName)
		{
			return Utils.GetString(detailed ? "ReinitializeCollectionDescriptionDetailed" : "ReinitializeCollectionDescription", collection, _items, itemTypeDisplayName ?? typeof(T).Name);
		}

		public override string ToString()
		{
			return $"Reset: Count={((_items != null) ? _items.Length : 0)}";
		}
	}

	private ICollection<T> _collection;

	private CollectionChange _change;

	private string _itemTypeDisplayName;

	public override object Target => _collection;

	public ICollection<T> Collection => _collection;

	private string DebuggerDisplay => _change.ToString();

	public CollectionChangeUndoUnit(ICollection<T> collection, NotifyCollectionChangedEventArgs changedArgs, string itemTypeDisplayName)
		: this(collection, CreateChange(changedArgs), itemTypeDisplayName)
	{
	}

	public CollectionChangeUndoUnit(ICollection<T> collection, NotifyCollectionChangedAction action, T changedItem, int index, string itemTypeDisplayName)
		: this(collection, CreateChange(changedItem, index, IsAddRemove(action)), itemTypeDisplayName)
	{
	}

	public CollectionChangeUndoUnit(ICollection<T> collection, NotifyCollectionChangedAction action, T[] items, int index, string itemTypeDisplayName)
		: this(collection, (CollectionChange)new AddRemoveBlock(items, index, IsAddRemove(action)), itemTypeDisplayName)
	{
	}

	public CollectionChangeUndoUnit(ICollection<T> collection, T[] items, string itemTypeDisplayName)
		: this(collection, (CollectionChange)new ResetChange(items), itemTypeDisplayName)
	{
	}

	public CollectionChangeUndoUnit(ICollection<T> collection, T oldItem, T newItem, int index, string itemTypeDisplayName)
		: this(collection, (CollectionChange)new Replacement(oldItem, newItem, index), itemTypeDisplayName)
	{
	}

	public CollectionChangeUndoUnit(ICollection<T> collection, T item, int oldIndex, int newIndex, string itemTypeDisplayName)
		: this(collection, (CollectionChange)new MoveItem(item, oldIndex, newIndex), itemTypeDisplayName)
	{
	}

	private CollectionChangeUndoUnit(ICollection<T> collection, CollectionChange change, string itemTypeDisplayName)
	{
		CoreUtilities.ValidateNotNull(collection);
		if (collection.IsReadOnly)
		{
			throw new ArgumentException(Utils.GetString("LE_TargetCollectionIsReadOnly", collection));
		}
		_collection = collection;
		_change = change;
		_itemTypeDisplayName = itemTypeDisplayName;
	}

	protected internal override bool Execute(UndoExecuteContext executeInfo)
	{
		if (_change != null)
		{
			return _change.Execute(_collection, executeInfo);
		}
		return false;
	}

	public override string GetDescription(UndoHistoryItemType itemType, bool detailed)
	{
		if (_change == null)
		{
			return null;
		}
		return _change.GetDescription(_collection, itemType, detailed, _itemTypeDisplayName);
	}

	protected internal override UndoMergeAction Merge(UndoMergeContext mergeInfo)
	{
		if (mergeInfo.UnitBeingAdded is CollectionChangeUndoUnit<T> collectionChangeUndoUnit && collectionChangeUndoUnit._collection == _collection && _change != null)
		{
			AddRemoveSingle addRemoveSingle = _change as AddRemoveSingle;
			AddRemoveSingle addRemoveSingle2 = collectionChangeUndoUnit._change as AddRemoveSingle;
			if (addRemoveSingle != null && addRemoveSingle2 != null && addRemoveSingle.WasAdd && addRemoveSingle.WasAdd != addRemoveSingle2.WasAdd && addRemoveSingle.Index == addRemoveSingle2.Index && EqualityComparer<T>.Default.Equals(addRemoveSingle.Item, addRemoveSingle2.Item))
			{
				_change = null;
				return UndoMergeAction.MergedRemoveUnit;
			}
		}
		return base.Merge(mergeInfo);
	}

	private static CollectionChange CreateChange(NotifyCollectionChangedEventArgs changedArgs)
	{
		CoreUtilities.ValidateNotNull(changedArgs, "changeArgs");
		switch (changedArgs.Action)
		{
		default:
			throw new InvalidOperationException(Utils.GetString("LE_ResetCollectionAction"));
		case NotifyCollectionChangedAction.Move:
			if (changedArgs.NewItems.Count != 1 || changedArgs.OldItems.Count != 1)
			{
				throw new NotSupportedException(Utils.GetString("LE_RangeCollectionAction"));
			}
			return new MoveItem((T)changedArgs.OldItems[0], changedArgs.OldStartingIndex, changedArgs.NewStartingIndex);
		case NotifyCollectionChangedAction.Replace:
			if (changedArgs.NewItems.Count != 1 || changedArgs.OldItems.Count != 1)
			{
				throw new NotSupportedException(Utils.GetString("LE_RangeCollectionAction"));
			}
			return new Replacement((T)changedArgs.OldItems[0], (T)changedArgs.NewItems[0], changedArgs.OldStartingIndex);
		case NotifyCollectionChangedAction.Add:
			if (changedArgs.NewItems.Count != 1)
			{
				throw new NotSupportedException(Utils.GetString("LE_RangeCollectionAction"));
			}
			return CreateChange((T)changedArgs.NewItems[0], changedArgs.NewStartingIndex, wasAdd: true);
		case NotifyCollectionChangedAction.Remove:
			if (changedArgs.OldItems.Count != 1)
			{
				throw new NotSupportedException(Utils.GetString("LE_RangeCollectionAction"));
			}
			return CreateChange((T)changedArgs.OldItems[0], changedArgs.OldStartingIndex, wasAdd: false);
		}
	}

	private static CollectionChange CreateChange(T item, int index, bool wasAdd)
	{
		return new AddRemoveSingle
		{
			Item = item,
			Index = index,
			WasAdd = wasAdd
		};
	}

	private static bool IsAddRemove(NotifyCollectionChangedAction action)
	{
		return action switch
		{
			NotifyCollectionChangedAction.Add => true, 
			NotifyCollectionChangedAction.Remove => false, 
			_ => throw new ArgumentException(Utils.GetString("LE_NeedAddRemoveAction"), "action"), 
		};
	}
}

	{
		CoreUtilities.ValidateNotNull(target, "target");
		CoreUtilities.ValidateNotNull(getSetProxy, "getSetter");
		CoreUtilities.ValidateNotNull(getSetProxy.Setter, "setter");
		_target = new WeakReference(target);
		_oldValue = oldValue;
		_newValue = newValue;
		_getSetProxy = getSetProxy;
		_propertyDisplayName = propertyDisplayName;
		_targetDisplayName = targetDisplayName;

		TTarget actualTarget = ActualTarget;
		if (actualTarget == null)
		{
			return false;
		}
		_getSetProxy.Setter(actualTarget, _oldValue);
		return true;
	}

	protected internal override UndoMergeAction Merge(UndoMergeContext mergeInfo)
	{
		if (mergeInfo.UnitBeingAdded is PropertyChangeUndoUnit<TTarget, TProperty> { ActualTarget: var actualTarget } propertyChangeUndoUnit)
		{
			TTarget actualTarget2 = ActualTarget;
			if (actualTarget == actualTarget2 && actualTarget != null && _getSetProxy.Equals(propertyChangeUndoUnit._getSetProxy) && !EqualityComparer<TProperty>.Default.Equals(_oldValue, propertyChangeUndoUnit._newValue))
			{
				_newValue = propertyChangeUndoUnit._newValue;
				return UndoMergeAction.Merged;
			}
		}
		return base.Merge(mergeInfo);
	}

public abstract class UndoManagerCommandBase : CommandBase
{
	public override bool CanExecute(object parameter)
	{
		return GetUndoManager(parameter) != null;
	}

	public override void Execute(object parameter)
	{
		UndoManager undoManager = GetUndoManager(parameter);
		if (undoManager != null && Execute(undoManager) && base.CommandSource != null)
		{
			base.CommandSource.Handled = true;
		}
	}

	protected abstract bool Execute(UndoManager manager);

	internal static UndoManager GetUndoManager(object parameter)
	{
		UndoManager undoManager = parameter as UndoManager;
		if (undoManager == null)
		{
			undoManager = UndoManager.FromReference(parameter);
		}
		return undoManager;
	}
}

public class ObservableCollectionExtendedWithUndo<T> : ObservableCollectionExtended<T>
{
	private UndoManager _undoManager;

	private T[] _originalItems;

	private string _itemTypeDisplayName;

	public UndoManager UndoManager => _undoManager;

	public ObservableCollectionExtendedWithUndo(string itemTypeDisplayName = null)
		: this(UndoManager.Current, itemTypeDisplayName)
	{
	}

	public ObservableCollectionExtendedWithUndo(UndoManager undoManager, string itemTypeDisplayName = null)
	{
		CoreUtilities.ValidateNotNull(undoManager, "undoManager");
		_itemTypeDisplayName = itemTypeDisplayName;
		_undoManager = undoManager;
	}

	protected override void ClearItems()
	{
		if (!base.IsUpdating)
		{
			T[] array = new T[base.Count];
			base.Items.CopyTo(array, 0);
			UndoManager.AddCollectionChange(this, NotifyCollectionChangedAction.Remove, array, 0, _itemTypeDisplayName);
		}
		base.ClearItems();
	}

	protected override void InsertItem(int index, T item)
	{
		if (!base.IsUpdating)
		{
			UndoManager.AddCollectionChange(this, NotifyCollectionChangedAction.Add, item, index, _itemTypeDisplayName);
		}
		base.InsertItem(index, item);
	}

	public override void InsertRange(int index, IEnumerable<T> collection)
	{
		if (!base.IsUpdating)
		{
			T[] dest = new T[CoreUtilities.GetCount(collection)];
			CoreUtilities.CopyTo(collection, dest, 0);
			UndoManager.AddCollectionChange(this, NotifyCollectionChangedAction.Add, dest, index, _itemTypeDisplayName);
		}
		base.InsertRange(index, collection);
	}

	protected override void MoveItem(int oldIndex, int newIndex)
	{
		if (!base.IsUpdating)
		{
			T item = base[oldIndex];
			UndoManager.AddCollectionChange(this, item, oldIndex, newIndex, _itemTypeDisplayName);
		}
		base.MoveItem(oldIndex, newIndex);
	}

	protected override void OnBeginUpdate()
	{
		_originalItems = new T[base.Count];
		CopyTo(_originalItems, 0);
		base.OnBeginUpdate();
	}

	protected override void OnEndUpdate()
	{
		AddEndUpdateSnapshot();
		base.OnEndUpdate();
	}

	protected override void RemoveItem(int index)
	{
		T changedItem = base[index];
		if (!base.IsUpdating)
		{
			UndoManager.AddCollectionChange(this, NotifyCollectionChangedAction.Remove, changedItem, index, _itemTypeDisplayName);
		}
		base.RemoveItem(index);
	}

	public override void RemoveRange(int index, int count)
	{
		if (index == 0 && count == base.Count)
		{
			Clear();
			return;
		}
		if (!base.IsUpdating)
		{
			T[] array = new T[count];
			if (base.Items is List<T> list)
			{
				list.CopyTo(index, array, 0, count);
			}
			else
			{
				for (int i = 0; i < count; i++)
				{
					array[i] = base[i + index];
				}
			}
			UndoManager.AddCollectionChange(this, NotifyCollectionChangedAction.Remove, array, index, _itemTypeDisplayName);
		}
		base.RemoveRange(index, count);
	}

	protected override void SetItem(int index, T item)
	{
		if (!base.IsUpdating)
		{
			T oldItem = base[index];
			UndoManager.AddCollectionChange(this, oldItem, item, index, _itemTypeDisplayName);
		}
		base.SetItem(index, item);
	}

	private void AddEndUpdateSnapshot()
	{
		T[] originalItems = _originalItems;
		_originalItems = null;
		if (originalItems != null && !CoreUtilities.AreEqual(originalItems, this))
		{
			UndoManager.AddCollectionChange(this, originalItems, _itemTypeDisplayName);
		}
	}
}
