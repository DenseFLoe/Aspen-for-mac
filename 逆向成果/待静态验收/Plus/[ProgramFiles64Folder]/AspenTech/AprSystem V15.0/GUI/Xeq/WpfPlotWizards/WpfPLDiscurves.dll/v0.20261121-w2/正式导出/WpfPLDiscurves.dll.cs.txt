		private static string GetAdsPathForStream(string streamName)
		{
			if (!string.IsNullOrEmpty(streamName))
			{
				List<string> list = (from x in streamName.Split('.')
					where !string.IsNullOrWhiteSpace(x)
					select x).ToList();
				if (list.Count > 1)
				{
					StringBuilder stringBuilder = new StringBuilder("Flowsheet.");
					for (int num = 0; num < list.Count - 1; num++)
					{
						stringBuilder.Append("Blocks.").Append(list[num]).Append(".SubFlowsheets.Simulation.");
					}
					stringBuilder.Append("Streams.").Append(list[list.Count - 1]);
					return stringBuilder.ToString();
				}
			}
			return $"Flowsheet.Streams.{streamName}";
		}

		{

			List<string> list = new List<string>(CurveList.GetSelectedCurves());
			if (list.Count > 0)
			{
				for (int i = 0; i < 31; i++)

					if (!(SelectedType == CurveName[i]))

						continue;
					}
					_wizardPlot.SetIndependentVariable(Context, XVarNames[i]);
					IHNode iHNode = Context.FindNode(YVarNames[i]);
					int count = iHNode.Elements.Count;
					for (int j = 0; j < list.Count; j++)
					{
						for (int k = 0; k < count; k++)
						{
							string text = iHNode.Elements[k, Type.Missing, Type.Missing, Type.Missing, Type.Missing].get_Name(Type.Missing);
							if (list[j] == text)
							{
								string depvar = YVarNames[i] + "/" + text;

								((IHPlotVal)_wizardPlot.PlotVal).set_Label(j, text);
								_wizardPlot.PlotVal.Title = SelectedType;
								break;
							}
						}
					}
					break;
				}
			}
			_help.SaveUnitsComboText(_context.Application.Tree, _wizardPlot, _currentUnit);
			_currentConfigValue = SaveConfigValue();
		}

		private void CreateTheoreticalMesh(int iFunctionType, double dblPositionParameter, double dblShapeParameter, double dblDMinUser, double dblDMaxUser, ref List<double> meshData)
		{
			double num;
			double num2;
			switch (iFunctionType)
			{
			case 0:
				num = dblPositionParameter;
				num2 = ((dblShapeParameter == 0.0) ? 0.0 : (Math.Pow(1E-05, 1.0 / dblShapeParameter) * dblPositionParameter));
				break;
			case 1:
				num = ((!(dblShapeParameter >= 0.01)) ? (5.0 * dblDMaxUser) : (Math.Pow(-1.0 * Math.Log(9.99999999995449E-06), 1.0 / dblShapeParameter) * dblPositionParameter));
				num2 = Math.Pow(-1.0 * Math.Log(0.99999), 1.0 / dblShapeParameter) * dblPositionParameter;
				break;
			case 2:
				num = dblPositionParameter + 4.1 * dblShapeParameter;
				num2 = Math.Max(dblPositionParameter - 4.1 * dblShapeParameter, 0.0);
				break;
			case 3:
				num = Math.Exp(Math.Log(dblPositionParameter) + 4.0 * dblShapeParameter);
				num2 = Math.Max(Math.Exp(Math.Log(dblPositionParameter) - 4.0 * dblShapeParameter), 0.0);
				break;
			default:
				return;
			}
			if (num > 10.0 * dblDMaxUser)
			{
				num = 5.0 * dblDMaxUser;
			}
			List<double> list = new List<double>();
			double num3 = (num - num2) / 199.0;
			for (int i = 0; i < 200; i++)
			{
				list.Add(num3 * (double)i + num2);
			}
			if (iFunctionType == 0)
			{
				list.Add(num + 1E-06);
			}
			if (num < dblDMaxUser - num3)
			{
				list.Add(dblDMaxUser - num3);
				list.Add(dblDMaxUser);
			}
			else if (num < dblDMaxUser)
			{
				list.Add(dblDMaxUser - num3);
			}
			if (num2 > dblDMinUser + num3)
			{
				list.Insert(0, dblDMinUser + num3);
				list.Insert(0, dblDMinUser);
			}
			else if (num2 > dblDMinUser)
			{
				list.Insert(0, dblDMinUser);
			}
			meshData = list;
		}

		public void CreateFraction(int iFunctionType, double dblPositionParameter, double dblShapeParameter, List<double> meshData, ref List<double> lstMDY, ref List<double> lstCMY)
		{
			lstCMY.Clear();
			lstMDY.Clear();
			double num = 0.0;
			double num2 = 0.0;
			double num3 = 0.0;
			switch (iFunctionType)
			{
			case 0:
			{
				num3 = ((!(meshData[0] >= 0.0) || !(meshData[0] <= dblPositionParameter)) ? 1.0 : Math.Pow(meshData[0] / dblPositionParameter, dblShapeParameter));
				lstCMY.Add(num3);
				for (int k = 1; k < meshData.Count(); k++)
				{
					num2 = ((!(meshData[k] >= 0.0) || !(meshData[k] <= dblPositionParameter)) ? 1.0 : Math.Pow(meshData[k] / dblPositionParameter, dblShapeParameter));
					num = num2 - num3;
					num3 = num2;
					if (meshData[k] - meshData[k - 1] > 0.0)
					{
						double item3 = num / (meshData[k] - meshData[k - 1]);
						lstMDY.Add(item3);
					}
					else
					{
						lstMDY.Add(0.0);
					}
					lstCMY.Add(num2);
				}
				break;
			}
			case 1:
			{
				if (dblPositionParameter <= 0.0)
				{
					break;
				}
				num3 = 1.0 - Math.Exp(-1.0 * Math.Pow(meshData[0] / dblPositionParameter, dblShapeParameter));
				lstCMY.Add(num3);
				for (int j = 1; j < meshData.Count(); j++)
				{
					num2 = 1.0 - Math.Exp(-1.0 * Math.Pow(meshData[j] / dblPositionParameter, dblShapeParameter));
					num = num2 - num3;
					num3 = num2;
					if (meshData[j] - meshData[j - 1] > 0.0)
					{
						double item2 = num / (meshData[j] - meshData[j - 1]);
						lstMDY.Add(item2);
					}
					else
					{
						lstMDY.Add(0.0);
					}
					lstCMY.Add(num2);
				}
				break;
			}
			case 2:
			{
				if (dblShapeParameter == 0.0)
				{
					break;
				}
				double dblX2 = (meshData[0] - dblPositionParameter) / (Math.Sqrt(2.0) * dblShapeParameter);
				num3 = 0.5 * (1.0 + Errf(dblX2));
				lstCMY.Add(num3);
				for (int l = 1; l < meshData.Count(); l++)
				{
					dblX2 = (meshData[l] - dblPositionParameter) / (Math.Sqrt(2.0) * dblShapeParameter);
					num2 = 0.5 * (1.0 + Errf(dblX2));
					num = num2 - num3;
					num3 = num2;
					if (meshData[l] - meshData[l - 1] > 0.0)
					{
						double item4 = num / (meshData[l] - meshData[l - 1]);
						lstMDY.Add(item4);
					}
					else
					{
						lstMDY.Add(0.0);
					}
					lstCMY.Add(num2);
				}
				break;
			}
			case 3:
			{
				if (dblShapeParameter == 0.0)
				{
					break;
				}
				double dblX = (Math.Log(meshData[0]) - Math.Log(dblPositionParameter)) / (Math.Sqrt(2.0) * dblShapeParameter);
