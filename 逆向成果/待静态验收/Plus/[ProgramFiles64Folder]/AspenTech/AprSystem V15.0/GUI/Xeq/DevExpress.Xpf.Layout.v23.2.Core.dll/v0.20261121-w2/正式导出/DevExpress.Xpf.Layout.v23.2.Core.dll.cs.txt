	public static class NotificationHelper
	{
		internal static Dictionary<object, Tracker<NotificationEventArgs>> trackers = new Dictionary<object, Tracker<NotificationEventArgs>>();

		public static IDisposable Subscribe(object source, IObserver<NotificationEventArgs> observer)
		{
			if (observer == null || source == null)
			{
				return null;
			}
			lock (trackers)
			{
				if (!trackers.TryGetValue(source, out var value))
				{
					CollectUnusedTrackers(trackers);
					value = new Tracker<NotificationEventArgs>();
					trackers.Add(source, value);
				}
				return value.Subscribe(observer);
			}
		}

		public static void Notify(object source, NotificationEventArgs action)
		{
			lock (trackers)
			{
				if (trackers.TryGetValue(source, out var value))
				{
					if (value.InUse)
					{
						value.Notify(action);
					}
					else
					{
						trackers.Remove(source);
					}
				}
			}
		}

		public static void EndNotification(object source)
		{
			lock (trackers)
			{
				if (trackers.TryGetValue(source, out var value))
				{
					value.EndNotification();
				}
				trackers.Remove(source);
			}
		}

		private static void CollectUnusedTrackers(IDictionary<object, Tracker<NotificationEventArgs>> trackers)
		{
			ICollection<object> collection = new List<object>();
			foreach (KeyValuePair<object, Tracker<NotificationEventArgs>> tracker in trackers)
			{
				if (!tracker.Value.InUse)
				{
					collection.Add(tracker.Key);
				}
			}
			foreach (object item in collection)
			{
				trackers.Remove(item);
			}
		}
	}
	internal class Tracker<T> : IObservable<T>
	{
		private class Unsubscriber : IDisposable
		{
			private ICollection<IObserver<T>> observers;

			private IObserver<T> observer;

			public Unsubscriber(ICollection<IObserver<T>> observers, IObserver<T> observer)
			{
				this.observers = observers;
				this.observer = observer;
			}

			public void Dispose()
			{
				if (observer != null && observers.Contains(observer))
				{
					observers.Remove(observer);
				}
			}
		}

		private ICollection<IObserver<T>> observers;

		public bool InUse => observers.Count > 0;

		public Tracker()
		{
			observers = new List<IObserver<T>>();
		}

		public IDisposable Subscribe(IObserver<T> observer)
		{
			if (!observers.Contains(observer))
			{
				observers.Add(observer);
			}
			return new Unsubscriber(observers, observer);
		}

		public void Notify(T element)
		{
			Exception ex = CheckError(element);
			IObserver<T>[] array = GetObservers();
			IObserver<T>[] array2 = array;
			foreach (IObserver<T> observer in array2)
			{
				if (ex != null)
				{
					observer.OnError(ex);
				}
				else
				{
					observer.OnNext(element);
				}
			}
		}

		protected virtual Exception CheckError(T element)
		{
			return null;
		}

		public void EndNotification()
		{
			IObserver<T>[] array = GetObservers();
			IObserver<T>[] array2 = array;
			foreach (IObserver<T> observer in array2)
			{
				if (observers.Contains(observer))
				{
					observer.OnCompleted();
				}
			}
			observers.Clear();
		}

		private IObserver<T>[] GetObservers()
		{
			IObserver<T>[] array = new IObserver<T>[observers.Count];
			observers.CopyTo(array, 0);
			return array;
		}
	}

	public static class ResizeHelper
	{
		private class ResizeInfo : BaseResizeInfo
		{
			private ResizeType hResize;

			private ResizeType vResize;

			private bool fHorizontal;

			private bool fVertical;

			private bool fChangeLocationX;

			private bool fChangeLocationY;

			private double corner = 10.0;

			public override ResizeType HResizeType => hResize;

			public override ResizeType VResizeType => vResize;

			public override bool Horizontal => fHorizontal;

			public override bool Vertical => fVertical;

			public override bool ChangeLocationX => fChangeLocationX;

			public override bool ChangeLocationY => fChangeLocationY;

			public ResizeInfo(Rect rect, Point pt)
				: base(rect, pt)
			{
				Calc(rect, pt);
			}

			public ResizeInfo(SizingAction sa)
				: base(default(Rect), default(Point))
			{
				vResize = sa.ToResizeType(isHorizontal: false);
				hResize = sa.ToResizeType(isHorizontal: true);
				fHorizontal = HResizeType == ResizeType.Left || HResizeType == ResizeType.Right;
				fVertical = VResizeType == ResizeType.Top || VResizeType == ResizeType.Bottom;
				fChangeLocationX = HResizeType == ResizeType.Left;
				fChangeLocationY = VResizeType == ResizeType.Top;
			}

			private void Calc(Rect r, Point p)
			{
				hResize = ResizeType.None;
				vResize = ResizeType.None;
				bool flag = p.Y < r.Top + r.Height * 0.5;
				bool flag2 = p.X < r.Left + r.Width * 0.5;
				bool flag3 = p.X < r.Left + corner && p.Y < r.Top + corner;
				bool flag4 = p.X > r.Right - corner && p.Y < r.Top + corner;
				bool flag5 = p.X > r.Right - corner && p.Y > r.Bottom - corner;
				bool flag6 = p.X < r.Left + corner && p.Y > r.Bottom - corner;
				bool flag7 = flag3 || flag4 || flag5 || flag6;
				bool flag8 = p.Y - p.X < r.Top - r.Left;
				bool flag9 = p.Y + p.X < r.Top + r.Right;
				bool flag10 = p.Y - p.X > r.Bottom - r.Right;
				bool flag11 = p.Y + p.X > r.Bottom + r.Left;
				if (flag7)
				{
					if (flag3)
					{
						vResize = ResizeType.Top;
						hResize = ResizeType.Left;
					}
					if (flag4)
					{
						vResize = ResizeType.Top;
						hResize = ResizeType.Right;
					}
					if (flag5)
					{
						vResize = ResizeType.Bottom;
						hResize = ResizeType.Right;
					}
					if (flag6)
					{
						vResize = ResizeType.Bottom;
						hResize = ResizeType.Left;
					}
				}
				else
				{
					if (flag8 && flag9 && flag)
					{
						vResize = ResizeType.Top;
					}
					if (flag10 && flag11 && !flag)
					{
						vResize = ResizeType.Bottom;
					}
					if (!flag8 && !flag11 && flag2)
					{
						hResize = ResizeType.Left;
					}
					if (!flag9 && !flag10 && !flag2)
					{
						hResize = ResizeType.Right;
					}
				}
				fHorizontal = HResizeType == ResizeType.Left || HResizeType == ResizeType.Right;
				fVertical = VResizeType == ResizeType.Top || VResizeType == ResizeType.Bottom;
				fChangeLocationX = HResizeType == ResizeType.Left;
				fChangeLocationY = VResizeType == ResizeType.Top;
			}

			bool changeLocationY = baseResizeInfo.ChangeLocationY;

			if (MathHelper.IsConstraintValid(maxSize.Height))

	public static class DropTypeHelper
	{
		private class DropInfo : BaseDropInfo
		{
			private DropType dropTypeCore;

			private bool fHorizontal;

			private bool fVertical;

			private Rect dropRectCore;

			public override DropType Type => dropTypeCore;

			public override bool Horizontal => fHorizontal;

			public override bool Vertical => fVertical;

			public override Rect DropRect => dropRectCore;

			public DropInfo(Rect rect, Point pt)
				: base(rect, pt)
			{
				if (base.ItemRect.Contains(base.DropPoint))
				{
					CalcWithCenterZone(base.ItemRect, base.DropPoint, 0.0);
				}
				else
				{
					CalcWithCenterZone(base.ItemRect, base.DropPoint);
				}
			}

			public DropInfo(Rect rect, Point pt, double factor, bool useCenterZoneCalculation)
				: base(rect, pt)
			{
				if (useCenterZoneCalculation)
				{
					CalcWithCenterZone(base.ItemRect, base.DropPoint, factor);
				}
				else
				{
					CalcWithRectangularSideZones(base.ItemRect, base.DropPoint, factor);
				}
			}

			private static DropType CalcDropType(Rect r, Point p)
			{
				DropType result = DropType.None;
				double num = ((r.Width != 0.0 && r.Height != 0.0) ? (r.Height / r.Width) : 1.0);
				double num2 = r.Left + r.Width / 2.0;
				double num3 = r.Top + r.Height / 2.0;
				double num4 = p.X - num2;
				double num5 = p.Y - num3;
				bool flag = Math.Abs(num5) <= (0.0 - num) * num4;
				bool flag2 = 0.0 - num5 >= num * Math.Abs(num4);
				bool flag3 = Math.Abs(num5) <= num * num4;
				bool flag4 = num5 >= num * Math.Abs(num4);
				if (flag)
				{
					result = DropType.Left;
				}
				else if (flag2)
				{
					result = DropType.Top;
				}
				else if (flag3)
				{
					result = DropType.Right;
				}
				else if (flag4)
				{
					result = DropType.Bottom;
				}
				return result;
			}

			private void CalcWithCenterZone(Rect r, Point p)
			{
				dropTypeCore = CalcDropType(r, p);
				switch (dropTypeCore)
				{
				case DropType.Top:
					dropRectCore = new Rect(r.Left, r.Top, r.Width, r.Height * 0.5);
					break;
				case DropType.Bottom:
					dropRectCore = new Rect(r.Left, r.Top + r.Height * 0.5, r.Width, r.Height * 0.5);
					break;
				case DropType.Left:
					dropRectCore = new Rect(r.Left, r.Top, r.Width * 0.5, r.Height);
					break;
				case DropType.Right:
					dropRectCore = new Rect(r.Left + r.Width * 0.5, r.Top, r.Width * 0.5, r.Height);
					break;
				}
				fHorizontal = Type == DropType.Left || Type == DropType.Right;
				fVertical = Type == DropType.Top || Type == DropType.Bottom;
			}

			private void CalcWithCenterZone(Rect r, Point p, double centerFactor)
			{
				dropTypeCore = DropType.None;
				if (!r.Contains(p))
				{
					return;
				}
				if (centerFactor != 0.0)
				{
					Rect rect = new Rect(r.Left + r.Width * 0.5 * (1.0 - centerFactor), r.Top + r.Height * 0.5 * (1.0 - centerFactor), r.Width * centerFactor, r.Height * centerFactor);
					if (rect.Contains(p))
					{
						dropTypeCore = DropType.Center;
						dropRectCore = rect;
						return;
					}
				}
				CalcWithCenterZone(r, p);
			}

			private void CalcWithRectangularSideZones(Rect r, Point p, double sideFactor)
			{
				dropTypeCore = DropType.None;
				if (!r.Contains(p))
				{
					return;
				}
				double num = r.Width * 0.5 * sideFactor;
				Rect rect = new Rect(r.Left, r.Top, num, r.Height);
				Rect rect2 = new Rect(r.Right - num, r.Top, num, r.Height);
				fHorizontal = true;
				if (rect.Contains(p))
				{
					dropTypeCore = DropType.Left;
					dropRectCore = new Rect(r.Left, r.Top, r.Width * 0.5, r.Height);
					return;
				}
				if (rect2.Contains(p))
				{
					dropTypeCore = DropType.Right;
					dropRectCore = new Rect(r.Left + r.Width * 0.5, r.Top, r.Width * 0.5, r.Height);
					return;
				}
				fHorizontal = false;
				fVertical = true;
				if (p.Y < r.Top + r.Height * 0.5)
				{
					dropTypeCore = DropType.Top;
					dropRectCore = new Rect(r.Left, r.Top, r.Width, r.Height * 0.5);
				}
				else
				{
					dropTypeCore = DropType.Bottom;
					dropRectCore = new Rect(r.Left, r.Top + r.Height * 0.5, r.Width, r.Height * 0.5);
				}
			}
		}

	public static class PlacementHelper

		public static Rect Arrange(Size size, Rect targetRect, Alignment alignment, Point offset)
		{
			if (alignment == Alignment.Fill)
			{
				return targetRect;
			}
			double x = GetLeft(size.Width, targetRect, alignment) + offset.X;
			double y = GetTop(size.Height, targetRect, alignment) + offset.Y;
			return new Rect(new Point(x, y), size);
		}

		public static Rect Arrange(Size size, Rect targetRect, Alignment alignment)
		{
			if (alignment == Alignment.Fill)
			{
				return targetRect;
			}
			double left = GetLeft(size.Width, targetRect, alignment);
			double top = GetTop(size.Height, targetRect, alignment);
			return new Rect(new Point(left, top), size);
		}

		private static double GetLeft(double width, Rect targetRect, Alignment alignment)
		{
			double result = targetRect.Left;
			switch (alignment)
			{
			case Alignment.TopCenter:
			case Alignment.MiddleCenter:
			case Alignment.BottomCenter:
				result = MathHelper.CenterRange(targetRect.Left, targetRect.Width, width);
				break;
			case Alignment.TopRight:
			case Alignment.MiddleRight:
			case Alignment.BottomRight:
				result = targetRect.Right - width;
				break;
			}
			return result;
		}

		private static double GetTop(double height, Rect targetRect, Alignment alignment)
		{
			double result = targetRect.Top;
			switch (alignment)
			{
			case Alignment.MiddleCenter:
			case Alignment.MiddleLeft:
			case Alignment.MiddleRight:
				result = MathHelper.CenterRange(targetRect.Top, targetRect.Height, height);
				break;
			case Alignment.BottomCenter:
			case Alignment.BottomLeft:
			case Alignment.BottomRight:
				result = targetRect.Bottom - height;
				break;
			}
			return result;
		}

	public static class MathHelper
	{
		public const double Epsilon = 1E-10;

		public static bool IsEmpty(double value)

				return IsZero(value);
			}
			return true;
		}

		{
			if (!double.IsNaN(value) && !double.IsInfinity(value))
			{
				return IsDimensionValid(value);
			}
			return false;
		}

		public static bool IsDimensionValid(double value)
		{
			return value > double.Epsilon;
		}

		public static bool IsZero(double value)
		{
			return IsZero(value, 1E-10);
		}

		public static double NormalizeConstraint(double value)
		{
			if (!IsConstraintValid(value))
			{
				return double.NaN;
			}
			return value;
		}

		public static Size MeasureSize(Size minSize, Size maxSize, Size measuredSize)

		public static Size MeasureMaxSize(Size[] maxSizes, bool fHorz)
		{
			double num = (fHorz ? 0.0 : double.MaxValue);
			double num2 = (fHorz ? double.MaxValue : 0.0);
			for (int i = 0; i < maxSizes.Length; i++)
			{
				if (fHorz)
				{
					num += CalcRealMaxConstraint(maxSizes[i].Width);
					num2 = Math.Min(num2, CalcRealMaxConstraint(maxSizes[i].Height));
				}
				else
				{
					num = Math.Min(num, CalcRealMaxConstraint(maxSizes[i].Width));
					num2 += CalcRealMaxConstraint(maxSizes[i].Height);
				}
			}
			if (num == double.MaxValue)
			{
				num = double.NaN;
			}
			if (num2 == double.MaxValue)
			{
				num2 = double.NaN;
			}
			return new Size(num, num2);
		}

		public static Size MeasureMinSize(Size[] minSizes, bool fHorz)
		{
			double num = 0.0;
			double num2 = 0.0;
			for (int i = 0; i < minSizes.Length; i++)
			{
				if (fHorz)
				{
					num += CalcRealMinConstraint(minSizes[i].Width);
					num2 = Math.Max(num2, CalcRealMinConstraint(minSizes[i].Height));
				}
				else
				{
					num = Math.Max(num, CalcRealMinConstraint(minSizes[i].Width));
					num2 += CalcRealMinConstraint(minSizes[i].Height);
				}
			}
			return new Size(num, num2);
		}

		public static Size MeasureMinSize(Size[] minSizes)
		{
			double num = 0.0;
			double num2 = 0.0;
			for (int i = 0; i < minSizes.Length; i++)
			{
				num = Math.Max(num, CalcRealMinConstraint(minSizes[i].Width));
				num2 = Math.Max(num2, CalcRealMinConstraint(minSizes[i].Height));
			}
			return new Size(num, num2);

		public static Size MeasureMaxSize(Size[] maxSizes)
		{
			double num = double.MaxValue;
			double num2 = double.MaxValue;
			for (int i = 0; i < maxSizes.Length; i++)
			{
				num = TopDimension(num, CalcRealMaxConstraint(maxSizes[i].Width));
				num2 = TopDimension(num2, CalcRealMaxConstraint(maxSizes[i].Height));

			if (num == double.MaxValue)
			{
				num = double.NaN;
			}
			if (num2 == double.MaxValue)
			{
				num2 = double.NaN;
			}
			return new Size(num, num2);

		public static double ValidateDimension(double min, double max, double dim)
		{
			if (double.IsNaN(dim))
			{
				return dim;
			}
			if (IsConstraintValid(max))
			{
				dim = Math.Min(max, dim);
			}
			if (IsConstraintValid(min))
			{
				dim = Math.Max(min, dim);
			}
			return dim;
		}

		public static double MeasureDimension(double min, double max, double dim)
		{
			if (IsConstraintValid(max))
			{
				dim = TopDimension(max, dim);
			}
			if (IsConstraintValid(min))
			{
				dim = Math.Max(min, dim);
			}
			return dim;
		}

		internal static double BottomDimension(double min, double dim)
		{
			if (double.IsNaN(dim))
			{
				return min;
			}
			if (!(min > dim))
			{
				return dim;
			}
			return min;
		}

		internal static double TopDimension(double max, double dim)
		{
			if (double.IsNaN(dim))
			{
				return max;
			}
			if (!(max < dim))
			{
				return dim;
			}
			return max;
		}

				return num;

			{
				return num;

			return 0.0;

				return new Rect(source.Left, source.Top, source.Width, Math.Max(0.0, target.Top - source.Top));

			return new Rect(source.Left, target.Bottom, source.Width, Math.Max(0.0, source.Bottom - target.Bottom));

		public static Rect CorrectBounds(Rect bounds, Rect constraints, Thickness threshold)
		{
			Rect result = bounds;
			if (constraints.Width * constraints.Height != 0.0)
			{
				double num = bounds.Left;
				double num2 = bounds.Top;
				if (bounds.Left + threshold.Right > constraints.Right)
				{
					num -= bounds.Left + threshold.Right - constraints.Right;
				}
				if (bounds.Top + threshold.Bottom > constraints.Bottom)
				{
					num2 -= bounds.Top + threshold.Bottom - constraints.Bottom;
				}
				if (num + threshold.Left < constraints.Left)
				{
					num -= num + threshold.Left - constraints.Left;
				}
				if (num2 + threshold.Top < constraints.Top)
				{
					num2 -= num2 + threshold.Top - constraints.Top;
				}
				result = new Rect(num, num2, bounds.Width, bounds.Height);
			}
			return result;
		}

		public static Size CorrectSize(Size size, Size desired)
		{
			Size result = size;
			if (desired.Width * desired.Height != 0.0)
			{
				result = new Size(Math.Max(desired.Width, size.Width), Math.Max(desired.Height, size.Height));
			}
			return result;
		}

	public abstract class BaseLayoutElement : BaseObject, ILayoutElement, IBaseObject, IDisposable, ISupportHierarchy<ILayoutElement>, ISupportVisitor<ILayoutElement>
	{
		protected static readonly ILayoutElement[] EmptyNodes = new ILayoutElement[0];

		internal ILayoutContainer parentCore;

		private ILayoutElement[] nodesCore;

		private bool isVisualStateInitialized;

		private object pressedHitResult;

		private object hotHitResult;

		private bool isDraggingCore;

		public bool IsDragging
		{
			get
			{
				return isDraggingCore;
			}
			set
			{
				if (isDraggingCore != value)
				{
					isDraggingCore = value;
					if (!isDraggingCore)
					{
						OnResetIsDragging();
					}
				}
			}
		}

		public virtual bool HitTestingEnabled => true;

		public virtual bool IsActive => true;

		public virtual bool IsEnabled => true;

		public bool IsReady { get; private set; }

		public virtual bool IsTabHeader => false;

		public Rect Bounds => ElementHelper.GetRect(this);

		public ILayoutContainer Container => parentCore;

		public ILayoutElement Parent => parentCore;

		public ILayoutElement[] Nodes
		{
			get
			{
				if (nodesCore == null)
				{
					nodesCore = GetNodesCore();
				}
				return nodesCore;
			}
		}

		public Point Location { get; protected internal set; }

		public Size Size { get; protected internal set; }

		protected override void OnDispose()
		{
			nodesCore = null;
			if (!IsDragging)
			{
				parentCore = null;
			}
			base.OnDispose();
		}

		protected virtual void OnResetIsDragging()
		{
			if (base.IsDisposing)
			{
				parentCore = null;
			}
		}

		public virtual void ResetState()
		{
			SetStateCore(ref pressedHitResult, null, State.Normal);
			SetStateCore(ref hotHitResult, null, State.Normal);
		}

		public bool HitTest(Point pt)
		{
			return HitTestCore(pt);
		}

		protected virtual bool HitTestCore(Point pt)
		{
			if (EnsureBounds())
			{
				return HitTestBounds(pt, Bounds);
			}
			return false;
		}

		protected virtual bool HitTestBounds(Point hitPoint, Rect bounds)
		{
			return bounds.Contains(hitPoint);
		}

		public void Invalidate()
		{
			IsReady = false;
		}

		public State GetState(object hitResult)
		{
			if (hitResult == null)
			{
				return State.Normal;
			}
			EnsureInitialState();
			State state = State.Normal;
			if (HitEquals(hitResult, pressedHitResult))
			{
				state |= State.Pressed;
			}
			if (HitEquals(hitResult, hotHitResult))
			{
				state |= State.Hot;
			}
			return state;
		}

		public void SetState(object hitResult, State state)
		{
			if (!base.IsDisposing)
			{
				EnsureInitialState();
				if ((state & State.Pressed) != State.Normal)
				{
					SetStateCore(ref pressedHitResult, hitResult, state);
				}
				if ((state & State.Hot) != State.Normal)
				{
					SetStateCore(ref hotHitResult, hitResult, state);
				}
			}
		}

		private void EnsureInitialState()
		{
			if (!isVisualStateInitialized)
			{
				isVisualStateInitialized = true;
				pressedHitResult = InitPressedState();
				hotHitResult = InitHotState();
			}
		}

		private void SetStateCore(ref object prevHitResult, object hitResult, State state)
		{
			if (!HitEquals(prevHitResult, hitResult))
			{
				object obj = prevHitResult;
				prevHitResult = hitResult;
				if (obj != null)
				{
					OnStateChanged(obj, GetState(obj));
				}
				if (hitResult != null)
				{
					OnStateChanged(hitResult, state);
				}
			}
		}

		protected virtual bool HitEquals(object prevHitResult, object hitResult)
		{
			return object.Equals(prevHitResult, hitResult);
		}

			}

		{
			LayoutElementHitInfo layoutElementHitInfo = null;
			using (IEnumerator<ILayoutElement> enumerator = GetEnumerator())

					BaseLayoutElement baseLayoutElement = (BaseLayoutElement)enumerator.Current;
					if (baseLayoutElement.HitTestingEnabled && baseLayoutElement.HitTest(pt))
					{

					}
				}
			}
			if (layoutElementHitInfo != null)
			{
				return layoutElementHitInfo;
			}
			return LayoutElementHitInfo.Empty;
		}

	public class LayoutElementEnumerator : IEnumerator<ILayoutElement>, IDisposable, IEnumerator
	{
		private Stack<ILayoutElement> Stack;

		private ILayoutElement Root;

		private ILayoutElement current;

		ILayoutElement IEnumerator<ILayoutElement>.Current => current;

		object IEnumerator.Current => current;

		public LayoutElementEnumerator(ILayoutElement root)
		{
			Stack = new Stack<ILayoutElement>(32);
			Root = root;
		}

		void IDisposable.Dispose()
		{
			ResetCore();
			Stack = null;
			Root = null;
			GC.SuppressFinalize(this);
		}

		bool IEnumerator.MoveNext()
		{
			if (current == null)
			{
				current = Root;
			}
			else
			{
				int num = current.Nodes.Length;
				if (num > 0)
				{
					for (int i = 0; i < num; i++)
					{
						ILayoutElement item = current.Nodes[num - 1 - i];
						Stack.Push(item);
					}
				}
				current = ((Stack.Count > 0) ? Stack.Pop() : null);
			}
			return current != null;
		}

		void IEnumerator.Reset()
		{
			ResetCore();
		}

		private void ResetCore()
		{
			if (Stack != null)
			{
				Stack.Clear();
			}
			current = null;
		}
	}
	public class IUIElementEnumerator : IEnumerator<IUIElement>, IDisposable, IEnumerator
	{
		private IUIElement Root;

		private Stack<IUIElement> Stack;

		private Predicate<IUIElement> Filter;

		private IUIElement current;

		object IEnumerator.Current => current;

		public IUIElement Current => current;

		public IUIElementEnumerator(IUIElement element)

		}

		public IUIElementEnumerator(IUIElement element, Predicate<IUIElement> filter)
		{
			Stack = new Stack<IUIElement>(8);
			Root = element;
			Filter = filter;
		}

		public void Dispose()
		{
			Reset();
			Root = null;
			Stack = null;
			Filter = null;
			GC.SuppressFinalize(this);
		}

		public bool MoveNext()
		{
			if (current == null)
			{
				current = Root;
			}
			else
			{
				if (current.Children != null)
				{
					IUIElement[] elements = current.Children.GetElements();
					if (elements.Length != 0)
					{
						for (int i = 0; i < elements.Length; i++)
						{
							IUIElement iUIElement = elements[elements.Length - 1 - i];
							if (Filter == null || Filter(iUIElement))
							{
								Stack.Push(iUIElement);
							}
						}
					}
				}
				current = ((Stack.Count > 0) ? Stack.Pop() : null);
			}
			return current != null;
		}

		public void Reset()
		{
			if (Stack != null)
			{
				Stack.Clear();
			}
			current = null;
		}

	public abstract class UILayoutElementFactory : ILayoutElementFactory
	{
		private class ElementHash
		{
			private class Sink
			{
				private IUIElement Key;

				private IDictionary<IUIElement, ILayoutElement> Hash;

				public Sink(IUIElement key, IDictionary<IUIElement, ILayoutElement> hash)
				{
					Key = key;
					Hash = hash;
				}

				public void Disposed(object sender, EventArgs e)
				{
					((ILayoutElement)sender).Disposed -= Disposed;
					Hash.Remove(Key);
					Hash = null;
					Key = null;
				}
			}

			private IDictionary<IUIElement, ILayoutElement> hash;

			public ILayoutElement this[IUIElement key] => hash[key];

			public int Count => hash.Count;

			public ElementHash()
			{
				hash = new Dictionary<IUIElement, ILayoutElement>();
			}

			public void Add(IUIElement key, ILayoutElement element)
			{
				hash.Add(key, element);
				element.Disposed += new Sink(key, hash).Disposed;
			}

			public bool TryGetValue(IUIElement key, out ILayoutElement element)
			{
				return hash.TryGetValue(key, out element);
			}

			public void Clear()
			{
				ILayoutElement[] array = new ILayoutElement[hash.Count];
				hash.Values.CopyTo(array, 0);
				for (int i = 0; i < array.Length; i++)
				{
					array[i].Dispose();
				}
			}
		}

		private readonly ElementHash Hash;

		public UILayoutElementFactory()
		{
			Hash = new ElementHash();
		}

		public ILayoutElement CreateLayoutHierarchy(object rootKey)
		{
			if (!(rootKey is IUIElement))
			{
				return null;
			}
			return CreateUIHierarchy((IUIElement)rootKey, Hash);
		}

		public ILayoutElement GetElement(object key)
		{
			if (!(key is IUIElement))
			{
				return null;
			}
			ILayoutElement element = null;
			if (!Hash.TryGetValue((IUIElement)key, out element))
			{
				return null;
			}
			return element;
		}

		private ILayoutElement CreateUIHierarchy(IUIElement rootKey, ElementHash hash)
		{
			if (hash.Count > 0)
			{
				hash.Clear();
			}
			if (rootKey == null)
			{
				return null;
			}
			using (IEnumerator<IUIElement> enumerator = GetUIEnumerator(rootKey))
			{
				while (enumerator.MoveNext())
				{
					IUIElement current = enumerator.Current;
					ILayoutElement element = CreateElement(current);
					ILayoutElement element2 = null;
					if (current.Scope != null && hash.TryGetValue(current.Scope, out element2) && element2 is BaseLayoutContainer baseLayoutContainer)
					{
						baseLayoutContainer.AddInternal(element);
					}
					hash.Add(current, element);
				}
			}
			return hash[rootKey];
		}

		protected abstract IEnumerator<IUIElement> GetUIEnumerator(IUIElement rootKey);

		protected abstract ILayoutElement CreateElement(IUIElement uiKey);
	}

	public delegate bool SelectionChangingHandler<T>(T item, bool selected);
	public delegate void SelectionChangedHandler<T>(T item, bool selected);
	public delegate T[] RequestSelectionRangeHandler<T>(T first, T last);
	public class SelectionInfo<T> : IDisposable
	{
		private bool isDisposing;

		private SelectionChangingHandler<T> SelectionChanging;

		private SelectionChangedHandler<T> SelectionChanged;

		private RequestSelectionRangeHandler<T> RequestSelectionRange;

		public SelectionMode Mode { get; set; }

		private IDictionary<object, T> Selection { get; set; }

		public T LastSelectedElement { get; private set; }

		public SelectionInfo(SelectionChangingHandler<T> selectionChanging, SelectionChangedHandler<T> selectionChanged, RequestSelectionRangeHandler<T> requestSelectionRange)
		{
			SelectionChanged = selectionChanged;
			SelectionChanging = selectionChanging;
			RequestSelectionRange = requestSelectionRange;
			Mode = SelectionMode.SingleItem;
			Selection = new Dictionary<object, T>();
		}

		public void Dispose()
		{
			if (!isDisposing)
			{
				isDisposing = true;
				SelectionChanged = null;
				RequestSelectionRange = null;
				ClearSelection();
				Selection = null;
			}
			GC.SuppressFinalize(this);
		}

		public void Select(T element)
		{
			if (element == null)
			{
				ClearSelection();
				return;
			}
			bool selectedState = GetSelectedState(element);
			switch (Mode)
			{
			case SelectionMode.SingleItem:
				ClearSelection(element);
				if (!selectedState && CanSelectionChanging(element, selected: true))
				{
					SelectElementCore(element);
				}
				break;
			case SelectionMode.MultipleItems:
				if (CanSelectionChanging(element, !selectedState))
				{
					if (!selectedState)
					{
						SelectElementCore(element);
					}
					else
					{
						UnSelectElementCore(element);
					}
				}
				break;
			case SelectionMode.ItemRange:
			{
				T[] array = RequestSelectionRange(LastSelectedElement, element);
				ClearSelection(array);
				foreach (T element2 in array)
				{
					if (!GetSelectedState(element2) && CanSelectionChanging(element2, selected: true))
					{
						SelectElementCore(element2);
					}
				}
				break;
			}
			}
			LastSelectedElement = element;
		}

		public bool GetSelectedState(T element)
		{
			object item = SelectionHelper.GetItem(element);
			if (item == null)
			{
				return true;
			}
			return Selection.ContainsKey(item);
		}

		private void SelectElementCore(T element)
		{
			Selection.Add(SelectionHelper.GetItem(element), element);
			RaiseSelectionChanged(element, selected: true);
		}

		private void UnSelectElementCore(T element)
		{
			Selection.Remove(SelectionHelper.GetItem(element));
			RaiseSelectionChanged(element, selected: false);
		}

		protected void ClearSelection()
		{
			T[] array = new T[Selection.Count];
			Selection.Values.CopyTo(array, 0);
			foreach (T element in array)
			{
				if (CanSelectionChanging(element, selected: false))
				{
					UnSelectElementCore(element);
				}
			}
		}

		protected void ClearSelection(T element)
		{
			T[] array = new T[Selection.Count];
			Selection.Values.CopyTo(array, 0);
			foreach (T element2 in array)
			{
				if (!object.Equals(SelectionHelper.GetItem(element2), SelectionHelper.GetItem(element)))
				{
					if (CanSelectionChanging(element2, selected: false))
					{
						UnSelectElementCore(element2);
					}
				}
				else
				{
					Selection[SelectionHelper.GetItem(element)] = element;
				}
			}
		}

		protected void ClearSelection(T[] range)
		{
			T[] array = new T[Selection.Count];
			Selection.Values.CopyTo(array, 0);
			object[] items = SelectionHelper.GetItems(range);
			foreach (T element in array)
			{
				int num = Array.IndexOf(items, SelectionHelper.GetItem(element));
				if (num == -1)
				{
					if (CanSelectionChanging(element, selected: false))
					{
						UnSelectElementCore(element);
					}
				}
				else
				{
					T val = range[num];
					Selection[SelectionHelper.GetItem(val)] = val;
				}
			}
		}

		protected bool CanSelectionChanging(T element, bool selected)
		{
			if (SelectionChanging != null)
			{
				return SelectionChanging(element, selected);
			}
			return true;
		}

		protected void RaiseSelectionChanged(T element, bool selected)
		{
			if (SelectionChanged != null)
			{
				SelectionChanged(element, selected);
			}
		}

		protected T[] RaiseRequestSelection(T first, T last)
		{
			T[] result = new T[0];
			if (RequestSelectionRange != null)
			{
				result = RequestSelectionRange(first, last);
			}
			return result;
		}
	}
	public static class SelectionHelper
	{
		public static T[] GetSelectionRange<T>(T[] baseRange, T first, T last) where T : class
		{
			object[] items = GetItems(baseRange);
			int num = Array.IndexOf(items, GetItem(first));
			int num2 = Array.IndexOf(items, GetItem(last));
			if (num == -1 && num2 == -1)
			{
				return new T[0];
			}
			if (num == -1 || num2 == -1)
			{
				return new T[1] { last };
			}
			int num3 = Math.Min(num, num2);
			int num4 = Math.Max(num, num2);
			T[] array = new T[num4 - num3 + 1];
			for (int i = num3; i < num4 + 1; i++)
			{
				array[i - num3] = baseRange[i];
			}
			return array;
		}

		public static object GetItem<T>(T element)
		{
			if (element is ISelectionKey selectionKey)
			{
				if (!selectionKey.AllowSelect)
				{
					return null;
				}
				return selectionKey.Item;
			}
			return element;
		}

		public static object GetViewKey<T>(T element)
		{
			if (!(element is ISelectionKey))
			{
				return element;
			}
			return (element as ISelectionKey).ViewKey;
		}

		public static object GetElementKey<T>(T element)
		{
			if (!(element is ISelectionKey))
			{
				return element;
			}
			return (element as ISelectionKey).ElementKey;
		}

		public static object[] GetItems<T>(T[] elements)
		{
			return Array.ConvertAll(elements, GetItem);
		}

			if (eventType != MouseEventType.MouseDown || ea.Buttons == MouseButtons.None)
			{
				return false;
			}
			if (ea.Buttons == MouseButtons.Right && element != null)
			{
				return !Selection.GetSelectedState(element);
			}
			return ea.Buttons == MouseButtons.Left;
		}

					case MouseEventType.MouseUp:

		{
			CheckResetPrevPressedElement(upHitInfo);
			bool flag = false;
			bool flag2 = false;
			if (ea.Buttons == MouseButtons.None && upHitInfo.Element != null)
			{
				if (ea.ChangedButtons == MouseButtons.Left)
				{
					if (upHitInfo.InClickPreviewBounds)

						DoClickPreviewAction(upHitInfo);
					}
					if (upHitInfo.InClickBounds || upHitInfo.InDoubleClickBounds)
					{
						flag = CheckClick(upHitInfo);
					}
				}
				if (ea.ChangedButtons == MouseButtons.Right && upHitInfo.InMenuBounds)
				{
					flag = DoMenuAction(upHitInfo);
					if (flag)
					{
						ea.Handled = true;
					}
				}
				flag2 = upHitInfo.Element.IsDisposing;
				if (flag)
				{
					prevHotInfo = (prevPressedInfo = null);
				}
			}
			else
			{
				fWaitForSecondClick = false;
			}
			if (!flag2)
			{
				SetElementPressedInfo(upHitInfo.Element, LayoutElementHitInfo.Empty);
				if (flag && upHitInfo.InControlBox)
				{
					SetActiveItem(base.Sender, upHitInfo.Element);
				}
			}
			OnMouseUp(upHitInfo);
			return flag;
		}

		protected bool ProcessMouseLeave()
		{
			CheckResetPrevHotElement(LayoutElementHitInfo.Empty);
			return false;
		}

		private bool CheckClick(LayoutElementHitInfo hi)
		{
			if (base.Sender != null && base.Sender.Adapter.GetIsDragging())
			{
				fWaitForSecondClick = false;
				return false;
			}
			if (fWaitForSecondClick)
			{
				if (hi.InDoubleClickBounds && IsDoubleClick(hi))
				{
					DoDoubleClickAction(hi);
					fWaitForSecondClick = false;
					return true;
				}
				SaveFirstClickInfo(hi);
			}
			else
			{
				SaveFirstClickInfo(hi);
			}
			if (hi.Element != null && (hi.Element.GetState(hi.HitResult) & State.Pressed) != State.Normal && hi.InClickBounds)
			{
				if (hi.Element != null)
				{
					hi.Element.ResetState();
				}
				if (DoClickAction(hi) || !hi.InDoubleClickBounds)
				{
					fWaitForSecondClick = false;
					return true;
				}
			}
			return false;
		}

		private bool IsDoubleClick(LayoutElementHitInfo hi)
		{
			if (base.Sender == lastClickView.Target && CheckLocation(hi.HitPoint))
			{
				return CheckTime(DateTime.Now.Ticks);
			}
			return false;
		}

		private bool CheckLocation(Point point)
		{
			bool flag = Math.Abs(point.X - lastClickPoint.X) <= (double)SystemInformation.DoubleClickWidth;
			bool flag2 = Math.Abs(point.Y - lastClickPoint.Y) <= (double)SystemInformation.DoubleClickHeight;
			return flag && flag2;
		}

		private bool CheckTime(long time)
		{
			return Math.Abs(lastClickTime - time) / 10000 <= SystemInformation.DoubleClickTime;
		}

		private void SaveFirstClickInfo(LayoutElementHitInfo hi)
		{
			lastClickPoint = hi.HitPoint;
			lastClickView = new WeakReference(base.Sender);
			lastClickTime = DateTime.Now.Ticks;
			fWaitForSecondClick = true;
		}

		{
			IUIInteractionServiceListener uIServiceListener = view.GetUIServiceListener<IUIInteractionServiceListener>();
			if (uIServiceListener == null || (ActiveItem == element && ActiveItem.IsActive) || !uIServiceListener.OnActiveItemChanging(element))
			{
				return;
			}
			UnSubscribe(ActiveItem);
			if (uIServiceListener.OnActiveItemChanged(element))
			{
				if (element != null && element.IsDisposing)
				{
					element = null;
				}
				activeItemCore = element;
				Subscribe(ActiveItem);
			}
			else
			{
				activeItemCore = null;
			}
		}

		private void Subscribe(ILayoutElement item)
		{
			if (item != null)
			{
				item.Disposed += OnActiveItemDisposed;
			}
		}

		private void UnSubscribe(ILayoutElement item)
		{
			if (item != null)
			{
				item.Disposed -= OnActiveItemDisposed;
			}
		}

		private void OnActiveItemDisposed(object sender, EventArgs e)
		{
			ILayoutElement layoutElement = sender as ILayoutElement;
			if (layoutElement == null)
			{
				layoutElement = ActiveItem;
			}
			UnSubscribe(layoutElement);
			if (prevHotInfo != null && prevHotInfo.Element == layoutElement)
			{
				prevHotInfo = null;
			}
			if (prevPressedInfo != null && prevPressedInfo.Element == layoutElement)
			{
				prevPressedInfo = null;
			}
			activeItemCore = null;
		}

		{
			if (prevHotInfo != null && prevHotInfo.Element != hitInfo.Element)
			{
				SetElementHotTrackedInfo(prevHotInfo.Element, LayoutElementHitInfo.Empty);
			}
			prevHotInfo = (hitInfo.IsHot ? hitInfo : null);
		}

		private void CheckResetPrevPressedElement(LayoutElementHitInfo hitInfo)
		{
			if (prevPressedInfo != null && prevPressedInfo.Element != hitInfo.Element)
			{
				SetElementPressedInfo(prevPressedInfo.Element, LayoutElementHitInfo.Empty);
			}
			prevPressedInfo = (hitInfo.IsPressed ? hitInfo : null);
			if (prevPressedInfo != null && (ActiveItem == null || prevPressedInfo.Element != ActiveItem))
			{
				ILayoutElement element = prevPressedInfo.Element;
				if (element != null)
				{
					element.Disposed += OnActiveItemDisposed;
				}
			}
		}

			element?.SetState(hitInfo.HitResult, State.Hot);

	public abstract class DragServiceState : IDragServiceState
	{
		private OperationType _lastInitializedOperation;

		public IDragService DragService { get; private set; }

		public abstract OperationType OperationType { get; }

		protected DragServiceState(IDragService service)
		{
			DragService = service;
		}

		public virtual void ProcessMouseDown(IView view, Point point)
		{
			DragService.SetState(OperationType.Regular);
		}

		public virtual void ProcessMouseUp(IView view, Point point)
		{
			DragService.SetState(OperationType.Regular);
		}

		public virtual void ProcessMouseMove(IView view, Point point)
		{
			DragService.SetState(OperationType.Regular);
		}

		public virtual void ProcessCancel(IView view)
		{
			Reset(view);
			if (OperationType != OperationType.Regular)
			{
				DoLeave(GetBehindView(), GetBehindOperation());
				DoCancel(view, OperationType);
			}
		}

		protected void Reset(IView view)
		{
			ResetCore();
			CancelInitializedOperation(view);
		}

		protected virtual void ResetCore()
		{
		}

		protected virtual bool ShouldCancelLastInitializedOperation(OperationType initializedOperation)
		{
			if (initializedOperation == OperationType.Regular || initializedOperation == OperationType.ClientDragging)
			{
				return false;
			}
			return true;
		}

		private void CancelInitializedOperation(IView view)
		{
			if (ShouldCancelLastInitializedOperation(_lastInitializedOperation))
			{
				view.GetUIServiceListener<IDragServiceListener>(_lastInitializedOperation)?.OnCancel();
				_lastInitializedOperation = OperationType.Regular;
			}
		}

		public virtual void ProcessKeyDown(IView view, Key key)
		{
			switch (key)
			{
			case Key.Escape:
				ProcessCancel(view);
				DragService.SetState(OperationType.Regular);
				break;
			case Key.LeftCtrl:
			case Key.RightCtrl:
				DoLeave(view, OperationType);
				DoLeave(GetBehindView(), GetBehindOperation());
				break;
			}
		}

		public virtual void ProcessKeyUp(IView view, Key key)
		{
			if (key == Key.LeftCtrl || key == Key.RightCtrl)
			{
				DoEnter(view, OperationType);
				DoEnter(GetBehindView(), GetBehindOperation());
			}
		}

		protected virtual IView GetBehindView()
		{
			return null;
		}

		protected virtual OperationType GetBehindOperation()
		{
			return OperationType.Regular;
		}

		protected void DoCancel(IView view, OperationType operation)
		{
			view?.GetUIServiceListener<IDragServiceListener>(operation)?.OnCancel();
		}

		protected void DoLeave(IView view, OperationType operation)
		{
			view?.GetUIServiceListener<IDragServiceListener>(operation)?.OnLeave();
		}

		protected void DoEnter(IView view, OperationType operation)
		{
			view?.GetUIServiceListener<IDragServiceListener>(operation)?.OnEnter();
		}

		public virtual void ProcessComplete(IView view)
		{
		}

		protected void TryLeaveAndEnter(IView viewFrom, OperationType from, IView viewTo, Point point, OperationType to)
		{
			IDragServiceListener dragServiceListener = null;
			if (viewFrom != null)
			{
				dragServiceListener = viewFrom.GetUIServiceListener<IDragServiceListener>(from);
			}
			IDragServiceListener dragServiceListener2 = null;
			bool flag = false;
			if (viewTo != null)
			{
				dragServiceListener2 = viewTo.GetUIServiceListener<IDragServiceListener>(to);
				flag = dragServiceListener2?.CanDrag(point, DragService.DragItem) ?? false;
			}
			if (flag || dragServiceListener2 == null)
			{
				dragServiceListener?.OnLeave();
			}
			if (flag)
			{
				dragServiceListener2.OnEnter();
			}
		}

		protected void DoDragging(IView view, Point point, OperationType type)
		{
			IDragServiceListener uIServiceListener = view.GetUIServiceListener<IDragServiceListener>(type);
			if (uIServiceListener != null && uIServiceListener.CanDrag(point, DragService.DragItem))
			{
				uIServiceListener.OnDragging(point, DragService.DragItem);
			}
		}

		protected void DoDragging(IView view, Point point)
		{
			DoDragging(view, point, OperationType);
		}

		protected void DoDrop(IView view, Point point, OperationType type)
		{
			IDragServiceListener uIServiceListener = view.GetUIServiceListener<IDragServiceListener>(type);
			if (uIServiceListener != null)
			{
				if (uIServiceListener.CanDrop(point, DragService.DragItem))
				{
					uIServiceListener.OnDrop(point, DragService.DragItem);
				}
				else
				{
					uIServiceListener.OnCancel();
				}
			}
		}

		protected void DoDrop(IView view, Point point)
		{
			DoDrop(view, point, OperationType);
		}

		protected bool InitializeOperation(IView view, Point point, ILayoutElement dragItem, OperationType type)
		{
			ILayoutElementBehavior elementBehavior = view.GetElementBehavior(dragItem);
			if (elementBehavior != null && elementBehavior.AllowDragging && elementBehavior.CanDrag(type))
			{
				DragService.DragItem = dragItem;
				DragService.DragSource = view;
				DragService.DragOrigin = view.ClientToScreen(point);
				view.GetUIServiceListener<IDragServiceListener>(type)?.OnInitialize(point, dragItem);
				_lastInitializedOperation = type;
				return true;
			}
			_lastInitializedOperation = OperationType.Regular;
			return false;
		}

		protected bool BeginOperation(OperationType operationType, IView view, ILayoutElement dragItem, Point startPoint)
		{
			ILayoutElementBehavior elementBehavior = view.GetElementBehavior(dragItem);
			if (elementBehavior != null && elementBehavior.AllowDragging && elementBehavior.CanDrag(operationType))
			{
				DragService.SetState(operationType);
				if (DragService.OperationType == operationType)
				{
					dragItem.ResetState();
					view.GetUIServiceListener<IDragServiceListener>(operationType)?.OnBegin(startPoint, dragItem);
					return true;
				}
			}
			return false;
		}

		protected void ResetOperation(OperationType type, IView view)
		{
			view.GetUIServiceListener<IDragServiceListener>(type)?.OnCancel();
			_lastInitializedOperation = OperationType.Regular;
		}
	}

	public static class DragServiceHelper
	{
		public static bool IsOutOfArea(Point pt, Point startPoint, int threshold)
		{
			double value = pt.X - startPoint.X;
			double value2 = pt.Y - startPoint.Y;
			if (MathHelper.IsZero(value, threshold))
			{
				return !MathHelper.IsZero(value2, threshold);
			}
			return true;
		}
	}
	public class RegularDragServiceState : DragServiceState
	{
		internal bool isDragging;

		internal bool isResizing;

		internal bool isReordering;

		internal bool isFloating;

		public sealed override OperationType OperationType => OperationType.Regular;

		public RegularDragServiceState(IDragService service)
			: base(service)
		{
		}

		public override void ProcessMouseDown(IView view, Point point)
		{
			LayoutElementHitInfo layoutElementHitInfo = view.Adapter.CalcHitInfo(view, point);
			ILayoutElement dragItem = view.GetDragItem(layoutElementHitInfo.Element);
			if (dragItem == null || layoutElementHitInfo.InControlBox)
			{
				return;
			}
			if (layoutElementHitInfo.InReorderingBounds)
			{
				isReordering = InitializeOperation(view, point, dragItem, OperationType.Reordering);
				if (isReordering)
				{
					return;
				}
			}
			if (layoutElementHitInfo.InDragBounds)
			{
				isFloating = InitializeOperation(view, point, dragItem, OperationType.Floating);
				if (isFloating)
				{
					return;
				}
				isDragging = InitializeOperation(view, point, dragItem, OperationType.FloatingMoving);
				if (isDragging)
				{
					return;
				}
				isDragging = InitializeOperation(view, point, dragItem, OperationType.ClientDragging);
				if (isDragging)
				{
					return;
				}
			}
			if (!layoutElementHitInfo.InResizeBounds)
			{
				return;
			}
			if (view.Type == HostType.Floating)
			{
				isResizing = InitializeOperation(view, point, dragItem, OperationType.FloatingResizing);
				if (isResizing)
				{
					return;
				}
			}
			isResizing = InitializeOperation(view, point, dragItem, OperationType.Resizing);
			_ = isResizing;
		}

		public override void ProcessMouseMove(IView view, Point point)
		{
			IView dragSource = base.DragService.DragSource;
			if (dragSource != null)
			{
				ILayoutElement dragItem = base.DragService.DragItem;
				Point dragOrigin = base.DragService.DragOrigin;
				Point point2 = view.ClientToScreen(point);
				Point startPoint = dragSource.ScreenToClient(point2);
				int threshold = ((isDragging && dragSource.Type == HostType.Floating) ? 1 : 5);
				if ((!DragServiceHelper.IsOutOfArea(point2, dragOrigin, threshold) || ((!isFloating || !BeginOperation(OperationType.Floating, dragSource, dragItem, point2)) && (!isDragging || ((dragSource.Type != HostType.Floating || !BeginOperation(OperationType.FloatingMoving, dragSource, dragItem, point2)) && !BeginOperation(OperationType.ClientDragging, dragSource, dragItem, startPoint))) && (!isReordering || !BeginOperation(OperationType.Reordering, dragSource, dragItem, startPoint)))) && isResizing && (dragSource.Type != HostType.Floating || !BeginOperation(OperationType.FloatingResizing, dragSource, dragItem, point2)))
				{
					BeginOperation(OperationType.Resizing, dragSource, dragItem, startPoint);
				}
			}
		}

		public override void ProcessMouseUp(IView view, Point point)
		{
			IView dragSource = base.DragService.DragSource;
			if (isResizing && dragSource != null && dragSource.Type == HostType.Floating)
			{
				ResetOperation(OperationType.FloatingResizing, view);
			}
			Reset(view);
		}

		protected override void ResetCore()
		{
			base.DragService.Reset();
			isDragging = (isResizing = (isReordering = (isFloating = false)));
			base.ResetCore();
		}
	}
	public class FloatingDragServiceState : DragServiceState
	{
		public sealed override OperationType OperationType => OperationType.Floating;

		public FloatingDragServiceState(IDragService service)
			: base(service)

		}

		public override void ProcessMouseMove(IView view, Point point)

			{

				BeginOperation(OperationType.FloatingMoving, view2, base.DragService.DragItem, startPoint);

			else
			{
				base.DragService.SetState(OperationType.Regular);
			}
		}
	}
	public abstract class ClientDragServiceStateBase : DragServiceState
	{
		private IView lastBehindView;

		private OperationType lastBehindViewOperation;

		public ClientDragServiceStateBase(IDragService service)
			: base(service)
		{
		}

		protected override IView GetBehindView()
		{
			return lastBehindView;
		}

		protected override OperationType GetBehindOperation()
		{
			return lastBehindViewOperation;
		}

		public override void ProcessMouseMove(IView view, Point point)
		{
			DoDragging(view, point);
			if (base.DragService.SuspendBehindDragging)
			{
				return;
			}
			Point point2 = view.ClientToScreen(point);
			OperationType operationType = OperationType.Regular;
			Point point3 = default(Point);
			IView behindView = GetBehindView(view, point2, point);
			if (behindView != null)
			{
				point3 = view.Adapter.GetBehindViewPoint(view, behindView, point2);
				LayoutElementHitInfo layoutElementHitInfo = behindView.Adapter.CalcHitInfo(behindView, point3);
				if (layoutElementHitInfo.InReorderingBounds && CheckReordering(behindView, point3))
				{
					operationType = OperationType.Reordering;
				}
			}
			if (lastBehindView != null && lastBehindView.IsDisposing)
			{
				lastBehindView = null;
			}
			if (behindView != lastBehindView || lastBehindViewOperation != operationType)
			{
				TryLeaveAndEnter(lastBehindView, lastBehindViewOperation, behindView, point3, operationType);
				lastBehindView = behindView;
				lastBehindViewOperation = operationType;
			}
			if (behindView != null)
			{
				DoDragging(behindView, point3, operationType);
			}
		}

		public override void ProcessMouseUp(IView view, Point point)
		{
			Point point2 = view.ClientToScreen(point);
			IView behindView = GetBehindView(view, point2, point);
			Point point3 = ((behindView != null) ? view.Adapter.GetBehindViewPoint(view, behindView, point2) : default(Point));
			DoDrop(view, point);
			if (behindView != null && !behindView.IsDisposing)
			{
				LayoutElementHitInfo layoutElementHitInfo = behindView.Adapter.CalcHitInfo(behindView, point3);
				DoDrop(behindView, point3, (layoutElementHitInfo.InReorderingBounds && CheckReordering(behindView, point3)) ? OperationType.Reordering : OperationType.Regular);
			}
			base.DragService.SetState(OperationType.Regular);
		}

		protected virtual IView GetBehindView(IView view, Point screenPoint, Point clientPoint)
		{
			return view.Adapter.GetBehindView(view, screenPoint);
		}

		private bool CheckReordering(IView view, Point point)
		{
			if (view is BaseView)
			{
				IDragServiceListener uIServiceListener = view.GetUIServiceListener<IDragServiceListener>(OperationType.Reordering);
				if (uIServiceListener != null)
				{
					if (uIServiceListener.CanDrag(point, base.DragService.DragItem))
					{
						return ((BaseView)view).CheckReordering(point);
					}
					return false;
				}
			}
			return true;
		}
	}

		protected IView CalculateEventProcessor(Point screenPoint, MouseEventType eventType, DevExpress.Xpf.Layout.Core.Platform.MouseEventArgs ea)
		{
			IView result = Receiver;
			if (Receiver == null)
			{
				result = base.Sender;
				ILayoutElementBehavior elementBehavior = base.Sender.GetElementBehavior(DragItem);
				if (OperationType == OperationType.Reordering)
				{
					OperationType operationType = CheckBehavior(base.Sender, DragItem, OperationType.FloatingMoving, OperationType.ClientDragging);
					if (operationType == OperationType.ClientDragging)
					{
						operationType = CheckBehavior(base.Sender, DragItem, OperationType.Floating, OperationType.ClientDragging);
					}
					if (elementBehavior != null && elementBehavior.AllowDragging && elementBehavior.CanDrag(operationType))
					{
						LeaveAndEnter(LastProcessor, OperationType, base.Sender, screenPoint, operationType);
					}
				}
				if (OperationType == OperationType.ClientDragging || OperationType == OperationType.Reordering)
				{
					OperationType operationType2 = CheckBehavior(base.Sender, DragItem, OperationType.FloatingMoving, OperationType.NonClientDragging);
					if (operationType2 == OperationType.NonClientDragging)
					{
						operationType2 = CheckBehavior(base.Sender, DragItem, OperationType.Floating, OperationType.NonClientDragging);
					}
					if (elementBehavior != null && elementBehavior.AllowDragging && elementBehavior.CanDrag(operationType2))
					{
						LeaveAndEnter(LastProcessor, OperationType, base.Sender, screenPoint, operationType2);
					}
				}
			}
			else
			{
				switch (OperationType)
				{
				case OperationType.NonClientDragging:
				{
					OperationType operationType4 = CheckReorderingOperation(Receiver, DragItem, screenPoint);
					ILayoutElementBehavior elementBehavior3 = base.Sender.GetElementBehavior(DragItem);
					if (elementBehavior3 != null && elementBehavior3.AllowDragging && elementBehavior3.CanDrag(operationType4))
					{
						LeaveAndEnter(base.Sender, OperationType.NonClientDragging, Receiver, screenPoint, operationType4);
					}
					break;
				}
				case OperationType.ClientDragging:
					LeaveAndEnter(LastProcessor, OperationType.ClientDragging, Receiver, screenPoint, OperationType.ClientDragging);
					break;
				case OperationType.Resizing:
				case OperationType.FloatingMoving:
				case OperationType.FloatingResizing:
					result = base.Sender;
					break;
				case OperationType.Reordering:
				{
					IView view = base.Sender;
					OperationType operationType3 = CheckReorderingOperation(Receiver, DragItem, screenPoint);
					ILayoutElementBehavior elementBehavior2 = view.GetElementBehavior(DragItem);
					if (eventType != MouseEventType.MouseUp && elementBehavior2 != null && elementBehavior2.AllowDragging && elementBehavior2.CanDrag(operationType3))
					{
						IViewAdapter viewAdapter = view.Adapter;
						LeaveAndEnter(LastProcessor, OperationType.Reordering, view, screenPoint, operationType3);
						if (view.IsDisposing)
						{
							view = viewAdapter.GetView(DragItem);
						}
					}
					result = view;
					break;
				}
				}
			}
			return result;
		}

		private OperationType CheckReorderingOperation(IView target, ILayoutElement dragItem, Point screenPoint)
		{
			Point pt = target.ScreenToClient(screenPoint);
			LayoutElementHitInfo layoutElementHitInfo = target.Adapter.CalcHitInfo(target, pt);
			if (layoutElementHitInfo.InReorderingBounds)
			{
				ReorderingListener uIServiceListener = target.GetUIServiceListener<ReorderingListener>(OperationType.Reordering);
				if (uIServiceListener != null && !uIServiceListener.CancelReordering(screenPoint, dragItem))
				{
					return OperationType.Reordering;
				}
			}
			return CheckBehavior(target, dragItem, OperationType.Floating, OperationType.ClientDragging);
		}

		private OperationType CheckBehavior(IView view, ILayoutElement dragItem, OperationType requested, OperationType allowed)
		{
			ILayoutElementBehavior elementBehavior = view.GetElementBehavior(dragItem);
			if (elementBehavior == null || !elementBehavior.CanDrag(requested))
			{
				return allowed;
			}
			return requested;
		}

		protected void LeaveAndEnter(IView viewFrom, OperationType from, IView viewTo, Point screenPoint, OperationType to)
		{
			if (viewFrom == viewTo && from == to)
			{
				return;
			}
			viewFrom?.GetUIServiceListener<IDragServiceListener>(from)?.OnLeave();
			if (from != to)
			{
				SetState(to);
				if (OperationType != to)
				{
					return;
				}
			}
			if (viewTo != null)
			{
				IDragServiceListener uIServiceListener = viewTo.GetUIServiceListener<IDragServiceListener>(to);
				Point point = viewTo.ScreenToClient(screenPoint);
				if (uIServiceListener != null && uIServiceListener.CanDrag(point, DragItem))
				{
					uIServiceListener.OnEnter();
				}
			}
		}

		protected virtual void ProcessMouseCore(IView processor, Point screenPoint, MouseEventType eventType, DevExpress.Xpf.Layout.Core.Platform.MouseEventArgs e)
		{
			if (processor == null)

				return;
			}
			Point point = processor.ScreenToClient(screenPoint);
			bool flag = false;
			switch (eventType)
			{
			case MouseEventType.MouseDown:
				flag = e.Buttons != MouseButtons.Left;
				if (!flag)
				{
					State.ProcessMouseDown(processor, point);
				}
				break;
			case MouseEventType.MouseMove:
				flag = e.Buttons != MouseButtons.Left;
				if (!flag)
				{
					State.ProcessMouseMove(processor, point);
				}
				break;
			case MouseEventType.MouseUp:
				flag = e.Buttons != MouseButtons.None;
				if (!flag)
				{
					State.ProcessMouseUp(processor, point);
				}
				break;
			case MouseEventType.MouseCaptureLost:
				flag = true;
				break;
			}
			if (flag)
			{
				State.ProcessCancel(processor);
				SetState(OperationType.Regular);
			}
		}

			operationType = CheckCanSuspendOperation(operationType);
			if (operationType2 == operationType)
			{
				return;
			}
			stateCore = GetStateFactory().Create(this, operationType);
			switch (operationType)
			{
			case OperationType.Regular:
				if (operationType2 != OperationType.Regular)
				{
					EndUpdating();
				}
				Reset();
				break;
			case OperationType.FloatingMoving:
				CheckSuspendBehindDragging(operationType2);
				break;
			case OperationType.Resizing:
			case OperationType.Reordering:
			case OperationType.Floating:
			case OperationType.ClientDragging:
				RethrowEvent = true;
				break;
			}
			if (operationType2 == OperationType.Regular)
			{
				BeginUpdating();
			}
		}

		private OperationType CheckCanSuspendOperation(OperationType operationType)
		{
			bool flag = false;
			switch (operationType)
			{
			case OperationType.Floating:
				flag = CheckSuspendFloating(operationType);
				break;
			case OperationType.ClientDragging:
				flag = CheckSuspendClientDragging(operationType);
				break;
			case OperationType.Resizing:
			case OperationType.FloatingResizing:
				flag = CheckSuspendResizing(operationType);
				break;
			case OperationType.Reordering:
				flag = CheckSuspendReordering(operationType);
				break;
			case OperationType.FloatingMoving:
				flag = CheckSuspendFloatingMoving(operationType);
				break;
			}
			if (flag)
			{
				State.ProcessCancel(DragSource);
				operationType = OperationType.Regular;
				Reset();
			}
			return operationType;
		}

	public class BaseObject : IDisposable
	{
		private int lockUpdateCounter;

		private bool isDisposingCore;

		public bool IsDisposing => isDisposingCore;

		public bool IsUpdateLocked => lockUpdateCounter > 0;

		public event EventHandler Changed;

		public event EventHandler Disposed;

		protected BaseObject()
		{
			OnCreate();
		}

		public void Dispose()
		{
			if (!IsDisposing)
			{
				isDisposingCore = true;
				BeginUpdate();
				OnDispose();
				CancelUpdate();
				RaiseDisposed(EventArgs.Empty);
				this.Changed = null;
				this.Disposed = null;
				GC.SuppressFinalize(this);
			}
		}

		protected virtual void OnCreate()
		{
		}

		protected virtual void OnDispose()
		{
		}

		public void BeginUpdate()
		{
			lockUpdateCounter++;
		}

		public void EndUpdate()
		{
			if (--lockUpdateCounter == 0)
			{
				OnUnlockUpdate();
			}
		}

		public void CancelUpdate()
		{
			lockUpdateCounter--;
		}

		protected void OnUnlockUpdate()
		{
			OnObjectChanged();
		}

		protected void OnObjectChanged()
		{
			if (!IsUpdateLocked && !IsDisposing)
			{
				BeginUpdate();
				OnUpdateObjectCore();
				RaiseChanged(EventArgs.Empty);
				CancelUpdate();
			}
		}

		protected virtual void OnUpdateObjectCore()
		{
		}

		protected void RaiseChanged(EventArgs e)
		{
			if (this.Changed != null)
			{
				this.Changed(this, e);
			}
		}

		protected void RaiseDisposed(EventArgs e)
		{
			if (this.Disposed != null)
			{
				this.Disposed(this, e);
			}
		}
	}

			}

			base.OnDispose();
		}

	public class LayoutElementHostAdapter : BaseObject, ILayoutElementHostAdapter, IDisposable
	{
		internal IDictionary<ILayoutElementHost, LayoutElementHitInfo> Context;

		private int lockEventProcessing;

		public ILayoutElementHost ActiveHost { get; private set; }

		public bool IsInEvent => lockEventProcessing > 0;

		protected override void OnCreate()
		{
			base.OnCreate();
			Context = CreateContext();
		}

		protected override void OnDispose()
		{
			Ref.Clear(ref Context);
			base.OnDispose();
		}

		protected virtual IDictionary<ILayoutElementHost, LayoutElementHitInfo> CreateContext()
		{
			return new Dictionary<ILayoutElementHost, LayoutElementHitInfo>();
		}

		public LayoutElementHitInfo CalcHitInfo(ILayoutElementHost host, Point hitPoint)
		{
			if (hitPoint == UIService.InvalidPoint)
			{
				return LayoutElementHitInfo.Empty;
			}
			LayoutElementHitInfo value = null;
			bool flag = Context.TryGetValue(host, out value);
			if (!flag || value.HitPoint != hitPoint || value.Element.IsDisposing)
			{
				Ensure(host);
				value = CalcHitInfoCore(host, hitPoint);
				if (flag)
				{
					Context[host] = value;
				}
				else
				{
					Context.Add(host, value);
				}
			}
			return value;
		}

		public bool HitTest(ILayoutElementHost host, Point hitPoint)
		{
			return HitTestCore(host, hitPoint);
		}

		protected virtual bool HitTestCore(ILayoutElementHost host, Point point)
		{
			ILayoutElement layoutElement = host?.LayoutRoot;
			if (layoutElement != null)
			{
				return HitTestRootElements(host, layoutElement, point);
			}
			return false;
		}

		private static bool HitTestRootElements(ILayoutElementHost host, ILayoutElement root, Point point)
		{
			ILayoutElement layoutElement = ((root.Nodes.Length != 0) ? root.Nodes[0] : null);
			if (!root.HitTest(point))
			{
				if (host.Type == HostType.AutoHide && layoutElement != null)
				{
					return layoutElement.HitTest(point);
				}
				return false;
			}
			return true;
		}

		protected virtual LayoutElementHitInfo CalcHitInfoCore(ILayoutElementHost host, Point point)
		{
			ILayoutElement layoutElement = host?.LayoutRoot;
			if (layoutElement == null)
			{
				return LayoutElementHitInfo.Empty;
			}
			return layoutElement.CalcHitInfo(point);
		}

		public void BeginEvent(ILayoutElementHost host)
		{
			BeginEventCore(host);
			Context.Clear();
		}

		public void EndEvent()
		{
			foreach (KeyValuePair<ILayoutElementHost, LayoutElementHitInfo> item in Context)
			{
				Invalidate(item.Key);
			}
			Context.Clear();
			EndEventCore();
		}

		private void BeginEventCore(ILayoutElementHost host)
		{
			lockEventProcessing++;
			ActiveHost = host;
			Ensure(ActiveHost);
		}

		private void EndEventCore()
		{
			ActiveHost = null;
			lockEventProcessing--;
		}

		protected void Ensure(ILayoutElementHost host)
		{
			if (host != null && host.LayoutRoot == null)
			{
				host.EnsureLayoutRoot();
			}
		}

		protected void Invalidate(ILayoutElementHost host)
		{
			ILayoutElement layoutElement = host?.LayoutRoot;
			if (layoutElement == null)
			{
				return;
			}
			foreach (ILayoutElement item in layoutElement)
			{
				item.Invalidate();
			}
		}
	}

		private static class StableSortingHelper
		{
			private static readonly Comparer<KeyValuePair<int, IView>> Comparer = Comparer<KeyValuePair<int, IView>>.Create(Compare);

			public static void Sort(IView[] views)
			{
				Array.Sort(views.Select((IView view, int index) => new KeyValuePair<int, IView>(index, view)).ToArray(), views, Comparer);
			}

			private static int Compare(KeyValuePair<int, IView> pair1, KeyValuePair<int, IView> pair2)
			{
				if (pair1.Value == pair2.Value)
				{
					return 0;
				}
				int num = CompareZOrder(pair1.Value.ZOrder, pair2.Value.ZOrder);
				if (num != 0)
				{
					return num;
				}
				return CompareZOrder(pair1.Key, pair2.Key);
			}

			private static int CompareZOrder(int order1, int order2)
			{
				return order2.CompareTo(order1);
			}
		}

		public void ProcessAction(ViewAction action)
		{
			if (!base.IsDisposing)
			{
				IView[] viewsSortedByZOrder = GetViewsSortedByZOrder();
				foreach (IView view in viewsSortedByZOrder)
				{
					ProcessActionCore(view, action);
				}
			}
		}

		public void ProcessAction(IView view, ViewAction action)
		{
			if (!base.IsDisposing)
			{
				ProcessActionCore(view, action);
			}

		protected void ProcessActionCore(IView view, ViewAction action)
		{
			if (view != null)
			{
				switch (action)
				{
				case ViewAction.Hiding:
					ActionService.Hide(view, immediately: false);
					break;
				case ViewAction.Hide:
					ActionService.Hide(view, immediately: true);
					break;
				case ViewAction.ShowSelection:
					ActionService.ShowSelection(view);
					break;
				case ViewAction.HideSelection:
					ActionService.HideSelection(view);
					break;
				}
			}
		}

		public IView GetView(object rootKey)
		{
			IView[] viewsSortedByZOrder = GetViewsSortedByZOrder();
			foreach (IView view in viewsSortedByZOrder)
			{
				if (view.RootKey == rootKey)
				{
					return view;
				}
			}
			return null;
		}

		public IView GetView(ILayoutElement element)
		{
			ILayoutElement root = ElementHelper.GetRoot(element);
			IView[] viewsSortedByZOrder = GetViewsSortedByZOrder();
			foreach (IView view in viewsSortedByZOrder)
			{
				if (view.LayoutRoot == root)
				{
					return view;
				}
			}
			return null;
		}

		public IView GetView(Point screenPoint)
		{
			IView[] viewsSortedByZOrder = GetViewsSortedByZOrder();
			foreach (IView view in viewsSortedByZOrder)
			{
				Ensure(view);
				if (view.IsActiveAndCanProcessEvent && view.ZOrder != -1)
				{
					Point hitPoint = view.ScreenToClient(screenPoint);
					if (HitTest(view, hitPoint))
					{
						return view;
					}
				}
			}
			return null;
		}

		public IView GetBehindView(IView source, Point screenPoint)
		{
			return GetBehindViewOverride(source, screenPoint);
		}

		protected virtual IView GetBehindViewOverride(IView source, Point screenPoint)
		{
			if (source == null)
			{
				return null;
			}
			IView[] viewsSortedByZOrder = GetViewsSortedByZOrder();
			bool flag = false;
			foreach (IView view in viewsSortedByZOrder)
			{
				if (!flag)
				{
					flag = view == source;
				}
				else
				{
					if (!flag)
					{
						continue;
					}
					Ensure(view);
					if (view.IsActiveAndCanProcessEvent)
					{
						Point hitPoint = view.ScreenToClient(screenPoint);
						if (HitTest(view, hitPoint))
						{
							return view;
						}
					}
				}
			}
			return null;
		}

		public Point GetBehindViewPoint(IView source, IView behindView, Point screenPoint)
		{
			return GetBehindViewPointOverride(source, behindView, screenPoint);
		}

		protected virtual Point GetBehindViewPointOverride(IView source, IView behindView, Point screenPoint)
		{
			return behindView.ScreenToClient(screenPoint);
		}

		{
			IView[] array = Views.ToArray();
			StableSortingHelper.Sort(array);
			return array;
		}

	public class BaseView : BaseLayoutElementHost, IView, ILayoutElementHost, IBaseObject, IDisposable, IUIServiceProvider
	{
		private IDictionary<object, IUIServiceListener> defaultListenersCore;

		private IDictionary<object, IUIServiceListener> customListenersCore;

		internal IViewAdapter adapterCore;

		private int? zOrderCore;

		public IViewAdapter Adapter => adapterCore;

		public int ZOrder
		{
			get
			{
				if (!zOrderCore.HasValue)
				{
					zOrderCore = CalcZOrder();
				}
				return zOrderCore.Value;
			}
		}

		public bool IsOnCurrentVirtualDesktop => IsOnCurrentVirtualDesktopCore();

		protected BaseView(ILayoutElementFactory factory, ILayoutElementBehavior behavior)
			: base(factory, behavior)
		{
		}

		protected override void OnCreate()
		{
			defaultListenersCore = new Dictionary<object, IUIServiceListener>();
			customListenersCore = new Dictionary<object, IUIServiceListener>();
			base.OnCreate();
		}

		protected override void OnDispose()
		{
			Ref.Clear(ref defaultListenersCore);
			Ref.Clear(ref customListenersCore);
			adapterCore = null;
			base.OnDispose();
		}

		public void InvalidateZOrder()
		{
			zOrderCore = null;
		}

		protected virtual int CalcZOrder()
		{
			return 0;
		}

		protected internal virtual bool CanSuspendDocking(ILayoutElement dragItem)
		{
			return false;
		}

		protected internal virtual bool CanSuspendFloating(ILayoutElement dragItem)
		{
			return false;
		}

		protected internal virtual bool CanSuspendClientDragging(ILayoutElement dragItem)
		{
			return false;
		}

		protected internal virtual bool CanSuspendResizing(ILayoutElement dragItem)
		{
			return false;
		}

		protected internal virtual bool CanSuspendReordering(ILayoutElement dragItem)
		{
			return false;
		}

		protected internal virtual bool CanSuspendFloatingMoving(ILayoutElement dragItem)
		{
			return false;
		}

		protected internal virtual bool CanSuspendBehindDragging(ILayoutElement dragItem)
		{
			return false;
		}

		protected internal virtual bool CheckReordering(Point point)
		{
			return true;
		}

		protected virtual bool CanHandleMouseCaptureLost()
		{
			return true;
		}

		public void OnKeyDown(Key key)
		{
			Adapter.ProcessKey(this, KeyEventType.KeyDown, key);
		}

		public void OnKeyUp(Key key)
		{
			Adapter.ProcessKey(this, KeyEventType.KeyUp, key);
		}

		protected virtual void BeginProcessEvent(object ea)
		{
		}

		protected virtual void EndProcessEvent(object ea)
		{
		}

		public void OnMouseEvent(MouseEventType eventType, MouseEventArgs ea = null)
		{
			BeginProcessEvent(ea);
			try
			{
				switch (eventType)
				{
				case MouseEventType.MouseDown:
					OnMouseDown(ea);
					break;
				case MouseEventType.MouseUp:
					OnMouseUp(ea);
					break;
				case MouseEventType.MouseMove:
					OnMouseMove(ea);
					break;
				case MouseEventType.MouseLeave:
					OnMouseLeave();
					break;
				case MouseEventType.MouseCaptureLost:
					OnMouseCaptureLost();
					break;
				}
			}
			finally
			{
				EndProcessEvent(ea);
			}
		}

		public void OnMouseDown(MouseEventArgs ea)
		{
			Adapter.ProcessMouseEvent(this, MouseEventType.MouseDown, ea);
		}

		public void OnMouseUp(MouseEventArgs ea)
		{
			Adapter.ProcessMouseEvent(this, MouseEventType.MouseUp, ea);
		}

		public void OnMouseMove(MouseEventArgs ea)
		{
			Adapter.ProcessMouseEvent(this, MouseEventType.MouseMove, ea);
		}

		public void OnMouseLeave()
		{
			Adapter.ProcessMouseEvent(this, MouseEventType.MouseLeave, null);
		}

		public void OnMouseCaptureLost()
		{
			if (CanHandleMouseCaptureLost())
			{
				Adapter.ProcessMouseEvent(this, MouseEventType.MouseCaptureLost, null);
			}
		}

		protected virtual bool CanHandleMouseDownCore()
		{
			return Adapter.DragService.DragItem != null;
		}

		public bool CanHandleMouseDown()
		{
			return CanHandleMouseDownCore();
		}

		protected virtual bool IsOnCurrentVirtualDesktopCore()
		{
			return true;
		}

		public void RegisterUIServiceListener(IUIServiceListener listener)
		{
			if (listener != null)
			{
				defaultListenersCore[listener.Key] = listener;
				listener.ServiceProvider = this;
			}
		}

		public ServiceListener GetUIServiceListener<ServiceListener>(object key) where ServiceListener : class, IUIServiceListener
		{
			return GetUIServiceListenerCore<ServiceListener>(key);
		}

		public ServiceListener GetUIServiceListener<ServiceListener>() where ServiceListener : class, IUIServiceListener
		{
			return GetUIServiceListenerCore<ServiceListener>(typeof(ServiceListener));
		}

		private ServiceListener GetUIServiceListenerCore<ServiceListener>(object key) where ServiceListener : class, IUIServiceListener
		{
			if (base.IsDisposing)
			{
				return null;
			}
			ServiceListener val = null;
			if (CanUseCustomServiceListener(key))
			{
				IUIServiceListener value = null;
				if (!customListenersCore.TryGetValue(key, out value))
				{
					value = GetCustomUIServiceListener<ServiceListener>(key);
					value.ServiceProvider = this;
					if (value is ServiceListener)
					{
						customListenersCore.Add(key, value);
					}
				}
				val = value as ServiceListener;
			}
			return val ?? GetDefaultUIServiceListener<ServiceListener>(key);
		}

		protected virtual bool CanUseCustomServiceListener(object key)
		{
			return false;
		}

		protected virtual ServiceListener GetCustomUIServiceListener<ServiceListener>(object key) where ServiceListener : class, IUIServiceListener
		{
			return null;
		}

		protected virtual ServiceListener GetDefaultUIServiceListener<ServiceListener>(object key) where ServiceListener : class, IUIServiceListener
		{
			IUIServiceListener value;
			return (ServiceListener)(defaultListenersCore.TryGetValue(key, out value) ? value : null);
		}
	}
	public class ViewCollection : BaseChangeableList<IView>
	{
		private static class AffinityHelper
		{
			public static void SetAffinity(IViewAdapter adapter, IView element)
			{
				if (AffinityHelperException.Assert(element))
				{
					((BaseView)element).adapterCore = adapter;
				}
			}
		}

		private IViewAdapter ownerCore;

		public ViewCollection(IViewAdapter owner)
		{
			ownerCore = owner;
		}

		protected override void OnDispose()
		{
			ownerCore = null;
			IView[] array = ToArray();
			for (int i = 0; i < array.Length; i++)
			{
				IView refToDispose = array[i];
				refToDispose.Disposed -= OnLayoutItemDisposed;
				Ref.Dispose(ref refToDispose);
			}
			base.OnDispose();
		}

		protected override void OnElementAdded(IView element)
		{
			base.OnElementAdded(element);
			AffinityHelper.SetAffinity(ownerCore, element);
			element.Disposed += OnLayoutItemDisposed;
		}

		protected override void OnElementRemoved(IView element)
		{
			element.Disposed -= OnLayoutItemDisposed;
			AffinityHelper.SetAffinity(null, element);
			base.OnElementRemoved(element);
		}

		private void OnLayoutItemDisposed(object sender, EventArgs ea)
		{
			RaiseCollectionChanged(new CollectionChangedEventArgs<IView>(sender as IView, CollectionChangedType.ElementDisposed));
			if (base.List != null)
			{
				Remove(sender as IView);
			}
		}
	}
