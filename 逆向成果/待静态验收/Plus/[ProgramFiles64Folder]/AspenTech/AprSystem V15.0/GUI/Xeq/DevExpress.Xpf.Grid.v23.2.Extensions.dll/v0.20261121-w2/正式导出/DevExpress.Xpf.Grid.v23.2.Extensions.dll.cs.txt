		{

		}

		public static CriteriaOperator MakeFilterSkipToken<T>(SortDefinition[] sortOrder, SortDefinition defaultUniqueSort, T lastFetchedRow)
		{
			if (!sortOrder.Any() && defaultUniqueSort == null)
			{
				throw new InvalidOperationException("At least 'defaultUniqueSort' argument should be specified to make a filter skip token");
			}
			if (lastFetchedRow == null)
			{
				return null;
			}
			sortOrder = SortDefinitionHelper.ConcatWithDefaultSortOrder(sortOrder, defaultUniqueSort);
			PropertyDescriptorCollection allProperties = TypeDescriptor.GetProperties(typeof(T));
			PropertyDescriptor[] properties = sortOrder.Select((SortDefinition x) => allProperties[x.PropertyName]).ToArray();
			object[] values = properties.Select((PropertyDescriptor x) => x.GetValue(lastFetchedRow)).ToArray();
			if (values.Last() == null)
			{
				throw new InvalidOperationException("Last sort value can't be null");
			}
			IEnumerable<CriteriaOperator> enumerable = Enumerable.Range(0, sortOrder.Length).Select(delegate(int i)
			{
				IEnumerable<BinaryOperator> first = Enumerable.Range(0, i).Select((Func<int, BinaryOperator>)((int j) => new BinaryOperator(sortOrder[j].PropertyName, values[j], (BinaryOperatorType)0)));
				CriteriaOperator singleElement = MakeOrderComparison(sortOrder[i], values[i], properties[i].PropertyType);
				return CriteriaOperator.And(((IEnumerable<CriteriaOperator>)first).Concat(singleElement.YieldIfNotNull<CriteriaOperator>()).ToArray());
			});
			return CriteriaOperator.Or(enumerable);
		}

		private static CriteriaOperator MakeOrderComparison(SortDefinition sort, object value, Type propertyType)
		{
			bool flag = propertyType.IsClass || Nullable.GetUnderlyingType(propertyType) != null;
			Func<object, BinaryOperatorType, BinaryOperator> func = (object obj, BinaryOperatorType type) => new BinaryOperator(sort.PropertyName, obj, type);
			if (sort.Direction == ListSortDirection.Ascending)
			{
				if (value == null)
				{
					return (CriteriaOperator)(object)func(null, (BinaryOperatorType)1);
				}
				return (CriteriaOperator)(object)func(value, (BinaryOperatorType)2);
			}
			if (value == null)
			{
				return null;
			}
			BinaryOperator val = func(value, (BinaryOperatorType)3);
			if (flag)
			{
				return CriteriaOperator.Or((CriteriaOperator)(object)val, (CriteriaOperator)(object)func(null, (BinaryOperatorType)0));
			}
			return (CriteriaOperator)(object)val;
		}
	}

		private class RowCell : HelperBase

			public double ColumnMinWidth => GetColumnMinWidth();

			public bool NewWidthIsMinWidth => NewWidth == ColumnMinWidth;

			public RowCell(ResizableGrid owner, ColumnBase gridColumn, ContentControl contentControl)
				: base(owner)
			{
				GridColumnControl = contentControl;
				GridColumn = gridColumn;
				Row = BandedViewBehavior.GetRow(GridColumn);
				RowSpan = BandedViewBehavior.GetRowSpan(GridColumn);
				Column = BandedViewBehavior.GetColumn(GridColumn);
				ColumnSpan = BandedViewBehavior.GetColumnSpan(GridColumn);
				ColumnDefinitions = new ColumnDefinitions();
				for (int i = Column; i < Column + ColumnSpan; i++)
				{
					ColumnDefinitions.Add(Owner.BandBehavior.ColumnDefinitions[i]);
				}
			}

			private double GetColumnMinWidth()
			{
				double num = 0.0;
				foreach (ColumnDefinition columnDefinition in ColumnDefinitions)
				{
					num += columnDefinition.MinWidth;
				}
				return Math.Max(num, 0.0);
			}

			private GridLength GetColumnWidth()
			{
				double num = 0.0;
				foreach (ColumnDefinition columnDefinition in ColumnDefinitions)
				{
					num += columnDefinition.Width.Value;
				}
				return new GridLength(num, GridUnitType.Star);
			}

			private void SetColumnWidth(GridLength value)
			{
				if (NewWidthIsMinWidth)
				{
					double value2 = value.Value / (double)ColumnDefinitions.Count;
					{
						foreach (ColumnDefinition columnDefinition in ColumnDefinitions)
						{
							columnDefinition.Width = new GridLength(value2, GridUnitType.Star);
						}
						return;
					}
				}
				double value3 = ColumnWidth.Value;
				double value4 = value.Value;
				foreach (ColumnDefinition columnDefinition2 in ColumnDefinitions)
				{
					double value5 = columnDefinition2.Width.Value;
					double num = value4 * value5 / value3;
					if (double.IsNaN(num))
					{
						num = 0.0;
					}
					columnDefinition2.Width = new GridLength(num, GridUnitType.Star);
				}
			}
		}

		private class Row : HelperBase

			public RowCell this[int index] => RowCellList[index];

			public int Count => RowCellList.Count;

			public double RowWidth
			{
				get
				{
					double num = 0.0;
					for (int i = 0; i < Count; i++)
					{
						num += this[i].ActualWidth;
					}
					return num - Owner.CorrectingLeftAreaWidth;
				}
			}

			public RowCell SelectedCell => this[SelectedIndex];

			public int SelectedIndex { get; private set; }

			public Row(ResizableGrid owner, int rowIndex)
				: base(owner)
			{
				RowCellList = new List<RowCell>();
				foreach (ContentControl child in Owner.Children)
				{
					ColumnBase columnBase = child.DataContext as ColumnBase;
					int row = BandedViewBehavior.GetRow(columnBase);
					int rowSpan = BandedViewBehavior.GetRowSpan(columnBase);
					if (row <= rowIndex && row + rowSpan > rowIndex)
					{
						RowCellList.Add(new RowCell(Owner, columnBase, child));
					}
					Comparison<RowCell> comparison = delegate(RowCell rowCell1, RowCell rowCell2)
					{
						if (rowCell1.Column == rowCell2.Column)
						{
							return 0;
						}
						return (rowCell1.Column >= rowCell2.Column) ? 1 : (-1);
					};
					RowCellList.Sort(comparison);
				}
			}

					if (this[i].Column == columnIndex)

						break;
					}
				}
			}

			public void Foreach(int startIndex, Action<RowCell> action)
			{

		public void Resize(ColumnBase gridColumn, double diff)
		{
			Rows rows = new Rows(this);
			CorrectColumnsWidth(rows);
			if (diff == 0.0 || double.IsNaN(diff))
			{
				return;
			}
			int column = BandedViewBehavior.GetColumn(gridColumn);
			int row = BandedViewBehavior.GetRow(gridColumn);
			Row row2 = rows[row];
			row2.SelectColumn(column);
			if (diff > 0.0)
			{
				DecreaseColumnsWidth(row2, row2.SelectedIndex + 1, ref diff);
				IncreaseSelectedColumnWidth(row2, diff);
			}
			if (diff < 0.0)
			{
				diff *= -1.0;
				if (Math.Abs(row2.SelectedCell.ActualWidth - row2.SelectedCell.ColumnMinWidth) <= 0.5)
				{
					return;
				}
				DecreaseSelectedColumnWidth(row2, ref diff);
				IncreaseColumnsWidth(row2, row2.SelectedIndex + 1, diff);
			}
			row2.Foreach(row2.SelectedIndex, delegate(RowCell rowCell)
			{
				double value = rowCell.NewWidth / row2.RowWidth;
				rowCell.ColumnWidth = new GridLength(value, GridUnitType.Star);
			});
		}

		private void IncreaseSelectedColumnWidth(Row row, double diff)
		{
			row.SelectedCell.NewWidth = row.SelectedCell.ActualWidth + diff;
		}

		private void DecreaseSelectedColumnWidth(Row row, ref double diff)
		{
			if (row.SelectedCell.ActualWidth - diff < row.SelectedCell.ColumnMinWidth)
			{
				diff = row.SelectedCell.ActualWidth - row.SelectedCell.ColumnMinWidth;
				row.SelectedCell.NewWidth = row.SelectedCell.ColumnMinWidth;
			}
			else
			{
				row.SelectedCell.NewWidth = row.SelectedCell.ActualWidth - diff;
			}
		}

		private void IncreaseColumnsWidth(Row row, int startIndex, double diff)
		{
			double one = 0.0;
			row.Foreach(startIndex, delegate(RowCell rowCell)
			{
				one += rowCell.ColumnWidth.Value;
			});
			row.Foreach(startIndex, delegate(RowCell rowCell)
			{
				double num = rowCell.ColumnWidth.Value / one;
				double num2 = diff * num;
				rowCell.NewWidth = rowCell.ActualWidth + num2;
			});
		}

		private void DecreaseColumnsWidth(Row row, int startIndex, ref double value)
		{
			double allWidth = 0.0;
			double allMinWidth = 0.0;
			row.Foreach(startIndex, delegate(RowCell rowCell)
			{
				allWidth += rowCell.ActualWidth;
				allMinWidth += rowCell.ColumnMinWidth;
			});
			if (allWidth - value < allMinWidth)
			{
				value = allWidth - allMinWidth;
			}
			row.Foreach(startIndex, delegate(RowCell rowCell)
			{
				rowCell.NewWidth = rowCell.ActualWidth;
			});
			if (value == 0.0)
			{
				return;
			}
			double one = 0.0;
			row.Foreach(startIndex, delegate(RowCell rowCell)
			{
				if (rowCell.NewWidth > rowCell.ColumnMinWidth)
				{
					one += rowCell.ColumnWidth.Value;
				}
			});
			if (one == 0.0)
			{
				return;
			}
			double one2 = 0.0;
			double diff = value;
			double diff2 = 0.0;
			int num = 0;
			while (true)
			{
				num++;
				row.Foreach(startIndex, delegate(RowCell rowCell)
				{
					double num2 = rowCell.ColumnWidth.Value / one;
					double num3 = diff * num2;
					rowCell.NewWidth -= num3;
					if (rowCell.NewWidth <= rowCell.ColumnMinWidth)
					{
						rowCell.NewWidth = rowCell.ColumnMinWidth;
						diff2 += rowCell.ColumnMinWidth - rowCell.NewWidth;
						one2 += rowCell.ColumnWidth.Value;
					}
				});
				if (diff2 != 0.0 && one2 != 0.0)
				{
					diff = diff2;
					one -= one2;
					diff2 = 0.0;
					one2 = 0.0;
					continue;
				}
				break;
			}
		}
	}

	internal class WeakDictionary<TKey, TValue> where TValue : class
	{
		private readonly Dictionary<WeakReference, WeakReference> internalDict = new Dictionary<WeakReference, WeakReference>();

		public TValue this[TKey key]
		{
			get
			{
				WeakReference weakKey = GetWeakKey(key);
				if (weakKey == null)
				{
					return null;
				}
				return internalDict[weakKey].Target as TValue;
			}
			set
			{
				WeakReference key2 = GetWeakKey(key) ?? new WeakReference(key);
				internalDict[key2] = new WeakReference(value);
			}
		}

		public IEnumerable<TValue> Values => internalDict.Values.Select((WeakReference wr) => wr.Target as TValue);

		private WeakReference GetWeakKey(TKey key)
		{
			List<WeakReference> list = internalDict.Keys.ToList();
			foreach (WeakReference item in list)
			{
				if (!item.IsAlive)
				{
					internalDict.Remove(item);
				}
				else if (item.Target.Equals(key))
				{
					return item;
				}
			}
			return null;
		}

		public void Remove(TKey key)
		{
			WeakReference weakKey = GetWeakKey(key);
			if (weakKey != null)
			{
				internalDict.Remove(weakKey);
			}
		}

		public bool TryGetValue(TKey key, out TValue value)
		{
			WeakReference weakKey = GetWeakKey(key);
			value = ((weakKey != null) ? (internalDict[weakKey].Target as TValue) : null);
			return weakKey != null;
		}
	}
	internal static class RootVisualsContainer
	{
		private static readonly WeakDictionary<DragDropManagerBase, UIElement> roots = new WeakDictionary<DragDropManagerBase, UIElement>();

		private static readonly List<WeakReference> RegisteredManagers = new List<WeakReference>();

		public static void Register(DragDropManagerBase manager)
		{
			RegisteredManagers.Add(new WeakReference(manager));
			if (manager.AssociatedControl != null)
			{
				if (manager.AssociatedControl.IsLoaded)
				{
					RegisterControl(manager.AssociatedControl);
				}
				manager.AssociatedControl.Loaded -= AssociatedControl_Loaded;
				manager.AssociatedControl.Loaded += AssociatedControl_Loaded;
			}
		}

		private static void AssociatedControl_Loaded(object sender, RoutedEventArgs e)
		{
			RegisterControl(sender);
		}

		private static void RegisterControl(object control)
		{
			WeakReference weakReference = RegisteredManagers.FirstOrDefault((WeakReference wr) => wr.IsAlive && ((DispatcherObject)(object)(wr.Target as DragDropManagerBase)).Dispatcher == Dispatcher.CurrentDispatcher && (wr.Target as DragDropManagerBase).AssociatedControl == control);
			if (weakReference != null && weakReference.Target is DragDropManagerBase manager)
			{
				RegisterManagerRoot(manager);
			}
		}

		private static bool RegisterManagerRoot(DragDropManagerBase manager)
		{
			FrameworkElement rootElement = manager.GetRootElement();
			if (manager.RootElement == null && rootElement == manager.AssociatedControl)
			{
				return false;
			}
			roots[manager] = rootElement;
			return true;
		}

		public static void Unregister(DragDropManagerBase manager)
		{
			roots.Remove(manager);
			RegisteredManagers.Remove(RegisteredManagers.FirstOrDefault((WeakReference wr) => wr.Target == manager));
		}

		public static List<UIElement> GetRootVisuals(DragDropManagerBase manager)
		{
			RegisterManagerRoot(manager);
			if (manager == null || !roots.TryGetValue(manager, out var value))
			{
				return null;
			}
			List<UIElement> rootVisuals = GetRootVisuals();
			rootVisuals.Remove(value);
			rootVisuals.Insert(0, value);
			return rootVisuals;
		}

		public static List<UIElement> GetRootVisuals()
		{
			return roots.Values.Distinct().ToList();
		}
	}

		private void PerformDropToView(DragDropManagerBase sourceManager, TableViewHitInfo hitInfo, Point pt, MoveRowsDelegate reorderDelegate, MoveRowsDelegate groupDelegateExtractor, MoveRowsDelegate addRowsDelegate, MoveRowsDelegate reorderGroupDelegate)
		{
			int rowHandle = hitInfo.RowHandle;
			if (GridControl.IsGroupRowHandle(rowHandle))
			{
				groupDelegateExtractor(sourceManager, rowHandle, base.HitElement);
			}
			else if (IsSortedButNotGrouped() || hitInfo.HitTest == TableViewHitTest.DataArea)
			{
				if (sourceManager.DraggingRows.Count > 0 && GetDataAreaElement(base.HitElement) != null && sourceManager != this)
				{
					addRowsDelegate(sourceManager, rowHandle, base.HitElement);
				}
				else
				{
					ClearDragInfo(sourceManager);
				}
			}
			else if (rowHandle == int.MinValue || rowHandle == -2147483645 || rowHandle == -2147483647)
			{
				ClearDragInfo(sourceManager);
			}
			else if (GridControl.GroupCount > 0)
			{
				int parentRowHandle = GridControl.GetParentRowHandle(rowHandle);
				if (ShouldReorderGroup())
				{
					reorderGroupDelegate(sourceManager, rowHandle, base.HitElement);
				}
				else
				{
					groupDelegateExtractor(sourceManager, parentRowHandle, base.HitElement);
				}
			}
			else
			{
				reorderDelegate(sourceManager, rowHandle, base.HitElement);
			}
		}

		private void MoveSelectedRows(DragDropManagerBase sourceManager, int insertRowHandle, DependencyObject hitElement)
		{
			object row = GridControl.GetRow(insertRowHandle);
			if (row == null || sourceManager.DraggingRows == null || sourceManager.DraggingRows.Contains(row))
			{
				return;
			}
			DropTargetType dropTargetTypeByHitElement = GetDropTargetTypeByHitElement(hitElement);
			if (dropTargetTypeByHitElement == DropTargetType.None)
			{
				return;
			}
			int num = ItemsSource.IndexOf(row) + ((dropTargetTypeByHitElement == DropTargetType.InsertRowsAfter) ? 1 : 0);
			bool flag = ItemsSource is DataView;
			GridControl.BeginDataUpdate();
			foreach (object draggingRow in sourceManager.DraggingRows)
			{
				if (ItemsSource == sourceManager.GetSource(draggingRow) && num > ItemsSource.IndexOf(sourceManager.GetObject(draggingRow)))
				{
					num--;
				}
				MoveRow(sourceManager, draggingRow, num);
				num++;
			}
			GridControl.EndDataUpdate();
			if (base.RestoreSelection)
			{
				int rowHandleByListIndex = GridControl.GetRowHandleByListIndex(ItemsSource.IndexOf(sourceManager.DraggingRows[0]));
				int rowHandleByListIndex2 = GridControl.GetRowHandleByListIndex(ItemsSource.IndexOf(sourceManager.DraggingRows[sourceManager.DraggingRows.Count - 1]));
				base.DataControl.SelectRange(rowHandleByListIndex, rowHandleByListIndex2);
			}
			else
			{
				base.DataControl.UnselectAll();
			}
			if (sourceManager.DraggingRows.Count > 0)
			{
				base.DataControl.CurrentItem = sourceManager.DraggingRows[0];
			}
		}

				DataRow dataRow = DevExpress.Xpf.Grid.DragDrop.Utils.CloneDataRow(dataView, obj2);
				DataRow dataRow2 = null;
				sourceManager.RemoveObject(obj);
				if (row.RowState != DataRowState.Detached)

				TreeListDropEventArgs e = RaiseDropEvent(sourceManager, pt);
				if (!e.Handled)
				{
					ValidateItemsSource();
					PerformDropToView(sourceManager, GetHitInfo(base.HitElement) as TreeListViewHitInfo, pt, MoveSelectedRows, null, base.AddRows);
				}
				RaiseDroppedEvent(sourceManager, e);
				base.OnDrop(sourceManager, source, pt);
			}
		}

		protected override GridViewHitInfoBase GetHitInfo(DependencyObject element)
		{
			return TreeListView.CalcHitInfo(element);
		}

		protected override void AddRow(DragDropManagerBase sourceManager, object row, int insertRowHandle)
		{
			if (DropStrategy == null)
			{
				UpdateDropStrategy();
			}
			DropStrategy.AddRow(this, sourceManager, row, insertRowHandle);
		}

		protected override void PerformDropToViewCore(DragDropManagerBase sourceManager)
		{
			TreeListViewHitInfo treeListViewHitInfo = GetHitInfo(base.HitElement) as TreeListViewHitInfo;
			if (BanDrop(treeListViewHitInfo.RowHandle, treeListViewHitInfo, sourceManager, GetDropTargetTypeByHitElement(base.HitElement)))
			{
				ClearDragInfo(sourceManager);
			}
			else
			{
				PerformDropToView(sourceManager, treeListViewHitInfo, base.LastPosition, SetReorderDropInfo, null, base.SetAddRowsDropInfo);
			}
		}

		private void PerformDropToView(DragDropManagerBase sourceManager, TreeListViewHitInfo hitInfo, Point pt, MoveRowsDelegate setreorderDelegate, MoveRowsDelegate moveToGroupDelegate, MoveRowsDelegate addRowsDelegate)
		{
			int rowHandle = hitInfo.RowHandle;
			if (hitInfo.HitTest == TreeListViewHitTest.DataArea)
			{
				if (sourceManager.DraggingRows.Count > 0 && GetDataAreaElement(base.HitElement) != null)
				{
					addRowsDelegate(sourceManager, rowHandle, base.HitElement);
				}
				else
				{
					ClearDragInfo(sourceManager);
				}
			}
			else if (rowHandle == int.MinValue || rowHandle == -2147483645 || rowHandle == -2147483647)
			{
				ClearDragInfo(sourceManager);
			}
			else
			{
				setreorderDelegate(sourceManager, rowHandle, base.HitElement);
			}
		}

		protected override DragDropDragOverEventArgs RaiseDragOverEvent(GridViewHitInfoBase hitInfo, DragDropManagerBase sourceManager, DropTargetType dropTargetType)
		{
			TreeListDragOverEventArgs e = new TreeListDragOverEventArgs(base.DataControl as GridDataControlBase, TreeListView.GetNodeByRowHandle(hitInfo.RowHandle), this, sourceManager, hitInfo as TreeListViewHitInfo, sourceManager.DraggingRows, dropTargetType)
			{
				Handled = false
			};
			if (DragOverEventHandler != null)
			{
				DragOverEventHandler(this, e);
			}
			return e;
		}

		protected override bool InternalBanDrop(int insertRowHandle, DragDropManagerBase sourceManager)
		{
			TreeListNode nodeByRowHandle = TreeListView.GetNodeByRowHandle(insertRowHandle);
			if (this == sourceManager)
			{
				foreach (TreeListNode draggingRow in sourceManager.DraggingRows)
				{
					if (nodeByRowHandle == draggingRow || (nodeByRowHandle != null && ((TreeListNodeBase)nodeByRowHandle).IsDescendantOf((TreeListNodeBase)(object)draggingRow)))
					{
						return true;
					}
				}
			}
			return base.InternalBanDrop(insertRowHandle, sourceManager);
		}

		private void MoveSelectedRows(DragDropManagerBase sourceManager, int insertRowHandle, DependencyObject hitElement)
		{
			TreeListNode nodeByRowHandle = TreeListView.GetNodeByRowHandle(insertRowHandle);
			object content = ((TreeListNodeBase)nodeByRowHandle).Content;
			if (content == null || sourceManager.DraggingRows == null || sourceManager.DraggingRows.Contains(content))
			{
				return;
			}
			DropTargetType dropTargetTypeByHitElement = GetDropTargetTypeByHitElement(hitElement);
			if (dropTargetTypeByHitElement != DropTargetType.None)
			{
				if (DropStrategy == null)
				{
					UpdateDropStrategy();
				}
				DropStrategy.MoveRows(this, sourceManager, nodeByRowHandle, dropTargetTypeByHitElement);
			}
		}

		protected virtual TreeListDropEventArgs RaiseDropEvent(DragDropManagerBase sourceManager, Point pt)
		{
			UIElement visibleHitTestElement = GetVisibleHitTestElement(pt);
			TreeListViewHitInfo treeListViewHitInfo = GetHitInfo((DependencyObject)visibleHitTestElement) as TreeListViewHitInfo;
			TreeListNode nodeByRowHandle = TreeListView.GetNodeByRowHandle(treeListViewHitInfo.RowHandle);
			DropTargetType dropTargetTypeByHitElement = GetDropTargetTypeByHitElement(visibleHitTestElement);
			return RaiseDropEvent(sourceManager, treeListViewHitInfo, nodeByRowHandle, sourceManager.DraggingRows, (treeListViewHitInfo.HitTest == TreeListViewHitTest.DataArea) ? DropTargetType.DataArea : dropTargetTypeByHitElement);
		}

		public void InsertObject(IList ItemsSource, TreeListNode node, DropTargetType dropTargetType, object obj, int index)
		{
			switch (dropTargetType)
			{
			case DropTargetType.InsertRowsAfter:
			case DropTargetType.InsertRowsBefore:
				ItemsSource.Insert(index, obj);
				break;
			case DropTargetType.InsertRowsIntoNode:
			case DropTargetType.DataArea:
				GetItemsSource(node).Add(obj);
				break;
			case DropTargetType.InsertRowsIntoGroup:
				break;
			}
		}

		public void InsertObject(DataRowCollection ItemsSource, TreeListNode node, DropTargetType dropTargetType, DataRow obj, int index)
		{
			switch (dropTargetType)
			{
			case DropTargetType.InsertRowsAfter:
			case DropTargetType.InsertRowsBefore:
				ItemsSource.InsertAt(obj, index);
				break;
			case DropTargetType.InsertRowsIntoNode:
			case DropTargetType.DataArea:
				if (TreeListView.TreeDerivationMode == TreeDerivationMode.Selfreference)
				{
					ItemsSource.Add(obj);
				}
				else
				{
					(GetItemsSource(node) as DataView).Table.Rows.Add(obj);
				}
				break;
			case DropTargetType.InsertRowsIntoGroup:
				break;
			}
		}

		private object GetObjectByRowHandle(int insertRowHandle)

			if (View.GetRowElementByRowHandle(insertRowHandle) is IGridDataRow gridDataRow)
			{

				{
					return null;
				}
				return gridDataRow.RowData.Row;
			}
			return null;
		}

		private void SetReorderDropInfo(DragDropManagerBase sourceManager, int insertRowHandle, DependencyObject hitElement)
		{
			FrameworkElement rowElement = GetRowElement(hitElement);
			FrameworkElement rowContentElement = GetRowContentElement(rowElement);
			TableDragIndicatorPosition dragIndicatorPositionForRowElement = GetDragIndicatorPositionForRowElement(rowContentElement);
			if (dragIndicatorPositionForRowElement != TableDragIndicatorPosition.None)
			{
				DropTargetType dropTargetType = GetDropTargetType(dragIndicatorPositionForRowElement);
				sourceManager.SetDropTargetType(dropTargetType);
				sourceManager.ViewInfo.DropTargetRow = GetObjectByRowHandle(insertRowHandle);
				sourceManager.ShowDropMarker(rowContentElement, dragIndicatorPositionForRowElement, GetRowContentIndent(rowElement));
			}
			else
			{
				ClearDragInfo(sourceManager);
			}
		}

		protected override TableDragIndicatorPosition GetDragIndicatorPositionForRowElement(FrameworkElement rowElement)
		{
			if (rowElement == null)
			{
				return TableDragIndicatorPosition.None;
			}
			double num = base.LastPosition.Y - LayoutHelper.GetRelativeElementRect((UIElement)rowElement, (UIElement)base.DataControl).Top;
			if (num < rowElement.ActualHeight / 4.0)
			{
				return TableDragIndicatorPosition.Top;
			}
			if (num > rowElement.ActualHeight * 0.75)
			{
				return TableDragIndicatorPosition.Bottom;
			}
			return TableDragIndicatorPosition.InRow;
		}

		}

		public override IList GetSource(object row)
		{
			if (TreeListView.TreeDerivationMode == TreeDerivationMode.Selfreference)
			{
				return ItemsSource;
			}
			if (!(row is TreeListNode node))
			{
				return null;
			}
			return GetParentItemsSource(node);
		}

		public override void RemoveObject(object obj)
		{
			if (obj != null)
			{
				if (ItemsSource != null)
				{
					base.RemoveObject(obj);
				}
				else
				{
					TreeListView.DeleteNode(obj as TreeListNode, deleteChildren: false);
				}
			}
		}

		private IList GetParentItemsSource(TreeListNode node)
		{
			if (node.ParentNode != null)
			{
				return GetItemsSource(node.ParentNode);
			}
			return ItemsSource;
		}

		private IList GetItemsSource(TreeListNode node)
		{
			if (TreeListView.TreeDerivationMode == TreeDerivationMode.Selfreference || node == null)
			{
				return ItemsSource;
			}
			if (TreeListView.TreeDerivationMode == TreeDerivationMode.ChildNodesSelector || TreeListView.TreeDerivationMode == TreeDerivationMode.HierarchicalDataTemplate)
			{
				return node.ItemsSource;
			}
			return null;
		}

		protected internal override StartDragEventArgs RaiseStartDragEvent(IndependentMouseEventArgs e)
		{
			if (StartDraggingEventHandler != null)
			{
				TreeListViewHitInfo treeListViewHitInfo = (base.MouseDownHitInfo ?? GetHitInfo(e)) as TreeListViewHitInfo;
				TreeListStartDragEventArgs e2 = new TreeListStartDragEventArgs(base.DataControl as GridDataControlBase, this, treeListViewHitInfo, TreeListView.GetNodeByRowHandle(treeListViewHitInfo.RowHandle))
				{
					CanDrag = true
				};
				StartDraggingEventHandler(this, e2);
				return e2;
			}
			return null;
		}

		protected override int GetOverRowHandle(UIElement source, Point pt)
		{
			UIElement visibleHitTestElement = GetVisibleHitTestElement(pt);
			return GetHitInfo((DependencyObject)visibleHitTestElement).RowHandle;
		}

		protected override void PerformAutoExpand()
		{
			TreeListView.ExpandNode(HoverRowHandle);
		}

		protected internal override bool CanStartDrag(MouseButtonEventArgs e)
		{
			if (PresentationSource.FromVisual(base.DataControl) == null)
			{
				return false;
			}
			if (View.IsEditing || View.IsEditFormVisible)
			{
				return false;
			}
			base.CanStartDrag(e);
			TreeListViewHitInfo treeListViewHitInfo = base.MouseDownHitInfo as TreeListViewHitInfo;
			return treeListViewHitInfo.InRow;
		}

		internal override string GetItemsSourceErrorText()
		{
			return "The TreeList's data source cannot be converted to an IList object, thus the TreeListDragDropManager cannot function properly.";
		}

		internal class SelectedItemsCollection : Collection<int>
		{
			private ArrayList itemValues = new ArrayList();

			private RowSelectionBehavior behavior;

			public IList SelectedItemValues => itemValues;

			public SelectedItemsCollection(RowSelectionBehavior behavior)
			{
				this.behavior = behavior;
			}

			protected override void ClearItems()
			{
				IList arg = itemValues;
				itemValues = new ArrayList();
				base.ClearItems();
				if (behavior.onSelectionChanged != null)
				{
					behavior.onSelectionChanged(new object[0], arg);
				}
			}

			protected override void InsertItem(int index, int item)
			{
				object obj = behavior.Source[item];
				itemValues.Insert(index, obj);
				base.InsertItem(index, item);
				if (behavior.onSelectionChanged != null)
				{
					behavior.onSelectionChanged(new object[1] { obj }, new object[0]);
				}
			}

			protected override void RemoveItem(int index)
			{
				object obj = itemValues[index];
				itemValues.RemoveAt(index);
				base.RemoveItem(index);
				if (behavior.onSelectionChanged != null)
				{
					behavior.onSelectionChanged(new object[0], new object[1] { obj });
				}
			}

			protected override void SetItem(int index, int item)
			{
				object obj = itemValues[index];
				object obj2 = behavior.Source[item];
				itemValues[index] = obj2;
				base.SetItem(index, item);
				if (behavior.onSelectionChanged != null)
				{
					behavior.onSelectionChanged(new object[1] { obj2 }, new object[1] { obj });
				}
			}

			public void GrowIndexes(int newItemIndex)
			{
				for (int i = 0; i < base.Count; i++)
				{
					if (base[i] >= newItemIndex)
					{
						base.SetItem(i, base[i] + 1);
					}
				}
			}

			public void ShrinkIndexes(int oldItemIndex)
			{
				int num = -1;
				for (int i = 0; i < base.Count; i++)
				{
					if (base[i] == oldItemIndex)
					{
						num = i;
						base.SetItem(i, -1);
					}
					else if (base[i] > oldItemIndex)
					{
						base.SetItem(i, base[i] - 1);
					}
				}
				if (num >= 0)
				{
					RemoveAt(num);
				}
			}
		}

		public static readonly DependencyProperty IsAllRowsSelectedProperty;

		public RowSelectionBehavior()

		private void OnIsAllRowsSelectedChanged()
		{
			if (!IsAllRowsSelected.HasValue)
			{
				return;
			}
			if (AllowSelectionSynchronize)
			{
				if (IsAllRowsSelected.Value)
				{
					GridControl.SelectAll();
				}
				else
				{
					GridControl.UnselectAll();
				}
				return;
			}
			lockUpdateIsAllRowsSelected = true;
			SetSelectionColumnCellsValue(allowSetCellValue: false, (int handle) => IsAllRowsSelected.Value);
			GridControl.RefreshData();
			lockUpdateIsAllRowsSelected = false;
		}

		private void SetSelectionColumnCellsValue(bool allowSetCellValue, Predicate<int> predicate)
		{
			for (int i = 0; i < ItemsSourceCount; i++)
			{
				int rowHandleByListIndex = GridControl.GetRowHandleByListIndex(i);
				if (allowSetCellValue)
				{
					GridControl.SetCellValue(rowHandleByListIndex, "_SelectAllColumn", predicate(rowHandleByListIndex));
				}
				else if (predicate(rowHandleByListIndex))
				{
					if (!SelectedItems.Contains(i))
					{
						SelectedItems.Add(i);
					}
				}
				else
				{
					SelectedItems.Remove(i);
				}
			}
		}

		protected override void OnAttached()
		{
			((AttachableObjectBase)this).OnAttached();
			GridControl.CustomUnboundColumnData += OnCustomUnboundColumnData;
			GridControl.FilterChanged += OnFilterChanged;
			((ObservableCollection<GridColumn>)(object)GridControl.Columns).CollectionChanged += Columns_CollectionChanged;
			base.AssociatedObject.CellValueChanging += OnCellValueChanging;
			GridControl.SelectionChanged += OnSelectionChanged;
			UpdateSelectAllColumn();
			if (base.AssociatedObject is SelectionView)
			{
				ItemsSourceChangedHelper = new ItemsSourceChangedHelper(this);
				ItemsSourceChangedHelper.CollectionChanged += OnItemsSourceCollectionChanged;
				ItemsSourceChangedHelper.SubscribeOnItemsSourceChanged(base.AssociatedObject as SelectionView);
			}
		}

		private void Columns_CollectionChanged(object sender, NotifyCollectionChangedEventArgs e)
		{
			UpdateSelectAllColumn();
		}

		private void UpdateSelectAllColumn()
		{
			foreach (GridColumn item in (Collection<GridColumn>)(object)GridControl.Columns)
			{
				if (item is SelectAllColumn)
				{
					(item as SelectAllColumn).RowSelectionBehavior = this;
				}
			}
		}

		private void OnItemsSourceCollectionChanged(object sender, ListChangedEventArgs e)
		{
			switch (e.ListChangedType)
			{
			case ListChangedType.ItemAdded:
				SelectedItems.GrowIndexes(e.NewIndex);
				UpdateIsAllRowsSelectedProperty();
				break;
			case ListChangedType.ItemDeleted:
				SelectedItems.ShrinkIndexes(e.OldIndex);
				UpdateIsAllRowsSelectedProperty();
				break;
			case ListChangedType.Reset:
				SelectedItems.Clear();
				UpdateIsAllRowsSelectedProperty();
				break;
			}
		}

		private void OnSelectionChanged(object sender, GridSelectionChangedEventArgs e)
		{
			if (lockSelectionChanged || !AllowSelectionSynchronize)
			{
				return;
			}
			List<int> selectedHandles = new List<int>(GridControl.GetSelectedRowHandles());
			bool flag = false;
			for (int i = 0; i < selectedHandles.Count; i++)
			{
				if (selectedHandles[i] < 0)
				{
					flag = true;
					break;
				}
			}
			if (GridControl.VisibleRowCount == selectedHandles.Count && flag)
			{
				GridControl.BeginSelection();
				base.AssociatedObject.DataControl.UnselectAll();
				for (int j = 0; j < ItemsSourceCount; j++)
				{
					GridControl.SelectItem(j);
				}
				GridControl.EndSelection();
				return;
			}
			SelectedItems.Clear();
			for (int num = selectedHandles.Count - 1; num >= 0; num--)
			{
				if (GridControl.IsGroupRowHandle(selectedHandles[num]))
				{
					selectedHandles.RemoveAt(num);
				}
				else
				{
					int listIndexByRowHandle = GridControl.GetListIndexByRowHandle(selectedHandles[num]);
					if (!SelectedItems.Contains(listIndexByRowHandle))
					{
						SelectedItems.Add(listIndexByRowHandle);
					}
				}
			}
			SetSelectionColumnCellsValue(allowSetCellValue: true, (int handle) => selectedHandles.Contains(handle));
		}

		protected override void OnDetaching()
		{
			if (base.AssociatedObject is SelectionView)
			{
				ItemsSourceChangedHelper.CollectionChanged -= OnItemsSourceCollectionChanged;
				ItemsSourceChangedHelper.UnsubscribeFromItemsSourceChanged(base.AssociatedObject as SelectionView);
			}
			((ObservableCollection<GridColumn>)(object)GridControl.Columns).CollectionChanged -= Columns_CollectionChanged;
			GridControl.CustomUnboundColumnData -= OnCustomUnboundColumnData;
			GridControl.FilterChanged -= OnFilterChanged;
			base.AssociatedObject.CellValueChanging -= OnCellValueChanging;
			GridControl.SelectionChanged -= OnSelectionChanged;
			((AttachableObjectBase)this).OnDetaching();
		}

		private void OnCellValueChanging(object sender, CellValueChangedEventArgs e)
		{
			if (e.Column.FieldName == "_SelectAllColumn")
			{
				base.AssociatedObject.PostEditor();
			}
		}

		private void OnFilterChanged(object sender, RoutedEventArgs e)
		{
			UpdateIsAllRowsSelectedProperty();
		}

		private void OnCustomUnboundColumnData(object sender, GridColumnDataEventArgs e)
		{
			if (e.Column != null && e.Column.FieldName == "_SelectAllColumn")
			{
				if (e.IsGetData)
				{
					e.Value = GetIsSelected(e.ListSourceRowIndex);
				}
				if (e.IsSetData)
				{
					SetIsSelected(e.ListSourceRowIndex, (bool)e.Value);
				}
			}
		}

		private void UpdateIsAllRowsSelectedProperty()
		{
			if (!lockUpdateIsAllRowsSelected)
			{
				bool flag = true;
				bool flag2 = true;
				flag = SelectedItems.Count >= ItemsSourceCount;
				flag2 = SelectedItems.Count == 0;
				if (!flag && !flag2)
				{
					IsAllRowsSelected = null;
				}
				if (flag)
				{
					IsAllRowsSelected = true;
				}
				if (flag2)
				{
					IsAllRowsSelected = false;
				}
			}
		}

		private bool GetIsSelected(int listIndex)
		{
			return SelectedItems.Contains(listIndex);
		}

		private void SetIsSelected(int listIndex, bool value)
		{
			lockSelectionChanged = true;
			int rowHandleByListIndex = GridControl.GetRowHandleByListIndex(listIndex);
			if (value)
			{
				if (!SelectedItems.Contains(listIndex))
				{
					SelectedItems.Add(listIndex);
					if (AllowSelectionSynchronize)
					{
						GridControl.SelectItem(rowHandleByListIndex);
					}
					GridControl.SetCellValue(rowHandleByListIndex, "_SelectAllColumn", true);
				}
			}
			else if (SelectedItems.Contains(listIndex))
			{
				SelectedItems.Remove(listIndex);
				if (AllowSelectionSynchronize)
				{
					GridControl.UnselectItem(rowHandleByListIndex);
				}
				GridControl.SetCellValue(rowHandleByListIndex, "_SelectAllColumn", false);
			}
			lockSelectionChanged = false;
			UpdateIsAllRowsSelectedProperty();
		}

		{

			{
				((RowSelectionBehavior)(object)d).OnIsAllRowsSelectedChanged();
			}));
			SelectionControlWrapper.Wrappers.Add(typeof(RowSelectionBehavior), typeof(GridRowSelectionBehaviorSelectionControlWrapper));
		}

	public class ItemsSourceChangedHelper
	{
		private IRowSelectionBehavior behavior;

		private SelectionView View { get; set; }

		private GridControl Grid { get; set; }

		private object ItemsSource { get; set; }

		public event ListChangedEventHandler CollectionChanged;

		public ItemsSourceChangedHelper(IRowSelectionBehavior behavior)
		{
			this.behavior = behavior;
		}

		private void RaiseCollectionChanged(ListChangedEventArgs e)
		{
			if (this.CollectionChanged != null)
			{
				this.CollectionChanged(this, e);
			}
		}

		public void SubscribeOnItemsSourceChanged(SelectionView view)
		{
			View = view;
			view.GridChanged += view_OnGridChanged;
			SubscribeOnNewGrid();
		}

		public void UnsubscribeFromItemsSourceChanged(SelectionView view)
		{
			view.GridChanged -= view_OnGridChanged;
			UnsubscribeFromOldGrid();
			View = null;
		}

		private void view_OnGridChanged(object sender, EventArgs e)
		{
			UnsubscribeFromOldGrid();
			SubscribeOnNewGrid();
		}

		private void SubscribeOnNewGrid()
		{
			if (View.Grid != null)
			{
				View.Grid.ItemsSourceChanged += Grid_ItemsSourceChanged;
				Grid = View.Grid;
				SubscribeOnGridItemsSourceChanged();
			}
		}

		private void UnsubscribeFromOldGrid()
		{
			if (Grid != null)
			{
				Grid.ItemsSourceChanged -= Grid_ItemsSourceChanged;
				UnsubscribeFromGridItemsSourceChanged();
				Grid = null;
			}
		}

		private void SubscribeOnGridItemsSourceChanged()
		{
			ItemsSource = behavior.Source;
			SubscribeOnCollectionChanged(ItemsSource);
		}

		private void UnsubscribeFromGridItemsSourceChanged()
		{
			if (ItemsSource != null)
			{
				UnsubscribeFromCollectionChanged(ItemsSource);
				ItemsSource = null;
			}
		}

		private void SubscribeOnCollectionChanged(object itemsSource)
		{
			if (itemsSource is INotifyCollectionChanged)
			{
				(itemsSource as INotifyCollectionChanged).CollectionChanged += ItemsSourceChangedHelper_CollectionChanged;
			}
			else if (itemsSource is IBindingList)
			{
				(itemsSource as IBindingList).ListChanged += ItemsSourceChangedHelper_ListChanged;
			}
		}

		private void UnsubscribeFromCollectionChanged(object itemsSource)
		{
			if (itemsSource is INotifyCollectionChanged)
			{
				(itemsSource as INotifyCollectionChanged).CollectionChanged -= ItemsSourceChangedHelper_CollectionChanged;
			}
			else if (itemsSource is IBindingList)
			{
				(itemsSource as IBindingList).ListChanged -= ItemsSourceChangedHelper_ListChanged;
			}
		}

		private void ItemsSourceChangedHelper_ListChanged(object sender, ListChangedEventArgs e)
		{
			RaiseCollectionChanged(e);
		}

		private void ItemsSourceChangedHelper_CollectionChanged(object sender, NotifyCollectionChangedEventArgs e)
		{
			ListChangedType listChangedType = ListChangedType.Reset;
			if (e.Action == NotifyCollectionChangedAction.Add)
			{
				listChangedType = ListChangedType.ItemAdded;
			}
			if (e.Action == NotifyCollectionChangedAction.Move)
			{
				listChangedType = ListChangedType.ItemMoved;
			}
			if (e.Action == NotifyCollectionChangedAction.Remove)
			{
				listChangedType = ListChangedType.ItemDeleted;
			}
			RaiseCollectionChanged(new ListChangedEventArgs(listChangedType, e.NewStartingIndex, e.OldStartingIndex));
		}

		private void Grid_ItemsSourceChanged(object sender, ItemsSourceChangedEventArgs e)
		{
			UnsubscribeFromGridItemsSourceChanged();
			SubscribeOnGridItemsSourceChanged();
		}
	}

	public class RowSelectionStrategy : SelectionStrategyRow
	{
		private bool isMouseLeftButtonDownEventRasing;

		private SelectionActionBase tempSelectionAction;

		private bool selectionActionWasChanged;

		private SelectionView SelectionView => view as SelectionView;

		public RowSelectionStrategy(GridViewBase view)
			: base(view)
		{
		}

		public override void OnBeforeProcessKeyDown(KeyEventArgs e)
		{
			if (!SelectionView.DisableStandardSelection)
			{
				base.OnBeforeProcessKeyDown(e);
			}
		}

		public override void OnBeforeMouseLeftButtonDown(DependencyObject originalSource)
		{
			if (!SelectionView.DisableStandardSelection)
			{
				if (Mouse.RightButton != MouseButtonState.Pressed)
				{
					isMouseLeftButtonDownEventRasing = true;
				}
				ColumnBase columnByTreeElement = view.GetColumnByTreeElement(originalSource);
				if (SelectionView.NavigationStyle == GridViewNavigationStyle.Cell && columnByTreeElement != null && columnByTreeElement.FieldName == "_SelectAllColumn")
				{
					SelectionView.SetEditorSetInactiveAfterClick(value: false);
				}
				else
				{
					base.OnBeforeMouseLeftButtonDown(originalSource);
				}
			}
		}

		public override void OnAssignedToGrid()
		{
			if (SelectionView.DisableStandardSelection)
			{
				SelectionView.DataControl.UnselectAll();
			}
			else
			{
				base.OnAssignedToGrid();
			}
		}

		public override void OnDataSourceReset()
		{
			if (SelectionView.DisableStandardSelection)
			{
				SelectionView.DataControl.UnselectAll();
			}
			else
			{
				base.OnDataSourceReset();
			}
		}

		public override void OnAfterMouseLeftButtonDown(IDataViewHitInfo hitInfo)
		{
			base.OnAfterMouseLeftButtonDown(hitInfo);
			if (!isMouseLeftButtonDownEventRasing)
			{
				return;
			}
			if (tempSelectionAction != null)
			{
				if (base.SelectionAction == null)
				{
					base.SelectionAction = tempSelectionAction;
				}
				tempSelectionAction = null;
			}
			if (base.SelectionAction != null && selectionActionWasChanged)
			{
				ExecuteSelectionAction();
				selectionActionWasChanged = false;
			}
			isMouseLeftButtonDownEventRasing = false;
		}

		public override void CreateMouseSelectionActions(int rowHandle, bool isIndicator)
		{
			base.CreateMouseSelectionActions(rowHandle, isIndicator);
			if (base.SelectionAction is OnlyThisSelectionAction && base.GridView.DataControl.GetSelectedRowHandles().Length > 1 && isMouseLeftButtonDownEventRasing)
			{
				tempSelectionAction = base.SelectionAction;
				base.SelectionAction = null;
				selectionActionWasChanged = true;
			}
		}

		public override void OnFocusedColumnChanged()
		{
			base.OnFocusedColumnChanged();
			if (selectionActionWasChanged && view.DataControl.CurrentColumn.FieldName == "_SelectAllColumn")
			{
				base.SelectionAction = new InvertSelectionAction(base.GridView);
			}
		}
	}

		{
			object content = ((TreeListNodeBase)insertNode).Content;
			targetManager.DataControl.BeginDataUpdate();
			TreeListNodeCollection treeListNodeCollection = null;
			int num = 0;
			switch (dropTargetType)
			{
			case DropTargetType.InsertRowsAfter:
			case DropTargetType.InsertRowsBefore:
				treeListNodeCollection = GetParentCollection(insertNode, targetManager.TreeListView);

			case DropTargetType.InsertRowsIntoNode:

				break;
			case DropTargetType.DataArea:

	public static class Utils
	{
		public static object GetPropertyValue(object obj, string propertyName)
		{
			if (obj is DataRow dataRow)
			{
				return dataRow[propertyName];
			}
			return GetPropertyDescriptor(obj, propertyName)?.GetValue(obj);
		}

		public static void SetPropertyValue(object obj, string propertyName, object value)
		{
			if (obj is DataRow dataRow)
			{
				dataRow[propertyName] = value;
			}
			else
			{
				GetPropertyDescriptor(obj, propertyName)?.SetValue(obj, value);
			}
		}

		private static PropertyDescriptor GetPropertyDescriptor(object obj, string propertyName)
		{
			//IL_000f: Unknown result type (might be due to invalid IL or missing references)
			//IL_0015: Expected O, but got Unknown
			if (propertyName.Contains("."))
			{
				return (PropertyDescriptor)new ComplexPropertyDescriptorReflection(obj, propertyName);
			}
			return TypeDescriptor.GetProperties(obj)[propertyName];
		}

		public static DataRow CloneDataRow(DataView dataView, object sourceMngObj)
		{
			DataTable table = dataView.Table;
			DataRow dataRow = table.NewRow();
			foreach (DataColumn column in table.Columns)
			{
				dataRow[column] = GetPropertyValue(sourceMngObj, column.ColumnName) ?? DBNull.Value;
			}
			return dataRow;
		}
	}
