	private void MakePathInEnv()
	{
		string location = GetType().Assembly.Location;
		FileInfo fileInfo = new FileInfo(location);
		if (fileInfo == null)
		{
			return;
		}
		location = fileInfo.DirectoryName;
		if (!string.IsNullOrWhiteSpace(location))
		{
			string environmentVariable = Environment.GetEnvironmentVariable("path");
			if (!string.IsNullOrWhiteSpace(environmentVariable) && environmentVariable.IndexOf(location, StringComparison.InvariantCultureIgnoreCase) < 0)
			{
				StringBuilder stringBuilder = new StringBuilder();
				stringBuilder.AppendFormat("{0};{1}", location, environmentVariable);
				environmentVariable = stringBuilder.ToString();
				Environment.SetEnvironmentVariable("path", environmentVariable);
			}
		}
	}

	public HtmlHelpLauncher()
	{
		MakePathInEnv();
		htmlHelpStrapper = HtmlHelpLauncherUnityBootstrapper.GetBootstrapper();
		htmlHelpStrapper.Run();
	}

	public void HtmlHelpInit(string strProduct, string strExtVersion, string strInVersion, string strLandingPage, string strProdDisplayName)
	{
		if (htmlHelpHelper == null)
		{
			htmlHelpHelper = UnityContainerExtensions.Resolve<IHtmlHelpHelper>(htmlHelpStrapper.Container, Array.Empty<ResolverOverride>());
			htmlHelpHelper.InitHtmlHelpSystem(strProduct, strExtVersion, strInVersion, strLandingPage, strProdDisplayName, string.Empty, bNoSelfExtract: false);
		}
	}

	public void HtmlHelpInit2(string strProduct, string strExtVersion, string strInVersion, string strLandingPage, string strProdDisplayName, string strHelpExtAppName, bool bNoSelfExtract)
	{
		if (htmlHelpHelper == null)
		{
			htmlHelpHelper = UnityContainerExtensions.Resolve<IHtmlHelpHelper>(htmlHelpStrapper.Container, Array.Empty<ResolverOverride>());
			htmlHelpHelper.InitHtmlHelpSystem(strProduct, strExtVersion, strInVersion, strLandingPage, strProdDisplayName, strHelpExtAppName, bNoSelfExtract);
		}
	}

	public void HtmlHelpInit3(string strProduct, string strExtVersion, string strInVersion, string strLandingPage, string strProdDisplayName, string strHelpExtAppName, bool bNoSelfExtract, bool displayMoreThanOnce)
	{
		if (displayMoreThanOnce)
		{
			htmlHelpHelper = UnityContainerExtensions.Resolve<IHtmlHelpHelper>(htmlHelpStrapper.Container, Array.Empty<ResolverOverride>());
			htmlHelpHelper.InitHtmlHelpSystem(strProduct, strExtVersion, strInVersion, strLandingPage, strProdDisplayName, strHelpExtAppName, bNoSelfExtract, displayMoreThanOnce);
		}
	}

	public void HtmlHelpClose()
	{
		if (htmlHelpHelper != null)
		{
			htmlHelpHelper.HtmlHelpDispose();
			htmlHelpHelper = null;
		}
	}

	public void HtmlHelpDisplayTopic(string strHelpTopicUrl)
	{
		if (htmlHelpHelper != null)
		{
			htmlHelpHelper.HtmlHelpDisplayF1Link(strHelpTopicUrl);
		}
	}

	public void SetVisible(bool visible)

		{
			htmlHelpHelper.SetVisible(visible);
		}
	}

	public void UseWinFormsBrowser(bool useBrowser)

			htmlHelpHelper.UseWinFormsBrowser = useBrowser;
		}
	}

	public void UseAspenTechStdIcon(bool useATStdIcon)

internal class HtmlHelpLauncherModuleCatalog : ModuleCatalog
{
	protected override void InnerLoad()
	{
		foreach (ModuleInfo item in from t in GetType().Assembly.GetExportedTypes().Where(typeof(IModule).IsAssignableFrom)
			where t != typeof(IModule)
			where !t.IsAbstract
			select CreateModuleInfo(t))
		{
			AddModule(item);
		}
		AddModule("AspenTech.HtmlHelp.HelpIntegration.HtmlHelpIntegrationModule", typeof(HtmlHelpIntegrationModule).AssemblyQualifiedName);
	}

	private ModuleInfo CreateModuleInfo(Type type)
	{
		if (type == null)
		{
			return null;
		}
		string name = type.Name;
		bool flag = false;
		CustomAttributeData customAttributeData = CustomAttributeData.GetCustomAttributes(type).FirstOrDefault((CustomAttributeData cad) => cad.Constructor.DeclaringType.FullName == typeof(ModuleAttribute).FullName);
		if (customAttributeData != null)
		{
			foreach (CustomAttributeNamedArgument namedArgument in customAttributeData.NamedArguments)
			{
				switch (namedArgument.MemberInfo.Name)
				{
				case "ModuleName":
					name = (string)namedArgument.TypedValue.Value;
					break;
				case "OnDemand":
					flag = (bool)namedArgument.TypedValue.Value;
					break;
				case "StartupLoaded":
					flag = !(bool)namedArgument.TypedValue.Value;
					break;
				}
			}
		}
		ModuleInfo moduleInfo = new ModuleInfo(name, type.AssemblyQualifiedName)
		{
			InitializationMode = (flag ? InitializationMode.OnDemand : InitializationMode.WhenAvailable),
			Ref = type.Assembly.CodeBase
		};
		foreach (CustomAttributeData item in from cad in CustomAttributeData.GetCustomAttributes(type)
			where cad.Constructor.DeclaringType.FullName == typeof(ModuleDependencyAttribute).FullName
			select cad)
		{
			moduleInfo.DependsOn.Add((string)item.ConstructorArguments[0].Value);
		}
		return moduleInfo;
	}
}

public enum CompositeEventRoutingStrategy
{
	Children,
	Parents,
	ChildrenThenParents,
	ParentsThenChildren,
	FirstOfParentsThenChildren,
	FirstOfChildrenThenParents,
	FirstOfParents,
	AllTopDown
}
public class RoutingEventAggregatorUnityExtension : UnityContainerExtension
{
	protected override void Initialize()
	{
	}

	private IEnumerable<IUnityContainer> EnumerateContainerChildren()
	{
		ILifetimeContainer lifetime = ((UnityContainerExtension)this).Context.Lifetime;
		if (lifetime == null)
		{
			yield break;
		}
		foreach (object item in (IEnumerable<object>)lifetime)
		{
			IUnityContainer val = (IUnityContainer)((item is IUnityContainer) ? item : null);
			if (val != null)
			{
				yield return val;
			}
		}
	}

	public static IEnumerable<IUnityContainer> EnumerateContainerChildren(IUnityContainer container)
	{
		RoutingEventAggregatorUnityExtension routingEventAggregatorUnityExtension = UnityContainerExtensions.Configure<RoutingEventAggregatorUnityExtension>(container);
		if (routingEventAggregatorUnityExtension == null)
		{
			yield break;
		}
		foreach (IUnityContainer item in routingEventAggregatorUnityExtension.EnumerateContainerChildren())
		{
			yield return item;
		}
	}
}
public class RoutingEventAggregator : IEventAggregator
{
	private readonly List<EventBase> _events = new List<EventBase>();

	public IUnityContainer Container { get; private set; }

	public RoutingEventAggregator(IUnityContainer container)
	{
		Container = container;
	}

	public TEventType GetEvent<TEventType>() where TEventType : EventBase
	{
		TEventType val = _events.FirstOrDefault((EventBase evt) => evt.GetType() == typeof(TEventType)) as TEventType;
		if (val == null)
		{
			val = UnityContainerExtensions.Resolve<TEventType>(Container, Array.Empty<ResolverOverride>());
			_events.Add(val);
		}
		return val;
	}

	internal void RoutedPublish<TPayload>(CompositeEventRoutingStrategy strategy, RoutedCompositePresentationEvent<TPayload> eventInstance, TPayload payload)
	{
		Type type = eventInstance.GetType();
		if (strategy == CompositeEventRoutingStrategy.AllTopDown)
		{
			IUnityContainer val = Container;
			while (val.Parent != null)
			{
				val = val.Parent;
			}
			PublishToContainer(val, type, payload);
			PublishToChildren(val, type, payload, stopAtFirstSubscriber: false);
			return;
		}
		bool flag = false;
		if (eventInstance.PublishToPeers(payload))
		{
			flag = true;
		}
		switch (strategy)
		{
		case CompositeEventRoutingStrategy.Children:
			PublishToChildren(Container, type, payload, stopAtFirstSubscriber: false);
			break;
		case CompositeEventRoutingStrategy.Parents:
			PublishToParents(Container, type, payload, stopAtFirstSubscriber: false);
			break;
		case CompositeEventRoutingStrategy.ChildrenThenParents:
			PublishToChildren(Container, type, payload, stopAtFirstSubscriber: false);
			PublishToParents(Container, type, payload, stopAtFirstSubscriber: false);
			break;
		case CompositeEventRoutingStrategy.ParentsThenChildren:
			PublishToParents(Container, type, payload, stopAtFirstSubscriber: false);
			PublishToChildren(Container, type, payload, stopAtFirstSubscriber: false);
			break;
		case CompositeEventRoutingStrategy.FirstOfChildrenThenParents:
			if (!flag && !PublishToChildren(Container, type, payload, stopAtFirstSubscriber: true))
			{
				PublishToParents(Container, type, payload, stopAtFirstSubscriber: true);
			}
			break;
		case CompositeEventRoutingStrategy.FirstOfParentsThenChildren:
			if (!flag && !PublishToParents(Container, type, payload, stopAtFirstSubscriber: true))
			{
				PublishToChildren(Container, type, payload, stopAtFirstSubscriber: true);
			}
			break;
		case CompositeEventRoutingStrategy.FirstOfParents:
			if (!flag)
			{
				PublishToParents(Container, type, payload, stopAtFirstSubscriber: true);
			}
			break;
		}
	}

	private bool PublishToParents<TPayload>(IUnityContainer container, Type eventType, TPayload payload, bool stopAtFirstSubscriber)
	{
		bool result = false;
		IUnityContainer parent = container.Parent;
		if (parent != null)
		{
			RoutingEventAggregator routingEventAggregator = null;
			try
			{
				routingEventAggregator = UnityContainerExtensions.Resolve<IEventAggregator>(parent, Array.Empty<ResolverOverride>()) as RoutingEventAggregator;
			}
			catch (Exception)
			{
			}
			if (routingEventAggregator != null && routingEventAggregator._events.FirstOrDefault((EventBase evt) => evt.GetType() == eventType) is RoutedCompositePresentationEvent<TPayload> routedCompositePresentationEvent && routedCompositePresentationEvent.PublishToPeers(payload))
			{
				result = true;
				if (stopAtFirstSubscriber)
				{
					return result;
				}
			}
			if (PublishToParents(parent, eventType, payload, stopAtFirstSubscriber))
			{
				result = true;
			}
		}
		return result;
	}

	private bool PublishToContainer<TPayload>(IUnityContainer container, Type eventType, TPayload payload)
	{
		bool result = false;
		RoutingEventAggregator routingEventAggregator = null;
		try
		{
			routingEventAggregator = UnityContainerExtensions.Resolve<IEventAggregator>(container, Array.Empty<ResolverOverride>()) as RoutingEventAggregator;
		}
		catch (Exception)
		{
		}
		if (routingEventAggregator != null && routingEventAggregator._events.FirstOrDefault((EventBase evt) => evt.GetType() == eventType) is RoutedCompositePresentationEvent<TPayload> routedCompositePresentationEvent && routedCompositePresentationEvent.PublishToPeers(payload))
		{
			result = true;
		}
		return result;
	}

	private bool PublishToChildren<TPayload>(IUnityContainer container, Type eventType, TPayload payload, bool stopAtFirstSubscriber)
	{
		bool result = false;
		foreach (IUnityContainer item in RoutingEventAggregatorUnityExtension.EnumerateContainerChildren(container))
		{
			RoutingEventAggregator routingEventAggregator = null;
			try
			{
				routingEventAggregator = UnityContainerExtensions.Resolve<IEventAggregator>(item, Array.Empty<ResolverOverride>()) as RoutingEventAggregator;
			}
			catch (Exception)
			{
			}
			if (routingEventAggregator != null && routingEventAggregator._events.FirstOrDefault((EventBase evt) => evt.GetType() == eventType) is RoutedCompositePresentationEvent<TPayload> routedCompositePresentationEvent && routedCompositePresentationEvent.PublishToPeers(payload))
			{
				result = true;
				if (stopAtFirstSubscriber)
				{
					break;
				}
			}
			if (PublishToChildren(item, eventType, payload, stopAtFirstSubscriber))
			{
				result = true;
				if (stopAtFirstSubscriber)
				{
					break;
				}
			}
		}
		return result;
	}
}
public class RoutedCompositePresentationEvent<TPayload> : CompositePresentationEvent<TPayload>
{
	public CompositeEventRoutingStrategy RoutingStrategy { get; set; }

	[Dependency]
	public IEventAggregator EventAggregator { get; set; }

	public RoutedCompositePresentationEvent()
	{
		RoutingStrategy = CompositeEventRoutingStrategy.ChildrenThenParents;
	}

	public override void Publish(TPayload payload)
	{
		Publish(RoutingStrategy, payload);
	}

	public void Publish(CompositeEventRoutingStrategy strategy, TPayload payload)
	{
		if (EventAggregator is RoutingEventAggregator routingEventAggregator)
		{
			routingEventAggregator.RoutedPublish(strategy, this, payload);
		}
		else
		{
			PublishToPeers(payload);
		}
	}

	public bool PublishToPeers(TPayload payload)
	{
		if (base.Subscriptions.Count == 0)
		{
			return false;
		}
		base.InternalPublish(payload);
		return true;
	}
