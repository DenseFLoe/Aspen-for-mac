		public static Assembly SafeLoad(string file, AssemblyLoadType type, int verifyflag, List<byte[]> tokens = null)
		{
			Assembly assembly = null;
			string text = string.Empty;
			try
			{
				bool flag = (verifyflag & 1) == 1;
				bool flag2 = (verifyflag & 2) == 2;

					bool fWasVerified = false;
					if (clrStrongName.StrongNameSignatureVerificationEx(file, fForceVerification: true, out fWasVerified) != 0 || !fWasVerified)
					{
						text = "Failed to verify signature for file: " + file;
					}
				}
				if (string.IsNullOrWhiteSpace(text))
				{
					switch (type)
					{
					case AssemblyLoadType.LoadFrom:
						assembly = Assembly.LoadFrom(file);
						break;
					case AssemblyLoadType.LoadFile:
						assembly = Assembly.LoadFile(file);
						break;
					default:
					{
						byte[] rawAssembly = File.ReadAllBytes(file);
						assembly = Assembly.Load(rawAssembly);
						break;
					}
					}
					if (!flag2 || tokens == null || tokens.Count == 0)
					{
						return assembly;
					}
					byte[] publicKeyToken = assembly.GetName().GetPublicKeyToken();
					foreach (byte[] token in tokens)
					{
						if (publicKeyToken.Length != token.Length)
						{
							continue;
						}
						bool flag3 = true;
						for (int i = 0; i < publicKeyToken.Length; i++)
						{
							if (publicKeyToken[i] != token[i])
							{
								flag3 = false;
								break;
							}
						}
						if (flag3)
						{
							return assembly;
						}
					}
					text = "Failed to verify token for file: " + file;
				}
			}
			catch (Exception innerException)
			{
				throw new Exception("Failed to load assembly: " + file, innerException);
			}
			throw new Exception(text);
		}

		public static bool CheckFileName(string fileName)
		{
			if (string.IsNullOrWhiteSpace(fileName))
			{
				return false;

				Action method = delegate
				{
					this._OnAddInError(this, new AddInDataReceivedEventArgs
					{
						Data = sortOutput.ToString()

				base.Dispatcher.BeginInvoke(method);
			}
		}

		public AddInInformation LoadOldAddIn(string pathOfTool, int verifyflag, List<byte[]> tokens)
		{
			AddInInformation addInInformation = new AddInInformation();
			if (Directory.Exists(pathOfTool))
			{
				try
				{
					addInInformation.Path = GetIEquipmentSelectionAssemblyPath(pathOfTool, verifyflag, tokens);
					if (!string.IsNullOrEmpty(addInInformation.Path))
					{
						addInInformation.AddInMode = LaunchToolMode.NewAppDomain;
					}
					else
					{
						addInInformation.AddInMode = LaunchToolMode.NoAddIn;
					}
				}
				catch (Exception)
				{
					addInInformation.Path = string.Empty;
					addInInformation.AddInMode = LaunchToolMode.NoAddIn;
				}
			}
			return addInInformation;
		}

		private static string GetIEquipmentSelectionAssemblyPath(string path, int verifyflag, List<byte[]> tokens)
		{
			string text = string.Empty;
			try
			{
				DirectoryInfo directoryInfo = new DirectoryInfo(path);
				FileInfo[] files = directoryInfo.GetFiles("*.dll", SearchOption.TopDirectoryOnly);
				FileInfo[] array = files;
				foreach (FileInfo fileInfo in array)
				{
					Assembly assembly = null;
					try
					{
						assembly = SharedUtility.SafeLoad(fileInfo.FullName, AssemblyLoadType.Load, verifyflag, tokens);
					}
					catch (Exception)
					{
					}
					if (!(assembly != null) || !(assembly.FullName != typeof(IEquipmentSelection).Assembly.FullName))
					{
						continue;
					}
					Type[] types = assembly.GetTypes();
					foreach (Type c in types)
					{
						if (typeof(IEquipmentSelection).IsAssignableFrom(c))
						{
							text = fileInfo.FullName;
							break;
						}
					}
					if (!string.IsNullOrEmpty(text))
					{
						break;
					}
				}
				if (string.IsNullOrEmpty(text))
				{
					files = directoryInfo.GetFiles("*.exe", SearchOption.TopDirectoryOnly);
					FileInfo[] array2 = files;
					foreach (FileInfo fileInfo2 in array2)
					{
						Assembly assembly2 = null;
						try
						{
							assembly2 = SharedUtility.SafeLoad(fileInfo2.FullName, AssemblyLoadType.Load, verifyflag, tokens);
						}
						catch (Exception)
						{
						}
						if (!(assembly2 != null) || !(assembly2.FullName != typeof(IEquipmentSelection).Assembly.FullName))
						{
							continue;
						}
						Type[] types2 = assembly2.GetTypes();
						foreach (Type c2 in types2)
						{
							if (typeof(IEquipmentSelection).IsAssignableFrom(c2))
							{
								text = fileInfo2.FullName;
								break;
							}
						}
						if (!string.IsNullOrEmpty(text))
						{
							break;
						}
					}
				}
			}
			catch (Exception)
			{
				text = string.Empty;
			}
			return text;
		}

	public static class Utilities
	{
		internal static IpcServerChannel RegisterServerChannel(string baseServerName)
		{
			BinaryServerFormatterSinkProvider binaryServerFormatterSinkProvider = new BinaryServerFormatterSinkProvider();
			binaryServerFormatterSinkProvider.TypeFilterLevel = TypeFilterLevel.Full;
			IpcServerChannel ipcServerChannel = new IpcServerChannel(null, baseServerName + "-" + Process.GetCurrentProcess().Id, binaryServerFormatterSinkProvider);
			ChannelServices.RegisterChannel(ipcServerChannel, ensureSecurity: false);
			return ipcServerChannel;
		}

		public static string GetAspenExchangeAddInPath(string pathOfAddInFolder)
		{
			string result = string.Empty;
			if (!string.IsNullOrEmpty(pathOfAddInFolder) && Directory.Exists(pathOfAddInFolder))
			{
				string text = Path.Combine(pathOfAddInFolder, "AspenExchangeAddIn.xml");
				if (File.Exists(text))
				{
					XmlDocument xmlDocument = new XmlDocument();
					xmlDocument.Load(text);
					XmlNodeList elementsByTagName = xmlDocument.GetElementsByTagName("AddIn");
					if (elementsByTagName != null && elementsByTagName.Count > 0)
					{
						XmlElement xmlElement = elementsByTagName[0] as XmlElement;
						if (xmlElement.HasAttribute("EntryPoint"))
						{
							string attribute = xmlElement.GetAttribute("EntryPoint");
							result = Path.Combine(pathOfAddInFolder, attribute);
						}
					}
				}
			}
			return result;
		}

		public static bool RunAsAddIn(string remotingURL, IAddInControl control)
		{
			if (control == null || control.AddInUI == null)
			{
				throw new ApplicationException("No instance for IAddInControl or AddInUI is not implement!");
			}
			ICrossProcAddInSite site = (ICrossProcAddInSite)Activator.GetObject(typeof(ICrossProcAddInSite), remotingURL);
			RegisterServerChannel("AspenExchange-WPFAddIn");
			bool normalExit = false;
			new Thread((ThreadStart)delegate
			{
				Thread.CurrentThread.IsBackground = true;
				Process processById = Process.GetProcessById(site.HostProcessId);
				if (processById != null)
				{
					processById.WaitForExit();
					if (!normalExit)
					{
						_ = "WARNING: Host process:" + site.HostProcessId + " quit unexpectedly!";
						Environment.Exit(2);
					}
				}
			}).Start();
			new AddIn(site, control);
			Dispatcher.Run();
			normalExit = true;
			return normalExit;
		}
