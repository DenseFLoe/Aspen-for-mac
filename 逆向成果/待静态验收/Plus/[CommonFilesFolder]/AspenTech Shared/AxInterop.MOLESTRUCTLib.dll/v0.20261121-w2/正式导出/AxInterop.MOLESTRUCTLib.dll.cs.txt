public delegate void _DMoleStructEvents_CalcStructEventHandler(object sender, _DMoleStructEvents_CalcStructEvent e);
public class _DMoleStructEvents_CalcStructEvent
{
	public int atom1;

	public string atom1Type;

	public int atom2;

	public string atom2Type;

	public string bondType;

	public _DMoleStructEvents_CalcStructEvent(int atom1, string atom1Type, int atom2, string atom2Type, string bondType)
	{
		this.atom1 = atom1;
		this.atom1Type = atom1Type;
		this.atom2 = atom2;
		this.atom2Type = atom2Type;
		this.bondType = bondType;
	}
}

[ClassInterface(ClassInterfaceType.None)]
public class AxMoleStructEventMulticaster : _DMoleStructEvents
{
	private AxMoleStruct parent;

	public AxMoleStructEventMulticaster(AxMoleStruct parent)
	{
		this.parent = parent;
	}

	public virtual void CalcStruct(int atom1, string atom1Type, int atom2, string atom2Type, string bondType)
	{
		_DMoleStructEvents_CalcStructEvent e = new _DMoleStructEvents_CalcStructEvent(atom1, atom1Type, atom2, atom2Type, bondType);
		parent.RaiseOnCalcStruct(parent, e);
	}

	public virtual void CleanGenTab()
	{
		EventArgs e = new EventArgs();
		parent.RaiseOnCleanGenTab(parent, e);
	}

	public virtual void CleanFormulaTab()

	}

	public virtual void FillFormula(string atom, int noAtom)
	{
		_DMoleStructEvents_FillFormulaEvent e = new _DMoleStructEvents_FillFormulaEvent(atom, noAtom);
		parent.RaiseOnFillFormula(parent, e);
	}

	public virtual void AfterImportStructure(string compName, string molFilePath, string molCompress)
	{
		_DMoleStructEvents_AfterImportStructureEvent e = new _DMoleStructEvents_AfterImportStructureEvent(compName, molFilePath, molCompress);
		parent.RaiseOnAfterImportStructure(parent, e);
	}

	public virtual void AfterSaveStructure(string moleFileString)
