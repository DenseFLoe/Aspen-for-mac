	public About()
	{
		object[] customAttributes = GetType().GetCustomAttributes(typeof(AboutSnapinAttribute), inherit: true);
		if (customAttributes.Length != 0)
		{
			AboutSnapinAttribute aboutSnapinAttribute = (AboutSnapinAttribute)customAttributes[0];
			customAttributes = aboutSnapinAttribute.SnapinType.GetCustomAttributes(typeof(SnapinInAttribute), inherit: true);
			if (customAttributes.Length != 0)
			{
				SnapinInAttribute snapinInAttribute = (SnapinInAttribute)customAttributes[0];
				m_Provider = snapinInAttribute.Provider;
				if (snapinInAttribute.Version != null)
				{
					m_Version = snapinInAttribute.Version;
				}
				else
				{
					m_Version = GetType().Module.Assembly.GetName().Version.ToString();
				}
			}
		}
		m_Mask = Color.Black;
		staticFolderIcons.Insert(0, "");
		staticFolderIcons.Insert(1, "");
		staticFolderIcons.Insert(2, "");
		staticFolderIcons.Insert(3, "");
	}

	private IntPtr TurnOnThemes()
	{
		ACTCTX pActCtx = default(ACTCTX);
		pActCtx.cbSize = (uint)Marshal.SizeOf(typeof(ACTCTX));
		pActCtx.dwFlags = 8u;
		pActCtx.lpSource = GetType().Module.Assembly.Location;
		pActCtx.wProcessorArchitecture = 0;
		pActCtx.wLangId = 0;
		pActCtx.lpAssemblyDirectory = null;
		pActCtx.lpResourceName = "ENABLETHEMES";
		pActCtx.lpApplicationName = null;
		pActCtx.hModule = IntPtr.Zero;
		IntPtr intPtr = CreateActCtx(ref pActCtx);
		uint lpCookie = 0u;
		bool flag = ActivateActCtx(intPtr, out lpCookie);
		return intPtr;
	}

	private bool HasThemes()
	{
		if (Environment.OSVersion.Version.Major >= 5 && Environment.OSVersion.Version.Minor > 0)
		{
			return true;
		}
		return false;
	}

	public void Remove(bool removeChildrenOnly)
	{
		if (HScopeItem != 0)
		{
			Snapin.ConsoleNameSpace.DeleteItem((uint)HScopeItem, Convert.ToInt32(!removeChildrenOnly));
		}
		foreach (BaseNode childNode in m_ChildNodes)
		{
			if (childNode.NumChildren != 0)
			{
				childNode.Remove(removeChildrenOnly: true);
			}
			Snapin.Unregister(childNode);
		}
		m_ChildNodes.Clear();
		if (!removeChildrenOnly)
		{
			if (ParentHScopeItem != 0 && Snapin.FindNodeByHScope(ParentHScopeItem) != null)
			{
				Snapin.ResultViewConsole.SelectScopeItem(ParentHScopeItem);
			}
			Snapin.Unregister(this);
		}
	}

	public void Insert(BaseNode parent)
	{
		IConsoleNameSpace2 consoleNameSpace = Snapin.ConsoleNameSpace;
		SCOPEDATAITEM a = GetScopeDataItem(parent);
		consoleNameSpace.InsertItem(ref a);
		HScopeItem = a.ID;
		ParentHScopeItem = parent.HScopeItem;
		ParentNode = parent;
	}

	public virtual SCOPEDATAITEM GetScopeDataItem(BaseNode parentNode)
	{
		return new SCOPEDATAITEM
		{
			mask = 110u,
			nImage = ClosedImageIndex,
			nOpenImage = OpenImageIndex,
			relativeID = parentNode.HScopeItem,
			displayname = (IntPtr)(-1),
			lParam = Cookie,
			cChildren = 1
		};
	}

	public void SetScopeViewItemImage()
	{
		IConsoleNameSpace2 consoleNameSpace = Snapin.ConsoleNameSpace;
		SCOPEDATAITEM a = new SCOPEDATAITEM
		{

	public static Type GetSystemTypeFromAssembly(string asmNameBeginning, string controlFullName)
	{
		Type type = null;
		try
		{
			type = Type.GetType(controlFullName, throwOnError: true);
			if (type != null)
			{
				return type;
			}
		}
		catch
		{
		}
		try
		{
			Assembly callingAssembly = Assembly.GetCallingAssembly();
			callingAssembly.GetType(controlFullName, throwOnError: true);
		}
		catch
		{
		}
		bool flag = false;
		Assembly[] assemblies = AppDomain.CurrentDomain.GetAssemblies();
		foreach (Assembly assembly in assemblies)
		{
			if (!assembly.FullName.StartsWith(asmNameBeginning))
			{
				continue;
			}
			try
			{
				flag = true;
				type = assembly.GetType(controlFullName, throwOnError: true);
				if (type != null)
				{
					return type;
				}
			}
			catch (Exception ex)
			{
				MessageBox.Show("Failed to get the type AspenTech.APED.FindCompoundGUI.DatabankSpecCtl " + ex.Message);
			}
		}
		if (flag)
		{
			return type;
		}
		AssemblyName[] referencedAssemblies = Assembly.GetCallingAssembly().GetReferencedAssemblies();
		foreach (AssemblyName assemblyName in referencedAssemblies)
		{
			string fullName = assemblyName.FullName;
			if (!fullName.StartsWith(asmNameBeginning))
			{
				continue;
			}
			flag = true;
			try
			{
				Assembly assembly2 = Assembly.Load(fullName);
				type = assembly2.GetType(controlFullName, throwOnError: true);
				if (type != null)
				{
					return type;
				}
			}
			catch (Exception ex2)
			{
				MessageBox.Show("Failed to get the type AspenTech.APED.FindCompoundGUI.DatabankSpecCtl " + ex2.Message);
			}
		}
		return type;
	}

	public int Notify(IntPtr lpDataObject, uint aevent, IntPtr arg, IntPtr param)
	{
		int result = 1;
		try
		{
			if (lpDataObject == (IntPtr)DataObject.CUSTOMOCX || lpDataObject == (IntPtr)DataObject.CUSTOMWEB)
			{
				return result;
			}
			IDataObject dataObject = null;
			if (lpDataObject != (IntPtr)0)
			{
				dataObject = (IDataObject)Marshal.GetObjectForIUnknown(lpDataObject);
			}
			result = 0;
			BaseNode baseNode;
			if (dataObject != null)
			{
				baseNode = ((DataObject)dataObject).Node;
			}
			else
			{
				RESULTDATAITEM selectedItem = m_Snapin.GetSelectedItem((IResultData)m_Console);
				if (selectedItem.nIndex != -1)
				{
					baseNode = m_Snapin.FindNode(selectedItem.lParam);
				}
				else
				{
					baseNode = m_Snapin.CurrentScopeNode;
					if (baseNode == null)
					{
						baseNode = m_Snapin.RootNode;
					}
				}
			}
			switch ((MMCN_Notifiy)aevent)
			{
			case MMCN_Notifiy.ACTIVATE:
			{
				bool bActivate = arg.ToInt32() != 0;
				baseNode.OnActivate(bActivate);
				break;
			}
			case MMCN_Notifiy.ADD_IMAGES:
			{
				IImageList il = (IImageList)Marshal.GetObjectForIUnknown(arg);
				BaseNode baseNode4 = m_Snapin.FindNodeByHScope((int)param);
				baseNode4.OnAddResultPaneImages(il);
				break;
			}
			case MMCN_Notifiy.BTN_CLICK:
			{
				int cmdId = param.ToInt32();
				baseNode.OnBtnClick(cmdId);
				break;
			}
			case MMCN_Notifiy.CLICK:
				baseNode.OnClick();
				break;
			case MMCN_Notifiy.COLUMN_CLICK:
			{
				int nCol = (int)arg;
				bool bAscending = (int)param == 0;
				baseNode.OnColumnClick(nCol, bAscending);
				break;
			}
			case MMCN_Notifiy.CUTORMOVE:
			{
				IDataObject ido2 = (IDataObject)Marshal.GetObjectForIUnknown(arg);
				if (!baseNode.OnCutOrRemove(ido2))
				{
					result = 1;
				}
				break;
			}
			case MMCN_Notifiy.DBLCLICK:
				result = 1;
				if (baseNode.OnDblClick())
				{
					result = 0;
				}
				break;
			case MMCN_Notifiy.DELETE:
				baseNode.OnDelete();
				break;
			case MMCN_Notifiy.DESELECT_ALL:
				baseNode.OnDeselectAll();
				break;
			case MMCN_Notifiy.EXPAND:
				if ((int)arg == 0)
				{
					baseNode.OnExpand();
				}
				else
				{
					baseNode.OnCollapse();
				}
				break;
			case MMCN_Notifiy.MENU_BTNCLICK:
			{
				MENUBUTTONDATA mENUBUTTONDATA = (MENUBUTTONDATA)Marshal.PtrToStructure(param, typeof(MENUBUTTONDATA));
				baseNode.OnMenuBtnClick(mENUBUTTONDATA.idCommand, mENUBUTTONDATA.x, mENUBUTTONDATA.y);

	public static object FindImage(string name)
	{
		if (name != null && name.Length > 0)
		{
			try
			{
				object obj = null;
				Assembly executingAssembly = Assembly.GetExecutingAssembly();
				if ((obj = LoadImage(executingAssembly, name)) != null)
				{
					return obj;
				}
				Assembly callingAssembly = Assembly.GetCallingAssembly();
				if ((obj = LoadImage(callingAssembly, name)) != null)
				{
					return obj;
				}
				Assembly[] assemblies = AppDomain.CurrentDomain.GetAssemblies();
				foreach (Assembly a in assemblies)
				{
					if ((obj = LoadImage(a, name)) != null)
					{
						return obj;
					}
				}
			}
			catch (Exception inner)
			{
				throw new SnapinException("Image " + name + " not found in app domain assemblies.", inner);
			}
		}
		return null;
	}

	protected static object LoadImage(Assembly a, string name)
	{
		string[] manifestResourceNames = a.GetManifestResourceNames();
		Stream manifestResourceStream = a.GetManifestResourceStream(name);
		if (manifestResourceStream != null)
		{
			if (Path.GetExtension(name).ToLower() == ".ico")
			{
				return new Icon(manifestResourceStream);
			}
			if (_TransparentColor == Color.Transparent)
			{
				return new Bitmap(manifestResourceStream);
			}
			Bitmap bitmap = new Bitmap(manifestResourceStream);
			bitmap.MakeTransparent(_TransparentColor);
			return bitmap;
		}
		return null;
	}

	public virtual IntPtr GetBitmapHandle(int index)
	{
		ImageEntry value = this[index];
		if (value.image != null && value.bitmapHandle == IntPtr.Zero)
		{
			Bitmap bm = ((!(value.image is Icon)) ? ((Bitmap)value.image) : ((Icon)value.image).ToBitmap());
			value.bitmapHandle = GetCompatibleBitmapHandle(bm);
			this[index] = value;
		}
		return value.bitmapHandle;
	}

	public virtual IntPtr GetIconHandle(int index)
	{
		ImageEntry value = this[index];
		if (value.image != null && value.iconHandle == IntPtr.Zero)
		{
			if (value.image is Icon)
			{
				value.iconHandle = ((Icon)value.image).Handle;
			}
			else
			{
				value.iconHandle = ((Bitmap)value.image).GetHicon();
			}
			this[index] = value;
		}
		return value.iconHandle;
	}

	internal IntPtr GetCompatibleBitmapHandle(Bitmap bm)
	{
		IntPtr hbitmap;
		IntPtr intPtr = (hbitmap = new Bitmap(bm).GetHbitmap());
		IntPtr hdc = CreateCompatibleDC(IntPtr.Zero);
		DIBSECTION lpvObject = default(DIBSECTION);
		int num = GetObject(hbitmap, Marshal.SizeOf(typeof(DIBSECTION)), ref lpvObject);
		BITMAPINFO lpbmi = new BITMAPINFO
		{
			bmiHeader = lpvObject.dsBmih
		};
		BITMAPINFO pbmi = default(BITMAPINFO);
		pbmi.bmiHeader = lpvObject.dsBmih;
		pbmi.bmiHeader.biBitCount = (ushort)GetDeviceCaps(hdc, 12);
		if (pbmi.bmiHeader.biBitCount != lpbmi.bmiHeader.biBitCount)
		{
			intPtr = CreateDIBSection(hdc, ref pbmi, 1u, out var _, IntPtr.Zero, 0u);
			SetDIBits(hdc, intPtr, 0u, lpvObject.dsBmih.biHeight, lpvObject.dsBm.bmBits, ref lpbmi, 1u);
			DeleteObject(hbitmap);
		}
		DeleteDC(hdc);
		return intPtr;
	}

			return;
		}
		int num = 1;
		foreach (DataRow row in m_table.Rows)
		{
			if (CheckItem(row))
			{
				if (row.RowState == DataRowState.Unchanged && refreshAll)
				{
					InsertItem(row, num);
				}
				else if (row.RowState == DataRowState.Added)
				{
					InsertItem(row, num);
				}
				else if (row.RowState == DataRowState.Modified)
				{
					SetItem(row, num);
				}
			}
			num++;
		}
		Table.AcceptChanges();
	}

	public override void OnFilterChange(MMC_FILTER_CHANGE_CODE code, int col)
	{
		switch (code)
		{
		case MMC_FILTER_CHANGE_CODE.MFCC_DISABLE:
			m_filters = null;
			break;
		case MMC_FILTER_CHANGE_CODE.MFCC_VALUE_CHANGE:
		{
			IConsole2 resultViewConsole = base.Snapin.ResultViewConsole;
			IHeaderCtrl2 headerCtrl = resultViewConsole as IHeaderCtrl2;
			uint pdwType = 0u;
			MMC_FILTERDATA data = new MMC_FILTERDATA
			{
				pszText = IntPtr.Zero
			};
			try
			{
				int num = 1000;
				data.pszText = Marshal.AllocCoTaskMem(num);
				data.cchTextMax = num;
				int columnFilter = headerCtrl.GetColumnFilter((uint)col, ref pdwType, ref data);
				string text = Marshal.PtrToStringAuto(data.pszText);
				if (m_filters == null)
				{
					m_filters = new Hashtable();
				}
				if (text.Length == 0 || text.Length >= num / 2)
				{
					m_filters.Remove(col);
				}
				else
				{
					m_filters[col] = text;
				}
				if (m_filters.Count == 0)
				{
					m_filters = null;
				}
				RemoveAllItems();
			}
			finally
			{
				if (!data.pszText.Equals(IntPtr.Zero))
				{
					Marshal.FreeCoTaskMem(data.pszText);
				}
			}
			break;
		}
		}
		base.OnFilterChange(code, col);
	}

	public virtual void RemoveAllItems()
	{
		try
		{
			IConsole2 resultViewConsole = base.Snapin.ResultViewConsole;
			if (resultViewConsole != null && resultViewConsole is IResultData resultData)
			{
				resultData.DeleteAllRsltItems();
				OnShowData(resultData, refreshAll: true);
			}
		}
		catch (Exception)
		{
		}
	}

	public virtual bool CheckItem(DataRow row)
	{
		if (m_filters == null)
		{
			return true;
		}
		object[] itemArray = row.ItemArray;
		for (int i = 0; i < itemArray.Length; i++)
		{
			if (m_filters[i] is string value && itemArray[i] != null && itemArray[i].ToString().IndexOf(value) == -1)
			{
				return false;
			}
		}
		return true;
	}

internal class SnapReg
{
	private ArrayList fSnapinRegInfos = new ArrayList();

	private Hashtable fSnapinAboutRegistrations = new Hashtable();

	public void UnRegister(Assembly asm)
	{
		Type[] types = asm.GetTypes();
		foreach (Type type in types)
		{
			if (Marshal.IsTypeVisibleFromCom(type) && type.IsClass && (type.IsPublic || type.IsNestedPublic) && !type.IsImport)
			{
				FindAttributes(type);
			}
		}
		PrepareAttribute();
		UnRegister();
	}

	public void Register(Assembly asm)
	{
		Type[] types = asm.GetTypes();
		foreach (Type type in types)
		{
			if (Marshal.IsTypeVisibleFromCom(type) && type.IsClass && (type.IsPublic || type.IsNestedPublic) && !type.IsImport)
			{
				FindAttributes(type);
			}
		}
		PrepareAttribute();
		Register();
	}

	private void FindAttributes(Type t)
	{
		string text = "{" + Marshal.GenerateGuidForType(t).ToString().ToUpper() + "}";
		object[] customAttributes = t.GetCustomAttributes(typeof(SnapinInAttribute), inherit: true);
		if (customAttributes.Length != 0 && customAttributes[0] is SnapinInAttribute fAttrib)
		{
			SnapinRegInfo snapinRegInfo = new SnapinRegInfo();
			snapinRegInfo.fAttrib = fAttrib;
			snapinRegInfo.fGUID = text.ToUpper();
			snapinRegInfo.fSnapinType = t;
			fSnapinRegInfos.Add(snapinRegInfo);
		}
		customAttributes = t.GetCustomAttributes(typeof(AboutSnapinAttribute), inherit: true);
		if (customAttributes.Length != 0 && customAttributes[0] is AboutSnapinAttribute aboutSnapinAttribute)
		{
			SnapinAboutRegistration snapinAboutRegistration = new SnapinAboutRegistration();
			snapinAboutRegistration.fAboutAttrib = aboutSnapinAttribute;
			snapinAboutRegistration.fGUID = text.ToUpper();
			fSnapinAboutRegistrations.Add(aboutSnapinAttribute.SnapinType.ToString(), snapinAboutRegistration);
		}
	}

	private void PrepareAttribute()
	{
		foreach (SnapinRegInfo fSnapinRegInfo in fSnapinRegInfos)
		{
			if (!(fSnapinAboutRegistrations[fSnapinRegInfo.fSnapinType.ToString()] is SnapinAboutRegistration fAboutReg))
			{
				throw new ApplicationException("Failed to locate a Snapin About object for the snapin: " + fSnapinRegInfo.fSnapinType.ToString());
			}
			fSnapinRegInfo.fAboutReg = fAboutReg;
		}
	}

	private void Register()
	{
		foreach (SnapinRegInfo fSnapinRegInfo in fSnapinRegInfos)
		{
			RegistryKey registryKey = Registry.LocalMachine.OpenSubKey("Software\\Microsoft\\MMC\\Snapins", writable: true);
			registryKey = registryKey.CreateSubKey(fSnapinRegInfo.fGUID);
			registryKey.SetValue("NameString", fSnapinRegInfo.fAttrib.Namestring);
			registryKey.SetValue("About", fSnapinRegInfo.fAboutReg.fGUID);
			if (fSnapinRegInfo.fAttrib.StandAlone)
			{
				registryKey.CreateSubKey("StandAlone");
			}
		}
	}

	private void UnRegister()
	{
		foreach (SnapinRegInfo fSnapinRegInfo in fSnapinRegInfos)
		{
			try
			{
				RegistryKey registryKey = Registry.LocalMachine.OpenSubKey("Software\\Microsoft\\MMC\\Snapins", writable: true);
				registryKey.DeleteSubKeyTree(fSnapinRegInfo.fGUID);
			}
			catch (ArgumentException)
			{
			}
			catch (SecurityException)
			{
			}
		}
	}
}

	public virtual void OnShowHeader(IHeaderCtrl2 header)
	{
		try
		{
			if (m_HeaderInfo != null)
			{
				for (int i = 0; i < m_HeaderInfo.Count; i++)
				{
					string title = m_HeaderInfo[i] as string;
					int nWidth = -1;
					header.InsertColumn(i, title, 0, nWidth);
				}
			}
		}
		catch (COMException ex)
		{
			throw ex;
		}
		catch (Exception ex2)
		{
			throw ex2;
		}
	}

	public virtual void OnShowData(IResultData ResultData)
	{
		if (m_List != null)
		{
			for (int i = 0; i < m_List.Count; i++)
			{
				InsertItem(m_List.IndexOfValue(i), i + 1);
			}
		}
	}

	public virtual void InsertItem(object row, int nRow)
	{
		if (row != null)
		{
			RESULTDATAITEM item = default(RESULTDATAITEM);
			row.ToString();
			item.mask = 22u;
			item.nImage = -1;
			item.str = (IntPtr)(-1);
			item.nCol = 0;
			item.lParam = GetItemLParam(nRow);
			ResultData.InsertItem(ref item);
		}
	}

	public virtual void SetItem(object row, int nRow)
	{
		if (row != null)
		{
			int itemLParam = GetItemLParam(nRow);
			RESULTDATAITEM item = default(RESULTDATAITEM);
			row.ToString();
			item.mask = 18u;
			item.str = (IntPtr)(-1);
			item.nCol = 0;
			ResultData.FindItemByLParam(itemLParam, out var pItemID);
			item.lParam = itemLParam;
			item.itemID = (int)pItemID;
			ResultData.SetItem(ref item);
		}
	}

	public virtual void UpdateItem(int lParam)
	{
		ResultData.FindItemByLParam(lParam, out var pItemID);
		ResultData.UpdateItem(pItemID);
	}

	public virtual void DeleteItem(int lParam)
	{
		ResultData.FindItemByLParam(lParam, out var pItemID);
		ResultData.DeleteItem(pItemID, 0);
	}

	public virtual void GetItem(out RESULTDATAITEM ResultDataItem, uint pItemID)
	{
		RESULTDATAITEM item = new RESULTDATAITEM
		{
			mask = 2u,
			itemID = (int)pItemID,
			nCol = 0
		};
		ResultData.GetItem(ref item);
		ResultDataItem = item;
	}

	public virtual RESULTDATAITEM GetSelectedItem()
	{
		return GetSelectedItem(-1);
	}

	public virtual RESULTDATAITEM GetSelectedItem(int nIndex)
	{
		RESULTDATAITEM item = new RESULTDATAITEM
		{
			mask = 8u,
			nCol = 0,
			nIndex = nIndex,
			nState = 2u
		};
		ResultData.GetNextItem(ref item);
		return item;
	}

	public virtual int GetTableRowID(int lParam)
	{
		return (lParam >> 16) - 1;
	}

	public virtual int GetItemLParam(int nRow)
	{
		return m_iCookie | (nRow << 16);
	}

	{
		if (m_ToolbarTable.ContainsKey(toolBarId))
		{
			return true;
		}
		try
		{
			ToolBarInfo value = new ToolBarInfo(toolBarId, buttons, hBitmaps, numBitMapsFromEachBmpFile);
			m_ToolbarTable.Add(toolBarId, value);
			return true;
		}
		catch (ArgumentException)
		{
			return false;
		}
		catch
		{
			return false;
		}
	}

	{
		return m_ToolbarTable.Count;
	}

	{
		m_ToolbarTable.Clear();
	}

	public int GetCurrentToolbarCount()
	{
		return m_CurrentToolbarsInfo.Count;
	}

	public int GetCurrentToolbarItemAtIndex(int index)
	{
		if (index < 0 || index >= m_CurrentToolbarsInfo.Count)
		{
			return -1;
		}
		return (int)m_CurrentToolbarsInfo[index];
	}

	public void ClearCurrentToolbarsInfo()
	{
		m_CurrentToolbarsInfo.Clear();
	}

	public bool AddCurrentToolbarsInfo(int toolBarId)
	{
		if (m_ToolbarTable.ContainsKey(toolBarId))
		{
			if (!m_CurrentToolbarsInfo.Contains(toolBarId))
			{
				m_CurrentToolbarsInfo.Add(toolBarId);
			}
			return true;
		}
		return false;
	}

	{
		try
		{
			ToolBarInfo toolBarInfo = (ToolBarInfo)m_ToolbarTable[toolBarId];
			for (int i = 0; i < toolBarInfo.m_hBitmaps.Length; i++)
			{
				pToolBar.AddBitmap(toolBarInfo.m_numBitmapsFromEachBmpHandle[i], toolBarInfo.m_hBitmaps[i], 16, 16, Color.Black.ToArgb());
			}
			MMCBUTTON[] array = new MMCBUTTON[toolBarInfo.m_NumButtons];
			for (int j = 0; j < toolBarInfo.m_NumButtons; j++)
			{
				array[j].idCommand = toolBarInfo.m_toolBarButtonInfo[j].idCommand;
				array[j].nBitmap = toolBarInfo.m_toolBarButtonInfo[j].bmpIndex;
				array[j].lpTooltipText = toolBarInfo.m_toolBarButtonInfo[j].strTooltip;
				array[j].fsState = toolBarInfo.m_toolBarButtonInfo[j].tbState;
				array[j].fsType = toolBarInfo.m_toolBarButtonInfo[j].tbStyle;
				array[j].lpButtonText = toolBarInfo.m_toolBarButtonInfo[j].strText;
				pToolBar.AddButtons(1, ref array[j]);
			}
			return true;
		}
		catch (Exception ex)
		{
			string message = ex.Message;
			return false;
		}
	}

internal class WizardThread
{
	protected WizardBase m_Form;

	protected DialogResult formResult;

	protected WizardPage[] m_Pages;

	protected WizardBase.WizardAction m_Action;

	protected string m_Title;

	protected Bitmap m_Logo;

	public DialogResult CachedDialogResult => m_Form.CachedDialogResult;

	public WizardThread(WizardPage[] pages, WizardBase.WizardAction act, string title, Bitmap logo)
	{
		m_Pages = pages;
		m_Action = act;
		m_Title = title;
		m_Logo = logo;
	}

	public void Run()
	{
		try
		{
			m_Form = new WizardBase(m_Title);
			for (int i = 0; i < m_Pages.Length; i++)
			{
				m_Form.AddWizardPage(m_Pages[i]);
			}
			m_Form.Logo = m_Logo;
			m_Form.InitPages();
			formResult = m_Form.ShowDialog();
			try
			{
				if (m_Form.CachedDialogResult == DialogResult.OK)
				{
					m_Action(m_Pages);
				}
			}
			finally
			{
				m_Form.Dispose();
			}
		}
		catch (Exception e)
		{
			WizardBase.ShowException("Caught an exception", e);
		}
	}
}
