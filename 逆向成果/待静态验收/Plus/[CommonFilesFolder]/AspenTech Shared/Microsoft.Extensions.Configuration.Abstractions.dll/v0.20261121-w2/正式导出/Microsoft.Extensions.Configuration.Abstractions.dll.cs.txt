		public static string? GetSectionKey(string? path)
		{
			if (string.IsNullOrEmpty(path))
			{
				return path;
			}
			int num = path.LastIndexOf(':');
			if (num >= 0)
			{
				return path.Substring(num + 1);
			}
			return path;
		}

		public static string? GetParentPath(string? path)
		{

			{
				return null;
			}
			int num = path.LastIndexOf(':');
			if (num >= 0)
			{
				return path.Substring(0, num);
			}
			return null;
