	public class LanguageList : ObservableCollection<LanguageItem>
	{
		private string regentry = "HKEY_CURRENT_USER\\Software\\AspenTech\\Engineering\\Language";

		private string _currentString;

		public LanguageItem CurrentSelection { get; set; }

		public string Description => Resources.strDescription;

		public string CurrentString
		{
			get
			{
				return _currentString;
			}
			set
			{
				_currentString = value;
				OnPropertyChanged(new PropertyChangedEventArgs("CurrentString"));
			}
		}

		public LanguageList()
		{
			Add(new LanguageItem(Resources.strEnglish, "eng"));
			Add(new LanguageItem(Resources.strJapanese, "jpn"));
			Add(new LanguageItem(Resources.strChinese, "chs"));
			Add(new LanguageItem(Resources.strRussian, "rus"));
			string text = Registry.GetValue(regentry, null, "eng") as string;
			if (text == null)
			{
				text = "eng";
			}
			string text2 = null;
			using (IEnumerator<LanguageItem> enumerator = GetEnumerator())
			{
				while (enumerator.MoveNext())
				{
					LanguageItem current = enumerator.Current;
					if (string.Compare(current.Code, text, StringComparison.InvariantCultureIgnoreCase) == 0)
					{
						text2 = current.Name;
						CurrentSelection = current;
						break;
					}
				}
			}
			if (text2 == null)
			{
				text2 = "English";
				Registry.SetValue(regentry, null, "eng");
			}
			CurrentString = Resources.strCurrent + text2;
		}

		public void SetRegistryToCurrent()
		{
			if (CurrentSelection != null)
			{
				Registry.SetValue(regentry, null, CurrentSelection.Code);
				CurrentString = Resources.strCurrent + CurrentSelection.Name;
			}
		}
	}
