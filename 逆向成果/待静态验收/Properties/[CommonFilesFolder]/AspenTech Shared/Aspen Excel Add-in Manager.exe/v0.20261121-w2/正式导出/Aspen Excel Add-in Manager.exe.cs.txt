			RegHive regHive = RegHive.Unspecified;
			regHive = (IsLM.Value ? RegHive.LocalMachine : RegHive.CurrentUser);

					ErrorDesc = Resources.ResourceManager.GetString("ASW_RegistryKeyAccessFailed");
					string messageBoxText = Resources.ResourceManager.GetString("ASW_DueToInsufficientAccessPrivileges") + "\r\n\r\n" + FriendlyName + "\r\n\r\n" + Resources.ResourceManager.GetString("ASW_TryRunningThisProgramAsAdministrator");

				}
				else
				{
					ErrorDesc = "Could not retrieve registry key";
				}
				return false;
			}
			try
			{
				regKey.SetValue("LoadBehavior", LoadBehaviorRegCode);
			}
			catch (Exception ex)

				ErrorDesc = "Error setting LoadBehavior - " + ex.Message;
				return false;

		protected RegistryKey GetRegKey(RegHive hiveToWriteTo, string progIdName, bool createIfNeeded, bool writeAccess, ref bool failDueToAccessRestrictrion)
		{
			RegistryKey registryKey = null;
			if (!IsLM.HasValue && hiveToWriteTo == RegHive.Unspecified)
			{
				return null;
			}
			StringBuilder stringBuilder = new StringBuilder();
			stringBuilder.Append("Software\\");
			stringBuilder.Append("Microsoft\\Office\\Excel\\Addins\\");
			stringBuilder.Append(progIdName);
			if (hiveToWriteTo == RegHive.Unspecified)
			{
				hiveToWriteTo = (IsLM.Value ? RegHive.LocalMachine : RegHive.CurrentUser);
			}
			RegistryHive hKey;
			switch (hiveToWriteTo)
			{
			case RegHive.Unspecified:
				return null;
			case RegHive.LocalMachine:
				hKey = RegistryHive.LocalMachine;
				break;
			case RegHive.CurrentUser:
				hKey = RegistryHive.CurrentUser;
				break;
			default:
				return null;
			}
			RegistryView registryView = ASWRegUtil.GetRegistryView(IsWow6432Node);
			RegistryKey registryKey2 = RegistryKey.OpenBaseKey(hKey, registryView);
			try
			{
				registryKey = registryKey2.OpenSubKey(stringBuilder.ToString(), writeAccess);
			}
			catch (SecurityException)
			{
				failDueToAccessRestrictrion = true;
			}
			catch (Exception)
			{
			}
			if (registryKey == null && createIfNeeded)
			{
				try
				{
					registryKey = registryKey2.CreateSubKey(stringBuilder.ToString());
				}
				catch
				{
				}
			}
			if (registryKey == null)
			{
				failDueToAccessRestrictrion = true;
				return null;
			}
			return registryKey;
		}

					LstASWAddins.Add(aspenAddinInfo);
					aspenAddinInfo.LoadAddin = regEntryIsAddinActive;
				}
			}
			RegistryKey addinRegKeyCU = _addinRegKeyCU;
			List<string> list = new List<string>();
			if (addinRegKeyCU == null)
			{
				ErrorDesc = "Could not retrieve Current User add-ins registry key";
				return false;
			}
			subKeyNames = addinRegKeyCU.GetSubKeyNames();
			foreach (string text3 in subKeyNames)
			{
				if (string.IsNullOrWhiteSpace(text3))
				{
					continue;
				}
				RegistryKey registryKey2 = addinRegKeyCU.OpenSubKey(text3);
				if (registryKey2 == null)
				{
					continue;
				}
				string text4 = text3;
				if (_dicAspenProgIdLmToVerInfo.ContainsKey(text4))
				{
					continue;
				}
				string fileLocation2 = GetFileLocation(text4);
				string text5 = ReadAddinKeyValDescription(registryKey2);
				string text6 = ReadAddinKeyValFriendlyName(registryKey2);
				if (string.IsNullOrWhiteSpace(text6))
				{
					text6 = text4;
				}
				if (string.IsNullOrWhiteSpace(text5))
				{
					text5 = text4;
				}
				int num2 = ReadAddinKeyValLoadBehaviorInt(registryKey2);
				if (num2 < 0)
				{
					continue;
				}
				AddinLoadBehavior regEntryLoadBehavior2 = GetRegEntryLoadBehavior(num2);
				bool regEntryIsAddinActive2 = GetRegEntryIsAddinActive(num2);
				if (regEntryLoadBehavior2 == AddinLoadBehavior.NotSetYet)
				{
					continue;
				}
				if (!_dicAspenProgIdToVerInfo.ContainsKey(text4))
				{
					AddinInfo addinInfo2 = new AddinInfo(IsWow6432Node, text4, text6, text5, fileLocation2, regEntryLoadBehavior2, regEntryIsAddinActive2);
					addinInfo2.IsLM = false;
					LstNonASWAddins.Add(addinInfo2);
					continue;
				}
				string strB = string.Empty;
				foreach (AspenAddinInfo lstASWAddin in LstASWAddins)
				{
					if (string.CompareOrdinal(lstASWAddin.ProgIDCU, text4) == 0)
					{
						strB = lstASWAddin.ProductName;
						break;
					}
				}
				bool flag = false;
				foreach (AspenAddinInfo lstASWAddin2 in LstASWAddins)
				{
					if (string.CompareOrdinal(lstASWAddin2.ProductName, strB) == 0)
					{
						if (string.CompareOrdinal(lstASWAddin2.ProgIDCU, text4) == 0)
						{
							flag = true;
							lstASWAddin2.LoadBehavior = regEntryLoadBehavior2;
							lstASWAddin2.ResetLoadAddinValue(regEntryIsAddinActive2);
							lstASWAddin2.FileLocation = fileLocation2;
						}
						else if (regEntryIsAddinActive2 && lstASWAddin2.LoadAddin && !list.Contains(lstASWAddin2.FriendlyName))
						{
							list.Add(lstASWAddin2.FriendlyName);
						}
					}
				}
				if (!flag)
				{
					AspenAddInVersionInfo aspenAddInVersionInfo2 = _dicAspenProgIdToVerInfo[text4];
					AspenAddinInfo aspenAddinInfo2 = new AspenAddinInfo(IsWow6432Node, aspenAddInVersionInfo2.ProgID, aspenAddInVersionInfo2.ProgIDLM, text6, text5, fileLocation2, regEntryLoadBehavior2, loadAddin: false, aspenAddInVersionInfo2.ProductName);

					aspenAddinInfo2.FriendlyNameLM = aspenAddInVersionInfo2.FriendlyNameLM;
					LstASWAddins.Add(aspenAddinInfo2);
					aspenAddinInfo2.ParentLst = LstASWAddins;
					aspenAddinInfo2.LoadAddin = regEntryIsAddinActive2;
				}
			}
			if (list.Count > 0)
			{
				StringBuilder stringBuilder = new StringBuilder();
				stringBuilder.Append("Multiple versions of the same add-in cannot be loaded by Excel at the same time.  To avoid a conflict, the 'Load Behavior' registry settings for following add-ins were changed:");
				stringBuilder.Append("\r\n\r\n");
				foreach (string item in list)
				{
					stringBuilder.AppendLine(item);
				}
				string appName = App.AppName;
				MessageBox.Show(stringBuilder.ToString(), appName, MessageBoxButton.OK, MessageBoxImage.Exclamation);
			}
			foreach (AspenAddinInfo lstASWAddin3 in LstASWAddins)
			{
				lstASWAddin3.Write(forceWrite: true);
			}
			return true;
		}

		private AddinLoadBehavior GetRegEntryLoadBehavior(int regVal)
		{
			Math.DivRem(regVal, 2, out var result);
			if (result != 0)
			{
				regVal--;
			}
			return (AddinLoadBehavior)regVal;
		}

		private bool GetRegEntryIsAddinActive(int regVal)
		{
			Math.DivRem(regVal, 2, out var result);
			return result != 0;
		}

					}
					if (!new List<string>(key.GetSubKeyNames()).Contains(subKeyName))
					{
						return string.Empty;
					}
					registryKey = key.OpenSubKey(subKeyName);
					if (registryKey == null)
					{
						return string.Empty;
					}
				}
				else
				{
					registryKey = key;
				}
				object value = registryKey.GetValue(valName);
				if (value == null)
				{
					return string.Empty;
				}
				return value.ToString();
			}
			catch
			{
				return string.Empty;
			}
		}
	}

		{
			if (LstAddinInfoASW != null)
			{
				foreach (AspenAddinInfo item in LstAddinInfoASW)
				{
					if (item.IsDirty)
					{
						return true;
					}
				}
			}
			if (LstAddinInfoNonASW != null)
			{
				foreach (AddinInfo item2 in LstAddinInfoNonASW)
				{
					if (item2.IsDirty)
					{
						return true;
					}
				}
			}
			return false;
		}

		private void RequestWriteAndCloseNow()
		{
			foreach (AddinInfo item in LstAddinInfoNonASW)
			{

			}
			foreach (AspenAddinInfo item2 in LstAddinInfoASW)
			{
				item2.Write();
			}
			if (this.OnCloseRequest != null)
			{
				this.OnCloseRequest(this, new EventArgs());
			}
		}
