		{
			get
			{
				if (HostFile == null)
				{
					return FullName;
				}
				string text = FullName;
				if (!text.StartsWith(FolderSeparator))
				{
					text = FolderSeparator + text;
				}
				return HostFile.HostedFullName + text;
			}

			{

					throw PrepareException(except);
				}
			}
			set
			{
				if (value == null)
				{
					throw new ArgumentNullException();
				}
				value = value.TrimEnd();
				if (value.Length == 0)
				{
					throw new ArgumentException("The new name is empty or contains only whitespace.");
				}
				if (value.IndexOfAny(new char[2]
				{
					Path.DirectorySeparatorChar,
					Path.AltDirectorySeparatorChar
				}) != -1)
				{
					throw new ArgumentException("The new name contains a path.");
				}
				if (value.IndexOfAny(InvalidChars) != -1)
				{
					throw new ArgumentException("The new name contains invalid characters.");
				}
				if (this is AbstractFolder && ((AbstractFolder)this).IsRoot)
				{
					throw new ItemIsRootFolderException(this);
				}
				try
				{
					using (new AutoBatchUpdate(BatchUpdateable))
					{
						DoName = value;
					}
				}
				catch (Exception except)
				{
					throw PrepareException(except);
				}
			}
		}

		public void Refresh(FileSystemEvents events, object userData)
		{
			if (events == null)
			{
				throw new ArgumentNullException("events");
			}
			try
			{
				FileSystemEventsSession fileSystemEventsSession = events.CreateSession(this, userData);
				fileSystemEventsSession.OnNewItemToProcess(this, null);
				ItemPair nextItemPair;
				do
				{
					nextItemPair = fileSystemEventsSession.GetNextItemPair(raiseBytesProgression: false);
					nextItemPair?.current.DoRefresh(fileSystemEventsSession);
				}
				while (nextItemPair != null);
			}
			catch (Exception except)
			{
				throw PrepareException(except);
			}
		}

				BuildList(session, destination, forward: true, recursive: true, includeFolders: true, new Filter[0]);
				using (new AutoBatchUpdate(destinationFolder.BatchUpdateable, events, userData))

		protected virtual FileSystemItem ListCopyTo(FileSystemEventsSession session, bool replaceExistingFiles)
		{
			ItemPair nextItemPair = session.GetNextItemPair(raiseBytesProgression: true);
			ItemPair itemPair = nextItemPair;
			do
			{
				if (itemPair != null)
				{
					ListCopyToCore(session, itemPair, replaceExistingFiles);
					itemPair = session.GetNextItemPair(raiseBytesProgression: true);
				}
			}
			while (itemPair != null);
			if (nextItemPair != null)
			{
				return nextItemPair.target;
			}
			throw new FileSystemInternalException("The firstItemPair is null in ListCopyTo");
		}

		protected virtual void ListCopyToCore(FileSystemEventsSession session, ItemPair currentItemPair, bool replaceExistingFiles)
		{
			bool flag = false;
			do
			{
				try
				{
					currentItemPair.current.DoCopyTo(session, currentItemPair.target, replaceExistingFiles);
					flag = false;
				}
				catch (Exception except)
				{
					session.OnUndoBytesProcessed();
					flag = session.OnException(except);
				}
			}
			while (flag);
			if (currentItemPair.current is AbstractFile)
			{
				session.OnBytesCompleted();
			}
		}

		protected virtual FileSystemItem ListMoveTo(FileSystemEventsSession session, bool replaceExistingFiles)
		{
			ItemPair itemPair = null;
			ItemPair nextItemPair;
			do
			{
				nextItemPair = session.GetNextItemPair(raiseBytesProgression: true);
				if (nextItemPair != null)
				{
					itemPair = nextItemPair;
					ListMoveToCore(session, nextItemPair, replaceExistingFiles);
				}
			}
			while (nextItemPair != null);
			if (itemPair != null)
			{
				return itemPair.target;
			}
			throw new FileSystemInternalException("The lastItemPair is null in ListMoveTo");
		}

		protected virtual void ListMoveToCore(FileSystemEventsSession session, ItemPair currentItemPair, bool replaceExistingFiles)
		{
			if (currentItemPair == null)
			{
				return;
			}
			bool flag = false;
			bool flag2 = false;
			do
			{
				try
				{
					currentItemPair.current.DoCopyTo(session, currentItemPair.target, replaceExistingFiles);
					flag2 = false;
					flag = true;
				}
				catch (Exception except)
				{
					session.OnUndoBytesProcessed();
					flag2 = session.OnException(except);
				}
			}
			while (flag2);
			if (flag)
			{
				do
				{
					try
					{
						bool flag3 = true;
						if (currentItemPair.current is AbstractFolder)
						{
							AbstractFolder abstractFolder = (AbstractFolder)currentItemPair.current;
							if (abstractFolder.GetItems(recursive: false).Length != 0)
							{
								flag3 = false;
							}
						}
						if (flag3)
						{
							currentItemPair.current.DoDelete(session);
						}
						flag2 = false;
					}
					catch (ItemDoesNotExistException)
					{
					}
					catch (Exception except2)
					{
						flag2 = session.OnException(except2);
					}
				}
				while (flag2);
			}
			if (currentItemPair.current is AbstractFile)
			{
				session.OnBytesCompleted();
			}
		}

		internal void ApplyPropertiesFrom(FileSystemEvents events, object userData, FileSystemItem sourceItem)
		{
			if (events == null)
			{
				throw new ArgumentNullException("events");
			}
			if (sourceItem == null)
			{
				throw new ArgumentNullException("sourceItem");
			}
			if (this is AbstractFolder { IsRoot: not false })
			{
				throw new ItemIsRootFolderException(this);
			}
			if (sourceItem is AbstractFolder { IsRoot: not false })
			{
				throw new ItemIsRootFolderException(sourceItem);
			}
			if ((Attributes & FileAttributes.ReadOnly) == FileAttributes.ReadOnly)
			{
				throw new ItemIsReadOnlyException(this);
			}
			try
			{
				FileSystemEventsSession fileSystemEventsSession = events.CreateSession(sourceItem, userData);
				using (new AutoBatchUpdate(BatchUpdateable, events, userData))
				{
					fileSystemEventsSession.OnNewItemToProcess(sourceItem, this);
					ItemPair nextItemPair;
					do
					{
						nextItemPair = fileSystemEventsSession.GetNextItemPair(raiseBytesProgression: false);
						nextItemPair?.target.DoApplyPropertiesFrom(fileSystemEventsSession, nextItemPair.current);

		protected override void DoCopyTo(FileSystemEventsSession session, FileSystemItem destination, bool replaceExistingFiles)
		{
			if (!base.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			if (IsSameAs(destination))
			{
				throw new FileSystemNotSupportedException("Cannot copy a file onto itself.", this);
			}
			if (destination.Exists)
			{
				if (!replaceExistingFiles)
				{
					throw new ItemAlreadyExistsException(destination);
				}
				if ((destination.Attributes & FileAttributes.ReadOnly) == FileAttributes.ReadOnly)
				{
					throw new ItemIsReadOnlyException(destination);
				}
			}
			AbstractFile abstractFile = (AbstractFile)destination;
			session.Events.BeginSubOperation();
			try
			{
				abstractFile.ApplyPropertiesFrom(session.Events, session.UserData, this);
			}
			finally
			{
				session.Events.EndSubOperation();
			}
			bool flag = false;
			try
			{
				using Stream stream = OpenRead(session.Events, session.UserData);
				Stream stream2 = null;
				session.Events.BeginSubOperation();
				try
				{
					if (abstractFile.Exists)
					{
						stream2 = abstractFile.OpenWrite(session.Events, session.UserData, overwrite: true);
					}
					else
					{
						stream2 = abstractFile.CreateWrite(session.Events, session.UserData);
						flag = true;
					}
				}
				finally
				{
					session.Events.EndSubOperation();
				}
				try
				{
					byte[] array = new byte[8192];
					int num = 0;
					int num2;
					while ((num2 = stream.Read(array, 0, array.Length)) > 0)
					{
						if (num > 0)
						{
							session.OnBytesProcessed(num);
						}
						num = num2;
						stream2.Write(array, 0, num2);
					}
				}
				catch (IOException inner)
				{
					throw new FileSystemIOException("An I/O exception occurred while copying the data from the item to its destination.", this, inner);
				}
				finally
				{
					stream2.Close();
				}
			}
			catch
			{
				if (flag)
				{
					try
					{
						abstractFile.Delete();
					}
					catch
					{
					}
				}
				throw;
			}
			session.Events.BeginSubOperation();
			try
			{
				ApplyPropertiesTo(session.Events, session.UserData, abstractFile);
			}
			finally
			{
				session.Events.EndSubOperation();
			}
		}

			}
			catch (Exception inner)
			{
				throw new FileSystemInternalException("Generated temporary filename was not unique.", file, inner);
			}
		}

		private AbstractFile GetFileFromRelativeName(FileSystemEventsSession session, string relativeName)
		{
			relativeName = relativeName.Replace(Path.AltDirectorySeparatorChar, Path.DirectorySeparatorChar);
			if (relativeName.StartsWith(Path.DirectorySeparatorChar.ToString()))
			{
				return base.RootFolder.GetFileFromRelativeName(session, relativeName.Remove(0, Path.DirectorySeparatorChar.ToString().Length));
			}
			checked
			{
				if (relativeName.StartsWith("." + Path.DirectorySeparatorChar))
				{
					return GetFileFromRelativeName(session, relativeName.Remove(0, ".".Length + Path.DirectorySeparatorChar.ToString().Length));
				}
				if (relativeName.StartsWith(".." + Path.DirectorySeparatorChar))
				{
					if (base.ParentFolder == null)
					{
						throw new FileSystemException("GetFile was called with a relative name refering to the folder's parent (\"..\") on a folder that has no parent.", this);
					}
					return base.ParentFolder.GetFileFromRelativeName(session, relativeName.Remove(0, "..".Length + Path.DirectorySeparatorChar.ToString().Length));
				}
				string empty = string.Empty;
				try
				{
					empty = StringCheck.ReplaceNull(Path.GetDirectoryName(relativeName));
				}
				catch (PathTooLongException inner)
				{
					throw new FileSystemIOException("The folder's full name is too long for relative name : " + relativeName, null, inner);
				}
				if (empty.Length != 0)
				{
					empty = (empty.EndsWith(Path.DirectorySeparatorChar.ToString()) ? empty : (empty + Path.DirectorySeparatorChar));
					string folderName = empty.Substring(0, empty.IndexOf(Path.DirectorySeparatorChar));
					string text = relativeName.Remove(0, relativeName.IndexOf(Path.DirectorySeparatorChar) + 1);
					if (!text.StartsWith(".." + Path.DirectorySeparatorChar) && !text.StartsWith(Path.DirectorySeparatorChar.ToString()))
					{
						session.OnScanningFolder(this);
					}
					AbstractFolder abstractFolder = DoGetFolder(session, folderName);
					return abstractFolder.GetFileFromRelativeName(session, text);
				}
				session.OnScanningFolder(this);
				return DoGetFile(session, relativeName);
			}
		}

		private AbstractFolder GetFolderFromRelativeName(FileSystemEventsSession session, string relativeName)
		{
			relativeName = relativeName.Replace(Path.AltDirectorySeparatorChar, Path.DirectorySeparatorChar);
			if (relativeName.StartsWith(Path.DirectorySeparatorChar.ToString()))
			{
				return base.RootFolder.GetFolderFromRelativeName(session, relativeName.Remove(0, Path.DirectorySeparatorChar.ToString().Length));
			}
			checked
			{
				if (relativeName.StartsWith("." + Path.DirectorySeparatorChar))
				{
					return GetFolderFromRelativeName(session, relativeName.Remove(0, ".".Length + Path.DirectorySeparatorChar.ToString().Length));
				}
				if (relativeName.StartsWith(".." + Path.DirectorySeparatorChar))
				{
					if (base.ParentFolder == null)
					{
						throw new FileSystemException("GetFolder was called with a relative name refering to the folder's parent (\"..\") on a folder that has no parent.", this);
					}
					return base.ParentFolder.GetFolderFromRelativeName(session, relativeName.Remove(0, "..".Length + Path.DirectorySeparatorChar.ToString().Length));
				}
				relativeName = (relativeName.EndsWith(Path.DirectorySeparatorChar.ToString()) ? relativeName : (relativeName + Path.DirectorySeparatorChar));
				string text = relativeName.Remove(0, relativeName.IndexOf(Path.DirectorySeparatorChar) + 1);
				if (text.Length != 0)
				{
					string folderName = relativeName.Substring(0, relativeName.IndexOf(Path.DirectorySeparatorChar));
					if (!text.StartsWith(".." + Path.DirectorySeparatorChar) && !text.StartsWith(Path.DirectorySeparatorChar.ToString()))
					{
						session.OnScanningFolder(this);
					}
					AbstractFolder abstractFolder = DoGetFolder(session, folderName);
					return abstractFolder.GetFolderFromRelativeName(session, text);
				}
				relativeName = relativeName.TrimEnd(Path.DirectorySeparatorChar);
				session.OnScanningFolder(this);
				return DoGetFolder(session, relativeName);
			}
		}

		protected virtual AbstractFile[] DoGetFiles(FileSystemEventsSession session, bool recursive, Filter[] filters)
		{
			FileSystemItem[] items = GetItems(session.Events, session.UserData, recursive: false);
			ArrayList arrayList = new ArrayList();
			FileSystemItem[] array = items;
			foreach (FileSystemItem fileSystemItem in array)
			{
				if (fileSystemItem is AbstractFile)
				{
					if (Filter.IsItemMatchingAll(fileSystemItem, filters))
					{
						arrayList.Add(fileSystemItem);
					}
				}
				else if (recursive)
				{
					AbstractFolder abstractFolder = fileSystemItem as AbstractFolder;
					if (Filter.IsItemMatchingAll(abstractFolder, FilterScope.Recurse, filters) != FilterResult.Reject)
					{
						arrayList.AddRange(abstractFolder.DoGetFiles(session, recursive: true, filters));
					}
				}
			}
			return (AbstractFile[])arrayList.ToArray(typeof(AbstractFile));
		}

		protected internal override void BuildList(FileSystemEventsSession session, FileSystemItem destination, bool forward, bool recursive, bool includeFolders, Filter[] filters)
		{
			FileSystemItem[] items = GetItems(session.Events, session.UserData, recursive: false);
			if (forward)
			{
				if (includeFolders && Filter.IsItemMatchingAll(this, filters))
				{
					session.OnNewItemToProcess(this, destination);
				}
				FileSystemItem[] array = items;
				foreach (FileSystemItem fileSystemItem in array)
				{
					if (fileSystemItem is AbstractFile)
					{
						if (destination != null)
						{
							fileSystemItem.BuildList(session, ((AbstractFolder)destination).GetFile(fileSystemItem.Name), forward: false, recursive: false, includeFolders: false, filters);
						}
						else
						{
							fileSystemItem.BuildList(session, null, forward: false, recursive: false, includeFolders: false, filters);
						}
					}
				}
				FileSystemItem[] array2 = items;
				foreach (FileSystemItem fileSystemItem2 in array2)
				{
					if (!(fileSystemItem2 is AbstractFolder))
					{
						continue;
					}
					if (recursive)
					{
						if (Filter.IsItemMatchingAll(fileSystemItem2, FilterScope.Recurse, filters) != FilterResult.Reject)
						{
							if (destination != null)
							{
								fileSystemItem2.BuildList(session, ((AbstractFolder)destination).GetFolder(fileSystemItem2.Name), forward: true, recursive: true, includeFolders, filters);
							}
							else
							{
								fileSystemItem2.BuildList(session, null, forward: true, recursive: true, includeFolders, filters);
							}
						}
					}
					else if (includeFolders && Filter.IsItemMatchingAll(fileSystemItem2, filters))
					{
						if (destination != null)
						{
							session.OnNewItemToProcess(fileSystemItem2, ((AbstractFolder)destination).GetFolder(fileSystemItem2.Name));
						}
						else
						{
							session.OnNewItemToProcess(fileSystemItem2, null);
						}
					}
				}
				return;
			}
			FileSystemItem[] array3 = items;
			foreach (FileSystemItem fileSystemItem3 in array3)
			{
				if (!(fileSystemItem3 is AbstractFolder))
				{
					continue;
				}
				if (recursive)
				{
					if (destination != null)
					{
						fileSystemItem3.BuildList(session, ((AbstractFolder)destination).GetFolder(fileSystemItem3.Name), forward: false, recursive: true, includeFolders, filters);
					}
					else
					{
						fileSystemItem3.BuildList(session, null, forward: false, recursive: true, includeFolders, filters);
					}
				}
				else if (includeFolders && Filter.IsItemMatchingAll(fileSystemItem3, filters))
				{
					if (destination != null)
					{
						session.OnNewItemToProcess(fileSystemItem3, ((AbstractFolder)destination).GetFolder(fileSystemItem3.Name));
					}
					else
					{
						session.OnNewItemToProcess(fileSystemItem3, null);
					}
				}
			}
			FileSystemItem[] array4 = items;
			foreach (FileSystemItem fileSystemItem4 in array4)
			{
				if (fileSystemItem4 is AbstractFile)
				{
					if (destination != null)
					{
						fileSystemItem4.BuildList(session, ((AbstractFolder)destination).GetFile(fileSystemItem4.Name), forward: false, recursive: false, includeFolders: false, filters);
					}
					else
					{
						fileSystemItem4.BuildList(session, null, forward: false, recursive: false, includeFolders: false, filters);
					}
				}
			}
			if (includeFolders && Filter.IsItemMatchingAll(this, filters))
			{
				session.OnNewItemToProcess(this, destination);
			}
		}

		protected abstract FileSystemItem[] DoGetChildItems(FileSystemEventsSession session);

		protected abstract AbstractFile DoGetFile(FileSystemEventsSession session, string fileName);

		protected abstract AbstractFolder DoGetFolder(FileSystemEventsSession session, string folderName);

		private Exception PrepareException(Exception except)
		{
			if (except is FileSystemException ex)
			{
				if (ex.Item == null)
				{
					ex.Item = this;
				}
				return ex;
			}
			if (ExceptionFilter.IsPublicException(except))
			{
				return except;
			}
			return new FileSystemInternalException(this, except);
		}
	}

	public class AutoBatchUpdate : MarshalByRefObject, IDisposable
	{
		private IBatchUpdateable m_batchUpdateable;

		private FileSystemEvents m_events;

		private object m_userData;

		public AutoBatchUpdate(object batchUpdateable)
		{
			m_batchUpdateable = batchUpdateable as IBatchUpdateable;
			if (m_batchUpdateable != null)
			{
				m_batchUpdateable.BeginUpdate();
			}
		}

		public AutoBatchUpdate(object batchUpdateable, FileSystemEvents events, object userData)
		{
			if (events == null)
			{
				throw new ArgumentNullException("events");
			}
			m_batchUpdateable = batchUpdateable as IBatchUpdateable;
			m_events = events;
			m_userData = userData;
			if (m_batchUpdateable != null)
			{
				m_batchUpdateable.BeginUpdate(events, userData);
			}
		}

		{
			if (m_batchUpdateable != null)
			{
				if (m_events == null)
				{
					m_batchUpdateable.EndUpdate();
				}
				else
				{
					m_batchUpdateable.EndUpdate(m_events, m_userData);
				}
			}

		protected override Stream DoOpenWrite(FileSystemEventsSession session, bool overwrite, FileShare share)
		{
			m_info.Refresh();
			if (!m_info.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			if ((m_info.Attributes & FileAttributes.ReadOnly) == FileAttributes.ReadOnly)
			{
				throw new ItemIsReadOnlyException(this);
			}
			try
			{
				m_knowsSize = false;
				if (overwrite)
				{
					return m_info.Open(FileMode.Create, FileAccess.Write, share);
				}
				Stream stream = m_info.Open(FileMode.Open, FileAccess.Write, share);
				try

					return stream;
				}
				catch

					stream.Close();
					return m_info.Open(FileMode.Append, FileAccess.Write, share);
				}
			}
			catch (IOException inner)
			{
				throw new FileSystemIOException("The file item could not be opened for writing.", this, inner);
			}
		}

		protected internal override void DoCreate(FileSystemEventsSession session)
		{
			m_info.Refresh();
			if (m_info.Exists)
			{
				throw new ItemAlreadyExistsException(this);
			}
			bool flag = false;
			try
			{
				m_info.Create().Close();
				flag = true;
				if (m_creationTime != DateTime.MinValue)
				{
					m_info.CreationTime = m_creationTime;
					m_knowsCreationTime = true;
				}
				if (m_lastWriteTime != DateTime.MinValue)
				{
					m_info.LastWriteTime = m_lastWriteTime;
					m_knowsLastWriteTime = true;
				}
				if (m_lastAccessTime != DateTime.MinValue)
				{
					m_info.LastAccessTime = m_lastAccessTime;
					m_knowsLastAccessTime = true;
				}
				try
				{
					m_info.Attributes = m_attributes;
					m_knowsAttributes = true;
				}
				catch (ArgumentException)
				{
					m_knowsAttributes = false;
				}
				m_size = 0L;
				m_knowsSize = true;
			}
			catch (Exception inner)
			{
				try
				{
					if (flag)
					{
						m_info.Delete();
					}
				}
				catch
				{
				}
				throw new FileSystemException("The physical file represented by the item could not be created.", this, inner);
			}
		}

		public override bool IsSameAs(FileSystemItem target)
		{
			if (target == null)
			{
				return false;
			}
			if (target is DiskFile)
			{
				return m_fullName == target.FullName;
			}
			return false;
		}

		protected override bool IsPathRooted(string path)
		{
			path = path.Replace(Path.AltDirectorySeparatorChar, Path.DirectorySeparatorChar);
			if (path.StartsWith(Path.DirectorySeparatorChar.ToString()))
			{
				return false;
			}
			try
			{
				return StringCheck.ReplaceNull(Path.GetPathRoot(path)).Length != 0;
			}
			catch (PathTooLongException inner)
			{
				throw new FileSystemIOException("The root path is too long for path : " + path, null, inner);
			}
		}

		private void SetFileInfo(FileInfo info)
		{
			m_info = info;
			if (m_info.Exists)
			{
				m_attributes = m_info.Attributes;
				m_knowsAttributes = true;
				m_creationTime = m_info.CreationTime;
				m_knowsCreationTime = true;
				m_lastAccessTime = m_info.LastAccessTime;
				m_knowsLastAccessTime = true;
				m_lastWriteTime = m_info.LastWriteTime;
				m_knowsLastWriteTime = true;
				m_size = m_info.Length;
				m_knowsSize = true;
			}
			else
			{
				m_knowsAttributes = false;
				m_knowsCreationTime = false;
				m_knowsLastAccessTime = false;
				m_knowsLastWriteTime = false;
				m_knowsSize = false;
			}
		}

					SetDirectoryInfo(new DirectoryInfo(m_fullNameForIO));

		public DiskFolder(string folderName)
		{
			if (folderName == null)
			{
				throw new ArgumentNullException("folderName");
			}
			folderName = folderName.TrimEnd();
			if (folderName.Length == 0)
			{
				throw new ArgumentException("The folder name is empty or contains only whitespace.", "folderName");
			}
			if (folderName.IndexOfAny(InvalidChars) != -1)
			{
				throw new ArgumentException("The folder name contains invalid characters.", "folderName");
			}
			folderName = folderName.Replace(Path.AltDirectorySeparatorChar, Path.DirectorySeparatorChar);
			try
			{
				m_fullName = Path.GetFullPath(folderName);
			}
			catch (Exception innerExcept)
			{
				ThrowException.ThrowArgumentException("The folder name is invalid.", "folderName", innerExcept);
			}
			if (!m_fullName.EndsWith(Path.DirectorySeparatorChar.ToString()))
			{
				m_fullName += Path.DirectorySeparatorChar;
			}
			string empty = string.Empty;
			try
			{
				empty = Path.GetPathRoot(m_fullName);
			}
			catch (PathTooLongException inner)
			{
				throw new FileSystemIOException("The folder's full name is too long for folder : " + base.FullName, null, inner);
			}
			if (!empty.EndsWith(Path.DirectorySeparatorChar.ToString()))
			{
				empty += Path.DirectorySeparatorChar;
			}
			if (m_fullName.Length == empty.Length)
			{
				m_fullNameForIO = m_fullName;
				m_name = string.Empty;
				m_isRoot = true;
			}
			else
			{
				m_fullNameForIO = m_fullName.TrimEnd(Path.DirectorySeparatorChar);
				m_name = StringCheck.ReplaceNull(Path.GetFileName(m_fullNameForIO));
			}
			try
			{
				SetDirectoryInfo(new DirectoryInfo(m_fullNameForIO));
			}
			catch (PathTooLongException inner2)
			{
				throw new FileSystemIOException("Unable to set DirectoryInfo for folder : " + base.FullName, null, inner2);
			}
		}

		internal DiskFolder(DirectoryInfo info)
		{
			if (info == null)
			{
				throw new ArgumentNullException("info");
			}
			m_fullNameForIO = info.FullName;
			if (m_fullNameForIO.EndsWith(Path.DirectorySeparatorChar.ToString()))
			{
				m_fullName = m_fullNameForIO;
			}
			else
			{
				m_fullName = m_fullNameForIO + Path.DirectorySeparatorChar;
			}
			m_isRoot = info.Parent == null;
			if (!m_isRoot)
			{
				m_name = info.Name;
			}
			SetDirectoryInfo(info);
		}

		protected override FileSystemItem[] DoGetChildItems(FileSystemEventsSession session)
		{
			m_info.Refresh();
			if (!m_info.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			DirectoryInfo[] directories = m_info.GetDirectories();
			FileInfo[] files = m_info.GetFiles();
			int num = directories.Length;
			checked
			{
				FileSystemItem[] array = new FileSystemItem[num + files.Length];
				for (int i = 0; i < num; i++)
				{
					array[i] = new DiskFolder(directories[i]);
				}
				for (int j = 0; j < files.Length; j++)
				{
					array[num + j] = new DiskFile(files[j]);
				}
				return array;
			}
		}

		protected internal override void DoDelete(FileSystemEventsSession session)
		{
			m_info.Refresh();
			if (!m_info.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			if ((m_info.Attributes & FileAttributes.ReadOnly) == FileAttributes.ReadOnly)
			{
				throw new ItemIsReadOnlyException(this);
			}
			try
			{
				m_info.Delete();
			}
			catch (Exception inner)
			{
				throw new FileSystemException("The file could not be deleted because it is currently in use.", this, inner);
			}
			m_creationTime = DateTime.MinValue;
			m_knowsCreationTime = true;
			m_lastAccessTime = DateTime.MinValue;
			m_knowsLastAccessTime = true;
			m_lastWriteTime = DateTime.MinValue;
			m_knowsLastWriteTime = true;
			m_attributes = FileAttributes.Directory | FileAttributes.Normal;
			m_knowsAttributes = true;
		}

		protected internal override void DoCreate(FileSystemEventsSession session)
		{
			m_info.Refresh();
			if (m_info.Exists)
			{
				throw new ItemAlreadyExistsException(this);
			}
			bool flag = false;
			try
			{
				m_info.Create();
				flag = true;
				if (m_creationTime != DateTime.MinValue)
				{
					m_info.CreationTime = m_creationTime;
					m_knowsCreationTime = true;
				}
				if (m_lastWriteTime != DateTime.MinValue)
				{
					m_info.LastWriteTime = m_lastWriteTime;
					m_knowsLastWriteTime = true;
				}
				if (m_lastAccessTime != DateTime.MinValue)
				{
					m_info.LastAccessTime = m_lastAccessTime;
					m_knowsLastAccessTime = true;
				}
				try
				{
					m_info.Attributes = m_attributes;
					m_knowsAttributes = true;
				}
				catch
				{
					m_knowsAttributes = false;
				}
			}
			catch (Exception inner)
			{
				try
				{
					if (flag)
					{
						m_info.Delete();
					}
				}
				catch
				{
				}
				throw new FileSystemException("The physical folder represented by the item could not be created.", this, inner);
			}
		}

		protected override void DoRefresh(FileSystemEventsSession session)
		{
			m_knowsAttributes = false;
			m_knowsCreationTime = false;
			m_knowsLastWriteTime = false;
			m_knowsLastAccessTime = false;
		}

		protected override AbstractFile DoGetFile(FileSystemEventsSession session, string fileName)
		{
			return new DiskFile(base.FullName + fileName);
		}

		protected override AbstractFolder DoGetFolder(FileSystemEventsSession session, string folderName)
		{
			return new DiskFolder(base.FullName + folderName);
		}

		public override bool IsSameAs(FileSystemItem target)
		{
			if (target == null)
			{
				return false;
			}
			if (target is DiskFolder)
			{
				return base.FullName == target.FullName;
			}
			return false;
		}

		protected override bool IsPathRooted(string path)
		{
			path = path.Replace(Path.AltDirectorySeparatorChar, Path.DirectorySeparatorChar);
			if (path.StartsWith(Path.DirectorySeparatorChar.ToString()))
			{
				return false;
			}
			try
			{
				return StringCheck.ReplaceNull(Path.GetPathRoot(path)).Length != 0;
			}
			catch (PathTooLongException inner)
			{
				throw new FileSystemIOException("The folder's full name is too long for folder : " + path, null, inner);
			}
		}

		private void SetDirectoryInfo(DirectoryInfo info)
		{
			m_info = info;
			if (m_info.Exists)
			{
				m_attributes = m_info.Attributes;
				m_knowsAttributes = true;
				m_creationTime = m_info.CreationTime;
				m_knowsCreationTime = true;
				m_lastAccessTime = m_info.LastAccessTime;
				m_knowsLastAccessTime = true;
				m_lastWriteTime = m_info.LastWriteTime;
				m_knowsLastWriteTime = true;
			}
			else

				m_knowsLastAccessTime = false;
				m_knowsLastWriteTime = false;
			}

		{
			return new FileSystemEventsSession(this, sender, userData);

			{
				m_subOperationCount++;
			}
		}

		{
			checked
			{
				m_subOperationCount--;
			}
		}

				this.ScanningFolder(sender, e);

		}

			ValidateParams(sender, e);

			{
				this.ItemException(sender, e);

	public class FileSystemEventsSession
	{
		protected ProgressionInfo m_currentFileBytes;

		protected ProgressionInfo m_allFilesBytes;

		protected ProgressionInfo m_allItems;

		protected ScanningFolderEventArgs m_scanningFolderArgs;

		protected ByteProgressionEventArgs m_byteProgressionArgs;

		protected ItemProgressionEventArgs m_itemProgressionArgs;

		protected ItemExceptionEventArgs m_itemExceptionArgs;

		protected FileSystemItem m_currentItem;

		protected FileSystemItem m_targetItem;

		protected FileSystemEvents m_events;

		protected object m_sender;

		protected object m_userData;

		protected Queue m_itemPairQueue = new Queue();

		protected SortedList m_itemPairList;

		private IComparer m_itemPairListComparer;

		public FileSystemEvents Events => m_events;

		public object UserData => m_userData;

		public FileSystemItem CurrentItem
		{
			get
			{
				return m_currentItem;
			}
			set
			{
				m_currentItem = value;
			}
		}

		public FileSystemItem TargetItem
		{
			get
			{
				return m_targetItem;
			}
			set
			{
				m_targetItem = value;
			}
		}

		public bool IsEmpty
		{
			get
			{
				if (m_itemPairList == null)
				{
					return m_itemPairQueue.Count == 0;
				}
				return m_itemPairList.Count == 0;
			}
		}

		protected internal FileSystemEventsSession(FileSystemEvents events, object sender, object userData)
		{
			m_events = events;
			m_sender = sender;
			m_userData = userData;
			m_currentFileBytes = new ProgressionInfo();
			m_allFilesBytes = new ProgressionInfo();
			m_allItems = new ProgressionInfo();
			m_scanningFolderArgs = new ScanningFolderEventArgs();
			m_byteProgressionArgs = new ByteProgressionEventArgs(m_currentFileBytes, m_allFilesBytes);
			m_itemProgressionArgs = new ItemProgressionEventArgs(m_allItems);
			m_itemExceptionArgs = new ItemExceptionEventArgs();
		}

		internal FileSystemEventsSession(FileSystemEvents events, object sender)
		{
			m_events = events;
			m_sender = sender;
		}

		internal void Reset()
		{
			m_itemPairQueue.Clear();
			m_itemPairList = null;
			m_itemPairListComparer = null;
		}

		public void OnScanningFolder(AbstractFolder currentFolder)
		{
			if (m_events.WantsScanningFolder)
			{
				m_scanningFolderArgs.CurrentItem = currentFolder;
				m_scanningFolderArgs.UserData = m_userData;
				try
				{
					m_events.OnScanningFolder(m_sender, m_scanningFolderArgs);
				}
				catch (Exception inner)
				{
					throw new TargetInvocationException(inner);
				}
				m_userData = m_scanningFolderArgs.UserData;
			}
		}

		public void OnNewItemToProcess(FileSystemItem currentItem, FileSystemItem targetItem)
		{
			if (currentItem != null || targetItem != null)
			{
				ItemPair itemPair = new ItemPair(currentItem, targetItem);
				if (m_itemPairList == null)
				{
					m_itemPairQueue.Enqueue(itemPair);
				}
				else
				{
					m_itemPairList.Add(itemPair, itemPair);
				}
			}
			checked
			{
				if (m_events.WantsByteProgression && currentItem is AbstractFile && currentItem.Exists)
				{
					long size = ((AbstractFile)currentItem).Size;
					if (size > 0)
					{
						m_allFilesBytes.Total += size;
					}
				}
				if (m_events.WantsItemProgression)
				{
					m_allItems.Total++;
				}
			}
		}

		public void OnProcessingNewItem(FileSystemItem current, FileSystemItem target, bool raiseBytesProgression)
		{
			CurrentItem = current;
			TargetItem = target;
			checked
			{
				if (m_events.WantsItemProgression)
				{
					m_itemProgressionArgs.CurrentItem = CurrentItem;
					m_itemProgressionArgs.TargetItem = TargetItem;
					m_itemProgressionArgs.UserData = m_userData;
					try
					{
						m_events.OnItemProgression(m_sender, m_itemProgressionArgs);
					}
					catch (Exception inner)
					{
						throw new TargetInvocationException(inner);
					}
					m_userData = m_itemProgressionArgs.UserData;
					m_allItems.Processed++;
				}
				if (!m_events.WantsByteProgression || !raiseBytesProgression)
				{
					return;
				}
				m_byteProgressionArgs.CurrentItem = CurrentItem;
				m_byteProgressionArgs.TargetItem = TargetItem;
				m_currentFileBytes.Total = 0L;
				m_currentFileBytes.Processed = 0L;
				if (current is AbstractFile && current.Exists)
				{
					long size = ((AbstractFile)current).Size;
					if (size > 0)
					{
						m_currentFileBytes.Total += size;
					}
				}
				m_byteProgressionArgs.UserData = m_userData;
				try
				{
					m_events.OnByteProgression(m_sender, m_byteProgressionArgs);
				}
				catch (Exception inner2)
				{
					throw new TargetInvocationException(inner2);
				}
				m_userData = m_byteProgressionArgs.UserData;
			}
		}

		public ItemPair GetNextItemPair(bool raiseBytesProgression)
		{
			if (IsEmpty)
			{
				if (m_allItems == null || m_allItems.Processed == m_allItems.Total)
				{
					OnCompletion();
				}
				return null;
			}
			ItemPair itemPair = null;
			if (m_itemPairList == null)
			{
				itemPair = (ItemPair)m_itemPairQueue.Dequeue();
			}
			else
			{
				itemPair = (ItemPair)m_itemPairList.GetByIndex(0);
				m_itemPairList.RemoveAt(0);
			}
			if (itemPair.current is AbstractFile)
			{
				OnProcessingNewItem(itemPair.current, itemPair.target, raiseBytesProgression);
			}
			else
			{
				OnProcessingNewItem(itemPair.current, itemPair.target, raiseBytesProgression: false);
			}
			return itemPair;
		}

		public void OnBytesProcessed(long increment)
		{
			if (!m_events.WantsByteProgression)
			{
				return;
			}
			int num;
			checked
			{
				num = (int)unchecked(m_currentFileBytes.Processed / 65536);
				m_currentFileBytes.Processed += increment;
				m_allFilesBytes.Processed += increment;
				m_byteProgressionArgs.UserData = m_userData;
			}
			if (m_currentFileBytes.Processed / 65536 > num || m_currentFileBytes.Percent == 100)
			{
				try
				{
					m_events.OnByteProgression(m_sender, m_byteProgressionArgs);
				}
				catch (Exception inner)
				{
					throw new TargetInvocationException(inner);
				}
			}
			m_userData = m_byteProgressionArgs.UserData;
		}

		public void OnBytesCompleted()
		{
			checked
			{
				if (m_events.WantsByteProgression && m_currentFileBytes.Total - m_currentFileBytes.Processed > 0)
				{
					OnBytesProcessed(m_currentFileBytes.Total - m_currentFileBytes.Processed);
				}
			}
		}

		public void OnUndoBytesProcessed()
		{
			checked
			{
				if (m_events.WantsByteProgression)
				{
					m_allFilesBytes.Processed -= m_currentFileBytes.Processed;
				}
			}
		}

		public bool OnException(Exception except)
		{
			if (!(except is FileSystemException) && !ExceptionFilter.IsPublicException(except))
			{
				except = new FileSystemInternalException(except);
			}
			if (m_events.WantsItemException)
			{
				m_itemExceptionArgs.CurrentItem = CurrentItem;
				m_itemExceptionArgs.TargetItem = TargetItem;
				m_itemExceptionArgs.Action = ItemExceptionAction.Abort;
				m_itemExceptionArgs.Exception = except;
				m_itemExceptionArgs.UserData = m_userData;
				try
				{
					m_events.OnItemException(m_sender, m_itemExceptionArgs);
				}
				catch (Exception inner)
				{
					throw new TargetInvocationException(inner);
				}
				m_userData = m_itemExceptionArgs.UserData;
				if (m_itemExceptionArgs.Action == ItemExceptionAction.Abort)
				{
					throw m_itemExceptionArgs.Exception;
				}
				return m_itemExceptionArgs.Action == ItemExceptionAction.Retry;
			}
			throw except;
		}

		public void OnCompletion()
		{
			if (m_events.WantsItemProgression && m_itemProgressionArgs.AllItems.Processed > 0)
			{
				m_itemProgressionArgs.UserData = m_userData;
				try
				{
					m_events.OnItemProgression(m_sender, m_itemProgressionArgs);
				}
				catch (Exception inner)
				{
					throw new TargetInvocationException(inner);
				}
				m_itemProgressionArgs.CurrentItem = null;
				m_itemProgressionArgs.TargetItem = null;
				m_userData = m_itemProgressionArgs.UserData;
			}
		}

		public bool AssignComparer(IComparer comparer)
		{
			if (comparer == null)
			{
				throw new ArgumentNullException("comparer");
			}
			if (m_itemPairList != null)
			{
				if (m_itemPairListComparer != comparer)
				{
					throw new FileSystemInternalException("Already using a different comparer.");
				}
				return false;
			}
			m_itemPairList = new SortedList(comparer);
			m_itemPairListComparer = comparer;
			while (m_itemPairQueue.Count > 0)
			{
				ItemPair itemPair = (ItemPair)m_itemPairQueue.Dequeue();
				m_itemPairList.Add(itemPair, itemPair);
			}
			return true;
		}
	}

		protected virtual ProductVersion LatestVersion(LicenseProduct product)
		{
			if (product == LicenseProduct.InvalidProduct)
			{
				return AllowedVersions[0];
			}
			ProductVersion[] allowedVersions = AllowedVersions;
			for (int i = 0; i < allowedVersions.Length; i++)
			{
				ProductVersion result = allowedVersions[i];
				if (result.Product == product)
				{
					return result;
				}
			}
			return AllowedVersions[0];
		}

		private string GetLicenseKeyFromRegistry()
		{
			string licenseKey = string.Empty;
			try
			{
				XceedLicenseManager lm = new XceedLicenseManager();
				RegistryKey registryKey = null;
				ProductVersion[] latestVersions = LatestVersions;
				for (int i = 0; i < latestVersions.Length; i++)
				{
					ProductVersion productVersion = latestVersions[i];
					if (!productVersion.SupportSampleKey)
					{
						continue;
					}
					registryKey = Registry.ClassesRoot.OpenSubKey("Licenses\\4507C5F2-9077-40DE-A621-64EC59BF6829\\" + XceedLicenseManager.GetProductCodeName(productVersion.Product));
					if (registryKey == null)
					{
						continue;
					}
					RegistryKey registryKey2 = registryKey;
					byte major = productVersion.Major;
					string text = major.ToString(CultureInfo.InvariantCulture);
					byte minor = productVersion.Minor;
					licenseKey = (string)registryKey2.GetValue(text + "." + minor.ToString(CultureInfo.InvariantCulture));
					registryKey.Close();
					if (licenseKey != null)
					{
						try
						{
							ValidateLicenseKey(ref lm, ref licenseKey, performNagIfNecessary: false);
						}
						catch (LicenseException)
						{
							continue;
						}
						break;
					}
				}
			}
			catch (SecurityException)
			{
			}
			if (licenseKey == null)
			{
				licenseKey = string.Empty;
			}
			m_isFromRegistry = true;
			return licenseKey;
		}

		private void ValidateLicenseKey(ref XceedLicenseManager lm, ref string licenseKey, bool performNagIfNecessary)
		{
			try
			{
				lm.AnalyzeLicense(licenseKey);
			}
			catch (XceedLicenseManagerException)
			{
				ThrowKeyInvalid();
			}
			bool flag = false;
			ProductVersion[] allowedVersions = AllowedVersions;
			for (int i = 0; i < allowedVersions.Length; i++)
			{
				ProductVersion productVersion = allowedVersions[i];
				if (productVersion.Product == lm.Product && productVersion.Major == lm.VerMajor && productVersion.Minor == lm.VerMinor)
				{
					flag = true;
					ProductVersion productVersion2 = LatestVersion(lm.Product);
					if ((productVersion.Major != productVersion2.Major || productVersion.Minor != productVersion2.Minor) && RunningOnDevelopperMachine && performNagIfNecessary)
					{
						DisplayRegisteredOlderVersion(productVersion.Major, productVersion.Minor, productVersion.Product);
					}
					break;
				}
			}
			if (!flag)
			{
				ThrowKeyInvalidProductVersion();
			}
			if (lm.Originator < LicenseOrigin.TrialVersion || lm.Originator > LicenseOrigin.TrialRenew9)
			{
				return;
			}
			checked
			{
				int days = new TimeSpan(lm.Date.Ticks - DateTime.Today.Ticks).Days;
				if (days <= 0)
				{
					if (RunningOnDevelopperMachine && performNagIfNecessary)
					{
						DisplayTrialExpired(-1 * default(TimeSpan).Days, lm.Product);
					}
					ThrowTrialKeyExpired();
				}
				else if (RunningOnDevelopperMachine && performNagIfNecessary)
				{
					DisplayTrial(days, lm.Product);
				}
			}
		}

			{

			}

			{
				return false;
			}
			if (m_cannotFindAnalyzer)
			{
				return false;
			}
			if (!RunningOnDevelopperMachine)
			{
				m_cannotFindAnalyzer = true;
				return false;
			}
			if (informationType == LicenseInformationType.FeatureRequireUpgrade)
			{
				if (m_performedFeratureAnalysis.Contains(informationType))
				{
					return false;
				}
				m_performedFeratureAnalysis.Add(informationType);
			}
			return true;
		}

		private bool DisplayInformation(string licenseNotificationPath, string arguments, LicenseProduct product)
		{
			Process process = null;
			ProcessStartInfo processStartInfo = null;
			try
			{
				processStartInfo = new ProcessStartInfo(licenseNotificationPath, arguments);
				processStartInfo.ErrorDialog = false;
				processStartInfo.UseShellExecute = true;
				process = new Process();
				process.StartInfo = processStartInfo;
				if (!File.Exists(licenseNotificationPath))
				{
					m_cannotFindAnalyzer = true;
					return false;
				}
				if (!process.Start())
				{
					m_cannotFindAnalyzer = true;
					return false;
				}
			}
			catch (Exception)
			{
				m_cannotFindAnalyzer = true;
				return false;
			}
			finally
			{
				process?.Dispose();
			}
			return true;
		}

		private static bool LastDisplayWasToday()
		{
			RegistryKey registryKey = null;
			try
			{
				registryKey = Registry.CurrentUser.CreateSubKey("Software\\Xceed Software\\LicenseAnalyzer");
				if (registryKey == null)
				{
					return false;
				}
				if (registryKey.GetValue("TrialVerification") is string s)
				{
					try
					{
						if (long.Parse(s) == DateTime.Today.Date.Ticks)
						{
							return true;
						}
					}
					catch
					{
						return true;
					}
				}
				registryKey.SetValue("TrialVerification", DateTime.Today.Date.Ticks);
				return false;
			}
			catch
			{
				return true;
			}
			finally
			{
				registryKey?.Close();
			}
		}

					return true;
				}

						{
							return true;
						}
					}
					catch
					{
						return true;
					}
				}
				registryKey.SetValue("OlderVersionState", true);
				return false;
			}
			catch
			{
				return true;
			}
			finally
			{
				registryKey?.Close();
			}
		}

	[Flags]
	public enum FilterScope
	{
		File = 1,
		Folder = 2,
		All = 3,
		Recurse = 4
	}
	public enum FilterResult
	{
		Reject,
		Accept,
		NotApplicable
	}
	public abstract class Filter
	{
		private FilterScope m_scope = FilterScope.File;

		public virtual FilterScope Scope
		{
			get
			{
				return m_scope;
			}
			set
			{
				m_scope = value;
			}
		}

		public static Filter[] GetFiltersFromParams(params object[] filters)
		{
			if (filters == null || filters.Length == 0)
			{
				return new Filter[0];
			}
			if (filters.GetType() == typeof(Filter[]))
			{
				return (Filter[])filters;
			}
			ArrayList arrayList = new ArrayList();
			foreach (object obj in filters)
			{
				if (obj is Array)
				{
					Array array = (Array)obj;
					if (array.GetType() == typeof(Filter[]))
					{
						arrayList.AddRange(array);
						continue;
					}
					foreach (object item in array)
					{
						arrayList.AddRange(GetFiltersFromParams(item));
					}
				}
				else if (obj is string)
				{
					string text = ((string)obj).TrimEnd();
					if (text.Length != 0)
					{
						arrayList.Add(new NameFilter(text));
					}
				}
				else if (obj is FileAttributes)
				{
					arrayList.Add(new AttributeFilter((FileAttributes)obj));
				}
				else
				{
					if (!(obj is Filter))
					{
						throw new ArgumentException("The filter parameter array contains an unrecognized type.", "filters");
					}
					arrayList.Add(obj);
				}
			}
			return (Filter[])arrayList.ToArray(typeof(Filter));
		}

		public static bool IsItemMatchingOne(FileSystemItem item, Filter[] filters)
		{
			if (item == null)
			{
				throw new ArgumentNullException("item");
			}
			if (filters == null)
			{
				throw new ArgumentNullException("filters");
			}
			bool flag = false;
			foreach (Filter filter in filters)
			{
				if (filter.IsItemInScope(item))
				{
					flag = true;
					if (filter.IsItemMatching(item))
					{
						return true;
					}
				}
			}
			return !flag;
		}

		public static FilterResult IsItemMatchingOne(FileSystemItem item, FilterScope applicableScope, Filter[] filters)
		{
			if (item == null)
			{
				throw new ArgumentNullException("item");
			}
			if (filters == null)
			{
				throw new ArgumentNullException("filters");
			}
			FilterResult result = FilterResult.NotApplicable;
			foreach (Filter filter in filters)
			{
				switch (filter.IsItemMatching(item, applicableScope))
				{
				case FilterResult.Accept:
					return FilterResult.Accept;
				case FilterResult.Reject:
					result = FilterResult.Reject;
					break;
				}
			}
			return result;
		}

		public static bool IsItemMatchingAll(FileSystemItem item, Filter[] filters)
		{
			if (item == null)
			{
				throw new ArgumentNullException("item");
			}
			if (filters == null)
			{
				throw new ArgumentNullException("filters");
			}
			foreach (Filter filter in filters)
			{
				if (filter.IsItemInScope(item) && !filter.IsItemMatching(item))
				{
					return false;
				}
			}
			return true;
		}

		public static FilterResult IsItemMatchingAll(FileSystemItem item, FilterScope applicableScope, Filter[] filters)
		{
			if (item == null)
			{
				throw new ArgumentNullException("item");
			}
			if (filters == null)
			{
				throw new ArgumentNullException("filters");
			}
			FilterResult result = FilterResult.NotApplicable;
			foreach (Filter filter in filters)
			{
				switch (filter.IsItemMatching(item, applicableScope))
				{
				case FilterResult.Reject:
					return FilterResult.Reject;
				case FilterResult.Accept:
					result = FilterResult.Accept;
					break;
				}
			}
			return result;
		}

		public abstract bool IsItemMatching(FileSystemItem item);

		public virtual FilterResult IsItemMatching(FileSystemItem item, FilterScope applicableScope)
		{
			if ((Scope & applicableScope) == 0)
			{
				return FilterResult.NotApplicable;
			}
			if (IsItemMatching(item))
			{
				return FilterResult.Accept;
			}
			return FilterResult.Reject;
		}

		protected Filter()
		{
		}

		protected Filter(FilterScope scope)
		{
			m_scope = scope;
		}

		protected bool IsItemInScope(FileSystemItem item)
		{
			if (item == null)
			{
				throw new ArgumentNullException("item");
			}
			if (item is AbstractFile)
			{
				return (Scope & FilterScope.File) == FilterScope.File;
			}
			if (item is AbstractFolder)
			{
				return (Scope & FilterScope.Folder) == FilterScope.Folder;
			}
			return false;
		}
	}

	public class DateTimeFilter : Filter
	{
		[Flags]
		public enum ApplicableDateTime
		{
			Creation = 1,
			LastWrite = 2,
			LastAccess = 4,
			All = 7
		}

		private DateTime m_minValue;

		private DateTime m_maxValue;

		private ApplicableDateTime m_applicableDateTimes;

		public DateTime MinDateTime => m_minValue;

		public DateTime MaxDateTime => m_maxValue;

		public ApplicableDateTime ApplicableDateTimes => m_applicableDateTimes;

		public override bool IsItemMatching(FileSystemItem item)
		{
			if (item == null)
			{
				throw new ArgumentNullException("item");
			}
			if ((ApplicableDateTimes & ApplicableDateTime.Creation) != 0 && (DateTime.Compare(item.CreationDateTime, m_minValue) < 0 || DateTime.Compare(item.CreationDateTime, m_maxValue) > 0))
			{
				return false;
			}
			if ((ApplicableDateTimes & ApplicableDateTime.LastAccess) != 0 && (DateTime.Compare(item.LastAccessDateTime, m_minValue) < 0 || DateTime.Compare(item.LastAccessDateTime, m_maxValue) > 0))
			{
				return false;
			}
			if ((ApplicableDateTimes & ApplicableDateTime.LastWrite) != 0 && (DateTime.Compare(item.LastWriteDateTime, m_minValue) < 0 || DateTime.Compare(item.LastWriteDateTime, m_maxValue) > 0))
			{
				return false;
			}
			return true;
		}

		public DateTimeFilter(DateTime minValue, DateTime maxValue)
			: this(minValue, maxValue, ApplicableDateTime.LastWrite, FilterScope.File)
		{
		}

		public DateTimeFilter(DateTime minValue, DateTime maxValue, ApplicableDateTime applicableDateTimes)
			: this(minValue, maxValue, applicableDateTimes, FilterScope.File)
		{
		}

		public DateTimeFilter(DateTime minValue, DateTime maxValue, FilterScope scope)
			: this(minValue, maxValue, ApplicableDateTime.LastWrite, scope)
		{
		}

		public DateTimeFilter(DateTime minValue, DateTime maxValue, ApplicableDateTime applicableDateTimes, FilterScope scope)
			: base(scope)
		{
			m_minValue = minValue;
			m_maxValue = maxValue;
			m_applicableDateTimes = applicableDateTimes;
		}
	}
	public class NameFilter : Filter
	{
		private string[] m_masks;

		private bool m_caseSensitive;

		public IEnumerable Masks => m_masks;

		public bool CaseSensitive => m_caseSensitive;

		public override bool IsItemMatching(FileSystemItem item)
		{
			if (item == null)
			{
				throw new ArgumentNullException("item");
			}
			foreach (string mask in Masks)
			{
				if (FileMatching.MatchName(item.Name, mask, CaseSensitive))
				{
					return true;
				}
			}
			return false;
		}

		public NameFilter(string mask)
			: this(mask, FilterScope.File)
		{
		}

		public NameFilter(string mask, FilterScope scope)
			: base(scope)
		{
			if (mask == null)
			{
				throw new ArgumentNullException("mask");
			}
			mask = mask.TrimEnd();
			if (mask.Length == 0)
			{
				throw new ArgumentException("The mask is an empty string, or contains only whitespace.", "mask");
			}
			if (mask.IndexOf(Path.DirectorySeparatorChar) != -1)
			{
				throw new ArgumentException("The mask contains a path.", "mask");
			}
			if (mask.StartsWith(">"))
			{
				m_caseSensitive = true;
				mask = mask.Remove(0, 1);
			}
			m_masks = mask.Split('|');
		}
	}

	public class IsolatedFile : AbstractFile
	{
		private string m_name = string.Empty;

		protected override long DoSize
		{
			get
			{
				if (!base.Exists)
				{
					throw new ItemDoesNotExistException(this);
				}
				return m_size;
			}
		}

		protected override string DoName
		{
			get
			{
				return m_name;
			}
			set
			{
				if (value == base.Name)
				{
					return;
				}
				if (base.Exists)
				{
					string fullName = base.ParentFolder.FullName;
					IsolatedFile isolatedFile = new IsolatedFile(fullName + value);
					isolatedFile.Create();
					bool flag = false;
					try
					{
						using (Stream stream = OpenRead())
						{
							using Stream stream2 = isolatedFile.OpenWrite(overwrite: true);
							byte[] array = new byte[65536];
							int count;
							while ((count = stream.Read(array, 0, array.Length)) > 0)
							{
								stream2.Write(array, 0, count);
							}
						}
						Delete();
						flag = true;
						m_fullName = isolatedFile.FullName;
						m_name = isolatedFile.Name;
						return;
					}
					catch (Exception ex)
					{
						if (!flag)
						{
							try
							{
								isolatedFile.Delete();
							}
							catch
							{
							}
						}
						if (ex is FileSystemIOException || ex is IOException)
						{
							throw new FileSystemException("An I/O exception occurred while renaming the item.", this, ex);
						}
						throw;
					}
				}
				m_fullName = base.ParentFolder.FullName + value;
				m_name = value;
			}
		}

		protected override bool DoHasAttributes => false;

		public IsolatedFile(string fileName)
		{
			if (fileName == null)
			{
				throw new ArgumentNullException("fileName");
			}
			fileName = fileName.TrimEnd();
			if (fileName.Length == 0)
			{
				throw new ArgumentException("The file name is empty or contains only whitespace.", "fileName");
			}
			if (fileName.IndexOfAny(InvalidChars) != -1)
			{
				throw new ArgumentException("The file name contains invalid characters.", "fileName");
			}
			fileName = fileName.Replace(Path.AltDirectorySeparatorChar, Path.DirectorySeparatorChar);
			if (fileName.EndsWith(Path.DirectorySeparatorChar.ToString()))
			{
				throw new ArgumentException("The fileName is not a file", "fileName");
			}
			fileName = (fileName.StartsWith(Path.DirectorySeparatorChar.ToString()) ? fileName : (Path.DirectorySeparatorChar + fileName));
			m_fullName = fileName;
			m_name = StringCheck.ReplaceNull(Path.GetFileName(fileName));
			if (m_name.Length == 0)
			{
				throw new ArgumentException("The file name represents a folder name.", "fileName");
			}
			try
			{
				RefreshSize();
			}
			catch
			{
			}
		}

		protected internal override void DoCreate(FileSystemEventsSession session)
		{
			if (base.Exists)
			{
				throw new ItemAlreadyExistsException(this);
			}
			if (!base.ParentFolder.Exists)
			{
				throw new FileSystemInternalException("An unexpected attempt was made to create an item in a folder that does not exist.", this);
			}
			try
			{
				IsolatedStorageFile domainStore = IsolatedFolder.m_domainStore;
				IsolatedStorageFileStream isolatedStorageFileStream = new IsolatedStorageFileStream(base.FullName, FileMode.CreateNew, domainStore);
				isolatedStorageFileStream.Close();
				m_size = 0L;
			}
			catch (IsolatedStorageException inner)
			{
				throw new FileSystemException("The physical file represented by the item could not be created.", this, inner);
			}
		}

		{
			if (!base.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			RefreshSize();
		}

		protected override Stream DoOpenRead(FileSystemEventsSession session, FileShare share)
		{

			{
				throw new ItemDoesNotExistException(this);

				return new IsolatedStorageFileStream(base.FullName, FileMode.Open, FileAccess.Read, share, domainStore);

			}
		}

		protected override Stream DoOpenWrite(FileSystemEventsSession session, bool overwrite, FileShare share)
		{
			if (!base.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			try
			{
				IsolatedStorageFile domainStore = IsolatedFolder.m_domainStore;
				if (overwrite)
				{
					return new IsolatedStorageFileStream(base.FullName, FileMode.Create, FileAccess.Write, share, domainStore);
				}
				return new IsolatedStorageFileStream(base.FullName, FileMode.Append, FileAccess.Write, share, domainStore);
			}
			catch (IsolatedStorageException inner)
			{
				throw new FileSystemIOException("The file item could not be opened for writing.", this, inner);
			}
		}

		public override bool IsSameAs(FileSystemItem target)
		{
			if (target == null)
			{
				return false;
			}
			if (target is IsolatedFile)
			{
				return base.FullName == target.FullName;
			}
			return false;
		}

		protected override bool IsPathRooted(string path)
		{
			return false;
		}

		protected internal override void DoDelete(FileSystemEventsSession session)
		{
			if (!base.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			try
			{
				IsolatedStorageFile domainStore = IsolatedFolder.m_domainStore;
				domainStore.DeleteFile(base.FullName);
				m_size = 0L;
			}
			catch (IsolatedStorageException inner)
			{
				throw new FileSystemException("The item could not be deleted.", this, inner);
			}
		}

		private void RefreshSize()
		{
			try
			{
				IsolatedStorageFile domainStore = IsolatedFolder.m_domainStore;
				using IsolatedStorageFileStream isolatedStorageFileStream = new IsolatedStorageFileStream(base.FullName, FileMode.Open, domainStore);
				m_size = isolatedStorageFileStream.Length;
			}
			catch
			{
				m_size = -1L;
			}
		}

			if (folderName.IndexOfAny(InvalidChars) != -1)
			{

			}
			folderName = folderName.Replace(Path.AltDirectorySeparatorChar, Path.DirectorySeparatorChar);
			folderName = (folderName.StartsWith(Path.DirectorySeparatorChar.ToString()) ? folderName : (Path.DirectorySeparatorChar + folderName));
			folderName = (folderName.EndsWith(Path.DirectorySeparatorChar.ToString()) ? folderName : (folderName + Path.DirectorySeparatorChar));
			m_fullName = folderName;
			if (m_fullName == Path.DirectorySeparatorChar.ToString())
			{

				m_isRoot = true;
			}

		{
			checked

					string[] fileNames = domainStore.GetFileNames(base.FullName + "*");
					string[] directoryNames = domainStore.GetDirectoryNames(base.FullName + "*");
					FileSystemItem[] array = new FileSystemItem[directoryNames.Length + fileNames.Length];
					for (int i = 0; i < directoryNames.Length; i++)
					{
						array[i] = new IsolatedFolder(base.FullName + directoryNames[i]);
					}
					for (int j = 0; j < fileNames.Length; j++)
					{
						array[directoryNames.Length + j] = new IsolatedFile(base.FullName + fileNames[j]);
					}
					return array;
				}
				catch (IsolatedStorageException inner)
				{
					throw new FileSystemException("The list of child items could not be retrieved from the folder.", this, inner);
				}
			}
		}

		}

	public static class Licenser
	{
		private static ArrayList LicenseList = new ArrayList();

		private static string _licenseKey = string.Empty;

		public static string LicenseKey
		{
			get
			{
				return _licenseKey;
			}
			set
			{
				if (value == null)
				{
					throw new ArgumentNullException("LicenseKey");
				}
				CreateNewLicense(value);
				_licenseKey = value;
			}
		}

		internal static void VerifyLicense()
		{
			if (LicenseList.Count == 0 && LicenseManager.UsageMode != LicenseUsageMode.Designtime)
			{
				CreateNewLicense(string.Empty);
			}
		}

		private static void CreateNewLicense(string licenseKey)
		{
			XceedLicense xceedLicense = FileSystemLicense.CreateLicense(licenseKey, typeof(Licenser));
			lock (LicenseList.SyncRoot)
			{
				foreach (FileSystemLicense license in LicenseList)
				{
					if (license.Product == xceedLicense.Product && license.VersionMajor == xceedLicense.VersionMajor && license.VersionMinor == xceedLicense.VersionMinor)
					{
						LicenseList.Remove(license);
						break;
					}
				}
				LicenseList.Add(xceedLicense);
			}
		}
	}

				throw new ItemAlreadyExistsException(this);
			}
			if (!file.Parent.Exists)
			{
				throw new FileSystemInternalException("An unexpected attempt was made to create an item in a folder that does not exist.", this);
			}
			try
			{
				file.Create();
			}
			catch (Exception except)
			{
				throw PrepareException(except);
			}
		}

		protected internal override void DoDelete(FileSystemEventsSession session)
		{
			MemoryFileHandler file = m_root.GetFile(m_fileName);
			if (!file.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			if ((file.Attributes & FileAttributes.ReadOnly) == FileAttributes.ReadOnly)
			{
				throw new ItemIsReadOnlyException(this);
			}
			try
			{
				file.Delete();
			}
			catch (Exception except)
			{
				throw PrepareException(except);
			}
		}

		protected override void DoRefresh(FileSystemEventsSession session)
		{
		}

		public override bool IsSameAs(FileSystemItem target)
		{
			if (target == null || !(target is MemoryFile))
			{
				return false;
			}
			return base.FullName == target.FullName;
		}

		protected override bool IsPathRooted(string path)
		{
			path = path.Replace(Path.AltDirectorySeparatorChar, Path.DirectorySeparatorChar);
			if (path.StartsWith(Path.DirectorySeparatorChar.ToString()))
			{
				return false;
			}
			string empty = string.Empty;
			try
			{
				empty = Path.GetPathRoot(path);
			}
			catch (PathTooLongException inner)
			{
				throw new FileSystemIOException("The root's full name is too long for item : " + path, null, inner);
			}
			if (empty != null)
			{
				return empty.Length > 0;
			}
			return false;
		}

		protected override Stream DoOpenRead(FileSystemEventsSession session, FileShare share)
		{
			MemoryFileHandler file = m_root.GetFile(m_fileName);
			if (!file.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			try
			{
				return file.OpenRead(share);
			}
			catch (Exception except)
			{
				throw PrepareException(except);
			}
		}

		protected override Stream DoOpenWrite(FileSystemEventsSession session, bool overwrite, FileShare share)
		{
			MemoryFileHandler file = m_root.GetFile(m_fileName);
			if (!file.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			if ((file.Attributes & FileAttributes.ReadOnly) == FileAttributes.ReadOnly)
			{
				throw new ItemIsReadOnlyException(this);
			}
			try
			{
				return file.OpenWrite(overwrite, share);
			}
			catch (Exception except)
			{
				throw PrepareException(except);
			}
		}

		private Exception PrepareException(Exception except)
		{
			if (except is FileSystemException ex)
			{
				if (ex.Item == null)
				{
					ex.Item = this;
				}
				return ex;
			}
			if (except is NotSupportedException)
			{
				return new FileSystemNotSupportedException(this, except);
			}
			if (ExceptionFilter.IsPublicException(except))
			{
				return except;
			}
			return new FileSystemInternalException(this, except);
		}
	}

	internal class MemoryFileHandler : MemoryHandler
	{
		private MemoryStream m_stream;

		private int m_readRef;

		private int m_writeRef;

		public override string FullName => base.Parent.FullName + base.Name;

		public override FileAttributes Attributes
		{
			get
			{
				return base.Attributes;
			}
			set
			{
				if ((value & FileAttributes.ReadOnly) == FileAttributes.ReadOnly && m_writeRef > 0)
				{
					throw new FileSystemException("A MemoryFile cannot be set read-only while it is opened for writing");
				}
				base.Attributes = value;
			}
		}

		public long Size
		{
			get
			{
				if (m_stream == null)
				{
					return 0L;
				}
				return m_stream.Length;
			}
		}

		public MemoryFileHandler(MemoryFolderHandler parent, string name)
			: base(parent, name)
		{
		}

		public MemoryFileHandler(MemoryFolderHandler parent)
			: base(parent, string.Empty)
		{
			ForceName("MF" + GetHashCode() + ".dat");
		}

		~MemoryFileHandler()
		{
			if (m_stream != null)
			{
				m_stream.SetLength(0L);
				m_stream.Close();
				m_stream = null;
			}
		}

		public override void Create()
		{
			m_exists = true;
			m_stream = new MemoryStream();
			DateTime now = DateTime.Now;
			if (!KnowsCreationTime)
			{
				base.CreationDateTime = now;
			}
			if (!KnowsLastAccessTime)
			{
				base.LastAccessDateTime = now;
			}
			if (!KnowsLastWriteTime)
			{
				base.LastWriteDateTime = now;
			}
			KnowsAttributes = false;
			KnowsCreationTime = false;
			KnowsLastAccessTime = false;
			KnowsLastWriteTime = false;
		}

		public override void Delete()
		{
			m_exists = false;
			base.CreationDateTime = DateTime.MinValue;
			base.LastAccessDateTime = DateTime.MinValue;
			base.LastWriteDateTime = DateTime.MinValue;
			if (m_stream != null)
			{
				m_stream.SetLength(0L);
				m_stream.Close();
				m_stream = null;
			}
		}

		public Stream OpenRead(FileShare share)
		{
			if (m_writeRef > 0)
			{
				throw new NotSupportedException("Cannot read and write from a MemoryFile.");
			}
			if (m_readRef > 0)
			{
				throw new NotSupportedException("Cannot open a MemoryFile for reading more than once at a time.");
			}
			m_stream.Seek(0L, SeekOrigin.Begin);
			MemoryHandlerStream memoryHandlerStream = new MemoryHandlerStream(m_stream, allowRead: true, allowWrite: false);
			checked
			{
				m_readRef++;
				memoryHandlerStream.ClosingStream += OnClosingStream;
				return memoryHandlerStream;
			}
		}

		public Stream OpenWrite(bool overwrite, FileShare share)
		{
			if (m_readRef > 0)
			{
				throw new NotSupportedException("Cannot read and write from a MemoryFile.");
			}
			if (m_writeRef > 0)
			{
				throw new NotSupportedException("Cannot open a MemoryFile for writing more than once at a time.");
			}
			if (overwrite)
			{
				m_stream.Close();
				m_stream = new MemoryStream();
			}
			else
			{
				m_stream.Seek(0L, SeekOrigin.End);
			}
			MemoryHandlerStream memoryHandlerStream = new MemoryHandlerStream(m_stream, allowRead: false, allowWrite: true);
			checked
			{
				m_writeRef++;
				memoryHandlerStream.ClosingStream += OnClosingStream;
				return memoryHandlerStream;
			}
		}

		private void OnClosingStream(object sender, EventArgs args)
		{
			DateTime now = DateTime.Now;
			checked
			{
				if (m_readRef > 0)
				{
					m_readRef--;
					if (m_readRef <= 0)
					{
						base.LastAccessDateTime = now;
					}
					return;
				}
				m_writeRef--;
				if (m_writeRef <= 0)
				{
					base.LastAccessDateTime = now;
					base.LastWriteDateTime = now;
				}
			}
		}
	}

	internal class MemoryFolderHandler : MemoryHandler
	{
		private Hashtable m_files = new Hashtable();

		private Hashtable m_folders = new Hashtable();

		public override string FullName => base.Parent.FullName + base.Name + Path.DirectorySeparatorChar;

		public MemoryFolderHandler(MemoryFolderHandler parent, string name)
			: base(parent, name)
		{
		}

		public override void Create()
		{
			m_exists = true;
		}

		public override void Delete()
		{
			m_exists = false;
		}

		public MemoryHandler[] GetChildItems()
		{
			ArrayList arrayList = new ArrayList(checked(m_files.Count + m_folders.Count));
			foreach (DictionaryEntry folder in m_folders)
			{
				MemoryHandler memoryHandler = (MemoryHandler)folder.Value;
				if (memoryHandler.Exists)
				{
					arrayList.Add(memoryHandler);
				}
			}
			foreach (DictionaryEntry file in m_files)
			{
				MemoryHandler memoryHandler2 = (MemoryHandler)file.Value;
				if (memoryHandler2.Exists)
				{
					arrayList.Add(memoryHandler2);
				}
			}
			return (MemoryHandler[])arrayList.ToArray(typeof(MemoryHandler));
		}

		public MemoryFileHandler GetFile()
		{
			MemoryFileHandler memoryFileHandler = new MemoryFileHandler(this);
			if (m_files.Contains(memoryFileHandler.Name))
			{
				throw new FileSystemInternalException("Failed to create a unique MemoryFile.");
			}
			m_files.Add(memoryFileHandler.Name, memoryFileHandler);
			return memoryFileHandler;
		}

		public MemoryFileHandler GetFile(string name)
		{
			if (name == null)
			{
				throw new ArgumentNullException("name");
			}
			checked
			{
				if (name.StartsWith(Path.DirectorySeparatorChar.ToString()))
				{
					name = name.Substring(1, name.Length - 1);
				}
				if (name.Length == 0)
				{
					throw new ArgumentException("The filename is empty.", "name");
				}
				int num = name.IndexOf(Path.DirectorySeparatorChar);
				string text = ((num == -1) ? name : name.Substring(0, num));
				string text2 = ((num == -1) ? string.Empty : name.Substring(num + 1, name.Length - num - 1));
				if (text.Length == 0 && text2.Length == 0)
				{
					throw new ArgumentException("The filename is empty.");
				}
				if (text2.Length == 0)
				{
					lock (m_files.SyncRoot)
					{
						MemoryFileHandler memoryFileHandler = (MemoryFileHandler)m_files[text];
						if (memoryFileHandler == null)
						{
							memoryFileHandler = new MemoryFileHandler(this, text);
							m_files.Add(text, memoryFileHandler);
						}
						return memoryFileHandler;
					}
				}
				MemoryFolderHandler memoryFolderHandler = null;
				lock (m_folders.SyncRoot)
				{
					memoryFolderHandler = (MemoryFolderHandler)m_folders[text];
					if (memoryFolderHandler == null)
					{
						memoryFolderHandler = new MemoryFolderHandler(this, text);
						m_folders.Add(text, memoryFolderHandler);
					}
				}
				return memoryFolderHandler.GetFile(text2);
			}
		}

		public MemoryFolderHandler GetFolder(string name)
		{
			if (name == null)
			{
				throw new ArgumentNullException("name");
			}
			checked
			{
				if (name.EndsWith(Path.DirectorySeparatorChar.ToString()))
				{
					name = name.Substring(0, name.Length - 1);
				}
				if (name.StartsWith(Path.DirectorySeparatorChar.ToString()))
				{
					name = name.Substring(1, name.Length - 1);
				}
				if (name.Length == 0)
				{
					return this;
				}
				int num = name.IndexOf(Path.DirectorySeparatorChar);
				string text = ((num == -1) ? name : name.Substring(0, num));
				string text2 = ((num == -1) ? string.Empty : name.Substring(num + 1, name.Length - num - 1));
				if (text.Length == 0 && text2.Length == 0)
				{
					return this;
				}
				MemoryFolderHandler memoryFolderHandler = null;
				lock (m_folders.SyncRoot)
				{
					memoryFolderHandler = (MemoryFolderHandler)m_folders[text];
					if (memoryFolderHandler == null)
					{
						memoryFolderHandler = new MemoryFolderHandler(this, text);
						m_folders.Add(text, memoryFolderHandler);
					}
				}
				if (text2.Length == 0)
				{
					return memoryFolderHandler;
				}
				return memoryFolderHandler.GetFolder(text2);
			}
		}

		internal void RenameItem(MemoryHandler handler, string newName)
		{
			if (handler is MemoryFolderHandler)
			{
				if (m_folders.Contains(newName) && ((MemoryHandler)m_folders[newName]).Exists)
				{
					throw new ItemAlreadyExistsException("Cannot rename item '" + handler.Name + "' to '" + newName + "'", null);
				}
				if (!m_folders.Contains(handler.Name))
				{
					throw new FileSystemInternalException("Attempting to rename a folder not in its parent's list.");
				}
				m_folders.Remove(handler.Name);
				if (!m_folders.Contains(newName))
				{
					m_folders.Add(newName, handler);
					return;
				}
				m_folders.Remove(newName);
				m_folders.Add(newName, handler);
			}
			else
			{
				if (m_files.Contains(newName) && ((MemoryHandler)m_files[newName]).Exists)
				{
					throw new ItemAlreadyExistsException("Cannot rename item '" + handler.Name + "' to '" + newName + "'", null);
				}
				if (!m_files.Contains(handler.Name))
				{
					throw new FileSystemInternalException("Attempting to rename a file not in its parent's list.");
				}
				m_files.Remove(handler.Name);
				if (!m_files.Contains(newName))
				{
					m_files.Add(newName, handler);
					return;
				}
				m_files.Remove(newName);
				m_files.Add(newName, handler);
			}
		}

	internal class MemoryRootHandler : MemoryFolderHandler
	{
		private static Hashtable mg_worlds = new Hashtable();

		private static long mg_lastCleanupCount;

		private string m_worldName = string.Empty;

		public override string FullName => m_worldName + ":" + Path.DirectorySeparatorChar;

		public override MemoryRootHandler Root => this;

		public string WorldName => m_worldName;

		private MemoryRootHandler(string worldName)
			: base(null, string.Empty)
		{
			if (worldName == null || worldName.Length == 0)
			{
				throw new FileSystemInternalException("Invalid virtual memory drive name.");
			}
			m_worldName = worldName;
			m_exists = true;
		}

		public MemoryRootHandler()
			: base(null, string.Empty)
		{
			int num = GetHashCode();
			m_worldName = "RAM" + num;
			checked
			{
				lock (mg_worlds.SyncRoot)
				{
					WeakReference weakReference = mg_worlds[m_worldName] as WeakReference;
					while (weakReference != null)
					{
						if (weakReference.IsAlive)
						{
							int num2 = ++num;
							m_worldName = "RAM" + num2;
							weakReference = mg_worlds[m_worldName] as WeakReference;
						}
						else
						{
							mg_worlds.Remove(m_worldName);
							weakReference = null;
						}
					}
					if (mg_lastCleanupCount + 50 < mg_worlds.Count)
					{
						CleanWorlds();
					}
					mg_worlds.Add(m_worldName, new WeakReference(this));
				}
				m_exists = true;
			}
		}

		public override void Create()
		{
		}

		public override void Delete()
		{
		}

		public static MemoryRootHandler GetRoot(string worldName)
		{
			lock (mg_worlds.SyncRoot)
			{
				MemoryRootHandler memoryRootHandler = null;
				if (mg_worlds.Contains(worldName))
				{
					if (!(mg_worlds[worldName] is WeakReference weakReference))
					{
						mg_worlds.Remove(worldName);
					}
					else
					{
						if (weakReference.Target is MemoryRootHandler result)
						{
							return result;
						}
						mg_worlds.Remove(worldName);
					}
				}
				if (checked(mg_lastCleanupCount + 50) < mg_worlds.Count)
				{
					CleanWorlds();
				}
				memoryRootHandler = new MemoryRootHandler(worldName);
				mg_worlds.Add(worldName, new WeakReference(memoryRootHandler));
				return memoryRootHandler;
			}
		}

		public static MemoryRootHandler GetRoot(ref string fullName)
		{
			string[] array = fullName.Split(new char[1] { ':' }, 2);
			if (array.Length != 2 || array[0].Length == 0 || array[1].Length == 0)
			{
				throw new ArgumentException("The memory file name must contain a root part and a file name.", "fullName");
			}
			MemoryRootHandler root = GetRoot(array[0]);
			fullName = array[1];
			return root;
		}

		private static void CleanWorlds()
		{
			ArrayList arrayList = new ArrayList(mg_worlds.Count);
			foreach (string key2 in mg_worlds.Keys)
			{
				WeakReference weakReference = mg_worlds[key2] as WeakReference;
				if (!weakReference.IsAlive)
				{
					arrayList.Add(key2);
				}
			}
			foreach (string item in arrayList)
			{
				mg_worlds.Remove(item);
			}
			mg_lastCleanupCount = mg_worlds.Count;
		}
	}

		{

		{
			return target == this;
		}

		protected override bool IsPathRooted(string path)
		{
			return true;
		}

		protected override Stream DoOpenRead(FileSystemEventsSession session, FileShare share)
		{
			if (!base.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			if (!m_stream.CanRead)
			{
				throw new FileSystemNotSupportedException("The attached stream does not support reading.", this);
			}
			if (m_writeRef > 0)
			{
				throw new FileSystemNotSupportedException("Cannot read and write at the same time from a StreamFile.", this);
			}
			if (m_readRef > 0)
			{
				throw new FileSystemNotSupportedException("Cannot open a StreamFile for reading more than once at a time.", this);
			}
			if (m_stream.CanSeek)
			{
				m_stream.Seek(0L, SeekOrigin.Begin);
			}
			MemoryHandlerStream memoryHandlerStream = new MemoryHandlerStream(m_stream, allowRead: true, allowWrite: false);
			checked
			{
				m_readRef++;
				memoryHandlerStream.ClosingStream += OnClosingStream;
				return memoryHandlerStream;
			}
		}

		protected override Stream DoOpenWrite(FileSystemEventsSession session, bool overwrite, FileShare share)
		{
			if (!base.Exists)
			{
				throw new ItemDoesNotExistException(this);
			}
			if ((base.Attributes & FileAttributes.ReadOnly) == FileAttributes.ReadOnly)
			{
				throw new ItemIsReadOnlyException(this);
			}
			if (!m_stream.CanWrite)
			{
				throw new FileSystemNotSupportedException("The attached stream does not support writing.", this);
			}
			if (m_readRef > 0)
			{
				throw new FileSystemNotSupportedException("Cannot read and write at the same time from a StreamFile.", this);
			}
			if (m_writeRef > 0)
			{
				throw new FileSystemNotSupportedException("Cannot open a StreamFile for writing more than once at a time.", this);
			}
			if (m_stream.CanSeek)
			{
				if (overwrite)
				{
					try
					{
						m_stream.Seek(0L, SeekOrigin.Begin);
						m_stream.SetLength(0L);
						m_stream.Flush();
					}
					catch
					{
					}
				}
				else
				{
					m_stream.Seek(0L, SeekOrigin.End);
				}
			}
			MemoryHandlerStream memoryHandlerStream = new MemoryHandlerStream(m_stream, allowRead: false, allowWrite: true);
			checked
			{
				m_writeRef++;
				memoryHandlerStream.ClosingStream += OnClosingStream;
				return memoryHandlerStream;
			}
		}

		private void OnClosingStream(object sender, EventArgs args)
		{
			checked
			{
				if (m_readRef > 0)
				{
					m_readRef--;
				}
				else
				{
					m_writeRef--;
				}
			}
		}
	}

	internal class FileMatching
	{
		public static bool MatchName(string fileName, string wildcards, bool caseSensitive)
		{
			if (fileName == null)
			{
				throw new ArgumentNullException("fileName");
			}
			if (wildcards == null)
			{
				throw new ArgumentNullException("wildCards");
			}
			return checked(RecursiveMatchName(fileName, wildcards, fileName.Length - 1, wildcards.Length - 1, onWildCard: false, caseSensitive));
		}

		internal static bool RecursiveMatchName(string fileName, string wildcards, int posFileName, int posWildcards, bool onWildCard, bool caseSensitive)
		{
			if (fileName == null)
			{
				throw new ArgumentNullException("fileName");
			}
			if (wildcards == null)
			{
				throw new ArgumentNullException("wildCards");
			}
			if (posFileName >= fileName.Length)
			{
				ThrowException.ThrowArgumentOutOfRangeException("posFileName", posFileName, "The position in the file name exceeds the file name's length.");
			}
			if (posWildcards >= wildcards.Length)
			{
				ThrowException.ThrowArgumentOutOfRangeException("posWildcards", posWildcards, "The position in the wildcard exceeds the wildcard's length.");
			}
			if (posWildcards < 0)
			{
				if (posFileName < 0 || onWildCard)
				{
					return true;
				}
				return false;
			}
			if (posFileName < 0)
			{
				return false;
			}
			checked
			{
				switch (wildcards[posWildcards])
				{
				case '*':
					return RecursiveMatchName(fileName, wildcards, posFileName, posWildcards - 1, onWildCard: true, caseSensitive);
				case '?':
					if (RecursiveMatchName(fileName, wildcards, posFileName - 1, posWildcards - 1, onWildCard: false, caseSensitive))
					{
						return true;
					}
					if (onWildCard)
					{
						return RecursiveMatchName(fileName, wildcards, posFileName - 1, posWildcards, onWildCard: true, caseSensitive);
					}
					return false;
				default:
					if (fileName[posFileName] == wildcards[posWildcards] || (!caseSensitive && char.ToUpper(fileName[posFileName]) == char.ToUpper(wildcards[posWildcards])))
					{
						if (RecursiveMatchName(fileName, wildcards, posFileName - 1, posWildcards - 1, onWildCard: false, caseSensitive))
						{
							return true;
						}
						if (onWildCard)
						{
							return RecursiveMatchName(fileName, wildcards, posFileName - 1, posWildcards, onWildCard: true, caseSensitive);
						}
						return false;
					}
					if (onWildCard)
					{
						return RecursiveMatchName(fileName, wildcards, posFileName - 1, posWildcards, onWildCard: true, caseSensitive);
					}
					return false;
				}
			}
		}
	}

	internal class WindowStream : Stream

		private bool m_disposed;

		private Stream m_innerStream;

		private long m_beginOffset;

		private long m_length;

		public override bool CanRead
		{
			get
			{
				CheckIfDisposed();

			}
		}

		public override bool CanSeek
		{
			get
			{
				CheckIfDisposed();
				return m_innerStream.CanSeek;
			}

		{
			get
			{
				CheckIfDisposed();
				return m_innerStream.CanWrite;
			}
		}

		public override long Length
		{
			get
			{
				CheckIfDisposed();
				if (m_length == -1)
				{
					if (m_innerStream.CanSeek)
					{
						return checked(m_innerStream.Length - m_beginOffset);
					}
					throw new NotSupportedException("Cannot get the length of this non-seekable stream.");
				}
				return m_length;
			}
		}

		public override long Position
		{
			get
			{
				CheckIfDisposed();
				CheckInnerStream();
				return checked(m_innerStream.Position - m_beginOffset);
			}
			set
			{
				CheckIfDisposed();
				if (value < 0)
				{
					ThrowException.ThrowArgumentOutOfRangeException("Position", value, "The posision in the stream cannot be set to a negative value.");
				}
				if (m_length != -1 && value > m_length)
				{
					ThrowException.ThrowArgumentOutOfRangeException("Position", value, "The position in the stream cannot be set beyond the end of the stream.");
				}
				m_innerStream.Position = checked(value + m_beginOffset);
			}
		}

		public WindowStream(Stream inner, long offset, long length)
		{
			if (inner == null)
			{
				throw new ArgumentNullException("inner");
			}
			m_innerStream = inner;
			m_beginOffset = offset;
			m_length = length;
			checked
			{
				if (m_innerStream.CanSeek)
				{
					if (m_innerStream.Position < offset)
					{
						m_innerStream.Position = offset;
					}
					else if (length != -1 && m_innerStream.Position > offset + length)
					{
						m_innerStream.Position = offset + length;
					}
				}
				CheckInnerStream();
			}
		}

		public WindowStream(Stream inner, long length)
			: this(inner, inner.Position, length)

		}

		public WindowStream(Stream inner)
			: this(inner, inner.Position, checked(inner.Length - inner.Position))

		}

			CheckIfDisposed();
			m_innerStream.Flush();
		}

		public override long Seek(long offset, SeekOrigin origin)
		{
			CheckIfDisposed();
			CheckInnerStream();
			if (!Enum.IsDefined(typeof(SeekOrigin), origin))
			{
				ThrowException.ThrowArgumentOutOfRangeException("origin", origin, "The seek origin is not a valid SeekOrigin member.");
			}
			long num = 0L;
			checked
			{
				switch (origin)
				{
				case SeekOrigin.Begin:
					num = offset;
					offset += m_beginOffset;
					break;
				case SeekOrigin.Current:
					num = m_innerStream.Position - m_beginOffset + offset;
					break;
				case SeekOrigin.End:
					if (m_length != -1)
					{
						num = m_length + offset;
						offset -= m_innerStream.Length - m_beginOffset - m_length;
					}
					else
					{
						num = m_innerStream.Length + offset;
					}
					break;
				}
				if (num < 0)
				{
					offset -= num;
				}
				else if (m_length != -1 && num > m_length)
				{
					offset -= num - m_length;
				}

			}
		}

		public override void SetLength(long value)
		{
			CheckIfDisposed();
			CheckInnerStream();

			{
				ThrowException.ThrowArgumentOutOfRangeException("count", count, "The count cannot be less than zero.");
			}
			checked
			{

				{
					ThrowException.ThrowArgumentOutOfRangeException("count", count, "The count exceeds the buffer's remaining length after offset.");
				}
				if (m_length != -1 && m_innerStream.Position - m_beginOffset + count > m_length)
				{
					count = (int)(m_length - (m_innerStream.Position - m_beginOffset));
				}
				return m_innerStream.Read(buffer, offset, count);
			}
		}

		public override void Write(byte[] buffer, int offset, int count)

			}
			checked

				if (count > buffer.Length - offset)
				{
					ThrowException.ThrowArgumentOutOfRangeException("count", count, "The count exceeds the buffer's remaining length after offset.");
				}
				long num = ((m_length != -1) ? m_innerStream.Length : 0);
				m_innerStream.Write(buffer, offset, count);
				if (m_length != -1)

					m_length += m_innerStream.Length - num;

			}
