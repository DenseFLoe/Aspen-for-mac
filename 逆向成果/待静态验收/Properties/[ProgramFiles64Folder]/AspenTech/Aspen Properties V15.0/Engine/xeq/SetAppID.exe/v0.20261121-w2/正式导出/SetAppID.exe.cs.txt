		public static void RegisterCapeOpenTypeLib(bool bRegister)
		{
			if (string.IsNullOrEmpty(myFilePath))
			{
				return;
			}
			string empty = string.Empty;
			string text = "CAPE-OPENv1-0-0.tlb";
			ITypeLib typeLib = null;
			IntPtr ppTLibAttr = IntPtr.Zero;
			if (LoadTypeLib(text, out typeLib) == 0)
			{
				if (bRegister)
				{
					try
					{
						RegisterTypeLib(typeLib, text, null);
					}
					catch (COMException ex)
					{
						Console.WriteLine("Error: " + ex.Message);
					}
				}
				else
				{
					try
					{
						typeLib.GetLibAttr(out ppTLibAttr);
						System.Runtime.InteropServices.ComTypes.TYPELIBATTR tYPELIBATTR = (System.Runtime.InteropServices.ComTypes.TYPELIBATTR)Marshal.PtrToStructure(ppTLibAttr, typeof(System.Runtime.InteropServices.ComTypes.TYPELIBATTR));
						Console.WriteLine("{0}.{1}", tYPELIBATTR.wMajorVerNum, tYPELIBATTR.wMinorVerNum);
						UnRegisterTypeLib(ref tYPELIBATTR.guid, tYPELIBATTR.wMajorVerNum, tYPELIBATTR.wMinorVerNum, tYPELIBATTR.lcid, tYPELIBATTR.syskind);
					}
					catch (COMException ex2)
					{
						Console.WriteLine("Error: " + ex2.Message);
					}
					finally
					{
						if (typeLib != null && ppTLibAttr != IntPtr.Zero)
						{
							typeLib.ReleaseTLibAttr(ppTLibAttr);
						}
					}
				}
			}
			string.IsNullOrEmpty(empty);
		}

						result = ((UnRegisterTypeLib(ref tYPELIBATTR.guid, tYPELIBATTR.wMajorVerNum, tYPELIBATTR.wMinorVerNum, tYPELIBATTR.lcid, tYPELIBATTR.syskind) != 0) ? ("Fail to unregister \"" + typeLibFile + "\".") : ("Unregister \"" + typeLibFile + "\" is successfully."));

					}

			}

			{

		private static string Setx64RegistryKey(string regKey, string regValue, int type, string dllOrTLBName)
		{
			string result = string.Empty;
			RegistryKey registryKey = null;
			string value = string.Empty;
			try
			{
				RegistryKey registryKey2 = RegistryKey.OpenBaseKey(RegistryHive.ClassesRoot, RegistryView.Registry64);
				if (registryKey2 != null)
				{
					switch (type)
					{
					case 1:
						registryKey = registryKey2.OpenSubKey(regKey);
						if (registryKey != null)
						{
							object value2 = registryKey.GetValue("");
							if (value2 != null)
							{
								value = Convert.ToString(value2);
							}
							if (!string.IsNullOrEmpty(value))
							{
								regKey = regKey.Replace("win64", "win32");
								registryKey = registryKey2.CreateSubKey(regKey, writable: true);
								registryKey.SetValue("", value);
							}
							registryKey.Close();
						}
						else
						{
							result = "Installation has not created Win64 key for " + dllOrTLBName + ". This indicates the installation has not finished Aspen Plus.";
						}
						break;
					case 2:
						registryKey = registryKey2.CreateSubKey(regKey, writable: true);
						registryKey.SetValue("", regValue);
						registryKey.Close();
						break;
					}
					registryKey2.Close();
				}
			}
			catch (Exception ex)
			{
				result = ex.Message;
			}
			return result;
		}

		private static string SetAppIDKey(string classID, string appID)
		{
			string text = string.Empty;
			string text2 = "AppID\\" + classID;
			string text3 = "CLSID\\" + classID;
			RegistryKey registryKey = null;
			_ = string.Empty;
			try
			{
				RegistryKey registryKey2 = RegistryKey.OpenBaseKey(RegistryHive.ClassesRoot, RegistryView.Registry64);
				if (registryKey2 != null)
				{
					registryKey = registryKey2.OpenSubKey(text2, writable: true);
					if (registryKey == null)
					{
						registryKey = registryKey2.CreateSubKey(text2, writable: true);
					}
					if (registryKey != null)
					{
						registryKey.SetValue("DllSurrogate", "");
						registryKey.Close();
					}
					else
					{
						text = "Create 'DllSurrogate' failed. ";
					}
					registryKey = registryKey2.OpenSubKey(text3, writable: true);
					if (registryKey == null)
					{
						registryKey = registryKey2.CreateSubKey(text3, writable: true);
					}
					if (registryKey == null)
					{
						text = ((!string.IsNullOrEmpty(text)) ? (text + "And create 'AppID' failed. ") : "Create 'AppID' failed. ");
					}
					else
					{
						registryKey.SetValue("AppID", appID);
						registryKey.Close();
					}
					registryKey2.Close();
				}
			}
			catch (Exception ex)
			{
				text = ex.Message;
			}
			return text;
		}

		{
			string text = "regtypelib.exe";
			if (!File.Exists(text) || !File.Exists(typeLibFile))
			{
				return "No file.";
			}
			string result = "";
			_ = string.Empty;
			try
			{
				Process process = new Process();
				process.StartInfo.UseShellExecute = false;
				process.StartInfo.RedirectStandardOutput = true;
				process.StartInfo.CreateNoWindow = true;
				process.StartInfo.FileName = text;
				process.StartInfo.Arguments = "\"" + typeLibFile + "\"";
				process.Start();
				result = process.StandardOutput.ReadToEnd();
				process.WaitForExit();
			}
			catch (Exception ex)
			{
				_ = ex.Message;
			}
			return result;
		}
