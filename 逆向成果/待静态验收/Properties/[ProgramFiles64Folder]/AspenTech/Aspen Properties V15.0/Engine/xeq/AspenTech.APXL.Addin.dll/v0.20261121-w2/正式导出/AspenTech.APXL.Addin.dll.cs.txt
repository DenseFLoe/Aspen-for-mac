		void IDTExtensibility2.OnDisconnection(ext_DisconnectMode RemoveMode, ref Array custom)
		{
			ExcelAppEventHandler.GetInstance().DisconnectEvents();
			Global.Application.Disconnect();
			_addInInstance = null;
			if (Utility.GetAddinKey())
			{
				Utility.DeleteAddinKey();
			}
			SLMHelper.CheckInTokenLicense();
			_ = string.Empty;
			Microsoft.Office.Interop.Excel.Application application = null;
			Workbook workbook = null;
			Sheets sheets = null;
			try
			{
				if (RemoveMode == ext_DisconnectMode.ext_dm_HostShutdown && _appObject != null)
				{
					application = _appObject as Microsoft.Office.Interop.Excel.Application;
					if (application != null)
					{
						workbook = application.ActiveWorkbook;
						if (workbook != null)
						{
							sheets = workbook.Sheets;
							if (sheets != null && sheets.Count > 0)
							{
								for (int i = 1; i <= sheets.Count; i++)
								{
									Worksheet worksheet = (dynamic)sheets.get_Item((object)i);
									if (worksheet != null)
									{
										Marshal.ReleaseComObject(worksheet);
									}
								}
							}
							if (!string.IsNullOrEmpty(PackageRangeManager.GetPropertyPackage(workbook)))
							{
								workbook.Close(Type.Missing, Type.Missing, Type.Missing);
							}
							else if (workbook.Saved)
							{
								workbook.Close(true, Type.Missing, Type.Missing);
							}
						}
						application.Quit();
						Process excelProcess = GetExcelProcess(application);
						excelProcess.Kill();
						excelProcess.WaitForExit();
					}
				}
			}
			catch (Exception ex)
			{
				_ = ex.Message;
			}
			finally
			{
				if (sheets != null)
				{
					Marshal.ReleaseComObject(sheets);
					sheets = null;
				}
				if (workbook != null)
				{
					Marshal.ReleaseComObject(workbook);
					workbook = null;
				}
				if (application != null)
				{
					Marshal.ReleaseComObject(application);
					application = null;
				}
			}
			_appObject = null;
			System.Windows.Forms.Application.DoEvents();
			GC.Collect();
			GC.WaitForPendingFinalizers();
		}

		public PropertiesWorkbookData GetActiveWorkbookData(Workbook wb = null, bool onSave = false)
		{
			if (UnderlyingObject == null)
			{
				return null;
			}
			if (wb == null)
			{
				wb = UnderlyingObject.ActiveWorkbook;
			}
			if (wb == null)
			{
				return null;
			}
			PropertiesWorkbookData propertiesWorkbookData = null;
			if (_workbookDataDictionary.ContainsKey(wb))
			{
				propertiesWorkbookData = _workbookDataDictionary[wb];
			}
			if (propertiesWorkbookData == null)
			{
				propertiesWorkbookData = new PropertiesWorkbookData(wb);
				propertiesWorkbookData.Initialize();
				if (onSave && string.IsNullOrWhiteSpace(propertiesWorkbookData.PackageName))
				{
					return null;
				}
				_workbookDataDictionary.Add(wb, propertiesWorkbookData);
			}
			if (_workbookDataDictionary.ContainsKey(wb))
			{
				propertiesWorkbookData = _workbookDataDictionary[wb];
			}
			StartBackgroundValidationThread();
			return propertiesWorkbookData;
		}

		public void Initialize(bool tryAgainOnFail = false)
		{
			if (!_underlyingObject.UserControl || _isXLAInitialized)
			{
				return;
			}
			_isXLAInitialized = true;
			Utility.EnsureXLALoaded();
			if (!XLAHandShake())
			{
				if (tryAgainOnFail)
				{
					_isXLAInitialized = false;
				}
				Global.DisableRibbon = true;
				RefreshRibbon();
				if (_underlyingObject.Visible && !tryAgainOnFail && Utility.GetAddinKey())
				{
					System.Windows.MessageBox.Show(AspenTech.APXL.Addin.Strings.Strings.PropertiesApp_FailedToLoad);
				}
			}
			else
			{
				Global.DisableRibbon = false;
				RefreshRibbon();
			}
		}

		{
			if (_temporaryDirectoriesToCleanUp == null)
			{
				return;
			}
			foreach (string item in _temporaryDirectoriesToCleanUp)

				try
				{
					Directory.Delete(item, recursive: true);
				}
				catch
				{
				}

			_temporaryDirectoriesToCleanUp.Clear();
		}

		public void WorkbookClosing(Workbook wb)
		{
			UnTrackWorkbook(wb);
			if (_workbookDataDictionary.ContainsKey(wb))
			{
				_workbookDataDictionary.Remove(wb);
				if (_workbookDataDictionary.Count > 0)
				{
					bool flag = false;
					foreach (KeyValuePair<Workbook, PropertiesWorkbookData> item in _workbookDataDictionary)
					{
						if (!string.IsNullOrWhiteSpace(item.Value.PackageName))
						{
							flag = true;
							break;
						}
					}
					if (!flag)
					{
						SLMHelper.CheckInTokenLicense();
					}
				}
			}
			XLAMacroHelper.RunMacro(XLAMacro.ReleaseObjects);
			CleanUpTempFiles();
			RefreshRibbon();
		}

		public bool CheckAspenPropertiesLicense()
		{
			bool result = true;
			try
			{
				if (SLMHelper.GetAspenPropertiesLicenseCookie() == 0 && !string.IsNullOrEmpty(SLMHelper.CheckOutTokenLicense()))
				{
					Global.LicenseError = "Aspen Properties Excel Calculator encounters a fatal licensing error and needs to close.";
					System.Windows.Forms.MessageBox.Show(Global.LicenseError, "Aspen Properties Excel Calculator Denial", MessageBoxButtons.OK, MessageBoxIcon.Exclamation);
					result = false;
					if (Global.Application.UnderlyingObject != null)
					{
						Microsoft.Office.Interop.Excel.Application underlyingObject = Global.Application.UnderlyingObject;
						if (underlyingObject != null)
						{
							try
							{
								object value = Missing.Value;
								underlyingObject.ActiveWorkbook.Close(false, value, value);
								if (underlyingObject.Workbooks.Count <= 0)
								{
									underlyingObject.Quit();
								}
								GC.Collect();
								GC.WaitForPendingFinalizers();
								Marshal.FinalReleaseComObject(underlyingObject.ActiveWorkbook);
								Marshal.FinalReleaseComObject(underlyingObject);
								SLMHelper.CheckInTokenLicense();
							}
							catch (Exception ex)
							{
								Global.LicenseError = ex.Message;
							}
						}
					}
					return result;
				}
			}
			catch (Exception ex2)
			{
				Global.LicenseError = ex2.Message;
			}
			return result;
		}

		public bool CanOpenWorkbook(Workbook wb)
		{
			if (wb == null)
			{
				return false;
			}
			if (_workbookDataDictionary == null)
			{
				_workbookDataDictionary = new Dictionary<Workbook, PropertiesWorkbookData>();
			}
			bool flag = false;
			foreach (KeyValuePair<Workbook, PropertiesWorkbookData> item in _workbookDataDictionary)
			{
				if ((!string.IsNullOrWhiteSpace(item.Value.PackageName) || !item.Value.IsEmpty()) && item.Key != wb && !string.IsNullOrEmpty(PackageRangeManager.GetPropertyPackage(wb)) && !string.IsNullOrWhiteSpace(item.Value.PackageName))
				{
					flag = true;
					break;
				}
			}
			return !flag;
		}

		public void ClosingExistingAPropWorkbook()
		{
			if (Global.Application.UnderlyingObject == null)
			{
				return;
			}
			Microsoft.Office.Interop.Excel.Application underlyingObject = Global.Application.UnderlyingObject;
			if (underlyingObject == null)
			{
				return;
			}
			try
			{
				foreach (Workbook workbook in underlyingObject.Workbooks)
				{
					if (!string.IsNullOrWhiteSpace(PackageRangeManager.GetPropertyPackage(workbook)))
					{
						object value = Missing.Value;
						workbook.Close(false, value, value);
						Marshal.FinalReleaseComObject(workbook);
						SLMHelper.CheckInTokenLicense();
					}
				}
				if (underlyingObject.Workbooks.Count <= 0)
				{
					underlyingObject.Quit();
					Marshal.FinalReleaseComObject(underlyingObject);
					Process excelProcess = GetExcelProcess(underlyingObject);
					excelProcess.Kill();
					excelProcess.WaitForExit();
				}
				GC.Collect();
				GC.WaitForPendingFinalizers();
			}
			catch (Exception ex)
			{
				Global.LicenseError = ex.Message;
			}
		}

		public void WorkbookOpening(Workbook wb)
		{
			if (wb != null && wb.Name.EndsWith(".xla"))
			{
				return;
			}
			Initialize();
			if (!CanOpenWorkbook(wb))
			{
				System.Windows.MessageBox.Show(AspenTech.APXL.Addin.Strings.Strings.PropertiesApp_PackageAlreadyOpen, AspenTech.APXL.Addin.Strings.Strings.PropertiesAddinDenial, MessageBoxButton.OK, MessageBoxImage.Exclamation);
				_ = Missing.Value;
				wb.Saved = true;
				wb.Close(false, Missing.Value, Missing.Value);
				return;
			}
			if (!string.IsNullOrWhiteSpace(PackageRangeManager.GetPropertyPackage(wb)) && !_workbookDataDictionary.ContainsKey(wb))
			{
				PropertiesWorkbookData propertiesWorkbookData = new PropertiesWorkbookData(wb);
				_workbookDataDictionary.Add(wb, propertiesWorkbookData);
				propertiesWorkbookData.Initialize();
			}
			RefreshRibbon();
			StartBackgroundValidationThread();
		}

		private void SLMHelper_TrackingCompleted(TrackingCompletedEventArgs e)
		{
			if (!_workbookDataDictionary.ContainsKey(e.Workbook))
			{
				_workbookDataDictionary.Add(e.Workbook, new PropertiesWorkbookData(e.Workbook)
				{
					Cookie = e.Cookie
				});
			}
			else
			{
				_workbookDataDictionary[e.Workbook].Cookie = e.Cookie;
			}
			StartBackgroundValidationThread();
		}

					}

							propertiesWorkbookData.ClearPackage();
						}

						{
							propertiesWorkbookData.EmbedPackage(text2);
						}
						else
						{
							propertiesWorkbookData.LinkPackage(text2);
						}
					}
				});
				break;
			case "APXL_SelectPackage":
				new SelectPackageDialog().ShowDialog();
				break;
			case "APXL_GetExcelBitness":
				result = ((!Utility.CheckExcelBit(Global.Version)) ? "x86" : "x64");
				break;
			case "APXL_GetCurrentVersion":
				result = Utility.GetCurrentVersion();

				break;
			case "APXL_GetWorkingFolder":

				result = Utility.ReportAddinResult(text);
				break;
			case "APXL_SetForegroundWindow":
				SetForegroundWindow(text);
				break;
			}

		}

		public static void Track(Workbook wb)
		{
			BackgroundWorker backgroundWorker = new BackgroundWorker();
			backgroundWorker.DoWork += delegate(object sender, DoWorkEventArgs args)
			{
				if (int.TryParse(checkoutTrackingLicense(), out var result))
				{
					args.Result = result;
				}
			};
			backgroundWorker.RunWorkerCompleted += delegate(object sender, RunWorkerCompletedEventArgs args)
			{
				if (args.Result != null)
				{
					OnTrackingCompleted(wb, (int)args.Result);
				}
			};
			backgroundWorker.RunWorkerAsync();
		}

				using MemoryStream memoryStream = new MemoryStream();
				using (GZipStream gZipStream = new GZipStream(stream, CompressionMode.Decompress))
				{
					gZipStream.CopyTo(memoryStream);
				}
				array = memoryStream.ToArray();
			}
			using MemoryStream memoryStream2 = new MemoryStream();
			BinaryFormatter binaryFormatter = new BinaryFormatter();
			memoryStream2.Write(array, 0, array.Length);
			memoryStream2.Seek(0L, SeekOrigin.Begin);
			return binaryFormatter.Deserialize(memoryStream2) as EmbeddedData;
		}

		private static IEnumerable<string> Split(string s, int size)
		{
			if (string.IsNullOrWhiteSpace(s) || size <= 0)
			{
				return null;
			}
			List<string> list = new List<string>();
			for (int i = 0; i < s.Length; i += size)
			{
				list.Add(s.Substring(i, Math.Min(size, s.Length - i)));
			}
			return list;
		}

					break;
				case DialogResult.Cancel:

					RemoveWb(Wb);
				}
			}

			Log("Exit -> WorkbookBeforeClose: Cancel = " + Cancel);
		}

		private void UpdateWbList()
		{
			if (_xlApp == null || _xlApp.Workbooks.Count <= 0)
			{
				return;
			}
			int num = 0;
			ArrayList arrayList = new ArrayList();
			foreach (Workbook workbook in _xlApp.Workbooks)
			{
				arrayList.Add(workbook.Name);
			}
			if (arrayList.Count <= 0 || wbList.Count != 0)
			{
				return;
			}
			if (wbList.Count == 0)
			{
				foreach (string item in arrayList)
				{
					wbList.Add(item, num);
				}
				return;
			}
			ArrayList arrayList2 = new ArrayList();
			foreach (string key4 in wbList.Keys)
			{
				if (!arrayList.Contains(key4))
				{
					arrayList2.Add(key4);
				}
			}
			foreach (string item2 in arrayList2)
			{
				wbList.Remove(item2);
			}
			foreach (string item3 in arrayList)
			{
				if (!wbList.Contains(item3))
				{
					wbList.Add(item3, num);
				}
			}
		}

						SetPackage(package);
						return;

					SetPackageRangeFormat(complete: false);

						return;

							return;
						}
						SetPackageRangeText(_propertyPackage);
						SetPackageRangeFormat(complete: true);
					}

						SetPackageRangeText(_propertyPackage);
						SetPackageRangeFormat(complete: true);
					}
				}
			}

	public static class StorageSheet
	{
		private static readonly string _storageSheetName = "AProp_XlStorage";

		public static void ClearStorageSheet(Workbook wb)
		{
			if (wb == null)
			{
				return;
			}
			Worksheet storageSheet = GetStorageSheet(wb);
			if (storageSheet == null)
			{
				return;
			}
			try
			{
				storageSheet.Cells.ClearContents();
				storageSheet.Delete();
				storageSheet = null;
			}
			catch
			{
			}
		}

		public static Worksheet GetStorageSheet(Workbook wb)
		{
			if (wb == null)
			{
				return null;
			}
			Worksheet result = null;
			Sheets worksheets = wb.Worksheets;
			try
			{
				if (worksheets != null)
				{
					long num = worksheets.Count;
					for (int i = 0; i < num; i++)
					{
						if (worksheets.get_Item((object)(i + 1)) is Worksheet worksheet && string.Compare(worksheet.Name, _storageSheetName) == 0)
						{
							result = worksheet;
							break;
						}
					}
				}
			}
			finally
			{
				ReleaseComObject(worksheets);
			}
			return result;
		}

		public static Worksheet CreateStorageSheet(Workbook wb)
		{
			if (wb == null)
			{
				return null;
			}
			Worksheet worksheet = GetStorageSheet(wb);
			if (worksheet == null)
			{
				Sheets sheets = null;
				try
				{
					sheets = wb.Worksheets;
					worksheet = (dynamic)sheets.Add(Type.Missing, Type.Missing, Type.Missing, Type.Missing);
					if (worksheet != null)
					{
						worksheet.Name = _storageSheetName;
						worksheet.Visible = XlSheetVisibility.xlSheetVeryHidden;
					}
				}
				finally
				{
					ReleaseComObject(sheets);
				}
			}
			return worksheet;
		}

		private static void ReleaseComObject(object obj)
		{
			if (obj != null && Marshal.IsComObject(obj))
			{
				Marshal.ReleaseComObject(obj);
				obj = null;
			}
		}
	}

		public void ExportEmbeddedFiles(List<string> files, string defaultPath = "")
		{
			if (FileData.Count == 0 || files == null || files.Count == 0)
			{
				return;
			}
			string empty = string.Empty;
			FolderBrowserDialog folderBrowserDialog = new FolderBrowserDialog();
			folderBrowserDialog.ShowNewFolderButton = true;
			folderBrowserDialog.Description = AspenTech.APXL.Addin.Strings.Strings.EmbeddedData_ChooseDirectory;
			if (!string.IsNullOrWhiteSpace(defaultPath))
			{
				folderBrowserDialog.SelectedPath = defaultPath;
			}
			if (folderBrowserDialog.ShowDialog() != DialogResult.OK)
			{
				return;
			}
			empty = folderBrowserDialog.SelectedPath;
			bool flag = false;
			foreach (KeyValuePair<string, FileInformation> fileDatum in FileData)
			{
				if (files.Contains(fileDatum.Key))
				{
					FileInfo fileInfo = new FileInfo(Path.Combine(empty, fileDatum.Key));
					if (fileInfo != null && fileInfo.Exists)
					{
						flag = true;
						break;
					}
				}
			}
			if (flag)
			{
				string embeddedData_FileConflict = AspenTech.APXL.Addin.Strings.Strings.EmbeddedData_FileConflict2;
				string embeddedData_Warning = AspenTech.APXL.Addin.Strings.Strings.EmbeddedData_Warning;
				if (System.Windows.Forms.MessageBox.Show(embeddedData_FileConflict, embeddedData_Warning, MessageBoxButtons.YesNo, MessageBoxIcon.Exclamation) == DialogResult.No)
				{
					return;
				}
			}
			foreach (KeyValuePair<string, FileInformation> fileDatum2 in FileData)
			{
				if (!files.Contains(fileDatum2.Key))
				{
					continue;
				}
				string path = Path.Combine(empty, fileDatum2.Key);
				byte[] array = (FileData.ContainsKey(fileDatum2.Key) ? FileData[fileDatum2.Key].Bytes : null);
				if (array == null)
				{
					System.Windows.Forms.MessageBox.Show(AspenTech.APXL.Addin.Strings.Strings.EmbeddedData_FileNotFound);
					break;
				}
				using MemoryStream memoryStream = new MemoryStream(array);
				try
				{
					Directory.CreateDirectory(Path.GetDirectoryName(path));
					using (FileStream destination = File.Create(path))
					{
						memoryStream.CopyTo(destination);
					}
					File.SetLastWriteTime(path, fileDatum2.Value.LastWriteTime);
				}
				catch
				{
				}
			}
		}

		public string ExportEmbeddedFiles(bool tempLocation = false, string defaultPath = "")
		{
			if (FileData.Count == 0)
			{
				return string.Empty;
			}
			string empty = string.Empty;
			if (!tempLocation)
			{
				if (!string.IsNullOrWhiteSpace(defaultPath))
				{
					empty = defaultPath;
				}
				else
				{
					FolderBrowserDialog folderBrowserDialog = new FolderBrowserDialog();
					folderBrowserDialog.ShowNewFolderButton = true;

						folderBrowserDialog.SelectedPath = defaultPath;
					}
					if (folderBrowserDialog.ShowDialog() != DialogResult.OK)
					{
						return string.Empty;
					}
					empty = folderBrowserDialog.SelectedPath;
				}
			}
			else
			{
				empty = Path.Combine(Path.GetTempPath(), Guid.NewGuid().ToString("N"));
			}
			bool flag = false;
			foreach (KeyValuePair<string, FileInformation> fileDatum in FileData)
			{
				FileInfo fileInfo = new FileInfo(Path.Combine(empty, fileDatum.Key));
				if (fileInfo != null && fileInfo.Exists)
				{
					flag = true;
					break;
				}
			}
			if (flag)
			{
				string embeddedData_FileConflict = AspenTech.APXL.Addin.Strings.Strings.EmbeddedData_FileConflict;
				string embeddedData_Warning = AspenTech.APXL.Addin.Strings.Strings.EmbeddedData_Warning;
				if (System.Windows.Forms.MessageBox.Show(embeddedData_FileConflict, embeddedData_Warning, MessageBoxButtons.YesNo, MessageBoxIcon.Exclamation) == DialogResult.No)
				{
					return string.Empty;
				}
			}
			string result = string.Empty;
			foreach (KeyValuePair<string, FileInformation> fileDatum2 in FileData)
			{
				string text = Path.Combine(empty, fileDatum2.Key);
				if (string.Compare(fileDatum2.Key, _rootFile, ignoreCase: true) == 0)
				{
					result = text;
				}
				byte[] array = (FileData.ContainsKey(fileDatum2.Key) ? FileData[fileDatum2.Key].Bytes : null);
				if (array == null)
				{
					System.Windows.Forms.MessageBox.Show(AspenTech.APXL.Addin.Strings.Strings.EmbeddedData_FileNotFound);
					return string.Empty;
				}
				using MemoryStream memoryStream = new MemoryStream(array);
				try
				{
					Directory.CreateDirectory(Path.GetDirectoryName(text));
					using (FileStream destination = File.Create(text))
					{
						memoryStream.CopyTo(destination);
					}
					File.SetLastWriteTime(text, fileDatum2.Value.LastWriteTime);
				}
				catch
				{
				}
			}
			return result;
		}

		public void EmbedFile(string filePath, bool isRoot = false)
		{
			if (string.IsNullOrWhiteSpace(filePath))
			{
				return;
			}
			string fileName = Path.GetFileName(filePath);
			if (isRoot)
			{
				_rootFile = fileName;
			}
			else if (string.Compare(fileName, _rootFile, ignoreCase: true) == 0)
			{
				System.Windows.Forms.MessageBox.Show("You cannot embed this file. The current property package has the same file name as the file you are trying to embed.", "Error", MessageBoxButtons.OK, MessageBoxIcon.Hand);
				return;
			}
			byte[] bytes = null;
			FileInfo fileInfo = new FileInfo(filePath);
			if (!fileInfo.Exists)
			{
				return;
			}
			try
			{
				using (FileStream fileStream = new FileStream(filePath, FileMode.Open, FileAccess.Read))
				{
					using MemoryStream memoryStream = new MemoryStream();
					fileStream.CopyTo(memoryStream);
					bytes = memoryStream.ToArray();
				}
				if (FileData.ContainsKey(fileName))
				{
					FileData.Remove(fileName);
				}
				FileData.Add(fileName, new FileInformation
				{
					Bytes = bytes,
					Name = fileInfo.Name,
					Size = fileInfo.Length / 1000,
					LastWriteTime = fileInfo.LastWriteTime
				});
			}
			catch (IOException)
			{
				System.Windows.Forms.MessageBox.Show("The file " + filePath + " cannot be accessed. Please make sure it is not open in another process before attempting to embed.", "Error", MessageBoxButtons.OK, MessageBoxIcon.Hand);
			}
		}
	}

		private void UpdateForm()
		{
			if (_data == null)
			{
				System.Windows.MessageBox.Show(AspenTech.APXL.Addin.Strings.Strings.SelectPackageDialog_NoData);
				Close();
				return;
			}
			try
			{
				_ignoreSelectionChanged = true;
				IsEmbedded = _data.PackageMode == PackageLinkMode.Embedded;
				IsLinked = _data.PackageMode == PackageLinkMode.Linked;
				IsCAPEOpen = _data.PackageMode == PackageLinkMode.CapeOpen;
				CapeOpenPackages = Global.Application.GetCAPEOPENPackages();
				if (IsEmbedded)
				{
					EmbeddedPackage = Path.GetFileName(_data.PackageName);
					LinkedPackage = string.Empty;
					SelectedCapeOpenPackage = string.Empty;
				}
				else if (IsCAPEOpen)
				{
					EmbeddedPackage = string.Empty;
					LinkedPackage = string.Empty;
					if (!string.IsNullOrWhiteSpace(_data.PackageName))
					{
						foreach (string capeOpenPackage in CapeOpenPackages)
						{
							if (string.Compare(_data.PackageName, capeOpenPackage, ignoreCase: true) == 0)
							{
								SelectedCapeOpenPackage = capeOpenPackage;
								break;
							}
						}
					}
					else
					{
						SelectedCapeOpenPackage = _data.PackageName;
					}
				}
				else
				{
					EmbeddedPackage = string.Empty;
					LinkedPackage = _data.PackageName;
					SelectedCapeOpenPackage = string.Empty;
				}
				Status = AspenTech.APXL.Addin.Strings.Strings.SelectPackageDialog_Ready;
			}
			finally
			{
				_ignoreSelectionChanged = false;
			}
		}

		private string PromptForNewFile()
		{
			string result = string.Empty;
			string text = Global.Application.UnderlyingObject.ActiveWorkbook.Path;
			if (string.IsNullOrWhiteSpace(text))
			{
				RegistryKey registryKey = RegistryKey.OpenBaseKey(RegistryHive.CurrentUser, RegistryView.Default);
				if (registryKey != null)
				{
					RegistryKey registryKey2 = registryKey.OpenSubKey("Software\\AspenTech\\Aspen Properties\\41.0\\mmgini", writable: false);
					if (registryKey2 != null)
					{
						object value = registryKey2.GetValue("startupdir");
						if (value != null)
						{
							text = value.ToString();
						}
					}
				}
			}
			Microsoft.Win32.OpenFileDialog openFileDialog = new Microsoft.Win32.OpenFileDialog();
			openFileDialog.InitialDirectory = text;
			openFileDialog.Multiselect = false;
			openFileDialog.Filter = "Aspen Properties Backup Files | *.aprbkp;";
			openFileDialog.FilterIndex = 3;
			openFileDialog.Title = AspenTech.APXL.Addin.Strings.Strings.SelectPackageDialog_Title;
			if (openFileDialog.ShowDialog() == true)
			{
				string fileName = openFileDialog.FileName;
				FileInfo fileInfo = new FileInfo(fileName);
				if (!fileInfo.Exists)
				{
					System.Windows.MessageBox.Show(AspenTech.APXL.Addin.Strings.Strings.SelectPackageDialog_FileNotExist, AspenTech.APXL.Addin.Strings.Strings.SelectPackageDialog_Error, MessageBoxButton.OK, MessageBoxImage.Hand);
				}
				else if (fileInfo.Length == 0L)
				{
					System.Windows.MessageBox.Show(AspenTech.APXL.Addin.Strings.Strings.SelectPackageDialog_FileNoContents, AspenTech.APXL.Addin.Strings.Strings.SelectPackageDialog_Error, MessageBoxButton.OK, MessageBoxImage.Hand);
				}
				else
				{
					result = fileName;
				}
			}
