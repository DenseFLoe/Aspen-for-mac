[Serializable]
[TypeConverter(typeof(CellBordersConverter))]
public class CellBorders : ICloneable
{
	private OGPen left;

	private OGPen top;

	private OGPen right;

	private OGPen bottom;

	public OGPen Left => left;

	public OGPen Top => top;

	public OGPen Right => right;

	public OGPen Bottom => bottom;

	public CellBorders()
	{
		left = new OGPen(BorderDashStyle.Dot, SystemColors.WindowFrame, 1);
		top = new OGPen(BorderDashStyle.Dot, SystemColors.WindowFrame, 1);
		right = new OGPen(BorderDashStyle.Dot, SystemColors.WindowFrame, 1);
		bottom = new OGPen(BorderDashStyle.Dot, SystemColors.WindowFrame, 1);
	}

	public CellBorders(OGPen pn)
	{
		left = pn;
		top = pn;
		right = pn;
		bottom = pn;
	}

	public CellBorders(OGPen LeftRight, OGPen TopBottom)
	{
		left = (right = LeftRight);
		top = (bottom = TopBottom);
	}

	public CellBorders(OGPen l, OGPen t, OGPen r, OGPen b)
	{
		top = t;
		bottom = b;
		left = l;
		right = r;
	}

	public bool IsAllBordersSame()
	{
		if (top == bottom && left == right && top == right)
		{
			return true;
		}
		return false;
	}

	public override int GetHashCode()
	{
		return Left.GetHashCode() ^ Top.GetHashCode() ^ Right.GetHashCode() ^ Bottom.GetHashCode();
	}

	public override bool Equals(object obj)
	{
		bool result = false;
		try
		{
			CellBorders cellBorders = (CellBorders)obj;
			if (cellBorders.Top == Top && cellBorders.Bottom == Bottom && cellBorders.Left == Left && cellBorders.Right == Right)
			{
				result = true;
			}
		}
		catch
		{
		}
		return result;
	}

	public object Clone()
	{
		CellBorders cellBorders = new CellBorders();
		cellBorders.top = top;
		cellBorders.bottom = bottom;
		cellBorders.left = left;
		cellBorders.right = right;
		return cellBorders;
	}

	public override string ToString()
	{
		return "Stingray.CellBorders";
	}
}

[TypeConverter(typeof(BrushConverter))]
[RefreshProperties(RefreshProperties.Repaint)]
public class OGBrush : ICloneable, ISerializable
{
	protected BrushStyle style = BrushStyle.Solid;

	protected Color foreColor = SystemColors.WindowText;

	protected Color backColor = SystemColors.Window;

	protected HatchStyle hatch = HatchStyle.None;

	[RefreshProperties(RefreshProperties.Repaint)]
	[Browsable(true)]
	[Category("OGBrush")]
	public BrushStyle Style => style;

	[Browsable(true)]
	[RefreshProperties(RefreshProperties.Repaint)]
	[Category("OGBrush")]
	public Color ForeColor => foreColor;

	[RefreshProperties(RefreshProperties.Repaint)]
	[Category("OGBrush")]
	[Browsable(true)]
	public Color BackColor => backColor;

	[Category("OGBrush")]
	[Browsable(true)]
	[RefreshProperties(RefreshProperties.Repaint)]
	[Editor(typeof(HatchStyleEditor), typeof(UITypeEditor))]
	public HatchStyle Hatch => hatch;

	[Browsable(false)]
	public string Description
	{
		get
		{
			string result = "Hollow";
			switch (Style)
			{
			case BrushStyle.Solid:
				result = "Solid," + ColorUtils.Color2String(BackColor);
				break;
			case BrushStyle.Hatched:
			{
				TypeConverter converter = TypeDescriptor.GetConverter(typeof(HatchStyle));
				result = "Hatched," + converter.ConvertToString(hatch);
				result = result + "," + ColorUtils.Color2String(BackColor);
				result = result + "," + ColorUtils.Color2String(ForeColor);
				break;
			}
			}
			return result;
		}
	}

	public OGBrush()
	{
	}

	public OGBrush(OGBrush br)
	{
		style = br.style;
		hatch = br.hatch;
		foreColor = br.foreColor;
		backColor = br.backColor;
	}

	public static OGBrush OGHollowBrush()
	{
		OGBrush oGBrush = new OGBrush();
		oGBrush.style = BrushStyle.Hollow;
		return oGBrush;
	}

	public static OGBrush OGSolidBrush(Color cl)
	{
		OGBrush oGBrush = new OGBrush();
		oGBrush.style = BrushStyle.Solid;
		oGBrush.foreColor = cl;
		return oGBrush;
	}

	public static OGBrush OGHatchedBrush(Color cl, HatchStyle ht)
	{
		OGBrush oGBrush = new OGBrush();
		oGBrush.style = BrushStyle.Hatched;
		oGBrush.foreColor = cl;
		oGBrush.hatch = ht;
		return oGBrush;
	}

	public static OGBrush OGHatchedBrush(Color fColor, Color bColor, HatchStyle ht)
	{
		OGBrush oGBrush = OGHatchedBrush(fColor, ht);
		oGBrush.backColor = bColor;
		return oGBrush;
	}

	public Brush MakeGdiPlusBrush()
	{
		if (Style == BrushStyle.Hollow)
		{
			return null;
		}
		if (Style == BrushStyle.Solid)
		{
			return new SolidBrush(BackColor);
		}
		return new HatchBrush((System.Drawing.Drawing2D.HatchStyle)(hatch - 1), ForeColor, BackColor);
	}

	public int GetBrushKey()
	{
		return GetHashCode();
	}

	public override bool Equals(object obj)
	{
		bool result = false;
		try
		{
			OGBrush oGBrush = (OGBrush)obj;
			if (oGBrush != null && oGBrush.BackColor == BackColor && oGBrush.ForeColor == ForeColor && oGBrush.Hatch == Hatch && oGBrush.Style == Style)
			{
				result = true;
			}
		}
		catch
		{
		}
		return result;
	}

	public override int GetHashCode()
	{
		return Style.GetHashCode() ^ Hatch.GetHashCode() ^ ForeColor.GetHashCode() ^ BackColor.GetHashCode();
	}

	public OGBrush(SerializationInfo info, StreamingContext context)
	{
		style = (BrushStyle)info.GetValue("OGBrush_Style", typeof(BrushStyle));
		if (style != BrushStyle.Hollow)
		{
			backColor = (Color)info.GetValue("OGBrush_BackColor", typeof(Color));
			if (style != BrushStyle.Solid)
			{
				foreColor = (Color)info.GetValue("OGBrush_ForeColor", typeof(Color));
				hatch = (HatchStyle)info.GetValue("OGBrush_Hatch", typeof(HatchStyle));
			}
		}
	}

	public void GetObjectData(SerializationInfo info, StreamingContext context)
	{
		info.AddValue("OGBrush_Style", Style);
		if (Style != BrushStyle.Hollow)
		{
			info.AddValue("OGBrush_BackColor", BackColor);
			if (Style != BrushStyle.Solid)
			{
				info.AddValue("OGBrush_ForeColor", ForeColor);
				info.AddValue("OGBrush_Hatch", Hatch);
			}
		}
	}

	public object Clone()
	{
		return new OGBrush(this);
	}

	public override string ToString()
	{
		string result = "";
		switch (Style)
		{
		case BrushStyle.Solid:
			result = "Solid," + ColorUtils.Color2String(BackColor);
			break;
		case BrushStyle.Hatched:
		{
			TypeConverter converter = TypeDescriptor.GetConverter(typeof(HatchStyle));
			result = converter.ConvertToString(hatch);
			result = result + "," + ColorUtils.Color2String(ForeColor);
			break;
		}
		}
		return result;
	}
}
[TypeConverter(typeof(FontConverter))]
[RefreshProperties(RefreshProperties.Repaint)]
public class OGFont : ICloneable, ISerializable
{
	protected bool includeItalic = false;

	protected bool includeStrikeOut = false;

	protected bool includeUnderLine = false;

	protected bool includeBold = false;

	protected bool includeOrientation = false;

	protected bool includeFaceName = false;

	protected bool includeSize = false;

	private BitVector32 fontBits;

	[Browsable(false)]
	public bool IncludeItalic
	{
		get
		{
			return includeItalic;
		}
		set
		{
			includeItalic = value;
		}
	}

	[Browsable(false)]
	public bool IncludeStrikeOut
	{
		get
		{
			return includeStrikeOut;
		}
		set
		{
			includeStrikeOut = value;
		}
	}

	[Browsable(false)]
	public bool IncludeUnderLine
	{
		get
		{
			return includeUnderLine;
		}
		set
		{
			includeUnderLine = value;
		}
	}

	[Browsable(false)]
	public bool IncludeBold
	{
		get
		{
			return includeBold;
		}
		set
		{
			includeBold = value;
		}
	}

	[Browsable(false)]
	public bool IncludeOrientation
	{
		get
		{
			return includeOrientation;
		}
		set
		{
			includeOrientation = value;
		}
	}

	[Browsable(false)]
	public bool IncludeFaceName
	{
		get
		{
			return includeFaceName;
		}
		set
		{
			includeFaceName = value;
		}
	}

	[Browsable(false)]
	public bool IncludeSize
	{
		get
		{
			return includeSize;
		}
		set
		{
			includeSize = value;
		}
	}

	[Category("OGFont")]
	[RefreshProperties(RefreshProperties.Repaint)]
	public bool Bold => Convert.ToBoolean(fontBits[StyleInfo.sBold]);

	[RefreshProperties(RefreshProperties.Repaint)]
	[Category("OGFont")]
	public bool Italic => Convert.ToBoolean(fontBits[StyleInfo.sItalic]);

	[Category("OGFont")]
	[RefreshProperties(RefreshProperties.Repaint)]
	public bool UnderLine => Convert.ToBoolean(fontBits[StyleInfo.sUnderLine]);

	[RefreshProperties(RefreshProperties.Repaint)]
	[Category("OGFont")]
	public bool StrikeOut => Convert.ToBoolean(fontBits[StyleInfo.sStrikeOut]);

	[RefreshProperties(RefreshProperties.Repaint)]
	[Category("OGFont")]
	public int Orientation => fontBits[StyleInfo.sOrientation];

	[Editor(typeof(FontNameEditor), typeof(UITypeEditor))]
	[RefreshProperties(RefreshProperties.Repaint)]
	[Category("OGFont")]
	public string FaceName
	{
		get
		{
			int index = fontBits[StyleInfo.sFnIndex];
			return StyleInfo.GetFaceName(index);
		}
	}

	[RefreshProperties(RefreshProperties.Repaint)]
	[Category("OGFont")]
	public int Size => fontBits[StyleInfo.sSize];

	[Browsable(false)]
	public string Description => ToString();

	[Browsable(false)]
	public bool IsEmpty => GetFontKey() == new OGFont().GetFontKey();

	public OGFont()
	{
		ref BitVector32 reference = ref fontBits;
		reference = new BitVector32(0);
		base..ctor();
	}

	public OGFont(string name, int sz)
	{
		ref BitVector32 reference = ref fontBits;
		reference = new BitVector32(0);
		base..ctor();
		SetFaceName(name);
		SetSize(sz);
	}

	public OGFont(string name, int sz, int ang)
	{
		ref BitVector32 reference = ref fontBits;
		reference = new BitVector32(0);
		base..ctor();
		SetFaceName(name);
		SetSize(sz);
		SetOrientation(ang);
	}

	public OGFont(string name, int sz, int ang, bool _bold, bool _italic, bool _underline, bool _strikeout)
	{
		ref BitVector32 reference = ref fontBits;
		reference = new BitVector32(0);
		base..ctor();
		SetFaceName(name);
		SetSize(sz);
		SetOrientation(ang);
		SetBold(_bold);
		SetItalic(_italic);
		SetUnderLine(_underline);
		SetStrikeOut(_strikeout);
	}

	public OGFont(Font ft)
	{
		ref BitVector32 reference = ref fontBits;
		reference = new BitVector32(0);
		base..ctor();
		SetBold(ft.Bold);
		SetItalic(ft.Italic);
		SetUnderLine(ft.Underline);
		SetStrikeOut(ft.Strikeout);
		SetSize((int)ft.SizeInPoints);
		SetFaceName(ft.Name);
	}

	public OGFont(OGFont ft)
	{
		ref BitVector32 reference = ref fontBits;
		reference = new BitVector32(0);
		base..ctor();
		if (ft.IncludeBold)
		{
			SetBold(ft.Bold);
		}
		if (ft.IncludeItalic)
		{
			SetItalic(ft.Italic);
		}
		if (ft.IncludeUnderLine)
		{
			SetUnderLine(ft.UnderLine);
		}
		if (ft.IncludeStrikeOut)
		{
			SetStrikeOut(ft.StrikeOut);
		}
		if (ft.IncludeSize)
		{
			SetSize(ft.Size);
		}
		if (ft.IncludeFaceName)
		{
			SetFaceName(ft.FaceName);
		}
		if (ft.IncludeOrientation)
		{
			SetOrientation(ft.Orientation);
		}
	}

	public bool IsSubSet(OGFont ft)
	{
		bool flag = true;
		if (flag && ft.IncludeBold)
		{
			flag &= IncludeBold && Bold == ft.Bold;
		}
		if (flag && ft.IncludeItalic)
		{
			flag &= IncludeItalic && Italic == ft.Italic;
		}
		if (flag && ft.IncludeUnderLine)
		{
			flag &= IncludeUnderLine && UnderLine == ft.UnderLine;
		}
		if (flag && ft.IncludeStrikeOut)
		{
			flag &= IncludeStrikeOut && StrikeOut == ft.StrikeOut;
		}
		if (flag && ft.IncludeOrientation)
		{
			flag &= IncludeOrientation && Orientation == ft.Orientation;
		}
		if (flag && ft.IncludeSize)
		{
			flag &= IncludeSize && Size == ft.Size;
		}
		if (flag && ft.IncludeFaceName)
		{
			flag &= IncludeFaceName && FaceName == ft.FaceName;
		}
		return flag;
	}

	public Font GetGdiPlusFont()
	{
		FontStyle fontStyle = FontStyle.Regular;
		if (Bold)
		{
			fontStyle |= FontStyle.Bold;
		}
		if (Italic)
		{
			fontStyle |= FontStyle.Italic;
		}
		if (UnderLine)
		{
			fontStyle |= FontStyle.Underline;
		}
		if (StrikeOut)
		{
			fontStyle |= FontStyle.Strikeout;
		}
		return new Font(FaceName, Size, fontStyle);
	}

	public static OGFont DefaultFont()
	{
		return new OGFont("Arial", 8, 0, _bold: false, _italic: false, _underline: false, _strikeout: false);
	}

	public void ChangeFont(OGFont ft, ModifyType mt)
	{
		int num;
		switch (mt)
		{
		case ModifyType.Exclude:
			if (ft.IncludeBold)
			{
				IncludeBold = false;
			}
			if (ft.IncludeItalic)
			{
				IncludeItalic = false;
			}
			if (ft.IncludeUnderLine)
			{
				IncludeUnderLine = false;
			}
			if (ft.IncludeStrikeOut)
			{
				IncludeStrikeOut = false;
			}
			if (ft.IncludeSize)
			{
				IncludeSize = false;
			}
			if (ft.IncludeOrientation)
			{
				IncludeOrientation = false;
			}
			if (ft.IncludeFaceName)
			{
				IncludeFaceName = false;
			}
			return;
		case ModifyType.Copy:
			SetBold(ft.Bold);
			SetItalic(ft.Italic);
			SetUnderLine(ft.UnderLine);
			SetStrikeOut(ft.StrikeOut);
			SetSize(ft.Size);
			SetOrientation(ft.Orientation);
			SetFaceName(ft.FaceName);
			return;
		default:
			num = ((mt == ModifyType.Copy) ? 1 : 0);
			break;
		case ModifyType.Override:
			num = 1;
			break;
		}
		bool flag = (byte)num != 0;
		if ((flag || !IncludeBold) && ft.IncludeBold)
		{
			SetBold(ft.Bold);
		}
		if ((flag || !IncludeItalic) && ft.IncludeItalic)
		{
			SetItalic(ft.Italic);
		}
		if ((flag || !IncludeUnderLine) && ft.IncludeUnderLine)
		{
			SetUnderLine(ft.UnderLine);
		}
		if ((flag || !IncludeStrikeOut) && ft.IncludeStrikeOut)
		{
			SetStrikeOut(ft.StrikeOut);
		}
		if ((flag || !IncludeSize) && ft.IncludeSize)
		{
			SetSize(ft.Size);
		}
		if ((flag || !IncludeOrientation) && ft.IncludeOrientation)
		{
			SetOrientation(ft.Orientation);
		}
		if ((flag || !IncludeFaceName) && ft.IncludeFaceName)
		{
			SetFaceName(ft.FaceName);
		}
	}

	public int GetFontKey()
	{
		return GetHashCode();
	}

	public override int GetHashCode()
	{
		return fontBits.GetHashCode();
	}

	public OGFont(SerializationInfo info, StreamingContext context)
	{
		ref BitVector32 reference = ref fontBits;
		reference = new BitVector32(0);
		base..ctor();
		SetFaceName((string)info.GetValue("OGFont_Name", typeof(string)));
		SetSize((int)info.GetValue("OGFont_Size", typeof(int)));
		SetOrientation((int)info.GetValue("OGFont_Orientation", typeof(int)));
		SetBold((bool)info.GetValue("OGFont_Bold", typeof(bool)));
		SetItalic((bool)info.GetValue("OGFont_italic", typeof(bool)));
		SetUnderLine((bool)info.GetValue("OGFont_UnderLine", typeof(bool)));
		SetStrikeOut((bool)info.GetValue("OGFont_StrikeOut", typeof(bool)));
	}

	public void GetObjectData(SerializationInfo info, StreamingContext context)
	{
		info.AddValue("OGFont_Name", FaceName);
		info.AddValue("OGFont_Size", Size);
		info.AddValue("OGFont_Orientation", Orientation);
		info.AddValue("OGFont_Bold", Bold);
		info.AddValue("OGFont_italic", Italic);
		info.AddValue("OGFont_UnderLine", UnderLine);
		info.AddValue("OGFont_StrikeOut", StrikeOut);
	}

	public object Clone()
	{
		return new OGFont(this);
	}

	public override string ToString()
	{
		string faceName = FaceName;
		return faceName + ", " + Size;
	}

	protected void SetBold(bool value)
	{
		fontBits[StyleInfo.sBold] = Convert.ToInt32(value);
		IncludeBold = true;
	}

	protected void SetItalic(bool value)
	{
		fontBits[StyleInfo.sItalic] = Convert.ToInt32(value);
		IncludeItalic = true;
	}

	protected void SetUnderLine(bool value)
	{
		fontBits[StyleInfo.sUnderLine] = Convert.ToInt32(value);
		IncludeUnderLine = true;
	}

	protected void SetStrikeOut(bool value)
	{
		fontBits[StyleInfo.sStrikeOut] = Convert.ToInt32(value);
		IncludeStrikeOut = true;
	}

	protected void SetOrientation(int value)
	{
		int num = ((value > 360) ? (value / 360) : 0);
		fontBits[StyleInfo.sOrientation] = value - num * 360;
		IncludeOrientation = true;
	}

	protected void SetFaceName(string value)
	{
		if (value != null && !(value == ""))
		{
			int faceNameIndex = StyleInfo.GetFaceNameIndex(value);
			if (faceNameIndex != -1)
			{
				fontBits[StyleInfo.sFnIndex] = faceNameIndex;
				IncludeFaceName = true;
			}
		}
	}

	protected void SetSize(int value)
	{
		if (value > 0 && value <= 1023)
		{
			fontBits[StyleInfo.sSize] = value;
			IncludeSize = true;
		}
	}
}
public enum BorderDashStyle
{
	None,
	Solid,
	Dash,
	Dot,
	DashDot,
	DashDotDot
}
[TypeConverter(typeof(PenConverter))]
public class OGPen : ICloneable, ISerializable
{
	private byte weight = 1;

	private Color borderColor = SystemColors.WindowFrame;

	private byte dashStyle = 0;

	[RefreshProperties(RefreshProperties.Repaint)]
	[Category("OGPen")]
	public byte Weight
	{
		get
		{
			return weight;
		}
		set
		{
			if (value <= byte.MaxValue)
			{
				weight = value;
				if (weight > 1 && DashStyle == BorderDashStyle.None)
				{
					DashStyle = BorderDashStyle.Solid;
				}
			}
		}
	}

	[Category("OGPen")]
	[RefreshProperties(RefreshProperties.Repaint)]
	public Color BorderColor
	{
		get
		{
			return borderColor;
		}
		set
		{
			borderColor = value;
			if (DashStyle == BorderDashStyle.None)
			{
				DashStyle = BorderDashStyle.Solid;
			}
		}
	}

	[RefreshProperties(RefreshProperties.Repaint)]
	[Editor(typeof(BorderDashStyleEditor), typeof(UITypeEditor))]
	[Category("OGPen")]
	public BorderDashStyle DashStyle
	{
		get
		{
			return dashStyle switch
			{
				1 => BorderDashStyle.Solid, 
				2 => BorderDashStyle.Dash, 
				3 => BorderDashStyle.Dot, 
				4 => BorderDashStyle.DashDot, 
				5 => BorderDashStyle.DashDotDot, 
				_ => BorderDashStyle.None, 
			};
		}
		set
		{
			switch (value)
			{
			default:
				dashStyle = 0;
				break;
			case BorderDashStyle.Solid:
				dashStyle = 1;

[TypeConverter(typeof(RangeConverter))]
public class Range : ICloneable, ISerializable
{
	protected int left;

	protected int top;

	protected int right;

	protected int bottom;

	protected RangeType rangeType;

	protected bool bValid = false;

	public int Left => left;

	public int Top => top;

	public int Right => right;

	public int Bottom => bottom;

	public RangeType RangeType => rangeType;

	public bool IsValid => bValid;

	public int Width => Right - Left + 1;

	public int Height => Bottom - Top + 1;

	public bool IsRows => rangeType == RangeType.rtRows;

	public bool IsCols => rangeType == RangeType.rtCols;

	public bool IsCells => rangeType == RangeType.rtCells;

	public bool IsTable => rangeType == RangeType.rtTable;

	protected Range()
	{
		left = (top = (right = (bottom = 0)));
		bValid = false;
	}

	public static Range InvalidRange()
	{
		return new Range();
	}

	public static Range Row(int nRow)
	{
		return Rows(nRow, nRow);
	}

	public static Range Rows(int nFrom, int nTo)
	{
		Range range = new Range();
		range.rangeType = RangeType.rtRows;
		range.top = Math.Min(nFrom, nTo);
		range.bottom = Math.Max(nFrom, nTo);
		range.left = (range.right = 0);
		range.bValid = true;
		return range;
	}

	public static Range Col(int nCol)
	{
		return Cols(nCol, nCol);
	}

	public static Range Cols(int nFrom, int nTo)
	{
		Range range = new Range();
		range.rangeType = RangeType.rtCols;
		range.left = Math.Min(nFrom, nTo);
		range.right = Math.Max(nFrom, nTo);
		range.top = (range.bottom = 0);
		range.bValid = true;
		return range;
	}

	public static Range Cell(int nRow, int nCol)
	{
		return Cells(nRow, nCol, nRow, nCol);
	}

	public static Range Cells(int nRow1, int nCol1, int nRow2, int nCol2)
	{
		Range range = new Range();
		range.rangeType = RangeType.rtCells;
		range.top = Math.Min(nRow1, nRow2);
		range.left = Math.Min(nCol1, nCol2);
		range.bottom = Math.Max(nRow1, nRow2);
		range.right = Math.Max(nCol1, nCol2);
		range.bValid = true;
		return range;
	}

	public static Range Table()
	{
		Range range = new Range();
		range.rangeType = RangeType.rtTable;
		range.bValid = true;
		return range;
	}

	public static Range FromLTRB(int nLeft, int nTop, int nRight, int nBottom)
	{
		return Cells(nTop, nLeft, nBottom, nRight);
	}

	public bool GetFirstCell(ref int nRow, ref int nCol)
	{
		if (!IsValid)
		{
			return false;
		}
		nRow = Top;
		nCol = Left;
		return true;
	}

	public bool GetNextCell(ref int nRow, ref int nCol)
	{
		return GetNextCell(ref nRow, ref nCol, bSortByRow: true);
	}

	public bool GetNextCell(ref int nRow, ref int nCol, bool bSortByRow)
	{
		if (!IsValid)
		{
			return false;
		}
		if (bSortByRow && ++nCol > Right)
		{
			if (++nRow > Bottom)
			{
				nRow = (nCol = 0);
			}
			else
			{
				nCol = Left;
			}
		}
		else if (!bSortByRow && ++nRow > Bottom)
		{
			if (++nCol > Right)
			{
				nRow = (nCol = 0);
			}
			else
			{
				nRow = Top;
			}
		}
		if (nRow <= 0)
		{
			return nCol > 0;
		}
		return true;
	}

	public void ExpandRange(int nFirstRow, int nFirstCol, int nRowCount, int nColCount)
	{
		if (IsValid)
		{
			switch (rangeType)
			{
			case RangeType.rtCols:
				top = nFirstRow;
				bottom = nRowCount;
				break;
			case RangeType.rtRows:
				left = nFirstCol;
				right = nColCount;
				break;
			case RangeType.rtTable:
				top = nFirstRow;
				bottom = nRowCount;
				left = nFirstCol;
				right = nColCount;
				break;
			}
			rangeType = RangeType.rtCells;
		}
	}

	public bool IsCellInRange(int nRow, int nCol)
	{
		if (!IsValid)
		{
			return false;
		}
		if ((!IsCols && (Top > nRow || nRow > Bottom)) || (!IsRows && (Left > nCol || nCol > Right)))
		{
			return IsTable;
		}
		return true;
	}

	public void InsertRows(int nRow, int nCount)
	{
		if (IsValid)
		{
			if (Top >= nRow)
			{
				top += nCount;
			}
			if (Bottom >= nRow)
			{
				bottom += nCount;
			}
		}
	}

	public void InsertCols(int nCol, int nCount)
	{
		if (IsValid)
		{
			if (Left >= nCol)
			{
				left += nCount;
			}
			if (Right >= nCol)
			{
				right += nCount;
			}
		}
	}

	public Range IntersectRange(Range r1)
	{
		return IntersectRange(this, r1);
	}

	public static Range IntersectRange(Range r1, Range r2)
	{
		if (!r1.IsValid || !r2.IsValid)
		{
			return InvalidRange();
		}
		Range range = new Range();
		if (r1.IsTable || r2.IsTable)
		{
			range = (Range)(r1.IsTable ? r2 : r1).Clone();
			range.bValid = true;
		}
		else if ((r1.IsRows && r2.IsCols) || (r1.IsCols && r2.IsRows))
		{
			range.left = (r1.IsCols ? r1 : r2).Left;
			range.right = (r1.IsCols ? r1 : r2).Right;
			range.top = (r1.IsCols ? r2 : r1).Top;
			range.bottom = (r1.IsCols ? r2 : r1).Bottom;
			range.rangeType = RangeType.rtCells;
			range.bValid = true;
		}
		else if (r1.IsCols || r2.IsCols)
		{
			range.left = Math.Max(r1.Left, r2.Left);
			range.right = Math.Min(r1.Right, r2.Right);
			range.top = (r1.IsCols ? r2 : r1).Top;
			range.bottom = (r1.IsCols ? r2 : r1).Bottom;
			range.rangeType = (r1.IsCols ? r2 : r1).RangeType;
			range.bValid = range.Left <= range.Right;
		}
		else if (r1.IsRows || r2.IsRows)
		{
			range.left = (r1.IsRows ? r2 : r1).Left;
			range.right = (r1.IsRows ? r2 : r1).Right;
			range.top = Math.Max(r2.Top, r1.Top);
			range.bottom = Math.Min(r2.Bottom, r1.Bottom);
			range.rangeType = (r1.IsRows ? r2 : r1).RangeType;
			range.bValid = range.Top <= range.Bottom;
		}
		else
		{
			range.left = Math.Max(r1.Left, r2.Left);
			range.right = Math.Min(r1.Right, r2.Right);
			range.top = Math.Max(r1.Top, r2.Top);
			range.bottom = Math.Min(r1.Bottom, r2.Bottom);
			range.bValid = range.Left <= range.Right && range.Top <= range.Bottom;
		}
		return range;
	}

	public Range UnionRange(Range r1)
	{
		return UnionRange(this, r1);
	}

	public static Range UnionRange(Range r1, Range r2)
	{
		if (!r1.IsValid || !r2.IsValid)
		{
			return InvalidRange();
		}
		Range range = new Range();
		if (r1.IsTable || r2.IsTable || (r1.IsCols && r2.IsRows) || (r1.IsRows && r2.IsCols))
		{
			return Table();
		}
		if (r1.IsCols || r2.IsCols)
		{
			return Cols(Math.Min(r1.Left, r2.Left), Math.Max(r1.Right, r2.Right));
		}
		if (r1.IsRows || r2.IsRows)
		{
			return Rows(Math.Min(r1.Top, r2.Top), Math.Max(r1.Bottom, r2.Bottom));
		}
		return Cells(Math.Min(r1.Top, r2.Top), Math.Min(r1.Left, r2.Left), Math.Max(r1.Bottom, r2.Bottom), Math.Max(r1.Right, r2.Right));
	}

	public override bool Equals(object obj)
	{
		bool result = false;
		try
		{
			Range range = (Range)obj;
			if (range.Left == Left && range.Top == Top && range.Right == Right && range.Bottom == Bottom && range.RangeType == RangeType)
			{
				result = true;
			}
		}
		catch
		{
		}
		return result;
	}

	public override int GetHashCode()
	{
		string value = Left.ToString() + Top + Right + Bottom;
		return Convert.ToInt32(value);
	}

	public object Clone()
	{
		Range range = FromLTRB(Left, Top, Right, Bottom);
		range.rangeType = RangeType;
		range.bValid = bValid;
		return range;
	}

	public override string ToString()
	{
		string result = "R" + Top + "C" + Left + ":R" + Bottom + "C" + Right;
		if (IsValid)
		{
			return result;
		}
		return "";
	}

	public Range(SerializationInfo info, StreamingContext context)
	{
		left = (int)info.GetValue("Range_Left", typeof(int));
		top = (int)info.GetValue("Range_Top", typeof(int));
		right = (int)info.GetValue("Range_Right", typeof(int));
		bottom = (int)info.GetValue("Range_Bottom", typeof(int));
		bValid = (bool)info.GetValue("Range_Valid", typeof(bool));
		rangeType = (RangeType)info.GetValue("Range_RANGETYPE", typeof(RangeType));
	}

	public void GetObjectData(SerializationInfo info, StreamingContext context)
	{
		info.AddValue("Range_LEFT", Left);
		info.AddValue("Range_TOP", Top);
		info.AddValue("Range_RIGHT", Right);
		info.AddValue("Range_BOTTOM", Bottom);
		info.AddValue("Range_Valid", bValid);
		info.AddValue("Range_TYPE", RangeType);
	}

	~Range()
	{
		base.Finalize();
	}
}

public abstract class DropDownListTypeEditor : UITypeEditor
{
	private IWindowsFormsEditorService fEdSvc = null;

	public DropDownListTypeEditor()
	{
	}

	public override UITypeEditorEditStyle GetEditStyle(ITypeDescriptorContext pContext)
	{
		if (pContext != null && pContext.Instance != null)
		{
			return UITypeEditorEditStyle.DropDown;
		}
		return base.GetEditStyle(pContext);
	}

	public override object EditValue(ITypeDescriptorContext pContext, IServiceProvider pProvider, object pValue)
	{
		if (pContext != null && pContext.Instance != null && pProvider != null)
		{
			try
			{
				fEdSvc = (IWindowsFormsEditorService)pProvider.GetService(typeof(IWindowsFormsEditorService));
				ListBox listBox = new ListBox();
				listBox.Click += ListBox_Click;
				CustomizeListBox(listBox);
				FillInList(pContext, pProvider, listBox);
				listBox.SelectedItem = pValue;
				fEdSvc.DropDownControl(listBox);
				return listBox.SelectedItem;
			}
			finally
			{
				fEdSvc = null;
			}
		}
		return pValue;
	}

	protected virtual void CustomizeListBox(ListBox lb)
	{
	}

	protected void ListBox_Click(object pSender, EventArgs pArgs)
	{
		fEdSvc.CloseDropDown();
	}

	protected abstract void FillInList(ITypeDescriptorContext pContext, IServiceProvider pProvider, ListBox pListBox);
}
public class FontNameEditor : DropDownListTypeEditor
{
	protected override void CustomizeListBox(ListBox lb)
	{
		lb.Sorted = true;
		lb.HorizontalScrollbar = true;
		lb.DrawMode = DrawMode.OwnerDrawVariable;
		lb.Font = new Font("Arial", 8.25f);
		lb.Width += 100;
		lb.Height += 100;
		lb.BorderStyle = BorderStyle.FixedSingle;
		lb.MeasureItem += OnMeasureItem;
		lb.DrawItem += OnDrawItem;
	}

	private void OnMeasureItem(object sender, MeasureItemEventArgs e)
	{
		ListBox listBox = (ListBox)sender;
		Font font = new Font(listBox.Font.Name, 13.75f);
		e.ItemHeight = e.Graphics.MeasureString("abcd", font).ToSize().Height;
		font.Dispose();
	}

	private void OnDrawItem(object sender, DrawItemEventArgs e)
	{
		e.DrawBackground();
		ListBox listBox = (ListBox)sender;
		string text = (string)listBox.Items[e.Index];
		DrawRectangularPortion(e, text);
		Brush brush = SystemBrushes.ControlText;
		if ((e.State & DrawItemState.Selected) == DrawItemState.Selected)
		{
			brush = SystemBrushes.HighlightText;
		}
		e.Graphics.DrawString(text, listBox.Font, brush, e.Bounds.Left + 35, e.Bounds.Top + 5);
		if ((e.State & DrawItemState.Focus) == DrawItemState.Focus)
		{
			e.DrawFocusRectangle();
		}
	}

	private void DrawRectangularPortion(DrawItemEventArgs e, string fontName)
	{
		Rectangle bounds = e.Bounds;
		bounds.Inflate(-2, -2);
		bounds.Width = 30;
		e.Graphics.FillRectangle(SystemBrushes.Highlight, bounds);
		e.Graphics.DrawRectangle(SystemPens.WindowFrame, bounds);
		Font font = GetFont(fontName);
		bounds.Inflate(-1, -1);
		StringFormat stringFormat = new StringFormat(StringFormatFlags.FitBlackBox | StringFormatFlags.NoWrap);
		e.Graphics.DrawString("abc", font, SystemBrushes.ControlText, bounds, stringFormat);
		stringFormat.Dispose();
		font.Dispose();
	}

	private Font GetFont(string fontName)
	{
		FontFamily fontFamily = new FontFamily(fontName);
		Font result = ((!fontFamily.IsStyleAvailable(FontStyle.Regular)) ? new Font(fontName, 9.75f, FontStyle.Italic) : new Font(fontName, 9.75f, FontStyle.Regular));
		fontFamily.Dispose();
		return result;
	}

	protected override void FillInList(ITypeDescriptorContext pContext, IServiceProvider pProvider, ListBox pListBox)
	{
		FontFamily[] families = StyleInfo.installedFonts.Families;
		foreach (FontFamily fontFamily in families)
		{
			pListBox.Items.Add(fontFamily.Name);
		}
	}
}

public class RangeConverter : TypeConverter
{
	public override bool CanConvertFrom(ITypeDescriptorContext context, Type sourceType)
	{
		if ((object)sourceType == typeof(string))
		{
			return true;
		}
		return base.CanConvertFrom(context, sourceType);
	}

	public override object ConvertFrom(ITypeDescriptorContext context, CultureInfo culture, object value)
	{
		if (value is string)
		{
			string[] array = ((string)value).Split(':');
			if (array.Length > 2)
			{
				return Range.InvalidRange();
			}
			int num = array[0].IndexOf('R');
			int num2 = array[0].IndexOf('C');
			int num3 = Convert.ToInt32(array[0].Substring(num + 1, num2 - 1));
			int num4 = Convert.ToInt32(array[0].Substring(num2 + 1));
			int nBottom = num3;
			int nRight = num4;
			if (array.Length > 1)
			{
				num = array[1].IndexOf('R');
				num2 = array[1].IndexOf('C');
				nBottom = Convert.ToInt32(array[1].Substring(num + 1, num2 - 1));
				nRight = Convert.ToInt32(array[1].Substring(num2 + 1));
			}
			return Range.FromLTRB(num4, num3, nRight, nBottom);
		}
		return base.ConvertFrom(context, culture, value);
	}

	public override bool CanConvertTo(ITypeDescriptorContext context, Type destinationType)
	{
		if ((object)destinationType == typeof(string))
		{
			return true;
		}
		return base.CanConvertTo(context, destinationType);
	}

	public override object ConvertTo(ITypeDescriptorContext context, CultureInfo culture, object value, Type destinationType)
	{
		if ((object)destinationType == typeof(string))
		{
			return ((Range)value).ToString();
		}
		return base.ConvertTo(context, culture, value, destinationType);
	}

	public override bool GetPropertiesSupported(ITypeDescriptorContext context)
	{
		return true;
	}

	public override PropertyDescriptorCollection GetProperties(ITypeDescriptorContext context, object value, Attribute[] attributes)
	{
		return TypeDescriptor.GetProperties(value, attributes);
	}

	public override bool GetCreateInstanceSupported(ITypeDescriptorContext context)
	{
		return false;
	}

	public override bool GetStandardValuesSupported(ITypeDescriptorContext context)
	{
		return false;
	}

	public override bool GetStandardValuesExclusive(ITypeDescriptorContext context)
	{
		return false;
	}
}

public class OGUtilities
{
	public static void SetAt(ref string s, int iIndex, int sCount, string pat)
	{
		s = s.Remove(iIndex, sCount);
		s = s.Insert(iIndex, pat);
	}

	public static void SetAt(ref string s, int iIndex, char pat)
	{
		SetAt(ref s, iIndex, 1, pat.ToString());
	}

	public static string Remove(string s, string pattern)
	{
		while (true)
		{
			int num = s.IndexOf(pattern);
			if (num == -1)
			{
				break;
			}
			s = s.Remove(num, pattern.Length);
		}
		return s;
	}

	public static int Count(string s, char pat)
	{
		int num = 0;
		foreach (char c in s)
		{
			if (c == pat)
			{
				num++;
			}
		}
		return num;
	}

	public static int Count(string s, string pat)
	{
		int num = 0;
		int startIndex = 0;
		while (true)
		{
			startIndex = s.IndexOf(pat, startIndex);
			if (startIndex == -1)
			{
				break;
			}
			num++;
			startIndex += pat.Length;
		}
		return num;
	}

	public static int GetDigitAfterIndex(int iPos, string p)
	{
		if (iPos <= 0)
		{
			return 0;
		}
		return CountDigits(p.Substring(0, iPos));
	}

	public static int GetDigitBeforeIndex(int iPos, string p)
	{
		if (iPos <= 1)
		{
			return 0;
		}
		return CountDigits(p.Substring(0, iPos)) - 1;
	}

	public static int GetDigitIndex(int iDigitNumber, string p)
	{
		if (p != "")
		{
			int num = 0;
			for (int i = 0; i < p.Length; i++)
			{
				if (char.IsDigit(p[i]) && ++num > iDigitNumber)
				{
					return i;
				}
			}
		}
		return -1;
	}

	public static int CountDigits(string s)
	{
		int num = 0;
		foreach (char c in s)
		{
			if (char.IsDigit(c))
			{
				num++;
			}
		}
		return num;
	}

	public static bool IsAllDigits(string s)
	{
		bool result = true;
		foreach (char c in s)
		{
			if (!char.IsDigit(c))
			{
				result = false;
				break;
			}
		}
		return result;
	}
}
